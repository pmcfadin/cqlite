1//! Partition lookup and index-based access methods for SSTableReader

4//! Summary.db, and Statistics.db readers.

13    /// Enhanced partition lookup using Index.db reader with promoted index support.
14    ///
15    /// `partition_key` must be the raw partition-key bytes as produced by
16    /// [`PartitionKey::to_bytes`](crate::storage::write_engine::mutation::PartitionKey::to_bytes):
17    ///
18    /// - **Single-component keys** — raw value bytes (UUID = 16 bytes, int = 4 BE bytes, etc.).
19    /// - **Multi-component (composite) keys** — `[len: u16 BE][value bytes][0x00]` per component,
20    ///   including a trailing `0x00` after the final component.
21    ///
22    /// The Index.db key_lookup map is keyed on these exact raw bytes (set when the BIG-format
23    /// parser was fixed in Issue #552).  The old digest-based path (which caused every lookup
24    /// to miss) has been removed.  On a miss the function returns `Ok(None)` so callers can
25    /// fall through to their existing sequential-scan fallback.
26    pub async fn lookup_partition_with_index(
27        &self,
28        partition_key: &[u8],
29    ) -> Result<Option<(u64, u32)>> {
30        if let Some(index_reader) = &self.index_reader {
31            // Direct raw-key lookup — O(1) HashMap lookup.
32            // Index.db entries are keyed on the raw partition key bytes since #552;
33            // no Murmur3 digest computation is needed or correct here.
34            if let Some(entry) = index_reader.lookup_partition(partition_key) {
35                debug!(
36                    "Found partition via Index.db raw-key lookup: offset={}, size={}",
37                    entry.data_offset, entry.data_size
38                );
39                return Ok(Some((entry.data_offset, entry.data_size)));
40            } else {
41                debug!(
42                    "Partition not found in Index.db for raw key (len={})",
43                    partition_key.len()
44                );
45            }
46        } else {
47            debug!("No Index.db reader available for partition lookup");
48        }
49        Ok(None)
50    }

52    /// Enhanced partition lookup using schema-driven key digest computation
53    pub async fn lookup_partition_with_schema_context(
54        &self,
55        partition_key: &[u8],
56        parsing_context: &ParsingContext,
57    ) -> Result<Option<(u64, u32)>> {
58        if let Some(index_reader) = &self.index_reader {
59            // Compute the schema-driven key digest for Index.db lookup
60            let key_digest =
61                self.compute_partition_key_digest_with_schema(partition_key, parsing_context)?;
62
63            // Use spec-compliant Index.db reader for partition lookup
64            if let Some(entry) = index_reader.lookup_partition(&key_digest) {
65                debug!(
66                    "Found partition via schema-driven Index.db: offset={}, size={}",
67                    entry.data_offset, entry.data_size
68                );
69                return Ok(Some((entry.data_offset, entry.data_size)));
70            }
71        }
72        Ok(None)
73    }

75    /// Enhanced partition iteration using Summary.db reader
76    ///
77    /// Note: Token-based range queries are not directly supported because Summary.db
78    /// does not store token values (Issue #218). Instead, this iterates all summary
79    /// entries and returns all partition data.
80    ///
81    /// For token-based filtering, compute tokens from partition keys after retrieval.
82    ///
83    /// ## Issue #500: Sequential-scan fallback for writer-produced SSTables
84    ///
85    /// The Summary.db → Index.db → Data.db lookup path depends on Index.db format
86    /// compatibility between writer and reader (digest format vs. raw-key format).
87    /// Locally written SSTables emit raw-key Index.db entries that the reader's
88    /// digest-based parser cannot resolve, so the lookup loop returns 0 entries
89    /// even though Summary.db enumerates the partitions correctly.
90    ///
91    /// When that happens we fall back to `sequential_scan`, which walks Data.db
92    /// directly. For V5CompressedLegacy NB SSTables (the format the writer emits),
93    /// `sequential_scan` uses the chunk-stitching path and returns every partition.
94    pub async fn iterate_all_partitions(&self) -> Result<Vec<(RowKey, Value)>> {
95        if let Some(summary_reader) = &self.summary_reader {
96            let entries = summary_reader.get_entries();
97            let mut results = Vec::new();
98
99            for entry in entries {
100                // Use Summary.db entry to find the corresponding Index.db entry
101                if let Some(_index_reader) = &self.index_reader {
102                    // The summary entry provides a position in Index.db
103                    // We need to read the partition data from Data.db
104
105                    // For now, use the partition key from the summary entry
106                    let partition_key_bytes = &entry.partition_key;
107
108                    // Look up the partition in Index.db to get the actual data offset
109                    if let Some((data_offset, data_size)) = self
110                        .lookup_partition_with_index(partition_key_bytes)
111                        .await?
112                    {
113                        // Convert Index.db relative offset to absolute file offset
114                        // Index.db offsets are relative to data section start (after compression header)
115                        let absolute_offset = data_offset + self.actual_header_size as u64;
116
117                        // Read and parse the actual partition data from Data.db
118                        match self
119                            .parse_partition_at_offset(absolute_offset, data_size)
120                            .await?
121                        {
122                            Some(partition_entries) => {
123                                for (row_key, value) in partition_entries {
124                                    results.push((row_key, value));
125                                }
126                            }
127                            None => {
128                                debug!("Failed to parse partition at offset {}", absolute_offset);
129                            }
130                        }
131                    }
132                } else {
133                    log::error!("Index reader not available for partition iteration");
134                    return Err(Error::corruption(
135                        "Index reader required for partition iteration - synthetic data not allowed for Issue #35",
136                    ));
137                }
138            }
139
140            // Only trust the index-based path when EVERY summary entry was resolved.
141            // Partial resolution silently drops the unresolved entries; defaulting to
142            // `sequential_scan` in that case is strictly safer and still correct on
143            // real Cassandra SSTables (sequential_scan returns the same partitions
144            // when the index resolves them all).
145            if results.len() == entries.len() && !entries.is_empty() {
146                debug!("Partition iteration found {} entries", results.len());
147                return Ok(results);
148            }
149
150            debug!(
151                "Index.db lookup resolved {}/{} summary entries; \
152                 falling back to sequential_scan (Issue #500)",
153                results.len(),
154                entries.len()
155            );
156        }
157
158        // Fallback path: sequential walk of Data.db.
159        // Used when Summary.db is absent OR when the Index.db lookup loop returned
160        // no entries (Issue #500: writer-produced SSTables emit a raw-key Index.db
161        // format the reader's digest-based parser does not resolve).
162        let table_id = self.scan_table_id();
163        let schema = self.schema.as_deref();
164        self.sequential_scan(&table_id, None, None, None, schema)
165            .await
166    }

168    /// Build the TableId used for fallback `sequential_scan` from header metadata.
169    ///
170    /// The reader populates `header.keyspace` / `header.table_name` from either the
171    /// SSTable header or the parent directory path. When the V5CompressedLegacy
172    /// stitching path is used, table_id matching is skipped, so any non-empty value
173    /// is accepted; for other formats this returns the qualified `keyspace.table`
174    /// form so the scan filter matches.
175    fn scan_table_id(&self) -> TableId {
176        let keyspace = &self.header.keyspace;
177        let table_name = &self.header.table_name;
178        if !keyspace.is_empty() && !table_name.is_empty() {
179            TableId::from(format!("{}.{}", keyspace, table_name))
180        } else if !table_name.is_empty() {
181            TableId::from(table_name.as_str())
182        } else {
183            TableId::from("default")
184        }
185    }

187    /// Token range iteration (deprecated - tokens not stored in Summary.db)
188    ///
189    /// This method is kept for API compatibility but simply delegates to
190    /// `iterate_all_partitions()` since Summary.db does not store token values.
191    /// Token filtering should be done by the caller after retrieval.
192    #[deprecated(
193        since = "0.1.0",
194        note = "Summary.db does not store tokens. Use iterate_all_partitions() and filter by computed tokens."

196    pub async fn iterate_token_range(
197        &self,
198        _start_token: i64,
199        _end_token: i64,
200    ) -> Result<Vec<(RowKey, Value)>> {
201        // Token values are not stored in Summary.db (Issue #218)
202        // Delegate to all-partition iteration
203        self.iterate_all_partitions().await
204    }

206    /// Get min/max timestamps from Statistics.db reader
207    pub async fn get_timestamp_range(&self) -> Result<Option<(i64, i64)>> {
208        if let Some(statistics_reader) = &self.statistics_reader {
209            let (min_ts, max_ts) = statistics_reader.timestamp_range();
210            debug!(
211                "Retrieved timestamp range from Statistics.db: {} to {}",
212                min_ts, max_ts
213            );
214            return Ok(Some((min_ts, max_ts)));
215        }
216        Ok(None)
217    }

236    /// Enhanced get method using spec readers for efficient lookup
237    pub async fn get_with_spec_readers(
238        &self,
239        table_id: &TableId,
240        key: &RowKey,
241    ) -> Result<Option<Value>> {
242        // Step 1: Use bloom filter for existence check
243        if let Some(bloom_filter) = &self.bloom_filter {
244            if !bloom_filter.might_contain(key.as_bytes()) {
245                debug!("Bloom filter indicates key does not exist");
246                return Ok(None);
247            }
248        }
249
250        // Step 2: Use Index.db reader for precise partition lookup
251        if let Some((offset, size)) = self.lookup_partition_with_index(key.as_bytes()).await? {
252            debug!("Using Index.db lookup: offset={}, size={}", offset, size);
253            return self.read_value_at_offset(offset, size).await;
254        }
255
256        // Step 3: Fallback to existing methods
257        debug!("Falling back to legacy lookup methods");
258        self.get(table_id, key).await
259    }
