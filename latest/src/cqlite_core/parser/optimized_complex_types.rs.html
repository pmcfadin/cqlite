1//! High-performance optimized complex type parsing for M3
2//!
3//! This module provides SIMD-optimized, vectorized parsing for complex types
4//! with performance targets:
5//! - Complex type parsing: <2x slower than primitive types
6//! - Memory usage: <1.5x increase for complex type storage
7//! - Throughput: >100 MB/s for complex type SSTable parsing
8//! - Latency: <10ms additional latency for complex type queries
9

54    /// Detect SIMD support on current platform
55    fn detect_simd_support() -> bool {
56        #[cfg(target_arch = "x86_64")]
57        {
58            is_x86_feature_detected!("sse2") && is_x86_feature_detected!("avx2")
59        }
60        #[cfg(not(target_arch = "x86_64"))]
61        {
62            false

66    /// Parse a list with vectorized optimizations
67    pub fn parse_optimized_list<'a>(&'a self, input: &'a [u8]) -> IResult<&'a [u8], Value> {
68        let start_time = std::time::Instant::now();
69
70        let (input, count) = parse_vint_length(input)?;
71        let (input, element_type) = map_res(be_u8, CqlTypeId::try_from)(input)?;
72
73        if count == 0 {
74            return Ok((input, Value::List(Vec::new())));
75        }
76
77        let result = if self.enable_simd && count >= self.batch_size {
78            self.parse_list_simd(input, count, element_type)
79        } else {
80            self.parse_list_sequential(input, count, element_type)
81        };
82
83        // Update metrics
84        self.metrics
85            .list_parse_count
86            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
87        self.metrics.list_parse_time_ns.fetch_add(
88            start_time.elapsed().as_nanos() as u64,
89            std::sync::atomic::Ordering::Relaxed,
90        );
91
92        result
93    }

95    /// SIMD-optimized list parsing for supported types
96    #[cfg(target_arch = "x86_64")]
97    fn parse_list_simd<'a>(
98        &self,
99        input: &'a [u8],
100        count: usize,
101        element_type: CqlTypeId,
102    ) -> IResult<&'a [u8], Value> {
103        match element_type {
104            CqlTypeId::Int => self.parse_int_list_simd(input, count),
105            CqlTypeId::Float => self.parse_float_list_simd(input, count),
106            CqlTypeId::BigInt => self.parse_bigint_list_simd(input, count),
107            _ => self.parse_list_sequential(input, count, element_type),
108        }
109    }

121    /// SIMD-optimized integer list parsing
122    #[cfg(target_arch = "x86_64")]
123    fn parse_int_list_simd<'a>(
124        &self,
125        mut input: &'a [u8],
126        count: usize,
127    ) -> IResult<&'a [u8], Value> {
128        let mut elements = Vec::with_capacity(count);
129        let mut remaining = count;
130
131        unsafe {
132            // Process 8 integers at a time using AVX2
133            while remaining >= 8 && input.len() >= 32 {
134                if is_x86_feature_detected!("avx2") {
135                    // Load 8 32-bit big-endian integers
136                    let chunk = _mm256_loadu_si256(input.as_ptr() as *const __m256i);
137
138                    // Convert from big-endian to little-endian
139                    let swapped = self.simd_bswap_epi32(chunk);
140
141                    // Extract values
142                    let values: [i32; 8] = std::mem::transmute(swapped);
143                    for &val in &values {
144                        elements.push(Value::Integer(val));
145                    }
146
147                    input = &input[32..];
148                    remaining -= 8;
149
150                    self.metrics
151                        .simd_operations
152                        .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
153                }
154            }
155        }
156
157        // Handle remaining elements sequentially
158        for _ in 0..remaining {
159            let (new_input, value) = parse_cql_value(input, CqlTypeId::Int)?;
160            elements.push(value);
161            input = new_input;
162        }
163
164        Ok((input, Value::List(elements)))
165    }

167    /// SIMD byte swap for 32-bit integers
168    #[cfg(target_arch = "x86_64")]
169    unsafe fn simd_bswap_epi32(&self, chunk: __m256i) -> __m256i {
170        // AVX2 byte swap using shuffle
171        unsafe {
172            let shuffle_mask = _mm256_set_epi8(
173                12, 13, 14, 15, 8, 9, 10, 11, 4, 5, 6, 7, 0, 1, 2, 3, 12, 13, 14, 15, 8, 9, 10, 11,
174                4, 5, 6, 7, 0, 1, 2, 3,
175            );
176            _mm256_shuffle_epi8(chunk, shuffle_mask)
177        }
178    }

180    /// SIMD-optimized float list parsing
181    #[cfg(target_arch = "x86_64")]
182    fn parse_float_list_simd<'a>(
183        &self,
184        mut input: &'a [u8],
185        count: usize,
186    ) -> IResult<&'a [u8], Value> {
187        let mut elements = Vec::with_capacity(count);
188        let mut remaining = count;
189
190        unsafe {
191            // Process 8 floats at a time using AVX2
192            while remaining >= 8 && input.len() >= 32 {
193                if is_x86_feature_detected!("avx2") {
194                    // Load 8 32-bit big-endian floats
195                    let chunk = _mm256_loadu_si256(input.as_ptr() as *const __m256i);
196
197                    // Convert from big-endian to little-endian
198                    let swapped = self.simd_bswap_epi32(chunk);
199                    let floats = _mm256_castsi256_ps(swapped);
200
201                    // Extract values
202                    let values: [f32; 8] = std::mem::transmute(floats);
203                    for &val in &values {
204                        elements.push(Value::Float(val as f64));
205                    }
206
207                    input = &input[32..];
208                    remaining -= 8;
209
210                    self.metrics
211                        .simd_operations
212                        .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
213                }
214            }
215        }
216
217        // Handle remaining elements sequentially
218        for _ in 0..remaining {
219            let (new_input, value) = parse_cql_value(input, CqlTypeId::Float)?;
220            elements.push(value);
221            input = new_input;
222        }
223
224        Ok((input, Value::List(elements)))
225    }

245                    let swapped = self.simd_bswap_epi64(chunk);

254                    remaining -= 4;
255
256                    self.metrics
257                        .simd_operations
258                        .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
259                }
260            }
261        }
262
263        // Handle remaining elements sequentially
264        for _ in 0..remaining {
265            let (new_input, value) = parse_cql_value(input, CqlTypeId::BigInt)?;
266            elements.push(value);
267            input = new_input;
268        }
269
270        Ok((input, Value::List(elements)))

304    /// Parse a map with optimized memory layout
305    pub fn parse_optimized_map<'a>(&self, input: &'a [u8]) -> IResult<&'a [u8], Value> {
306        let start_time = std::time::Instant::now();
307
308        let (input, count) = parse_vint_length(input)?;
309        let (input, key_type) = map_res(be_u8, CqlTypeId::try_from)(input)?;
310        let (input, value_type) = map_res(be_u8, CqlTypeId::try_from)(input)?;
311
312        if count == 0 {
313            return Ok((input, Value::Map(Vec::new())));
314        }
315
316        // Use pre-allocated buffer for efficiency
317        let mut map = Vec::with_capacity(count);
318        let mut remaining_input = input;
319
320        // Batch processing for better cache locality
321        let batch_size = self.batch_size.min(count);
322        for chunk_start in (0..count).step_by(batch_size) {
323            let chunk_end = (chunk_start + batch_size).min(count);
324
325            for _ in chunk_start..chunk_end {
326                // Parse key with length prefix (same as regular map parser)
327                let (new_input, key_length) = parse_vint(remaining_input)?;
328                if key_length < 0 {
329                    return Err(nom::Err::Error(nom::error::Error::new(
330                        remaining_input,
331                        nom::error::ErrorKind::Verify,
332                    )));
333                }
334                let (new_input, key_data) = take(key_length as usize)(new_input)?;
335                let key = super::types::parse_cql_value_raw(key_data, key_type)?.1;
336
337                // Parse value with length prefix (same as regular map parser)
338                let (new_input, value_length) = parse_vint(new_input)?;
339                if value_length < 0 {
340                    return Err(nom::Err::Error(nom::error::Error::new(
341                        new_input,
342                        nom::error::ErrorKind::Verify,
343                    )));
344                }
345                let (new_input, value_data) = take(value_length as usize)(new_input)?;
346                let value = super::types::parse_cql_value_raw(value_data, value_type)?.1;
347
348                map.push((key, value));
349                remaining_input = new_input;
350            }
351        }
352
353        // Update metrics
354        self.metrics
355            .map_parse_count
356            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
357        self.metrics.map_parse_time_ns.fetch_add(
358            start_time.elapsed().as_nanos() as u64,
359            std::sync::atomic::Ordering::Relaxed,
360        );
361
362        Ok((remaining_input, Value::Map(map)))
363    }

365    /// Parse a UDT with field caching optimization
366    pub fn parse_optimized_udt<'a>(&self, input: &'a [u8]) -> IResult<&'a [u8], Value> {
367        let start_time = std::time::Instant::now();
368
369        let (input, type_name_len) = parse_vint_length(input)?;
370        let (input, type_name_bytes) = take(type_name_len)(input)?;
371        let type_name = String::from_utf8(type_name_bytes.to_vec()).map_err(|_| {
372            nom::Err::Error(nom::error::Error::new(input, nom::error::ErrorKind::Verify))
373        })?;
374
375        let (input, field_count) = parse_vint_length(input)?;
376
377        // Use HashMap with pre-allocated capacity for better performance
378        let mut fields = HashMap::with_capacity(field_count);
379        let mut remaining_input = input;
380
381        for _ in 0..field_count {
382            let (new_input, field_name_len) = parse_vint_length(remaining_input)?;
383            let (new_input, field_name_bytes) = take(field_name_len)(new_input)?;
384            let field_name = String::from_utf8(field_name_bytes.to_vec()).map_err(|_| {
385                nom::Err::Error(nom::error::Error::new(
386                    new_input,
387                    nom::error::ErrorKind::Verify,
388                ))
389            })?;
390
391            let (new_input, field_type) = map_res(be_u8, CqlTypeId::try_from)(new_input)?;
392            let (new_input, field_value) = parse_cql_value(new_input, field_type)?;
393
394            fields.insert(field_name, field_value);
395            remaining_input = new_input;
396        }
397
398        // Update metrics
399        self.metrics
400            .udt_parse_count
401            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
402        self.metrics.udt_parse_time_ns.fetch_add(
403            start_time.elapsed().as_nanos() as u64,
404            std::sync::atomic::Ordering::Relaxed,
405        );
406
407        let udt_value = crate::types::UdtValue {
408            type_name,
409            keyspace: "default".to_string(), // TODO: Get from context
410            fields: fields
411                .into_iter()
412                .map(|(name, value)| crate::types::UdtField {
413                    name,
414                    value: Some(value),
415                })
416                .collect(),
417        };
418        Ok((remaining_input, Value::Udt(udt_value)))
419    }

421    /// Parse a set (optimized similar to list)
422    pub fn parse_optimized_set<'a>(&'a self, input: &'a [u8]) -> IResult<&'a [u8], Value> {
423        // Sets are stored similarly to lists, so we can reuse list optimization
424        let (input, list_value) = self.parse_optimized_list(input)?;
425
426        if let Value::List(elements) = list_value {
427            Ok((input, Value::Set(elements)))
428        } else {
429            Err(nom::Err::Error(nom::error::Error::new(
430                input,
431                nom::error::ErrorKind::Verify,
432            )))
433        }
434    }

436    /// Parse a tuple with memory-efficient layout
437    pub fn parse_optimized_tuple<'a>(&self, input: &'a [u8]) -> IResult<&'a [u8], Value> {
438        let (input, element_count) = parse_vint_length(input)?;
439
440        let mut elements = Vec::with_capacity(element_count);
441        let mut remaining_input = input;
442
443        for _ in 0..element_count {
444            let (new_input, element_type) = map_res(be_u8, CqlTypeId::try_from)(remaining_input)?;
445            let (new_input, element) = parse_cql_value(new_input, element_type)?;
446
447            elements.push(element);
448            remaining_input = new_input;
449        }
450
451        Ok((remaining_input, Value::Tuple(elements)))
452    }

459    /// Generate performance report
460    pub fn generate_performance_report(&self) -> String {
461        use std::sync::atomic::Ordering;
462
463        let list_count = self.metrics.list_parse_count.load(Ordering::Relaxed);
464        let list_time = self.metrics.list_parse_time_ns.load(Ordering::Relaxed);
465        let map_count = self.metrics.map_parse_count.load(Ordering::Relaxed);
466        let map_time = self.metrics.map_parse_time_ns.load(Ordering::Relaxed);
467        let udt_count = self.metrics.udt_parse_count.load(Ordering::Relaxed);
468        let udt_time = self.metrics.udt_parse_time_ns.load(Ordering::Relaxed);
469        let simd_ops = self.metrics.simd_operations.load(Ordering::Relaxed);
470
471        format!(
472            "Complex Type Parser Performance Report\n\
473            =======================================\n\
474            SIMD Support: {}\n\
475            Batch Size: {}\n\n\
476            Lists Parsed: {} (avg: {:.2} μs each)\n\
477            Maps Parsed: {} (avg: {:.2} μs each)\n\
478            UDTs Parsed: {} (avg: {:.2} μs each)\n\
479            SIMD Operations: {}\n\n\
480            Total Parse Time: {:.2} ms\n\
481            Average Throughput: {:.2} MB/s",
482            self.enable_simd,
483            self.batch_size,
484            list_count,
485            if list_count > 0 {
486                list_time as f64 / list_count as f64 / 1000.0
487            } else {
488                0.0
489            },
490            map_count,
491            if map_count > 0 {
492                map_time as f64 / map_count as f64 / 1000.0
493            } else {
494                0.0
495            },
496            udt_count,
497            if udt_count > 0 {
498                udt_time as f64 / udt_count as f64 / 1000.0
499            } else {
500                0.0
501            },
502            simd_ops,
503            (list_time + map_time + udt_time) as f64 / 1_000_000.0,
504            self.calculate_throughput()
505        )
506    }
507
508    /// Calculate average throughput in MB/s
509    fn calculate_throughput(&self) -> f64 {
510        use std::sync::atomic::Ordering;
511
512        let total_time_ns = self.metrics.list_parse_time_ns.load(Ordering::Relaxed)
513            + self.metrics.map_parse_time_ns.load(Ordering::Relaxed)
514            + self.metrics.udt_parse_time_ns.load(Ordering::Relaxed);
515
516        if total_time_ns > 0 {
517            // Estimate data processed (rough approximation)
518            let estimated_bytes = (self.metrics.list_parse_count.load(Ordering::Relaxed) * 100)
519                + (self.metrics.map_parse_count.load(Ordering::Relaxed) * 200)
520                + (self.metrics.udt_parse_count.load(Ordering::Relaxed) * 150);
521
522            (estimated_bytes as f64 * 1_000_000_000.0) / (total_time_ns as f64 * 1_000_000.0)
523        } else {
524            0.0
525        }
526    }
