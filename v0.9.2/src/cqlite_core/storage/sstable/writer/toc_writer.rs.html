1//! TOC.txt writer - writes table of contents
2//!
3//! Generates the TOC.txt component listing all SSTable files.
4//! Acts as a publication barrier - SSTable is not visible until TOC.txt exists.
5//!
6//! Critical requirements:
7//! - MUST be written LAST (after all other components)
8//! - Lists all component files
9//! - Plain text format, one component per line
10//! - Self-referential: TOC.txt must list itself
11
12use std::fs::File;
13use std::io::{BufWriter, Write};
14use std::path::PathBuf;
15
16use crate::error::{Error, Result};
17use crate::storage::sstable::directory::types::SSTableComponent;

19/// Entry representing a component to be written to TOC.txt
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub struct ComponentEntry {
22    /// Component type (e.g., Data, Index, etc.)
23    pub component: SSTableComponent,
24}
25
26impl ComponentEntry {
27    /// Create a new component entry
28    pub fn new(component: SSTableComponent) -> Self {
29        Self { component }
30    }
31
32    /// Get the filename for this component
33    pub fn filename(&self) -> &'static str {
34        self.component.file_extension()
35    }
36}

38/// TOC.txt component writer
39///
40/// Writes the table of contents listing all SSTable component files.
41/// This file acts as the publication barrier - an SSTable is not considered
42/// complete and readable until TOC.txt exists.
43///
44/// # Format
45///
46/// Plain text file with one component filename per line:
47/// ```text
48/// Data.db
49/// Index.db
50/// Filter.db
51/// Statistics.db
52/// Summary.db
53/// CompressionInfo.db
54/// Digest.crc32
55/// TOC.txt
56/// ```
57///
58/// # Critical Requirements
59///
60/// 1. TOC.txt MUST be written LAST after all other components
61/// 2. TOC.txt MUST include itself in the listing
62/// 3. Component order in the file doesn't matter (implementation orders for consistency)
63///
64/// # Example
65///
66/// ```rust,ignore
67/// let mut components = vec![
68///     ComponentEntry::new(SSTableComponent::Data),
69///     ComponentEntry::new(SSTableComponent::Index),
70///     ComponentEntry::new(SSTableComponent::Statistics),
71///     ComponentEntry::new(SSTableComponent::Filter),
72///     ComponentEntry::new(SSTableComponent::Summary),
73///     ComponentEntry::new(SSTableComponent::CompressionInfo),
74///     ComponentEntry::new(SSTableComponent::Digest),
75/// ];
76///
77/// let writer = TocWriter::new(toc_path);
78/// writer.write(&components)?;
79/// ```
80#[derive(Debug)]
81pub struct TocWriter {
82    /// Path to the TOC.txt file to write
83    path: PathBuf,
84}
85
86impl TocWriter {
87    /// Create a new TOC.txt writer
88    ///
89    /// # Arguments
90    ///
91    /// * `path` - Full path to the TOC.txt file (e.g., "nb-1-big-TOC.txt")
92    pub fn new(path: PathBuf) -> Self {
93        Self { path }
94    }
95
96    /// Write TOC.txt file with the given component list
97    ///
98    /// Automatically adds TOC.txt itself to the listing if not already present.
99    /// Components are written in a consistent order for deterministic output.
100    ///
101    /// # Arguments
102    ///
103    /// * `components` - List of components to include in TOC.txt
104    ///
105    /// # Returns
106    ///
107    /// Ok(()) on success, Error on I/O failure
108    ///
109    /// # Errors
110    ///
111    /// Returns Error::Storage if:
112    /// - Unable to create the TOC.txt file
113    /// - Unable to write component names
114    /// - Unable to flush/sync the file
115    pub fn write(&self, components: &[ComponentEntry]) -> Result<()> {
116        // Ensure TOC itself is in the list (add if missing)
117        let mut final_components: Vec<ComponentEntry> = components.to_vec();
118
119        if !final_components
120            .iter()
121            .any(|e| e.component == SSTableComponent::TOC)
122        {
123            final_components.push(ComponentEntry::new(SSTableComponent::TOC));
124        }
125
126        // Sort components for deterministic output (matches Cassandra behavior)
127        // Order: Data, Statistics, Digest, TOC, CompressionInfo, Filter, Index, Summary
128        final_components.sort_by_key(|entry| match entry.component {
129            SSTableComponent::Data => 0,
130            SSTableComponent::Statistics => 1,
131            SSTableComponent::Digest => 2,
132            SSTableComponent::TOC => 3,
133            SSTableComponent::CompressionInfo => 4,
134            SSTableComponent::Filter => 5,
135            SSTableComponent::Index => 6,
136            SSTableComponent::Summary => 7,
137            SSTableComponent::Partitions => 8,
138            SSTableComponent::Rows => 9,
139        });
140
141        // Create file with buffered writing
142        let file = File::create(&self.path).map_err(|e| {
143            Error::storage(format!(
144                "Failed to create TOC.txt file {:?}: {}",
145                self.path, e
146            ))
147        })?;
148
149        let mut writer = BufWriter::new(file);
150
151        // Write each component filename on its own line
152        for entry in &final_components {
153            writeln!(writer, "{}", entry.filename()).map_err(|e| {
154                Error::storage(format!(
155                    "Failed to write component {} to TOC.txt: {}",
156                    entry.filename(),
157                    e
158                ))
159            })?;
160        }
161
162        // Flush buffer and sync to disk (durability requirement)
163        writer
164            .flush()
165            .map_err(|e| Error::storage(format!("Failed to flush TOC.txt: {}", e)))?;
166
167        let file = writer
168            .into_inner()
169            .map_err(|e| Error::storage(format!("Failed to extract file from buffer: {}", e)))?;
170
171        file.sync_all()
172            .map_err(|e| Error::storage(format!("Failed to sync TOC.txt to disk: {}", e)))?;
173
174        Ok(())
175    }
176}

184    #[test]
185    fn test_toc_writer_basic() {
186        let temp_dir = TempDir::new().unwrap();
187        let toc_path = temp_dir.path().join("nb-1-big-TOC.txt");
188
189        let components = vec![
190            ComponentEntry::new(SSTableComponent::Data),
191            ComponentEntry::new(SSTableComponent::Index),
192            ComponentEntry::new(SSTableComponent::Statistics),
193        ];
194
195        let writer = TocWriter::new(toc_path.clone());
196        writer.write(&components).unwrap();
197
198        // Verify file exists
199        assert!(toc_path.exists());
200
201        // Read and verify contents
202        let contents = fs::read_to_string(&toc_path).unwrap();
203        let lines: Vec<&str> = contents.lines().collect();
204
205        // Should include the 3 components plus TOC.txt itself
206        assert_eq!(lines.len(), 4);
207
208        // Verify TOC.txt is included
209        assert!(lines.contains(&"TOC.txt"));
210        assert!(lines.contains(&"Data.db"));
211        assert!(lines.contains(&"Index.db"));
212        assert!(lines.contains(&"Statistics.db"));
213    }
