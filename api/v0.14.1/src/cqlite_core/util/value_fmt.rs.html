1//! Value formatting shared by output writers
2//! Implements the Value → String mapping per QUERY_RESULT_CONTRACT.md
3//!
4//! This module provides stable, cqlsh-compatible formatting for all CQL value
5//! types.  It originally lived in `cqlite-cli/src/output/value_fmt.rs` and was
6//! moved into core (Issue #683) so that the Parquet export writer — which uses
7//! it for textual fallbacks such as inet and duration — can live in
8//! `cqlite-core` together with its formatting helpers.  The CLI re-exports it
9//! unchanged from `cqlite_cli::output::value_fmt`.
10

15/// ValueFormatter provides cqlsh-compatible string formatting for CQL values
16pub struct ValueFormatter;

18/// Lowercase hex digits for the lookup-table UUID/hex encoders.
19const HEX_LOWER: &[u8; 16] = b"0123456789abcdef";
20
21impl ValueFormatter {
22    /// Returns `true` when the value represents a genuine CQL `NULL`.
23    ///
24    /// This is the authoritative null predicate used by output writers to decide
25    /// whether to emit an empty CSV field. It replaces the fragile
26    /// `format_value(v) == "null"` sentinel, which wrongly collapsed a literal
27    /// text value `"null"` to an empty field (issue #1499).
28    ///
29    /// A genuine null in a frozen column is represented as
30    /// `Value::Frozen(Box::new(Value::Null))` (which `format_value` renders as the
31    /// bare string `"null"`), so `is_null` unwraps `Value::Frozen` recursively
32    /// before checking for `Value::Null`. This restores the pre-#1499 CSV
33    /// null-output contract for frozen columns without re-introducing the original
34    /// bug: a literal text value `Value::Text("null")` is NOT null and still emits
35    /// `"null"`. `Value::Null` and `Value::Frozen(..)` are the only genuine-null
36    /// representations that format to the bare string `"null"` (a UDT/collection
37    /// with null members formats as `{field: null}`/`[null]`, not `"null"`).
38    #[inline]
39    pub fn is_null(value: &Value) -> bool {
40        match value {
41            Value::Null => true,
42            Value::Frozen(inner) => Self::is_null(inner),
43            _ => false,
44        }
45    }

47    /// Append the formatted string representation of `value` to `out`.
48    ///
49    /// Byte-for-byte identical to `format_value`, but writes into a caller-owned
50    /// scratch buffer so hot loops (CSV rows) can reuse one allocation across all
51    /// cells instead of allocating a fresh `String` per cell (issue #1499). Scalar
52    /// hot paths are written directly; rarer complex types delegate to
53    /// `format_value` for a single append.
54    pub fn format_into(value: &Value, out: &mut String) {
55        use std::fmt::Write as _;
56        match value {
57            Value::Null => out.push_str("null"),
58            Value::Boolean(b) => out.push_str(if *b { "true" } else { "false" }),
59            Value::TinyInt(i) => {
60                let _ = write!(out, "{}", i);
61            }
62            Value::SmallInt(i) => {
63                let _ = write!(out, "{}", i);
64            }
65            Value::Integer(i) => {
66                let _ = write!(out, "{}", i);
67            }
68            Value::BigInt(i) => {
69                let _ = write!(out, "{}", i);
70            }
71            Value::Counter(i) => {
72                let _ = write!(out, "{}", i);
73            }
74            Value::Text(s) => out.push_str(s),
75            Value::Uuid(bytes) => Self::format_uuid_into(bytes, out),
76            // Complex / rarer types: single append via the owned formatter. This
77            // keeps output byte-identical without duplicating their logic.
78            other => out.push_str(&Self::format_value(other)),
79        }
80    }

82    /// Encode a 16-byte UUID as lowercase hyphenated text into `out` using a hex
83    /// lookup table (no per-cell `format!` machinery). Shared by `format_uuid`
84    /// and the JSON writer (issue #1499).
85    pub fn format_uuid_into(bytes: &[u8; 16], out: &mut String) {
86        out.reserve(36);
87        for (i, b) in bytes.iter().enumerate() {
88            // Hyphens after bytes 4, 6, 8, and 10 (1-based).
89            if matches!(i, 4 | 6 | 8 | 10) {
90                out.push('-');
91            }
92            out.push(HEX_LOWER[(b >> 4) as usize] as char);
93            out.push(HEX_LOWER[(b & 0x0f) as usize] as char);
94        }
95    }

97    /// Format a Value to its string representation according to the contract specification
98    ///
99    /// # Contract Guarantees
100    /// - UUID/TimeUUID: lowercase hyphenated (e.g., "a8f167f0-ebe7-4f20-a386-31ff138bec3b")
101    /// - Timestamps: `YYYY-MM-DD HH:MM:SS[.fff][+0000]`, default UTC
102    /// - Collections: list `[a, b]`, set `{a, b}`, map `{k: v}`
103    /// - Blob: `0x`-prefixed lowercase hex
104    /// - Boolean: `true`/`false`
105    /// - Numbers: standard Rust formatting, avoid scientific notation unless necessary
106    pub fn format_value(value: &Value) -> String {
107        match value {
108            Value::Null => "null".to_string(),
109
110            // Boolean: lowercase true/false
111            Value::Boolean(b) => b.to_string(),
112
113            // Integer types: standard decimal formatting
114            Value::TinyInt(i) => i.to_string(),
115            Value::SmallInt(i) => i.to_string(),
116            Value::Integer(i) => i.to_string(),
117            Value::BigInt(i) => i.to_string(),
118            Value::Counter(i) => i.to_string(),
119
120            // Floating point: avoid scientific notation for reasonable ranges
121            Value::Float32(f) => Self::format_float32(*f),
122            Value::Float(f) => Self::format_float64(*f),
123
124            // Text: output as-is (no quotes for CLI display)
125            Value::Text(s) => s.clone(),
126
127            // Blob: 0x-prefixed lowercase hex
128            Value::Blob(bytes) => format!("0x{}", hex::encode(bytes)),
129
130            // Timestamp: milliseconds since epoch → YYYY-MM-DD HH:MM:SS.fff+0000
131            Value::Timestamp(millis) => Self::format_timestamp(*millis),
132
133            // Date: days since epoch → YYYY-MM-DD
134            Value::Date(days) => Self::format_date(*days),
135
136            // Time: nanoseconds since midnight → HH:MM:SS.nnnnnnnnn
137            Value::Time(nanos) => Self::format_time(*nanos),
138
139            // UUID: lowercase hyphenated format
140            Value::Uuid(bytes) => Self::format_uuid(bytes),
141
142            // Varint: arbitrary precision integer as decimal string
143            Value::Varint(bytes) => Self::format_varint(bytes),
144
145            // Decimal: scale + unscaled value → decimal string
146            Value::Decimal { scale, unscaled } => Self::format_decimal(*scale, unscaled),
147
148            // Duration: months, days, nanoseconds → "XmoYdZns" format
149            Value::Duration {
150                months,
151                days,
152                nanos,
153            } => Self::format_duration(*months, *days, *nanos),
154
155            // JSON: serialize to JSON string
156            Value::Json(json_value) => json_value.to_string(),
157
158            // Collections
159            Value::List(elements) => Self::format_list(elements),
160            Value::Set(elements) => Self::format_set(elements),
161            Value::Map(pairs) => Self::format_map(pairs),
162            Value::Tuple(fields) => Self::format_tuple(fields),
163
164            // User Defined Type
165            Value::Udt(udt) => Self::format_udt(udt),
166
167            // Frozen: unwrap and format inner value
168            Value::Frozen(inner) => Self::format_value(inner),
169
170            // Tombstone: special marker (should rarely appear in query results)
171            Value::Tombstone(info) => format!("<deleted@{}>", info.deletion_time),
172
173            // Inet: IPv4 or IPv6 address
174            Value::Inet(bytes) => Self::format_inet(bytes),
175        }
176    }

180    /// Format float32 avoiding scientific notation for reasonable ranges
181    fn format_float32(f: f32) -> String {
182        if f.is_nan() {
183            "NaN".to_string()
184        } else if f.is_infinite() {
185            if f.is_sign_positive() {
186                "Infinity".to_string()
187            } else {
188                "-Infinity".to_string()
189            }
190        } else if f.abs() < 1e-6 || f.abs() > 1e10 {
191            format!("{:e}", f)
192        } else {
193            format!("{}", f)
194        }
195    }

214    /// Format timestamp (milliseconds since epoch) as YYYY-MM-DD HH:MM:SS.fff+0000
215    fn format_timestamp(millis: i64) -> String {
216        // Use from_timestamp_millis to correctly handle pre-epoch timestamps
217        // (truncating division was incorrect for negative values)
218        if let Some(datetime) = DateTime::from_timestamp_millis(millis) {
219            // Format with milliseconds: YYYY-MM-DD HH:MM:SS.fff+0000
220            datetime.format("%Y-%m-%d %H:%M:%S%.3f+0000").to_string()
221        } else {
222            format!("<invalid-timestamp:{}>", millis)
223        }
224    }
225
226    /// Format date (days since Unix epoch) as YYYY-MM-DD
227    fn format_date(days: i32) -> String {
228        // Unix epoch: 1970-01-01
229        let epoch = DateTime::from_timestamp(0, 0)
230            .map(|dt| dt.date_naive())
231            .unwrap_or_else(|| {
232                // Fallback: construct epoch date directly if timestamp fails
233                // Ultimate fallback for Date value formatting
234                chrono::NaiveDate::from_ymd_opt(1970, 1, 1).unwrap_or(chrono::NaiveDate::MIN)
235            });
236
237        if let Some(date) = epoch.checked_add_signed(chrono::Duration::days(days as i64)) {
238            date.format("%Y-%m-%d").to_string()
239        } else {
240            format!("<invalid-date:{}>", days)
241        }
242    }

244    /// Format time (nanoseconds since midnight) as HH:MM:SS.nnnnnnnnn
245    fn format_time(nanos: i64) -> String {
246        if nanos < 0 {
247            return format!("<invalid-time:{}>", nanos);
248        }
249
250        let total_secs = nanos / 1_000_000_000;
251        let hours = total_secs / 3600;
252        let minutes = (total_secs % 3600) / 60;
253        let seconds = total_secs % 60;
254        let remaining_nanos = nanos % 1_000_000_000;
255
256        if hours >= 24 {
257            return format!("<invalid-time:{}>", nanos);
258        }
259
260        format!(
261            "{:02}:{:02}:{:02}.{:09}",
262            hours, minutes, seconds, remaining_nanos
263        )
264    }

273    /// Format varint as decimal string
274    fn format_varint(bytes: &[u8]) -> String {
275        if bytes.is_empty() {
276            return "0".to_string();
277        }
278
279        // Use from_signed_bytes_be to handle both positive and negative values correctly
280        let result = num_bigint::BigInt::from_signed_bytes_be(bytes);
281        result.to_string()
282    }

284    /// Format decimal (scale + unscaled value) as decimal string
285    fn format_decimal(scale: i32, unscaled: &[u8]) -> String {
286        if unscaled.is_empty() {
287            return "0".to_string();
288        }
289
290        // Sanity ceiling (issue #1754), kept consistent with the Node binding's
291        // `decimal_to_string`. A Cassandra `decimal` unscaled value is a Java
292        // BigInteger, so it is legitimately arbitrary-precision; we must NOT call a
293        // merely-large-but-well-formed value "corrupt". The only hard cost is the
294        // single `BigInt` → decimal-string base conversion, which is superlinear in
295        // the digit count. A 32 KB magnitude (~79k decimal digits) still converts
296        // in tens of milliseconds; only a genuinely pathological magnitude beyond
297        // that could stall a render, so fail closed ONLY above the ceiling.
298        // Infallible signature: this stays total (never panics).
299        const DECIMAL_MAX_UNSCALED_BYTES: usize = 32 * 1024;
300        if unscaled.len() > DECIMAL_MAX_UNSCALED_BYTES {
301            return format!(
302                "<corrupt-decimal:scale={scale},unscaled_len={}bytes>",
303                unscaled.len()
304            );
305        }
306
307        // Convert the unscaled bytes to a BigInt. Cassandra encodes the unscaled
308        // value as a two's-complement big-endian BigInteger, which is exactly what
309        // `from_signed_bytes_be` decodes (positive when the high bit is clear).
310        let bigint = num_bigint::BigInt::from_signed_bytes_be(unscaled);
311        // ONE base-10 conversion — the sole superlinear step; every branch below is
312        // a single O(digits) pass over the resulting string (no repeated division,
313        // no scale-width padding blowup).
314        let mut decimal_str = bigint.to_string();
315        let is_neg = decimal_str.starts_with('-');
316        if is_neg {
317            decimal_str = decimal_str[1..].to_string();
318        }
319
320        // Faithful, bounded exponent form for over-bound cases (issue #1754). Two
321        // triggers, both of which would otherwise require an O(digits)-wide
322        // positional expansion:
323        //   - a large-but-valid unscaled magnitude (thousands+ of digits), and
324        //   - a pathological `scale` used as a `repeat`/padding width (which at
325        //     `scale == i32::MIN` would also overflow `(-scale)`).
326        // Rendering `<digits>e<-scale>` (value = unscaled × 10^(−scale)) preserves
327        // EVERY digit exactly with no unbounded padding. Legitimate, normal-size
328        // decimals fall through to the byte-identical positional form below.
329        const DECIMAL_POSITIONAL_MAX_BYTES: usize = 1024;
330        const SCALE_RENDER_CAP: usize = 1_000_000;
331        if unscaled.len() > DECIMAL_POSITIONAL_MAX_BYTES
332            || scale.unsigned_abs() as usize > SCALE_RENDER_CAP
333        {
334            let body = if is_neg {
335                format!("-{decimal_str}")
336            } else {
337                decimal_str
338            };
339            // `unsigned_abs()`/`i64` avoid the `(-scale)` overflow at `i32::MIN`.
340            return if scale == 0 {
341                body
342            } else {
343                format!("{body}e{}", -(scale as i64))
344            };
345        }
346
347        // Insert decimal point based on scale
348        if scale <= 0 {
349            // Scale <= 0: multiply by 10^(-scale). `unsigned_abs()` avoids the
350            // `(-scale)` overflow panic at `scale == i32::MIN`.
351            decimal_str.push_str(&"0".repeat(scale.unsigned_abs() as usize));
352        } else if scale as usize >= decimal_str.len() {
353            // Need leading zeros
354            let leading_zeros = scale as usize - decimal_str.len() + 1;
355            decimal_str = format!("0.{}{}", "0".repeat(leading_zeros - 1), decimal_str);
356        } else {
357            // Insert decimal point
358            let pos = decimal_str.len() - scale as usize;
359            decimal_str.insert(pos, '.');
360        }
361
362        if is_neg {
363            format!("-{}", decimal_str)
364        } else {
365            decimal_str
366        }
367    }

369    /// Format duration as "XmoYdZns" (cqlsh format)
370    fn format_duration(months: i32, days: i32, nanos: i64) -> String {
371        let mut parts = Vec::new();
372
373        if months != 0 {
374            parts.push(format!("{}mo", months));
375        }
376        if days != 0 {
377            parts.push(format!("{}d", days));
378        }
379        if nanos != 0 {
380            parts.push(format!("{}ns", nanos));
381        }
382
383        if parts.is_empty() {
384            "0ns".to_string()
385        } else {
386            parts.join("")
387        }
388    }

392        let formatted_elements: Vec<String> = elements.iter().map(Self::format_value).collect();

402    /// Format map as {k1: v1, k2: v2}
403    fn format_map(pairs: &[(Value, Value)]) -> String {
404        let formatted_pairs: Vec<String> = pairs
405            .iter()
406            .map(|(k, v)| format!("{}: {}", Self::format_value(k), Self::format_value(v)))
407            .collect();
408        format!("{{{}}}", formatted_pairs.join(", "))

417    /// Format UDT as {field1: value1, field2: value2}
418    fn format_udt(udt: &crate::types::UdtValue) -> String {
419        let formatted_fields: Vec<String> = udt
420            .fields
421            .iter()
422            .map(|field| {
423                let value_str = field
424                    .value
425                    .as_ref()
426                    .map(Self::format_value)
427                    .unwrap_or_else(|| "null".to_string());
428                format!("{}: {}", field.name, value_str)
429            })
430            .collect();
431        format!("{{{}}}", formatted_fields.join(", "))
432    }

434    /// Format inet address (IPv4 or IPv6)
435    fn format_inet(bytes: &[u8]) -> String {
436        if bytes.len() == 4 {
437            // IPv4
438            let addr = Ipv4Addr::new(bytes[0], bytes[1], bytes[2], bytes[3]);
439            addr.to_string()
440        } else if bytes.len() == 16 {
441            // IPv6
442            let mut octets = [0u8; 16];
443            octets.copy_from_slice(bytes);
444            let addr = Ipv6Addr::from(octets);
445            addr.to_string()
446        } else {
447            format!("<invalid-inet:{}-bytes>", bytes.len())
448        }
449    }

762    /// Issue #1754: a corrupt SSTable can carry a pathological DECIMAL `scale`.
763    /// `scale == i32::MIN` used to overflow-panic at `(-scale) as usize` (and
764    /// under `overflow-checks` in debug builds), and a huge positive/negative
765    /// scale would allocate an unbounded string. Formatting must stay total —
766    /// never panic — and render bounded output instead.
767    #[test]
768    fn test_decimal_pathological_scale_is_bounded_not_panic() {
769        // i32::MIN scale: the negation-overflow reproducer.
770        let d_min = Value::Decimal {
771            scale: i32::MIN,
772            unscaled: vec![0x01],
773        };
774        let s = ValueFormatter::format_value(&d_min);
775        assert!(
776            s.contains('e'),
777            "extreme scale renders in exponent form: {s}"
778        );
779        assert!(
780            s.len() < 64,
781            "must not materialize an unbounded string: {s}"
782        );
783
784        // Huge positive scale: no multi-hundred-megabyte padding allocation.
785        let d_max = Value::Decimal {
786            scale: i32::MAX,
787            unscaled: vec![0x01],
788        };
789        let s2 = ValueFormatter::format_value(&d_max);
790        assert!(
791            s2.len() < 64,
792            "must not materialize an unbounded string: {s2}"
793        );
794    }

796    /// Issue #1754 (follow-up correctness fix): a large-but-WELL-FORMED unscaled
797    /// magnitude (thousands of digits, above the 1024-byte positional threshold
798    /// but within the sanity ceiling) must render FAITHFULLY (precision-preserving
799    /// exponent form) and FAST — NOT be misclassified as corruption. This is the
800    /// behavior the owner mandated: an arbitrary-precision BigInteger value is not
801    /// corrupt just for being big.
802    #[test]
803    fn test_decimal_large_valid_renders_faithfully_fast() {
804        // 2 KB magnitude (~4930 decimal digits): over the positional threshold,
805        // under the ceiling. 0x7f keeps the high bit clear (a large POSITIVE
806        // value; all-0xff would be two's-complement -1). Scale 4 → ×10^-4.
807        let unscaled = vec![0x7fu8; 2048];
808        let start = std::time::Instant::now();
809        let s = ValueFormatter::format_value(&Value::Decimal { scale: 4, unscaled });
810        let elapsed = start.elapsed();
811        assert!(
812            !s.starts_with("<corrupt-decimal:"),
813            "a well-formed large decimal must not be called corrupt: {s}"
814        );
815        // Precision-preserving exponent form: all significant digits + "e-4".
816        assert!(
817            s.ends_with("e-4"),
818            "expected exponent form, got: {}",
819            &s[..40]
820        );
821        let digits = s.trim_end_matches("e-4");
822        // 2 KB of 0xff is a ~4933-digit integer — every digit is preserved.
823        assert!(
824            digits.len() >= 4900 && digits.chars().all(|c| c.is_ascii_digit()),
825            "expected full digit string, got {} chars",
826            digits.len()
827        );
828        assert!(
829            elapsed < std::time::Duration::from_millis(500),
830            "expected fast single-conversion render, took {elapsed:?}"
831        );
832    }
