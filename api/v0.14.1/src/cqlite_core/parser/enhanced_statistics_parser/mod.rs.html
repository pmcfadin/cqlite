1//! Enhanced Statistics.db parser for Cassandra 5.0 'nb' format
2//!
3//! # Implementation Status (Issue #162)
4//!
5//! This module provides **MINIMAL PARSING** of nb-format Statistics.db files to support
6//! delta-coded timestamp decoding in V5CompressedLegacy parser.
7//!
8//! ## Current Implementation
9//!
10//! Parses ONLY the EncodingStats fields required for delta decoding:
11//! - Header (32 bytes): version, data_length, checksum, metadata
12//! - EncodingStats section: partitioner, minTimestamp, minLocalDeletionTime, minTTL
13//!
14//! All other statistics (row counts, histograms, column stats, etc.) are populated with
15//! placeholder values. This is sufficient for V5CompressedLegacy parser baseline values.
16//!
17//! ## Previous Implementation (REMOVED)
18//!
19//! The previous implementation violated the no-heuristics mandate (Issue #28) by fabricating
20//! statistics from header metadata. It was removed and replaced with this minimal real-data
21//! parser that extracts only what's needed from the actual binary format.
22//!
23//! ## Deferred to Future Milestones
24//!
25//! Complete Statistics.db parsing including:
26//! - Row count statistics and distribution histograms
27//! - Column-level statistics and cardinality estimates
28//! - Partition size histograms and percentiles
29//! - Compression ratio and performance metrics
30//! - Checksum validation (header.checksum field not yet validated)
31//!
32//! ## References
33//!
34//! - Issue #162: Fix Statistics reader for Cassandra 5 nb format
35//! - Issue #28: No-heuristics mandate for modern Cassandra 5.0 paths
36//! - Issue #105: Remove heuristic estimation from enhanced_statistics_parser.rs
37//! - `docs/development/rust_developer_guide.md`: Architecture decisions
38//!
39//! # Module layout (Issue #1125 split)
40//!
41//! - [`header`] — 32-byte file header + Table-of-Contents (component offsets).
42//! - [`encoding_stats`] — EncodingStats (min timestamp/deletion-time/TTL) decode.
43//! - [`serialization_header`] — table-schema (partition/clustering/static/regular columns).
44//! - [`marshal_type`] — Cassandra marshal-type → CQL conversion + `ColumnInfo` builders.
45//! - this `mod.rs` — the public entry points that orchestrate the above.

73/// Parse minimal nb-format statistics data for delta-coding baseline (Issue #162)
74///
75/// This implementation parses ONLY the EncodingStats fields required for delta decoding:
76/// - partitioner (string)
77/// - minTimestamp (VInt)
78/// - minLocalDeletionTime (VInt)
79/// - minTTL (VInt)
80///
81/// All other fields (histograms, column stats, etc.) are skipped to minimize complexity.
82/// This is sufficient for V5CompressedLegacy parser which needs baseline values for
83/// delta-coded timestamps and TTLs.
84///
85/// # Format (observed from real nb-format Statistics.db files)
86///
87/// After 32-byte header:
88/// - metadata_type (u32 BE) = 0x00000003 (indicates EncodingStats section)
89/// - data_length (VInt) - length of remaining data
90/// - partitioner_length (VInt) - length of partitioner class name string
91/// - partitioner (UTF-8 string) - e.g., "org.apache.cassandra.dht.Murmur3Partitioner"
92/// - additional_metadata (various VInts) - skipped
93/// - minTimestamp (VInt, microseconds)
94/// - minLocalDeletionTime (VInt, seconds)
95/// - minTTL (VInt, seconds)
96///
97/// # Returns
98///
99/// Partial statistics with only TimestampStatistics populated from real data.
100#[allow(clippy::type_complexity)]
101pub fn parse_nb_format_statistics_data(
102    input: &[u8],
103    header: &StatisticsHeader,
104    full_input: &[u8],
105    // VG3 plumbing: gates are threaded here so version-sensitive decisions in
106    // parse_encoding_stats_vuints (e.g. has_uint_deletion_time) can be flipped
107    // without re-deriving gates from the filename.
108    // Pass `None` from callers that do not have gates (standalone tools, tests).
109    gates: Option<&VersionGates>,
110) -> Result<(
111    RowStatistics,
112    TimestampStatistics,
113    Vec<super::header::ColumnInfo>,
114    Vec<super::header::ColumnInfo>,
115    Vec<super::header::ColumnInfo>,
116)> {
117    // Get HEADER offset from TOC (Issue #216)
118    let header_offset = parse_statistics_toc_for_header_offset(full_input);
119
120    // Parse the EncodingStats section from the data following the header
121    let result = parse_minimal_encoding_stats(input, full_input, header_offset, gates);
122
123    match result {
124        Ok((
125            _,
126            (
127                min_timestamp,
128                min_deletion_time,
129                min_ttl,
130                partition_columns,
131                clustering_columns,
132                regular_columns,
133            ),
134        )) => {
135            // Populate the authoritative row/partition counts from the STATS
136            // component (issue #1325). We reuse the single source of truth added
137            // in #944 — `repair_metadata::read_table_counts` — rather than
138            // re-deriving the walk here:
139            //   * `partition_count` = Σ `estimatedPartitionSize` histogram bucket
140            //     counts (self-describing; always decodable).
141            //   * `total_rows`      = STATS `totalRows` via the version-gated walk,
142            //     `None` when not authoritatively traversable (e.g. a clustered
143            //     covered-Slice whose bound values are not modeled).
144            // No-heuristics mandate (#28): when a value is NOT authoritatively
145            // reachable we leave the documented 0 rather than fabricating a count.
146            //   * `total_rows`: `None` from `read_table_counts` → 0 (meaning
147            //     "not authoritatively reachable", never a guessed count).
148            //   * `live_rows`: STATS carries no per-SSTable live-row count that is
149            //     authoritatively distinguishable from `total_rows` here, so it is
150            //     left as 0 (documented "not authoritatively derivable") rather
151            //     than guessed. (Consumers that need an upper bound use
152            //     `total_rows`; see cqlite-flight `read_table_counts` doctrine.)
153            //   * `avg_rows_per_partition`: computed only when BOTH counts are
154            //     authoritative and `partition_count > 0`; otherwise left 0.0.
155            // `full_input` is the raw Statistics.db buffer (with TOC), exactly the
156            // input `read_table_counts` expects. This is strictly additive — an
157            // Err is logged and the fields stay at 0 (a Statistics.db that parses
158            // today must keep parsing).
159            //
160            // No-heuristics gate discipline (#28, #1325 roborev finding): the
161            // version-gated `read_table_counts` walk MUST run with gates that match
162            // the file's actual on-disk layout — Statistics.db does NOT self-describe
163            // its SSTable version (the version comes from the filename → the gates).
164            // The `nb` and `oa`/`da` STATS min/max blocks have DIFFERENT layouts, so
165            // synthesizing `nb` gates for a file whose format we have not
166            // authoritatively established would mis-walk an `oa`/`da` buffer and could
167            // read a BOGUS nonzero `totalRows`. This function is the SINGLE entry point
168            // for `parse_statistics_with_fallback`, which is a public API reachable with
169            // `None` gates for ANY format (standalone tools, tests) — there is no
170            // format-detection dispatch that guarantees only `nb` bytes arrive here.
171            //
172            // Therefore we pass the caller's gates through UNCHANGED and NEVER
173            // synthesize a version:
174            //   * `Some(gates)` — the caller (e.g. `StatisticsReader::open`) derived
175            //     these from the filename, so they are authoritative for the file's
176            //     real version (nb / oa / da). The gated walk reaches `totalRows`.
177            //   * `None` — the format is UNKNOWN. We cannot authoritatively establish
178            //     it, so `read_table_counts` stops before the gated min/max block and
179            //     reports `total_rows = None`. We surface that as `0` meaning "not
180            //     authoritatively available", NEVER a guessed count. This is strictly
181            //     safer than the old `nb` synthesis, which could fabricate a nonzero
182            //     count for an oa/da file.
183            let (total_rows, partition_count) =
184                match crate::parser::repair_metadata::read_table_counts(full_input, gates) {
185                    // `total_rows.unwrap_or(0)`: a `None` here means STATS does NOT
186                    // authoritatively expose `totalRows` (e.g. an unmodeled
187                    // improvedMinMax covered-Slice bound blocks the walk). The 0 is
188                    // therefore "not authoritatively available from STATS", NOT a
189                    // measured zero — never a guessed count (#1325, no-heuristics #28).
190                    Ok(counts) => (counts.total_rows.unwrap_or(0), counts.partition_count),
191                    Err(e) => {
192                        tracing::debug!(
193                            "Best-effort authoritative row/partition-count decode failed; \
194                             leaving RowStatistics counts at 0: {:?}",
195                            e
196                        );
197                        (0, 0)
198                    }
199                };
200            let avg_rows_per_partition = if partition_count > 0 && total_rows > 0 {
201                total_rows as f64 / partition_count as f64
202            } else {
203                0.0
204            };
205
206            let row_stats = RowStatistics {
207                total_rows,
208                // `live_rows = 0` here means "NOT authoritatively available from
209                // STATS for nb", NOT a measured zero. STATS carries no per-SSTable
210                // live-row count that is authoritatively distinguishable from
211                // `total_rows`, so per #1325's acceptance criteria this unreachable
212                // field stays "0-with-documented-meaning" rather than being guessed
213                // or aliased to `total_rows` (no-heuristics mandate #28). Changing
214                // this field to `Option` is a broader API change tracked separately.
215                live_rows: 0,
216                tombstone_count: 0,
217                partition_count,
218                avg_rows_per_partition,
219                row_size_histogram: vec![],
220            };
221
222            let timestamp_stats = TimestampStatistics {
223                min_timestamp,
224                // `None` = maxTimestamp NOT yet authoritatively available (issue
225                // #1653). The real `maxTimestamp` is recovered from STATS field 4
226                // by the best-effort post-pass in `parse_enhanced_statistics_file`
227                // (issue #1729), which sets `Some(..)`. The inner EncodingStats
228                // parser does not reach field 4, so it stays `None` here — an
229                // honest "unavailable", NOT the old `i64::MIN` sentinel or the even
230                // older `min_timestamp` alias (both lied about the max). A consumer
231                // that needs a real max must treat `None` as unavailable.
232                max_timestamp: None,
233                min_deletion_time,
234                // Genuinely parsed by the STATS post-pass (#1073/#1011), which
235                // overwrites this below. Until then use the authoritative
236                // `NO_DELETION_TIME` (`i64::MAX`) "no deletions recorded" value so a
237                // post-pass failure fails CLOSED (never classified fully-expired),
238                // NOT the old `= min_deletion_time` placeholder which reported the
239                // MIN as the max (issue #1653).
240                max_deletion_time: i64::MAX,
241                min_ttl,
242                // `None` = maxTTL not authoritatively available (issue #1653). The
243                // enhanced parser does not decode a per-SSTable max TTL, so it is
244                // honestly `None`, NOT the old `= min_ttl` alias (which lied).
245                max_ttl: None,
246                // `None` = rows-with-TTL count not authoritatively available (issue
247                // #1653), NOT a fabricated `0` (which claimed "no rows have a TTL").
248                rows_with_ttl: None,
249            };
250
251            // Table/partition/compression statistics are NOT decoded by the
252            // enhanced (nb) parser. They are left `None` on `SSTableStatistics`
253            // (built by the caller) rather than fabricated with all-zero /
254            // "unknown" placeholders that were type-indistinguishable from real
255            // values (issue #1653; no-heuristics/no-fabrication mandate #28).
256            Ok((
257                row_stats,
258                timestamp_stats,
259                partition_columns,
260                clustering_columns,
261                regular_columns,
262            ))
263        }
264        Err(e) => {
265            tracing::debug!(
266                "Failed to parse minimal EncodingStats from Statistics.db: {:?}",
267                e
268            );
269            Err(Error::UnsupportedFormat(format!(
270                "Failed to parse minimal nb-format Statistics.db EncodingStats: {:?}. \
271                         This is required for delta-coded timestamp decoding. \
272                         Header checksum: 0x{:08x}, data_length: {}",
273                e, header.checksum, header.data_length
274            )))
275        }
276    }
277}

279/// Main enhanced parser for real Statistics.db files (minimal implementation for Issue #162)
280///
281/// This function parses the header and minimal EncodingStats fields from nb-format
282/// Statistics.db files. Only timestamp-related fields are extracted; all other
283/// statistics (histograms, column stats, etc.) are populated with placeholder values.
284///
285/// This is sufficient for V5CompressedLegacy parser which requires min_timestamp,
286/// min_local_deletion_time, and min_ttl for delta decoding baseline.
287///
288/// # Arguments
289///
290/// * `gates` - Optional [`VersionGates`] for version-sensitive decoding decisions
291///   (VG1 plumbing).  Pass `None` from standalone tools/tests to use nb-compatible
292///   defaults; pass `Some(&gates)` from `SSTableReader` to enable VG3 gating.
293///
294/// # Returns
295///
296/// SSTableStatistics with only header and timestamp_stats populated from real data.
297pub fn parse_enhanced_statistics_file<'a>(
298    input: &'a [u8],
299    gates: Option<&VersionGates>,
300) -> nom::IResult<&'a [u8], SSTableStatistics> {
301    // Parse the 32-byte header
302    let (remaining, header) = parse_nb_format_header(input)?;
303
304    // Parse minimal statistics data (EncodingStats + SerializationHeader columns)
305    // Pass full input for TOC-based HEADER offset lookup (Issue #216)
306    let result = parse_nb_format_statistics_data(remaining, &header, input, gates);
307
308    match result {
309        Ok((row_stats, mut timestamp_stats, partition_columns, clustering_columns, columns)) => {
310            tracing::debug!(
311                "Successfully parsed Statistics.db serialization header: {} partition keys, {} clustering keys, {} regular columns",
312                partition_columns.len(),
313                clustering_columns.len(),
314                columns.len()
315            );
316
317            // Best-effort STATS-extras decode over the FULL buffer (with TOC):
318            // recover the authoritative maxLocalDeletionTime (the inner parser
319            // leaves it as a placeholder equal to min_deletion_time), the
320            // authoritative maxTimestamp (issue #1729 — the inner parser leaves
321            // it as the `i64::MIN` "unavailable" sentinel), and the estimated
322            // tombstone-drop-times histogram (Issue #1073). This is strictly
323            // additive — a Statistics.db that parses today must keep parsing, so
324            // an Err here is logged and the placeholders are kept. When
325            // maxTimestamp is unavailable (extras.max_timestamp == None, i.e.
326            // Cassandra's Long.MIN_VALUE sentinel) we deliberately leave
327            // `max_timestamp == None` in place (issue #1653) rather than
328            // substituting the min — consumers requiring a real max must not
329            // proceed.
330            let tombstone_drop_times =
331                match crate::parser::repair_metadata::parse_stats_extras(input, gates) {
332                    Ok(extras) => {
333                        // Only set max_deletion_time. Do NOT touch min_deletion_time
334                        // (the EncodingStats min baseline consumed elsewhere).
335                        timestamp_stats.max_deletion_time = extras.max_local_deletion_time;
336                        // `extras.max_timestamp` is already `Option`: `Some` only
337                        // when the authoritative max was decoded (#1729). Propagate
338                        // it directly — `None` stays `None` (issue #1653).
339                        if let Some(max_ts) = extras.max_timestamp {
340                            timestamp_stats.max_timestamp = Some(max_ts);
341                        }
342                        // Authoritative maxTTL (issue #1537): the inner nb walk only
343                        // decodes the EncodingStats minTTL baseline and honestly
344                        // leaves `max_ttl = None`. Recover the true maximum from the
345                        // STATS-extras walk so the compaction TTL-expiry LDT floor
346                        // (`compute_expiry_ttl_ldt_floor`) can lower the output's
347                        // min-LDT delta baseline below a creation-time (`ldt - ttl`)
348                        // tombstone. `None` (no expiring cells) stays `None`.
349                        if extras.max_ttl.is_some() {
350                            timestamp_stats.max_ttl = extras.max_ttl;
351                        }
352                        extras.tombstone_drop_times
353                    }
354                    Err(e) => {
355                        tracing::debug!(
356                            "Best-effort STATS-extras decode failed; keeping placeholders: {:?}",
357                            e
358                        );
359                        vec![]
360                    }
361                };
362
363            let statistics = SSTableStatistics {
364                header,
365                row_stats,
366                timestamp_stats,
367                column_stats: vec![],
368                // NOT decoded by the enhanced (nb) parser → honestly `None`
369                // rather than fabricated all-zero / "unknown" placeholders
370                // (issue #1653; no-heuristics mandate #28).
371                table_stats: None,
372                partition_stats: None,
373                compression_stats: None,
374                metadata: std::collections::HashMap::new(),
375                serialization_header_columns: columns,
376                serialization_header_partition_keys: partition_columns,
377                serialization_header_clustering_keys: clustering_columns,
378                tombstone_drop_times,
379            };
380
381            Ok((remaining, statistics))
382        }
383        Err(e) => {
384            // Convert Error to nom::Err
385            tracing::warn!("Failed to parse nb-format Statistics.db: {}", e);
386            Err(nom::Err::Error(nom::error::Error::new(
387                input,
388                nom::error::ErrorKind::Verify,
389            )))
390        }
391    }
392}

394/// Enhanced statistics reader with fallback (minimal implementation for Issue #162)
395///
396/// Attempts to parse nb-format Statistics.db with minimal EncodingStats extraction.
397/// This provides the minimum fields needed for delta-coded timestamp decoding.
398///
399/// # Arguments
400///
401/// * `gates` - Optional [`VersionGates`] threaded from `SSTableReader` for VG3
402///   version-sensitive decoding.  Pass `None` from standalone tools/tests; the
403///   nb-compatible behaviour is used when `gates` is `None`.
404///
405/// # Returns
406///
407/// SSTableStatistics with minimal fields populated, or error if parsing fails.
408pub fn parse_statistics_with_fallback<'a>(
409    input: &'a [u8],
410    gates: Option<&VersionGates>,
411) -> nom::IResult<&'a [u8], SSTableStatistics> {
412    // Try the minimal enhanced parser
413    parse_enhanced_statistics_file(input, gates)
414}
