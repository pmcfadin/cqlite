1//! Repair-state metadata decode from the `Statistics.db` STATS component
2//! (issue #988, epic #968).
3//!
4//! Apache Cassandra persists three repair-coordination fields inside the STATS
5//! `MetadataType` component of `Statistics.db`:
6//!
7//!   * `repairedAt` (`long`) — the repair timestamp, `0` for an unrepaired
8//!     SSTable. Rendered by `sstablemetadata` as `Repaired at: <n>`.
9//!   * `pendingRepair` (`UUID`, nullable) — the in-flight (incremental) repair
10//!     session id, `null` for an SSTable not part of a pending repair. Rendered
11//!     as `Pending repair: --` when null.
12//!   * `isTransient` (`boolean`) — whether the SSTable holds only transiently
13//!     replicated data. Rendered as `IsTransient: false`.
14//!
15//! # Scope — persisted-metadata parse/report ONLY
16//!
17//! This module decodes and **reports** the *persisted* repair state. It does
18//! **not** implement, and must not be read as implying, repair coordination,
19//! incremental-repair session tracking, or **repair-aware compaction /
20//! tombstone purging**. Exposing `repairedAt` here establishes nothing about
21//! whether tombstones are safe to purge against a repair boundary — that is a
22//! separate correctness concern that lives in the compaction layer.
23//!
24//! # What is decoded from real bytes
25//!
26//! `repairedAt` is decoded directly from the STATS component. It is reachable by
27//! a fully *self-describing* forward walk over the leading STATS fields (two
28//! `EstimatedHistogram`s and one `TombstoneHistogram`, every one length-prefixed
29//! and free of any column-type dependency), so it is decoded for **every**
30//! storage format (nb / oa / da) without needing the serialization header.
31//!
32//! `pendingRepair` and `isTransient` sit *after* the version-gated
33//! `improvedMinMax` / `legacyMinMax` block and the variable `commitLogIntervals`
34//! set, both of which require version-gated / type-aware skipping to traverse.
35//!
36//! # Full repair-state walk (issue #1021)
37//!
38//! When the caller supplies authoritative [`VersionGates`],
39//! [`parse_repair_metadata`] now performs that version-gated forward walk and
40//! decodes `pendingRepair` (UUID) and `isTransient` **from real bytes** in
41//! addition to `repairedAt`. The walk is driven ENTIRELY by the version gates
42//! (`hasLegacyMinMax` / `hasImprovedMinMax` / `hasUIntDeletionTime` /
43//! `hasPendingRepair` / `hasIsTransient`) read from the parsed descriptor — no
44//! heuristics, no type guessing (#28). Mirrors the exact field order of
45//! cassandra-5.0.0 `StatsMetadata.StatsMetadataSerializer.serialize`, the same
46//! layout the `da` STATS writer emits (`build_stats_component_da`).
47//!
48//! When `gates` is `None` the walk past `repairedAt` is NOT attempted (the
49//! min/max block and the commit-log-interval shape are version-dependent and
50//! cannot be traversed without the descriptor), so `pendingRepair` /
51//! `isTransient` are reported honestly as [`RepairField::Unparsed`] rather than a
52//! fabricated `null` / `false` — a real SSTable that carried a pending-repair
53//! UUID or transient flag is never silently misreported as absent.

61/// Cassandra `MetadataType.STATS` ordinal (MetadataType.java).
62const METADATA_TYPE_STATS: u32 = 2;

64/// A STATS repair field that may either have been decoded from real bytes
65/// (`Decoded`) or is honestly reported as not-yet-decoded (`Unparsed`).
66///
67/// This exists so the public API never fabricates a concrete value for a field
68/// this module does not actually walk from the STATS bytes. A real SSTable
69/// carrying a pending-repair UUID or a transient flag is reported as `Unparsed`
70/// — never silently misreported as the absent/false default — until a
71/// type-aware forward walk past `improvedMinMax` + `commitLogIntervals` is
72/// implemented (which requires a fixture that does not exist in the corpus; see
73/// module docs).
74#[derive(Debug, Clone, PartialEq, Eq)]
75pub enum RepairField<T> {
76    /// The field was decoded directly from the STATS-component bytes.
77    Decoded(T),
78    /// The field was not decoded; its real value is unknown. Callers must NOT
79    /// treat this as the absent/default state.
80    Unparsed,
81}
82
83impl<T> RepairField<T> {
84    /// The decoded value, or `None` when the field is `Unparsed`. Callers that
85    /// need to distinguish "decoded as absent" from "not decoded" must match on
86    /// the variant directly rather than using this.
87    pub fn decoded(&self) -> Option<&T> {
88        match self {
89            RepairField::Decoded(v) => Some(v),
90            RepairField::Unparsed => None,
91        }
92    }
93
94    /// Whether this field was decoded from real bytes.
95    pub fn is_decoded(&self) -> bool {
96        matches!(self, RepairField::Decoded(_))
97    }
98}

100/// Repair-coordination metadata persisted in the `Statistics.db` STATS
101/// component. Read-side / report-only (see module docs).
102#[derive(Debug, Clone, PartialEq, Eq)]
103pub struct RepairMetadata {
104    /// `repairedAt` timestamp; `0` for an unrepaired SSTable. Decoded directly
105    /// from the STATS-component bytes (genuinely walked, so concrete).
106    pub repaired_at: i64,
107
108    /// `pendingRepair` session UUID (`Decoded(Some(uuid))`), an explicitly
109    /// decoded null (`Decoded(None)`), or `Unparsed` when this module did not
110    /// decode it from the STATS bytes.
111    ///
112    /// Decoded from real bytes by the version-gated forward walk when
113    /// authoritative [`VersionGates`] are supplied. `hasPendingRepair` is
114    /// `>= na`, so the field is present and decoded for every Cassandra 5.0
115    /// format CQLite reads (`nb`/`oa`/`da`). `Unparsed` is reserved for "present
116    /// but not safely reachable" (an unmodeled clustering comparator) or when
117    /// `gates` is `None` (the walk past `repairedAt` is not attempted). See

121    /// `isTransient` flag (`Decoded(bool)`) or `Unparsed` when not decoded.
122    ///
123    /// Decoded from real bytes by the version-gated forward walk when
124    /// authoritative [`VersionGates`] are supplied. `hasIsTransient` is `>= na`,
125    /// so the field is present and decoded for every supported Cassandra 5.0
126    /// format (`nb`/`oa`/`da`). `Unparsed` is reserved for "present but not
127    /// safely reachable" or when `gates` is `None`.

151/// Number of CRC bytes Cassandra writes after EACH metadata component. The
152/// `Statistics.db` MetadataSerializer emits a 4-byte CRC32 (over the component's
153/// own bytes) immediately after every component body — including the last.
154/// Verified empirically against real nb/oa/da fixtures: for a component at TOC
155/// offset `o` whose successor is at offset `next`, the bytes `[next-4, next)`
156/// are `crc32(buf[o..next-4])`; for the final component the CRC occupies the
157/// last 4 bytes of the file. The STATS component must never be decoded into this
158/// CRC, so both the next-component bound and the last-component bound subtract it.
159const METADATA_COMPONENT_CRC_LEN: usize = 4;
160
161/// The located STATS component, as a `[start, end)` byte range within the
162/// `Statistics.db` buffer. `end` is the *next* component's offset minus the
163/// 4-byte per-component CRC that sits between STATS and that component (the
164/// smallest component offset strictly greater than `start`), or — when STATS is
165/// the last component — the end of the file minus the trailing 4-byte CRC. The
166/// decode cursor is constructed over ONLY this slice so that a truncated STATS
167/// body or an over-long internal length field fails closed instead of spilling
168/// into the CRC or the following component.
169#[derive(Debug, Clone, Copy, PartialEq, Eq)]
170struct StatsComponentBounds {
171    start: usize,
172    end: usize,
173}
174
175/// Locate the byte range of the STATS (`MetadataType` ordinal 2) component from
176/// the `Statistics.db` Table of Contents.
177///
178/// The TOC carries `(type, offset)` pairs for every metadata component. The
179/// STATS component's end is derived authoritatively from those same offsets —
180/// the next component begins where STATS must end — rather than from the rest of
181/// the file. When STATS is the final component, its end is the start of the
182/// trailing CRC.
183///
184/// Every Cassandra 5.0 format CQLite reads (`nb` BIG, `oa` BIG, `da` BTI) is
185/// `>= na`, so `hasMetadataChecksum` is always true and
186/// `MetadataSerializer.serialize` always writes a 4-byte CRC32 after each
187/// component body (`MetadataSerializer.java:110-111,117`). The body bound
188/// therefore always subtracts that CRC. (Pre-`na` BIG versions — `ma`–`me` — are
189/// out of scope: CQLite is a Cassandra 5.0 reader and does not open them.)
190///
191/// Returns:
192///   * `Ok(None)` ONLY when the TOC is well-formed but carries NO STATS
193///     component (nothing to decode → the caller reports the unrepaired
194///     default).
195///   * `Ok(Some(bounds))` for a valid STATS range.
196///   * `Err(Corruption)` for a malformed/truncated TOC (too short for the
197///     header, component count out of range, or a TOC that overruns the
198///     buffer), OR when a STATS entry IS present but its derived range is
199///     invalid (offset past EOF, inverted, or zero-length) — fail closed so a
200///     corrupt `Statistics.db` is never silently reported as unrepaired.
201fn stats_component_bounds(input: &[u8]) -> Result<Option<StatsComponentBounds>> {
202    // Count this TOC walk (issue #1658 A5 bench instrumentation — no decode effect).
203    crate::parser::toc_walk_metrics::record_toc_walk();
204    // A malformed or truncated TOC must FAIL CLOSED rather than be silently
205    // reported as "no STATS component" (which would misreport corrupt repair
206    // metadata as the unrepaired default). `Ok(None)` is reserved exclusively
207    // for a well-formed TOC that genuinely carries no STATS entry (below).
208    if input.len() < 8 {
209        return Err(Error::Corruption(format!(
210            "Statistics.db too short for a metadata TOC header: {} bytes",
211            input.len()
212        )));
213    }
214    let num_components = u32::from_be_bytes([input[0], input[1], input[2], input[3]]);
215    if num_components == 0 || num_components > 100 {
216        return Err(Error::Corruption(format!(
217            "Statistics.db TOC component count out of range: {num_components}"
218        )));
219    }
220    let toc_start = 8usize;
221    let entry_size = 8usize;
222    let toc_size = match (num_components as usize)
223        .checked_mul(entry_size)
224        .and_then(|n| n.checked_add(toc_start))
225    {
226        Some(n) => n,
227        None => {
228            return Err(Error::Corruption(format!(
229                "Statistics.db TOC size overflow for {num_components} components"
230            )))
231        }
232    };
233    if input.len() < toc_size {
234        return Err(Error::Corruption(format!(
235            "Statistics.db TOC truncated: need {toc_size} bytes, have {}",
236            input.len()
237        )));
238    }
239
240    let mut stats_off: Option<usize> = None;
241    let mut offsets: Vec<usize> = Vec::with_capacity(num_components as usize);
242    for i in 0..num_components as usize {
243        let entry = match i
244            .checked_mul(entry_size)
245            .and_then(|n| n.checked_add(toc_start))
246        {
247            Some(e) => e,
248            None => {
249                return Err(Error::Corruption(format!(
250                    "Statistics.db TOC entry offset overflow at index {i}"
251                )))
252            }
253        };
254        let ty = u32::from_be_bytes([
255            input[entry],
256            input[entry + 1],
257            input[entry + 2],
258            input[entry + 3],
259        ]);
260        let off = u32::from_be_bytes([
261            input[entry + 4],
262            input[entry + 5],
263            input[entry + 6],
264            input[entry + 7],
265        ]) as usize;
266        offsets.push(off);
267        if ty == METADATA_TYPE_STATS {
268            stats_off = Some(off);
269        }
270    }
271
272    // No STATS component in the TOC → nothing to decode (not an error).
273    let Some(start) = stats_off else {
274        return Ok(None);
275    };
276
277    // The STATS body ends where the *next* component (or the file) begins. For
278    // The STATS body ends 4 bytes before the *next* component begins: every
279    // supported Cassandra 5.0 format (`nb`/`oa`/`da`, all `>= na`) writes a
280    // per-component CRC32 between each component body and the next component's
281    // offset (`MetadataSerializer.java:110-111`). Use the smallest component
282    // offset strictly greater than `start`, minus that CRC; when STATS is the last
283    // component, bound by the file end minus the trailing CRC. Both cases exclude
284    // the 4-byte CRC so it is never decoded as metadata.
285    let next_boundary = offsets
286        .iter()
287        .copied()
288        .filter(|&o| o > start)
289        .min()
290        .unwrap_or(input.len());
291    let end = next_boundary.saturating_sub(METADATA_COMPONENT_CRC_LEN);
292
293    // A STATS entry exists but its derived range is invalid (offset past EOF,
294    // inverted, or empty): this is genuine corruption — fail closed rather than
295    // silently reporting the unrepaired default or building a cursor over
296    // garbage.
297    if start >= end || end > input.len() {
298        return Err(Error::Corruption(format!(
299            "STATS component range invalid: start {start}, derived end {end}, \
300             Statistics.db {} bytes",
301            input.len()
302        )));
303    }
304
305    Ok(Some(StatsComponentBounds { start, end }))
306}

308/// A bounded cursor over the STATS component bytes that fails closed (explicit
309/// error) on any read past the end rather than panicking.
310struct Cursor<'a> {
311    bytes: &'a [u8],
312    pos: usize,
313}
314
315impl<'a> Cursor<'a> {
316    fn new(bytes: &'a [u8]) -> Self {
317        Cursor { bytes, pos: 0 }
318    }
319
320    fn need(&self, n: usize) -> Result<usize> {
321        let end = self
322            .pos
323            .checked_add(n)
324            .ok_or_else(|| Error::Corruption("STATS cursor overflow".to_string()))?;
325        if end > self.bytes.len() {
326            return Err(Error::Corruption(format!(
327                "STATS component truncated: need {n} bytes at offset {} but only {} remain",
328                self.pos,
329                self.bytes.len().saturating_sub(self.pos),
330            )));
331        }
332        Ok(end)
333    }
334
335    fn skip(&mut self, n: usize) -> Result<()> {
336        let end = self.need(n)?;
337        self.pos = end;
338        Ok(())
339    }
340
341    fn read_i32(&mut self) -> Result<i32> {
342        let end = self.need(4)?;
343        let v = i32::from_be_bytes([
344            self.bytes[self.pos],
345            self.bytes[self.pos + 1],
346            self.bytes[self.pos + 2],
347            self.bytes[self.pos + 3],
348        ]);
349        self.pos = end;
350        Ok(v)
351    }
352
353    fn read_i64(&mut self) -> Result<i64> {
354        let end = self.need(8)?;
355        let mut buf = [0u8; 8];
356        buf.copy_from_slice(&self.bytes[self.pos..end]);
357        self.pos = end;
358        Ok(i64::from_be_bytes(buf))
359    }
360
361    fn read_u32(&mut self) -> Result<u32> {
362        let end = self.need(4)?;
363        let v = u32::from_be_bytes([
364            self.bytes[self.pos],
365            self.bytes[self.pos + 1],
366            self.bytes[self.pos + 2],
367            self.bytes[self.pos + 3],
368        ]);
369        self.pos = end;
370        Ok(v)
371    }
372
373    fn read_f64(&mut self) -> Result<f64> {
374        let end = self.need(8)?;
375        let mut buf = [0u8; 8];
376        buf.copy_from_slice(&self.bytes[self.pos..end]);
377        self.pos = end;
378        Ok(f64::from_be_bytes(buf))
379    }
380
381    fn read_u8(&mut self) -> Result<u8> {
382        let end = self.need(1)?;
383        let v = self.bytes[self.pos];
384        self.pos = end;
385        Ok(v)
386    }
387
388    fn read_u16(&mut self) -> Result<u16> {
389        let end = self.need(2)?;
390        let v = u16::from_be_bytes([self.bytes[self.pos], self.bytes[self.pos + 1]]);
391        self.pos = end;
392        Ok(v)
393    }
394
395    /// Read a 16-byte raw UUID (the two `long`s Cassandra writes via
396    /// `UUIDSerializer`: most-significant then least-significant bits, each BE).
397    fn read_uuid(&mut self) -> Result<[u8; 16]> {
398        let end = self.need(16)?;
399        let mut buf = [0u8; 16];
400        buf.copy_from_slice(&self.bytes[self.pos..end]);
401        self.pos = end;
402        Ok(buf)
403    }
404
405    /// Read an unsigned VInt (Cassandra `VIntCoding.readUnsignedVInt`):
406    /// the first byte's leading 1-bits give the number of EXTRA bytes; the
407    /// remaining low bits of the first byte are the high-order value bits.
408    fn read_unsigned_vint(&mut self) -> Result<u64> {
409        let first = self.read_u8()?;
410        let extra = first.leading_ones() as usize;
411        if extra == 0 {
412            return Ok(first as u64);
413        }
414        if extra > 8 {
415            return Err(Error::Corruption(format!(
416                "invalid unsigned VInt: {extra} extra bytes"
417            )));
418        }
419        let mut value: u64 = 0;
420        for _ in 0..extra {
421            let b = self.read_u8()?;
422            value = (value << 8) | b as u64;
423        }
424        // The first byte contributes `8 - extra` low bits (its bits below the
425        // `extra` leading 1s and the separator 0). For extra == 8 there are no
426        // first-byte value bits.
427        if extra < 8 {
428            let mask = (1u8 << (8 - extra)) as u64;
429            // bits below the separator bit
430            let first_bits = first as u64 & (mask - 1);
431            value |= first_bits << (8 * extra);
432        }
433        Ok(value)
434    }
435
436    /// Read `n` raw bytes as an owned `Vec`, bounded by the cursor (fails closed
437    /// on overrun). Used for the clusteringTypes UTF-8 names.
438    fn read_bytes(&mut self, n: usize) -> Result<Vec<u8>> {
439        let end = self.need(n)?;
440        let v = self.bytes[self.pos..end].to_vec();
441        self.pos = end;
442        Ok(v)
443    }
444}

461/// Skip an `EstimatedHistogram`: `i32 count`, then `count` × (`i64 offset`,
462/// `i64 count`). Self-describing — no column-type knowledge required.
463fn skip_estimated_histogram(c: &mut Cursor) -> Result<()> {
464    let count = c.read_i32()?;
465    if count < 0 {
466        return Err(Error::Corruption(format!(
467            "negative EstimatedHistogram bucket count {count}"
468        )));
469    }
470    let bytes = (count as usize)
471        .checked_mul(16)
472        .ok_or_else(|| Error::Corruption("EstimatedHistogram size overflow".to_string()))?;
473    c.skip(bytes)
474}

476/// Skip a `TombstoneHistogram`: `i32 maxBinSize`, `i32 size`, then `size` bins.
477///
478/// `modern` selects the entry width: the modern (`oa`/`da`)
479/// `HistogramSerializer` writes `i64 point + i32 value` (12 bytes); the legacy
480/// (`nb`/`mc`) serializer writes `f64 point + i64 value` (16 bytes). The choice
481/// follows `TombstoneHistogram.getSerializer(version)` — the only format-gated
482/// decision needed to reach `repairedAt`.
483fn skip_tombstone_histogram(c: &mut Cursor, modern: bool) -> Result<()> {
484    let _max_bin_size = c.read_i32()?;
485    let size = c.read_i32()?;
486    if size < 0 {
487        return Err(Error::Corruption(format!(
488            "negative TombstoneHistogram size {size}"
489        )));
490    }
491    let entry_width = if modern { 12 } else { 16 };
492    let bytes = (size as usize)
493        .checked_mul(entry_width)
494        .ok_or_else(|| Error::Corruption("TombstoneHistogram size overflow".to_string()))?;
495    c.skip(bytes)
496}

498/// Whether this version uses the modern tombstone-histogram entry encoding.
499///
500/// `TombstoneHistogram.getSerializer` resolves to the modern `HistogramSerializer`
501/// for `oa`+ (incl. BTI `da`) and the legacy serializer for older versions.
502fn uses_modern_tombstone_histogram(gates: &VersionGates) -> bool {
503    match gates {
504        // The oa-only `hasUIntDeletionTime` gate cleanly separates the modern
505        // (oa/da) histogram encoding from the legacy (nb) one.
506        VersionGates::Big(g) => g.has_uint_deletion_time,
507        VersionGates::Bti(_) => true,
508    }
509}

511/// The version-gate flags the post-`repairedAt` forward walk needs, projected
512/// out of [`VersionGates`] so the walk depends only on the AUTHORITATIVE parsed
513/// descriptor (no heuristics, #28). Every flag mirrors a single Cassandra
514/// `BigFormat`/`BtiFormat` version gate consumed by
515/// `StatsMetadata.StatsMetadataSerializer.serialize`.
516#[derive(Debug, Clone, Copy)]
517struct RepairWalkGates {
518    /// Modern tombstone-histogram entry width (`oa`+/`da`).
519    modern_histogram: bool,
520    /// `hasImprovedMinMax` — emits clusteringTypes + a covered `Slice` in place
521    /// of the legacy min/max clustering-value lists.
522    has_improved_min_max: bool,
523    /// `hasLegacyMinMax` — emits min/max clustering-value lists (each
524    /// `i32 count` then `count` × vint-length value).
525    has_legacy_min_max: bool,
526    /// `hasPendingRepair` (`na`+/`da`) — the `pendingRepair` nullable-UUID field
527    /// is present in the body.
528    has_pending_repair: bool,
529    /// `hasIsTransient` (`na`+/`da`) — the `isTransient` boolean is present.
530    has_is_transient: bool,
531}
532
533impl RepairWalkGates {
534    fn from(gates: &VersionGates) -> Self {
535        match gates {
536            VersionGates::Big(g) => RepairWalkGates {
537                modern_histogram: g.has_uint_deletion_time,
538                has_improved_min_max: g.has_improved_min_max,
539                has_legacy_min_max: g.has_legacy_min_max,
540                has_pending_repair: g.has_pending_repair,
541                has_is_transient: g.has_is_transient,
542            },
543            VersionGates::Bti(g) => RepairWalkGates {
544                modern_histogram: true,
545                has_improved_min_max: g.has_improved_min_max,
546                has_legacy_min_max: g.has_legacy_min_max,
547                has_pending_repair: g.has_pending_repair,
548                has_is_transient: g.has_is_transient,
549            },
550        }
551    }
552}

554/// Skip the legacy min/max clustering-value lists (`hasLegacyMinMax`).
555///
556/// Each list is `i32 count` then `count` × a value written with
557/// `ByteBufferUtil.writeWithShortLength` — an UNSIGNED 16-bit length prefix
558/// (`writeShort`) then the raw bytes (cassandra-5.0.0
559/// `StatsMetadata.StatsMetadataSerializer.serialize` legacy branch). Verified
560/// against a real `nb` `system_schema` fixture (clustering values present).
561fn skip_legacy_min_max(c: &mut Cursor) -> Result<()> {
562    for _ in 0..2 {
563        let count = c.read_i32()?;
564        if count < 0 {
565            return Err(Error::Corruption(format!(
566                "negative legacy min/max clustering count {count}"
567            )));
568        }
569        for _ in 0..count {
570            let len = c.read_u16()? as usize;
571            c.skip(len)?;
572        }
573    }
574    Ok(())
575}

577/// Attempt to skip the improvedMinMax block (`hasImprovedMinMax`): a
578/// clusteringTypes list (unsigned-VInt count, then each type a vint-length UTF-8
579/// string) followed by the covered `Slice` (two `ClusteringBound`s, each
580/// `byte kind` + `short size` + `size` comparator-encoded values).
581///
582/// The covered-clustering bound VALUES are RAW comparator-encoded (no per-value
583/// length prefix), exactly as cassandra-5.0.0
584/// `ClusteringPrefix.Serializer.serializeValuesWithoutSize` writes them: 32-value
585/// header batches then each present value's bytes via `AbstractType.writeValue`
586/// (fixed-width raw, or `writeWithVIntLength` for variable types). We skip them
587/// authoritatively by resolving each clustering column's `valueLengthIfFixed()`
588/// from the persisted `clusteringTypes` and mirroring
589/// `ClusteringBoundOrBoundary.Serializer.skipValues` (see [`repair_clustering`]).
590///
591/// Returns `Ok(true)` when the whole block (types + covered Slice) was skipped, so
592/// the walk may continue to `pendingRepair`/`isTransient`. Returns `Ok(false)`
593/// ONLY when a clustering type is not modeled (its fixed-vs-variable nature is
594/// unknown) or a bound advertises more values than the types describe — i.e. the
595/// length of some value cannot be computed. In that case the caller reports the
596/// trailing fields honestly as `Unparsed` rather than guessing (#28). Empty
597/// bounds (`Slice.ALL`, the no-clustering shape) skip cleanly to `Ok(true)`.
598fn try_skip_improved_min_max(c: &mut Cursor) -> Result<bool> {
599    // clusteringTypes list → resolved per-column value layouts. The raw UTF-8
600    // type names are read through the same bounded cursor.
601    let layouts = {
602        // Borrow the cursor mutably inside the closure by reading bytes directly;
603        // read_clustering_type_layouts drives both the vint reads and the name
604        // reads through `c`, but the closure also needs `c`, so we inline the
605        // loop here instead of passing a borrow twice.
606        let count = ByteSkip::read_unsigned_vint(c)? as usize;
607        let mut v = Vec::with_capacity(count.min(64));
608        for _ in 0..count {
609            let len = ByteSkip::read_unsigned_vint(c)? as usize;
610            let bytes = c.read_bytes(len)?;
611            let name = std::str::from_utf8(&bytes).map_err(|_| {
612                Error::Corruption("clusteringTypes entry is not valid UTF-8".to_string())
613            })?;
614            v.push(resolve_clustering_value_layout(name));
615        }
616        v
617    };
618    // coveredClustering = Slice: start bound, then end bound, each skipped per the
619    // resolved layouts. A bool result distinguishes "skipped" from "unmodeled".
620    skip_covered_slice(c, &layouts)
621}

623/// Skip the `commitLogIntervals` IntervalSet: `i32 size` then `size` × an
624/// interval (`CommitLogPosition` start + end = 2 × (`i64 segmentId` +
625/// `i32 position`) = 24 bytes per interval).
626fn skip_commit_log_intervals(c: &mut Cursor) -> Result<()> {
627    let size = c.read_i32()?;
628    if size < 0 {
629        return Err(Error::Corruption(format!(
630            "negative commitLogIntervals size {size}"
631        )));
632    }
633    let bytes = (size as usize)
634        .checked_mul(24)
635        .ok_or_else(|| Error::Corruption("commitLogIntervals size overflow".to_string()))?;
636    c.skip(bytes)
637}

639/// Decode the repair-state metadata from a raw `Statistics.db` buffer.
640///
641/// `repairedAt` is decoded from the STATS component for every format. When
642/// authoritative [`VersionGates`] are supplied, the full version-gated forward
643/// walk continues past `repairedAt` and decodes `pendingRepair` and
644/// `isTransient` from real bytes too (issue #1021). When `gates` is `None`, the
645/// legacy (nb) tombstone-histogram width is assumed to reach `repairedAt`, and
646/// the two later fields are reported honestly as [`RepairField::Unparsed`] (the
647/// min/max block and commit-log-interval shape are version-dependent and cannot
648/// be traversed without the descriptor).
649///
650/// # Errors
651///
652/// Returns an error only when the STATS component is present but its fields are
653/// truncated/corrupt, OR overrun the STATS component's authoritative end bound —
654/// strict callers can rely on this to fail closed (a truncated body or an
655/// over-long internal length never spills into the trailing CRC or the following
656/// component). A *missing* STATS component is reported as the unrepaired default
657/// with `repaired_at_decoded = false`. A STATS entry that IS present but whose
658/// derived byte range is invalid (offset past EOF, inverted) is treated as
659/// corruption and fails closed.
660pub fn parse_repair_metadata(input: &[u8], gates: Option<&VersionGates>) -> Result<RepairMetadata> {
661    let walk = gates.map(RepairWalkGates::from);
662
663    let Some(bounds) = stats_component_bounds(input)? else {
664        return Ok(RepairMetadata::unrepaired_default());
665    };
666
667    let modern_histogram = walk.map(|w| w.modern_histogram).unwrap_or(false);
668
669    // Bound the cursor over ONLY the STATS component slice (start..end, with the
670    // trailing CRC and following components excluded). Any read past this slice
671    // fails closed with Error::Corruption.
672    let mut c = Cursor::new(&input[bounds.start..bounds.end]);
673
674    // 1-2. estimatedPartitionSize + estimatedCellPerPartitionCount.
675    skip_estimated_histogram(&mut c)?;
676    skip_estimated_histogram(&mut c)?;
677
678    // 3. commitLogUpperBound: i64 segmentId + i32 position.
679    c.skip(8 + 4)?;
680
681    // 4. minTimestamp, maxTimestamp.
682    c.skip(8 + 8)?;
683
684    // 5. min/maxLocalDeletionTime. Both encodings (nb 2× i32, oa/da 2× u32)
685    //    occupy 8 bytes total, so the width does not branch here.
686    c.skip(4 + 4)?;
687
688    // 6. minTTL, maxTTL.
689    c.skip(4 + 4)?;
690
691    // 7. compressionRatio (f64).
692    c.skip(8)?;
693
694    // 8. estimatedTombstoneDropTime (TombstoneHistogram) — the only format-gated
695    //    field width on the path to repairedAt.
696    skip_tombstone_histogram(&mut c, modern_histogram)?;
697
698    // 9. sstableLevel (i32), then repairedAt (i64).
699    let _sstable_level = c.read_i32()?;
700    let repaired_at = c.read_i64()?;
701
702    // Without authoritative gates the post-repairedAt min/max block and the
703    // commit-log-interval shape cannot be traversed; report the two later fields
704    // honestly as Unparsed rather than a fabricated null / false.
705    let Some(walk) = walk else {
706        return Ok(RepairMetadata {
707            repaired_at,
708            pending_repair: RepairField::Unparsed,
709            is_transient: RepairField::Unparsed,
710            repaired_at_decoded: true,
711        });
712    };
713
714    // 10. min/max clustering block. Cassandra serializes the improvedMinMax
715    //     variant when present, otherwise the legacy variant; a version that
716    //     carries neither (pre-mb) writes nothing here.
717    //
718    //     The improvedMinMax covered-clustering Slice carries comparator-encoded
719    //     bound VALUES that this repair-only decoder does not model. When those
720    //     values are present (a table with clustering data) the walk cannot
721    //     safely continue, so the two later fields are reported honestly as
722    //     `Unparsed` rather than mis-decoded.
723    if walk.has_improved_min_max {
724        if !try_skip_improved_min_max(&mut c)? {
725            return Ok(RepairMetadata {
726                repaired_at,
727                pending_repair: RepairField::Unparsed,
728                is_transient: RepairField::Unparsed,
729                repaired_at_decoded: true,
730            });
731        }
732    } else if walk.has_legacy_min_max {
733        skip_legacy_min_max(&mut c)?;
734    }
735
736    // 11. hasLegacyCounterShards (bool).
737    c.skip(1)?;
738
739    // 12. totalColumnsSet (long), totalRows (long).
740    c.skip(8 + 8)?;
741
742    // 13. commitLogLowerBound (CommitLogPosition) + commitLogIntervals
743    //     (IntervalSet). `hasCommitLogLowerBound` is `>= mb` and
744    //     `hasCommitLogIntervals` is `>= mc` in Cassandra; both gates are below
745    //     `na`, so every Cassandra 5.0 format CQLite reads (`nb`/`oa`/`da`) always
746    //     carries both fields. Skip them unconditionally. (The pre-`mc` absent
747    //     case is out of scope — CQLite is a Cassandra 5.0 reader.)
748    c.skip(8 + 4)?; // commitLogLowerBound CommitLogPosition: i64 segmentId + i32 position
749    skip_commit_log_intervals(&mut c)?;
750
751    // 14. pendingRepair: a presence byte (writeBoolean), then a 16-byte UUID
752    //     when present. `hasPendingRepair` is `version.compareTo("na") >= 0`
753    //     (cassandra-5.0.0 `BigFormat`/`BtiFormat`), so it is ALWAYS true for
754    //     every Cassandra 5.0 format CQLite reads (`nb`/`oa`/`da`) and the field
755    //     is always present and decoded. The `else` is unreachable for supported
756    //     formats (pre-`na` BIG `ma`–`me` is out of scope — CQLite is a Cassandra
757    //     5.0 reader); it fails closed via `Unparsed` rather than fabricating a
758    //     value, which `classify_inputs` rejects.
759    let pending_repair = if walk.has_pending_repair {
760        let present = c.read_u8()?;
761        if present != 0 {
762            RepairField::Decoded(Some(c.read_uuid()?))
763        } else {
764            RepairField::Decoded(None)
765        }
766    } else {
767        RepairField::Unparsed
768    };
769
770    // 15. isTransient (bool). Same `>= na` gate as pendingRepair, so always
771    //     present and decoded for supported Cassandra 5.0 formats. The `else` is
772    //     unreachable for `nb`/`oa`/`da`; it fails closed (`Unparsed`).
773    let is_transient = if walk.has_is_transient {
774        RepairField::Decoded(c.read_u8()? != 0)
775    } else {
776        RepairField::Unparsed
777    };
778
779    Ok(RepairMetadata {
780        repaired_at,
781        pending_repair,
782        is_transient,
783        repaired_at_decoded: true,
784    })
785}

787/// Authoritative per-SSTable row/partition counts decoded from the STATS
788/// component of `Statistics.db` (issue #944).
789///
790/// Both fields come straight from cassandra-5.0.0
791/// `StatsMetadata.StatsMetadataSerializer.serialize` — no heuristics (#28):
792///
793///   * `partition_count` is the sum of the bucket counts of the
794///     `estimatedPartitionSize` `EstimatedHistogram` (the FIRST STATS field). That
795///     histogram records one observation per partition, so Σ counts is exactly the
796///     SSTable's partition count — the same value `SSTableReader` exposes via
797///     `getEstimatedPartitionSize().count()`. It is reachable by a fully
798///     self-describing walk and needs NO version gates.
799///   * `total_rows` is the `totalRows` `long` written near the end of the STATS
800///     body (field 12). Reaching it requires the version-gated forward walk
801///     (`improvedMinMax`/`legacyMinMax` block + commit-log intervals), so it is
802///     `Some` only when authoritative [`VersionGates`] are supplied AND the walk
803///     could traverse the min/max block. For a clustered table whose covered-Slice
804///     bound values are not modeled, `total_rows` is honestly `None` rather than a
805///     guessed value.
806#[derive(Debug, Clone, Copy, PartialEq, Eq)]
807pub struct TableCounts {
808    /// Σ `estimatedPartitionSize` histogram bucket counts = partition count.
809    pub partition_count: u64,
810    /// `totalRows` from the STATS body, when the gated walk could reach it.
811    pub total_rows: Option<u64>,
812}
813
814/// Decode authoritative [`TableCounts`] from a raw `Statistics.db` buffer.
815///
816/// `partition_count` is always decoded (self-describing leading histogram).
817/// `total_rows` is decoded only when `gates` is `Some` and the version-gated walk
818/// to field 12 succeeds; otherwise it is `None` (never fabricated).
819///
820/// # Errors
821///
822/// Returns `Error::Corruption` when the STATS component is present but truncated
823/// or structurally invalid (mirrors [`parse_repair_metadata`] — fails closed). A
824/// `Statistics.db` with NO STATS component yields all-zero counts
825/// (`partition_count == 0`, `total_rows == None`).
826pub fn read_table_counts(input: &[u8], gates: Option<&VersionGates>) -> Result<TableCounts> {
827    let walk = gates.map(RepairWalkGates::from);
828
829    let Some(bounds) = stats_component_bounds(input)? else {
830        return Ok(TableCounts {
831            partition_count: 0,
832            total_rows: None,
833        });
834    };
835
836    let modern_histogram = walk.map(|w| w.modern_histogram).unwrap_or(false);
837    let mut c = Cursor::new(&input[bounds.start..bounds.end]);
838
839    // 1. estimatedPartitionSize histogram: read it (don't skip) so we can sum the
840    //    per-partition bucket counts. Σ counts = partition count (authoritative).
841    let partition_count = sum_estimated_histogram_counts(&mut c)?;
842
843    // 2. estimatedCellPerPartitionCount histogram — skip.
844    skip_estimated_histogram(&mut c)?;
845
846    // Without authoritative gates the path to totalRows (the gated min/max block)
847    // cannot be traversed; report partition_count alone and total_rows = None.
848    let Some(walk) = walk else {
849        return Ok(TableCounts {
850            partition_count,
851            total_rows: None,
852        });
853    };
854
855    // 3-8. commitLogUpperBound, timestamps, deletion times, TTLs, compressionRatio,
856    //       tombstone histogram — same fixed/gated skips as the repair walk.
857    c.skip(8 + 4)?; // commitLogUpperBound: i64 segmentId + i32 position
858    c.skip(8 + 8)?; // minTimestamp, maxTimestamp
859    c.skip(4 + 4)?; // min/maxLocalDeletionTime
860    c.skip(4 + 4)?; // minTTL, maxTTL
861    c.skip(8)?; // compressionRatio (f64)
862    skip_tombstone_histogram(&mut c, modern_histogram)?;
863
864    // 9. sstableLevel (i32), repairedAt (i64).
865    c.skip(4 + 8)?;
866
867    // 10. version-gated min/max clustering block. The improvedMinMax covered-Slice
868    //     bound values are not always modeled; when they cannot be skipped, totalRows
869    //     is unreachable and reported honestly as None.
870    if walk.has_improved_min_max {
871        if !try_skip_improved_min_max(&mut c)? {
872            return Ok(TableCounts {
873                partition_count,
874                total_rows: None,
875            });
876        }
877    } else if walk.has_legacy_min_max {
878        skip_legacy_min_max(&mut c)?;
879    }
880
881    // 11. hasLegacyCounterShards (bool).
882    c.skip(1)?;
883
884    // 12. totalColumnsSet (long), then totalRows (long) — READ totalRows.
885    c.skip(8)?; // totalColumnsSet
886    let total_rows = c.read_i64()?;
887    let total_rows = u64::try_from(total_rows).ok();
888
889    Ok(TableCounts {
890        partition_count,
891        total_rows,
892    })
893}

895/// Read an `EstimatedHistogram` and return the SUM of its bucket counts (`i64`
896/// each). Layout: `i32 bucketCount`, then `bucketCount` × (`i64 offset`,
897/// `i64 count`). Σ counts = number of observations (one per partition for the
898/// estimatedPartitionSize histogram). Saturating add: a real SSTable never
899/// overflows u64, but never wrap into a tiny count that would mislead the gate.
900fn sum_estimated_histogram_counts(c: &mut Cursor) -> Result<u64> {
901    let count = c.read_i32()?;
902    if count < 0 {
903        return Err(Error::Corruption(format!(
904            "negative EstimatedHistogram bucket count {count}"
905        )));
906    }
907    let mut sum: u64 = 0;
908    for _ in 0..count {
909        let _offset = c.read_i64()?;
910        let bucket = c.read_i64()?;
911        if bucket < 0 {
912            return Err(Error::Corruption(format!(
913                "negative EstimatedHistogram bucket value {bucket}"
914            )));
915        }
916        sum = sum.saturating_add(bucket as u64);
917    }
918    Ok(sum)
919}

921/// Cassandra's canonical "no deletion" local-deletion-time sentinel,
922/// normalized to `i64` for both legacy (nb) and modern (oa/da) encodings.
923///
924/// `LivenessInfo.NO_DELETION_TIME` is `Cell.MAX_DELETION_TIME` =
925/// `Integer.MAX_VALUE` for legacy (signed `i32`) SSTables, and `0xFFFF_FFFF`
926/// (the `u32` saturation value) for modern SSTables that store the field as an
927/// unsigned 32-bit integer. `sstablemetadata` prints `Long.MAX_VALUE`
928/// (`9223372036854775807`) for "no tombstones"; we normalize to that here so
929/// the decoded `max_local_deletion_time` matches the reference tool's printed
930/// integer in BOTH cases.
931const NO_DELETION_TIME: i64 = i64::MAX;
932
933/// Additional STATS-component facts decoded by [`parse_stats_extras`] that are
934/// not part of the repair-coordination metadata: the SSTable's maximum local
935/// deletion time and the estimated tombstone-drop-times histogram.
936///
937/// Decoded best-effort: a `Statistics.db` that parses today must keep parsing,
938/// so callers treat an `Err` from [`parse_stats_extras`] as "leave the existing
939/// placeholders" rather than propagating it.
940#[derive(Debug, Clone)]
941pub struct StatsExtras {
942    /// SSTable `maxLocalDeletionTime`, normalized so that the version-specific
943    /// "no tombstones" sentinel (`i32::MAX` for nb, `0xFFFF_FFFF` for oa/da)
944    /// maps to [`NO_DELETION_TIME`] (`i64::MAX`); any real deletion time is the
945    /// seconds-since-epoch value as `i64`.
946    pub max_local_deletion_time: i64,
947    /// `estimatedTombstoneDropTime` histogram as `(point, count)` pairs. Empty
948    /// when the histogram carries no bins (the common case for SSTables with no
949    /// tombstones).
950    pub tombstone_drop_times: Vec<(i64, u64)>,
951    /// Authoritative SSTable `maxTimestamp` (write timestamp, microseconds)
952    /// decoded from STATS field 4 (issue #1729). `None` when the SSTable
953    /// carries no live timestamp — Cassandra's `MetadataCollector` seeds the
954    /// tracker with `Long.MIN_VALUE`, so an on-disk value of [`i64::MIN`] is the
955    /// "no timestamps recorded" sentinel and is surfaced fail-closed as `None`
956    /// (consumers that require a real max — e.g. the #1388 drop gate — must NOT
957    /// proceed as if a max were known). Any other value is the true maximum.
958    ///
959    /// This intentionally replaces the historical `max_timestamp = min_timestamp`
960    /// placeholder in [`crate::parser::enhanced_statistics_parser`], which
961    /// silently reported the MIN write timestamp as the max.
962    pub max_timestamp: Option<i64>,
963    /// Authoritative SSTable `maxTTL` (seconds) decoded from STATS field 9
964    /// (issue #1537). Cassandra's `MetadataCollector` seeds the TTL tracker with
965    /// `Cell.NO_TTL` (`0`) and updates it only from EXPIRING cells, so a maxTTL of
966    /// `0` means "no expiring cells" and is surfaced fail-closed as `None`. Any
967    /// positive value is the true maximum TTL across the SSTable's expiring cells.
968    /// The compaction TTL-expiry LDT floor (issue #1537,
969    /// [`crate::storage::write_engine::merge::compute_expiry_ttl_ldt_floor`]) needs
970    /// this authoritative maximum: the enhanced (nb) parser only decodes the
971    /// EncodingStats `minTTL` baseline and honestly leaves `max_ttl = None`, so
972    /// this best-effort STATS-extras walk is the only source of the true maximum.
973    pub max_ttl: Option<i64>,
974}

990/// Decode the SSTable `maxLocalDeletionTime` and the estimated
991/// tombstone-drop-times histogram from a raw `Statistics.db` buffer.
992///
993/// This walks the same self-describing leading STATS fields as
994/// [`parse_repair_metadata`], but reads (rather than skips) field 5
995/// (min/maxLocalDeletionTime) and field 8 (the tombstone histogram).
996///
997/// When `gates` is `None`, the legacy (nb) encoding is assumed for BOTH the
998/// max-LDT sentinel comparison and the histogram entry width (nb-compatible
999/// default, matching the rest of the minimal Statistics parser and the
1000/// `merge.rs` caller).
1001///
1002/// # Returns
1003///
1004/// * `Ok(StatsExtras::default())` — max-LDT = [`NO_DELETION_TIME`], empty
1005///   histogram — when the buffer carries no STATS component (nothing to decode).
1006/// * `Ok(extras)` with the decoded facts otherwise.
1007///
1008/// # Errors
1009///
1010/// Returns an error when the STATS component is present but its leading
1011/// (self-describing) fields are truncated/corrupt or overrun the component's
1012/// authoritative end bound. Best-effort callers must treat this as "leave the
1013/// existing placeholders" and MUST NOT propagate it (a `Statistics.db` that
1014/// parses today must keep parsing).
1015pub fn parse_stats_extras(input: &[u8], gates: Option<&VersionGates>) -> Result<StatsExtras> {
1016    let Some(bounds) = stats_component_bounds(input)? else {
1017        // No STATS component → nothing to decode. Report the canonical
1018        // "no tombstones" default rather than failing.
1019        return Ok(StatsExtras::default());
1020    };
1021
1022    let modern = gates.map(uses_modern_tombstone_histogram).unwrap_or(false);
1023
1024    // Bound the cursor over ONLY the STATS component slice (start..end, CRC and
1025    // following components excluded). Any read past this slice fails closed.
1026    let mut c = Cursor::new(&input[bounds.start..bounds.end]);
1027
1028    // 1-2. estimatedPartitionSize + estimatedCellPerPartitionCount.
1029    skip_estimated_histogram(&mut c)?;
1030    skip_estimated_histogram(&mut c)?;
1031
1032    // 3. commitLogUpperBound: i64 segmentId + i32 position.
1033    c.skip(8 + 4)?;
1034
1035    // 4. minTimestamp, maxTimestamp (both i64 BE microseconds). Read (rather
1036    //    than skip) maxTimestamp: it is the authoritative SSTable max write
1037    //    timestamp (issue #1729). Both encodings (nb / oa / da) lay these out
1038    //    identically here, so no version gating is needed for this field.
1039    c.skip(8)?; // minTimestamp
1040    let raw_max_timestamp = c.read_i64()?;
1041    let max_timestamp = decode_max_timestamp(raw_max_timestamp);
1042
1043    // 5. min/maxLocalDeletionTime. Width is 4 bytes each in both encodings; the
1044    //    signedness/sentinel differs by version (handled below).
1045    let _min_ldt = c.read_u32()?;
1046    let raw_max_ldt = c.read_u32()?;
1047    let max_local_deletion_time = decode_max_local_deletion_time(raw_max_ldt, modern);
1048
1049    // 6. minTTL (skip), maxTTL (read). Both are `int` (4 bytes BE) in every
1050    //    encoding (nb / oa / da). maxTTL is the authoritative maximum TTL across
1051    //    the SSTable's expiring cells (issue #1537); `0` (Cassandra `Cell.NO_TTL`,
1052    //    the tracker seed) means "no expiring cells" and is surfaced as `None`.
1053    c.skip(4)?; // minTTL
1054    let raw_max_ttl = c.read_i32()?;
1055    let max_ttl = if raw_max_ttl > 0 {
1056        Some(i64::from(raw_max_ttl))
1057    } else {
1058        None
1059    };
1060
1061    // 7. compressionRatio (f64).
1062    c.read_f64()?;
1063
1064    // 8. estimatedTombstoneDropTime (TombstoneHistogram).
1065    let tombstone_drop_times = read_tombstone_histogram(&mut c, modern)?;
1066
1067    Ok(StatsExtras {
1068        max_local_deletion_time,
1069        tombstone_drop_times,
1070        max_timestamp,
1071        max_ttl,
1072    })
1073}

1075/// Normalize a raw 4-byte `maxLocalDeletionTime` to a canonical `i64`.
1076///
1077/// * modern (oa/da): the field is an unsigned `u32`; the sentinel is
1078///   `0xFFFF_FFFF`.
1079/// * legacy (nb): the field is a signed `i32`; the sentinel is `i32::MAX`.
1080///
1081/// The version sentinel maps to [`NO_DELETION_TIME`] (`i64::MAX`); any real
1082/// deletion time (always `< 2^31` seconds-since-epoch) is returned as its
1083/// integer value.
1084/// Cassandra's `MetadataCollector` seeds its timestamp `MinMaxLongTracker` with
1085/// `Long.MIN_VALUE` for the max; an SSTable that recorded no live write
1086/// timestamp serializes that sentinel verbatim to STATS field 4. We treat it as
1087/// "no authoritative maxTimestamp" rather than a real (nonsensical) maximum.
1088pub(crate) const NO_MAX_TIMESTAMP_SENTINEL: i64 = i64::MIN;
1089
1090/// Normalize a raw STATS `maxTimestamp` (i64 BE microseconds) into an
1091/// authoritative value, or `None` when the SSTable recorded no write timestamp.
1092///
1093/// The `Long.MIN_VALUE` sentinel maps to `None` (fail-closed: consumers that
1094/// require a real max must not proceed as if one were known). Every other value
1095/// — including a valid large or negative real timestamp — is returned verbatim.
1096///
1097/// Shared by both parsers (enhanced nb walk here, and the legacy fixed-width
1098/// `parse_timestamp_statistics`, issue #1653) so the sentinel→`None` mapping
1099/// never drifts between paths.
1100pub(crate) fn decode_max_timestamp(raw: i64) -> Option<i64> {
1101    if raw == NO_MAX_TIMESTAMP_SENTINEL {
1102        None
1103    } else {
1104        Some(raw)
1105    }
1106}

1109    if modern {
1110        if raw == u32::MAX {
1111            NO_DELETION_TIME
1112        } else {
1113            raw as i64
1114        }
1115    } else {

1117        if signed == i32::MAX {
1118            NO_DELETION_TIME
1119        } else {
1120            signed as i64
1121        }
1122    }

1125/// Read a `TombstoneHistogram` as `(point, count)` pairs.
1126///
1127/// Header: `i32 maxBinSize`, `i32 size` (bin count, must be `>= 0`). Each bin is
1128/// `f64 point + i64 value` (16 bytes) for legacy (nb), or `i64 point + i32 value`
1129/// (12 bytes) for modern (oa/da). The legacy `f64` point is cast to `i64`
1130/// (Cassandra rounds drop-times to integer bucket boundaries, so the cast is
1131/// exact). An empty histogram yields an empty `Vec`.
1132fn read_tombstone_histogram(c: &mut Cursor, modern: bool) -> Result<Vec<(i64, u64)>> {
1133    let _max_bin_size = c.read_i32()?;
1134    let size = c.read_i32()?;
1135    if size < 0 {
1136        return Err(Error::Corruption(format!(
1137            "negative TombstoneHistogram size {size}"
1138        )));
1139    }
1140    // Bound the allocation by the bytes actually present BEFORE reserving:
1141    // a corrupt Statistics.db can advertise a huge positive `size`, and
1142    // `Vec::with_capacity(size)` would otherwise attempt a massive (or aborting)
1143    // allocation. Each bin is `entry_width` bytes; validate that all bins fit in
1144    // the remaining cursor span (fail-closed via `need`) so the reserve below is
1145    // capped by real data, not the untrusted header.
1146    let entry_width = if modern { 12usize } else { 16usize };
1147    let required = (size as usize)
1148        .checked_mul(entry_width)
1149        .ok_or_else(|| Error::Corruption("TombstoneHistogram size overflow".to_string()))?;
1150    c.need(required)?;
1151    let mut out = Vec::with_capacity(size as usize);
1152    for _ in 0..size {
1153        if modern {
1154            let point = c.read_i64()?;
1155            let value = c.read_i32()?;
1156            out.push((point, value as u64));
1157        } else {
1158            let point = c.read_f64()?;
1159            let value = c.read_i64()?;
1160            out.push((point as i64, value as u64));
1161        }
1162    }
1163    Ok(out)
1164}
