1//! SSTable compaction execution: K-way merge start + atomic finalize.
2//!
3//! Extracted verbatim from `write_engine/maintenance.rs` (issue #1120, epic
4//! #1116) as a behavior-preserving split. Owns the merge state-machine entry
5//! points driven by `maintenance_step`: `start_merge` (opens the K-way merger +
6//! tmp output writer) and the blocking/async finalizers that atomically publish
7//! the merged SSTable and reclaim the inputs. `WriteEngine`'s fields are
8//! reachable here because this is a sibling module in the same crate.
9

11use super::{merge, KWayMerger, WriteEngine};
12use crate::error::{Error, Result};
13use std::path::PathBuf;
14use std::time::Instant;

16// Test-only crash-injection seam (issue #1393).
17//
18// When set on the thread that drives `finalize_merge_async`, the finalizer
19// aborts *after* the tmp component files have been written but *before* the
20// atomic publication rename — and deliberately leaves the `.compaction-tmp-*`
21// directory on disk, exactly as a real process crash mid-compaction would. The
22// startup orphan sweep is then exercised end-to-end against genuine partial
23// output. It is a `thread_local` (not a global) so parallel tests cannot see
24// each other's injection: the sweep e2e test runs `maintenance_step`
25// synchronously with no ambient runtime, so `block_on_async` polls the
26// finalizer on the very thread that set the flag.
27#[cfg(test)]
28thread_local! {
29    pub(crate) static FAIL_COMPACTION_BEFORE_RENAME: std::cell::Cell<bool> =
30        const { std::cell::Cell::new(false) };
31}

33// Test-only publication-barrier fsync fault seams (issue #1959).
34//
35// These let a test force the directory durability `fsync` to fail at each of
36// the two publication points inside `finalize_merge_async`, exercising the
37// rollback branches that a real EIO/ENOSPC on the directory handle would hit:
38//
39//   * `FAIL_COMPACTION_DIR_FSYNC_BEFORE_TOC` — fault the fsync performed AFTER
40//     the non-TOC component renames but BEFORE the TOC rename (step 2b). A
41//     failure here must roll back the already-renamed components and abort with
42//     `Error::Storage`, so no partially-published set is left behind and the
43//     inputs stay intact.
44//   * `FAIL_COMPACTION_DIR_FSYNC_AFTER_TOC` — fault the fsync performed AFTER
45//     the TOC rename but BEFORE the inputs are deleted (step 2c). A failure here
46//     must roll back the whole published set (INCLUDING the just-renamed TOC) so
47//     no visible-but-not-durable TOC survives, and the inputs — still the only
48//     durable copy — are never deleted.
49//
50// Like `FAIL_COMPACTION_BEFORE_RENAME` these are `thread_local` so parallel
51// tests cannot see each other's injection; the driving test polls the finalizer
52// synchronously on the same thread that set the flag.
53#[cfg(test)]
54thread_local! {
55    pub(crate) static FAIL_COMPACTION_DIR_FSYNC_BEFORE_TOC: std::cell::Cell<bool> =
56        const { std::cell::Cell::new(false) };
57    pub(crate) static FAIL_COMPACTION_DIR_FSYNC_AFTER_TOC: std::cell::Cell<bool> =
58        const { std::cell::Cell::new(false) };
59}

61impl WriteEngine {
62    /// Start a new merge operation (M5.2 helper, Issue #474)
63    ///
64    /// ## Atomicity design
65    ///
66    /// The SSTableWriter is pointed at a temporary directory (`tmp_dir`) that lives
67    /// alongside the final SSTable directory. After `writer.finish()` succeeds, each
68    /// output component is atomically renamed into the final directory. Input files are
69    /// deleted only after all renames complete. This guarantees:
70    ///
71    /// - A crash before renames: tmp files are incomplete; inputs intact.
72    /// - A crash mid-rename: at worst a partial output component exists in the final
73    ///   directory, but the old inputs are still there and the TOC.txt (publication
74    ///   barrier) has not been renamed yet, so the partial output is never discovered
75    ///   by readers scanning for `TOC.txt`.
76    /// - A crash after all renames but before input deletion: a harmless duplicate
77    ///   exists until next compaction.
78    ///
79    /// `purge_safe` (#921 finding 1): `true` only when `input_paths` spans every
80    /// SSTable for the table (a major/full compaction), allowing gc_grace
81    /// tombstone purging without risking resurrection of data shadowed in a
82    /// non-included overlapping SSTable. `false` keeps every tombstone.
83    ///
84    /// `max_purgeable_timestamp` (#935): for a PARTIAL compaction (`purge_safe ==
85    /// false`), the min write timestamp across the non-included overlapping
86    /// SSTables — see [`merge::compute_max_purgeable_timestamp`]. When `Some`, a
87    /// tombstone older than every outside SSTable is purged even in a partial
88    /// compaction; `None` keeps the conservative #921 behavior (no purging).
89    #[tracing::instrument(name = "compaction.start_merge", level = "debug", skip(self, input_paths, max_purgeable_timestamp, outside_paths), fields(inputs = input_paths.len()))]
90    pub(crate) fn start_merge(
91        &mut self,
92        input_paths: Vec<PathBuf>,
93        purge_safe: bool,
94        max_purgeable_timestamp: Option<i64>,
95        outside_paths: Vec<PathBuf>,
96    ) -> Result<()> {
97        tracing::info!(
98            "Starting compaction merge of {} SSTables",
99            input_paths.len()
100        );
101
102        // gc_grace / gcBefore cutoff (issue #845): the fully-expired drop-set
103        // (issue #1388) needs the same cutoff the merger's purge stage uses, so
104        // compute it up front. `now_secs` is reused below for the merger.
105        let now_secs = std::time::SystemTime::now()
106            .duration_since(std::time::UNIX_EPOCH)
107            .map(|d| d.as_secs() as i64)
108            .unwrap_or(0);
109        let gc_before_secs = merge::compute_gc_before(&self.config.schema, now_secs);
110
111        // Fully-expired SSTable drop (issue #1388): compute the subset of the
112        // selected inputs that are provably fully expired (authoritative
113        // `Statistics.db` `max_deletion_time < gcBefore`, no cell scan) AND
114        // overlap-safe against the outside set (`max_timestamp` below the outside
115        // min write timestamp — the same #935 coarse global bound). For a full
116        // compaction `outside_paths` is empty ⇒ +inf bound ⇒ every fully-expired
117        // input is droppable. These SSTables are EXCLUDED from the merger's input
118        // list below (never read/decoded — the perf win) and reclaimed after the
119        // merged output publishes.
120        let drop_set = merge::fully_expired_sstables(&input_paths, &outside_paths, gc_before_secs);
121        // Subtract the drop-set from the merger inputs (with the all-dropped guard);
122        // shared with the CLI one-shot path.
123        let (input_paths, dropped_whole) = merge::split_merge_and_dropped(&input_paths, drop_set);
124
125        // Measure total bytes read (sum of Data.db file sizes as an approximation)
126        let bytes_read: u64 = input_paths
127            .iter()
128            .map(|p| std::fs::metadata(p).map(|m| m.len()).unwrap_or(0))
129            .sum();
130
131        let output_generation = self.generation;
132
133        // Final SSTable directory: data_dir/keyspace/table/
134        let sstable_dir = self
135            .config
136            .data_dir
137            .join(&self.config.schema.keyspace)
138            .join(&self.config.schema.table);
139
140        // Temporary root: data_dir/.compaction-tmp-{gen}/
141        //
142        // Placing the tmp root inside `data_dir` (not inside `sstable_dir`) keeps
143        // the path simple and guarantees the rename is within the same filesystem.
144        // The SSTableWriter appends `keyspace/table/` internally, so component files
145        // land at: data_dir/.compaction-tmp-{gen}/keyspace/table/nb-{gen}-big-*.db
146        let tmp_dir = self
147            .config
148            .data_dir
149            .join(format!(".compaction-tmp-{}", output_generation));
150
151        // Create the tmp root (SSTableWriter::finish will create the subdirs)
152        std::fs::create_dir_all(&tmp_dir).map_err(|e| {
153            Error::Storage(format!(
154                "Failed to create compaction tmp directory {:?}: {}",
155                tmp_dir, e
156            ))
157        })?;
158
159        // #850: derive the effective compaction schema by reading static-row
160        // presence from the input SSTable headers. If a static column was dropped
161        // from the current schema but an input still carries it, re-add it so the
162        // merger decodes the static cells and the writer emits the static prelude
163        // (header-driven static presence). Byte-identical to `config.schema` when
164        // no such column exists.
165        let mut effective_schema =
166            merge::effective_compaction_schema(&self.config.schema, &input_paths);
167        // #929: normalize bare-name UDT columns to their UserType(...) marshal on
168        // the EFFECTIVE (decode) schema by copying each column's exact UserType
169        // marshal from the input headers. The compaction reader decides complex-
170        // vs-simple from this schema's `data_type` (is_complex_column), so a bare
171        // name would make it misdecode/drop pre-existing per-field UDT cells
172        // (roborev #1009/#1013/#1015/#1017/#1019/#1021). Derived ENTIRELY from the
173        // input headers, so it runs unconditionally (a registry is needed only at
174        // flush time); errors on a mixed encoding rather than corrupting.
175        // `write_schema` derives from this via `for_compaction_output`. The
176        // configured registry handles a UDT column added after the inputs were
177        // written (absent from every input header) so the output is not emitted
178        // as a bare/BytesType column (roborev #1023).
179        merge::apply_udt_marshals_from_inputs(
180            &mut effective_schema,
181            &input_paths,
182            self.config.udt_registry.as_ref(),
183        )?;
184
185        // Create K-way merger.
186        //
187        // gc_grace / gcBefore purging (issue #845): `gcBefore = now -
188        // gc_grace_seconds` and `now_secs` were computed above (also used by the
189        // #1388 fully-expired drop-set). `compute_gc_before` returns `None` when the
190        // table declares no valid `gc_grace_seconds`, disabling purging (and, above,
191        // dropping) — preserving the pre-#845 behavior. `now_secs` is the wall-clock
192        // used for both the purge boundary and TTL evaluation.
193        //
194        // Overlap-safety gate (#921 finding 1): only a major/full compaction
195        // (one that spans every candidate SSTable for the table) may purge
196        // tombstones. A partial compaction sets `purge_safe = false`, which
197        // makes the gc_grace purge stage a strict no-op and cannot resurrect
198        // data shadowed in a non-included overlapping SSTable.
199
200        // #921 finding 2: mirror `compact_sstables`' dropped-column survivor
201        // pre-pass. Decode with `effective_schema` (retains dropped columns so
202        // their cells parse and can be purged), but WRITE with a post-drop
203        // schema: a background compaction that purges the LAST cell/tombstone of
204        // a dropped column must NOT emit that column in the output
205        // SerializationHeader (stale header columns misalign a post-drop reader).
206        // The pre-pass uses the SAME gc cutoff AND `purge_safe` as the write
207        // merger below so the two make IDENTICAL purge decisions — a tombstone
208        // purged in the write pass is also purged here and never counted as a
209        // surviving cell. Reuses the exact helper `compact_sstables` uses.
210        let retained_dropped = if effective_schema.dropped_columns.is_empty() {
211            std::collections::HashSet::new()
212        } else {
213            merge::compute_surviving_dropped_columns(
214                input_paths.clone(),
215                &effective_schema,
216                gc_before_secs,
217                Some(now_secs),
218                purge_safe,
219                max_purgeable_timestamp,
220            )?
221        };
222        // `write_schema` inherits the #929 UDT normalization from the already-
223        // normalized `effective_schema` (for_compaction_output clones columns).
224        let write_schema = effective_schema.for_compaction_output(&retained_dropped);
225
226        // Issue #1234: thread the configured UDT registry onto the merge readers so
227        // a top-level `frozen<UDT>` value decodes structurally during background
228        // compaction instead of erroring out and dropping the partition.
229        let merger = KWayMerger::new_with_gc_and_registry(
230            input_paths.clone(),
231            &effective_schema,
232            gc_before_secs,
233            Some(now_secs),
234            self.config.udt_registry.clone(),
235        )?
236        .with_purge_safe(purge_safe)
237        // #935: overlap-aware purging for a partial compaction. `purge_safe`
238        // overrides this in the merger (full compaction → +inf bound), so passing
239        // the partial bound unconditionally is safe.
240        .with_max_purgeable_timestamp(max_purgeable_timestamp);
241
242        // Repair-state preservation + mixed-state rejection (issue #1021).
243        //
244        // Cassandra partitions compaction candidates by repair state and never
245        // mixes repaired / unrepaired / pending-repair SSTables in one
246        // compaction. CQLite cannot reproduce the repair-boundary tombstone
247        // constraints, so it MUST NOT silently merge across that boundary: read
248        // each input's persisted repair state from its Statistics.db, reject a
249        // mixed set with a typed error, and PRESERVE the single shared state into
250        // the merged output below. Authoritative metadata only (no heuristics).
251        let repair_state = merge::classify_inputs(&input_paths)?;
252
253        // Point the SSTableWriter at the tmp root; it will write to
254        // tmp_dir/keyspace/table/nb-{gen}-big-*.db. Use the post-drop
255        // `write_schema` (NOT `effective_schema`) so fully-purged dropped
256        // columns are stripped from the output header (#921 finding 2).
257        let mut writer = crate::storage::sstable::writer::SSTableWriter::new(
258            tmp_dir.clone(),
259            output_generation,
260            &write_schema,
261        )?;
262        // Compaction output (issue #1222): emit the uncompressed-BIG CRC.db with
263        // Cassandra's compaction-only trailing empty-final-chunk CRC32 = 0. The
264        // flush path leaves this unset, keeping its CRC.db byte-identical.
265        writer.mark_compaction_output();
266
267        // Carry the inputs' shared repair state into the merged output's
268        // Statistics.db (issue #1021). For a normal unrepaired corpus this is the
269        // unrepaired default (a no-op vs the previous hardcoded zeros).
270        writer.set_repair_state(
271            repair_state.repaired_at,
272            repair_state.pending_repair,
273            repair_state.is_transient,
274        );
275
276        // Two-pass compaction (issue #729): compute output FINAL encoding baselines
277        // by reading the minimum values from each input SSTable's Statistics.db.
278        // The output baseline must be ≤ all per-partition values from all inputs.
279        let (baseline_min_ts, mut baseline_min_ldt, baseline_min_ttl) =
280            merge::compute_baseline_min(&input_paths);
281        // Issue #1537: unlike `compact_sstables_with_registry` — whose
282        // `now_secs: Option<i64>` parameter can be `None` (expiry disabled), so it
283        // gates the floor application on `now_secs.is_some()` — this WriteEngine
284        // path computes `now_secs` locally above as a plain `i64` (unwrap_or(0))
285        // and ALWAYS builds the merger with the literal `Some(now_secs)` (see the
286        // `KWayMerger::new_with_gc_and_registry` construction above). Expiry is
287        // therefore unconditionally active here, so no `is_some()` gate is needed
288        // (and `now_secs` is not an `Option`, so a literal mirror would not even
289        // compile). An expired expiring cell converts to a creation-time
290        // (`ldt - ttl`) tombstone whose LDT sits BELOW the input-derived baseline,
291        // so lower the LDT baseline to a provably-safe floor so the DataWriter's
292        // unsigned LDT-delta never underflows. No-op when no input carries
293        // expiring cells.
294        if let Some(floor) = merge::compute_expiry_ttl_ldt_floor(&input_paths) {
295            baseline_min_ldt = baseline_min_ldt.min(floor);
296        }
297        writer.pre_seed_encoding_baselines(baseline_min_ts, baseline_min_ldt, baseline_min_ttl);
298
299        // Increment generation for next operation
300        self.generation += 1;
301
302        self.active_merge = Some(ActiveMerge {
303            merger,
304            writer,
305            input_paths,
306            tmp_dir,
307            sstable_dir,
308            rows_merged: 0,
309            bytes_read,
310            started_at: Instant::now(),
311            effective_schema,
312            dropped_whole,
313            // Stage 4 (#1668): no partition is mid-drain when a merge starts.
314            pending_partition: None,
315        });
316
317        Ok(())
318    }

320    /// Finalize the active merge - blocking version (M5.2 helper).
321    ///
322    /// Bridges to the async finalizer via [`merge::block_on_async`], which is
323    /// safe to call from within an active Tokio runtime (e.g. the CLI's
324    /// `#[tokio::main]` worker threads). A nested `Handle::block_on` here would
325    /// otherwise panic with "Cannot start a runtime from within a runtime"
326    /// (Issue #587).
327    pub(crate) fn finalize_merge_blocking(&mut self, report: &mut MaintenanceReport) -> Result<()> {
328        merge::block_on_async(self.finalize_merge_async(report))
329    }
330
331    /// Finalize the active merge - async version (M5.2 helper, Issue #474)
332    ///
333    /// ## Atomicity protocol
334    ///
335    /// 1. `writer.finish()` flushes all component files to the tmp directory.
336    /// 2. Each component file is renamed from `tmp_dir/` to `sstable_dir/`.
337    ///    The TOC.txt rename is performed **last** (it is the publication barrier:
338    ///    readers discover SSTables by scanning for `TOC.txt`).
339    /// 3. Only after all renames succeed are the input SSTable files deleted.
340    /// 4. The now-empty tmp directory is removed.
341    ///
342    /// If any step 2 rename fails, the partially-renamed output components are
343    /// cleaned up and an error is returned. The input SSTables remain intact.
344    ///
345    /// Single-boundary error recording (issue #1037): this is an *unrecorded*
346    /// inner helper. Any error returned here escapes through
347    /// `maintenance_step_inner` to the public `maintenance_step`, which wraps the
348    /// whole step in `record_result("compaction", ..)` and counts it exactly
349    /// once. Recording here too would double-count finalize failures.
350    #[tracing::instrument(name = "compaction.finalize", level = "debug", skip(self, report))]
351    async fn finalize_merge_async(&mut self, report: &mut MaintenanceReport) -> Result<()> {
352        let merge = match self.active_merge.take() {
353            Some(m) => m,
354            None => return Ok(()),
355        };
356
357        let input_count = merge.input_paths.len() as u64;
358        let merge_rows = merge.rows_merged;
359        let elapsed = merge.started_at.elapsed();
360        tracing::info!(
361            "Finalizing compaction merge: {} rows, {:?} elapsed",
362            merge.rows_merged,
363            elapsed
364        );
365
366        // Step 1: Finish writing all components to the tmp directory.
367        // If this fails the tmp directory may contain partial files, but inputs are safe.
368        let tmp_info = match merge.writer.finish().await {
369            Ok(info) => info,
370            Err(e) => {
371                // Clean up tmp directory on failure (best effort)
372                let _ = std::fs::remove_dir_all(&merge.tmp_dir);
373                return Err(Error::Storage(format!(
374                    "Compaction merge write failed (inputs intact): {}",
375                    e
376                )));
377            }
378        };
379
380        tracing::info!(
381            "Compaction tmp output: {} bytes, {} partitions",
382            tmp_info.data_size,
383            tmp_info.partition_count
384        );
385
386        // Test-only crash injection (issue #1393): simulate a process crash after
387        // the tmp component files are on disk but before the publication rename.
388        // We intentionally do NOT clean up `merge.tmp_dir` — a real crash leaves
389        // it for the startup orphan sweep to reclaim. Inputs are still intact
390        // (no rename or delete has run yet).
391        #[cfg(test)]
392        if FAIL_COMPACTION_BEFORE_RENAME.with(|f| f.get()) {
393            return Err(Error::Storage(
394                "injected compaction crash before publication rename (test seam, issue #1393)"
395                    .to_string(),
396            ));
397        }
398
399        // Step 2: Atomically rename each component from the tmp sub-directory to
400        // the final SSTable directory. Because both directories are under the same
401        // `data_dir`, the rename is within the same filesystem (POSIX atomic).
402        // We rename TOC.txt last because it is the publication barrier.
403        let sstable_dir = &merge.sstable_dir;
404        // The configured data root — the top of the leaf→data-root ancestor
405        // chain fsynced at publication (issue #1959, symmetry with the flush
406        // barrier). Cloned so it does not alias the `&mut self` borrows below.
407        let data_root = self.config.data_dir.clone();
408
409        // Ensure the final SSTable directory exists (it normally does, but handle
410        // the edge case where it was created by start_merge only in the tmp path).
411        // This defensive `create_dir_all` can materialize `sstable_dir` (and even
412        // an ancestor) whose PARENT dirent is not otherwise fsynced by this path;
413        // the leaf→data-root chain fsync at publication (steps 2b/2c below) makes
414        // every ancestor dirent durable, so the newly created directory cannot be
415        // lost on a crash after the inputs are deleted (issue #1959).
416        std::fs::create_dir_all(sstable_dir).map_err(|e| {
417            Error::Storage(format!(
418                "Failed to create SSTable directory {:?}: {}",
419                sstable_dir, e
420            ))
421        })?;
422
423        // Helper: map a tmp component path to its final destination.
424        // tmp_info paths look like: data_dir/.compaction-tmp-N/keyspace/table/nb-N-big-X.db
425        // Final destination:        data_dir/keyspace/table/nb-N-big-X.db
426        let make_rename = |src: &PathBuf| -> Result<(PathBuf, PathBuf)> {
427            let filename = src
428                .file_name()
429                .ok_or_else(|| Error::Storage("Component path has no filename".to_string()))?;
430            let dst = sstable_dir.join(filename);
431            Ok((src.clone(), dst))
432        };
433
434        // Build list of (src, dst) renames. TOC.txt goes last (publication barrier).
435        let mut renames: Vec<(PathBuf, PathBuf)> = Vec::new();
436
437        // Non-TOC components first. Index.db/Summary.db are BIG-only (BTI writers
438        // report them as None — issue #908); only rename them when present.
439        // Filter.db is optional too: a table with a disabled bloom filter
440        // (bloom_filter_fp_chance = 1.0, AlwaysPresentFilter) emits NO Filter.db,
441        // so there is nothing to rename. Including a non-existent path here would
442        // fail the atomic publish (Issue #852).
443        let mut non_toc: Vec<&PathBuf> = vec![&tmp_info.data_path];
444        if let Some(ref p) = tmp_info.index_path {
445            non_toc.push(p);
446        }
447        if let Some(ref p) = tmp_info.filter_path {
448            non_toc.push(p);
449        }
450        if let Some(ref p) = tmp_info.summary_path {
451            non_toc.push(p);
452        }
453        non_toc.push(&tmp_info.stats_path);
454        non_toc.push(&tmp_info.digest_path);
455        // CRC.db is emitted for uncompressed BIG output (issue #1197); rename it
456        // when present so the published SSTable keeps its per-chunk CRC file.
457        if let Some(ref p) = tmp_info.crc_path {
458            non_toc.push(p);
459        }
460        for src in non_toc {
461            renames.push(make_rename(src)?);
462        }
463        if let Some(ref p) = tmp_info.partitions_path {
464            renames.push(make_rename(p)?);
465        }
466        if let Some(ref ci_path) = tmp_info.compression_info_path {
467            renames.push(make_rename(ci_path)?);
468        }
469        // TOC.txt is the publication barrier: rename it LAST and — per the
470        // crash-safety invariant — only AFTER the non-TOC component dirents are
471        // durable (issue #1959). Keep it OUT of the `renames` list so the
472        // intervening directory fsync can be sequenced between the components and
473        // the TOC.
474        let toc_rename = make_rename(&tmp_info.toc_path)?;
475
476        // Perform the renames. On failure, remove any already-renamed files so
477        // we don't leave a half-published SSTable, then return the error.
478        // Time the rename/publication-barrier phase (issue #1037).
479        let finalize_start = Instant::now();
480        let mut renamed: Vec<PathBuf> = Vec::with_capacity(renames.len() + 1);
481
482        // Best-effort rollback: undo the renames done so far and drop the tmp dir.
483        let rollback = |renamed: &[PathBuf], tmp_dir: &std::path::Path| {
484            for dst in renamed {
485                let _ = std::fs::remove_file(dst);
486            }
487            let _ = std::fs::remove_dir_all(tmp_dir);
488        };
489
490        // Fsync the full leaf→data-root ancestor chain of the published SSTable
491        // directory (issue #1959). This mirrors the flush durability barrier
492        // (`durability::finalize_flush_durability`), which fsyncs the same chain
493        // via `dirs_to_sync`: not just the leaf `sstable_dir` (its component
494        // dirents) but every ancestor up to the data root, so the leaf's own
495        // entry in its parent — potentially just created by the defensive
496        // `create_dir_all` above — is durable too. Each `sync_directory` is a
497        // no-op on non-Unix (the Unix-only invariant, matching the flush path).
498        let fsync_publish_chain = || -> Result<()> {
499            for dir in super::durability::dirs_to_sync(sstable_dir, &data_root) {
500                super::durability::sync_directory(&dir)?;
501            }
502            Ok(())
503        };
504
505        // Step 2a: rename every NON-TOC component into the final directory.
506        for (src, dst) in &renames {
507            if let Err(e) = std::fs::rename(src, dst) {
508                let err = Error::Storage(format!(
509                    "Atomic rename of {:?} to {:?} failed (rolling back, inputs intact): {}",
510                    src, dst, e
511                ));
512                rollback(&renamed, &merge.tmp_dir);
513                return Err(err);
514            }
515            tracing::debug!(
516                "Renamed {:?} → {:?}",
517                src.file_name().unwrap_or_default(),
518                dst.file_name().unwrap_or_default()
519            );
520            renamed.push(dst.clone());
521        }
522
523        // Step 2b: directory durability barrier BEFORE publishing the TOC
524        // (issue #1959). Component *contents* were fsynced in `writer.finish()`,
525        // but the renames only altered dirents; fsync the destination directory
526        // (and its ancestors) so every non-TOC component's dirent is durable
527        // before the TOC — the barrier readers scan for — is renamed in.
528        // Sequencing the fsync here means the TOC dirent can only become durable
529        // AFTER the components it names, so a crash can never expose a visible
530        // TOC that references a not-yet-durable component. NOTE: this invariant
531        // holds on Unix only — `sync_directory` is a no-op on non-Unix, where the
532        // per-file fsyncs are the durability guarantee (issue #1392 / #1959).
533        #[cfg(test)]
534        if FAIL_COMPACTION_DIR_FSYNC_BEFORE_TOC.with(|f| f.get()) {
535            rollback(&renamed, &merge.tmp_dir);
536            return Err(Error::Storage(
537                "injected directory fsync fault before TOC rename (test seam, issue #1959)"
538                    .to_string(),
539            ));
540        }
541        if let Err(e) = fsync_publish_chain() {
542            rollback(&renamed, &merge.tmp_dir);
543            return Err(e);
544        }
545
546        // Step 2c: publish by renaming TOC.txt last, then fsync the LEAF
547        // directory AGAIN so the TOC dirent itself is durable BEFORE the input
548        // SSTables (the only other durable copy of this data) are deleted below.
549        // Without this second fsync, a crash after the inputs are removed but
550        // before the TOC dirent reached the device would lose the SSTable
551        // entirely (issue #1959). Same Unix-only scope as step 2b: a no-op on
552        // non-Unix, where the per-file fsyncs are the durability guarantee.
553        //
554        // ASYMMETRY vs step 2b (issue #1959): 2b fsyncs the FULL leaf→data-root
555        // chain because the non-TOC renames + the defensive `create_dir_all` can
556        // touch ancestor dirents that must be durable before publication; between
557        // 2b and 2c the ONLY dirent that changed is the TOC entry in the LEAF
558        // `sstable_dir` (the TOC rename). The ancestor dirents were already made
559        // durable by 2b, so re-fsyncing the whole chain here is redundant I/O —
560        // fsyncing only the leaf is sufficient to make the TOC dirent durable.
561        {
562            let (src, dst) = &toc_rename;
563            if let Err(e) = std::fs::rename(src, dst) {
564                let err = Error::Storage(format!(
565                    "Atomic rename of {:?} to {:?} failed (rolling back, inputs intact): {}",
566                    src, dst, e
567                ));
568                rollback(&renamed, &merge.tmp_dir);
569                return Err(err);
570            }
571            renamed.push(dst.clone());
572        }
573        #[cfg(test)]
574        if FAIL_COMPACTION_DIR_FSYNC_AFTER_TOC.with(|f| f.get()) {
575            // Roll back the WHOLE published set, including the just-renamed TOC,
576            // so no visible-but-not-durable TOC survives; the inputs are never
577            // deleted (they remain the sole durable copy).
578            rollback(&renamed, &merge.tmp_dir);
579            return Err(Error::Storage(
580                "injected directory fsync fault after TOC rename (test seam, issue #1959)"
581                    .to_string(),
582            ));
583        }
584        // Leaf-only (see the step 2c asymmetry note above): only the TOC dirent in
585        // `sstable_dir` changed since the full-chain fsync at 2b.
586        if let Err(e) = super::durability::sync_directory(sstable_dir) {
587            rollback(&renamed, &merge.tmp_dir);
588            return Err(e);
589        }
590
591        // Finalize/rename latency in seconds (issue #1037): the atomic
592        // publication barrier just completed successfully.
593        crate::observability::record_histogram(
594            crate::observability::catalog::COMPACTION_FINALIZE_DURATION,
595            finalize_start.elapsed().as_secs_f64(),
596            &[],
597        );
598
599        // Step 3: All renames succeeded. The new SSTable is now visible.
600        // Delete input SSTable files. If deletion fails we log a warning but do
601        // NOT return an error — the merge output is correct.
602        //
603        // Issue #591 (write-while-mapped / Windows policy): the inputs were read
604        // through buffered I/O (never memory-mapped) and the merge has fully
605        // drained every source through its streaming producer by this point
606        // (issue #827), so the merger holds no mapping over them — deleting them
607        // cannot fault with SIGBUS. `delete_sstable_files` removes each
608        // input's TOC.txt first (unpublishing it) and is best-effort on the data
609        // components, so a component still pinned by a concurrent mapped reader on
610        // Windows becomes an invisible orphan reclaimed on the next startup rather
611        // than a hard failure or a source of duplicate rows.
612        for input_path in &merge.input_paths {
613            if let Err(e) = self.delete_sstable_files(input_path) {
614                tracing::warn!(
615                    "Failed to delete compaction input {:?}: {} \
616                     (merge output is valid; inputs will be re-evaluated next cycle)",
617                    input_path,
618                    e
619                );
620            }
621        }
622
623        // Reclaim the DROPPED-WHOLE SSTables (issue #1388): proven fully expired +
624        // overlap-safe, they were EXCLUDED from the merger (never read) so their
625        // reclamation happens here — AFTER the merged output published — via the
626        // same best-effort component-delete path as the merged inputs. A failure is
627        // an invisible orphan (TOC.txt removed first) reclaimed on next startup.
628        //
629        // This surface already deleted EVERY merge input above, so pass
630        // `already_deleted = merge.input_paths`: in the degenerate all-expired case
631        // the SSTable the all-dropped guard retained as a merge input is ALSO in
632        // `dropped_whole`, and the loop above already reclaimed it — the dedup skips
633        // it here to avoid a double-delete + spurious orphan warning (roborev #1388).
634        crate::storage::write_engine::merge::reclaim_dropped_whole(
635            &merge.dropped_whole,
636            &merge.input_paths,
637            |dropped| {
638                if let Err(e) = self.delete_sstable_files(dropped) {
639                    tracing::warn!(
640                        "Failed to delete dropped-whole compaction input {:?}: {} \
641                         (merge output is valid; leftover is an invisible orphan)",
642                        dropped,
643                        e
644                    );
645                }
646            },
647        );
648
649        // Step 4: Remove the now-empty tmp directory (best effort).
650        if let Err(e) = std::fs::remove_dir_all(&merge.tmp_dir) {
651            tracing::debug!(
652                "Failed to remove compaction tmp directory {:?}: {}",
653                merge.tmp_dir,
654                e
655            );
656        }
657
658        // The final Data.db path is in sstable_dir (renamed from tmp)
659        let final_data_path = sstable_dir.join(
660            tmp_info
661                .data_path
662                .file_name()
663                .ok_or_else(|| Error::Storage("Data.db path has no filename".to_string()))?,
664        );
665
666        // Compute total bytes written across ALL output SSTable components (not just Data.db).
667        // We stat the final paths (post-rename) so we measure what was actually persisted.
668        let stat_final = |p: &PathBuf| -> u64 {
669            let filename = p.file_name().unwrap_or_default();
670            std::fs::metadata(sstable_dir.join(filename))
671                .map(|m| m.len())
672                .unwrap_or(0)
673        };
674        // Index.db/Summary.db are BIG-only (BTI reports None — issue #908) and
675        // Filter.db is optional (disabled bloom filter omits it, Issue #852), so
676        // only stat the components that were actually written.
677        let mut byte_paths: Vec<&PathBuf> = vec![&tmp_info.data_path];
678        if let Some(ref p) = tmp_info.index_path {
679            byte_paths.push(p);
680        }
681        if let Some(ref p) = tmp_info.filter_path {
682            byte_paths.push(p);
683        }
684        if let Some(ref p) = tmp_info.summary_path {
685            byte_paths.push(p);
686        }
687        byte_paths.push(&tmp_info.stats_path);
688        byte_paths.push(&tmp_info.digest_path);
689        if let Some(ref p) = tmp_info.crc_path {
690            byte_paths.push(p);
691        }
692        if let Some(ref p) = tmp_info.partitions_path {
693            byte_paths.push(p);
694        }
695        let total_bytes_written: u64 = byte_paths.iter().map(|p| stat_final(p)).sum::<u64>()
696            + tmp_info
697                .compression_info_path
698                .as_ref()
699                .map(stat_final)
700                .unwrap_or(0);
701
702        // Update per-step report
703        report.completed_merges.push(final_data_path);
704        report.bytes_written += total_bytes_written;
705        // Record the dropped-whole set (issue #1388, R4), distinct from the merged
706        // inputs, so the drop decision is assertable from the plan/stats.
707        report
708            .dropped_whole
709            .extend(merge.dropped_whole.iter().cloned());
710
711        // Update cumulative lifetime stats
712        self.cumulative_stats.compactions_completed += 1;
713        self.cumulative_stats.sstables_merged_in += merge.input_paths.len() as u64;
714        self.cumulative_stats.sstables_produced += 1;
715        self.cumulative_stats.bytes_read += merge.bytes_read;
716        self.cumulative_stats.bytes_written += total_bytes_written;
717        self.cumulative_stats.rows_merged += merge.rows_merged;
718        self.cumulative_stats.total_time += elapsed;
719
720        // Compaction completion metrics (issue #1037): full-compaction duration
721        // (pairs with COMPACTION_ROWS_MERGED — emitted incrementally per
722        // maintenance step — for rows/sec throughput), bytes written across all
723        // output components, and SSTables in/out. Rows are NOT re-counted here to
724        // avoid double-counting the per-step COMPACTION_ROWS_MERGED increments.
725        let _ = merge_rows;
726        {
727            use crate::observability::{self as obs, catalog};
728            obs::record_histogram(catalog::COMPACTION_DURATION, elapsed.as_secs_f64(), &[]);
729            obs::add_counter(catalog::COMPACTION_BYTES_WRITTEN, total_bytes_written, &[]);
730            obs::add_counter(catalog::COMPACTION_SSTABLES_IN, input_count, &[]);
731            obs::add_counter(catalog::COMPACTION_SSTABLES_OUT, 1, &[]);
732        }
733
734        tracing::info!(
735            "Compaction complete: merged {} inputs → 1 output ({} bytes total across all components, {} rows, {:?})",
736            merge.input_paths.len(),
737            total_bytes_written,
738            merge.rows_merged,
739            elapsed
740        );
741
742        Ok(())
743    }
744}
