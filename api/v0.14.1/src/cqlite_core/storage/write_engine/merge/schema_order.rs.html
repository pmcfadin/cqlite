1//! Schema-aware heap-direct ordering (issue #1668, stages 5b-5c-i) — the
2//! "deferred crux" from stage 3a, solved for real.
3//!
4//! Stage 3a proved cross-group emission order was safe pre-5c-i only because
5//! [`super::KWayMerger::step_streaming`] drained a `Vec<MergeEntry>` that
6//! `merge_partition_rows` had already schema-aware-sorted (`merged.sort_by(|a,
7//! b| ck_a.compare(ck_b, &self.schema) ...)`) as its unconditional last step.
8//! [`super::KWayMerger`]'s `heap` field now uses [`SchemaOrderedEntry`]
9//! directly (stage 5c-i), so the heap's OWN pop order is schema-correct —
10//! see [`SchemaOrderedEntry`]'s `Ord` impl.
11//!
12//! ## Why a bounded lookahead/reorder buffer is NOT sufficient
13//!
14//! The issue's stage-5b framing offered two shapes: make the heap genuinely
15//! schema-aware, OR prove a small bounded lookahead is sufficient. The bound
16//! does not exist for a general schema: consider a single `DESC` clustering
17//! column with N distinct values. The fallback (ASC-only) order pops them
18//! ascending; the schema-correct (DESC) order needs the LARGEST value
19//! FIRST — which is the LAST value the fallback order would ever produce.
20//! Emitting the correct first item therefore requires having already seen
21//! ALL N items — the required lookahead is O(N), not a fixed constant
22//! independent of partition size. This is why the comparator itself must be
23//! schema-aware.
24//!
25//! ## Design: `Arc<TableSchema>`, not a borrowed lifetime
26//!
27//! `MergeEntry::Ord` cannot be made schema-aware directly: `Ord::cmp(&self,
28//! &other) -> Ordering` takes no extra context, and `ClusteringKey::compare`
29//! needs `&TableSchema`. A borrowed `&'a TableSchema` on every heap entry
30//! would make `KWayMerger` self-referential once the heap is a LONG-LIVED
31//! struct field (the heap borrowing a sibling field of the SAME struct) —
32//! not expressible in safe Rust. [`SchemaOrderedEntry`] instead holds an
33//! `Arc<TableSchema>` (cheap to clone per entry — a refcount bump, not a
34//! deep copy), populated from [`super::KWayMerger`]'s ADDITIVE `schema_arc`
35//! field (the pre-existing plain `schema: TableSchema` field is untouched,
36//! keeping every OTHER existing `self.schema` use-site in `mod.rs`
37//! unaffected by this change).

48/// Heap element pairing a [`MergeEntry`] with the schema it should be
49/// ordered against. `Ord` mirrors [`MergeEntry::Ord`] EXACTLY (same
50/// token/key-bytes/run_index tiebreaks) except at the clustering-key step,
51/// where it defers to the schema-aware [`ClusteringKey::compare`] — falling
52/// back to the plain `Ord` on error, the SAME fallback `merge_partition_rows`
53/// and `write_partition` already use for a malformed clustering key (more
54/// components than the schema declares).
55#[cfg(feature = "write-support")]
56#[derive(Debug, Clone)]
57pub(crate) struct SchemaOrderedEntry {
58    pub(crate) entry: MergeEntry,
59    pub(crate) schema: Arc<TableSchema>,
60}
61
62#[cfg(feature = "write-support")]
63impl SchemaOrderedEntry {
64    pub(crate) fn new(entry: MergeEntry, schema: Arc<TableSchema>) -> Self {
65        Self { entry, schema }
66    }
67}

69#[cfg(feature = "write-support")]
70impl PartialEq for SchemaOrderedEntry {
71    fn eq(&self, other: &Self) -> bool {
72        self.cmp(other) == Ordering::Equal
73    }
74}

76#[cfg(feature = "write-support")]
77impl Eq for SchemaOrderedEntry {}

79#[cfg(feature = "write-support")]
80impl PartialOrd for SchemaOrderedEntry {
81    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
82        Some(self.cmp(other))
83    }
84}

86#[cfg(feature = "write-support")]
87impl Ord for SchemaOrderedEntry {
88    fn cmp(&self, other: &Self) -> Ordering {
89        match self.entry.key.token.cmp(&other.entry.key.token) {
90            Ordering::Equal => match self.entry.key.key.cmp(&other.entry.key.key) {
91                Ordering::Equal => {
92                    match (&self.entry.clustering_key, &other.entry.clustering_key) {
93                        (None, None) => self.entry.run_index.cmp(&other.entry.run_index),
94                        (None, Some(_)) => Ordering::Less,
95                        (Some(_), None) => Ordering::Greater,
96                        (Some(a), Some(b)) => {
97                            // THE schema-aware substitution for MergeEntry's
98                            // fallback `a.cmp(b)`.
99                            match a.compare(b, &self.schema).unwrap_or_else(|_| a.cmp(b)) {
100                                Ordering::Equal => self.entry.run_index.cmp(&other.entry.run_index),
101                                other_ord => other_ord,
102                            }
103                        }
104                    }
105                }
106                other_ord => other_ord,
107            },
108            other_ord => other_ord,
109        }
110    }
111}

113/// Push every entry onto a schema-aware min-heap and pop them all, in order
114/// (issue #1668, stage 5b proof — still used by this module's own tests as a
115/// standalone equivalence check independent of a real `KWayMerger`). Test-only
116/// now that stage 5c-i wires `SchemaOrderedEntry` directly into
117/// `KWayMerger.heap` for production use.
118///
119/// Proves cross-group emission order can be correct DIRECTLY off a heap —
120/// no `merged.sort_by` needed afterward.
121#[cfg(all(test, feature = "write-support"))]
122pub(crate) fn schema_ordered_pop_all(
123    entries: Vec<MergeEntry>,
124    schema: &TableSchema,
125) -> Vec<MergeEntry> {
126    use std::cmp::Reverse;
127    use std::collections::BinaryHeap;
128
129    let schema_arc = Arc::new(schema.clone());
130    let mut heap: BinaryHeap<Reverse<SchemaOrderedEntry>> = BinaryHeap::new();
131    for entry in entries {
132        heap.push(Reverse(SchemaOrderedEntry::new(entry, schema_arc.clone())));
133    }
134    let mut out = Vec::with_capacity(heap.len());
135    while let Some(Reverse(wrapped)) = heap.pop() {
136        out.push(wrapped.entry);
137    }
138    out
139}

266    /// The grouping-contiguity property (issue #1668's "the crux") re-verified
267    /// for THIS comparator: `ClusteringKey::compare` is ALSO a valid total
268    /// order (transitive, antisymmetric — `compare_values` is the same
269    /// canonical value-ordering function used throughout the write path), so
270    /// same-clustering-key entries must still pop contiguously.
271    #[test]
272    fn heap_groups_contiguously_by_schema_aware_ord() {
273        let schema = single_col_schema(ClusteringOrder::Desc);
274        let entries = vec![
275            live_entry(2, 1, ClusteringKey::single("ck", Value::Integer(1)), 100),
276            live_entry(0, 1, ClusteringKey::single("ck", Value::Integer(0)), 300),
277            live_entry(1, 1, ClusteringKey::single("ck", Value::Integer(2)), 50),
278            live_entry(1, 1, ClusteringKey::single("ck", Value::Integer(0)), 200),
279            live_entry(2, 1, ClusteringKey::single("ck", Value::Integer(2)), 60),
280            live_entry(0, 1, ClusteringKey::single("ck", Value::Integer(1)), 150),
281            live_entry(0, 1, ClusteringKey::single("ck", Value::Integer(2)), 70),
282            live_entry(2, 1, ClusteringKey::single("ck", Value::Integer(0)), 250),
283            live_entry(1, 1, ClusteringKey::single("ck", Value::Integer(1)), 120),
284        ];
285
286        let popped = schema_ordered_pop_all(entries, &schema);
287        let popped_cks: Vec<i32> = popped.iter().map(ck_value).collect();
288
289        let mut closed: Vec<i32> = Vec::new();
290        let mut current: Option<i32> = None;
291        for &c in &popped_cks {
292            if current != Some(c) {
293                if let Some(prev) = current.take() {
294                    assert!(
295                        !closed.contains(&prev),
296                        "clustering key {prev} reappeared non-contiguously"
297                    );
298                    closed.push(prev);
299                }
300                current = Some(c);
301            }
302        }
303        let mut distinct: Vec<i32> = Vec::new();
304        for &c in &popped_cks {
305            if !distinct.contains(&c) {
306                distinct.push(c);
307            }
308        }
309        assert_eq!(distinct.len(), 3, "expected 3 distinct clustering keys");
310    }

312    /// THE proof: a `DESC` clustering column where the required lookahead to
313    /// emit the FIRST correctly-ordered item is the WHOLE partition (see the
314    /// module doc) — a bounded reorder buffer cannot solve this, only a
315    /// schema-aware comparator can. Assert the heap POPS the Cassandra-
316    /// correct descending order directly, with NO subsequent sort — an
317    /// independent expected-sequence check, not just "matches some other
318    /// path" (which could be equally wrong).
319    #[test]
320    fn schema_ordered_pop_all_matches_desc_clustering_order() {
321        let schema = single_col_schema(ClusteringOrder::Desc);
322        let entries = vec![
323            live_entry(0, 1, ClusteringKey::single("ck", Value::Integer(0)), 100),
324            live_entry(0, 1, ClusteringKey::single("ck", Value::Integer(1)), 100),
325            live_entry(0, 1, ClusteringKey::single("ck", Value::Integer(2)), 100),
326        ];
327
328        let popped = schema_ordered_pop_all(entries, &schema);
329        let order: Vec<i32> = popped.iter().map(ck_value).collect();
330        assert_eq!(
331            order,
332            vec![2, 1, 0],
333            "heap-direct pop order must be DESCENDING with NO final sort"
334        );
335    }

370    /// Absent-trailing-component (NULL-first) + `DESC` second column — the
371    /// SAME residual case stage 3a proved doesn't leak through the (already-
372    /// sorted-Vec-draining) `step_streaming` design. Here it is proven at
373    /// the HEAP level directly: absent-`ck2` sorts first (NULL-first,
374    /// regardless of `ck2`'s DESC-ness), then present values by DESC.
375    #[test]
376    fn schema_ordered_pop_all_matches_absent_trailing_component_order() {
377        let schema = two_col_schema();
378        let entries = vec![
379            live_entry(0, 1, ck_multi(&[("ck1", 5), ("ck2", 1)]), 100),
380            live_entry(0, 1, ck_multi(&[("ck1", 5)]), 100), // absent ck2
381            live_entry(0, 1, ck_multi(&[("ck1", 5), ("ck2", 2)]), 100),
382        ];
383
384        let popped = schema_ordered_pop_all(entries, &schema);
385        let ck2_markers: Vec<Option<i32>> = popped
386            .iter()
387            .map(|e| {
388                e.clustering_key.as_ref().and_then(|k| {
389                    k.columns
390                        .iter()
391                        .find(|(name, _)| name == "ck2")
392                        .map(|(_, v)| match v {
393                            Value::Integer(v) => *v,
394                            other => panic!("expected Integer ck2, got {other:?}"),
395                        })
396                })
397            })
398            .collect();
399
400        assert_eq!(
401            ck2_markers,
402            vec![None, Some(2), Some(1)],
403            "absent-ck2 must sort first (NULL-first), then DESC by ck2 — \
404             directly off the heap, no final sort"
405        );
406    }

408    /// Stage 5c-i finding: even though the heap's OWN pop order is now
409    /// schema-aware (this module), `merge_partition_rows`'s intermediate
410    /// `clustered_rows: BTreeMap<Option<ClusteringKey>, Vec<MergeEntry>>`
411    /// STILL discards that arrival order — a `BTreeMap`'s iteration order is
412    /// governed ENTIRELY by its own `Ord` impl on the key type (here
413    /// `ClusteringKey`'s FALLBACK, non-schema-aware `Ord`), independent of
414    /// insertion order. Inserting keys in schema-correct (DESC) arrival order
415    /// and then iterating the map yields the FALLBACK (ASC) order, not the
416    /// arrival order — proving `merge_partition_rows`'s final
417    /// `merged.sort_by(schema-aware compare)` is STILL load-bearing and NOT
418    /// yet redundant. This is a general property of `BTreeMap`, demonstrated
419    /// directly against the SAME type shape `clustered_rows` uses, without
420    /// touching (or removing anything from) `merge_partition_rows` itself.
421    #[test]
422    fn clustered_rows_btreemap_discards_schema_aware_arrival_order() {
423        use std::collections::BTreeMap;
424
425        // Simulate the NOW-schema-aware heap's arrival order for a DESC
426        // clustering column: 2, 1, 0 (descending — schema-correct).
427        let mut clustered_rows: BTreeMap<Option<ClusteringKey>, i32> = BTreeMap::new();
428        for v in [2, 1, 0] {
429            clustered_rows.insert(Some(ClusteringKey::single("ck", Value::Integer(v))), v);
430        }
431
432        let iterated: Vec<i32> = clustered_rows.into_values().collect();
433
434        // The map's OWN key Ord (fallback, ASC) governs iteration — NOT
435        // insertion order — so it comes out ascending, the OPPOSITE of the
436        // schema-correct DESC order the heap delivered.
437        assert_eq!(
438            iterated,
439            vec![0, 1, 2],
440            "BTreeMap iteration order is governed by the key's Ord, not \
441             insertion order — this is WHY merge_partition_rows's final \
442             merged.sort_by(schema-aware compare) is still necessary after \
443             stage 5c-i's schema-aware heap wiring: `clustered_rows` re-\
444             imposes the FALLBACK order regardless of arrival order"
445        );
446    }
