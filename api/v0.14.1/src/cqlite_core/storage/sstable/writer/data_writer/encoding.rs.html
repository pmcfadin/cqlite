13/// The column name targeted by a cell operation, if any (issue #927: used by
14/// mixed-stream reconciliation). `DeleteRow` targets no specific column.
15pub(crate) fn merged_op_column(
16    op: &crate::storage::write_engine::mutation::CellOperation,
17) -> Option<&str> {
18    use crate::storage::write_engine::mutation::CellOperation;
19    match op {
20        CellOperation::Write { column, .. }
21        | CellOperation::WriteWithTtl { column, .. }
22        | CellOperation::Delete { column, .. }
23        | CellOperation::WriteComplexElement { column, .. }
24        | CellOperation::ComplexDeletion { column, .. } => Some(column.as_str()),
25        CellOperation::DeleteRow => None,
26    }

29/// The `localDeletionTime` (seconds, GC clock) to stamp a cell tombstone with for
30/// an op carried by `mutation` (#921 finding 2).
31///
32/// A `CellOperation::Delete` that carries its OWN `local_deletion_time` (set by
33/// the compaction merge→rewrite path from the SOURCE cell tombstone's surfaced
34/// LDT) is honored VERBATIM so a surviving within-grace cell tombstone keeps its
35/// original GC clock across compactions. For every other op — and for a `Delete`
36/// with no surfaced LDT (`None`) — fall back to the mutation's
37/// [`effective_local_deletion_time`](crate::storage::write_engine::mutation::Mutation::effective_local_deletion_time),
38/// preserving the historical behavior (WAL / CQL-DELETE paths, row-derived LDT).
39/// Choose the `localDeletionTime` the cell/row tombstone of `op` is actually
40/// emitted with.
41///
42/// A `CellOperation::Delete { local_deletion_time: Some(ldt), .. }` carries an
43/// explicit per-cell LDT (#921 finding 2) that the DataWriter stamps VERBATIM;
44/// every other op (a `Delete` without a per-cell LDT, or a `DeleteRow`) derives
45/// the LDT from the enclosing mutation. The SSTable writer's STATS/baseline
46/// collection MUST record this exact value (not just
47/// `mutation.effective_local_deletion_time()`), otherwise a per-cell LDT below
48/// the mutation-derived value underflows the Data.db delta encoding, and one
49/// above it leaves Statistics.db min/max/histogram describing a tombstone that
50/// was never written (#921 finding 2 — roborev Medium).
51pub(crate) fn op_cell_local_deletion_time(
52    op: &crate::storage::write_engine::mutation::CellOperation,
53    mutation: &Mutation,
54) -> i32 {
55    use crate::storage::write_engine::mutation::CellOperation;
56    match op {
57        CellOperation::Delete {
58            local_deletion_time: Some(ldt),
59            ..
60        } => *ldt,
61        _ => mutation.effective_local_deletion_time(),
62    }
63}

65/// The write timestamp (microseconds) to stamp on the static cell emitted for
66/// `op` within `mutation` (issue #1018).
67///
68/// A `Write`/`WriteWithTtl`/`Delete` op carries its OWN per-cell write timestamp
69/// when the compaction merge→mutation path recorded one in
70/// [`Mutation::cell_write_timestamps`](crate::storage::write_engine::mutation::Mutation::cell_write_timestamps)
71/// (a surviving live cell's writetime, or a static cell tombstone's
72/// `markedForDeleteAt`). It is resolved via
73/// [`Mutation::cell_write_timestamp`](crate::storage::write_engine::mutation::Mutation::cell_write_timestamp),
74/// which falls back to the mutation's row `timestamp_micros` when no per-cell
75/// override exists. Every other op (and the no-override case) resolves to exactly
76/// `mutation.timestamp_micros`, so the single-writetime behavior is unchanged.
77///
78/// Shared by `collect_static_operations` (the compaction/partition path) and
79/// `DataWriter::write_static_row` (the public single-mutation entry point) so both
80/// resolve per-cell static writetimes IDENTICALLY — without it the public entry
81/// point would rewrite older static cells (live OR cell tombstones) to the row max.
82pub(crate) fn op_cell_write_timestamp(
83    op: &crate::storage::write_engine::mutation::CellOperation,
84    mutation: &Mutation,
85) -> i64 {
86    use crate::storage::write_engine::mutation::CellOperation;
87    match op {
88        CellOperation::Write { column, .. }
89        | CellOperation::WriteWithTtl { column, .. }
90        | CellOperation::Delete { column, .. } => mutation.cell_write_timestamp(column),
91        _ => mutation.timestamp_micros,
92    }
93}

95/// Generate a version-1 TimeUUID for use as a list cell path.
96///
97/// List elements in Cassandra use TimeUUIDs as cell paths to maintain insertion order.
98/// Each call with a different `element_index` produces a monotonically increasing UUID.
99///
100/// # Arguments
101/// * `timestamp_micros` - Mutation timestamp in microseconds since Unix epoch
102/// * `element_index` - Index of the element within the list (for monotonic ordering)
103pub(crate) fn generate_list_cell_path_timeuuid(
104    timestamp_micros: i64,
105    element_index: u64,
106) -> [u8; 16] {
107    // UUID v1 timestamp: 100-nanosecond intervals since UUID epoch (Oct 15, 1582)
108    // Offset from Unix epoch to UUID epoch in 100-ns units
109    const UUID_EPOCH_OFFSET: u64 = 0x01B2_1DD2_1381_4000;
110
111    let ts_100ns = (timestamp_micros as u64) * 10 + element_index;
112    let uuid_ts = ts_100ns + UUID_EPOCH_OFFSET;
113
114    // Extract time fields per RFC 4122
115    let time_low = (uuid_ts & 0xFFFF_FFFF) as u32;
116    let time_mid = ((uuid_ts >> 32) & 0xFFFF) as u16;
117    let time_hi = ((uuid_ts >> 48) & 0x0FFF) as u16 | 0x1000; // version 1
118
119    // Fixed clock_seq and node for deterministic output
120    let clock_seq: u16 = 0x80; // variant bits (10xx) + seq=0
121    let node: [u8; 6] = [0x00, 0x00, 0x00, 0x00, 0x00, 0x00];
122
123    let mut uuid = [0u8; 16];
124    uuid[0..4].copy_from_slice(&time_low.to_be_bytes());
125    uuid[4..6].copy_from_slice(&time_mid.to_be_bytes());
126    uuid[6..8].copy_from_slice(&time_hi.to_be_bytes());
127    uuid[8] = (clock_seq >> 8) as u8;
128    uuid[9] = (clock_seq & 0xFF) as u8;
129    uuid[10..16].copy_from_slice(&node);
130
131    uuid
132}

159/// Serialize a Value to bytes for cell storage
160///
161/// This follows Cassandra's type-specific serialization rules.
162pub(crate) fn serialize_value(value: &Value) -> Result<Vec<u8>> {
163    match value {
164        Value::Null => Ok(Vec::new()),
165        Value::Boolean(b) => Ok(vec![if *b { 1 } else { 0 }]),
166        Value::TinyInt(n) => Ok(vec![*n as u8]),
167        Value::SmallInt(n) => Ok(n.to_be_bytes().to_vec()),
168        Value::Integer(n) => Ok(n.to_be_bytes().to_vec()),
169        Value::BigInt(n) => Ok(n.to_be_bytes().to_vec()),
170        Value::Counter(n) => Ok(n.to_be_bytes().to_vec()),
171        Value::Float32(f) => Ok(f.to_bits().to_be_bytes().to_vec()),
172        Value::Float(f) => Ok(f.to_bits().to_be_bytes().to_vec()),
173        Value::Text(s) => Ok(s.as_bytes().to_vec()),
174        Value::Blob(bytes) => Ok(bytes.clone()),
175        Value::Timestamp(millis) => Ok(millis.to_be_bytes().to_vec()),
176        Value::Date(days) => {
177            // Cassandra DATE: stored as unsigned int with Integer.MIN_VALUE offset
178            let stored = days.wrapping_sub(i32::MIN) as u32;
179            Ok(stored.to_be_bytes().to_vec())
180        }
181        Value::Time(nanos) => Ok(nanos.to_be_bytes().to_vec()),
182        Value::Uuid(bytes) => Ok(bytes.to_vec()),
183        Value::Inet(bytes) => Ok(bytes.clone()),
184        Value::Varint(bytes) => Ok(bytes.clone()),
185        Value::Decimal { scale, unscaled } => {
186            let mut result = Vec::new();
187            result.extend_from_slice(&scale.to_be_bytes());
188            result.extend_from_slice(unscaled);
189            Ok(result)
190        }
191        Value::Duration {
192            months,
193            days,
194            nanos,
195        } => {
196            let mut result = Vec::new();
197            // Cassandra DurationType stores three signed VInts, not fixed-width ints.
198            encode_signed(*months as i64, &mut result);
199            encode_signed(*days as i64, &mut result);
200            encode_signed(*nanos, &mut result);
201            Ok(result)
202        }
203        Value::Udt(udt_value) => {
204            // Construct UdtTypeDef from UdtValue fields by inferring types
205            let mut schema =
206                UdtTypeDef::new(udt_value.keyspace.clone(), udt_value.type_name.clone());
207
208            // Infer field types from values
209            for field in &udt_value.fields {
210                let field_type = infer_cql_type_from_value(field.value.as_ref());
211                schema = schema.with_field(field.name.clone(), field_type, true);
212            }
213
214            let serializer = TypeSerializer::new();
215            serializer.serialize_udt(value, &schema)
216        }
217        Value::List(elements) => {
218            // ListType preserves insertion order — do NOT sort.
219            let mut buf = Vec::new();
220            buf.extend_from_slice(&len_as_i32(elements.len())?.to_be_bytes());
221            for elem in elements {
222                let elem_bytes = serialize_collection_element(elem, "Collection")?;
223                buf.extend_from_slice(&len_as_i32(elem_bytes.len())?.to_be_bytes());
224                buf.extend_from_slice(&elem_bytes);
225            }
226            Ok(buf)
227        }
228        Value::Set(elements) => {
229            // Cassandra SetType is a sorted collection: a frozen set serializes its
230            // elements in the SetType element-type comparator's order (issue #1254,
231            // #1275). Order by `compare_collection_elements` — SIGNED for numeric
232            // element types, unsigned-byte otherwise — decided from the element
233            // `Value`s (authoritative type), the SAME single comparator the
234            // non-frozen SET cell path uses (`write_set_complex_cells` in
235            // complex.rs). A raw serialized-byte sort (the pre-#1275 behavior) put
236            // negatives last, e.g. `frozen<set<int>>` of {-1,0,1} serialized in the
237            // wrong order. Then serialize each element in that order
238            // (`serialize_collection_element` rejects Value::Null).
239            let mut ordered: Vec<&Value> = elements.iter().collect();
240            ordered.sort_by(|a, b| compare_collection_elements(a, b));
241            let serialized: Vec<Vec<u8>> = ordered
242                .iter()
243                .map(|e| serialize_collection_element(e, "Collection"))
244                .collect::<Result<Vec<_>>>()?;
245
246            let mut buf = Vec::new();
247            buf.extend_from_slice(&len_as_i32(serialized.len())?.to_be_bytes());
248            for elem_bytes in &serialized {
249                buf.extend_from_slice(&len_as_i32(elem_bytes.len())?.to_be_bytes());
250                buf.extend_from_slice(elem_bytes);
251            }
252            Ok(buf)
253        }
254        Value::Map(entries) => {
255            // Cassandra MapType is a sorted collection: a frozen map serializes its
256            // entries in the KEY-type comparator's order (issue #1254, #1275). Order
257            // by `compare_collection_elements` over the KEY `Value`s — SIGNED for
258            // numeric key types, unsigned-byte otherwise — the SAME single
259            // comparator the non-frozen MAP cell path uses (`write_map_complex_cells`
260            // in complex.rs). A raw serialized-KEY-byte sort (the pre-#1275 behavior)
261            // ordered a `map<int,…>` with negative keys wrong. Then serialize each
262            // key/value in that order.
263            let mut ordered: Vec<&(Value, Value)> = entries.iter().collect();
264            ordered.sort_by(|a, b| compare_collection_elements(&a.0, &b.0));
265            let serialized: Vec<(Vec<u8>, Vec<u8>)> = ordered
266                .iter()
267                .map(|(key, val)| {
268                    if matches!(key, Value::Null) {
269                        return Err(Error::InvalidInput(
270                            "MAP keys cannot be null (CQL semantics)".to_string(),
271                        ));
272                    }
273                    Ok((serialize_value(key)?, serialize_value(val)?))
274                })
275                .collect::<Result<Vec<_>>>()?;
276
277            let mut buf = Vec::new();
278            buf.extend_from_slice(&len_as_i32(serialized.len())?.to_be_bytes());
279            for (key_bytes, val_bytes) in &serialized {
280                buf.extend_from_slice(&len_as_i32(key_bytes.len())?.to_be_bytes());
281                buf.extend_from_slice(key_bytes);
282                buf.extend_from_slice(&len_as_i32(val_bytes.len())?.to_be_bytes());
283                buf.extend_from_slice(val_bytes);
284            }
285            Ok(buf)
286        }
287        Value::Tuple(fields) => {
288            let mut buf = Vec::new();
289            for field in fields {
290                match field {
291                    Value::Null => buf.extend_from_slice(&(-1i32).to_be_bytes()),
292                    other => {
293                        let field_bytes = serialize_value(other)?;
294                        buf.extend_from_slice(&len_as_i32(field_bytes.len())?.to_be_bytes());
295                        buf.extend_from_slice(&field_bytes);
296                    }
297                }
298            }
299            Ok(buf)
300        }
301        Value::Frozen(inner) => serialize_value(inner),
302        _ => Err(Error::InvalidInput(format!(
303            "Unsupported value type for serialization: {:?}",
304            value
305        ))),
306    }
307}

309/// Infer CQL type from a Value instance
310///
311/// Used for UDT serialization when schema context is not available.
312/// Empty collections still fall back to `text` because there is no element
313/// value available to inspect.
314pub(crate) fn infer_cql_type_from_value(value: Option<&Value>) -> CqlType {
315    match value {
316        None | Some(Value::Null) => CqlType::Text, // Default for NULL
317        Some(Value::Boolean(_)) => CqlType::Boolean,
318        Some(Value::TinyInt(_)) => CqlType::TinyInt,
319        Some(Value::SmallInt(_)) => CqlType::SmallInt,
320        Some(Value::Integer(_)) => CqlType::Int,
321        Some(Value::BigInt(_)) => CqlType::BigInt,
322        Some(Value::Float32(_)) => CqlType::Float,
323        Some(Value::Float(_)) => CqlType::Double,
324        Some(Value::Text(_)) => CqlType::Text,
325        Some(Value::Blob(_)) => CqlType::Blob,
326        Some(Value::Timestamp(_)) => CqlType::Timestamp,
327        Some(Value::Date(_)) => CqlType::Date,
328        Some(Value::Time(_)) => CqlType::Time,
329        Some(Value::Uuid(_)) => CqlType::Uuid,
330        Some(Value::Inet(_)) => CqlType::Inet,
331        Some(Value::Varint(_)) => CqlType::Varint,
332        Some(Value::Decimal { .. }) => CqlType::Decimal,
333        Some(Value::Duration { .. }) => CqlType::Duration,
334        Some(Value::Counter(_)) => CqlType::Counter,
335        Some(Value::List(elements)) => CqlType::List(Box::new(
336            elements
337                .first()
338                .map(|elem| infer_cql_type_from_value(Some(elem)))
339                .unwrap_or(CqlType::Text),
340        )),
341        Some(Value::Set(elements)) => CqlType::Set(Box::new(
342            elements
343                .first()
344                .map(|elem| infer_cql_type_from_value(Some(elem)))
345                .unwrap_or(CqlType::Text),
346        )),
347        Some(Value::Map(entries)) => {
348            let (key_type, value_type) = entries
349                .first()
350                .map(|(key, value)| {
351                    (
352                        infer_cql_type_from_value(Some(key)),
353                        infer_cql_type_from_value(Some(value)),
354                    )
355                })
356                .unwrap_or((CqlType::Text, CqlType::Text));
357            CqlType::Map(Box::new(key_type), Box::new(value_type))
358        }
359        Some(Value::Tuple(fields)) => CqlType::Tuple(
360            fields
361                .iter()
362                .map(|field| infer_cql_type_from_value(Some(field)))
363                .collect(),
364        ),
365        Some(Value::Udt(udt)) => CqlType::Udt(
366            udt.type_name.clone(),
367            udt.fields
368                .iter()
369                .map(|field| {
370                    (
371                        field.name.clone(),
372                        infer_cql_type_from_value(field.value.as_ref()),
373                    )
374                })
375                .collect(),
376        ),
377        Some(Value::Frozen(inner)) => {
378            CqlType::Frozen(Box::new(infer_cql_type_from_value(Some(inner))))
379        }
380        Some(Value::Tombstone(_)) => CqlType::Text, // Tombstones shouldn't appear in UDT fields
381        Some(Value::Json(_)) => CqlType::Text,      // JSON is stored as text
382    }
383}

398pub(crate) fn is_static_row_mutation(mutation: &Mutation, schema: &TableSchema) -> bool {
399    if mutation.clustering_key.is_some() || !schema.columns.iter().any(|column| column.is_static) {
400        return false;
401    }
402
403    mutation.operations.iter().all(|operation| match operation {
404        crate::storage::write_engine::mutation::CellOperation::Write { column, .. }
405        | crate::storage::write_engine::mutation::CellOperation::WriteWithTtl { column, .. }
406        | crate::storage::write_engine::mutation::CellOperation::Delete { column, .. }
407        | crate::storage::write_engine::mutation::CellOperation::WriteComplexElement {
408            column,
409            ..
410        }
411        | crate::storage::write_engine::mutation::CellOperation::ComplexDeletion {
412            column, ..
413        } => schema
414            .columns
415            .iter()
416            .find(|candidate| candidate.name == *column)
417            .map(|candidate| candidate.is_static)
418            .unwrap_or(false),
419        crate::storage::write_engine::mutation::CellOperation::DeleteRow => true,
420    })
421}

448/// Returns true if `column` is part of the primary key — a partition-key or
449/// clustering-key column.
450///
451/// Primary-key columns are encoded positionally (the partition key and the row's
452/// clustering prefix); they must NEVER be written as regular cells. The compaction
453/// path can surface a clustering column as a `Write` op (the merger keeps the
454/// clustering cell for its own read-back, and `merge_entry_to_mutation` turns it
455/// into a `Write`); emitting it as a cell writes the value a second time and
456/// corrupts the row body for strict readers (#857). The writer drops such ops.
457pub(crate) fn is_primary_key_column(column: &str, schema: &TableSchema) -> bool {
458    schema.partition_keys.iter().any(|k| k.name == column)
459        || schema.clustering_keys.iter().any(|k| k.name == column)
460}

462/// Collect and merge static-column operations from all mutations in a partition.
463///
464/// Scans every mutation (regardless of whether it has a clustering key) and
465/// collects operations that target static columns.  Last-write-wins by
466/// `timestamp_micros` when the same column is written more than once.
467///
468/// Mutations at or before `shadow_floor` (the partition tombstone's deletion
469/// timestamp) are skipped: their static cells are shadowed and an sstable
470/// must be internally reconciled (see `DataWriter::write_partition`).
471///
472/// Returns the merged operations in an unspecified order (the writer will
473/// sort them by schema column order when building the row body). Each op
474/// carries the originating mutation's timestamp and explicit local deletion
475/// time (Issue #764) so a surviving older static delete keeps its own LDT
476/// instead of inheriting the newest static mutation's value.
477///
478/// Issue #1668, stage 5c-ii: a thin wrapper over [`StaticOpsTracker`] — the
479/// SAME per-mutation last-write-wins fold, now factored into a running
480/// tracker so a future incremental writer entry point (stage 5c-iv) can feed
481/// it one cluster group at a time instead of requiring the whole `&[Mutation]`
482/// slice upfront. Byte-identical to the prior whole-slice implementation.
483pub(crate) fn collect_static_operations(
484    mutations: &[Mutation],
485    schema: &TableSchema,
486    shadow_floor: Option<i64>,
487) -> Vec<StaticMergedOp> {
488    let mut tracker = StaticOpsTracker::new();
489    for mutation in mutations {
490        tracker.feed(mutation, schema, shadow_floor);
491    }
492    tracker.finish()
493}

495/// Derive a static row's liveness timestamp + TTL from the SURVIVING merged
496/// static ops (issue #1018, roborev HIGH).
497///
498/// The static row carries no row-level liveness in the emitted bytes (#1196 —
499/// the writetime rides on each static CELL), but the partition path still
500/// threads a `(latest_ts, ttl)` pair into `write_static_row_with_prev_size`.
501/// That pair MUST be derived from the ops that actually SURVIVED
502/// `collect_static_operations` — each carrying its own per-cell
503/// `timestamp_micros` after the per-cell shadow floor — NOT from the set of
504/// mutations that merely cleared the floor on their ROW MAX. A static `Write`
505/// whose per-cell writetime is `<= shadow_floor` is already dropped from
506/// `merged`, so it cannot contribute liveness/TTL here; deriving from the
507/// mutations' row max could otherwise resurrect a shadowed static cell's
508/// writetime. Returns the max per-cell timestamp and the TTL of the op holding
509/// it (last-write-wins). `None` when there are no surviving ops.
510pub(crate) fn static_liveness_from_ops(ops: &[StaticMergedOp]) -> Option<(i64, Option<u32>)> {
511    ops.iter()
512        .max_by_key(|mop| mop.timestamp_micros)
513        .map(|mop| (mop.timestamp_micros, mop.row_ttl_seconds))
514}

516/// Whether a range tombstone's clustering range covers the given clustering key.
517pub(crate) fn range_tombstone_covers(
518    rt: &RangeTombstone,
519    clustering_key: Option<&ClusteringKey>,
520    schema: &TableSchema,
521) -> bool {
522    use std::cmp::Ordering;
523
524    let Some(ck) = clustering_key else {
525        return false;
526    };
527    let cmp = |bound: &ClusteringKey| ck.compare(bound, schema).unwrap_or_else(|_| ck.cmp(bound));
528
529    let after_start = match &rt.start {
530        ClusteringBound::Inclusive(b) => cmp(b) != Ordering::Less,
531        ClusteringBound::Exclusive(b) => cmp(b) == Ordering::Greater,
532        ClusteringBound::Bottom => true,
533        ClusteringBound::Top => false,
534    };
535    let before_end = match &rt.end {
536        ClusteringBound::Inclusive(b) => cmp(b) != Ordering::Greater,
537        ClusteringBound::Exclusive(b) => cmp(b) == Ordering::Less,
538        ClusteringBound::Top => true,
539        ClusteringBound::Bottom => false,
540    };
541    after_start && before_end
542}

544/// Serialize value for clustering key (type-specific encoding)
545///
546/// Fixed-width types: raw bytes (no length prefix)
547/// Variable-width types: VInt length + bytes
548pub(crate) fn serialize_value_for_clustering(
549    value: &Value,
550    comparator: &ComparatorType,
551) -> Result<Vec<u8>> {
552    match (value, comparator) {
553        // Fixed-width types (no length prefix)
554        (Value::Boolean(b), ComparatorType::Boolean) => Ok(vec![if *b { 1 } else { 0 }]),
555        (Value::TinyInt(n), ComparatorType::TinyInt) => Ok(n.to_be_bytes().to_vec()),
556        (Value::SmallInt(n), ComparatorType::SmallInt) => Ok(n.to_be_bytes().to_vec()),
557        (Value::Integer(n), ComparatorType::Int) => Ok(n.to_be_bytes().to_vec()),
558        (Value::BigInt(n), ComparatorType::BigInt) => Ok(n.to_be_bytes().to_vec()),
559        (Value::Float32(f), ComparatorType::Float32) => Ok(f.to_bits().to_be_bytes().to_vec()),
560        (Value::Float(f), ComparatorType::Float) => Ok(f.to_bits().to_be_bytes().to_vec()),
561        (Value::Timestamp(millis), ComparatorType::Timestamp) => Ok(millis.to_be_bytes().to_vec()),
562        (Value::Date(days), ComparatorType::Date) => {
563            // Cassandra DATE in clustering keys: stored as unsigned int with Integer.MIN_VALUE offset
564            let stored = days.wrapping_sub(i32::MIN) as u32;
565            let mut result = Vec::new();
566            encode_unsigned(4, &mut result);
567            result.extend_from_slice(&stored.to_be_bytes());
568            Ok(result)
569        }
570        (Value::Uuid(bytes), ComparatorType::Uuid) => Ok(bytes.to_vec()),
571
572        // Variable-width types (VInt length + bytes)
573        (Value::Text(s), ComparatorType::Text) => {
574            let bytes = s.as_bytes();
575            let mut result = Vec::new();
576            encode_unsigned(bytes.len() as u64, &mut result);
577            result.extend_from_slice(bytes);
578            Ok(result)
579        }
580        (Value::Blob(bytes), ComparatorType::Blob) => {
581            let mut result = Vec::new();
582            encode_unsigned(bytes.len() as u64, &mut result);
583            result.extend_from_slice(bytes);
584            Ok(result)
585        }
586
587        // Frozen collections as clustering keys: serialize the full collection bytes with VInt length prefix
588        (Value::Frozen(inner), _) => {
589            let bytes = serialize_value(inner)?;
590            let mut result = Vec::new();
591            encode_unsigned(bytes.len() as u64, &mut result);
592            result.extend_from_slice(&bytes);
593            Ok(result)
594        }
595
596        _ => Err(Error::InvalidInput(format!(
597            "Type mismatch or unsupported clustering type: value={:?}, comparator={:?}",
598            value, comparator
599        ))),
600    }
601}

603/// Serialize a `ClusteringKey` as a Cassandra `ClusteringPrefix` byte sequence.
604///
605/// Format (same as the clustering prefix written in Data.db rows):
606/// ```text
607/// [header: unsigned VInt]   ← 2 bits per column: 00=present, 10=null
608/// [value bytes…]            ← type-specific bytes for each PRESENT column
609/// ```
610///
611/// Returns `Err` if a clustering column type is unknown; the caller falls back
612/// to `[0x00]` (empty header VInt, valid for "no columns") in that case.
613pub(super) fn serialize_clustering_prefix_to_vec(
614    clustering_key: &ClusteringKey,
615    schema: &TableSchema,
616) -> Result<Vec<u8>> {
617    let mut header = 0u64;
618    for (i, (_, value)) in clustering_key.columns.iter().enumerate() {
619        let state: u64 = match value {
620            Value::Null => 2, // NULL
621            _ => 0,           // PRESENT
622        };
623        header |= state << (i * 2);
624    }
625
626    let mut buf: Vec<u8> = Vec::new();
627    encode_unsigned(header, &mut buf);
628
629    for (i, (_, value)) in clustering_key.columns.iter().enumerate() {
630        if !matches!(value, Value::Null) {
631            if i >= schema.clustering_keys.len() {
632                return Err(crate::error::Error::Schema(format!(
633                    "Clustering key index {} out of range (schema has {})",
634                    i,
635                    schema.clustering_keys.len()
636                )));
637            }
638            let cluster_col = &schema.clustering_keys[i];
639            let comparator = ComparatorType::from_data_type(&cluster_col.data_type)?;
640            let value_bytes = serialize_value_for_clustering(value, &comparator)?;
641            buf.extend_from_slice(&value_bytes);
642        }
643    }
644
645    Ok(buf)
646}
