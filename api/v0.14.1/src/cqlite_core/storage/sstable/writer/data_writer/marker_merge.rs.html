1//! Incremental rows+markers interleave (issue #1668, stage 5c-iii).
2//!
3//! `sort_partition_items` (`partition.rs`) re-sorts the WHOLE combined
4//! `rows ++ markers` sequence from scratch every time, even though `rows`
5//! already arrives in clustering order (from `merge_clustering_rows`, which
6//! itself requires pre-sorted `mutations` — see stage 5a) and the range-
7//! tombstone markers are typically FEW (the stage-1 carrier pre-scan already
8//! established this). [`merge_rows_and_markers`] replaces that full re-sort
9//! with a classic two-way MERGE: sort the (small) marker list once, then
10//! interleave it with the already-sorted rows in one forward pass — no
11//! re-sorting of the combined, potentially partition-sized sequence.
12//!
13//! The comparator itself, [`partition_item_cmp`], is extracted VERBATIM from
14//! `sort_partition_items`'s prior inline closure (moved, not duplicated) so
15//! BOTH the old full-sort path and this new merge path always agree — a
16//! future divergence between the two would be a single-function fix, not a
17//! two-place hunt.
18//!
19//! ## Why a merge is safe here (no tie-break ambiguity)
20//!
21//! A `Row` at a given clustering value always carries `weight = 0`
22//! (`CLUSTERING` ordinal), while a `Marker` at ANY clustering value carries
23//! `weight = ±1` — never `0`. So a `Row` and a `Marker` can never compare
24//! EQUAL under [`partition_item_cmp`] (the tuple always differs at the
25//! `weight` field), which is exactly why choosing "take from `rows` on a
26//! non-`Greater` comparison" in [`merge_rows_and_markers`]'s tie-break never
27//! actually has to arbitrate a real tie between the two streams — only
28//! marker-vs-marker or row-vs-row comparisons could tie, and those are each
29//! already resolved within their OWN pre-sorted stream before the merge
30//! step ever compares across streams.

34/// Sort key for clustering-ordered emission: `(position class, clustering
35/// values, bound weight, kind ordinal)`. class: -1 = before all rows
36/// (Bottom), 0 = positioned by clustering values, 1 = after all rows (Top).
37/// The weight orders a marker relative to a row at the SAME clustering value
38/// (`ClusteringPrefix.Kind.comparedToClustering`): an inclusive-start /
39/// exclusive-end bound sorts before the row, an inclusive-end / exclusive-
40/// start bound after it.
41///
42/// The final element is the Cassandra `ClusteringPrefix.Kind` ordinal, used
43/// as a strict tiebreak so that — at an equal clustering point and equal
44/// weight — the CLOSING bound of one range always sorts immediately before
45/// the matching OPENING bound of the next, regardless of the order the
46/// `range_tombstones` arrived in (issue #1220, roborev finding 2).
47fn sort_class<'a, 'b>(item: &'b PartitionItem<'a>) -> (i8, Option<&'b ClusteringKey>, i8, u8) {
48    match item {
49        PartitionItem::Row(row) => (0, row.clustering_key, 0, CLUSTERING),
50        PartitionItem::Marker { bound, is_open, .. } => {
51            let kind = bound_kind_ordinal(bound, *is_open);
52            match bound {
53                ClusteringBound::Inclusive(ck) => {
54                    (0, Some(ck), if *is_open { -1 } else { 1 }, kind)
55                }
56                ClusteringBound::Exclusive(ck) => {
57                    (0, Some(ck), if *is_open { 1 } else { -1 }, kind)
58                }
59                ClusteringBound::Bottom => (-1, None, 0, kind),
60                ClusteringBound::Top => (1, None, 0, kind),
61            }
62        }
63        PartitionItem::Boundary {
64            kind, clustering, ..
65        } => (0, Some(clustering), 0, *kind),
66    }
67}

69/// The Cassandra `ClusteringPrefix.Kind` ordinal for a range-tombstone BOUND.
70/// Used purely as a sort tiebreak (see [`sort_class`]); the canonical
71/// ordering is the one decoded on the read path (`row_decoder/block_emit.rs`)
72/// and the constants in `data_writer/mod.rs`:
73///   0 EXCL_END_BOUND · 1 INCL_START_BOUND · 6 INCL_END_BOUND · 7 EXCL_START_BOUND.
74fn bound_kind_ordinal(bound: &ClusteringBound, is_open: bool) -> u8 {
75    match (is_open, bound) {
76        (true, ClusteringBound::Inclusive(_)) => INCL_START_BOUND, // 1
77        (false, ClusteringBound::Exclusive(_)) => EXCL_END_BOUND,  // 0
78        (false, ClusteringBound::Inclusive(_)) => INCL_END_BOUND,  // 6
79        (true, ClusteringBound::Exclusive(_)) => EXCL_START_BOUND, // 7
80        // Open-ended bounds are positioned by `class` (Bottom/Top), never
81        // reach a value-level tiebreak; report their side's kind for
82        // completeness.
83        (true, ClusteringBound::Bottom | ClusteringBound::Top) => INCL_START_BOUND,
84        (false, ClusteringBound::Bottom | ClusteringBound::Top) => INCL_END_BOUND,
85    }
86}

88/// Total order over [`PartitionItem`]s (rows + bound markers), schema-aware.
89/// Moved VERBATIM out of `sort_partition_items`'s prior inline closure
90/// (issue #1668, stage 5c-iii) so the full-sort and merge paths share ONE
91/// comparator.
92pub(super) fn partition_item_cmp(
93    a: &PartitionItem,
94    b: &PartitionItem,
95    schema: &TableSchema,
96) -> std::cmp::Ordering {
97    let (class_a, ck_a, weight_a, kind_a) = sort_class(a);
98    let (class_b, ck_b, weight_b, kind_b) = sort_class(b);
99    class_a
100        .cmp(&class_b)
101        .then_with(|| match (ck_a, ck_b) {
102            (Some(x), Some(y)) => x.compare(y, schema).unwrap_or_else(|_| x.cmp(y)),
103            _ => std::cmp::Ordering::Equal,
104        })
105        .then(weight_a.cmp(&weight_b))
106        .then(kind_a.cmp(&kind_b))
107}

109/// Merge already-sorted `rows` with the (typically few) markers derived from
110/// `range_tombstones`, WITHOUT re-sorting the combined sequence from scratch
111/// (issue #1668, stage 5c-iii).
112///
113/// `rows` MUST already be in clustering order (guaranteed by
114/// `merge_clustering_rows`'s pre-sorted `mutations` input — see stage 5a).
115/// The markers are built from `range_tombstones` and sorted ONCE (cheap: this
116/// list is small), then interleaved with `rows` in a single forward two-way
117/// merge pass. Produces the IDENTICAL sequence `sort_partition_items` (the
118/// full re-sort) computes for the same inputs — proven by this module's
119/// tests.
120///
121/// NOT yet called by any production path: `write_partition_with_index_blocks`
122/// still calls `sort_partition_items` (the full re-sort) — swapping to this
123/// merge is stage 5c-iv's job, once the writer's incremental entry point
124/// exists to actually feed it row-by-row. `#[allow(dead_code)]` until then,
125/// matching the crate's convention for proof-only surface pending wiring
126/// (see `schema_order.rs`'s stage-5b history).
127#[allow(dead_code)]
128pub(super) fn merge_rows_and_markers<'a>(
129    rows: Vec<PartitionItem<'a>>,
130    range_tombstones: &'a [RangeTombstone],
131    schema: &TableSchema,
132) -> Vec<PartitionItem<'a>> {
133    let mut markers: Vec<PartitionItem<'a>> = Vec::with_capacity(range_tombstones.len() * 2);
134    for rt in range_tombstones {
135        markers.push(PartitionItem::Marker {
136            bound: &rt.start,
137            is_open: true,
138            deletion_time: rt.deletion_time,
139            local_deletion_time: rt.local_deletion_time,
140        });
141        markers.push(PartitionItem::Marker {
142            bound: &rt.end,
143            is_open: false,
144            deletion_time: rt.deletion_time,
145            local_deletion_time: rt.local_deletion_time,
146        });
147    }
148    // The marker list is small (typically far fewer than the row count), so
149    // sorting IT (not the combined sequence) stays cheap.
150    markers.sort_by(|a, b| partition_item_cmp(a, b, schema));
151
152    let mut out = Vec::with_capacity(rows.len() + markers.len());
153    let mut rows_iter = rows.into_iter().peekable();
154    let mut markers_iter = markers.into_iter().peekable();
155    loop {
156        match (rows_iter.peek(), markers_iter.peek()) {
157            (Some(r), Some(m)) => {
158                // A Row and a Marker never compare Equal (see the module
159                // doc), so this branch never has to arbitrate a real tie.
160                if partition_item_cmp(r, m, schema) != std::cmp::Ordering::Greater {
161                    // SAFETY-free `unwrap`: `peek()` just proved `Some`.
162                    if let Some(row) = rows_iter.next() {
163                        out.push(row);
164                    }
165                } else if let Some(marker) = markers_iter.next() {
166                    out.push(marker);
167                }
168            }
169            (Some(_), None) => {
170                out.extend(rows_iter.by_ref());
171                break;
172            }
173            (None, Some(_)) => {
174                out.extend(markers_iter.by_ref());
175                break;
176            }
177            (None, None) => break,
178        }
179    }
180    out
181}

190    fn schema_one_clustering(order: ClusteringOrder) -> TableSchema {
191        TableSchema {
192            keyspace: "ks".to_string(),
193            table: "t".to_string(),
194            partition_keys: vec![KeyColumn {
195                name: "id".to_string(),
196                data_type: "int".to_string(),
197                position: 0,
198            }],
199            clustering_keys: vec![ClusteringColumn {
200                name: "ck".to_string(),
201                data_type: "int".to_string(),
202                position: 0,
203                order,
204            }],
205            columns: vec![],
206            comments: std::collections::HashMap::new(),
207            dropped_columns: std::collections::HashMap::new(),
208        }
209    }

211    fn ck(v: i32) -> ClusteringKey {
212        ClusteringKey::single("ck", Value::Integer(v))
213    }
214
215    fn row_item(v: i32) -> PartitionItem<'static> {
216        // A minimal RowWrite: no ops, no deletion — only clustering_key
217        // matters for the interleave/sort comparison under test.
218        PartitionItem::Row(RowWrite {
219            clustering_key: Some(Box::leak(Box::new(ck(v)))),
220            liveness_ts: Some(0),
221            ttl_seconds: None,
222            row_deletion: None,
223            ops: Vec::new(),
224            complex_element_ops: Vec::new(),
225        })
226    }
227
228    fn row_key(item: &PartitionItem<'_>) -> Option<i32> {
229        match item {
230            PartitionItem::Row(row) => match row.clustering_key.and_then(|k| k.columns.first()) {
231                Some((_, Value::Integer(v))) => Some(*v),
232                _ => None,
233            },
234            _ => None,
235        }
236    }
237
238    fn is_marker(item: &PartitionItem<'_>) -> bool {
239        matches!(item, PartitionItem::Marker { .. })
240    }

242    /// THE proof: a partition with range tombstones interleaved with rows —
243    /// the incremental merge must produce the IDENTICAL sequence the full
244    /// re-sort (`sort_partition_items`) computes for the SAME inputs.
245    /// Independently reasoned expected shape (not just old==new): with a
246    /// range tombstone covering `[1, 3]` (inclusive both ends) and rows at
247    /// 0, 1, 2, 3, 4, the OPEN marker must sort strictly before row 1 (the
248    /// inclusive-start weight is -1, ahead of the row's weight 0) and the
249    /// CLOSE marker strictly after row 3 (inclusive-end weight is +1).
250    #[test]
251    fn merge_matches_full_sort_for_range_tombstone_interleaved_with_rows() {
252        let schema = schema_one_clustering(ClusteringOrder::Asc);
253        let rt = RangeTombstone {
254            start: ClusteringBound::Inclusive(ck(1)),
255            end: ClusteringBound::Inclusive(ck(3)),
256            deletion_time: 1000,
257            local_deletion_time: 100,
258        };
259        let range_tombstones = vec![rt];
260
261        // Incremental merge (stage 5c-iii).
262        let merged =
263            merge_rows_and_markers((0..5).map(row_item).collect(), &range_tombstones, &schema);
264
265        // Full re-sort (today's path): rows + 2 markers, then sort_by the
266        // SAME shared comparator.
267        let mut combined: Vec<PartitionItem> = (0..5).map(row_item).collect();
268        for rt in &range_tombstones {
269            combined.push(PartitionItem::Marker {
270                bound: &rt.start,
271                is_open: true,
272                deletion_time: rt.deletion_time,
273                local_deletion_time: rt.local_deletion_time,
274            });
275            combined.push(PartitionItem::Marker {
276                bound: &rt.end,
277                is_open: false,
278                deletion_time: rt.deletion_time,
279                local_deletion_time: rt.local_deletion_time,
280            });
281        }
282        combined.sort_by(|a, b| partition_item_cmp(a, b, &schema));
283
284        // Independent expectation: open marker between row 0 and row 1;
285        // close marker between row 3 and row 4.
286        let shape: Vec<(Option<i32>, bool)> =
287            merged.iter().map(|i| (row_key(i), is_marker(i))).collect();
288        assert_eq!(
289            shape,
290            vec![
291                (Some(0), false),
292                (None, true), // open marker, before row 1
293                (Some(1), false),
294                (Some(2), false),
295                (Some(3), false),
296                (None, true), // close marker, after row 3
297                (Some(4), false),
298            ],
299            "merge must place the open marker before row 1 and the close \
300             marker after row 3 (inclusive-both-ends range tombstone)"
301        );
302
303        // Merge output must be IDENTICAL to the full re-sort's output.
304        let merged_shape: Vec<(Option<i32>, bool)> =
305            merged.iter().map(|i| (row_key(i), is_marker(i))).collect();
306        let combined_shape: Vec<(Option<i32>, bool)> = combined
307            .iter()
308            .map(|i| (row_key(i), is_marker(i)))
309            .collect();
310        assert_eq!(
311            merged_shape, combined_shape,
312            "incremental merge must produce the IDENTICAL interleaved \
313             sequence as the full re-sort for the same inputs"
314        );
315    }
