1//! `SSTableWriter`-level incremental partition-write wiring (issue #1668,
2//! stage 5c-iv part 2).
3//!
4//! Wraps `DataWriter::begin_partition_incremental` (stage 5c-iv part 1) with
5//! the SAME bookkeeping `SSTableWriter::write_partition` performs around it
6//! (token-order validation, key-range/baseline stats, and — once the
7//! `DataWriter`-level session finishes — Index.db/Filter.db/Summary.db/BTI
8//! registration and partition-count/observability counters), split into two
9//! calls so the CALLER (a streaming merge consumer) can drive the
10//! `DataWriter`-level session directly in between, feeding one cluster group
11//! at a time instead of assembling a whole `Vec<Mutation>` first:
12//!
13//! ```text
14//! let mut session = output_writer.begin_partition_incremental(&key, tombstone, &rts)?;
15//! if schema_has_static { session.feed_static_row(&ops, ts, schema)?; }
16//! for mutation in cluster_groups { session.feed_row(&mutation, schema)?; }
17//! let (offset, blocks, emit) = session.finish(schema)?;
18//! output_writer.complete_partition_incremental(&key, tombstone, offset, &blocks, emit)?;
19//! ```
20//!
21//! `begin_partition_incremental` returns `data_writer::IncrementalPartitionWriter<'w, 'r>`
22//! DIRECTLY — its lifetime `'w` is tied to the WHOLE `&'w mut SSTableWriter`
23//! borrow (Rust cannot narrow a returned borrow to "just one field" across a
24//! function boundary), so the caller cannot use `output_writer` for anything
25//! else while the session is alive. This is not a real constraint: the
26//! caller only ever calls `session.feed_row`/`finish` during that window,
27//! never anything else on `output_writer` — and once `finish()` consumes the
28//! session, NLL proves the borrow has ended, freeing `output_writer` for the
29//! `complete_partition_incremental` call. Splitting `finish()`'s Data.db-only
30//! result (offset/blocks/emit — plain owned values, no borrow) out from the
31//! remaining bookkeeping is exactly what makes this composable.
32

43impl SSTableWriter {
44    /// This writer's OWN schema (issue #1668, stage 5c-iv part 2) — the
45    /// header/column layout ACTUALLY on disk (e.g. a compaction's
46    /// `write_schema`, with fully-purged dropped columns already stripped),
47    /// as opposed to a merge layer's DECODE-time schema (e.g.
48    /// `KWayMerger::schema`/`effective_schema`, which still declares a
49    /// dropped column so its cells can be decoded and purge-evaluated).
50    ///
51    /// `write_partition` never needed this exposed: it always used its OWN
52    /// `&self.schema` internally for emission, decoupled from whatever schema
53    /// the caller used to decode/reconcile the `Vec<Mutation>` it was handed.
54    /// The incremental streaming path's `feed_row`/`feed_static_row`/`finish`
55    /// take an EXPLICIT schema argument instead (issue #1668 stage 5c-iv part
56    /// 1), so a caller assembling that argument from the WRONG schema (its
57    /// own decode-time schema, not this writer's) silently corrupts the
58    /// row/header encoding — passing MORE (or fewer) columns than the header
59    /// this writer already committed to declares. Callers MUST pass
60    /// `writer.schema()` here, never their own decode-time schema.
61    pub(crate) fn schema(&self) -> &TableSchema {
62        &self.schema
63    }
64
65    /// Begin an incremental partition write (issue #1668, stage 5c-iv part
66    /// 2). Performs the SAME pre-work `write_partition` does before handing
67    /// off to `DataWriter` (token-order validation, key-range tracking), then
68    /// delegates to `DataWriter::begin_partition_incremental`.
69    ///
70    /// # Precondition: baselines must be pre-seeded
71    ///
72    /// Unlike `write_partition` (which has the whole partition's
73    /// `Vec<Mutation>` in hand and can fold its stats BEFORE pushing a
74    /// baseline to `DataWriter`), this entry point streams mutations in one
75    /// at a time and cannot know a partition's minimum LDT/timestamp before
76    /// it starts emitting bytes for that partition — buffering ahead to find
77    /// out would defeat the entire point of this streaming path (bounding
78    /// peak memory to `max_row_width × k`, issue #1668). So this method
79    /// requires `self.baselines_locked` (i.e. the caller already called
80    /// `pre_seed_encoding_baselines` with the true minimum computed over ALL
81    /// inputs) and asserts it rather than silently risking an LDT-delta
82    /// underflow panic deep inside `DataWriter`. This is not a real
83    /// constraint on the one production caller: `compact_sstables_with_registry`
84    /// always pre-seeds baselines before calling `KWayMerger::merge`. A
85    /// caller that has not pre-seeded (e.g. a unit test constructing a bare
86    /// `SSTableWriter::new`) must call `pre_seed_encoding_baselines` first —
87    /// exactly as production does — or use `write_partition` instead.
88    ///
89    /// The caller must NOT use `self` again until the returned session is
90    /// consumed via `IncrementalPartitionWriter::finish` and this writer's
91    /// `complete_partition_incremental` is called with its result — see the
92    /// module doc.
93    pub(crate) fn begin_partition_incremental<'w, 'r>(
94        &'w mut self,
95        key: &DecoratedKey,
96        partition_tombstone: Option<&PartitionTombstone>,
97        range_tombstones: &'r [RangeTombstone],
98    ) -> Result<IncrementalPartitionWriter<'w, 'r>> {
99        debug_assert!(
100            self.baselines_locked,
101            "begin_partition_incremental requires pre-seeded encoding baselines \
102             (call pre_seed_encoding_baselines before streaming partitions) — \
103             see the precondition doc on this method, issue #1668"
104        );
105        if let Some(last_token) = self.last_token {
106            if key.token <= last_token {
107                return Err(Error::InvalidInput(format!(
108                    "Partitions must be written in token order: got token {} after {}",
109                    key.token, last_token
110                )));
111            }
112        }
113        self.last_token = Some(key.token);
114        self.stats.update_key_range(&key.key);
115
116        // No baseline push here (unlike `write_partition`): `baselines_locked`
117        // is required (see doc above), so `write_partition`'s
118        // `if !self.baselines_locked { update_stats_from_metadata }` gate
119        // would always be a no-op anyway.
120
121        self.data_writer.begin_partition_incremental(
122            key,
123            partition_tombstone,
124            range_tombstones,
125            &self.schema,
126        )
127    }
128
129    /// Finalize the SSTableWriter-level bookkeeping for a partition written
130    /// via the incremental entry point (issue #1668, stage 5c-iv part 2).
131    /// Call this AFTER `IncrementalPartitionWriter::finish` has already
132    /// returned its `(offset, blocks, emit)` — this method never touches
133    /// `DataWriter` again, only the OTHER `SSTableWriter` fields (Index.db,
134    /// Filter.db, Summary.db, BTI, partition-count/observability), mirroring
135    /// `write_partition`'s tail EXACTLY (moved, not duplicated in spirit —
136    /// see `write_partition`'s own comments for the rationale behind each
137    /// step).
138    ///
139    /// `partition_stats` is the caller's per-partition fold (every mutation
140    /// of this partition folded through `stats_fold::fold_mutation_stats` as
141    /// it streamed by — the caller cannot fold directly into
142    /// `self.stats` while the `IncrementalPartitionWriter` session holds an
143    /// exclusive borrow of `self`). Merged in here, once the session's borrow
144    /// has ended, via `stats_fold::merge_stats_fold` so the FINAL
145    /// `StatisticsMetadata` (max values / tombstone histogram / live-LDT
146    /// sentinel / partition-level-deletion flag) matches `write_partition`'s
147    /// byte-for-byte, even though `baselines_locked` already made the
148    /// DataWriter-baseline half of the old push a no-op.
149    pub(crate) fn complete_partition_incremental(
150        &mut self,
151        key: &DecoratedKey,
152        partition_tombstone: Option<&PartitionTombstone>,
153        data_offset: u64,
154        promoted_blocks: &[PromotedIndexBlock],
155        emit_counts: PartitionEmitCounts,
156        partition_stats: &StatisticsMetadata,
157    ) -> Result<()> {
158        stats_fold::merge_stats_fold(&mut self.stats, partition_stats);
159        self.stats.row_count += emit_counts.rows;
160        self.stats.column_count += emit_counts.columns;
161
162        let partition_serialized_size = self.data_writer.position().saturating_sub(data_offset);
163        self.stats
164            .record_partition(partition_serialized_size, emit_counts.columns);
165
166        let entry_info =
167            self.index_writer
168                .add_partition_with_promoted(key, data_offset, promoted_blocks)?;
169
170        if let Some(ref mut filter) = self.filter_writer {
171            filter.add_key(key);
172        }
173
174        self.queue_bti_partition(key, data_offset, promoted_blocks, partition_tombstone);
175
176        self.summary_writer.note_partition(key);
177        if self.summary_sample_counter % self.summary_sample_interval == 0 {
178            self.summary_writer
179                .add_entry(key, entry_info.index_offset)?;
180        }
181        self.summary_sample_counter += 1;
182        self.partition_count += 1;
183        self.stats.increment_partition_count();
184
185        crate::observability::add_counter(crate::observability::catalog::WRITE_PARTITIONS, 1, &[]);
186
187        Ok(())
188    }

190    /// Begin a CROSS-CALL RESUMABLE incremental partition write (issue
191    /// #1668, stage 5c-iv part 3) — for a caller that must be able to
192    /// return control to an outer scheduler mid-partition and resume later
193    /// (`WriteEngine::maintenance_step`'s budget-driven pause/resume, stage
194    /// 4), unlike [`Self::begin_partition_incremental`]'s session, which
195    /// borrows `self` for its whole lifetime and therefore cannot survive a
196    /// function return.
197    ///
198    /// Same pre-work and the same pre-seeded-baselines precondition as
199    /// [`Self::begin_partition_incremental`] — see its doc for the
200    /// rationale, which applies unchanged here.
201    pub(crate) fn begin_streaming_partition(
202        &mut self,
203        key: &DecoratedKey,
204        partition_tombstone: Option<&PartitionTombstone>,
205        range_tombstones: &[RangeTombstone],
206    ) -> Result<StreamingPartitionSession> {
207        debug_assert!(
208            self.baselines_locked,
209            "begin_streaming_partition requires pre-seeded encoding baselines \
210             (call pre_seed_encoding_baselines before streaming partitions) — \
211             see begin_partition_incremental's precondition doc, issue #1668"
212        );
213        if let Some(last_token) = self.last_token {
214            if key.token <= last_token {
215                return Err(Error::InvalidInput(format!(
216                    "Partitions must be written in token order: got token {} after {}",
217                    key.token, last_token
218                )));
219            }
220        }
221        self.last_token = Some(key.token);
222        self.stats.update_key_range(&key.key);
223
224        self.data_writer.begin_streaming_partition(
225            key,
226            partition_tombstone,
227            range_tombstones,
228            &self.schema,
229        )
230    }

232    /// Feed one clustering row into a [`StreamingPartitionSession`] (issue
233    /// #1668, stage 5c-iv part 3). Borrows `self.data_writer` for just this
234    /// one call — unlike [`IncrementalPartitionWriter`], `session` itself
235    /// never holds a borrow of `self`, so it can be stored (e.g. on
236    /// `ActiveMerge`) and fed across as many separate calls as needed.
237    pub(crate) fn feed_streaming_row(
238        &mut self,
239        session: &mut StreamingPartitionSession,
240        mutation: &Mutation,
241    ) -> Result<()> {
242        session.feed_row(&mut self.data_writer, mutation, &self.schema)
243    }
244
245    /// Feed the static-row prelude into a [`StreamingPartitionSession`]
246    /// (issue #1668, stage 5c-iv part 3) — called at most once, before any
247    /// [`Self::feed_streaming_row`] call for the same session.
248    pub(crate) fn feed_streaming_static_row(
249        &mut self,
250        session: &mut StreamingPartitionSession,
251        merged: &[StaticMergedOp],
252        first_mutation_ts: i64,
253    ) -> Result<()> {
254        session.feed_static_row(
255            &mut self.data_writer,
256            merged,
257            first_mutation_ts,
258            &self.schema,
259        )
260    }
261
262    /// Finalize a [`StreamingPartitionSession`] (issue #1668, stage 5c-iv
263    /// part 3) — returns the same `(offset, blocks, emit)` shape
264    /// [`IncrementalPartitionWriter::finish`] does; the caller must still
265    /// call [`Self::complete_partition_incremental`] with the result (that
266    /// method is shared by both sessions — it only needs the returned
267    /// values, not the session type).
268    pub(crate) fn finish_streaming_partition(
269        &mut self,
270        session: StreamingPartitionSession,
271    ) -> Result<(u64, Vec<PromotedIndexBlock>, PartitionEmitCounts)> {
272        session.finish(&mut self.data_writer, &self.schema)
273    }
