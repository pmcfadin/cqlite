1//! Shared per-mutation statistics fold (issue #1668, stage 5c-iv part 2).
2//!
3//! [`SSTableWriter::write_partition`] folds every mutation of a partition into
4//! `StatisticsMetadata` (min/max timestamp, LDT, TTL, the tombstone-drop-time
5//! histogram, the live-LDT sentinel, and the partition-level-deletion flag)
6//! BEFORE writing any bytes for that partition. The incremental streaming path
7//! (`KWayMerger::merge`, `writer/incremental.rs`) sees mutations one at a time
8//! and cannot buffer a whole partition to reproduce that ordering, but the
9//! fold itself is a pure function of ONE mutation plus the running
10//! accumulator — extracted here so both paths call the exact same logic and
11//! can never drift.
12//!
13//! [`SSTableWriter::write_partition`]: super::SSTableWriter::write_partition
14
15use crate::storage::sstable::writer::stats_writer::StatisticsMetadata;
16use crate::storage::write_engine::mutation::{CellOperation, Mutation};

18/// Fold one mutation's timestamp/TTL/local-deletion-time/tombstone information
19/// into `stats`. Mirrors the per-mutation loop body that used to live inline in
20/// [`super::SSTableWriter::write_partition`] verbatim — same chokepoints
21/// (`update_timestamp`, `update_local_deletion_time`, `update_ttl`,
22/// `note_live_local_deletion_time`, `mark_partition_level_deletion`), same
23/// per-`CellOperation` handling, so folding every mutation of a partition
24/// through this function (in any order — the chokepoints are commutative
25/// min/max folds, issue #1668 stage 5a) reproduces `write_partition`'s final
26/// `StatisticsMetadata` byte-for-byte.
27///
28/// Callers: `write_partition` (once per mutation, whole-partition buffered)
29/// and the incremental streaming path (once per mutation, as it streams
30/// through — see `writer/incremental.rs` doc comments for how the streamed
31/// partition-scoped fold is merged into the running `SSTableWriter`-wide
32/// `stats` at partition end).
33pub(crate) fn fold_mutation_stats(stats: &mut StatisticsMetadata, mutation: &Mutation) {
34    stats.update_timestamp(mutation.timestamp_micros);
35    // Issue #1018: simple `Write`/`WriteWithTtl`/`Delete` cells may carry their
36    // OWN (lower) per-cell timestamps in `Mutation::cell_write_timestamps` (a
37    // live cell's writetime OR a cell tombstone's markedForDeleteAt) and are
38    // emitted with an explicit `min_timestamp` delta. Fold every per-cell
39    // timestamp into the stats BEFORE emitting cells so `min_timestamp` can
40    // never exceed an emitted cell's actual timestamp (which would underflow
41    // the unsigned-VInt delta). Mirrors the pre-pass fold in
42    // `compute_mutations_baseline_stats`.
43    if let Some(cell_ts) = &mutation.cell_write_timestamps {
44        for ts in cell_ts.values() {
45            stats.update_timestamp(*ts);
46        }
47    }
48    if let Some(ttl) = mutation.ttl_seconds {
49        stats.update_ttl(ttl as i32);
50        let now_seconds = std::time::SystemTime::now()
51            .duration_since(std::time::UNIX_EPOCH)
52            .map(|d| d.as_secs() as i32)
53            .unwrap_or(0);
54        let local_deletion_time = now_seconds.saturating_add(ttl as i32);
55        stats.update_local_deletion_time(local_deletion_time);
56    }
57    // Track local deletion times for tombstones and TTL cells. Issue #764:
58    // row/cell tombstones use the caller-supplied `local_deletion_time` when
59    // present, else the timestamp-derived value.
60    for op in &mutation.operations {
61        match op {
62            CellOperation::WriteWithTtl {
63                ttl_seconds,
64                local_deletion_time,
65                ..
66            } => {
67                stats.update_ttl(*ttl_seconds as i32);
68                // Issue #1538: honor the authoritative per-cell LDT VERBATIM
69                // when present (a surviving expiring cell preserved through
70                // compaction); `None` keeps the historical `now + ttl`
71                // derivation.
72                let local_deletion_time = match local_deletion_time {
73                    Some(ldt) => *ldt,
74                    None => {
75                        let now_seconds = std::time::SystemTime::now()
76                            .duration_since(std::time::UNIX_EPOCH)
77                            .map(|d| d.as_secs() as i32)
78                            .unwrap_or(0);
79                        now_seconds.saturating_add(*ttl_seconds as i32)
80                    }
81                };
82                stats.update_local_deletion_time(local_deletion_time);
83            }
84            op @ (CellOperation::Delete { .. } | CellOperation::DeleteRow) => {
85                // Issue #764 / #921 finding 2: record the EXACT LDT the
86                // tombstone is emitted with, via the same helper the emit path
87                // uses, so stats and Data.db bytes agree exactly.
88                let local_deletion_time =
89                    crate::storage::sstable::writer::data_writer::op_cell_local_deletion_time(
90                        op, mutation,
91                    );
92                stats.update_local_deletion_time(local_deletion_time);
93            }
94            // Issue #887: a `ComplexDeletion` marker is physically written with
95            // its OWN `marked_for_delete_at` / `local_deletion_time`, which may
96            // fall outside the row's own timestamp/LDT range.
97            CellOperation::ComplexDeletion {
98                marked_for_delete_at,
99                local_deletion_time,
100                ..
101            } => {
102                stats.update_timestamp(*marked_for_delete_at);
103                stats.update_local_deletion_time(*local_deletion_time);
104            }
105            // Issue #887: a per-element complex cell carries its OWN explicit
106            // timestamp/ttl/local_deletion_time.
107            CellOperation::WriteComplexElement {
108                timestamp_micros,
109                ttl_seconds,
110                local_deletion_time,
111                is_deleted,
112                ..
113            } => {
114                stats.update_timestamp(*timestamp_micros);
115                if let Some(ttl) = ttl_seconds {
116                    stats.update_ttl(*ttl as i32);
117                }
118                if let Some(ldt) = local_deletion_time {
119                    stats.update_local_deletion_time(*ldt);
120                }
121                // Issue #1728 (roborev finding 2): a LIVE complex element
122                // carries Cassandra's `NO_DELETION_TIME` sentinel just like a
123                // live simple `Write`.
124                if !*is_deleted && ttl_seconds.is_none() && local_deletion_time.is_none() {
125                    stats.note_live_local_deletion_time();
126                }
127            }
128            // Issue #1728: a live, non-TTL `Write` cell carries Cassandra's
129            // `Cell.NO_DELETION_TIME` sentinel as its localDeletionTime.
130            CellOperation::Write { value, .. } => {
131                if mutation.ttl_seconds.is_none() && !matches!(value, crate::types::Value::Null) {
132                    stats.note_live_local_deletion_time();
133                }
134            }
135        }
136    }
137    // Track stats for partition tombstones.
138    if let Some(pt) = &mutation.partition_tombstone {
139        stats.update_timestamp(pt.deletion_time);
140        stats.update_local_deletion_time(pt.local_deletion_time);
141        stats.mark_partition_level_deletion();
142    }
143    // Track stats for range tombstones.
144    for rt in &mutation.range_tombstones {
145        stats.update_timestamp(rt.deletion_time);
146        stats.update_local_deletion_time(rt.local_deletion_time);
147    }
148    // Issue #1721: a decoupled row tombstone (#932
149    // `Mutation::row_tombstone = Some((deletion_time, ldt))`) is emitted as a
150    // `HAS_DELETION` row stamped with its OWN `(deletion_time, ldt)` —
151    // DECOUPLED from `timestamp_micros`, so the per-cell/mutation folds above
152    // never see it.
153    if let Some((deletion_time, ldt)) = mutation.row_tombstone {
154        stats.update_timestamp(deletion_time);
155        stats.update_local_deletion_time(ldt);
156    }
157}

159/// Fold `from`'s accumulated range/flags into `into` (issue #1668 stage
160/// 5c-iv part 2). Used to merge a partition-scoped fold (accumulated while
161/// streaming a partition's mutations one at a time) into the SSTable-wide
162/// running `StatisticsMetadata` at partition end, once the incremental
163/// session's exclusive borrow of the writer has ended.
164///
165/// Min/max fields are commutative folds (stage 5a), so feeding both of
166/// `from`'s min and max back through the same chokepoints on `into`
167/// reproduces exactly what folding every mutation directly into `into` would
168/// have produced — PROVIDED `from`'s own untouched-default sentinels are
169/// excluded first. `update_timestamp` already self-filters its untouched
170/// defaults (`i64::MAX`/`i64::MIN`, both treated as the LIVE/NO_DELETION
171/// marker, issue #851), so timestamps are safe to feed unconditionally. LDT
172/// and TTL are NOT symmetric:
173///   * `update_local_deletion_time` filters `i32::MAX` (live sentinel) but
174///     NOT `i32::MIN` (`from.max_local_deletion_time`'s untouched default
175///     when no tombstone was ever folded) — guarded explicitly below, else an
176///     empty `from` would drag `into.min_local_deletion_time` down to
177///     `i32::MIN`.
178///   * `update_ttl` only filters non-positive values, so `from.min_ttl`'s
179///     untouched default (`i32::MAX`, itself positive) must be excluded
180///     explicitly, else it would corrupt `into.max_ttl` to `i32::MAX`.
181///
182/// The live-LDT sentinel itself is NOT re-derived through
183/// `update_local_deletion_time` (which filters `i32::MAX` OUT) —
184/// `from.max_local_deletion_time == i32::MAX` is used as the equivalent "saw
185/// a live cell" signal instead, since `note_live_local_deletion_time` is the
186/// ONLY setter that can assign `i32::MAX` to `max_local_deletion_time` (a
187/// real tombstone LDT is always filtered before it can reach the max).
188pub(crate) fn merge_stats_fold(into: &mut StatisticsMetadata, from: &StatisticsMetadata) {
189    into.update_timestamp(from.min_timestamp);
190    into.update_timestamp(from.max_timestamp);
191
192    into.update_local_deletion_time(from.min_local_deletion_time);
193    if from.max_local_deletion_time > i32::MIN {
194        into.update_local_deletion_time(from.max_local_deletion_time);
195    }
196    if from.max_local_deletion_time == i32::MAX {
197        into.note_live_local_deletion_time();
198    }
199
200    if from.min_ttl != i32::MAX {
201        into.update_ttl(from.min_ttl);
202    }
203    into.update_ttl(from.max_ttl);
204
205    if from.has_partition_level_deletions {
206        into.mark_partition_level_deletion();
207    }
208}

345    /// The relevant `StatisticsMetadata` fields `fold_mutation_stats`/
346    /// `merge_stats_fold` actually touch, for equivalence comparison (no
347    /// `PartialEq` on the full struct — the histogram/key-range/repair fields
348    /// are untouched by this fold and irrelevant here).
349    #[derive(Debug, PartialEq)]
350    struct FoldSnapshot {
351        min_timestamp: i64,
352        max_timestamp: i64,
353        min_local_deletion_time: i32,
354        max_local_deletion_time: i32,
355        min_ttl: i32,
356        max_ttl: i32,
357        has_partition_level_deletions: bool,
358    }
359
360    impl From<&StatisticsMetadata> for FoldSnapshot {
361        fn from(s: &StatisticsMetadata) -> Self {
362            Self {
363                min_timestamp: s.min_timestamp,
364                max_timestamp: s.max_timestamp,
365                min_local_deletion_time: s.min_local_deletion_time,
366                max_local_deletion_time: s.max_local_deletion_time,
367                min_ttl: s.min_ttl,
368                max_ttl: s.max_ttl,
369                has_partition_level_deletions: s.has_partition_level_deletions,
370            }
371        }
372    }

374    /// Correctness proof (issue #1668 stage 5c-iv part 2): folding every
375    /// mutation of a partition DIRECTLY into one accumulator (what
376    /// `write_partition` does — the old behavior) must produce the IDENTICAL
377    /// final min/max/flag aggregates as the incremental path's split —
378    /// folding disjoint SUBSETS of the same mutations into SEPARATE
379    /// partition-scoped accumulators (simulating streaming them across
380    /// several `feed_row`/`feed_static_row` calls) and then merging those
381    /// sub-folds back together via `merge_stats_fold` (simulating
382    /// `complete_partition_incremental`).
383    #[test]
384    fn split_and_merge_matches_direct_fold_for_every_mutation_kind() {
385        let mutations = representative_mutations();
386
387        // "write_partition"-equivalent: fold every mutation directly into one
388        // running accumulator.
389        let mut direct = StatisticsMetadata::new();
390        for m in &mutations {
391            fold_mutation_stats(&mut direct, m);
392        }
393
394        // Incremental-equivalent: split into three arbitrary, non-trivial
395        // partition-scoped sub-folds (as if fed across three separate
396        // `feed_row`/`feed_static_row` calls before merging at partition
397        // end), then merge them all into a fresh running accumulator.
398        let mut part_a = StatisticsMetadata::new();
399        let mut part_b = StatisticsMetadata::new();
400        let mut part_c = StatisticsMetadata::new();
401        for (i, m) in mutations.iter().enumerate() {
402            match i % 3 {
403                0 => fold_mutation_stats(&mut part_a, m),
404                1 => fold_mutation_stats(&mut part_b, m),
405                _ => fold_mutation_stats(&mut part_c, m),
406            }
407        }
408        let mut merged = StatisticsMetadata::new();
409        merge_stats_fold(&mut merged, &part_a);
410        merge_stats_fold(&mut merged, &part_b);
411        merge_stats_fold(&mut merged, &part_c);
412
413        assert_eq!(
414            FoldSnapshot::from(&direct),
415            FoldSnapshot::from(&merged),
416            "splitting the fold across partition-scoped sub-accumulators and \
417             merging must reproduce write_partition's direct fold exactly"
418        );
419
420        // Sanity: the live sentinel and partition-deletion flag must actually
421        // have been exercised by this fixture, else the equality above would
422        // pass vacuously without proving the sentinel-handling guards work.
423        assert_eq!(
424            direct.max_local_deletion_time,
425            i32::MAX,
426            "live_write_row must have lifted the live-LDT sentinel"
427        );
428        assert!(
429            direct.has_partition_level_deletions,
430            "partition_only must have set the partition-level-deletion flag"
431        );
432    }

434    /// Guard: an EMPTY partition-scoped sub-fold (a partition with no
435    /// tombstones/TTLs/live cells reaching this fold at all) must merge as a
436    /// true no-op — proving the untouched-default-sentinel guards in
437    /// `merge_stats_fold` (`max_local_deletion_time > i32::MIN`, `min_ttl !=
438    /// i32::MAX`) actually prevent corruption, not just coincidentally pass
439    /// on the representative fixture above.
440    #[test]
441    fn merging_an_empty_sub_fold_is_a_no_op() {
442        let mut into = StatisticsMetadata::new();
443        fold_mutation_stats(
444            &mut into,
445            &Mutation::new(
446                table(),
447                pk(),
448                Some(ck(1)),
449                vec![CellOperation::WriteWithTtl {
450                    column: "v".to_string(),
451                    value: Value::Text("x".to_string()),
452                    ttl_seconds: 60,
453                    local_deletion_time: Some(1_000),
454                }],
455                100,
456                Some(60),
457            ),
458        );
459        let before = FoldSnapshot::from(&into);
460
461        let empty = StatisticsMetadata::new();
462        merge_stats_fold(&mut into, &empty);
463
464        assert_eq!(
465            FoldSnapshot::from(&into),
466            before,
467            "merging a never-folded (default) StatisticsMetadata must not \
468             change min_local_deletion_time to i32::MIN or max_ttl to i32::MAX"
469        );
470    }
