1//! Authoritative zstd frame-header parsing (issue #1414).
2//!
3//! CQLite ships **no-dictionary** zstd only. A zstd **dictionary**-compressed
4//! chunk is a valid but UNSUPPORTED feature — not corruption — and must be
5//! rejected fail-closed BEFORE a plain decode is attempted. Detection is done by
6//! parsing the frame header per the zstd frame format (RFC 8878 §3.1), never by
7//! guessing from byte patterns (no-heuristics mandate, issue #28).
8
9use super::compression_info::CompressionInfo;
10
11/// The zstd frame magic number as it appears on the wire (little-endian of
12/// `0xFD2FB528`). Every zstd frame begins with these four bytes.
13pub(crate) const ZSTD_MAGIC_LE: [u8; 4] = [0x28, 0xB5, 0x2F, 0xFD];
14
15/// Detect an unsupported zstd dictionary and return the typed rejection message,
16/// or `None` when no dictionary is present (the normal decode path proceeds).
17///
18/// Checks AUTHORITATIVE metadata first (a dictionary option in
19/// `CompressionInfo.db`), then the frame header (a nonzero `Dictionary_ID`). Both
20/// are authoritative sources — never a byte-pattern guess (issue #1414, #28). The
21/// caller wraps the message in `Error::UnsupportedFormat`, failing closed BEFORE
22/// attempting a plain decode.
23pub(crate) fn zstd_dictionary_rejection(
24    info: &CompressionInfo,
25    compressed_data: &[u8],
26    chunk_index: usize,
27) -> Option<String> {
28    let chunk_offset = info.chunk_offsets.get(chunk_index).copied().unwrap_or(0);
29    if let Some(option) = info.zstd_dictionary_option() {
30        return Some(format!(
31            "zstd dictionary compression (CompressionInfo.db option '{option}') is unsupported \
32             for chunk {chunk_index} at offset 0x{chunk_offset:x}"
33        ));
34    }
35    if let Some(dict_id) = zstd_frame_dictionary_id(compressed_data) {
36        return Some(format!(
37            "zstd dictionary compression (Dictionary_ID={dict_id}) is unsupported for chunk \
38             {chunk_index} at offset 0x{chunk_offset:x}"
39        ));
40    }
41    None
42}

44/// Parse the AUTHORITATIVE zstd frame header to extract a nonzero `Dictionary_ID`.
45///
46/// Per the zstd frame format (RFC 8878 §3.1): a frame starts with the 4-byte
47/// magic `0xFD2FB528` (little-endian on disk), then a 1-byte
48/// `Frame_Header_Descriptor` whose low two bits are the `Dictionary_ID_flag`
49/// (values 0/1/2/3 map to 0/1/2/4 `Dictionary_ID` bytes) and whose bit 5 is the
50/// `Single_Segment_flag` (a 1-byte `Window_Descriptor` follows the descriptor
51/// iff that flag is 0). The `Dictionary_ID`, when present, is a little-endian
52/// unsigned integer immediately after the (optional) window descriptor.
53///
54/// Returns `Some(id)` only for a well-formed frame carrying a NONZERO dictionary
55/// ID — authoritative evidence of dictionary compression, which CQLite does not
56/// support (issue #1414). Returns `None` for a plain (no-dictionary) frame, a
57/// non-zstd byte run, or a header too short to decide; the normal decode path
58/// then handles those. This is pure frame-header parsing per the spec — NOT a
59/// byte-pattern heuristic (no-heuristics mandate, issue #28).
60///
61/// Known sub-case: a frame with `Dictionary_ID_flag == 0` uses an out-of-band
62/// dictionary (the `Dictionary_ID` is omitted from the frame header per RFC 8878
63/// §3.1.1.1.1), so this returns `None` for it; such frames are still handled
64/// fail-closed by the `CompressionInfo.db` dictionary-option metadata path
65/// (`zstd_dictionary_option`) and, absent that, error out during decode — never
66/// returning wrong rows.
67pub(crate) fn zstd_frame_dictionary_id(frame: &[u8]) -> Option<u32> {
68    if frame.len() < 5 || frame[0..4] != ZSTD_MAGIC_LE {
69        return None;
70    }
71    let frame_header_descriptor = frame[4];
72    // Low two bits: Dictionary_ID_flag → 0/1/2/4 ID bytes.
73    let did_size = match frame_header_descriptor & 0x03 {
74        0 => return None, // no Dictionary_ID field present
75        1 => 1usize,
76        2 => 2usize,
77        _ => 4usize, // flag value 3
78    };
79    // Bit 5: Single_Segment_flag. A Window_Descriptor byte follows the descriptor
80    // iff this flag is 0.
81    let single_segment = (frame_header_descriptor & 0x20) != 0;
82    let did_start = 5 + usize::from(!single_segment);
83    let did_end = did_start.saturating_add(did_size);
84    if frame.len() < did_end {
85        return None;
86    }
87    let mut id: u32 = 0;
88    for (i, &b) in frame[did_start..did_end].iter().enumerate() {
89        id |= u32::from(b) << (8 * i);
90    }
91    (id != 0).then_some(id)
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97
98    #[test]
99    fn zstd_frame_dictionary_id_parses_authoritatively() {
100        // Magic 0xFD2FB528 (LE on disk): 28 B5 2F FD.
101        const MAGIC: [u8; 4] = ZSTD_MAGIC_LE;
102
103        // Not a zstd frame / too short → None.
104        assert_eq!(zstd_frame_dictionary_id(&[]), None);
105        assert_eq!(
106            zstd_frame_dictionary_id(&[0x00, 0x01, 0x02, 0x03, 0x04]),
107            None
108        );
109
110        // Dictionary_ID_flag = 0 → no ID field → None (plain frame).
111        // FHD 0x20 sets Single_Segment_flag (no window byte), flag bits 0.
112        assert_eq!(
113            zstd_frame_dictionary_id(&[MAGIC[0], MAGIC[1], MAGIC[2], MAGIC[3], 0x20]),
114            None
115        );
116
117        // 1-byte Dictionary_ID, single-segment (no window descriptor):
118        // FHD = 0x21 → Single_Segment=1, DID flag=1 (1 ID byte). ID byte = 0x2A.
119        let frame_1b = [MAGIC[0], MAGIC[1], MAGIC[2], MAGIC[3], 0x21, 0x2A];
120        assert_eq!(zstd_frame_dictionary_id(&frame_1b), Some(0x2A));
121
122        // 4-byte Dictionary_ID, single-segment: FHD = 0x23 → DID flag=3 (4 bytes).
123        // LE 0x04 0x03 0x02 0x01 → 0x01020304.
124        let frame_4b = [
125            MAGIC[0], MAGIC[1], MAGIC[2], MAGIC[3], 0x23, 0x04, 0x03, 0x02, 0x01,
126        ];
127        assert_eq!(zstd_frame_dictionary_id(&frame_4b), Some(0x0102_0304));
128
129        // NOT single-segment: FHD = 0x01 → DID flag=1 (1 byte) AND a
130        // Window_Descriptor byte precedes the ID. window=0x00, ID=0x2A.
131        let frame_win = [MAGIC[0], MAGIC[1], MAGIC[2], MAGIC[3], 0x01, 0x00, 0x2A];
132        assert_eq!(zstd_frame_dictionary_id(&frame_win), Some(0x2A));
133
134        // Header truncated before the ID bytes → None (defer to the decode path).
135        let truncated = [MAGIC[0], MAGIC[1], MAGIC[2], MAGIC[3], 0x23, 0x04];
136        assert_eq!(zstd_frame_dictionary_id(&truncated), None);
137
138        // Explicit zero Dictionary_ID → None (a present-but-zero ID is no dict).
139        let frame_zero = [MAGIC[0], MAGIC[1], MAGIC[2], MAGIC[3], 0x21, 0x00];
140        assert_eq!(zstd_frame_dictionary_id(&frame_zero), None);
141    }
142}
