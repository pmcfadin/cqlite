1//! C3 single-walk BTI partition-lookup memo (issue #1574).
2//!
3//! A single-candidate `WHERE pk = ?` point read against a BTI SSTable descends the
4//! `Partitions.db` trie twice for the SAME key — once for the candidate prune
5//! (`might_contain_partition`) and once for the seek. The resolved uncompressed
6//! `Data.db` offset (or authoritative absence) is a pure function of the immutable
7//! trie + key, so [`lookup_partition_via_bti_trie`] stores it in a single-entry
8//! reader-local slot ([`SSTableReader::bti_lookup_memo`]) and the seek reuses it
9//! without a second descent — keeping `TRIE_WALKS` at 1 per point read. A stale or
10//! absent slot (a different key, or a concurrent read) simply misses and re-walks,
11//! never a wrong result. This is deliberately NOT a cross-lookup key/offset cache
12//! (that is Epic B/B4); it is bounded to one entry.
13//!
14//! Split out of `partition_lookup.rs` (campsite / epic #1116): keeps that file under
15//! the source size threshold.
16//!
17//! [`lookup_partition_via_bti_trie`]: SSTableReader::lookup_partition_via_bti_trie
18
19use super::SSTableReader;
20
21impl SSTableReader {
22    /// Emit the BTI presence-oracle observability counters (`READ_BLOOM_CHECKS`
23    /// and `READ_PARTITION_LOOKUP`) for a resolved BTI partition lookup, keyed on
24    /// whether the key was found. Used by the C3 single-walk memo-hit path so a
25    /// reused resolution records the same presence decision a fresh descent does
26    /// (only the `TRIE_WALKS` descent is skipped, not the presence accounting).
27    pub(super) fn emit_bti_presence_counters(found: bool) {
28        use crate::observability::{self as obs, catalog};
29        let result = if found { "hit" } else { "miss" };
30        obs::add_counter(
31            catalog::READ_BLOOM_CHECKS,
32            1,
33            &[
34                (catalog::attr::RESULT, result.into()),
35                (catalog::attr::SSTABLE_FORMAT, "bti".into()),
36            ],
37        );
38        obs::add_counter(
39            catalog::READ_PARTITION_LOOKUP,
40            1,
41            &[
42                (catalog::attr::RESULT, result.into()),
43                (catalog::attr::LOOKUP_ROUTE, "bti_trie".into()),
44                (catalog::attr::SSTABLE_FORMAT, "bti".into()),
45            ],
46        );
47    }
48
49    /// C3 same-key memo read: returns `Some(resolved)` when the most recent BTI
50    /// resolution was for this exact `partition_key` (so the caller can skip a
51    /// second trie descent), or `None` on a miss / different key / poisoned lock.
52    /// The memoized value is a pure function of the immutable trie + key.
53    pub(super) fn bti_lookup_memo_get(&self, partition_key: &[u8]) -> Option<Option<u64>> {
54        let guard = self.bti_lookup_memo.lock().ok()?;
55        match guard.as_ref() {
56            Some((k, resolved)) if k.as_ref() == partition_key => Some(*resolved),
57            _ => None,
58        }
59    }
60
61    /// C3 same-key memo write: record this key's resolved offset for reuse by the
62    /// immediately-following seek. Best effort — a poisoned lock is a no-op (the
63    /// seek just re-walks, still correct).
64    pub(super) fn bti_lookup_memo_store(&self, partition_key: &[u8], resolved: Option<u64>) {
65        if let Ok(mut guard) = self.bti_lookup_memo.lock() {
66            *guard = Some((Box::from(partition_key), resolved));
67        }
68    }
69}

71#[cfg(test)]
72mod tests {
73    /// The single-slot memo's match logic (same-key hit / different-key miss / the
74    /// stored resolution round-trips) mirrors [`SSTableReader::bti_lookup_memo_get`]
75    /// on the same `Option<(Box<[u8]>, Option<u64>)>` slot the reader field holds.
76    /// The reader-level wiring (`TRIE_WALKS == 1`) is covered end-to-end by
77    /// `tests/issue_1574_bti_single_walk.rs`.
78    #[test]
79    fn memo_slot_same_key_hits_different_key_misses() {
80        // store a resolution for key-A
81        let mut slot: Option<(Box<[u8]>, Option<u64>)> = Some((Box::from(&b"key-A"[..]), Some(63)));
82
83        // The same-key match predicate used by `bti_lookup_memo_get`.
84        let get = |slot: &Option<(Box<[u8]>, Option<u64>)>, key: &[u8]| -> Option<Option<u64>> {
85            match slot.as_ref() {
86                Some((k, resolved)) if k.as_ref() == key => Some(*resolved),
87                _ => None,
88            }
89        };
90
91        assert_eq!(get(&slot, b"key-A"), Some(Some(63)), "same key hits");
92        assert_eq!(get(&slot, b"key-B"), None, "different key misses");
93
94        // An absence (None) resolution is a valid, distinct memoized value.
95        slot = Some((Box::from(&b"key-C"[..]), None));
96        assert_eq!(get(&slot, b"key-C"), Some(None), "absence round-trips");
97        assert_eq!(get(&slot, b"key-A"), None, "overwritten key misses");
98    }
99}
