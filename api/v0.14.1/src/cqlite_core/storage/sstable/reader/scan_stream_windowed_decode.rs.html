1//! IO-half chunk decode for the windowed streaming scan (issue #1940, D2).
2//!
3//! Split out of `scan_stream_windowed.rs` to keep that source file under the
4//! campsite-rule size limit (epic #1116). `decode_scan_chunk` is the one place the
5//! windowed-scan IO half turns a freshly-read, CRC-verified compressed chunk into
6//! the refcounted decompressed `Bytes` substrate shipped on the chunk channel,
7//! while handing the compressed buffer back so the feed loop recycles it as the
8//! next read's scratch (≤1 alloc/chunk). Included via
9//! `#[path = "scan_stream_windowed_decode.rs"] impl`-carrying module in the parent.
10
11use super::SSTableReader;
12use crate::Result;
13
14impl SSTableReader {
15    /// Decode ONE compressed chunk on the IO half (issue #1940, D2): CRC was
16    /// already verified inside the read path, so this does cache-lookup-or-decompress
17    /// and returns the refcounted decompressed `Bytes` substrate shipped on the
18    /// channel — while HANDING BACK the compressed buffer (`compressed`) so the feed
19    /// loop can RECYCLE it as the next read's scratch (no per-chunk compressed-buffer
20    /// allocation). The single decode plane (`chunk_source`) still owns the one
21    /// decompress call. Two cases consume the buffer instead of handing it back
22    /// (returning an empty scratch, so a fresh buffer is minted next read): an
23    /// incompressible-raw chunk (stored uncompressed by Cassandra, compressed len
24    /// ≥ `max_compressed_length`) is passed through zero-copy `Vec`→`Bytes`; and the
25    /// no-compressor case (raw/uncompressed NB scan with no `CompressionInfo` — the
26    /// read buffer already holds finished bytes) is MOVED into the B1 cache zero-copy
27    /// via `Bytes::from(Vec)`, never `to_vec()`-copied (issue #1940 BLOCKER-1;
28    /// uncompressed is CQLite's own write-surface format, a first-class path).
29    /// `chunk_index` is the ABSOLUTE chunk index (the feed reads from data-section
30    /// start in order), matching the prior parse-half keying.
31    pub(super) fn decode_scan_chunk(
32        &self,
33        chunk_index: usize,
34        max_compressed_length: usize,
35        compressed: Vec<u8>,
36    ) -> Result<(bytes::Bytes, Vec<u8>)> {
37        use crate::storage::sstable::compression::Compression;
38
39        // Incompressible-raw chunks (stored uncompressed by Cassandra) skip the
40        // cache and are passed through zero-copy; the buffer is consumed.
41        if compressed.len() >= max_compressed_length {
42            return Ok((bytes::Bytes::from(compressed), Vec::new()));
43        }
44        let key = crate::storage::cache::ChunkKey::new(
45            self.chunk_cache_id ^ super::super::data_access::NS_WINDOWED_CHUNK,
46            chunk_index as u64,
47        );
48        // Cache hit → refcount-bump clone (B1 contract), never a memcpy; the
49        // compressed buffer is recycled unused. Warm scans must take the hit rather
50        // than re-decompress and overwrite (issue #1598 roborev Medium).
51        if let Some(hit) = self.chunk_cache.get(&key) {
52            return Ok((hit, compressed));
53        }
54        // No compressor (raw/uncompressed NB scan, no CompressionInfo): the read
55        // buffer already holds the finished, uncompressed chunk bytes. MOVE it into
56        // the B1 cache as zero-copy `Bytes` (`Bytes::from(Vec)` reuses the read
57        // buffer's heap allocation) rather than `to_vec()`-copying it — this is
58        // CQLite's own uncompressed write-surface format, a first-class path (issue
59        // #1940 BLOCKER-1). Nothing to recycle on the raw path (the buffer became
60        // the cached substrate), so the returned scratch is empty; a fresh read
61        // buffer is minted for the next chunk. No decompress here, so the
62        // decode-thread probe does NOT fire (it pins where decompression runs).
63        if self.compression_reader.is_none() {
64            return Ok((self.chunk_cache.insert(key, compressed), Vec::new()));
65        }
66        // Miss → decompress from the BORROWED slice (so we keep `compressed` to
67        // recycle) into the single decode plane, cache the resident `Bytes`.
68        let compression = self
69            .compression_reader
70            .as_ref()
71            .map(|cr| Compression::new(*cr.algorithm()))
72            .transpose()?;
73        // Runtime-placement guard (issue #1940): record the thread this decode
74        // actually decompresses on — placed at the REAL decompress site (past the
75        // cache-hit / incompressible-raw / no-compressor early exits, none of which
76        // decompress) so a guard test can prove decompression runs on a
77        // spawn_blocking thread, NOT an async worker (the D2 substrate moved
78        // decompression into the IO-half feed loop, which must stay off the reactor
79        // for EVERY backend). Compiled only under `scan-offload-probe`.
80        #[cfg(feature = "scan-offload-probe")]
81        super::probe::record_decode_thread();
82        let comp_info_dummy = crate::storage::sstable::compression_info::CompressionInfo {
83            algorithm: String::new(),
84            option_pairs: vec![],
85            chunk_length: 0,
86            max_compressed_length: max_compressed_length as u32,
87            data_length: 0,
88            chunk_offsets: vec![],
89        };
90        let chunk_source = super::super::chunk_source::ChunkSource::new(
91            self.point_source.as_ref(), // unused by decode_borrowed
92            &comp_info_dummy,
93            compression.as_ref(),
94            &self.chunk_cache,
95            0, // unused
96            0, // unused
97            super::super::data_access::NS_WINDOWED_CHUNK,
98            self.chunk_cache_id,
99        );
100        let decoded = chunk_source.decode_borrowed(key, &compressed)?;
101        Ok((decoded, compressed))
102    }
103}
