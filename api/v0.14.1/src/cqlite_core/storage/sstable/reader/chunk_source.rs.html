1//! Single chunk decode plane: read → CRC → decompress → B1 cache (issue #1598, Epic G).
2//!
3//! The point-read (BTI get, BIG point), windowed-scan, and BIG-reverse decode paths all
4//! resolve their `Compression::decompress` here. The `iterate_all_partitions` and
5//! `sequential_scan` decode sites (`parse_partition_at_offset` in `parsing/mod.rs`,
6//! `parse_block_entries` in `parsing/block_entries.rs`) remain on the legacy
7//! `self.file` + `compression_reader` model (not `ReadAt` + `CompressionInfo` + chunk-index)
8//! and are a scoped follow-up (#2165) to route through ChunkSource. Architecture test:
9//! `tests/chunk_decode_single_plane.rs`.
10
11use crate::storage::cache::{ChunkKey, DecompressedChunkCache};
12use crate::storage::sstable::compression::Compression;
13use crate::storage::sstable::compression_info::CompressionInfo;
14use crate::storage::sstable::reader::block_io::read_compressed_chunk_at;
15use crate::storage::sstable::reader::data_access::DECOMPRESS_CALLS;
16use crate::storage::sstable::reader::read_at::ReadAt;
17use crate::{Error, Result};
18use bytes::Bytes;
19use std::sync::atomic::Ordering;

21/// Single chunk decode plane: positioned read → CRC → decompress → B1 cache.
22///
23/// Composes the C2 read+CRC primitive (`read_compressed_chunk_at`) + the best-of-breed
24/// decompress+cache tail (moved from the BTI block). Every query-path chunk read funnels
25/// through one of these entry points, each of which decompresses ONLY CRC-validated bytes:
26/// - `chunk(index)`: whole-chunk read (self read+CRC) for the BTI/windowed-scan paths
27/// - `decode_and_cache`: shared decompress+cache tail for sites that already have
28///   CRC-validated compressed bytes (windowed-scan blocking half; the compressed
29///   offset-read window of `read_compressed_offset_window`, issue #1773)
30/// - `decompress_only`: uncached decompress for the BIG reverse path and the
31///   compressed offset-read window (both already CRC-validated upstream)
32pub(crate) struct ChunkSource<'a> {
33    /// Positioned read source (C2)
34    source: &'a dyn ReadAt,
35    /// Chunk offsets/sizes/algorithm
36    comp_info: &'a CompressionInfo,
37    /// Decompressor (None => raw passthrough)
38    compression: Option<&'a Compression>,
39    /// B1 decompressed-chunk cache
40    cache: &'a DecompressedChunkCache,
41    /// Data.db file size
42    file_size: u64,
43    /// Header offset (always 0 for NB/BTI)
44    header_offset: u64,
45    /// Cache-key namespace salt (NS_*)
46    namespace: u64,
47    /// Stable reader identity
48    cache_id: u64,
49}
50
51impl<'a> ChunkSource<'a> {
52    /// Construct a ChunkSource for positioned chunk reads.
53    #[allow(clippy::too_many_arguments)]
54    pub(crate) fn new(
55        source: &'a dyn ReadAt,
56        comp_info: &'a CompressionInfo,
57        compression: Option<&'a Compression>,
58        cache: &'a DecompressedChunkCache,
59        file_size: u64,
60        header_offset: u64,
61        namespace: u64,
62        cache_id: u64,
63    ) -> Self {
64        Self {
65            source,
66            comp_info,
67            compression,
68            cache,
69            file_size,
70            header_offset,
71            namespace,
72            cache_id,
73        }
74    }
75
76    /// Whole-chunk read: positioned read → CRC → decompress → B1 cache.
77    ///
78    /// Used by the BTI target-chunk path (self-reading) and can be called by the windowed
79    /// scan if it moves to positioned reads. Returns `Ok(None)` at EOF.
80    pub(crate) fn chunk(&self, index: usize) -> Result<Option<Bytes>> {
81        // Build cache key from absolute chunk index in this namespace
82        let key = ChunkKey::new(self.cache_id ^ self.namespace, index as u64);
83
84        // B1 cache hit: Arc clone, no read, no decompress
85        if let Some(hit) = self.cache.get(&key) {
86            return Ok(Some(hit));
87        }
88
89        // Cache miss: positioned read + CRC (C2 primitive)
90        let compressed = match read_compressed_chunk_at(
91            self.source,
92            self.comp_info,
93            index,
94            self.file_size,
95            self.header_offset,
96        )? {
97            Some(c) => c,
98            None => return Ok(None), // EOF
99        };
100
101        // Incompressible-raw passthrough or decompress, then cache
102        let incompressible = compressed.len() >= self.comp_info.max_compressed_length as usize;
103        Ok(Some(self.decode_and_cache(
104            key,
105            compressed,
106            incompressible,
107        )?))
108    }
109
110    /// Shared decompress+cache tail: decompress (or raw-passthrough) → insert → Arc.
111    ///
112    /// Called by `chunk()` after its own read+CRC, and directly by the windowed-scan
113    /// blocking half (which receives CRC-validated compressed bytes over a channel).
114    /// Together with `decompress_only`, this is the ONLY place on the query path where
115    /// `Compression::decompress` is called — the architecture test proves it.
116    pub(crate) fn decode_and_cache(
117        &self,
118        key: ChunkKey,
119        compressed: Vec<u8>,
120        incompressible: bool,
121    ) -> Result<Bytes> {
122        let decompressed = if incompressible {
123            // Stored uncompressed by Cassandra: pass raw bytes through (no decompress counter)
124            compressed
125        } else if let Some(compression) = self.compression {
126            // Decompress: the single query-path decompress call site
127            let d = compression.decompress(&compressed).map_err(|e| {
128                Error::corruption(format!(
129                    "ChunkSource: failed to decompress chunk (key={:?}): {}",
130                    key, e
131                ))
132            })?;
133            DECOMPRESS_CALLS.fetch_add(1, Ordering::Relaxed);
134            // CHUNK_PATH_ALLOCS (consumer E3/#1940): the decompress OUTPUT buffer is
135            // a per-chunk copy-chain heap allocation. It is the ONE surviving
136            // allocation after the D2 substrate work — it flows zero-copy into the
137            // B1 cache as `Bytes` (no `Arc::from` re-copy) and is the refcounted
138            // substrate the window borrows. Recording it here (not the old
139            // compressed-read-buffer site, which D2 turned into a reused scratch)
140            // makes the ≤1-alloc/chunk bound measurable. No-op in release.
141            crate::storage::sstable::read_work_counters::record_chunk_path_alloc();
142            d
143        } else {
144            // No compression reader: treat raw bytes as decompressed
145            compressed
146        };
147
148        // Insert into B1 cache (Vec→Arc conversion happens once here) and return
149        Ok(self.cache.insert(key, decompressed))
150    }
151
152    /// Decompress a CRC-validated COMPRESSIBLE chunk from a BORROWED slice, cache
153    /// it, and return the resident [`Bytes`] — leaving the compressed input buffer
154    /// owned by the caller so it can be RECYCLED as a per-loop scratch (issue #1940,
155    /// D2). This is the windowed-scan IO half's decode entry: unlike
156    /// [`decode_and_cache`](Self::decode_and_cache) (which takes the compressed `Vec`
157    /// by value and drops it), this borrows `&compressed`, so the caller keeps the
158    /// compressed buffer and reuses it for the next chunk read — the compressed-read
159    /// side then performs no per-chunk allocation, and the ONE surviving copy-chain
160    /// allocation is the decompress output (recorded here). Callers handle the
161    /// incompressible-raw and no-compressor cases directly (they move the buffer);
162    /// this method is only for the decompress path (`self.compression` is `Some`).
163    /// A `None` compressor is treated as a raw passthrough of a COPY (rare/never on
164    /// the windowed path, where compression is always resolved).
165    pub(crate) fn decode_borrowed(&self, key: ChunkKey, compressed: &[u8]) -> Result<Bytes> {
166        let decompressed = if let Some(compression) = self.compression {
167            let d = compression.decompress(compressed).map_err(|e| {
168                Error::corruption(format!(
169                    "ChunkSource: failed to decompress chunk (key={:?}): {}",
170                    key, e
171                ))
172            })?;
173            DECOMPRESS_CALLS.fetch_add(1, Ordering::Relaxed);
174            // CHUNK_PATH_ALLOCS (consumer E3/#1940): the decompress OUTPUT buffer is
175            // the ONE surviving per-chunk copy-chain allocation after D2 — it flows
176            // zero-copy into the B1 cache as `Bytes` (no `Arc::from` re-copy) and is
177            // the refcounted substrate the window borrows. No-op in release.
178            crate::storage::sstable::read_work_counters::record_chunk_path_alloc();
179            d
180        } else {
181            compressed.to_vec()
182        };
183        Ok(self.cache.insert(key, decompressed))
184    }
185
186    /// Decompress-only helper for the BIG reverse path: decompress without caching.
187    ///
188    /// Preserves the current uncached behavior of `pull_reverse_chunk` — no B1 cache
189    /// insertion, no DECOMPRESS_CALLS counter (separate work_counters::add_chunk_decompressed).
190    /// This exists ONLY to consolidate the `Compression::decompress` call site into this
191    /// module while changing zero runtime behavior on the reverse path.
192    pub(crate) fn decompress_only(
193        compression: Option<&Compression>,
194        compressed: Vec<u8>,
195    ) -> Result<Vec<u8>> {
196        if let Some(c) = compression {
197            c.decompress(&compressed).map_err(|e| {
198                Error::corruption(format!(
199                    "ChunkSource: reverse-path decompress failed: {}",
200                    e
201                ))
202            })
203        } else {
204            Ok(compressed)
205        }
206    }
207}
