191/// Best-effort total physical RAM in bytes, or `None` when it cannot be
192/// determined on this platform. Used by [`DiskAccessMode::Auto`] to decide when
193/// a file is large enough to warrant page-cache-bypassing direct I/O.
194fn system_memory_bytes() -> Option<u64> {
195    #[cfg(unix)]
196    {
197        // SAFETY: `sysconf` is a pure query with no pointer arguments.
198        let pages = unsafe { libc::sysconf(libc::_SC_PHYS_PAGES) };
199        let page_size = unsafe { libc::sysconf(libc::_SC_PAGESIZE) };
200        if pages > 0 && page_size > 0 {
201            return Some((pages as u64).saturating_mul(page_size as u64));
202        }
203        None
204    }
205    #[cfg(not(unix))]
206    {
207        None
208    }
209}

211/// Decide which disk-access backend to use for a Data.db file.
212///
213/// Pure function (system memory injected) so the [`DiskAccessMode::Auto`]
214/// heuristic can be unit-tested deterministically. Resolution rules:
215/// - explicit `Buffered`/`Mmap`/`Direct` are returned unchanged (the caller
216///   applies graceful fallback if the OS refuses the backend);
217/// - `Auto` returns `Buffered` for files below `mmap_min_size_bytes`, `Direct`
218///   when the file exceeds `memory_fraction` of `system_memory` (and memory is
219///   known and direct I/O is available on this platform), otherwise `Mmap`.
220///
221/// The deprecated `use_mmap` flag / `CQLITE_USE_MMAP` env is folded in by the
222/// caller (it promotes a `Buffered` request to `Mmap`), so it is not an input
223/// here; an explicit mode always takes precedence.
224fn resolve_disk_access_mode(
225    configured: DiskAccessMode,
226    file_size: u64,
227    mmap_min_size_bytes: u64,
228    memory_fraction: f64,
229    system_memory: Option<u64>,
230    direct_io_available: bool,
231) -> DiskAccessMode {
232    // Zero-length files cannot be mapped and have nothing to read directly;
233    // always use buffered I/O for them regardless of the requested mode.
234    if file_size == 0 {
235        return DiskAccessMode::Buffered;
236    }
237    match configured {
238        DiskAccessMode::Buffered => DiskAccessMode::Buffered,
239        DiskAccessMode::Mmap => DiskAccessMode::Mmap,
240        DiskAccessMode::Direct => DiskAccessMode::Direct,
241        DiskAccessMode::Auto => {
242            if file_size < mmap_min_size_bytes {
243                return DiskAccessMode::Buffered;
244            }
245            let fraction = if memory_fraction.is_finite() && memory_fraction > 0.0 {
246                memory_fraction.min(1.0)
247            } else {
248                0.5
249            };
250            if direct_io_available {
251                if let Some(mem) = system_memory {
252                    let threshold = (mem as f64 * fraction) as u64;
253                    if file_size > threshold {
254                        return DiskAccessMode::Direct;
255                    }
256                }
257            }
258            DiskAccessMode::Mmap
259        }
260    }
261}

275/// Resolve a [`PrefetchMode`] into the concrete advice the mmap backend should
276/// issue, or `None` for "no advice".
277///
278/// `memmap2::Advice` / `Mmap::advise` (madvise) are Unix-only, so this and its
279/// single call site are gated to `#[cfg(unix)]`. On non-Unix targets the mmap
280/// backend simply issues no read-ahead advice.
281///
282/// [`PrefetchMode::Auto`] deliberately issues **no** madvise (issue #1143).
283/// `MADV_SEQUENTIAL` couples aggressive read-ahead with *drop-behind*: pages are
284/// evicted from the page cache as soon as the scan moves past them. In isolation
285/// that is fine (mmap scans are ~40% faster than buffered), but under concurrent
286/// write load the page-cache pressure means the just-dropped pages are gone by
287/// the time an overlapping scan needs them again, so re-reads take *synchronous*
288/// major page faults on the tokio worker thread and the read-side p99 tail blows
289/// up (~2x regression). Relying on the kernel's default read-ahead (no
290/// drop-behind) keeps the isolated mmap win while letting the page cache retain
291/// hot pages, which collapses that tail. Callers who genuinely want the
292/// drop-behind behaviour can still request `Sequential` explicitly
293/// (`CQLITE_PREFETCH=sequential` / [`StorageConfig::prefetch`]).
294#[cfg(unix)]
295fn mmap_advice_for(prefetch: PrefetchMode) -> Option<memmap2::Advice> {
296    match prefetch {
297        // No madvise: rely on the kernel's default read-ahead. Chosen for `Auto`
298        // to avoid `MADV_SEQUENTIAL` drop-behind evicting hot pages under
299        // concurrent write load (issue #1143).
300        PrefetchMode::Off | PrefetchMode::Auto => None,
301        // Explicit opt-in to aggressive read-ahead + drop-behind. Best for a
302        // one-shot full scan that will not be re-read and should not pin the
303        // whole file in the page cache.
304        PrefetchMode::Sequential => Some(memmap2::Advice::Sequential),
305        PrefetchMode::WillNeed => Some(memmap2::Advice::WillNeed),
306    }
307}

309/// Process-wide count of currently-open [`SSTableReader`]s, the source value
310/// for the [`SSTABLES_OPEN`](crate::observability::catalog::SSTABLES_OPEN)
311/// gauge. Tracked PER FORMAT so the gauge — which carries the
312/// [`cqlite.sstable.format`](crate::observability::catalog::attr::SSTABLE_FORMAT)
313/// attribute — reports the correct live count for each `big`/`bti` label series
314/// instead of a single global total stamped onto every series. Incremented when
315/// [`SSTableReader::open`] succeeds and decremented when a reader is dropped, so
316/// each gauge series reflects live readers regardless of the `observability`
317/// feature state (the helper calls are no-ops when off).
318static SSTABLES_OPEN_COUNT_BIG: AtomicI64 = AtomicI64::new(0);
319static SSTABLES_OPEN_COUNT_BTI: AtomicI64 = AtomicI64::new(0);
320
321/// Select the per-format open-reader counter for a `sstable_format_label()`
322/// value (`"big"` / `"bti"`). Defaults to the BIG counter for any unexpected
323/// label so the value is never silently dropped; the label set is bounded to
324/// the two [`sstable_format_label`](SSTableReader::sstable_format_label) returns.
325fn sstables_open_count_for(format: &str) -> &'static AtomicI64 {
326    match format {
327        "bti" => &SSTABLES_OPEN_COUNT_BTI,
328        _ => &SSTABLES_OPEN_COUNT_BIG,
329    }
330}

333    /// Open an SSTable file for reading.
334    ///
335    /// Instrumented (epic #1031 / #1034): wraps the open in a
336    /// `sstable.reader.open` span, increments the [`SSTABLES_OPEN`] gauge on
337    /// success, and records an error on the `reader` subsystem when open fails.
338    ///
339    /// [`SSTABLES_OPEN`]: crate::observability::catalog::SSTABLES_OPEN
340    pub async fn open(path: &Path, config: &Config, platform: Arc<Platform>) -> Result<Self> {
341        // Back-compat: existing callers and sibling crates get a FRESH per-reader
342        // decompressed-chunk cache sized from config (issue #1567). Production
343        // reads route through `SSTableManager`, which calls `open_with_cache` with
344        // its SHARED instance so all readers of a dataset share one cache.
345        //
346        // Honor the `config.memory.block_cache.enabled` toggle (issue #1568): when
347        // disabled this yields a genuine no-op cache so the direct reader path
348        // bypasses caching identically to the manager path, instead of the toggle
349        // being decorative here. Reuses the shared `build_chunk_cache` helper.
350        let cache = super::build_chunk_cache(config);
351        Self::open_with_cache(path, config, platform, cache).await
352    }

370    }

394    /// Cancel-aware [`open_with_cache`](Self::open_with_cache) (issue #2383 fix C):
395    /// same as it but threads `cancel` into the Index.db parse. See
396    /// [`open_cancellable`](Self::open_cancellable).
397    pub async fn open_with_cache_cancellable(
398        path: &Path,
399        config: &Config,
400        platform: Arc<Platform>,
401        cache: Arc<crate::storage::cache::DecompressedChunkCache>,
402        cancel: crate::storage::scan_cancel::ScanCancel,
403    ) -> Result<Self> {
404        use crate::observability::{self as obs, catalog};
405        use tracing::Instrument as _;
406
407        let span = tracing::debug_span!(
408            "sstable.reader.open",
409            file_size = tracing::field::Empty,
410            sstable_format = tracing::field::Empty,
411        );
412
413        // Instrument the future rather than holding an entered guard across the
414        // `.await`: entering a span guard and then awaiting can attach unrelated
415        // async work scheduled on this task to the span. `Instrument` enters the
416        // span only while this specific future is polled.
417        let result = Self::open_inner(path, config, platform, cache, cancel)
418            .instrument(span.clone())
419            .await;
420        match &result {
421            Ok(reader) => {
422                let format = reader.sstable_format_label();
423                span.record("file_size", reader.stats.file_size);
424                span.record("sstable_format", format);
425                // SSTABLES_OPEN is a snapshot gauge of the live reader count;
426                // record the current PER-FORMAT count after this open succeeds so
427                // the format-attributed gauge series stays correct under mixed
428                // BIG/BTI readers.
429                let now = sstables_open_count_for(format).fetch_add(1, Ordering::Relaxed) + 1;
430                obs::record_gauge(
431                    catalog::SSTABLES_OPEN,
432                    now,
433                    &[(catalog::attr::SSTABLE_FORMAT, format.into())],
434                );
435            }
436            Err(e) => {
437                // Record the error WHILE the open span is current so
438                // `mark_span_error` marks THIS `sstable.reader.open` span. The
439                // instrumented future has already completed here, so the span is
440                // no longer entered; `in_scope` re-enters it for the duration of
441                // the error-recording call.
442                span.in_scope(|| obs::record_error(e, "reader"));
443            }
444        }
445        result
446    }

448    /// Open implementation; see [`open`](Self::open) for the instrumented wrapper.
449    async fn open_inner(
450        path: &Path,
451        config: &Config,
452        platform: Arc<Platform>,
453        chunk_cache: Arc<crate::storage::cache::DecompressedChunkCache>,
454        cancel: crate::storage::scan_cancel::ScanCancel,
455    ) -> Result<Self> {
456        // Retain the open-time Config before any local `config` shadowing so
457        // `perform_integrity_check` can delegate to `verify::verify_sstable`
458        // (single source of truth, issue #1283) under the same config.
459        let open_config = config.clone();
460        // #1249 (spec R1): reject below-floor versions BEFORE any file I/O.
461        // Gates derive solely from the filename, so this is the earliest point
462        // that can enforce the na+ floor — a pre-`na` (BIG) or non-`da` (BTI)
463        // descriptor is rejected with a typed `UnsupportedVersion` before we
464        // open/mmap/read the body (it never surfaces an I/O/parse error). Only a
465        // structurally-unparseable descriptor falls back to nb-compatible BIG
466        // gates (preserving existing tolerance for odd-but-5.0 unit-test paths);
467        // the gates do not change parsing decisions until VG3 flips behaviour.
468        let version_gates = Arc::new(match VersionGates::from_path(path) {
469            Ok(gates) => gates,
470            // A parsed-but-below-floor version is FATAL — never degrade it.
471            Err(e @ Error::UnsupportedVersion { .. }) => return Err(e),
472            Err(e) => {
473                tracing::debug!(
474                    "SSTableReader::open: could not derive VersionGates from {:?} ({}); \
475                     defaulting to nb-compatible BIG gates",
476                    path,
477                    e
478                );
479                VersionGates::Big(BigVersionGates::nb_fallback())
480            }
481        });
482
483        // Resolve the disk-access backend (buffered / mmap / direct, or auto)
484        // from config + environment overrides. See `Config::storage.disk_access_mode`.
485        let mut reader_config = SSTableReaderConfig::default();
486        reader_config.use_mmap = config.storage.use_mmap || mmap_enabled_via_env();
487        reader_config.mmap_min_size_bytes = config.storage.mmap_min_size_bytes;
488
489        let file_size = tokio::fs::metadata(path).await?.len();
490
491        // The explicit mode comes from env > config. The legacy `use_mmap` flag
492        // is folded in by promoting an otherwise-`Buffered` request to `Mmap` so
493        // the old opt-in keeps working (`Auto` and explicit non-buffered modes
494        // are left untouched, so a real backend decision always wins).
495        let configured_mode = disk_access_mode_via_env().unwrap_or(config.storage.disk_access_mode);
496        let configured_mode =
497            if reader_config.use_mmap && matches!(configured_mode, DiskAccessMode::Buffered) {
498                DiskAccessMode::Mmap
499            } else {
500                configured_mode
501            };
502
503        let resolved_mode = resolve_disk_access_mode(
504            configured_mode,
505            file_size,
506            reader_config.mmap_min_size_bytes as u64,
507            config.storage.direct_io_memory_fraction,
508            system_memory_bytes(),
509            direct_io_available(),
510        );
511
512        // Build both the shared point-read source and the per-scan factory from
513        // the same backend decision so concurrent scans get independent cursors
514        // (issue #815). `ScanSource::Mapped` shares the same `Arc<Mmap>` so no
515        // extra mapping is created per scan. Every non-buffered backend degrades
516        // gracefully to buffered I/O if the OS/filesystem refuses it.
517        let prefetch = prefetch_mode_via_env().unwrap_or(config.storage.prefetch);
518        let (source, scan_source) = Self::build_block_sources(
519            path,
520            file_size,
521            resolved_mode,
522            prefetch,
523            config.storage.direct_io_prefetch_bytes,
524        )
525        .await?;
526        let file = Arc::new(Mutex::new(source));
527
528        // Build the POINT-READ positional source ONCE at open (issue #1573, C2).
529        // It shares the reader's `Arc<Mmap>` when the backend is mmap (no extra
530        // mapping / fd); for buffered/direct it holds one dedicated read-only fd,
531        // which is exactly the "open the fd once, positioned-read thereafter"
532        // contract that removes the BTI per-lookup `open(2)`. Every non-mmap
533        // backend degrades gracefully to a plain positioned fd if the faster
534        // backend is refused, mirroring `build_block_sources`.
535        let point_source: Arc<dyn read_at::ReadAt> = match &scan_source {
536            ScanSource::Mapped(mmap) => Arc::new(read_at::MmapReadAt::new(mmap.clone())),
537            #[cfg(unix)]
538            ScanSource::Direct { .. } => match read_at::DirectReadAt::open(path, file_size) {
539                Ok(d) => Arc::new(d) as Arc<dyn read_at::ReadAt>,
540                Err(e) => {
541                    tracing::warn!(
542                        "Direct-I/O point source for {} failed ({}); using buffered pread",
543                        path.display(),
544                        e
545                    );
546                    Arc::new(read_at::PlainFileReadAt::open(path, file_size)?)
547                }
548            },
549            ScanSource::Buffered { .. } => {
550                Arc::new(read_at::PlainFileReadAt::open(path, file_size)?)
551            }
552        };
553
554        // Parse header - read available bytes, not a fixed size
555        // NOTE: For NB format files (Cassandra 4.x+), Data.db often contains compressed row data
556        // with no embedded header. The header.rs module detects this via filename pattern and
557        // returns a minimal header loaded from Statistics.db instead.
558        let header_size = std::cmp::min(4096, file_size as usize);
559        let mut header_buffer = vec![0u8; header_size];
560        {
561            let mut file_guard = file.lock().await;
562            let bytes_read = file_guard.read(&mut header_buffer).await?;
563            header_buffer.truncate(bytes_read);
564        }
565
566        // VG5 / Issue #831 / #909: BTI ("da") read support.
567        //
568        // BTI SSTables use a Partitions.db trie + Rows.db row index instead of
569        // Index.db/Summary.db. We load BOTH (tiny) tries fully into memory here so
570        // the point-lookup path (`lookup_partition_via_bti_trie` /
571        // `bti_point_lookup`) can walk them for O(log n) partition resolution:
572        //   - Partitions.db resolves a partition key to either a direct Data.db
573        //     offset (NARROW partition) or a positive RowsOffset (WIDE partition).
574        //   - Rows.db, indexed by that RowsOffset, recovers the wide partition's
575        //     Data.db position (issue #909/#910).
576        //
577        // Loading Partitions.db + Rows.db is the ONLY BTI-specific step in open():
578        // the rest of the flow (header / compression / Statistics-driven schema)
579        // tolerates the absent Index.db/Summary.db gracefully (those loaders
580        // return None).
581        let (bti_partitions_db, bti_rows_db) = if matches!(*version_gates, VersionGates::Bti(_)) {
582            let base = extract_sstable_base_name(path).ok_or_else(|| {
583                Error::unsupported_format(format!(
584                    "BTI (da) SSTable '{}' has a non-standard filename; cannot derive the \
585                         sibling Partitions.db name required for trie point lookup (#831).",
586                    path.display()
587                ))
588            })?;
589            let parent = path.parent().unwrap_or_else(|| Path::new("."));
590            let partitions_path = parent.join(format!("{}-Partitions.db", base));
591            let partitions_bytes = tokio::fs::read(&partitions_path).await.map_err(|e| {
592                Error::unsupported_format(format!(
593                    "BTI (da) SSTable '{}' is missing its sibling Partitions.db trie \
594                         (expected '{}'): {}. BTI read support requires Partitions.db for \
595                         partition-key point lookup (#831).",
596                    path.display(),
597                    partitions_path.display(),
598                    e
599                ))
600            })?;
601
602            // Rows.db carries the within-partition row index for WIDE
603            // partitions (issue #909/#910). It is ALWAYS emitted for a BTI
604            // SSTable (possibly 0 bytes for a narrow-only table), so a missing
605            // Rows.db is a structural error. A 0-byte file is valid: no
606            // partition resolved to a positive RowsOffset, so the point-lookup
607            // path never indexes into it.
608            let rows_path = parent.join(format!("{}-Rows.db", base));
609            let rows_bytes = tokio::fs::read(&rows_path).await.map_err(|e| {
610                Error::unsupported_format(format!(
611                    "BTI (da) SSTable '{}' is missing its sibling Rows.db row-index trie \
612                         (expected '{}'): {}. BTI read support requires Rows.db to resolve \
613                         wide-partition point lookups (#909/#910).",
614                    path.display(),
615                    rows_path.display(),
616                    e
617                ))
618            })?;
619
620            (Some(Arc::new(partitions_bytes)), Some(Arc::new(rows_bytes)))
621        } else {
622            let none: Option<Arc<Vec<u8>>> = None;
623            (none.clone(), none)
624        };
625
626        use tracing::Instrument;
627
628        let config = crate::cql::config::ParserConfig::default();
629        let parser = SSTableParser::new(config)?;
630        // Parse the header using enhanced version detection - strict error propagation.
631        // VersionGates are passed so VG3 can flip version-sensitive parsing decisions
632        // inside header parsing without re-deriving gates from the filename.
633        let header = parse_header_with_version_detection(&header_buffer, path, &version_gates)
634            .instrument(tracing::debug_span!("sstable.reader.open.header_parse"))
635            .await
636            .map_err(|e| {
637                Error::corruption(format!(
638                    "Failed to parse SSTable header for file '{}': {}. This indicates either \
639                     file corruption or an unsupported SSTable format. File size: {} bytes, \
640                     header buffer size: {} bytes.",
641                    path.display(),
642                    e,
643                    file_size,
644                    header_buffer.len()
645                ))
646            })?;
647        let header_size = calculate_actual_header_size(&header, &header_buffer)?;
648
649        // Schema extraction deferred until after Statistics.db columns are loaded (Issue #163)
650        // See schema extraction code after statistics_reader loading below
651
652        // Seek to start of data section
653        {
654            let mut file_guard = file.lock().await;
655            file_guard
656                .seek(std::io::SeekFrom::Start(header_size as u64))
657                .await?;
658        }
659
660        // Load CompressionInfo.db ONCE for chunked decompression (if it exists).
661        // This is the single authoritative parse per open (issue #1597 / G1); the
662        // component name is derived deterministically from `SsTableDescriptor`
663        // (one `exists()` probe, not the legacy ~25-generation scan).
664        let compression_info = Self::load_compression_info_metadata(path, &platform).await?;
665
666        // Derive the compression reader (algorithm only) from that single parsed
667        // result — no second parse. `algorithm_enum()` is fallible only for a name
668        // `parse()` would already have rejected, so no `unwrap`/`expect` is needed.
669        let compression_reader = match &compression_info {
670            Some(info) => Some(CompressionReader::new(info.algorithm_enum()?)),
671            None => None,
672        };
673
674        // Load CRC.db per-chunk checksums for uncompressed BIG read-time integrity
675        // (issue #1396). Only for uncompressed tables (compression_info None);
676        // compressed tables carry inline per-chunk CRCs instead.
677        let crc_reader = if compression_info.is_none() {
678            Self::load_crc_reader(path, &header, file_size).await?
679        } else {
680            None
681        };
682
683        // Pre-validate component architecture for better error handling
684        let components = Self::detect_component_files(path).await?;
685        if !components.is_empty() {
686            let integrity_issues = Self::validate_component_integrity(path, &components).await?;
687            if !integrity_issues.is_empty() {
688                tracing::warn!(
689                    "Component integrity issues detected but proceeding with loading: {:?}",
690                    integrity_issues
691                );
692            }
693        }
694
695        // Integrated in-Data.db index only; separate Index.db is parsed once by load_index_reader (#2385/#2395).
696        let index = Self::load_index(&file, &header)
697            .instrument(tracing::debug_span!("sstable.reader.open.load_index"))
698            .await?;
699
700        // Load bloom filter if available (supports both integrated and component-based)
701        let bloom_filter = Self::load_bloom_filter(&file, &header, &platform, path)
702            .instrument(tracing::debug_span!("sstable.reader.open.load_bloom"))
703            .await?;
704
705        // Load spec readers for enhanced metadata and lookups. Distinguish an absent
706        // Index.db from a present-but-unloadable one (issue #2302) so the full
707        // enumeration can WARN loud on the latter instead of silently full-scanning.
708        let (index_reader, index_present_but_unloadable) =
709            match Self::load_index_reader(path, &platform, &cancel)
710                .instrument(tracing::debug_span!(
711                    "sstable.reader.open.load_index_reader"
712                ))
713                .await?
714            {
715                component_loading::IndexLoadOutcome::Loaded(reader) => (Some(*reader), false),
716                component_loading::IndexLoadOutcome::Absent => (None, false),
717                component_loading::IndexLoadOutcome::PresentButUnloadable => (None, true),
718            };
719        let summary_reader = Self::load_summary_reader(path, &platform)
720            .instrument(tracing::debug_span!("sstable.reader.open.load_summary"))
721            .await;
722        let statistics_reader = Self::load_statistics_reader(path, &platform)
723            .instrument(tracing::debug_span!("sstable.reader.open.load_statistics"))
724            .await?;
725
726        // Extract SerializationHeader columns from Statistics.db (Issue #163)
727        // This enables schema extraction for V5CompressedLegacy format
728        let mut header = header; // Make mutable to populate columns
729        if let Some(ref stats_reader) = statistics_reader {
730            let statistics = stats_reader.statistics();
731            let partition_columns = &statistics.serialization_header_partition_keys;
732            let clustering_columns = &statistics.serialization_header_clustering_keys;
733            let regular_columns = &statistics.serialization_header_columns;
734
735            if !partition_columns.is_empty()
736                || !clustering_columns.is_empty()
737                || !regular_columns.is_empty()
738            {
739                tracing::debug!(
740                    "Populating header columns from Statistics.db SerializationHeader: {} partition keys, {} clustering keys, {} regular columns",
741                    partition_columns.len(),
742                    clustering_columns.len(),
743                    regular_columns.len()
744                );
745
746                let mut merged_columns = Vec::with_capacity(
747                    partition_columns.len() + clustering_columns.len() + regular_columns.len(),
748                );
749                merged_columns.extend_from_slice(partition_columns);
750                merged_columns.extend_from_slice(clustering_columns);
751                merged_columns.extend_from_slice(regular_columns);
752
753                header.columns = merged_columns;
754            }
755        }
756
757        // Extract schema from header for V5.0+ formats (after columns are populated)
758        let schema = if matches!(
759            header.cassandra_version,
760            CassandraVersion::V5_0NewBig
761                | CassandraVersion::V5_0Bti
762                | CassandraVersion::V5_0DataFormat
763                | CassandraVersion::V5_0FormatC
764                | CassandraVersion::V5_0FormatD
765                | CassandraVersion::V5_0FormatE
766                | CassandraVersion::V5_0FormatF
767                | CassandraVersion::V5_0FormatG
768        ) {
769            match TableSchema::from_sstable_header(&header) {
770                Ok(s) => {
771                    tracing::debug!(
772                        "Extracted schema from SSTable header: {}.{} ({} columns, {} partition keys, {} clustering keys)",
773                        s.keyspace,
774                        s.table,
775                        s.columns.len(),
776                        s.partition_keys.len(),
777                        s.clustering_keys.len()
778                    );
779                    Some(Arc::new(s))
780                }
781                Err(e) => {
782                    tracing::warn!(
783                        "Failed to extract schema from SSTable header for {}: {}. Schema-aware parsing will not be available.",
784                        path.display(),
785                        e
786                    );
787                    None
788                }
789            }
790        } else {
791            // Legacy formats don't have schema in header
792            None
793        };
794
795        // Derive block_count from CompressionInfo.db when available — this is the
796        // authoritative source for compressed SSTables (no-heuristics mandate #28).
797        // Each entry in chunk_offsets corresponds to one compressed block in Data.db.
798        let block_count = compression_info
799            .as_ref()
800            .map(|ci| ci.chunk_offsets.len() as u64)
801            .unwrap_or(0);
802
803        let stats = SSTableReaderStats {
804            file_size,
805            entry_count: header.stats.row_count,
806            table_count: 1, // Will be updated as we discover tables
807            block_count,
808            index_size: 0,        // Will be updated if index is loaded
809            bloom_filter_size: 0, // Will be updated if bloom filter is loaded
810            compression_ratio: header.stats.compression_ratio,
811            cache_hit_rate: 0.0,
812        };
813
814        // Extract generation from filename or use default
815        let generation = extract_generation_from_path(path);
816
817        // Stable per-reader cache identity (issue #1567): hash the immutable
818        // file path + generation. Authoritative reader identity, never byte
819        // content — combined with a per-site salt to form the cache key.
820        let chunk_cache_id = {
821            use std::hash::{Hash, Hasher};
822            let mut h = std::collections::hash_map::DefaultHasher::new();
823            path.hash(&mut h);
824            generation.hash(&mut h);
825            h.finish()
826        };
827
828        // Per-reader key→partition-offset cache (issue #1570, B4), built from the
829        // open-time config BEFORE `open_config` is moved into the struct field.
830        let key_offset_cache = super::build_key_offset_cache(&open_config);
831
832        // Cache the immutable `[first_key, last_key]` endpoint tokens ONCE at open
833        // (issue #1576, Epic C/C5 perf finding) so `partition_key_out_of_range`
834        // only hashes the QUERY key on the hot point-read path instead of
835        // re-hashing the two fixed endpoints on every read. Armed only when
836        // Summary.db is present with two non-empty endpoints — the exact condition
837        // the short-circuit itself requires. Uses the same Cassandra Murmur3 token
838        // as the hot path, so the cached values are byte-identical.
839        let endpoint_tokens = summary_reader.as_ref().and_then(|s| {
840            let first = s.get_first_key();
841            let last = s.get_last_key();
842            (!first.is_empty() && !last.is_empty()).then(|| {
843                (
844                    crate::util::cassandra_murmur3::cassandra_murmur3_token(first),
845                    crate::util::cassandra_murmur3::cassandra_murmur3_token(last),
846                )
847            })
848        });
849
850        Ok(Self {
851            file_path: arc_swap::ArcSwap::from_pointee(path.to_path_buf()),
852            file,
853            scan_source,
854            point_source,
855            header,
856            parser,
857            index,
858            bloom_filter,
859            compression_reader,
860            config: reader_config,
861            open_config,
862            platform,
863            stats,
864            #[cfg(feature = "tombstones")]
865            tombstone_merger: TombstoneMerger::new(),
866            generation,
867            actual_header_size: header_size,
868            index_reader,
869            index_present_but_unloadable,
870            summary_reader,
871            endpoint_tokens,
872            statistics_reader,
873            schema_registry: None, // Will be set by set_schema_registry() after construction
874            schema,
875            #[cfg(feature = "state_machine")]
876            registry_schema: None, // Resolved by resolve_registry_schema() at wiring time (#1692)
877            udt_registry: None, // Will be set when available for UDT-aware parsing
878            scan_cancel: crate::storage::scan_cancel::ScanCancel::default(), // #2264: set via set_scan_cancel
879            compression_info: compression_info.map(Arc::new),
880            crc_reader,
881            verified_uncompressed_chunks: std::sync::Mutex::new(std::collections::HashSet::new()),
882            version_gates,
883            bti_partitions_db,
884            bti_rows_db,
885            chunk_cache,
886            chunk_cache_id,
887            bti_partition_offsets: std::sync::OnceLock::new(),
888            bti_lookup_memo: std::sync::Mutex::new(None),
889            key_offset_cache,
890        })
891    }

927    /// Open a buffered point-read source and its per-scan factory. Shared by the
928    /// buffered backend and as the graceful fallback for mmap/direct.
929    async fn open_buffered_sources(
930        path: &Path,
931        file_size: u64,
932    ) -> Result<(BlockSource, ScanSource)> {
933        // A5 read-work counter (FILE_OPENS; consumer C2): one per open(2) that mints
934        // a reader fd — here the buffered cold-open / graceful-fallback site. No-op
935        // in release (design.md Decision 1/2).
936        crate::storage::sstable::read_work_counters::record_file_open();
937        Ok((
938            BlockSource::buffered_sized(File::open(path).await?, file_size),
939            ScanSource::Buffered {
940                file_len: file_size,
941            },
942        ))
943    }

945    /// Build the shared point-read [`BlockSource`] and the per-scan
946    /// [`ScanSource`] for a resolved [`DiskAccessMode`].
947    ///
948    /// `prefetch` selects read-ahead advice for mmap and (together with
949    /// `prefetch_bytes`) the direct-I/O read-ahead window. Each non-buffered
950    /// backend degrades gracefully to buffered I/O if the OS or filesystem
951    /// refuses it (mmap can fail on network mounts; direct I/O is unsupported on
952    /// some platforms/filesystems), so opening never fails purely because a
953    /// faster backend is unavailable.
954    async fn build_block_sources(
955        path: &Path,
956        file_size: u64,
957        mode: DiskAccessMode,
958        prefetch: PrefetchMode,
959        prefetch_bytes: usize,
960    ) -> Result<(BlockSource, ScanSource)> {
961        // With prefetch off, use the minimal aligned read-ahead the direct
962        // backend still requires (a single block, rounded up inside the cursor).
963        // Otherwise clamp the configured window so a sub-chunk
964        // `direct_io_prefetch_bytes` cannot make one compression chunk straddle
965        // many aligned refills (issue #1596, F6). The actual per-SSTable chunk
966        // length is not parsed yet at this open-time construction site, so the
967        // clamp floors against Cassandra's default 64 KiB chunk — which fully
968        // protects the default case (and the 1 MiB default window is already
969        // above the floor, so the clamp is a no-op there).
970        let direct_window = if matches!(prefetch, PrefetchMode::Off) {
971            1
972        } else {
973            prefetch_window::clamp_direct_prefetch_window(
974                prefetch_bytes,
975                prefetch_window::DEFAULT_COMPRESSION_CHUNK_BYTES,
976            )
977        };
978        match mode {
979            DiskAccessMode::Buffered => Self::open_buffered_sources(path, file_size).await,
980            DiskAccessMode::Mmap => match Self::map_file(path) {
981                Ok(mmap) => {
982                    tracing::debug!(
983                        "Opened {} via memory map ({} bytes)",
984                        path.display(),
985                        file_size
986                    );
987                    // Best-effort read-ahead advice (Unix-only; madvise has no
988                    // Windows equivalent here). Failure is non-fatal.
989                    #[cfg(unix)]
990                    if let Some(advice) = mmap_advice_for(prefetch) {
991                        if let Err(e) = mmap.advise(advice) {
992                            tracing::debug!(
993                                "madvise({:?}) on {} failed: {}",
994                                advice,
995                                path.display(),
996                                e
997                            );
998                        }
999                    }
1000                    let mmap = Arc::new(mmap);
1001                    Ok((BlockSource::mapped(mmap.clone()), ScanSource::Mapped(mmap)))
1002                }
1003                Err(e) => {
1004                    tracing::warn!(
1005                        "Memory-mapping {} failed ({}); falling back to buffered I/O",
1006                        path.display(),
1007                        e
1008                    );
1009                    Self::open_buffered_sources(path, file_size).await
1010                }
1011            },
1012            DiskAccessMode::Direct => {
1013                #[cfg(unix)]
1014                {
1015                    match source::DirectCursor::open(path, direct_window) {
1016                        Ok(cursor) => {
1017                            tracing::debug!(
1018                                "Opened {} via direct I/O ({} bytes, {}-byte window)",
1019                                path.display(),
1020                                file_size,
1021                                direct_window
1022                            );
1023                            Ok((
1024                                BlockSource::direct(cursor),
1025                                ScanSource::Direct {
1026                                    window: direct_window,
1027                                    file_len: file_size,
1028                                },
1029                            ))
1030                        }
1031                        Err(e) => {
1032                            tracing::warn!(
1033                                "Direct I/O on {} failed ({}); falling back to buffered I/O",
1034                                path.display(),
1035                                e
1036                            );
1037                            Self::open_buffered_sources(path, file_size).await
1038                        }
1039                    }
1040                }
1041                #[cfg(not(unix))]
1042                {
1043                    let _ = direct_window;
1044                    tracing::warn!(
1045                        "Direct I/O is unavailable on this platform; using buffered I/O for {}",
1046                        path.display()
1047                    );
1048                    Self::open_buffered_sources(path, file_size).await
1049                }
1050            }
1051            // `resolve_disk_access_mode` never yields `Auto`; handle defensively.
1052            DiskAccessMode::Auto => Self::open_buffered_sources(path, file_size).await,
1053        }
1054    }

1056    /// Memory-map an SSTable file read-only.
1057    ///
1058    /// # Safety / correctness
1059    ///
1060    /// The returned [`Mmap`](memmap2::Mmap) aliases the file's bytes for its
1061    /// entire lifetime. SSTables are immutable once written, and CQLite treats
1062    /// them as read-only inputs, so this matches Cassandra's own mmap read
1063    /// strategy. Mutating the underlying file while a reader is open is
1064    /// undefined behaviour — callers must not do so.
1065    ///
1066    /// Note that only the initial mapping is fallible here. Once mapped, a later
1067    /// page fault — caused by truncation, deletion, or a network/overlay
1068    /// filesystem hiccup — raises `SIGBUS` and **cannot** be recovered as an
1069    /// `io::Error`. This is why mmap is opt-in and gated on immutable local
1070    /// files; see [`Config`]'s `storage.use_mmap` for the full constraints.
1071    fn map_file(path: &Path) -> Result<memmap2::Mmap> {
1072        // A5 read-work counter (FILE_OPENS; consumer C2): one per open(2) that mints
1073        // a reader fd — here the mmap cold-open site. No-op in release (design.md
1074        // Decision 1/2).
1075        crate::storage::sstable::read_work_counters::record_file_open();
1076        let std_file = std::fs::File::open(path)?;
1077        // SAFETY: read-only mapping of a file assumed immutable for the
1078        // reader's lifetime; see the function-level note above.
1079        let mmap = unsafe { memmap2::MmapOptions::new().map(&std_file)? };
1080        Ok(mmap)
1081    }

1083    /// Load CompressionInfo.db metadata for chunked reading
1084    async fn load_compression_info_metadata(
1085        path: &Path,
1086        _platform: &Arc<Platform>,
1087    ) -> Result<Option<CompressionInfo>> {
1088        use tokio::fs::File;
1089        use tokio::io::AsyncReadExt;
1090
1091        // Try to find CompressionInfo.db in same directory.
1092        let parent_dir = path.parent().unwrap_or(Path::new("."));
1093        // Derive the base name via the descriptor parser, which handles
1094        // hyphenated-UUID ids (e.g. "da-00000000-0000-0000-0000-000000000001-bti").
1095        // A fixed parts[0..3] split mangles those and looks for the wrong
1096        // "*-CompressionInfo.db", silently treating compressed data as
1097        // uncompressed (roborev #970).
1098        let base_name = crate::storage::sstable::version_gate::SsTableDescriptor::parse(path)
1099            .ok()
1100            .map(|d| format!("{}-{}-{}", d.version, d.sstable_id, d.format.as_str()));
1101
1102        if let Some(base) = base_name {
1103            let compression_info_path = parent_dir.join(format!("{}-CompressionInfo.db", base));
1104            if compression_info_path.exists() {
1105                let mut file = File::open(&compression_info_path).await?;
1106                let mut buffer = Vec::new();
1107                file.read_to_end(&mut buffer).await?;
1108
1109                // Fail-fast (issue #1001): a CompressionInfo.db that is present but
1110                // names an unknown/unsupported compressor (or is otherwise malformed)
1111                // MUST hard-error at reader-open time, BEFORE any Data.db chunk is read —
1112                // never silently fall back to the uncompressed path. The error carries
1113                // the offending component path for diagnosis. A genuinely uncompressed
1114                // SSTable has no CompressionInfo.db at all and returns Ok(None) below.
1115                let info = CompressionInfo::parse(&buffer).map_err(|e| {
1116                    Error::UnsupportedFormat(format!(
1117                        "Failed to parse CompressionInfo.db at {}: {}",
1118                        compression_info_path.display(),
1119                        e
1120                    ))
1121                })?;
1122                tracing::debug!(
1123                    "Loaded CompressionInfo: algorithm={}, chunk_length={}, chunks={}",
1124                    info.algorithm,
1125                    info.chunk_length,
1126                    info.chunk_offsets.len()
1127                );
1128                return Ok(Some(info));
1129            }
1130        }
1131
1132        Ok(None)
1133    }

1135    /// Load the `CRC.db` per-chunk checksum sidecar for read-time integrity of
1136    /// **uncompressed** BIG SSTables (issue #1396).
1137    ///
1138    /// Called only when `compression_info` is `None` (compressed tables carry
1139    /// inline per-chunk CRCs and no `CRC.db`). Returns:
1140    /// - `Some(crc)` for an uncompressed BIG SSTable that ships a `CRC.db`
1141    ///   (Cassandra writes one for every uncompressed BIG table) — verification is
1142    ///   then default-on for every uncompressed chunk read.
1143    /// - `None` for BTI (`da`) tables (Cassandra emits no `CRC.db`) or an
1144    ///   uncompressed BIG table whose `CRC.db` is absent. The absent case is the
1145    ///   owner-pinned **warn-and-proceed** decision (design D4): a `tracing::warn!` is
1146    ///   emitted so the missing integrity component is visible, and the read
1147    ///   proceeds unverified rather than hard-failing.
1148    ///
1149    /// A present-but-malformed `CRC.db` is a hard [`Error`] at open time (never a
1150    /// silent fall-through to unverified reads), mirroring the CompressionInfo.db
1151    /// fail-fast posture (#1001).
1152    async fn load_crc_reader(
1153        path: &Path,
1154        header: &SSTableHeader,
1155        data_len: u64,
1156    ) -> Result<Option<Arc<crc::CrcDb>>> {
1157        // BTI (`da`) never ships a CRC.db; nothing to load.
1158        if matches!(header.cassandra_version, CassandraVersion::V5_0Bti) {
1159            return Ok(None);
1160        }
1161
1162        let parent_dir = path.parent().unwrap_or(Path::new("."));
1163        let Some(base) = compression::extract_sstable_base_name(path) else {
1164            return Ok(None);
1165        };
1166        let crc_path = parent_dir.join(format!("{}-CRC.db", base));
1167        if !crc_path.exists() {
1168            // Absent CRC.db on an uncompressed BIG SSTable: warn-and-proceed
1169            // (owner-pinned, design D4). Cassandra 5.0 writes a CRC.db for every
1170            // uncompressed BIG SSTable, so its absence is notable but not fatal —
1171            // reads proceed unverified rather than hard-failing.
1172            tracing::warn!(
1173                "CRC.db absent for uncompressed SSTable {} — proceeding without \
1174                 read-time per-chunk CRC verification (warn-and-proceed, issue #1396)",
1175                path.display()
1176            );
1177            return Ok(None);
1178        }
1179
1180        let crc = crc::CrcDb::open(&crc_path, data_len).await.map_err(|e| {
1181            Error::corruption(format!(
1182                "Failed to parse CRC.db at {}: {}",
1183                crc_path.display(),
1184                e
1185            ))
1186        })?;
1187        tracing::debug!(
1188            "Loaded CRC.db for {}: chunk_size={}, chunks={}",
1189            path.display(),
1190            crc.chunk_size(),
1191            crc.chunk_count()
1192        );
1193        Ok(Some(Arc::new(crc)))
1194    }

1310    /// Rebind this reader's lazy-scan path to `new_path` WITHOUT re-opening or
1311    /// re-parsing (issue #2383, the #2356 "rebind-by-inode" direction).
1312    ///
1313    /// The caller MUST have already established that `new_path` is the SAME
1314    /// on-disk generation as this reader by AUTHORITATIVE identity — matching
1315    /// `(device, inode)` and size (issue #28 no-heuristics; Cassandra snapshot
1316    /// files are hardlinks to the immutable SSTable, so a same-inode candidate is
1317    /// byte-identical). The already-open point/scan handles reference the shared
1318    /// inode and stay valid across the swap; only [`Self::new_scan_cursor`]'s
1319    /// per-scan `File::open` reads this path, so pointing it at a LIVE hardlink
1320    /// restores the #2352 ENOENT protection with zero parse cost.
1321    ///
1322    /// ## In-flight isolation invariant (roborev-1654 HIGH, adjudicated)
1323    ///
1324    /// Mutating this shared `Arc<SSTableReader>`'s path while a concurrent request
1325    /// scans it does NOT break the in-flight request's isolation, because the
1326    /// rebind is byte-transparent and monotone-toward-live:
1327    /// 1. **Same immutable inode.** The sole caller (`warm::registry::rebind`)
1328    ///    fires only after `rebuild::rebind_matches` proves
1329    ///    `(device, inode, generation)` + size equality; every rebind target is a
1330    ///    hardlink to the SAME immutable SSTable inode, so any path this reader
1331    ///    carries yields byte-identical data (issue #28 no-heuristics).
1332    /// 2. **Atomic swap.** `file_path` is an `arc_swap::ArcSwap<PathBuf>`; a
1333    ///    concurrent `file_path()` sees either the old or the new whole `PathBuf`,
1334    ///    never a torn value.
1335    /// 3. **Dead → live only.** A rebind happens only when the current path is
1336    ///    dead and an identity-matching LIVE hardlink exists, so an in-flight
1337    ///    request's NEXT `new_scan_cursor` `File::open` is strictly MORE likely to
1338    ///    succeed after the rebind than before (pre-rebind it would ENOENT).
1339    /// 4. **No semantics off the path.** No `file_path()` consumer re-derives
1340    ///    keyspace/table/schema from the path after `open`; the scan path uses it
1341    ///    ONLY for `File::open` (bytes) and all parsed read state (header,
1342    ///    compression info, CRC, index, generation) lives on `&self`, fixed at
1343    ///    open — the swapped path changes which hardlink is read, never what the
1344    ///    bytes mean.
1345    pub fn rebind_path(&self, new_path: &Path) {
1346        self.file_path.store(Arc::new(new_path.to_path_buf()));
1347    }

1486impl Drop for SSTableReader {
1487    fn drop(&mut self) {
1488        use crate::observability::{self as obs, catalog};
1489        // Keep the per-format live-reader count and the SSTABLES_OPEN gauge in
1490        // sync as readers are released. Use a single atomic read-modify-write
1491        // (`fetch_sub`) on the SAME per-format counter `open()` incremented: a
1492        // load-then-store would race concurrent drops and lose decrements,
1493        // drifting the gauge permanently high. `open()` only ever increments on
1494        // success, so this never underflows in practice.
1495        let format = self.sstable_format_label();
1496        let now = sstables_open_count_for(format).fetch_sub(1, Ordering::Relaxed) - 1;
1497        obs::record_gauge(
1498            catalog::SSTABLES_OPEN,
1499            now,
1500            &[(catalog::attr::SSTABLE_FORMAT, format.into())],
1501        );
1502    }
1503}
