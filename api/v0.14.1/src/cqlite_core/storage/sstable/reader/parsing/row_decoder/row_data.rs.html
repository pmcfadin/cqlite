3impl V5CompressedLegacyParser {
4    /// Parse row data (header + cells) and return cells with new offset
5    ///
6    /// V5CompressedLegacy format stores cells WITHOUT column names in schema column order.
7    /// Schema is REQUIRED to determine which column each value belongs to.
8    ///
9    /// Returns: `ParsedRow` = `(cells, row_header, new_offset, is_static)` where
10    /// `is_static` is `true` when the row's `EXTENDED_IS_STATIC` flag was set.
11    /// Static rows must be merged into clustering rows by the caller, not emitted directly.
12    #[allow(clippy::too_many_arguments)]
13    pub(super) fn parse_row_data_with_offset(
14        &self,
15        data: &[u8],
16        offset: usize,
17        schema: Option<&TableSchema>,
18        reader: &crate::storage::sstable::reader::types::SSTableReader,
19        want_cell_metadata: bool,
20        resolution: &RowColumnResolution,
21        // Issue #1741 (Finding 1): read-side shadow for per-cell tombstone/TTL
22        // filtering. `Some` only on user-facing SELECT reads (`read_shadowing`);
23        // `None` for every physical consumer (compaction / delta-scan / tests), which
24        // stay byte-unchanged.
25        shadow: Option<&PartitionShadow>,
26    ) -> Result<ParsedRow> {
27        self.parse_row_data_with_offset_impl(
28            data,
29            offset,
30            schema,
31            reader,
32            want_cell_metadata,
33            None,
34            resolution,
35            shadow,
36        )
37    }
38
39    /// Implementation of [`Self::parse_row_data_with_offset`] with an optional
40    /// per-column complex-element collector for the compaction read path
41    /// (epic #899). When `compaction_complex_out` is `Some`, every complex
42    /// column's per-element cells + complex deletion are captured into it
43    /// alongside the normal collapsed-`Value` cells. On user-facing reads it is
44    /// `None` and behavior is byte-unchanged.
45    #[allow(clippy::too_many_arguments)]
46    pub(super) fn parse_row_data_with_offset_impl(
47        &self,
48        data: &[u8],
49        mut offset: usize,
50        schema: Option<&TableSchema>,
51        reader: &crate::storage::sstable::reader::types::SSTableReader,
52        want_cell_metadata: bool,
53        mut compaction_complex_out: Option<&mut CompactionComplexColumns>,
54        resolution: &RowColumnResolution,
55        shadow: Option<&PartitionShadow>,
56    ) -> Result<ParsedRow> {
57        // Issue #1642 (K3): positional cell vector, built directly in
58        // serialization-header (schema) column order. Determinism comes from
59        // CONSTRUCTION — the emit path no longer allocates a per-row `HashMap` nor
60        // alphabetically re-sorts each row. The interned column-name handle
61        // (issue #1334) is a schema-owned `Arc<str>` shared across every cell/row,
62        // so populating a cell with its name stays an `Arc::clone` refcount bump,
63        // NOT a per-cell heap `String` allocation. Pre-sized to the on-disk data
64        // column count PLUS the clustering-key count so the common case does not
65        // reallocate — clustering-key cells are pushed FIRST (below), before the
66        // data columns, so sizing to data columns alone reallocates on a clustered
67        // table (issue #1642).
68        let mut cells: RowCells =
69            Vec::with_capacity(resolution.columns_for(false).len() + resolution.clustering_len());
70        // Parallel per-cell write metadata map (populated alongside `cells`).
71        // Only allocated when the caller actually needs WRITETIME/TTL metadata
72        // (i.e. `want_cell_metadata == true`).  On the normal read path this stays
73        // `None` so that zero HashMap allocations or inserts occur per cell.
74        let mut cell_meta: Option<HashMap<String, CellWriteMetadata>> = if want_cell_metadata {
75            Some(HashMap::new())
76        } else {
77            None
78        };
79
80        // DS4 (Issue #700): Per-column complex collection metadata.  Only allocated when
81        // want_cell_metadata is true (same gate as cell_meta to avoid hot-path overhead).
82        let mut complex_col_meta: Option<HashMap<String, ComplexColumnMeta>> = if want_cell_metadata
83        {
84            Some(HashMap::new())
85        } else {
86            None
87        };
88
89        let schema = schema.ok_or_else(|| {
90            Error::schema(format!(
91                "V5CompressedLegacy: Schema required for {}.{} (cells stored without column names)",
92                self.keyspace, self.table_name
93            ))
94        })?;
95
96        debug!(
97            "V5CompressedLegacy: Starting row data parse at offset {} with {} schema columns",
98            offset,
99            schema.columns.len()
100        );
101
102        // ISSUE #213 FIX: Correct parsing order for V5CompressedLegacy format
103        //
104        // The CORRECT format from Cassandra UnfilteredSerializer.java is:
105        //   1. [row_flags: u8]
106        //   2. [extended_flags: u8 if 0x80 set]
107        //   3. [clustering_prefix: variable]  ← BEFORE row_size!
108        //   4. [row_size: VInt]
109        //   5. [prev_size: VInt]
110        //   6. [row_body: timestamp, ttl, deletion, bitmap, cells]
111        //
112        // The previous code parsed row_size BEFORE clustering, which caused
113        // clustering key bytes to be misinterpreted as row_size (often 0).
114
115        // Step 1: Parse row flags (1-2 bytes)
116        let (row_flags, extended_flags, flags_size) = self.parse_row_flags(data, offset)?;
117        offset += flags_size;
118
119        // Issue #258 fix: Check if this is a static row (no clustering prefix)
120        // Per Cassandra UnfilteredSerializer.java lines 114-122, 190-191:
121        // Static rows have the IS_STATIC bit (0x01) set in extended flags and
122        // do NOT have a clustering prefix - skip directly to row_size.
123        let is_static = extended_flags
124            .map(|ef| (ef & EXTENDED_IS_STATIC) != 0)
125            .unwrap_or(false);
126
127        // Step 2: Parse clustering prefix BEFORE row_size (Issue #213 fix)
128        // This is the critical change - clustering comes AFTER flags but BEFORE row_size
129        // EXCEPT for static rows which have no clustering prefix at all.
130        let (clustering_values, offset) = if !is_static {
131            self.parse_clustering_prefix(data, offset, schema)?
132        } else {
133            tracing::debug!(
134                "V5CompressedLegacy: Static row detected (extended_flags=0x{:02x}), skipping clustering prefix",
135                extended_flags.unwrap_or(0)
136            );
137            (vec![], offset)
138        };
139
140        tracing::debug!(
141            "V5CompressedLegacy: Parsed {} clustering values after flags, now at offset {} (is_static={})",
142            clustering_values.len(),
143            offset,
144            is_static
145        );
146
147        // Issue #229 FIX: Add clustering key values to cells HashMap
148        //
149        // Cassandra stores clustering keys separately from regular columns, but they
150        // must be included in the result for proper query output. Without this fix,
151        // tables with clustering keys show fallback column names because the clustering
152        // values weren't being added to the cells HashMap.
153        for (i, _ck) in schema.clustering_keys.iter().enumerate() {
154            if i < clustering_values.len() {
155                // Issue #1334: reuse the interned clustering-key name handle
156                // (an `Arc::clone`) rather than cloning the schema `String`.
157                if let Some(name) = resolution.clustering_name(i) {
158                    cells.push((Arc::clone(name), clustering_values[i].clone()));
159                }
160            }
161        }
162
163        // Issue #1741 (Finding 1): per-cell shadow context for this row. `Some` only
164        // on the user-facing SELECT path. `covering` is the deletion (µs) that shadows
165        // any data older than it (partition tombstone folded with the open range
166        // tombstone when this row's clustering falls inside it); `now` is the read
167        // clock for per-cell TTL expiry. A data cell / whole collection whose
168        // effective write ts <= `covering`, or which is TTL-expired at `now`, is
169        // dropped below (no new per-cell allocation — the insert is made conditional).
170        let cell_ctx: Option<(Option<i64>, i64)> =
171            shadow.map(|sh| sh.cell_context(&clustering_values));
172
173        // Step 3: Parse row metadata (row_size, prev_size, timestamps, etc.)
174        //
175        // CRITICAL (Issue #237): Save offset where row_size VInt STARTS.
176        // The row_size value is measured from AFTER this VInt is consumed.
177        // Formula: next_offset = (row_metadata_offset + row_size_vint_len) + row_size
178        // This offset is right after the clustering prefix (which was already parsed).
179        let row_metadata_offset = offset;
180        let (mut row_header, row_size) =
181            self.parse_row_metadata(data, offset, row_flags, extended_flags)?;
182
183        // CRITICAL VALIDATION: row_size must be reasonable
184        //
185        // In V5CompressedLegacy format, row_size should never exceed the block size (typically 16KB).
186        // If row_size is unreasonably large, it indicates either:
187        // 1. Partition tombstone or deletion marker (no actual row data)
188        // 2. Format parsing error (landed at wrong offset)
189        // 3. Corrupted data
190        //
191        // In all cases, we should skip this partition rather than panic.
192        const MAX_REASONABLE_ROW_SIZE: u64 = 1_000_000; // 1MB max (very generous)
193        if row_size > MAX_REASONABLE_ROW_SIZE {
194            return Err(Error::corruption(format!(
195                "V5CompressedLegacy: Unreasonably large row_size={} at offset {} (max: {}). Likely partition tombstone or format error.",
196                row_size,
197                offset,
198                MAX_REASONABLE_ROW_SIZE
199            )));
200        }
201
202        // Row payloads can span multiple compressed chunks in V5CompressedLegacy format.
203        // The reader has already stitched all chunks together (see get_all_entries()),
204        // so row_size is valid across chunk boundaries. We MUST NOT validate against
205        // individual chunk sizes as rows naturally span chunks in Cassandra's format.
206        // This is NOT corruption - it's the intended file layout.
207
208        tracing::debug!(
209            "V5CompressedLegacy: Parsed row metadata at offset {}: header_size={} bytes, row_size={} bytes, timestamp={:?}, ttl={:?}, deletion={:?}",
210            offset, row_header.header_size, row_size, row_header.timestamp, row_header.ttl, row_header.local_deletion_time
211        );
212
213        // Row tombstone detection (Issue #191, Phase 2) + coexistence (Issue #932).
214        //
215        // A row with ROW_HAS_DELETION (0x10) carries a row-level deletion. HISTORICALLY
216        // such a row had NO cells, so the parser skipped cell parsing entirely. But a
217        // row deletion can COEXIST with surviving cells (cells written strictly after
218        // the deletion — issue #932): the writer emits HAS_DELETION + HAS_TIMESTAMP +
219        // the surviving cells. We must therefore skip cell parsing ONLY when there are
220        // no cell bytes after the row header; otherwise fall through to the normal cell
221        // loop so the surviving cells are read (the row header still carries the
222        // deletion via `local_deletion_time` / `marked_for_delete_at`).
223        //
224        // Calculate offset after row data (based on row_size from header).
225        //
226        // CRITICAL FIX (Issue #237): row_size is measured from AFTER the row_size VInt,
227        // not from where it starts. This matches Cassandra's getFilePointer() semantics:
228        //   next_position = row_size_value + position_after_reading_row_size_vint
229        let after_row_offset =
230            (row_metadata_offset + row_header.row_size_vint_len) + row_size as usize;
231        // Cell data (if any) begins right after the row header. When that start
232        // reaches the row boundary there are no cells — a pure row tombstone.
233        let cell_data_start = row_metadata_offset + row_header.header_size;
234        let has_cell_bytes = cell_data_start < after_row_offset;
235
236        if row_header.local_deletion_time.is_some() && !has_cell_bytes {
237            tracing::debug!(
238                "V5CompressedLegacy: Pure row tombstone (deletion_time={:?}), skipping cell parsing",
239                row_header.local_deletion_time
240            );
241
242            // Validate we have enough data
243            if after_row_offset > data.len() {
244                let remaining = data
245                    .len()
246                    .saturating_sub(row_metadata_offset + row_header.row_size_vint_len);
247                return Err(Error::corruption(format!(
248                    "V5CompressedLegacy: Not enough bytes for row data at offset {} (need {}, have {})",
249                    row_metadata_offset + row_header.row_size_vint_len,
250                    row_size,
251                    remaining
252                )));
253            }
254
255            let next_offset = after_row_offset;
256            tracing::debug!(
257                "V5CompressedLegacy: Skipped tombstoned row, next offset = {}",
258                next_offset
259            );
260
261            // Return empty cells for tombstoned row (no cell metadata, no complex meta)
262            return Ok((
263                cells,
264                cell_meta,
265                Some(row_header),
266                next_offset,
267                is_static,
268                None,
269            ));
270        }
271
272        // Advance offset past row metadata to start of cell data
273        let mut offset = offset + row_header.header_size;
274
275        tracing::debug!(
276            "V5CompressedLegacy: Cell data starts at offset {}, first 32 bytes: {}",
277            offset,
278            hex::encode(&data[offset..std::cmp::min(offset + 32, data.len())])
279        );
280
281        // Cell flags validation: First byte should be valid cell flags (0x00-0x1F) for simple cells
282        // Common flags: 0x00 (basic cell), 0x08 (USE_ROW_TIMESTAMP), 0x04 (HAS_EMPTY_VALUE)
283        // Deleted cells have 0x01 (IS_DELETED), expiring cells have 0x02 (IS_EXPIRING)
284        //
285        // NOTE: For complex columns (non-frozen collections), the first byte is a VInt for the
286        // cell count, which may have values > 0x1F. This is normal and not an error.
287        // The validation below is only accurate for tables with all simple cells.
288        if offset < data.len() {
289            let first_byte = data[offset];
290            if first_byte <= 0x1F {
291                debug!(
292                    "V5CompressedLegacy: Valid cell flags 0x{:02x} at offset {} after row header",
293                    first_byte, offset
294                );
295            } else {
296                debug!(
297                    "V5CompressedLegacy: First byte 0x{:02x} at offset {} (> 0x1F) - may be VInt for complex column cell count",
298                    first_byte, offset
299                );
300            }
301        }
302
303        // CRITICAL: V5CompressedLegacy format stores cells WITHOUT column names
304        // or column IDs in the binary data. Cells appear in SCHEMA DEFINITION ORDER
305        // (the order columns were defined in CREATE TABLE), NOT alphabetical order.
306        //
307        // NULL/missing columns are handled by:
308        // - Checking for cell marker (0x08) before attempting to parse
309        // - If no marker found or parse fails, column is NULL (not present)
310        // - Continue to next column in schema order
311        //
312        // This implementation uses schema definition order directly, which is the
313        // correct approach per Cassandra 5.0 SerializationHeader semantics.
314
315        // CRITICAL FIX (Issue #191): cells are stored in SERIALIZATION HEADER column
316        // order (Statistics.db serialization header — alphabetical by
317        // ColumnIdentifier/comparator), NOT CQL schema order. Partition/clustering
318        // keys carry no cell data and are excluded; dropped columns (present on disk
319        // but absent from the supplied schema) are RETAINED in order so their bytes
320        // are consumed and trailing columns stay byte-aligned (issue #1080 Part 2).
321        //
322        // Issue #702: the on-disk column group differs by row kind, so the resolution
323        // exposes a separate ordering for static vs regular rows.
324        //
325        // Issue #1046 (the true hoist): this header→schema resolution is CONSTANT for
326        // every row in the block, so it is built ONCE (in `RowColumnResolution::build`
327        // at the top of the per-block driver) and reused here. The per-row body now
328        // performs ZERO schema-lookup allocations — no per-row `HashMap`, no per-row
329        // `String` clone, no per-row `Vec` of columns.
330        let columns_in_order: &[ColumnToParse] = resolution.columns_for(is_static);
331
332        // Apply the missing_columns_bitmap INLINE (no per-row `Vec`): a column at
333        // on-disk index `idx` is present iff `idx >= 64` (beyond the u64 bitmap,
334        // treated present) or its bit is clear. The bitmap is indexed by the ON-DISK
335        // column order, which is exactly `columns_in_order` (dropped columns retained),
336        // so the index alignment is preserved.
337        let missing_bitmap = row_header.missing_columns_bitmap;
338        let is_present = |idx: usize| -> bool {
339            match missing_bitmap {
340                Some(bitmap) => idx >= 64 || (bitmap & (1u64 << idx)) == 0,
341                None => true,
342            }
343        };
344
345        tracing::debug!("V5CompressedLegacy: Parsing cells in SERIALIZATION HEADER ORDER starting at offset {} (row header was {} bytes, {} on-disk columns, bitmap={:?})", offset, row_header.header_size, columns_in_order.len(), missing_bitmap);
346        tracing::debug!(
347            "V5CompressedLegacy: Cell data hex (first 64 bytes): {}",
348            hex::encode(&data[offset..std::cmp::min(offset + 64, data.len())])
349        );
350
351        // Issue #221: Check if row has complex deletion info for non-frozen collections
352        let has_complex_deletion = (row_flags & ROW_HAS_COMPLEX_DELETION) != 0;
353        if has_complex_deletion {
354            tracing::debug!("V5CompressedLegacy: Row has HAS_COMPLEX_DELETION flag (0x40) set");
355        }
356
357        // Issue #1741 read-side shadowing aggregate. Computed from scalars the cell
358        // parser already returns (no new allocation, no extra HashMap): the max data
359        // cell write timestamp, the max expiring-cell expiry, and whether any data
360        // cell is live-forever. Stashed on `row_header` after the loop and consulted
361        // by the read emit paths to hide partition/range-tombstone-shadowed and
362        // TTL-expired rows. Kept off the write/compaction reconciliation path.
363        let mut agg_max_cell_ts: Option<i64> = None;
364        let mut agg_max_expires_at: Option<i64> = None;
365        let mut agg_has_live_forever = false;
366
367        for (col_idx, ctp) in columns_in_order.iter().enumerate() {
368            // Skip columns marked MISSING by the row's bitmap (inline, no per-row
369            // allocation). `col_idx` is the ON-DISK column index — exactly what the
370            // bitmap is indexed by — so this is identical to the prior pre-filtered
371            // `columns_to_parse` Vec, just without materializing it.
372            if !is_present(col_idx) {
373                continue;
374            }
375            let header_type: Option<&str> = ctp.header_type;
376
377            // A column present on disk but ABSENT from the supplied schema is a
378            // DROPPED column (issue #1080 Part 2): its bytes MUST be consumed to
379            // keep the trailing columns aligned, but it is NOT emitted. We decode
380            // it with a synthetic Column whose `data_type` is the AUTHORITATIVE
381            // on-disk header marshal string (the only type metadata we have), then
382            // discard the value. `emit` gates every cell/metadata insertion below.
383            let emit = ctp.schema.is_some();
384            // Synthetic column used ONLY to consume a DROPPED column's bytes; built
385            // from the on-disk header marshal type. Initialized lazily (deferred
386            // binding) so the common schema-present path pays no allocation — the
387            // dropped-column path is rare. The holder outlives the borrow below.
388            let dropped_column_holder;
389            let column: &crate::schema::Column = match ctp.schema {
390                Some(c) => c,
391                None => {
392                    dropped_column_holder = crate::schema::Column {
393                        name: format!("__dropped_col_{col_idx}"),
394                        data_type: header_type.unwrap_or("blob").to_string(),
395                        nullable: true,
396                        default: None,
397                        is_static,
398                    };
399                    &dropped_column_holder
400                }
401            };
402
403            // Issue #1081: the AUTHORITATIVE complex-ness / complex-decode type.
404            // Prefer the on-disk SerializationHeader marshal type (carries
405            // `UserType(...)` for a non-frozen UDT, which the supplied schema's
406            // bare short form cannot express); fall back to the supplied schema
407            // type only on the header-empty path (no-heuristics: both are
408            // authoritative metadata, never guessed from bytes — issue #28). For a
409            // dropped column the synthetic `column.data_type` IS the header marshal
410            // type, so this resolves identically either way.
411            let complex_type: &str = header_type.unwrap_or(&column.data_type);
412
413            if offset >= data.len() {
414                tracing::debug!(
415                    "V5CompressedLegacy: Reached end of data at column {} ('{}'), parsed {}/{} on-disk cells",
416                    col_idx,
417                    column.name,
418                    cells.len(),
419                    columns_in_order.len()
420                );
421                break;
422            }
423
424            // Issue #221: Branch based on column type - complex columns need special parsing
425            // Issue #693: simple columns return 4-tuple including cell timestamp / expiration;
426            //             complex columns return 2-tuple and inherit the row-level timestamp.
427            if ctp.is_complex {
428                tracing::debug!(
429                    "V5CompressedLegacy: Column '{}' is complex (non-frozen collection / multicell UDT), using parse_complex_column",
430                    column.name
431                );
432                // Epic #899: on the compaction read path collect per-element
433                // cells + the real complex deletion into `compaction_complex_out`
434                // (otherwise this is the user-facing read path: collapsed value
435                // only, byte-unchanged). Elements inherit the row liveness
436                // timestamp when they carry USE_ROW_TIMESTAMP.
437                let parse_result = if compaction_complex_out.is_some() {
438                    let row_ts = row_header.timestamp.unwrap_or(0);
439                    let mut element_buf = Vec::new();
440                    self.parse_complex_column_inner(
441                        data,
442                        offset,
443                        column,
444                        complex_type,
445                        has_complex_deletion,
446                        row_ts,
447                        Some(&mut element_buf),
448                        // Compaction / delta read path: NO per-element filtering (it
449                        // needs every element for reconciliation) — byte-unchanged.
450                        None,
451                    )
452                    .map(|(value, new_offset, col_meta)| {
453                        if emit {
454                            if let Some(ref mut out) = compaction_complex_out {
455                                // Capture the whole-collection collapsed value for the
456                                // byte-neutral Phase A output path (roborev #863).
457                                out.insert(
458                                    column.name.clone(),
459                                    (col_meta.complex_deletion, element_buf, value.clone()),
460                                );
461                            }
462                        }
463                        (value, new_offset, col_meta)
464                    })
465                } else {
466                    // Issue #1741 (per-element filtering): on the user-facing SELECT
467                    // read path (`cell_ctx.is_some()`), thread the covering deletion +
468                    // read clock + row-liveness ts into the element loop so shadowed /
469                    // TTL-expired elements are skipped from the emitted container.
470                    // `None` when no shadow context is active (tombstone-free reads and
471                    // physical consumers) — byte-unchanged.
472                    let element_filter = cell_ctx.map(|(cover, now)| ElementShadow {
473                        cover,
474                        now,
475                        row_ts: row_header.timestamp,
476                        // Issue #1741: the effective row expiry a USE_ROW_TTL
477                        // collection element inherits — computed EXACTLY like the
478                        // scalar USE_ROW_TTL cell path (see the `None if use_row_ttl`
479                        // arm below): the pk-liveness localExpirationTime
480                        // (`liveness_expires_at_seconds`, year-2106-safe i64 from
481                        // HAS_TTL), falling back to the row `local_deletion_time`
482                        // re-read UNSIGNED on oa/da when the GC clock is post-2038.
483                        row_expires_at: row_header.liveness_expires_at_seconds.or_else(|| {
484                            row_header.local_deletion_time.map(|s| {
485                                if self.has_uint_deletion_time() {
486                                    (s as u32) as i64
487                                } else {
488                                    s as i64
489                                }
490                            })
491                        }),
492                        // Issue #2038 (round 3): row-liveness TTL seconds, paired
493                        // with `row_expires_at` above so a `USE_ROW_TTL` collection
494                        // element's per-cell-metadata expiry can be resolved
495                        // EXACTLY like the scalar `USE_ROW_TTL` cell path's
496                        // `(row_header.ttl, row_expiry)` pairing (~line 726 below).
497                        row_ttl_seconds: row_header.ttl,
498                    });
499                    self.parse_complex_column(
500                        data,
501                        offset,
502                        column,
503                        complex_type,
504                        has_complex_deletion,
505                        reader,
506                        element_filter,
507                    )
508                };
509                match parse_result {
510                    Ok((value, new_offset, col_meta)) => {
511                        tracing::debug!(
512                            "V5CompressedLegacy:   ✓ Complex column {} '{}' = {:?}, consumed {} bytes",
513                            col_idx, column.name, value, new_offset - offset
514                        );
515                        // Normal read-path (WRITETIME/TTL queries): use the row-level liveness
516                        // timestamp unchanged.  Cassandra's WRITETIME(non_frozen_collection) on
517                        // the standard read path returns the row timestamp, not per-element max.
518                        // The delta-scan path computes its own max-element-writetime from
519                        // ComplexColumnMeta (stored separately below) and never reads this field
520                        // for collection columns.  Do NOT mutate this with max_element_writetime
521                        // here — that would silently change WRITETIME(col) on the ordinary path
522                        // (roborev Finding 1).
523                        if emit {
524                            // Issue #1741 (Finding 2): the collection's effective max
525                            // write ts is the newest of its element-level timestamps and
526                            // the row liveness ts inherited by USE_ROW_TIMESTAMP elements.
527                            // Folding `max_element_writetime` (not just the row ts) means a
528                            // collection element NEWER than a partition/range tombstone
529                            // keeps the row visible even when the row ts predates the
530                            // tombstone. `max_element_writetime` folds ALL live elements
531                            // (including shadow-dropped ones), so a wholly-shadowed
532                            // collection still contributes its element ts here.
533                            let coll_eff_ts = col_meta
534                                .max_element_writetime
535                                .max(row_header.timestamp.unwrap_or(i64::MIN));
536                            // Issue #1741 (per-element filtering): the shadowed/expired
537                            // elements have ALREADY been skipped from `value` inside
538                            // `parse_complex_column_inner`. The collection now reads as
539                            // ABSENT (null) iff the read-side filter EMPTIED it (every
540                            // element was shadowed/expired: `shadow_filtered_element_count
541                            // > 0` AND the container is empty). A collection empty for any
542                            // OTHER reason (genuinely empty / all element-tombstones, where
543                            // `shadow_filtered_element_count == 0`) keeps its prior
544                            // behavior, so tombstone-free reads and physical consumers are
545                            // byte-unchanged. `cover == None` never filters an element, so
546                            // a non-shadowing read never treats a collection as absent.
547                            let collection_absent = col_meta.shadow_filtered_element_count > 0
548                                && Self::complex_value_is_empty(&value);
549                            // Fold the collection into the ROW aggregate REGARDLESS of
550                            // absence (so a wholly-shadowed collection is still recognised
551                            // as shadowed by `row_hidden`, and Finding 2's newest-element ts
552                            // keeps a surviving row visible). Its expiring elements
553                            // contribute their expiry; `has_live_forever_element` is already
554                            // computed POST-filter, so a wholly-shadowed collection never
555                            // sets live-forever.
556                            if coll_eff_ts != i64::MIN {
557                                agg_max_cell_ts = Some(
558                                    agg_max_cell_ts.map_or(coll_eff_ts, |m| m.max(coll_eff_ts)),
559                                );
560                            }
561                            if let Some(e) = col_meta.max_element_expires_at {
562                                agg_max_expires_at =
563                                    Some(agg_max_expires_at.map_or(e, |m| m.max(e)));
564                            }
565                            if col_meta.has_live_forever_element {
566                                agg_has_live_forever = true;
567                            }
568                            if !collection_absent {
569                                if let Some(ref mut meta_map) = cell_meta {
570                                    let row_ts = row_header.timestamp.unwrap_or(0);
571                                    // Issue #2038: surface the collection's expiry so
572                                    // `TTL(non_frozen_collection/UDT)` is not always
573                                    // `null` (the complex-cell analogue of the scalar
574                                    // #1743 fix at ~line 736 below). Authoritative,
575                                    // no-heuristics: `visible_uniform_expiration` is
576                                    // `Some` ONLY when every VISIBLE element shares the
577                                    // identical explicit expiry (the `ExpiryHomogeneity`
578                                    // tracker in `complex_column.rs` — roborev Medium
579                                    // finding); a mixed/heterogeneous collection, or one
580                                    // with no expiring element, stays `None` rather than
581                                    // over-approximating with a single element's TTL.
582                                    let expiration = col_meta.visible_uniform_expiration.clone();
583                                    meta_map.insert(
584                                        column.name.clone(),
585                                        CellWriteMetadata {
586                                            write_timestamp_micros: row_ts,
587                                            expiration,
588                                        },
589                                    );
590                                }
591                                // DS4 (Issue #700): Store ComplexColumnMeta for delta-scan.
592                                if let Some(ref mut ccm_map) = complex_col_meta {
593                                    ccm_map.insert(column.name.clone(), col_meta);
594                                }
595                                // Issue #1334/#1642: interned name handle (Arc::clone),
596                                // pushed positionally in schema column order.
597                                cells.push((Arc::clone(&ctp.name), value));
598                            }
599                        }
600                        offset = new_offset;
601                    }
602                    Err(e) => {
603                        tracing::debug!(
604                            "V5CompressedLegacy:   ✗ Complex column {} '{}' at offset {} FAILED: {}",
605                            col_idx, column.name, offset, e
606                        );
607                        break;
608                    }
609                }
610            } else {
611                // Issue #1741: peek the cell flags (offset points at the flags byte)
612                // to detect USE_ROW_TTL (0x10), which makes a cell with no explicit
613                // expiry inherit the ROW's expiry rather than being live-forever.
614                let cell_flags = data.get(offset).copied().unwrap_or(0);
615                match self.parse_cell_value_schema_order(
616                    data,
617                    offset,
618                    column,
619                    header_type,
620                    // J1 (issue #1635): the precomputed per-column dispatch tag,
621                    // resolved once per block — no per-cell `to_lowercase`.
622                    Some(&ctp.kind),
623                    reader,
624                ) {
625                    Ok((value, cell_own_ts, cell_exp, new_offset)) => {
626                        tracing::debug!(
627                            "V5CompressedLegacy:   ✓ Column {} '{}' ({}) = {:?}, consumed {} bytes",
628                            col_idx,
629                            column.name,
630                            column.data_type,
631                            value,
632                            new_offset - offset
633                        );
634                        // Only compute and store per-cell metadata when the caller requested it.
635                        // On the normal read hot-path (want_cell_metadata == false), cell_meta is
636                        // None and this entire block is skipped — zero allocations per cell.
637                        // `emit` is false for a DROPPED column (issue #1080 Part 2): we still
638                        // advanced `offset` to consume its bytes, but emit no cell/metadata.
639                        if emit {
640                            // Issue #1741 read-side shadowing aggregate (scalars only,
641                            // no allocation). A cell tombstone is not live data. A live
642                            // cell contributes its effective write timestamp and, if
643                            // expiring (explicit per-cell TTL, or USE_ROW_TTL inheriting
644                            // the row's expiry), its expiry; otherwise it is live-forever.
645                            // Computed BEFORE the metadata block, which moves `cell_exp`.
646                            let mut dropped = false;
647                            if !matches!(value, Value::Tombstone(_)) {
648                                let eff_ts = cell_own_ts.or(row_header.timestamp);
649                                // A USE_ROW_TTL cell inherits the ROW's expiry. For a
650                                // TTL-bearing INSERT that expiry is the pk-liveness
651                                // localExpirationTime (`liveness_expires_at_seconds`,
652                                // from HAS_TTL); `local_deletion_time` is set only by
653                                // HAS_DELETION (row tombstone), so it is the fallback.
654                                let use_row_ttl = (cell_flags & 0x10) != 0;
655                                let eff_exp: Option<i64> = match cell_exp.as_ref() {
656                                    Some(e) => Some(e.expires_at_seconds),
657                                    None if use_row_ttl => {
658                                        // Liveness expiry is already the year-2106-safe
659                                        // i64 (unsigned-reinterpreted at decode, #1741 F1).
660                                        // The `local_deletion_time` fallback is stored as
661                                        // the `(u32) as i32` on-disk representation, so on
662                                        // oa/da a post-2038 GC clock must be re-read UNSIGNED
663                                        // here too rather than sign-extended negative.
664                                        row_header.liveness_expires_at_seconds.or_else(|| {
665                                            row_header.local_deletion_time.map(|s| {
666                                                if self.has_uint_deletion_time() {
667                                                    (s as u32) as i64
668                                                } else {
669                                                    s as i64
670                                                }
671                                            })
672                                        })
673                                    }
674                                    None => None,
675                                };
676                                // Issue #1741 (Finding 1): per-cell shadow/TTL filter.
677                                // Drop this data cell from the EMITTED map when its
678                                // effective write ts is shadowed by the covering deletion
679                                // (`eff_ts <= cover`) or it is TTL-expired at `now`.
680                                // `cell_ctx` is `None` for physical reads → never drops →
681                                // byte-unchanged.
682                                if let Some((cover, now)) = cell_ctx {
683                                    dropped = PartitionShadow::cell_shadowed_or_expired(
684                                        cover, now, eff_ts, eff_exp,
685                                    );
686                                }
687                                // Fold this cell into the ROW aggregate that drives the
688                                // row-level `row_hidden` decision. A dropped (shadowed/
689                                // expired) cell STILL contributes its ts + expiry — so a
690                                // row reduced to nothing is recognised as shadowed/expired
691                                // (its max ts stays `<= covering` / its expiry stays past)
692                                // rather than looking like an empty/truncated parse — but
693                                // it is NEVER counted live-forever (it is not live).
694                                if let Some(t) = eff_ts {
695                                    agg_max_cell_ts = Some(agg_max_cell_ts.map_or(t, |m| m.max(t)));
696                                }
697                                match eff_exp {
698                                    Some(e) => {
699                                        agg_max_expires_at =
700                                            Some(agg_max_expires_at.map_or(e, |m| m.max(e)));
701                                    }
702                                    None => {
703                                        if !dropped {
704                                            agg_has_live_forever = true;
705                                        }
706                                    }
707                                }
708                            }
709                            if dropped {
710                                offset = new_offset;
711                                continue;
712                            }
713                            // Only compute and store per-cell metadata when the caller requested it.
714                            // On the normal read hot-path (want_cell_metadata == false), cell_meta is
715                            // None and this entire block is skipped — zero allocations per cell.
716                            if let Some(ref mut meta_map) = cell_meta {
717                                // Resolve effective write timestamp:
718                                // use cell's own timestamp when present, else row-level liveness timestamp.
719                                let effective_ts = cell_own_ts
720                                    .unwrap_or_else(|| row_header.timestamp.unwrap_or(0));
721                                // Resolve expiration: explicit per-cell TTL wins;
722                                // else a USE_ROW_TTL (0x10) cell inherits the ROW's
723                                // expiry (issue #1743). For a statement-level `USING
724                                // TTL` INSERT that expiry is the pk-liveness
725                                // localExpirationTime (`liveness_expires_at_seconds`,
726                                // from row `HAS_TTL`) — NOT `local_deletion_time`, the
727                                // GC-grace clock set only by a row tombstone
728                                // (`HAS_DELETION`), `None` for a plain TTL INSERT (so
729                                // the old pairing produced `None` → `TTL(col)` null).
730                                // Mirrors the #1741 shadow-path expiry resolution,
731                                // incl. the oa/da unsigned LDT fallback reinterpret.
732                                let use_row_ttl = (cell_flags & 0x10) != 0;
733                                let row_level_exp = if use_row_ttl {
734                                    let row_expiry =
735                                        row_header.liveness_expires_at_seconds.or_else(|| {
736                                            row_header.local_deletion_time.map(|s| {
737                                                if self.has_uint_deletion_time() {
738                                                    (s as u32) as i64
739                                                } else {
740                                                    s as i64
741                                                }
742                                            })
743                                        });
744                                    match (row_header.ttl, row_expiry) {
745                                        (Some(ttl_s), Some(exp_s)) => Some(CellExpiration {
746                                            ttl_seconds: ttl_s,
747                                            expires_at_seconds: exp_s,
748                                        }),
749                                        _ => None,
750                                    }
751                                } else {
752                                    None
753                                };
754                                let effective_exp = cell_exp.or(row_level_exp);
755                                meta_map.insert(
756                                    column.name.clone(),
757                                    CellWriteMetadata {
758                                        write_timestamp_micros: effective_ts,
759                                        expiration: effective_exp,
760                                    },
761                                );
762                            }
763                            // Issue #1334/#1642: interned name handle (Arc::clone),
764                            // pushed positionally in schema column order.
765                            cells.push((Arc::clone(&ctp.name), value));
766                        }
767                        offset = new_offset;
768                    }
769                    Err(e) => {
770                        tracing::debug!(
771                            "V5CompressedLegacy:   ✗ Column {} '{}' ({}) at offset {} FAILED: {}",
772                            col_idx,
773                            column.name,
774                            column.data_type,
775                            offset,
776                            e
777                        );
778                        // CRITICAL FIX: Stop parsing remaining columns when we hit an error
779                        // The offset doesn't advance here, but we exit the loop cleanly
780                        // rather than continuing with invalid offset
781                        break;
782                    }
783                }
784            }
785        }
786
787        // Issue #1741: stash the read-side shadowing aggregate onto the header.
788        row_header.max_data_cell_timestamp = agg_max_cell_ts;
789        row_header.max_data_cell_expires_at = agg_max_expires_at;
790        row_header.has_live_forever_data_cell = agg_has_live_forever;
791
792        tracing::debug!(
793            "V5CompressedLegacy: Parsed {}/{} on-disk columns (missing columns are NULL)",
794            cells.len(),
795            columns_in_order.len()
796        );
797        tracing::debug!(
798            "V5CompressedLegacy: Cell column names (positional order): {:?}",
799            cells.iter().map(|(name, _)| name).collect::<Vec<_>>()
800        );
801
802        debug!("V5CompressedLegacy: Parsed total of {} cells", cells.len());
803
804        // Calculate offset after cell data (based on row_size from header)
805        //
806        // CRITICAL (Issue #237): row_size is measured from AFTER the row_size VInt,
807        // not from where it starts. This matches Cassandra's getFilePointer() semantics:
808        //   next_position = row_size_value + position_after_reading_row_size_vint
809        //
810        // Formula: (row_metadata_offset + row_size_vint_len) + row_size
811        //
812        // There is NO trailing field in V5CompressedLegacy format - the next partition/row
813        // starts immediately after row_size bytes from this position.
814        let row_size_counted_from = row_metadata_offset + row_header.row_size_vint_len;
815        let after_cells_offset = row_size_counted_from + row_size as usize;
816
817        // Validate we have enough data
818        if after_cells_offset > data.len() {
819            let remaining = data.len().saturating_sub(row_size_counted_from);
820            return Err(Error::corruption(format!(
821                "V5CompressedLegacy: Not enough bytes for row data at offset {} (need {}, have {})",
822                row_size_counted_from, row_size, remaining
823            )));
824        }
825
826        // No trailing field - next partition/row starts immediately
827        let next_offset = after_cells_offset;
828
829        debug!(
830            "V5CompressedLegacy: Row complete - row_size={} bytes, next offset = {} (counted from {}, is_static={})",
831            row_size, next_offset, row_size_counted_from, is_static
832        );
833
834        Ok((
835            cells,
836            cell_meta,
837            Some(row_header),
838            next_offset,
839            is_static,
840            complex_col_meta,
841        ))
842    }

844    /// Test-only helper that parses the cell header (flags + conditional temporal
845    /// metadata) and returns the offset at which the value bytes begin.
846    ///
847    /// This mirrors the logic in `parse_cell_value_schema_order` for the conditional
848    /// sections (Steps 1-3), but stops before the value parse.  It is used by the
849    /// S1 audit verification tests (Issue #623) to confirm that:
850    ///   - USE_ROW_TIMESTAMP (0x08) causes the timestamp VInt to be ABSENT
851    ///   - USE_ROW_TTL (0x10) without IS_EXPIRING causes LDT/TTL to be ABSENT
852    ///
853    /// Returns `(flags, value_start_offset)`.
854    #[cfg(test)]
855    pub(super) fn parse_cell_header_end_offset(
856        &self,
857        data: &[u8],
858        start_offset: usize,
859    ) -> Result<(u8, usize)> {
860        const CELL_IS_DELETED: u8 = 0x01;
861        const CELL_IS_EXPIRING: u8 = 0x02;
862        const CELL_USE_ROW_TIMESTAMP: u8 = 0x08;
863        const CELL_USE_ROW_TTL: u8 = 0x10;
864
865        if start_offset >= data.len() {
866            return Err(Error::corruption(
867                "cell_header_end_offset: no flags byte".to_string(),
868            ));
869        }
870        let flags = data[start_offset];
871        let mut offset = start_offset + 1;
872
873        let is_deleted = (flags & CELL_IS_DELETED) != 0;
874        let is_expiring = (flags & CELL_IS_EXPIRING) != 0;
875        let use_row_timestamp = (flags & CELL_USE_ROW_TIMESTAMP) != 0;
876        let use_row_ttl = (flags & CELL_USE_ROW_TTL) != 0;
877
878        // Step 1: skip timestamp VInt if not using row timestamp.
879        // Skip-only: byte advancement is identical for vint/vuint, but use the
880        // UNSIGNED variant to match the writer encoding (roborev #863).
881        if !use_row_timestamp {
882            let (remaining, _ts_delta) = parse_vuint(&data[offset..]).map_err(|e| {
883                Error::corruption(format!(
884                    "cell_header_end_offset: failed to parse timestamp VInt: {:?}",
885                    e
886                ))
887            })?;
888            offset += data[offset..].len() - remaining.len();
889        }
890        // Step 2: skip LDT VUInt if not using row TTL and (deleted or expiring)
891        if !use_row_ttl && (is_deleted || is_expiring) {
892            let (remaining, _ldt_delta) = parse_vuint(&data[offset..]).map_err(|e| {
893                Error::corruption(format!(
894                    "cell_header_end_offset: failed to parse LDT VUInt: {:?}",
895                    e
896                ))
897            })?;
898            offset += data[offset..].len() - remaining.len();
899        }
900        // Step 3: skip TTL VUInt if not using row TTL and expiring
901        if !use_row_ttl && is_expiring {
902            let (remaining, _ttl_delta) = parse_vuint(&data[offset..]).map_err(|e| {
903                Error::corruption(format!(
904                    "cell_header_end_offset: failed to parse TTL VUInt: {:?}",
905                    e
906                ))
907            })?;
908            offset += data[offset..].len() - remaining.len();
909        }
910
911        Ok((flags, offset))
912    }
