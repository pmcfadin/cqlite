1//! Byte-comparable key decoding for Cassandra 5.0+ (CEP-25)
2//!
3//! This module implements decoding of byte-comparable encoded keys used in
4//! Cassandra 5.0's 'newbig' format. Byte-comparable encoding preserves the
5//! sort order of typed values when compared lexicographically as bytes.
6//!
7//! ## Format Overview
8//!
9//! Keys are encoded as a sequence of components separated by markers:
10//! - `0x40` (NEXT_COMPONENT): Separates components
11//! - `0x38` (TERMINATOR): Marks end of key
12//! - `0x3E` (NULL_MARKER): Represents null component
13//! - `0x00 0xFE`: Escape sequence for literal zero byte
14//! - `0x00 0xFF`: End-of-variable-length-data marker
15//!
16//! ## Type-Specific Encodings
17//!
18//! - **Fixed signed integers** (int, bigint): Sign bit flipped for ordering
19//! - **Text/Blob**: Escape sequences with `0x00 0xFF` terminator
20//! - **UUID**: Direct 16-byte encoding
21//!
22//! ## References
23//!
24//! - CEP-25: https://cwiki.apache.org/confluence/display/CASSANDRA/CEP-25
25//! - ByteComparable.md: https://github.com/apache/cassandra/blob/trunk/src/java/org/apache/cassandra/utils/bytecomparable/ByteComparable.md

30/// Byte-comparable format constants (from CEP-25)
31pub const NEXT_COMPONENT: u8 = 0x40;
32pub const TERMINATOR: u8 = 0x38;
33pub const NULL_MARKER: u8 = 0x3E;
34pub const ESCAPE_00: u8 = 0x00;
35pub const ESCAPE_FE: u8 = 0xFE; // Literal zero
36pub const ESCAPE_FF: u8 = 0xFF; // End of variable-length data
37
38/// Decode a byte-comparable encoded composite key
39///
40/// Returns a vector of component byte arrays, one for each key component.
41/// Each component is already decoded (unescaped, with markers removed).
42///
43/// # Arguments
44/// * `input` - Raw bytes containing byte-comparable encoded key
45///
46/// # Returns
47/// * `Ok((remaining, components))` - Remaining data and decoded components
48/// * `Err(_)` - Parse error with corruption details
49///
50/// # Format
51/// ```text
52/// [component_1_data] 0x40 [component_2_data] 0x40 ... 0x38
53/// ```
54pub fn decode_byte_comparable_key(input: &[u8]) -> IResult<&[u8], Vec<Vec<u8>>> {
55    let mut components = Vec::new();
56    let mut offset = 0;
57
58    loop {
59        if offset >= input.len() {
60            // Reached end without terminator - return what we have
61            break;
62        }
63
64        // Check for terminator first
65        if input[offset] == TERMINATOR {
66            offset += 1;
67            break;
68        }
69
70        // Extract next component
71        match extract_component(&input[offset..]) {
72            Ok((component_data, consumed)) => {
73                components.push(component_data);
74                offset += consumed;
75
76                // Check what comes after component
77                if offset < input.len() {
78                    match input[offset] {
79                        NEXT_COMPONENT => {
80                            // More components follow
81                            offset += 1;
82                        }
83                        TERMINATOR => {
84                            // End of key
85                            offset += 1;
86                            break;
87                        }
88                        _ => {
89                            // Unexpected byte - may be start of next component
90                            // Some formats may not have explicit separators
91                            continue;
92                        }
93                    }
94                }
95            }
96            Err(_e) => {
97                return Err(nom::Err::Error(nom::error::Error::new(
98                    &input[offset..],
99                    nom::error::ErrorKind::Fail,
100                )));
101            }
102        }
103    }
104
105    Ok((&input[offset..], components))
106}

108/// Extract a single component up to separator (0x40) or terminator (0x38)
109///
110/// Handles escape sequences for variable-length data.
111///
112/// # Returns
113/// * `Ok((data, consumed))` - Decoded component data and bytes consumed
114/// * `Err(_)` - Parse error
115fn extract_component(input: &[u8]) -> Result<(Vec<u8>, usize)> {
116    let mut data = Vec::new();
117    let mut i = 0;
118
119    while i < input.len() {
120        match input[i] {
121            NEXT_COMPONENT => {
122                // Component separator - end of component
123                return Ok((data, i));
124            }
125            TERMINATOR => {
126                // End of key - end of component
127                return Ok((data, i));
128            }
129            NULL_MARKER => {
130                // Null component
131                return Ok((vec![], i + 1));
132            }
133            ESCAPE_00 => {
134                // Escape sequence
135                if i + 1 >= input.len() {
136                    return Err(Error::corruption("Incomplete escape sequence"));
137                }
138                match input[i + 1] {
139                    ESCAPE_FE => {
140                        // Literal 0x00
141                        data.push(0x00);
142                        i += 2;
143                    }
144                    ESCAPE_FF => {
145                        // End of variable-length data (acts as component terminator)
146                        return Ok((data, i + 2));
147                    }
148                    _ => {
149                        // Unknown escape - treat as regular byte
150                        data.push(input[i]);
151                        i += 1;
152                    }
153                }
154            }
155            byte => {
156                data.push(byte);
157                i += 1;
158            }
159        }
160    }
161
162    // Reached end of input - return what we collected
163    Ok((data, i))
164}

166/// Decode a signed 32-bit integer component
167///
168/// Fixed-size encoding (4 bytes) with sign bit flipped for ordering.
169/// Negative numbers have high bit = 0, positive have high bit = 1.
170///
171/// **NOTE**: This function is used for schema-aware key decoding when type
172/// information is available from the schema. In schemaless parsing, keys are
173/// returned as raw bytes. The `#[allow(dead_code)]` annotation is present
174/// because this function is called conditionally based on schema availability.
175///
176/// # Arguments
177/// * `data` - 4-byte component data
178///
179/// # Returns
180/// * `Ok(value)` - Decoded i32 value
181/// * `Err(_)` - Invalid length or parse error
182#[allow(dead_code)] // Used for type-specific decoding when schema is available
183pub fn decode_int32_component(data: &[u8]) -> Result<i32> {
184    if data.len() != 4 {
185        return Err(Error::corruption(format!(
186            "Invalid int32 component length: {} (expected 4)",
187            data.len()
188        )));
189    }
190
191    // Read big-endian and un-flip sign bit
192    let encoded = u32::from_be_bytes([data[0], data[1], data[2], data[3]]);
193    let value = (encoded ^ 0x8000_0000) as i32;
194    Ok(value)
195}

197/// Decode a signed 64-bit integer component (bigint or timestamp)
198///
199/// Fixed-size encoding (8 bytes) with sign bit flipped for ordering.
200///
201/// **NOTE**: This function is used for schema-aware key decoding when type
202/// information is available. The `#[allow(dead_code)]` annotation is present
203/// because usage is conditional on schema availability.
204///
205/// # Arguments
206/// * `data` - 8-byte component data
207///
208/// # Returns
209/// * `Ok(value)` - Decoded i64 value
210/// * `Err(_)` - Invalid length or parse error
211#[allow(dead_code)] // Used for type-specific decoding when schema is available
212pub fn decode_bigint_component(data: &[u8]) -> Result<i64> {
213    if data.len() != 8 {
214        return Err(Error::corruption(format!(
215            "Invalid bigint component length: {} (expected 8)",
216            data.len()
217        )));
218    }
219
220    // Read big-endian and un-flip sign bit
221    let encoded = u64::from_be_bytes([
222        data[0], data[1], data[2], data[3], data[4], data[5], data[6], data[7],
223    ]);
224    let value = (encoded ^ 0x8000_0000_0000_0000) as i64;
225    Ok(value)
226}

228/// Decode a text component with escape sequences
229///
230/// Variable-length text with `0x00 0xFF` terminator and `0x00 0xFE` escaping.
231/// This function validates UTF-8 encoding.
232///
233/// **NOTE**: This function is used for schema-aware key decoding when type
234/// information is available. The `#[allow(dead_code)]` annotation is present
235/// because usage is conditional on schema availability.
236///
237/// # Arguments
238/// * `data` - Already-decoded component bytes (escape sequences removed)
239///
240/// # Returns
241/// * `Ok(text)` - Decoded UTF-8 string
242/// * `Err(_)` - Invalid UTF-8
243#[allow(dead_code)] // Used for type-specific decoding when schema is available
244pub fn decode_text_component(data: &[u8]) -> Result<String> {
245    String::from_utf8(data.to_vec())
246        .map_err(|e| Error::corruption(format!("Invalid UTF-8 in text component: {}", e)))
247}

253    #[test]
254    fn test_decode_simple_key() {
255        // Example: "AMZN" (4 bytes) + separator + date bytes + terminator
256        let input = &[
257            0x41, 0x4D, 0x5A, 0x4E, // "AMZN"
258            0x40, // NEXT_COMPONENT
259            0x80, 0x00, 0x00, 0x00, 0x00, 0x4F, 0x88, 0x00, // Encoded date (8 bytes)
260            0x38, // TERMINATOR
261        ];
262
263        let (remaining, components) = decode_byte_comparable_key(input).unwrap();
264        assert_eq!(components.len(), 2);
265        assert_eq!(components[0], b"AMZN");
266        assert_eq!(
267            components[1],
268            &[0x80, 0x00, 0x00, 0x00, 0x00, 0x4F, 0x88, 0x00]
269        );
270        assert_eq!(remaining.len(), 0);
271    }
