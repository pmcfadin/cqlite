1//! Cross-generation read reconciliation for the SSTable manager (issues #883,
2//! #885, #957, #1579).
3//!
4//! When a table directory holds more than one SSTable generation, plain
5//! concatenation of each reader's live rows is wrong: the same
6//! `(partition, clustering)` row can appear in several generations, and a
7//! row/cell tombstone in a newer generation suppresses only its OWN generation's
8//! copy — so concatenation duplicates overwritten rows and resurrects rows deleted
9//! in a later generation. These helpers reconcile across generations with the
10//! authoritative [`KWayMerger`](crate::storage::write_engine::KWayMerger) — the
11//! same last-write-wins + tombstone-shadowing logic compaction uses — so the read
12//! path returns Cassandra's merged, deduplicated, tombstone-honouring result.
13//!
14//! Three drivers, ONE reconciliation kernel (the merger's `step()`):
15//!
16//! - [`merge_generations_for_read`] — materializing plain read (`scan`,
17//!   partition-targeted point reads). Collects the reconciled rows into a `Vec`
18//!   in Cassandra token order.
19//! - [`merge_generations_for_read_with_metadata`] — the `WRITETIME`/`TTL`
20//!   projection sibling; additionally surfaces the winning cell's per-cell write
21//!   metadata.
22//! - [`stream_generations_for_read`] — the STREAMING plain read (`scan_stream`,
23//!   issue #1579 / D3). Feeds each stepped partition straight into a bounded
24//!   channel instead of collecting the whole table, so live heap is O(one
25//!   partition + channel) and time-to-first-row is O(first partition), not
26//!   O(full merge).
27//!
28//! All three yield partitions in the merger's `DecoratedKey` order = `(token,
29//! key)`, which is byte-identical to [`scan_merge::sort_by_token_order`]. The
30//! materializing paths still apply that stable sort as a no-op guard; the
31//! streaming path relies on the merger's order directly (issue #1579).
32//!
33//! Extracted from `sstable/mod.rs` (issue #1116 campsite split): behaviour of the
34//! two materializing helpers is unchanged apart from the new `target_key`
35//! partition-targeting parameter (issue #1579, point-read path).
36//!
37//! The whole module is gated on `write-support` at its `mod` declaration in the
38//! parent (`sstable/mod.rs`).

63/// Post-merge read-time visibility for the multi-generation read path (issue #1849).
64///
65/// A `SELECT` over a table directory with more than one SSTable generation routes
66/// through the [`KWayMerger`], which performs cross-generation last-write-wins +
67/// tombstone RECONCILIATION but NOT read-time TTL expiry / partition-deletion
68/// visibility. Historically that let a multi-gen read return TTL-expired (or
69/// partition-shadowed) cells as live, even after #1741 fixed the single-gen path.
70///
71/// This runs the merger's already-reconciled output through the SAME single-gen
72/// [`PartitionShadow`] per-cell decision (`cell_shadowed_or_expired`) and row-level
73/// partition-shadow decision (`merged_row_shadowed_by_partition`), so there is ONE
74/// read-visibility implementation across single- and multi-generation reads. It adds
75/// READ visibility only — reconciliation and the compaction/write path are untouched
76/// (AC6). Range-tombstone shadowing of covered older cells is already applied by the
77/// merger (#933); read-time cell/row TTL expiry is the gap this closes.
78struct ReadShadow {
79    /// Read-time TTL clock (epoch seconds), captured ONCE per scan so a scan crossing
80    /// an expiration-second boundary decides every row with the same `now` — matching
81    /// the single-gen parser's per-scan clock capture (issue #1849 AC4). Honours the
82    /// `CQLITE_TTL_NOW_OVERRIDE_SECS` debug test seam via [`now_epoch_secs`].
83    now_secs: i64,
84    /// Names of the primary-key (partition + clustering) columns, surfaced by the
85    /// compaction reader as pseudo-cells. Excluded from the row-level DATA-cell
86    /// timestamp aggregate so only real data cells drive the partition-shadow row
87    /// decision — matching the single-gen fold, which ignores pk/ck pseudo-cells.
88    key_columns: HashSet<String>,
89}
90
91impl ReadShadow {
92    fn new(schema: &crate::schema::TableSchema, now_secs: i64) -> Self {
93        let mut key_columns =
94            HashSet::with_capacity(schema.partition_keys.len() + schema.clustering_keys.len());
95        for k in &schema.partition_keys {
96            key_columns.insert(k.name.clone());
97        }
98        for k in &schema.clustering_keys {
99            key_columns.insert(k.name.clone());
100        }
101        Self {
102            now_secs,
103            key_columns,
104        }
105    }
106
107    /// Filter one merged `RowData::Live` row's cells for read visibility given the
108    /// partition-tombstone `cover` (`markedForDeleteAt` µs, or `None`). Returns `None`
109    /// when the WHOLE row is hidden (partition-shadowed), else the surviving cells
110    /// with cell tombstones plus TTL-expired / partition-shadowed data cells dropped.
111    fn filter_live(&self, cover: Option<i64>, cells: Vec<CellData>) -> Option<Vec<CellData>> {
112        let now = self.now_secs;
113        let mut kept = Vec::with_capacity(cells.len());
114        // Fold over REAL data cells (kept AND dropped) so a partition tombstone that
115        // shadows every data cell is recognised at the row level — mirrors the
116        // single-gen `agg_max_cell_ts` fold in `row_data.rs`.
117        let mut max_data_ts: Option<i64> = None;
118        for cell in cells {
119            // A cell tombstone is never live data (a deleted column is absent).
120            if matches!(cell.value, Value::Tombstone(_)) {
121                continue;
122            }
123            // Primary-key (partition + clustering) pseudo-cells are STRUCTURAL: the
124            // compaction reader surfaces clustering columns as simple cells and both
125            // the single-gen read loop and the merger's `apply_partition_shadowing`
126            // leave them untouched (never subjected to the read-time shadow/expiry
127            // drop). Keep them verbatim and out of the `max_data_ts` fold, so a row
128            // kept alive by a newer DATA cell under a partition tombstone never loses
129            // its clustering-key value (a pseudo-cell whose own `ts <= cover` must NOT
130            // be stripped) — otherwise the multi-gen path would emit a malformed row
131            // diverging from the single-gen path (issue #1849; roborev multi-gen finding).
132            if self.key_columns.contains(&cell.column) {
133                kept.push(cell);
134                continue;
135            }
136            let eff_ts = Some(cell.timestamp);
137            let eff_exp = cell_expiry_secs(&cell);
138            let dropped = PartitionShadow::cell_shadowed_or_expired(cover, now, eff_ts, eff_exp);
139            if let Some(t) = eff_ts {
140                max_data_ts = Some(max_data_ts.map_or(t, |m| m.max(t)));
141            }
142            if !dropped {
143                kept.push(cell);
144            }
145        }
146        if merged_row_shadowed_by_partition(cover, max_data_ts) {
147            return None;
148        }
149        Some(kept)
150    }
151}

153/// The read-time expiry instant (epoch seconds) of a merged cell, or `None` when it
154/// is not an expiring cell. An expiring cell's `localDeletionTime` is its
155/// `localExpirationTime`; it is reinterpreted UNSIGNED so a post-2038 expiry stored
156/// as a negative `i32` bit pattern (oa/da `hasUIntDeletionTime`) is not wrapped
157/// negative and wrongly treated as long-expired — matching the single-gen unsigned
158/// LDT handling in `row_data.rs` (issue #1849 AC4).
159fn cell_expiry_secs(cell: &CellData) -> Option<i64> {
160    cell.ttl?;
161    cell.local_deletion_time.map(|s| (s as u32) as i64)
162}

164/// The partition-level `markedForDeleteAt` (µs) carried by the synthetic
165/// partition-tombstone carrier entry of a merged partition (issue #1072), or `None`
166/// when the partition has no tombstone. Used as the read-side partition-shadow cover
167/// across the partition's rows (issue #1849).
168fn partition_cover(rows: &[MergeEntry]) -> Option<i64> {
169    rows.iter()
170        .find_map(|e| e.partition_deletion.map(|(mfda, _ldt)| mfda))
171}

173/// Convert one reconciled partition's merge rows into the live scan rows the read
174/// path emits: drop cell tombstones, drop row tombstones, drop rows left empty.
175///
176/// The single emission kernel shared by the materializing plain merge and the
177/// streaming driver so tombstone filtering can never drift between them (issue
178/// #1334: emit the interned-name `ScanRow` carrier the read path consumes).
179fn partition_live_rows(
180    row_key: &RowKey,
181    rows: Vec<MergeEntry>,
182    shadow: &ReadShadow,
183) -> Vec<(RowKey, ScanRow)> {
184    // Issue #1849: partition-tombstone cover for this partition's rows, applied as a
185    // read-side shadow floor alongside per-cell read-time TTL expiry (post-merge).
186    let cover = partition_cover(&rows);
187    let mut out = Vec::new();
188    for entry in rows {
189        match entry.row_data {
190            RowData::Live { cells } => {
191                // Read-time visibility: drop cell tombstones + TTL-expired /
192                // partition-shadowed data cells; `None` hides the whole row.
193                let Some(surviving) = shadow.filter_live(cover, cells) else {
194                    continue;
195                };
196                let row_cells: RowCells = surviving
197                    .into_iter()
198                    .map(|c| (Arc::from(c.column.as_str()), c.value))
199                    .collect();
200                if !row_cells.is_empty() {
201                    out.push((row_key.clone(), ScanRow::Row(row_cells)));
202                }
203            }
204            // Row tombstone: the row is deleted across all generations — suppress.
205            RowData::Tombstone { .. } => {}
206        }
207    }
208    out
209}

211/// Reconcile multiple SSTable generations into the single authoritative live-row
212/// set, returning them materialized in Cassandra token order (issue #883).
213///
214/// This drives the same [`KWayMerger`](crate::storage::write_engine::KWayMerger)
215/// the compaction path uses, so reconciliation is byte-for-byte the
216/// last-write-wins + tombstone-shadowing logic (`merge_partition_rows`): per-cell
217/// LWW by write timestamp, row/cell tombstones shadow older cells, and
218/// fully-deleted rows are dropped. The merger manages its own reader
219/// threads/runtimes internally, so it runs on a blocking task.
220///
221/// `start_key`/`end_key` bound the merged output to the same inclusive
222/// `[start_key, end_key]` key range the per-reader
223/// [`scan`](reader::SSTableReader::scan) applies (skip `key < start`, skip
224/// `key > end`, using `RowKey`'s `Ord`) — Issue #957. The range filter runs
225/// before `limit`, matching the per-reader scan order (range then limit).
226///
227/// `target_key` restricts the output to a SINGLE partition (the point-read path,
228/// issue #1579): keep only the partition whose raw key bytes equal `target_key`
229/// and STOP as soon as it is found. Partition keys are unique in the merger's
230/// output, so once the target is seen nothing later can match — this avoids
231/// converting every other partition to `ScanRow` and decoding past the target,
232/// while remaining byte-identical to the caller's former
233/// `retain(|r| r.key == partition_key)`. `target_key` and the range bounds are
234/// not combined by any current caller (point reads pass `None`/`None` bounds).
235///
236/// With `None` target and `None`/`None` bounds the output is byte-for-byte the
237/// full reconciled set.
238pub(super) async fn merge_generations_for_read(
239    reader_list: &[Arc<reader::SSTableReader>],
240    schema: &crate::schema::TableSchema,
241    start_key: Option<&RowKey>,
242    end_key: Option<&RowKey>,
243    limit: Option<usize>,
244    target_key: Option<&RowKey>,
245) -> Result<Vec<(RowKey, ScanRow)>> {
246    // Own the bounds/target so the merge body can use them without borrowing
247    // across the await; cheap clone of the key bytes.
248    let start_key = start_key.cloned();
249    let end_key = end_key.cloned();
250    let target_key = target_key.map(|k| k.as_bytes().to_vec());
251
252    let paths = ordered_generation_paths(reader_list);
253    let schema = schema.clone();
254
255    let mut merged = tokio::task::spawn_blocking(move || -> Result<Vec<(RowKey, ScanRow)>> {
256        // Issue #1849: capture the read-time TTL clock ONCE per scan.
257        let shadow = ReadShadow::new(&schema, now_epoch_secs());
258        let mut merger = KWayMerger::new(paths, &schema)?;
259        let mut out = Vec::new();
260        while let MergeStep::Partition { key, rows } = merger.step()? {
261            let row_key = RowKey::new(key.key.clone());
262
263            // Partition-targeted point read (#1579): keep ONLY the target and stop.
264            if let Some(ref target) = target_key {
265                if row_key.as_bytes() != target.as_slice() {
266                    continue;
267                }
268                out.extend(partition_live_rows(&row_key, rows, &shadow));
269                break;
270            }
271
272            // Full / range scan: inclusive [start, end] on `RowKey` order (#957).
273            if let Some(ref start) = start_key {
274                if &row_key < start {
275                    continue;
276                }
277            }
278            if let Some(ref end) = end_key {
279                if &row_key > end {
280                    continue;
281                }
282            }
283            out.extend(partition_live_rows(&row_key, rows, &shadow));
284        }
285        Ok(out)
286    })
287    .await
288    .map_err(|e| crate::Error::Storage(format!("cross-generation read merge task: {e}")))??;
289
290    // Match the plain-scan contract: the merger already emits partitions in
291    // Cassandra token order with clustering rows contiguous within a partition.
292    // Preserve that with a stable TOKEN-order sort (issue #1580), then apply
293    // LIMIT. The sort is a no-op ordering-wise when the input is already
294    // token-ordered; it only guards against a stray divergence.
295    scan_merge::sort_by_token_order(&mut merged, limit, |(k, _)| k);
296    Ok(merged)
297}

299/// Metadata-aware sibling of [`merge_generations_for_read`] for the
300/// `WRITETIME(col)` / `TTL(col)` projection path (Issue #885).
301///
302/// Reconciles multiple SSTable generations with the same
303/// [`KWayMerger`](crate::storage::write_engine::KWayMerger) (per-cell LWW +
304/// row/cell tombstone shadowing) and additionally surfaces the **winning** cell's
305/// per-cell write metadata:
306///
307/// - `write_timestamp_micros` comes straight from the winning `CellData`
308///   (`reconcile_cluster` keeps each surviving cell's own timestamp), so it is the
309///   WRITETIME of the cell that actually won cross-generation LWW.
310/// - `expiration` (TTL) is recovered best-effort from the per-reader
311///   `scan_with_cell_metadata` outputs: for each surviving `(key, column)` we take
312///   the newest reader-surfaced metadata and attach its expiration only when its
313///   timestamp matches the merge winner. Absent/mismatched ⇒ `None`.
314///
315/// `start_key`/`end_key`/`target_key`/`limit` behave exactly as in
316/// [`merge_generations_for_read`], keeping this definitionally in lockstep with
317/// the plain helper. When `target_key` is `Some`, the best-effort TTL scan is also
318/// bounded to that single partition (issue #1579, point-read path).
319pub(super) async fn merge_generations_for_read_with_metadata(
320    reader_list: &[Arc<reader::SSTableReader>],
321    schema: &crate::schema::TableSchema,
322    start_key: Option<&RowKey>,
323    end_key: Option<&RowKey>,
324    limit: Option<usize>,
325    target_key: Option<&RowKey>,
326) -> Result<Vec<(RowKey, ScanRow, HashMap<String, CellWriteMetadata>)>> {
327    // Own the bounds so the merge body can use them without borrowing across the
328    // await; cheap clone of the key bytes. Mirrors the plain helper.
329    let owned_start = start_key.cloned();
330    let owned_end = end_key.cloned();
331    let target_bytes = target_key.map(|k| k.as_bytes().to_vec());
332
333    // Best-effort TTL source: gather each reader's own per-cell metadata and keep,
334    // per (row-key bytes, column), the entry with the newest write timestamp. The
335    // merger surfaces accurate WRITETIME but no TTL, so this recovers expiration
336    // for the winning cell when the reader format carries it. For a point read the
337    // scan is bounded to the target partition (#1579); otherwise to the read range.
338    let table_id = CqlTableId::from(format!("{}.{}", schema.keyspace, schema.table).as_str());
339    let mut ttl_lookup: HashMap<(Vec<u8>, String), CellWriteMetadata> = HashMap::new();
340    for reader in reader_list {
341        let (ttl_start, ttl_end): (Option<&RowKey>, Option<&RowKey>) = match target_key {
342            Some(t) => (Some(t), Some(t)),
343            None => (owned_start.as_ref(), owned_end.as_ref()),
344        };
345        let per_reader = reader
346            .scan_with_cell_metadata(&table_id, ttl_start, ttl_end, None, Some(schema))
347            .await?;
348        for (row_key, _value, meta) in per_reader {
349            for (column, cell_meta) in meta {
350                ttl_lookup
351                    .entry((row_key.as_bytes().to_vec(), column))
352                    .and_modify(|existing| {
353                        if cell_meta.write_timestamp_micros > existing.write_timestamp_micros {
354                            *existing = cell_meta.clone();
355                        }
356                    })
357                    .or_insert(cell_meta);
358            }
359        }
360    }
361
362    let paths = ordered_generation_paths(reader_list);
363    let merge_schema = schema.clone();
364    let start_key = owned_start;
365    let end_key = owned_end;
366    let target_for_merge = target_bytes;
367
368    let merged_rows = tokio::task::spawn_blocking(move || -> Result<Vec<MergedMetaRow>> {
369        // Issue #1849: capture the read-time TTL clock ONCE per scan.
370        let shadow = ReadShadow::new(&merge_schema, now_epoch_secs());
371        let mut merger = KWayMerger::new(paths, &merge_schema)?;
372        let mut out = Vec::new();
373        while let MergeStep::Partition { key, rows } = merger.step()? {
374            let row_key = RowKey::new(key.key.clone());
375
376            // Partition-targeted point read (#1579): keep ONLY the target and stop.
377            if let Some(ref target) = target_for_merge {
378                if row_key.as_bytes() != target.as_slice() {
379                    continue;
380                }
381                push_metadata_rows(&key.key, rows, &mut out, &shadow);
382                break;
383            }
384
385            // Full / range scan: inclusive [start, end] on `RowKey` order (#957).
386            if let Some(ref start) = start_key {
387                if &row_key < start {
388                    continue;
389                }
390            }
391            if let Some(ref end) = end_key {
392                if &row_key > end {
393                    continue;
394                }
395            }
396            push_metadata_rows(&key.key, rows, &mut out, &shadow);
397        }
398        Ok(out)
399    })
400    .await
401    .map_err(|e| crate::Error::Storage(format!("cross-generation metadata merge task: {e}")))??;
402
403    // Attach per-cell metadata: WRITETIME from the merge winner, TTL recovered from
404    // the reader lookup only when its timestamp matches the winner.
405    let mut results: Vec<(RowKey, ScanRow, HashMap<String, CellWriteMetadata>)> =
406        Vec::with_capacity(merged_rows.len());
407    for (key_bytes, value, timestamps) in merged_rows {
408        let mut meta_map: HashMap<String, CellWriteMetadata> =
409            HashMap::with_capacity(timestamps.len());
410        for (column, write_ts) in timestamps {
411            let expiration = ttl_lookup
412                .get(&(key_bytes.clone(), column.clone()))
413                .filter(|m| m.write_timestamp_micros == write_ts)
414                .and_then(|m| m.expiration.clone());
415            meta_map.insert(
416                column,
417                CellWriteMetadata {
418                    write_timestamp_micros: write_ts,
419                    expiration,
420                },
421            );
422        }
423        results.push((RowKey::new(key_bytes), value, meta_map));
424    }
425
426    // Stable TOKEN-order sort (issue #1580), then LIMIT — identical ordering to the
427    // plain merge path; the metadata payload rides in the tuple's tail.
428    scan_merge::sort_by_token_order(&mut results, limit, |(k, _, _)| k);
429    Ok(results)
430}

432/// Convert one reconciled partition's merge rows into metadata-carrying rows: the
433/// live cells plus each surviving cell's `(column, write_timestamp)`. Shared by the
434/// range and point-targeted branches of the metadata merge so the tombstone
435/// filtering + timestamp capture live once.
436fn push_metadata_rows(
437    key_bytes: &[u8],
438    rows: Vec<MergeEntry>,
439    out: &mut Vec<MergedMetaRow>,
440    shadow: &ReadShadow,
441) {
442    // Issue #1849: same read-side partition cover + per-cell TTL/shadow filter as the
443    // plain path, so the WRITETIME/TTL projection agrees with `SELECT *`.
444    let cover = partition_cover(&rows);
445    for entry in rows {
446        if let RowData::Live { cells } = entry.row_data {
447            // `filter_live` drops cell tombstones + TTL-expired / partition-shadowed
448            // data cells; `None` hides the whole row (partition-shadowed).
449            let Some(surviving) = shadow.filter_live(cover, cells) else {
450                continue;
451            };
452            let mut row_cells: RowCells = Vec::with_capacity(surviving.len());
453            let mut timestamps: Vec<(String, i64)> = Vec::with_capacity(surviving.len());
454            for c in surviving {
455                timestamps.push((c.column.clone(), c.timestamp));
456                row_cells.push((Arc::from(c.column.as_str()), c.value));
457            }
458            if !row_cells.is_empty() {
459                out.push((key_bytes.to_vec(), ScanRow::Row(row_cells), timestamps));
460            }
461        }
462        // Row tombstones suppress the row entirely (no emission).
463    }
464}

466/// STREAMING cross-generation reconciliation for `scan_stream` (issue #1579 / D3).
467///
468/// Runs the authoritative [`KWayMerger`](crate::storage::write_engine::KWayMerger)
469/// on a blocking task and feeds each stepped partition's live rows STRAIGHT into a
470/// bounded channel via `blocking_send` (backpressure preserved), instead of
471/// collecting the entire reconciled table, sorting it, and dribbling it. Live heap
472/// is therefore O(one partition + channel), and the first row is available after
473/// the first partition rather than after the whole merge.
474///
475/// Reconciliation is byte-identical to [`merge_generations_for_read`] (shared
476/// `partition_live_rows` kernel), and the emission order is the merger's
477/// `DecoratedKey` = `(token, key)` order — byte-identical to the collect+sort
478/// path's [`scan_merge::sort_by_token_order`] output (issue #1579 ordering
479/// guardrail).
480///
481/// Construction (`KWayMerger::new`, which opens the input files) happens on the
482/// blocking task; its success/failure is signalled back over a oneshot BEFORE any
483/// streaming, so the caller can FALL BACK to the lazy per-reader streaming merge on
484/// a construction error exactly as the materializing `scan` falls back to
485/// concatenation. A runtime `step()` error mid-stream is delivered as an `Err`
486/// item on the channel (the consumer sees it), matching the lazy path's read-error
487/// behaviour.
488///
489/// This is a deliberate, documented error-path asymmetry (issue #1579): the
490/// caller's fallback-to-concatenation only ever applies to the CONSTRUCTION
491/// failure above (nothing has been streamed yet, so falling back cannot mix
492/// reconciled and unreconciled rows). A `step()` failure after some partitions
493/// were already emitted downstream is NEVER retried/fallen-back — it ends the
494/// stream via the `Err` channel item instead. That is safer than it sounds: the
495/// materializing `merge_generations_for_read` has no equivalent mid-collection
496/// failure signal — a `step()` error there simply propagates the whole call as
497/// `Err` before any partial `Vec` is returned to the caller. The streaming
498/// driver's `Err` item preserves the same "no half-reconciled result surfaces
499/// silently" guarantee while still letting the caller observe exactly how many
500/// good rows it already received.
501#[cfg(not(feature = "tombstones"))]
502pub(super) async fn stream_generations_for_read(
503    reader_list: &[Arc<reader::SSTableReader>],
504    schema: &crate::schema::TableSchema,
505    start_key: Option<&RowKey>,
506    end_key: Option<&RowKey>,
507    buffer_size: usize,
508) -> Result<mpsc::Receiver<Result<(RowKey, ScanRow)>>> {
509    let start_key = start_key.cloned();
510    let end_key = end_key.cloned();
511    let paths = ordered_generation_paths(reader_list);
512    let schema = schema.clone();
513
514    let (ready_tx, ready_rx) = oneshot::channel::<Result<()>>();
515    let (out_tx, out_rx) = mpsc::channel::<Result<(RowKey, ScanRow)>>(buffer_size.max(1));
516
517    tokio::task::spawn_blocking(move || {
518        // Issue #1849: capture the read-time TTL clock ONCE per scan.
519        let shadow = ReadShadow::new(&schema, now_epoch_secs());
520        let mut merger = match KWayMerger::new(paths, &schema) {
521            Ok(m) => {
522                // Signal readiness; if the caller already dropped, stop.
523                if ready_tx.send(Ok(())).is_err() {
524                    return;
525                }
526                m
527            }
528            Err(e) => {
529                // Let the caller fall back to the lazy per-reader streaming merge.
530                let _ = ready_tx.send(Err(e));
531                return;
532            }
533        };
534
535        loop {
536            let step = match merger.step() {
537                Ok(s) => s,
538                Err(e) => {
539                    let _ = out_tx.blocking_send(Err(e));
540                    return;
541                }
542            };
543            let (key, rows) = match step {
544                MergeStep::Partition { key, rows } => (key, rows),
545                MergeStep::Complete => return,
546            };
547
548            let row_key = RowKey::new(key.key.clone());
549            if let Some(ref start) = start_key {
550                if &row_key < start {
551                    continue;
552                }
553            }
554            if let Some(ref end) = end_key {
555                if &row_key > end {
556                    continue;
557                }
558            }
559
560            let live = partition_live_rows(&row_key, rows, &shadow);
561            // Issue #1579: the streaming producer holds ONE partition's rows
562            // resident at a time — record the window (not the whole table) so the
563            // memory guard observes O(window).
564            stream_merge_probe::record_resident(live.len() as u64);
565            for entry in live {
566                if out_tx.blocking_send(Ok(entry)).is_err() {
567                    return; // consumer dropped
568                }
569            }
570        }
571    });
572
573    match ready_rx.await {
574        Ok(Ok(())) => Ok(out_rx),
575        Ok(Err(e)) => Err(e),
576        Err(_) => Err(crate::Error::Storage(
577            "cross-generation streaming merge task ended before signalling readiness".to_string(),
578        )),
579    }
580}
581
582/// The merger expects inputs ordered newest → oldest (run_index 0 = newest) for
583/// its stable tie-break; the reader `Vec` order is discovery-dependent, so sort
584/// explicitly by generation descending and collect the input `Data.db` paths.
585fn ordered_generation_paths(reader_list: &[Arc<reader::SSTableReader>]) -> Vec<PathBuf> {
586    let mut ordered: Vec<&Arc<reader::SSTableReader>> = reader_list.iter().collect();
587    ordered.sort_by(|a, b| b.generation.cmp(&a.generation));
588    ordered.iter().map(|r| r.file_path()).collect()
589}

598    fn shadow(now_secs: i64) -> ReadShadow {

612    fn live_cell(column: &str, ts: i64) -> CellData {

616    fn expiring_cell(column: &str, ts: i64, ldt: i32) -> CellData {
617        let mut c = CellData::new(column.to_string(), Value::Integer(1), ts);

619        c.local_deletion_time = Some(ldt);

622
623    /// A TTL-expired data cell (past `localDeletionTime`) is dropped while a
624    /// live-forever sibling survives; the row itself stays (not partition-shadowed).
625    #[test]
626    fn filter_live_drops_ttl_expired_cell_keeps_live() {
627        let now = 2_000_000i64;
628        let cells = vec![
629            live_cell("name", 100),
630            expiring_cell("token", 100, 1_000), // expired: 1000 <= now
631        ];
632        let kept = shadow(now).filter_live(None, cells).expect("row visible");
633        let names: Vec<&str> = kept.iter().map(|c| c.column.as_str()).collect();
634        assert_eq!(names, vec!["name"], "expired `token` must be dropped");
635    }

637    /// A cell tombstone is never live data (dropped like the single-gen path).
638    #[test]
639    fn filter_live_drops_cell_tombstone() {
640        let mut tomb = CellData::new("gone".to_string(), Value::Integer(0), 100);
641        tomb.value = Value::Tombstone(Box::new(crate::types::TombstoneInfo {
642            deletion_time: 100,
643            tombstone_type: crate::types::TombstoneType::CellTombstone,
644            local_deletion_time: 0,
645            ttl: None,
646            range_start: None,
647            range_end: None,
648        }));
649        let kept = shadow(0)

656    /// A row whose every data cell is shadowed by the partition tombstone (all data
657    /// older than `markedForDeleteAt`) is hidden entirely (`None`); a row with a cell
658    /// strictly newer than the deletion survives.
659    #[test]
660    fn filter_live_partition_cover_hides_fully_shadowed_row() {
661        let cover = Some(2_000i64);
662        // All data older/equal to the cover → whole row hidden.
663        let hidden =
664            shadow(0).filter_live(cover, vec![live_cell("a", 1_000), live_cell("b", 2_000)]);
665        assert!(hidden.is_none(), "fully-shadowed row must be hidden");
666        // A cell strictly newer than the cover → row survives (and keeps newer cell).
667        let kept = shadow(0)
668            .filter_live(cover, vec![live_cell("a", 1_000), live_cell("b", 3_000)])
669            .expect("row visible");
670        let names: Vec<&str> = kept.iter().map(|c| c.column.as_str()).collect();
671        assert_eq!(names, vec!["b"], "older `a` shadowed, newer `b` survives");
672    }

674    /// A row kept alive by a DATA cell newer than the partition tombstone (a
675    /// post-delete resurrecting UPDATE) must retain its clustering-key pseudo-cell
676    /// even when that pseudo-cell's own write timestamp is `<= cover`. The shadow/
677    /// expiry drop is STRUCTURALLY skipped for key columns (matching the single-gen
678    /// path + the merger), so the surviving row is never emitted missing its
679    /// clustering-key value (issue #1849; roborev multi-gen finding).
680    #[test]
681    fn filter_live_keeps_clustering_key_under_partition_cover() {
682        let cover = Some(2_000i64);
683        // `ck` is a clustering pseudo-cell with ts <= cover (would be dropped if the
684        // shadow test were applied to it); `data` is a newer resurrecting cell; `old`
685        // is a stale data cell that MUST be shadowed away.
686        let cells = vec![
687            live_cell("ck", 1_000),
688            live_cell("old", 1_000),
689            live_cell("data", 3_000),
690        ];
691        let kept = shadow_with_keys(0, &["ck"])
692            .filter_live(cover, cells)
693            .expect("row survives via newer `data` cell");
694        let names: Vec<&str> = kept.iter().map(|c| c.column.as_str()).collect();
695        assert!(
696            names.contains(&"ck"),
697            "clustering-key pseudo-cell must be retained, got {names:?}"
698        );
699        assert!(
700            names.contains(&"data"),
701            "newer resurrecting data cell must survive, got {names:?}"
702        );
703        assert!(
704            !names.contains(&"old"),
705            "stale data cell shadowed by the partition tombstone must be dropped, got {names:?}"
706        );
707    }
