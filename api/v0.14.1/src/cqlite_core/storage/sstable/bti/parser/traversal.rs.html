1//! Full trie traversal (DFS) primitives — issue #832.
2//!
3//! These free functions operate purely on in-memory trie bytes plus a root
4//! offset (mirroring [`walk_bti_trie`](super::partitions::walk_bti_trie)).  They
5//! enumerate *every* leaf/payload in **byte-comparable order** by performing an
6//! explicit, stack-based depth-first search whose termination bounds are all
7//! relative to the trie's own byte length (see [`dfs_collect_in_order`]).
8//!
9//! In-order semantics (matches Cassandra `Walker`/`ReverseValueIterator`):
10//!   - A node that carries its own payload sorts BEFORE all of its children
11//!     (the key that terminates here is a prefix of every continuation).
12//!   - Children are visited in ascending transition-byte order.
13//!
14//! The reconstructed key for each emitted entry is the concatenation of the
15//! transition bytes from the root down to (and including) the node that carries
16//! the payload.  This is a *byte-comparable / token* encoding, NOT the original
17//! partition/clustering key — callers that need the original key must resolve it
18//! from Data.db.  The payload OFFSETS, however, are definitive.
19

28/// Read the root offset from the 8-byte big-endian footer of a BTI file and
29/// load the entire trie (everything before the footer) into memory.
30///
31/// Real Cassandra 5.0 BTI files (`Partitions.db` / `Rows.db`) have **no
32/// header** — the root node's absolute offset is the last 8 bytes of the file.
33/// This is the footer-based loader used by the traversal iterators; it does NOT
34/// rely on any whole-file BTI header (a real BTI file has none — the root offset
35/// lives in the footer, not a byte-0 header).
36///
37/// Returns `(trie_data, root_offset)`.
38pub(crate) fn load_bti_trie_via_footer<R: Read + Seek>(
39    reader: &mut R,
40) -> BtiResult<(Vec<u8>, usize)> {
41    let file_size = reader.seek(SeekFrom::End(0))?;
42    if file_size < 8 {
43        return Err(Error::Parse(format!(
44            "BTI file too small ({file_size} bytes; need at least 8 for footer)"
45        )));
46    }
47
48    reader.seek(SeekFrom::End(-8))?;
49    let mut footer = [0u8; 8];
50    reader.read_exact(&mut footer)?;
51    let root_offset = u64::from_be_bytes(footer);
52
53    let trie_size = file_size - 8;
54    if root_offset >= trie_size {
55        return Err(Error::Parse(format!(
56            "BTI file: root_offset {root_offset} >= trie_size {trie_size}"
57        )));
58    }
59
60    reader.seek(SeekFrom::Start(0))?;
61    let mut trie_data = vec![0u8; trie_size as usize];
62    reader.read_exact(&mut trie_data)?;
63    Ok((trie_data, root_offset as usize))
64}

66/// Collect the ascending-transition-byte child list `(transition_byte, child_offset)`
67/// for a parsed BTI node, ready for in-order DFS.
68///
69/// Crucially this iterates `Dense` children directly by index (transition byte =
70/// `start_byte + i`) and **skips** the `None` slots, which are the Dense "no
71/// transition" sentinels (raw delta `0`).  A `Some(ptr)` slot is a real child
72/// and is always emitted — *including* one whose absolute offset is `0` (the
73/// first-written leaf in BTI's bottom-up layout legitimately lives at offset 0).
74/// (The former `get_transitions` helper returned an empty Vec for Dense nodes
75/// and must NOT be used for traversal — now removed.)
76///
77/// `Sparse` transitions are returned in their stored order; the parser preserves
78/// the on-disk ascending order, and we sort defensively.
79///
80/// Shared with [`super::rows_floor`], whose O(key-length) floor/ceiling walk
81/// follows the SAME ascending `(transition_byte, absolute_child_offset)` child
82/// order the DFS uses (issue #1647 / L1).
83pub(super) fn ordered_children(node: &BtiNode) -> Vec<(u8, usize)> {
84    match &node.data {
85        BtiNodeData::PayloadOnly { .. } => Vec::new(),
86        BtiNodeData::Single { transition } => {
87            vec![(transition.byte, transition.child.distance as usize)]
88        }
89        BtiNodeData::Sparse { transitions } => {
90            let mut out: Vec<(u8, usize)> = transitions
91                .iter()
92                .map(|t| (t.byte, t.child.distance as usize))
93                .collect();
94            // Defensive: BTI Sparse transitions are stored ascending; enforce it.
95            out.sort_by_key(|&(b, _)| b);
96            out
97        }
98        BtiNodeData::Dense {
99            start_byte,
100            children,
101        } => {
102            let mut out = Vec::new();
103            for (i, child) in children.iter().enumerate() {
104                // `None` is the Dense "no transition" sentinel (raw delta 0):
105                // skip it.  `Some(ptr)` is a real child and is always emitted,
106                // even when `ptr.distance == 0` — a real child can live at
107                // absolute trie offset 0 (the first-written leaf).
108                if let Some(ptr) = child {
109                    let transition_byte = start_byte.wrapping_add(i as u8);
110                    out.push((transition_byte, ptr.distance as usize));
111                }
112            }
113            out
114        }
115    }
116}

118/// Generic in-order DFS over a BTI trie, decoding each node payload with
119/// `decode_payload` and invoking `visit(&reconstructed_key, payload)` in
120/// byte-comparable order — the key is handed to the visitor as a **borrowed
121/// slice** into the single reusable path buffer, so nothing is copied per node.
122///
123/// This is the zero-clone core (issue #1649 / L2): a SINGLE mutable `path`
124/// accumulates the transition bytes from the root to the current node, and the
125/// visitor sees that borrow directly.  Callers that need an owned key call
126/// `.to_vec()` inside their visitor (paying **per emitted result**, never per
127/// child edge — [`dfs_collect_in_order`]); offset-only callers (e.g. the
128/// next-partition seek-bound resolver) drop the key and allocate **nothing**
129/// per node ([`dfs_collect_partition_locations`]).
130///
131/// The traversal is iterative (explicit stack), bounds-checks every offset, and
132/// enforces a **visited-offset guard** plus three independent limits so corrupt or
133/// cyclic tries produce an error rather than an infinite loop, unbounded memory, or
134/// a panic.  The visited-offset guard fires FIRST for cycles/reconvergence: a BTI
135/// trie is a TREE (each node written once, referenced by exactly one parent), so no
136/// offset is legitimately entered twice during a full DFS.  Rejecting the first
137/// re-entry bounds the PENDING OP STACK — a small cyclic/high-fan-out node cannot
138/// push `Enter` ops faster than the total-work counter climbs toward a large padded
139/// trie length (issue #1629 adversarial reconvergence).  **Every remaining bound is
140/// relative to the trie's own byte length (`trie_data.len()`), never a fixed
141/// constant.**  The reason: on an acyclic trie every node on a root→node path is
142/// distinct and occupies at least one trie byte, so any legitimately reconstructed
143/// path length — and any legitimate node depth — is `<= trie_data.len()`.  A path
144/// or depth exceeding the trie's own byte length is structurally impossible
145/// without revisiting a node, i.e. proof of a cycle/corruption.  Bounding by
146/// `trie_data.len()` therefore can NEVER reject a key that legitimately fits in
147/// the file (unlike a fixed cap such as the old 128-byte or `u16::MAX` limits,
148/// which spuriously rejected long-but-legal byte-comparable encoded key paths —
149/// issue #1629), while still catching every cycle:
150///
151/// 1. **Total work** — a running `nodes_visited` counter across ALL paths.  Every
152///    distinct node occupies at least one trie byte, so `nodes_visited >
153///    trie_data.len()` proves reconvergence/cycling and terminates adversarial
154///    tries whose paths stay individually short but whose total node visits blow
155///    up.  This is the airtight cycle guard.
156/// 2. **Per-path key-path length** — `path.len() > trie_data.len()` is a
157///    structurally impossible reconstructed path on an acyclic trie.
158/// 3. **Node depth** — the number of nodes on the current root→node path (stack
159///    depth, root = 1).  A legal leaf's depth is at most `trie_data.len() + 1`
160///    (one distinct node per byte, plus the root count), so a greater depth
161///    proves a cycle.
162///
163/// Each guard carries a distinct error message so corruption modes stay
164/// diagnosable.
165pub(crate) fn dfs_visit_in_order<T, F, V>(
166    trie_data: &[u8],
167    root_offset: usize,
168    mut decode_payload: F,
169    mut visit: V,
170) -> BtiResult<()>
171where
172    // Issue #1650 (L3): the DFS parses each node ONCE and hands the borrowed
173    // `&BtiNode` to `decode_payload`, so a payload-bearing internal node is no
174    // longer parsed twice (once for its payload, once for its children).
175    F: FnMut(&[u8], usize, &BtiNode) -> BtiResult<Option<T>>,
176    V: FnMut(&[u8], T),
177{
178    // Op-based iterative DFS.  A SINGLE mutable `path` accumulates the transition
179    // bytes from the root down to the current node: `Enter` pushes a node's
180    // transition byte (and schedules the matching `Pop`), `Pop` backtracks it.  The
181    // visitor sees `path` as a BORROWED slice — we never copy it per node.  Owned-key
182    // callers copy inside the visitor (per emitted result); offset-only callers copy
183    // nothing.  This avoids the O(depth^2) prefix copying that a per-edge
184    // `key_bytes.clone()` incurs on long key paths (a 70 000-byte legal chain would
185    // otherwise copy ~2.45 GB; issue #1629 roborev).
186    enum DfsOp {
187        Enter {
188            node_offset: usize,
189            // The transition byte from the parent to this node (root = `None`).
190            transition: Option<u8>,
191            // Number of NODES on the root→this-node path (root = 1).
192            depth: usize,
193        },
194        // Backtrack: remove the transition byte pushed by the matching `Enter`.
195        Pop,
196    }
197
198    // The single reusable root→current-node transition-byte path.
199    let mut path: Vec<u8> = Vec::new();
200    let mut stack: Vec<DfsOp> = vec![DfsOp::Enter {
201        node_offset: root_offset,
202        transition: None,
203        depth: 1,
204    }];
205
206    // Total nodes entered across ALL paths — the cycle / reconvergence guard.
207    let mut nodes_visited: usize = 0;
208
209    // Per-offset visited bitset (1 bit per trie byte) — the FIRST-re-entry guard.
210    // A BTI trie is structurally a TREE: every node is written once and referenced
211    // by exactly one parent, so during a full DFS no node offset is legitimately
212    // entered more than once.  Rejecting the first re-entry bounds the PENDING OP
213    // STACK: it stops a small cyclic/reconvergent node with high fan-out (e.g. a
214    // Dense node whose up-to-256 children all point back at itself or a shared
215    // descendant) from pushing `Enter` ops far faster than `nodes_visited` climbs
216    // toward a large padded `trie_data.len()` — which the total-work guard alone
217    // would let exhaust memory before firing (issue #1629 adversarial reconvergence).
218    // A bit-per-offset respects the <128 MB memory target (1/8th of a `Vec<bool>`).
219    let mut visited = vec![0u8; trie_data.len().div_ceil(8)];
220
221    while let Some(op) = stack.pop() {
222        let DfsOp::Enter {
223            node_offset,
224            transition,
225            depth: node_depth,
226        } = op
227        else {
228            // `DfsOp::Pop`: this node's whole subtree is done — backtrack its
229            // transition byte.  Every `Enter` that pushed a byte scheduled exactly
230            // one `Pop`, so this always removes exactly the byte this frame pushed
231            // (the discarded `Option` is always `Some`); ignore it explicitly.
232            let _ = path.pop();
233            continue;
234        };
235
236        // Extend the shared path with this node's transition byte and schedule its
237        // removal AFTER the entire subtree is processed.  The `Pop` is pushed
238        // BEFORE this node's children so, LIFO, it runs once every child subtree
239        // has been fully walked.  The root (transition `None`) contributes no byte
240        // and therefore needs no `Pop`.
241        if let Some(b) = transition {
242            path.push(b);
243            stack.push(DfsOp::Pop);
244        }
245
246        nodes_visited = nodes_visited.saturating_add(1);
247        // Issue #1618 (H5): count every node the DFS enters (L1/L3: <40 nodes visited).
248        crate::storage::sstable::read_work_counters::record_bti_node_visited();
249        if nodes_visited > trie_data.len() {
250            return Err(Error::Parse(format!(
251                "BTI DFS exceeded total work bound ({nodes_visited} nodes visited > \
252                 trie size {}; corrupt or cyclic trie)",
253                trie_data.len()
254            )));
255        }
256        if path.len() > trie_data.len() {
257            return Err(Error::Parse(format!(
258                "BTI DFS key path exceeds trie size {} (corrupt or cyclic trie)",
259                trie_data.len()
260            )));
261        }
262        if node_depth > trie_data.len().saturating_add(1) {
263            return Err(Error::Parse(format!(
264                "BTI DFS exceeded node depth (path of {node_depth} nodes > trie size {} + 1; \
265                 corrupt or cyclic trie)",
266                trie_data.len()
267            )));
268        }
269        if node_offset >= trie_data.len() {
270            return Err(Error::Parse(format!(
271                "BTI DFS: node_offset {node_offset} out of bounds (trie size {})",
272                trie_data.len()
273            )));
274        }
275
276        // Visited-offset guard: reject the FIRST re-entry of any node offset.
277        // Fires BEFORE this node's children are decoded/scheduled, so a cycle or a
278        // DAG-style reconvergence cannot inflate the op stack (or `nodes_visited`)
279        // past a single extra visit.  Runs after the OOB check so `node_offset` is
280        // a valid index into the bitset.
281        let word = node_offset >> 3;
282        let bit = 1u8 << (node_offset & 7);
283        if visited[word] & bit != 0 {
284            return Err(Error::Parse(format!(
285                "BTI DFS revisited node offset {node_offset} (corrupt or cyclic trie)"
286            )));
287        }
288        visited[word] |= bit;
289
290        // Parse this node ONCE (issue #1650 / L3): the parsed node drives BOTH the
291        // payload decode below and the child descent, so a payload-bearing internal
292        // node is no longer parsed twice per visit.
293        let node = parse_bti_node_for_traversal(trie_data, node_offset)?;
294
295        // 1) Emit this node's own payload (if any) BEFORE descending — the key
296        //    terminating here sorts before any continuation.  The visitor receives
297        //    `path` as a borrowed slice; any copy is the visitor's choice (per
298        //    emitted result), never per child edge.
299        if let Some(payload) = decode_payload(trie_data, node_offset, &node)? {
300            visit(&path, payload);
301        }
302
303        // 2) Descend children in ASCENDING transition-byte order.  Because the
304        //    stack is LIFO, push their `Enter` ops in DESCENDING order.  The `Pop`
305        //    scheduled above already sits below these ops, so it fires after the
306        //    entire subtree and backtracks this node's transition byte.
307        let children = ordered_children(&node);
308        let child_depth = node_depth.saturating_add(1);
309        for &(transition_byte, child_offset) in children.iter().rev() {
310            stack.push(DfsOp::Enter {
311                node_offset: child_offset,
312                transition: Some(transition_byte),
313                depth: child_depth,
314            });
315        }
316    }
317
318    Ok(())
319}

321/// Thin owned-key wrapper over [`dfs_visit_in_order`]: collects
322/// `(reconstructed_key, decoded_payload)` in byte-comparable order.  Each result
323/// pays ONE `to_vec()` for its key (inherent to the owned return type), never a
324/// per-child-edge copy.  Offset-only callers should use [`dfs_visit_in_order`]
325/// directly (e.g. [`dfs_collect_partition_locations`]) to allocate nothing.
326pub(crate) fn dfs_collect_in_order<T, F>(
327    trie_data: &[u8],
328    root_offset: usize,
329    decode_payload: F,
330) -> BtiResult<Vec<(Vec<u8>, T)>>
331where
332    F: FnMut(&[u8], usize, &BtiNode) -> BtiResult<Option<T>>,
333{
334    let mut results: Vec<(Vec<u8>, T)> = Vec::new();
335    dfs_visit_in_order(trie_data, root_offset, decode_payload, |key, payload| {
336        results.push((key.to_vec(), payload));
337    })?;
338    Ok(results)
339}

341/// Enumerate every partition entry in a `Partitions.db` trie in byte-comparable
342/// order: `(reconstructed_token_key, BtiPartitionLocation)`.
343pub(crate) fn dfs_collect_partition_entries(
344    trie_data: &[u8],
345    root_offset: usize,
346) -> BtiResult<Vec<(Vec<u8>, BtiPartitionLocation)>> {
347    dfs_collect_in_order(trie_data, root_offset, |data, off, node| {
348        read_node_payload(data, off, Some(node))
349    })
350}

352/// Offset-only counterpart to [`dfs_collect_partition_entries`]: enumerate every
353/// partition [`BtiPartitionLocation`] in byte-comparable order WITHOUT
354/// materializing the reconstructed token keys.
355///
356/// The reconstructed key is handed to the visitor as a borrowed slice and
357/// dropped, so a caller that needs only the locations (e.g. the next-partition
358/// seek-bound resolver in `partition_successor`) pays **zero** per-entry key-`Vec`
359/// allocations — only the single `Vec<BtiPartitionLocation>` grows (issue #1649).
360pub(crate) fn dfs_collect_partition_locations(
361    trie_data: &[u8],
362    root_offset: usize,
363) -> BtiResult<Vec<BtiPartitionLocation>> {
364    let mut locations: Vec<BtiPartitionLocation> = Vec::new();
365    dfs_visit_in_order(
366        trie_data,
367        root_offset,
368        |data, off, node| read_node_payload(data, off, Some(node)),
369        |_key, location| locations.push(location),
370    )?;
371    Ok(locations)
372}

374/// Enumerate **all** partitions in a real Cassandra 5.0 `Partitions.db` BTI file
375/// (issue #832), in byte-comparable order.
376///
377/// This is the headerless public entry point: the trie is loaded via the 8-byte
378/// footer (a real BTI file has no whole-file header).  Each
379/// returned tuple is `(reconstructed_token_key, BtiPartitionLocation)`; the
380/// offset is definitive, the key is a byte-comparable token prefix (see the DFS
381/// module note).
382///
383/// Returns an empty Vec for a `< 8`-byte (e.g. empty) file.
384pub fn iterate_partitions_in_bti_file<R: Read + Seek>(
385    reader: &mut R,
386) -> BtiResult<Vec<(Vec<u8>, BtiPartitionLocation)>> {
387    let file_size = reader.seek(SeekFrom::End(0))?;
388    if file_size < 8 {
389        return Ok(Vec::new());
390    }
391    let (trie_data, root_offset) = load_bti_trie_via_footer(reader)?;
392    dfs_collect_partition_entries(&trie_data, root_offset)
393}

395/// Offset-only counterpart to [`iterate_partitions_in_bti_file`]: enumerate every
396/// partition [`BtiPartitionLocation`] in a real Cassandra 5.0 `Partitions.db` BTI
397/// file in byte-comparable order, WITHOUT reconstructing (or allocating) the
398/// token keys (issue #1649 / L2).
399///
400/// Use this when the reconstructed byte-comparable token key is not needed (the
401/// offsets are definitive) — the DFS then performs **zero** per-partition
402/// key-`Vec` allocations. Returns an empty Vec for a `< 8`-byte (e.g. empty) file.
403pub fn iterate_partition_locations_in_bti_file<R: Read + Seek>(
404    reader: &mut R,
405) -> BtiResult<Vec<BtiPartitionLocation>> {
406    let file_size = reader.seek(SeekFrom::End(0))?;
407    if file_size < 8 {
408        return Ok(Vec::new());
409    }
410    let (trie_data, root_offset) = load_bti_trie_via_footer(reader)?;
411    dfs_collect_partition_locations(&trie_data, root_offset)
412}

437    /// Dense16 (ordinal 11): [0xB0|pf] [start] [len-1] [range * 2-byte deltas]

440        let mut v = vec![0xB0 | (payload_flags & 0x0F), start, len - 1];

442            v.extend_from_slice(&d.to_be_bytes());
443        }

555    /// Build a `Partitions.db`-style trie that is a single-transition **chain**
556    /// of `n_links` `SingleNoPayload4` nodes descending to a `PayloadOnly`
557    /// partition leaf at offset 0.  Returns `(trie_bytes, root_offset)`.
558    ///
559    /// Because every BTI transition encodes exactly one key byte, the leaf's
560    /// reconstructed key path is `n_links` bytes long and the root→leaf node
561    /// depth is `n_links + 1`.  `partition_leaf(hash, -1)` yields
562    /// `DataOffset(0)` at the leaf.
563    fn partition_chain_to_leaf(n_links: usize) -> (Vec<u8>, usize) {
564        let mut trie = Vec::new();
565        trie.extend_from_slice(&partition_leaf(0x11, -1)); // leaf at offset 0 (3 bytes)
566        let mut child_off = 0usize;
567        for i in 0..n_links {
568            let node_off = trie.len();
569            let delta = node_off - child_off;
570            assert!(
571                delta <= 0x0F,
572                "chain link delta {delta} does not fit a SingleNoPayload4 nibble"
573            );
574            // SingleNoPayload4 (ordinal 1): [0x10|delta] [transition]; no payload.
575            trie.push(0x10 | (delta as u8 & 0x0F));
576            trie.push((i % 255) as u8 + 1); // transition byte (nonzero, varies)
577            child_off = node_off;
578        }
579        let root = trie.len() - 2; // last chain node
580        (trie, root)
581    }

705    /// Issue #1629 (roborev): the total-work guard scales with the ENTIRE trie byte
706    /// length, so it does NOT bound the PENDING OP STACK.  A small high-fan-out node
707    /// that self-loops (or reconverges), sitting in a LARGELY PADDED trie, can push
708    /// `Enter` ops far faster than `nodes_visited` climbs toward `trie_data.len()` —
709    /// exhausting memory before the work bound fires.  The visited-offset guard must
710    /// reject the FIRST re-entry, so this returns Err (the revisit/cyclic message)
711    /// after visiting a handful of nodes, never exploding the op stack.
712    #[test]
713    fn dfs_padded_reconvergence_rejected_before_stack_blowup() {
714        // Sparse8 at offset 0 with 200 transitions, EVERY delta = 0 ⇒ every child
715        // resolves to offset 0 (self-loop).  Without the visited guard, each entry
716        // would push 200 more `Enter` ops while `nodes_visited` rose by only 1, so
717        // the op stack would balloon to ~200 × trie_len before the total-work bound
718        // tripped.
719        let fan_out: usize = 200;
720        let mut trie = vec![0x50u8, fan_out as u8]; // Sparse8, no payload
721        trie.extend((1..=fan_out).map(|b| b as u8)); // 200 distinct ascending transitions
722        trie.resize(trie.len() + fan_out, 0); // 200 deltas, all 0 → self-loop
723        let live_len = trie.len();
724        // Pad the trie MUCH larger than the live node so the old total-work bound
725        // (== trie_data.len()) would have allowed ~100k self-visits.
726        trie.resize(live_len + 100_000, 0);
727        assert!(
728            trie.len() > live_len * 100,
729            "padded trie must dwarf the live node so total-work alone would not save us"
730        );
731
732        let err = dfs_collect_partition_entries(&trie, 0)
733            .expect_err("a padded high-fan-out self-loop must error, not blow up the stack");
734        let msg = format!("{err}");
735        assert!(
736            msg.contains("revisited node offset"),
737            "the visited guard must fire first (before total work), got: {msg}"
738        );
739    }
