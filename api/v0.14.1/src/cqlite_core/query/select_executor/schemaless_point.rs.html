1//! Schema-less point-lookup classification for the SELECT executor (issue #1750).
2//!
3//! Split out of `lookup.rs` (campsite rule, epic #1116) so the over-threshold
4//! classifier file does not grow. This holds ONLY the structural schema-less
5//! point-read recogniser + its single-component key encoder; the schema-aware
6//! `classify_partition_lookup` stays in `lookup.rs`.
7//!
8//! # Known schemaless-seek limitations (tracked in #2200)
9//!
10//! These are ACCEPTED limitations of the schema-less single-component seek path.
11//! Each is fail-SAFE (it yields the correct 0 rows or declines to a full scan, never
12//! a wrong non-empty answer) and none is fixed here — they are documented so the
13//! follow-up issue and the next reader are grounded:
14//!
15//! 1. **WRITETIME()/TTL() projection + schemaless pk read → 0 rows.** When a
16//!    `WHERE pk = <lit>` also projects cell metadata (WRITETIME/TTL), the seek path
17//!    is NOT taken (`execute.rs` gates it on `!include_cell_metadata`) and the
18//!    metadata full-scan cannot reconstruct the partition-key column schema-less, so
19//!    the post-scan predicate backstop rejects every row → 0 rows
20//!    (`execute.rs:465`).
21//! 2. **Post-seek guard false-negative for a TIMESTAMP pk (and any seek-firing type
22//!    outside the numeric `values_equal`/`as_f64` set) with a numeric literal.** The
23//!    seek finds the partition, but the post-seek guard's
24//!    `values_equal(Timestamp, BigInt)` is `false`, so the valid row is rejected →
25//!    0 rows. The seek still self-verifies on the raw key bytes, so this is a
26//!    false-negative (0 rows), never a wrong row.
27//! 3. **`coerce_value_for_comparator` has no Date/Varint/Decimal arm for numeric
28//!    literals — by design.** Those types decline via a serialize error rather than
29//!    risk a wrong-width key, so a `WHERE date_pk = <numeric_lit>` full-scans (safe)
30//!    instead of taking the typed seek.

35/// The classified schema-less single-component point-key seek (issue #1750).
36///
37/// Carries the raw on-disk partition-key `bytes` to seek by AND the authoritative
38/// pk component (`name` + `cql_type`) needed to (a) reconstruct the seeked row's
39/// partition-key column schema-less and (b) re-evaluate the predicate against it
40/// (the post-seek guard for roborev 3784 FINDING 1).
41#[derive(Debug, Clone, PartialEq, Eq)]
42pub(super) struct SchemalessPointSeek {
43    /// Raw single-component partition-key bytes to seek by.
44    pub bytes: Vec<u8>,
45    /// The header-synthesised pk column name (`id` / `partition_key`).
46    pub pk_name: String,
47    /// The authoritative CQL type of the pk component.
48    pub pk_cql_type: String,
49}

51/// Classify a SCHEMA-LESS point read against AUTHORITATIVE partition-key metadata
52/// (issue #1750 regression fix, then re-scoped to stop over-firing).
53///
54/// When NO schema is available, `classify_partition_lookup` can only report
55/// `NoSchema` and fall back to a full scan. For a `WHERE pk = <literal>` that scan
56/// CANNOT reconstruct the partition-key column (Cassandra never serialises the pk
57/// column value in the cell payload, and `build_row_from_scan` needs the schema to
58/// decode it from the row key), so the post-scan predicate backstop rejects EVERY
59/// row and the read returns 0 rows. Serving such a read by a key-byte-targeted seek
60/// (which never re-evaluates the predicate) fixes that — but ONLY when the equality
61/// column really is the partition key. A `WHERE <regular_col> = <literal>` seeks a
62/// nonexistent partition and returns 0 rows, whereas a full scan correctly matches
63/// the regular-column cell (regular cells ARE decodable schema-less). So the
64/// targeted seek must fire ONLY for a metadata-CONFIRMED partition-key equality.
65///
66/// The confirmation is by ELIMINATION from authoritative metadata (the Statistics.db
67/// SerializationHeader, [`PartitionKeyShape`]) — NEVER a text/name guess (#28).
68/// Cassandra does not serialise the pk column NAME, but the header authoritatively
69/// gives the pk-component count, the clustering-key count, and the REAL names of the
70/// non-key columns. The seek fires ONLY when:
71///   * the predicates are EXACTLY one non-token single-value `column = <literal>`,
72///   * the value has an unambiguous single-component key encoding,
73///   * the table has exactly ONE partition-key component and ZERO clustering keys
74///     (a single-component point key), AND
75///   * the predicate column is ABSENT from the authoritative non-key column names —
76///     so it can ONLY be the sole partition key.
77///
78/// Any other shape — or a `shape` we could not resolve (`None`) — returns `None`, so
79/// the caller keeps the honest full-scan path (correct for regular-column equalities
80/// and safe when metadata is unavailable). The targeted read is self-verifying: the
81/// storage seek only returns rows whose raw partition key equals these bytes.
82pub(super) fn classify_schemaless_point_lookup(
83    predicates: &[SSTablePredicate],
84    shape: Option<&PartitionKeyShape>,
85) -> Option<SchemalessPointSeek> {
86    use super::super::select_optimizer::SSTableFilterOp;
87
88    // Exactly one predicate, a non-token single-value equality.
89    let [predicate] = predicates else {
90        return None;
91    };
92    if predicate.is_token() || !matches!(predicate.operation, SSTableFilterOp::Equal) {
93        return None;
94    }
95    let [value] = predicate.values.as_slice() else {
96        return None;
97    };
98
99    // Metadata-confirm the predicate column is the SOLE partition key by
100    // elimination — a single-component pk, no clustering keys, and the column is
101    // NOT one of the authoritative non-key column names. Without a resolvable
102    // shape we cannot confirm, so we do NOT seek (full-scan stays correct).
103    let shape = shape?;
104    let is_confirmed_sole_pk = shape.partition_key_count == 1
105        && shape.clustering_key_count == 0
106        && !shape.non_key_column_names.contains(&predicate.column);
107    if !is_confirmed_sole_pk {
108        return None;
109    }
110
111    // The pk component's authoritative CQL TYPE (from the SerializationHeader
112    // `keyType`, never a guess — #28). Without a provable single-component type we
113    // DECLINE the seek (full-scan stays correct) rather than encode a
114    // possibly-wrong-width key (issue #1750, roborev 3784 FINDING 2).
115    let component = shape.single_pk_component.as_ref()?;
116
117    // Encode the literal through the TYPED single-component partition-key codec
118    // for that pk type — the SAME path the schema-aware Targeted seek uses — so a
119    // parsed integer literal (`Value::BigInt`) encodes to the width Cassandra
120    // wrote (`int` → 4 bytes, `bigint` → 8, `uuid` → 16, `text` → utf8). An
121    // unencodable literal/type yields `None` → decline the seek.
122    let bytes = crate::storage::partition_key_codec::encode_single_component_key_typed(
123        value,
124        &component.cql_type,
125    )
126    .ok()?;
127
128    Some(SchemalessPointSeek {
129        bytes,
130        pk_name: component.name.clone(),
131        pk_cql_type: component.cql_type.clone(),
132    })
133}

135/// Build the user-visible `QueryRow` for ONE row returned by a schema-less
136/// single-component point seek, reconstructing the partition-key column and
137/// applying the post-seek predicate guard (issue #1750, roborev 3784 FINDING 1).
138///
139/// The seek self-verifies on raw KEY bytes but never re-checks the predicate
140/// COLUMN, and the classifier admits the predicate column BY ELIMINATION (absent
141/// from the authoritative non-key names) — which also admits a misspelled or
142/// nonexistent column. If such a literal happens to encode to a real partition
143/// key's bytes, the seek would return that partition's row for a column the user
144/// never has. So we:
145///   1. reconstruct the sole pk column from the raw key under its authoritative
146///      TYPE and header-synthesised NAME (schema-less `build_row_from_scan` omits
147///      the pk), and
148///   2. re-evaluate the predicate against the materialised row — a predicate on
149///      the pk name matches; one on a nonexistent/misspelled column is `Unknown`
150///      and rejects the row (yielding the correct 0 rows).
151///
152/// Returns `None` for a tombstoned/absent row, a pk that cannot be decoded, or a
153/// row the predicate rejects. The projection is honoured for output (the pk is
154/// materialised for the guard regardless, then dropped from output if the
155/// projection excludes it) so this matches the schema-aware path's row shape.
156pub(super) fn finalize_schemaless_seek_row(
157    key: crate::types::RowKey,
158    value: crate::types::ScanRow,
159    projection: &[String],
160    predicates: &[SSTablePredicate],
161    seek: &SchemalessPointSeek,
162) -> Option<crate::query::result::QueryRow> {
163    use super::predicate::evaluate_predicates;
164    use super::row_build::build_row_from_scan;
165
166    // Decode the sole pk value from the raw key under its authoritative type
167    // (BEFORE `key` is moved into the row builder). A decode failure means we
168    // cannot validate the row, so drop it (honest 0 rather than an unvalidated 1).
169    let comparator = crate::types::ComparatorType::from_data_type(&seek.pk_cql_type).ok()?;
170    let pk_value =
171        crate::storage::partition_key_codec::deserialize_value_bytes(&key.0, &comparator).ok()?;
172
173    // Schema-less row build (schema = None) surfaces the decoded non-key cells but
174    // NOT the pk column.
175    let mut row = build_row_from_scan(key, value, projection, None)?;
176
177    // Materialise the pk column under its authoritative name for the guard.
178    let pk_name: std::sync::Arc<str> = seek.pk_name.as_str().into();
179    let already_present = row.values.contains_key(&pk_name);
180    if !already_present {
181        row.values.insert(pk_name.clone(), pk_value);
182    }
183
184    // Post-seek guard: re-evaluate the predicate. A predicate on the pk name
185    // matches the reconstructed value; one on a nonexistent/misspelled column is
186    // Unknown → row rejected (the FINDING 1 over-firing this removes).
187    if !evaluate_predicates(&row, predicates).ok()? {
188        return None;
189    }
190
191    // Honour projection for OUTPUT: drop the pk again if we injected it only for
192    // the guard and the projection does not include it (empty projection = all).
193    if !already_present
194        && !projection.is_empty()
195        && !projection.iter().any(|p| p.as_str() == seek.pk_name)
196    {
197        row.values.remove(&pk_name);
198    }
199
200    Some(row)
201}

244    /// Issue #1750 (regression fix): a schema-less `col = <literal>` whose column is
245    /// a metadata-CONFIRMED sole partition key yields the TYPED single-component key
246    /// bytes for a targeted seek. Confirmation is by elimination from authoritative
247    /// metadata — never a pk-name/text guess.
248    #[test]
249    fn classify_schemaless_point_lookup_targets_confirmed_pk_equality() {
250        let shape = simple_table_shape();

255            .expect("a single UUID `=` on the confirmed pk must classify");
256        assert_eq!(seek.bytes, uuid.to_vec(), "raw 16-byte uuid key");
257        assert_eq!(seek.pk_name, "id");
258        assert_eq!(seek.pk_cql_type, "uuid");
259
260        // roborev 3784 FINDING 2: a parsed integer literal arrives as `Value::BigInt`
261        // (the SELECT parser widens every integer); on an `int` pk it must encode to
262        // Cassandra's 4-byte `int` key, NOT an 8-byte one.
263        let int_shape = int_pk_shape();
264        let int_pred = SSTablePredicate::column(
265            "partition_key",
266            SSTableFilterOp::Equal,
267            vec![Value::BigInt(42)],
268        );
269        let int_seek =
270            classify_schemaless_point_lookup(std::slice::from_ref(&int_pred), Some(&int_shape))
271                .expect("int pk `=` must classify");
272        assert_eq!(
273            int_seek.bytes,
274            42i32.to_be_bytes().to_vec(),
275            "a BigInt literal on an int pk must encode to the 4-byte int key (roborev 3784)",
276        );

435    /// roborev 3784 FINDING 1: the post-seek guard re-evaluates the predicate. A
436    /// `WHERE <pk_name> = <value>` matches the reconstructed pk column and returns
437    /// the row; a `WHERE <nonexistent_col> = <value>` (admitted by elimination) is
438    /// Unknown against the materialised row and yields 0 rows.
439    #[test]
440    fn finalize_seek_row_guards_on_reconstructed_pk() {
441        // int pk = 42, one regular col `v`.
442        let seek = SchemalessPointSeek {
443            bytes: 42i32.to_be_bytes().to_vec(),
444            pk_name: "partition_key".to_string(),
445            pk_cql_type: "int".to_string(),
446        };
447        let key = RowKey::new(42i32.to_be_bytes().to_vec());
448        let value = scan_row(&[("v", Value::Text("hi".to_string()))]);
449
450        // Predicate on the pk name → matches → row returned, pk reconstructed.
451        let pk_pred = vec![Pred::column(
452            "partition_key",
453            SSTableFilterOp::Equal,
454            vec![Value::BigInt(42)],
455        )];
456        let row: QueryRow =
457            finalize_schemaless_seek_row(key.clone(), value.clone(), &[], &pk_pred, &seek)
458                .expect("pk-name predicate must match the reconstructed row");
459        assert_eq!(row.values.get("partition_key"), Some(&Value::Integer(42)));
460        assert_eq!(row.values.get("v"), Some(&Value::Text("hi".to_string())));
461
462        // Predicate on a nonexistent column → Unknown against the materialised row
463        // → row REJECTED (the FINDING 1 over-firing this removes).
464        let bogus_pred = vec![Pred::column(
465            "not_a_column",
466            SSTableFilterOp::Equal,
467            vec![Value::BigInt(42)],
468        )];
469        assert!(
470            finalize_schemaless_seek_row(key, value, &[], &bogus_pred, &seek).is_none(),
471            "a nonexistent predicate column must yield 0 rows post-seek (roborev 3784 FINDING 1)",
472        );
473    }
