1//! Streaming SELECT producer (`execute_streaming_background`) — issue #1578 split.
2//!
3//! Relocated verbatim from `execute.rs` (epic #1116 file-size split) so the
4//! materializing runner and the streaming producer live in separate files. As a
5//! child module of `select_executor` this reaches `mod.rs`'s private items
6//! directly; logic, ordering, and error handling are unchanged.
7
8use super::{
9    build_row_from_scan_cached, classify_partition_lookup, evaluate_predicates,
10    honest_targeted_path, partition_key_digest, sort_rows_by_token, validate_token_predicates,
11    PartitionKeyCache, PartitionLookupOutcome,
12};
13use super::{
14    AccessPath, ExecutionStep, QueryRow, Result, SelectExecutor, StorageEngine, TableId,
15    TableSchema,
16};
17use std::sync::Arc;
18use tokio::sync::mpsc;
19
20impl SelectExecutor {
21    /// Background task: Execute streaming scan and send rows through channel
22    pub(super) async fn execute_streaming_background(
23        storage: Arc<StorageEngine>,
24        // Issue #1587 (E5): schema resolved ONCE per query by `execute_streaming`
25        // and moved into this task — no per-scan-step registry lock + deep clone.
26        query_schema: Option<Arc<TableSchema>>,
27        _table_id: TableId,
28        execution_steps: Vec<ExecutionStep>,
29        tx: mpsc::Sender<Result<QueryRow>>,
30        buffer_size: usize,
31    ) -> Result<()> {
32        // Issue #581: LIMIT/OFFSET must be enforced by the producer in the
33        // streaming path. The `ExecutionStep::Limit` arm previously only logged a
34        // message and relied on a consumer that never applied it, so
35        // `execute_streaming` yielded the full result set regardless of LIMIT.
36        // Extract the bound up front (steps are ordered with Limit after the scan)
37        // and stop sending once it is satisfied — mirroring `execute_limit`
38        // (drain OFFSET, then truncate to `count`) row-by-row so the producer
39        // stops scanning early.
40        let limit = execution_steps.iter().find_map(|step| match step {
41            ExecutionStep::Limit { count, offset } => Some((*count, offset.unwrap_or(0))),
42            _ => None,
43        });
44        let (limit_count, mut offset_remaining) = match limit {
45            Some((count, offset)) => (Some(count), offset),
46            None => (None, 0),
47        };
48
49        // A `LIMIT 0` means no rows can ever be sent; return before scanning.
50        if limit_count == Some(0) {
51            return Ok(());
52        }
53
54        // Issue #757: PER PARTITION LIMIT caps rows per partition before the
55        // query-wide LIMIT/OFFSET. The scan yields rows grouped by partition
56        // key, so we track the current partition and reset the counter at each
57        // boundary. Issue #1590 (E8): the boundary is compared on the partition
58        // key's 128-bit `partition_key_digest` (a heap-free hash of the key bytes
59        // we already hold) as a FAST pre-check, then confirmed by EXACT byte
60        // equality against the current partition's raw bytes — stored ONCE when
61        // the boundary advances, never cloned per row. This keeps correctness
62        // independent of digest collisions (a collision between two DISTINCT
63        // partitions never shares a counter).
64        let per_partition_limit = execution_steps.iter().find_map(|step| match step {
65            ExecutionStep::PerPartitionLimit { count } => Some(*count),
66            _ => None,
67        });
68        let mut current_partition: Option<(u128, Vec<u8>)> = None;
69        let mut partition_count: u64 = 0;
70
71        let mut sent: u64 = 0;
72
73        for step in &execution_steps {
74            match step {
75                ExecutionStep::SSTableScan {
76                    table,
77                    predicates,
78                    projection,
79                    ..
80                } => {
81                    let schema_opt = query_schema.as_deref();
82
83                    // FINDING 2 (Issue #955 follow-up): reject a `token(...)` whose
84                    // columns are not the full partition key in declared order
85                    // before scanning (same rule as the materializing path).
86                    validate_token_predicates(predicates, schema_opt)?;
87
88                    // Issue #949: a fully-constrained `WHERE pk = ?` is served by a
89                    // partition-targeted lookup that prunes SSTables via bloom/BTI,
90                    // instead of streaming a scan over every SSTable. The resulting
91                    // rows are sent through the same per-row pipeline below
92                    // (predicates, PER PARTITION LIMIT, OFFSET, LIMIT). Note
93                    // `scan_partition` reconciles across SSTable generations like the
94                    // materializing `scan()` (last-write-wins + tombstone shadowing),
95                    // which is the authoritative read semantics; it does not merely
96                    // mirror `scan_stream`'s per-key merge.
97                    let lookup = classify_partition_lookup(predicates, schema_opt);
98                    if let PartitionLookupOutcome::Targeted(ref pk_bytes) = lookup {
99                        // Issue #960: the streaming analogue of the materializing
100                        // partition-targeted lookup. Epic #951 (honest paths): the
101                        // `tombstones` build's `scan_partition` is a full-scan +
102                        // retain with NO prune, reported via `engaged == false`; only
103                        // claim `StreamingPartitionLookup` when it really pruned.
104                        let (rows, engaged) =
105                            storage.scan_partition(table, pk_bytes, schema_opt).await?;
106                        crate::query::access_path::record(honest_targeted_path(
107                            AccessPath::StreamingPartitionLookup,
108                            engaged,
109                        ));
110                        // Issue #1817: hoist the partition-key decode across a
111                        // partition's rows (a lookup returns one partition's rows).
112                        let mut pk_cache = PartitionKeyCache::default();
113                        for (key, value) in rows {
114                            let part_sig =
115                                per_partition_limit.map(|_| partition_key_digest(&key.0));
116                            let Some(row) = build_row_from_scan_cached(
117                                key,
118                                value,
119                                projection,
120                                schema_opt,
121                                &mut pk_cache,
122                            ) else {
123                                continue;
124                            };
125                            if !evaluate_predicates(&row, predicates)? {
126                                continue;
127                            }
128                            if let (Some(cap), Some(sig)) = (per_partition_limit, part_sig) {
129                                // Fast digest pre-check, then EXACT byte confirm so a
130                                // digest collision between DISTINCT partitions never
131                                // shares a counter (issue #1590).
132                                let same = matches!(
133                                    &current_partition,
134                                    Some((d, bytes))
135                                        if *d == sig && bytes.as_slice() == row.key.as_bytes()
136                                );
137                                if !same {
138                                    // Clone the key bytes ONCE per boundary, not per row.
139                                    current_partition = Some((sig, row.key.as_bytes().to_vec()));
140                                    partition_count = 0;
141                                }
142                                if partition_count >= cap {
143                                    continue;
144                                }
145                                partition_count += 1;
146                            }
147                            if offset_remaining > 0 {
148                                offset_remaining -= 1;
149                                continue;
150                            }
151                            if tx.send(Ok(row)).await.is_err() {
152                                return Ok(());
153                            }
154                            sent += 1;
155                            if let Some(count) = limit_count {
156                                if sent >= count {
157                                    return Ok(());
158                                }
159                            }
160                        }
161                        // This SSTableScan step is fully served by the lookup.
162                        continue;
163                    }
164
165                    // Issue #955: `WHERE pk IN (...)` over the complete key is the
166                    // union of N partition-targeted lookups. Gather them, sort by
167                    // token to match full-scan order, then drive the same per-row
168                    // pipeline (predicates, PER PARTITION LIMIT, OFFSET, LIMIT).
169                    if let PartitionLookupOutcome::MultiTargeted(ref pk_keys) = lookup {
170                        // Epic #951 (honest paths): each lookup reports whether it
171                        // pruned. On the `tombstones` build every call full-scans
172                        // (`engaged == false`); claim `MultiPartitionLookup` only when
173                        // the lookups actually pruned, else report the honest fallback.
174                        let mut combined = Vec::new();
175                        let mut all_engaged = true;
176                        for pk_bytes in pk_keys {
177                            let (rows, engaged) =
178                                storage.scan_partition(table, pk_bytes, schema_opt).await?;
179                            all_engaged &= engaged;
180                            combined.extend(rows);
181                        }
182                        crate::query::access_path::record(honest_targeted_path(
183                            AccessPath::MultiPartitionLookup,
184                            all_engaged,
185                        ));
186                        sort_rows_by_token(&mut combined);
187                        // Issue #1817: hoist the partition-key decode across a
188                        // partition's rows (combined is token-sorted, so a
189                        // partition's rows are contiguous).
190                        let mut pk_cache = PartitionKeyCache::default();
191                        for (key, value) in combined {
192                            let part_sig =
193                                per_partition_limit.map(|_| partition_key_digest(&key.0));
194                            let Some(row) = build_row_from_scan_cached(
195                                key,
196                                value,
197                                projection,
198                                schema_opt,
199                                &mut pk_cache,
200                            ) else {
201                                continue;
202                            };
203                            if !evaluate_predicates(&row, predicates)? {
204                                continue;
205                            }
206                            if let (Some(cap), Some(sig)) = (per_partition_limit, part_sig) {
207                                // Fast digest pre-check, then EXACT byte confirm so a
208                                // digest collision between DISTINCT partitions never
209                                // shares a counter (issue #1590).
210                                let same = matches!(
211                                    &current_partition,
212                                    Some((d, bytes))
213                                        if *d == sig && bytes.as_slice() == row.key.as_bytes()
214                                );
215                                if !same {
216                                    // Clone the key bytes ONCE per boundary, not per row.
217                                    current_partition = Some((sig, row.key.as_bytes().to_vec()));
218                                    partition_count = 0;
219                                }
220                                if partition_count >= cap {
221                                    continue;
222                                }
223                                partition_count += 1;
224                            }
225                            if offset_remaining > 0 {
226                                offset_remaining -= 1;
227                                continue;
228                            }
229                            if tx.send(Ok(row)).await.is_err() {
230                                return Ok(());
231                            }
232                            sent += 1;
233                            if let Some(count) = limit_count {
234                                if sent >= count {
235                                    return Ok(());
236                                }
237                            }
238                        }
239                        // This SSTableScan step is fully served by the lookups.
240                        continue;
241                    }
242
243                    // Issue #960: the streaming path did not take a targeted
244                    // lookup; report the honest fallback reason. `lookup` is the
245                    // `Fallback` arm here (the `Targeted`/`MultiTargeted` arms
246                    // returned above via `continue`).
247                    if let PartitionLookupOutcome::Fallback(reason) = lookup {
248                        crate::query::access_path::record(AccessPath::FallbackFullScan { reason });
249                    }
250
251                    // Issue #790: pull rows lazily from a bounded streaming scan
252                    // instead of materializing the full result `Vec`. The reader
253                    // parses one entry at a time into this channel, so live heap
254                    // stays bounded by `buffer_size` rather than O(result rows).
255                    // Issue #1592: consume the BATCHED streaming surface — one
256                    // async wake per batch, not per row. Flattening each batch
257                    // yields the same rows in the same order as `scan_stream`.
258                    let mut scan_stream = storage
259                        .scan_stream_batched(table, None, None, schema_opt, buffer_size)
260                        .await?;
261
262                    // Issue #1817: hoist the partition-key decode across a
263                    // partition's rows. The cache persists ACROSS batches (a
264                    // partition may straddle a batch boundary), so a partition's
265                    // key is decoded once regardless of batching.
266                    let mut pk_cache = PartitionKeyCache::default();
267                    while let Some(batch) = scan_stream.recv().await {
268                        for (key, value) in batch? {
269                            // Capture the partition-key digest before `key` is moved
270                            // into row construction (only when needed).
271                            let part_sig =
272                                per_partition_limit.map(|_| partition_key_digest(&key.0));
273                            let Some(row) = build_row_from_scan_cached(
274                                key,
275                                value,
276                                projection,
277                                schema_opt,
278                                &mut pk_cache,
279                            ) else {
280                                continue;
281                            };
282
283                            if !evaluate_predicates(&row, predicates)? {
284                                continue;
285                            }
286
287                            // Apply PER PARTITION LIMIT: cap matching rows per
288                            // partition, before OFFSET/LIMIT (Cassandra semantics).
289                            if let (Some(cap), Some(sig)) = (per_partition_limit, part_sig) {
290                                // Fast digest pre-check, then EXACT byte confirm so a
291                                // digest collision between DISTINCT partitions never
292                                // shares a counter (issue #1590).
293                                let same = matches!(
294                                    &current_partition,
295                                    Some((d, bytes))
296                                        if *d == sig && bytes.as_slice() == row.key.as_bytes()
297                                );
298                                if !same {
299                                    // Clone the key bytes ONCE per boundary, not per row.
300                                    current_partition = Some((sig, row.key.as_bytes().to_vec()));
301                                    partition_count = 0;
302                                }
303                                if partition_count >= cap {
304                                    continue;
305                                }
306                                partition_count += 1;
307                            }
308
309                            // Apply OFFSET: skip the first `offset_remaining` matches.
310                            if offset_remaining > 0 {
311                                offset_remaining -= 1;
312                                continue;
313                            }
314
315                            // Send row through channel (with backpressure). Consumer drop ends the scan.
316                            if tx.send(Ok(row)).await.is_err() {
317                                return Ok(());
318                            }
319                            sent += 1;
320
321                            // Apply LIMIT: stop scanning once `count` rows have been
322                            // sent. Dropping `scan_stream` here signals the producer
323                            // (via a closed channel) to stop parsing early.
324                            if let Some(count) = limit_count {
325                                if sent >= count {
326                                    return Ok(());
327                                }
328                            }
329                        }
330                    }
331                }
332                ExecutionStep::Limit { .. } | ExecutionStep::PerPartitionLimit { .. } => {
333                    // Enforced inline during the scan above (see the bounds
334                    // extracted before the loop).
335                }
336                // Projection and predicate filtering are pushed into SSTableScan above.
337                ExecutionStep::Project { .. } | ExecutionStep::Filter { .. } => {}
338                _ => {
339                    // Data-safety (issue #1694): log the step's variant name only,
340                    // never its contents (which carry query literals/values).
341                    tracing::warn!(
342                        "Streaming execution: skipping unsupported step {}",
343                        step.variant_name()
344                    );
345                }
346            }
347        }
348
349        Ok(())
350    }
