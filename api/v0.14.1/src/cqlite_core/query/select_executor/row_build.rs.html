2//!
3//! [`build_row_from_scan`] turns a raw `(RowKey, ScanRow)` scan entry into a
4//! `QueryRow`, reconstructing partition-key columns from the raw key. It is part
5//! of the public surface (re-exported via `query::mod`) so other readers (e.g.
6//! the Arrow Flight compaction-merge producer) assemble rows identically.

18/// Split a `TableId` of the form `"keyspace.table"` into its parts.
19///
20/// If no dot is present, the whole name becomes the table component and the
21/// keyspace is `None`.
22pub(super) fn parse_table_id(table_id: &TableId) -> (Option<String>, String) {
23    let table_str = table_id.name();
24    match table_str.rfind('.') {
25        Some(dot) => (
26            Some(table_str[..dot].to_string()),
27            table_str[dot + 1..].to_string(),
28        ),
29        None => (None, table_str.to_string()),
30    }
31}

33/// Parse a CQL type string (e.g. `"list<int>"`, `"text"`) into a [`CqlType`].
34///
35/// Returns `None` when the type string cannot be parsed (unknown or malformed
36/// types). Used to populate `ColumnInfo::cql_type` from the schema's string
37/// representation, satisfying the no-heuristics mandate (Issue #28).
38pub(super) fn parse_cql_type_str(type_str: &str) -> Option<CqlType> {
39    let parser = ComplexTypeParser::new();
40    parser
41        .parse_type(type_str)
42        .ok()
43        .map(|parsed| parsed.cql_type)
44}

46/// Build a `ColumnInfo` from a schema column name + CQL type string, deriving
47/// the flat `DataType` from the parsed `CqlType` (Issue #674). Centralises the
48/// "name + type string → ColumnInfo" pattern shared by the SELECT-* and
49/// explicit-projection column builders.
50pub(super) fn column_info_from_type_str(
51    name: String,
52    type_str: &str,
53    position: usize,
54    table_name: Option<String>,
55) -> ColumnInfo {
56    let cql_type_opt = parse_cql_type_str(type_str);
57    let data_type = cql_type_opt
58        .as_ref()
59        .map(cql_type_to_data_type)
60        .unwrap_or(crate::types::DataType::Text);
61    let mut col_info = ColumnInfo {
62        name,
63        data_type,
64        nullable: true,
65        position,
66        table_name,
67        cql_type: None,
68    };
69    if let Some(cql_type) = cql_type_opt {
70        col_info = col_info.with_cql_type(cql_type);
71    }
72    col_info
73}

75/// A decoded partition's identity + its interned `(name, value)` partition-key
76/// columns (Issue #1817). The identity is the raw key bytes PAIRED WITH a cheap
77/// schema fingerprint (`pk_schema_fingerprint`), so a hit requires BOTH to match.
78/// Names are interned once per partition.
79type DecodedPartitionKey = (Arc<[u8]>, u64, Vec<(Arc<str>, Value)>);
80
81/// Cheap fingerprint of a schema's partition-key column list — the `(name, type,
82/// position)` of every partition-key column, in order (Issue #1817, roborev).
83///
84/// The decoded partition-key columns depend ONLY on the raw key bytes and this
85/// list, so two schemas with identical partition-key columns decode identically
86/// and may safely share a cache entry, while any difference (column names, types,
87/// count, order) yields a different fingerprint → a cache MISS and re-decode.
88/// This closes the cross-schema footgun where a cache reused across two tables
89/// whose partition keys share raw bytes but differ in schema would return columns
90/// decoded for the WRONG schema.
91fn pk_schema_fingerprint(schema: &crate::schema::TableSchema) -> u64 {
92    use std::hash::{Hash, Hasher};
93    let mut hasher = std::collections::hash_map::DefaultHasher::new();
94    schema.partition_keys.len().hash(&mut hasher);
95    for col in &schema.partition_keys {
96        col.name.hash(&mut hasher);
97        col.data_type.hash(&mut hasher);
98        col.position.hash(&mut hasher);
99    }
100    hasher.finish()
101}

103/// Cross-row memoization of the decoded partition-key columns (Issue #1817).
104///
105/// Partition-key column VALUES are constant within a partition, but
106/// [`build_row_from_scan`] historically re-decoded them from the raw key bytes
107/// for EVERY row. Because a scan yields the rows of one partition consecutively
108/// (token-ordered full scans, partition-targeted lookups, and the Flight k-way
109/// merge all group a partition's rows), memoizing the last partition's decoded
110/// columns makes the byte-parse + name-intern happen ONCE per partition
111/// (`O(partitions)`) rather than once per row (`O(rows)`). The decoded values are
112/// `clone`d into each row (an `Arc` ref-count bump for the name + a `Value`
113/// clone — each row needs its own copy), so the materialized rows are
114/// byte-identical to the prior per-row decode.
115///
116/// If the input is NOT partition-ordered the cache simply misses and re-decodes,
117/// so correctness never depends on ordering — only the decode COUNT does.
118///
119/// A cache hit also requires the requesting schema's partition-key fingerprint to
120/// match the cached one (roborev): `PartitionKeyCache` is publicly re-exported and
121/// could be reused across tables whose partition keys share raw bytes but differ
122/// in schema — matching the fingerprint makes any such difference a MISS so the
123/// columns are never decoded for the wrong schema.
124#[derive(Default)]
125pub struct PartitionKeyCache {
126    /// Raw partition-key bytes + partition-key schema fingerprint of the last
127    /// decoded partition and its decoded `(interned name, value)` columns
128    /// (unfiltered by projection — the projection filter is applied when cloning
129    /// into each row, so a cache entry is reusable across queries with different
130    /// projections).
131    decoded: Option<DecodedPartitionKey>,
132}
133
134impl PartitionKeyCache {
135    /// Return the decoded partition-key columns for `key_bytes`, decoding through
136    /// the canonical codec (and interning names once) only on a cache MISS — i.e.
137    /// once per distinct partition when rows arrive partition-grouped.
138    fn columns_for<'a>(
139        &'a mut self,
140        key_bytes: &Arc<[u8]>,
141        schema: &crate::schema::TableSchema,
142    ) -> &'a [(Arc<str>, Value)] {
143        let fingerprint = pk_schema_fingerprint(schema);
144        let hit = self.decoded.as_ref().is_some_and(|(cached, fp, _)| {
145            *fp == fingerprint && cached.as_ref() == key_bytes.as_ref()
146        });
147        if !hit {
148            // Test-only decode/name-derivation counter (Issue #1817): a cache MISS
149            // is exactly one decode. A partition-grouped scan of N rows over P
150            // partitions increments this P times, not N — the O(partitions) pin.
151            #[cfg(test)]
152            super::PARTITION_KEY_DECODES.with(|c| c.set(c.get() + 1));
153
154            let cols = match crate::storage::partition_key_codec::decode_partition_key_columns(
155                key_bytes, schema,
156            ) {
157                // Intern each name into a shared `Arc<str>` ONCE (per partition),
158                // so cloning into a row is a ref-count bump, not a `String` alloc.
159                Ok(pk_columns) => pk_columns
160                    .into_iter()
161                    .map(|(name, value)| (Arc::<str>::from(name), value))
162                    .collect(),
163                // Surface — never silently swallow — a decode failure, so a
164                // missing partition-key column can't ship invisibly (Issue #586).
165                // Cache the empty result so a failing partition is not re-decoded
166                // (and not re-warned) once per row.
167                Err(e) => {
168                    tracing::warn!(
169                        "Failed to reconstruct partition-key columns from row key \
170                         (len={} bytes) for {}.{}: {}",
171                        key_bytes.len(),
172                        schema.keyspace,
173                        schema.table,
174                        e
175                    );
176                    Vec::new()
177                }
178            };
179            self.decoded = Some((Arc::clone(key_bytes), fingerprint, cols));
180        }
181        match &self.decoded {
182            Some((_, _, cols)) => cols,
183            None => &[],
184        }
185    }
186}

188/// Build a `QueryRow` from a single `(RowKey, ScanRow)` produced by storage scan,
189/// applying optional projection and synthesising partition-key columns from the
190/// raw key bytes when a schema is available.
191///
192/// Partition-key columns are never stored in the cell payload, so they are
193/// reconstructed from the raw row key via the canonical
194/// [`crate::storage::partition_key_codec::decode_partition_key_columns`] (the
195/// same codec the write engine uses). This is the fix for Issue #586: the
196/// previous decoder assumed a `u16` length prefix for every TEXT key, which is
197/// only correct for composite components — a single-component TEXT partition key
198/// is raw bytes, so its column was silently dropped from scan-built rows.
199///
200/// Returns `None` for tombstoned rows (so the caller can `continue`).
201///
202/// Exposed publicly so other readers (e.g. the Arrow Flight server's compaction
203/// merge producer) can assemble rows identically to the SELECT path, guaranteeing
204/// output parity. The `row` carries the decoded non-partition-key cells as the
205/// single [`ScanRow`] row carrier (issue #1334); partition-key columns are
206/// reconstructed from `key`.
207///
208/// This is a thin wrapper over [`build_row_from_scan_cached`] with a single-use
209/// [`PartitionKeyCache`], so a one-off build decodes the key exactly once (as
210/// before). Loops over many rows should call [`build_row_from_scan_cached`] with
211/// a shared cache to hoist the per-partition decode (Issue #1817).
212pub fn build_row_from_scan(
213    key: RowKey,
214    row: ScanRow,
215    projection: &[String],
216    schema: Option<&crate::schema::TableSchema>,
217) -> Option<QueryRow> {
218    let mut pk_cache = PartitionKeyCache::default();
219    build_row_from_scan_cached(key, row, projection, schema, &mut pk_cache)
220}

222/// [`build_row_from_scan`] with a caller-owned [`PartitionKeyCache`] that hoists
223/// the partition-key decode across the rows of a partition (Issue #1817).
224///
225/// The output is byte-identical to [`build_row_from_scan`] — the cache only
226/// avoids re-decoding the partition key for consecutive rows that share it.
227pub fn build_row_from_scan_cached(
228    key: RowKey,
229    row: ScanRow,
230    projection: &[String],
231    schema: Option<&crate::schema::TableSchema>,
232    pk_cache: &mut PartitionKeyCache,
233) -> Option<QueryRow> {
234    // Suppress tombstoned / absent rows from user-visible output. A row tombstone
235    // or null row reaches here as `ScanRow::Marker` (Issue #505); it must never
236    // appear in query results. A live row is always `ScanRow::Row`, so there is
237    // exactly ONE row-carrier path — a live row's column values can never silently
238    // fall through to a non-row fallback (issue #1334 / roborev H2).
239    let cells = row.into_cells()?;
240
241    // Issue #1584: pre-size the value map to the upper bound of inserts — the
242    // decoded cell count plus the reconstructed partition-key columns. This is a
243    // single sized allocation with no rehash growth (for `SELECT *` it is exact;
244    // for a narrower projection it slightly over-reserves, still one alloc).
245    let pk_hint = schema.map(|s| s.partition_keys.len()).unwrap_or(0);
246    let mut row_values: HashMap<Arc<str>, Value> = HashMap::with_capacity(cells.len() + pk_hint);
247    let project = |name: &str| projection.is_empty() || projection.iter().any(|p| p == name);
248
249    // Issue #1334: the decoder carries interned `Arc<str>` column-name handles in
250    // the row carrier; move them straight into `QueryRow.values` (an `Arc` move —
251    // NO `String` re-allocation of the name).
252    for (name, col_value) in cells {
253        if project(&name) {
254            row_values.insert(name, col_value);
255        }
256    }
257    // Cassandra never serialises partition-key columns in the cell payload;
258    // reconstruct them from the raw row key when the schema is known. We decode
259    // through the canonical codec shared with the write engine so single-component
260    // (raw bytes) and composite (`[u16 len][bytes][0x00]`) keys are handled
261    // identically on both paths (Issue #586). Issue #1817: the decode is memoized
262    // per partition by `pk_cache`, so consecutive rows of a partition clone the
263    // already-decoded columns instead of re-parsing the key bytes.
264    if let Some(schema) = schema {
265        for (name, value) in pk_cache.columns_for(&key.0, schema) {
266            if project(name) {
267                row_values.insert(Arc::clone(name), value.clone());
268            }
269        }
270    }
271
272    Some(QueryRow {
273        values: row_values,
274        key,
275        metadata: Default::default(),
276        cell_metadata: None,
277    })
278}

526    /// Issue #1817 (roborev): `PartitionKeyCache` is publicly re-exported and could
527    /// be reused across tables whose partition keys happen to share the SAME raw key
528    /// bytes but have DIFFERENT schemas. A hit keyed only on the raw bytes would

536        // (`id_a` vs `id_b`). A bytes-only cache would return `id_a` for schema B.
537        let schema_a = single_pk_schema("id_a", "text");
538        let schema_b = single_pk_schema("id_b", "text");
539        let key_bytes: Arc<[u8]> = Arc::from(b"shared-key".as_slice());
540

542
543        // First call: schema A. Columns must be decoded for A.
544        let cols_a: Vec<(Arc<str>, Value)> = cache.columns_for(&key_bytes, &schema_a).to_vec();
545        assert_eq!(cols_a.len(), 1);
546        assert_eq!(cols_a[0].0.as_ref(), "id_a");

573    /// SUPPRESSES a `ScanRow::Marker` but SURFACES a live `ScanRow::Row`. This is
574    /// exactly why the CLI `read`/`inspect`/`benchmark` bulletproof-reader fallback
575    /// producers must NOT wrap a LIVE synthetic value in `Marker` — doing so drops
576    /// it from user-visible output. The SAME live value must survive as a `Row`.
577    #[test]
578    fn live_value_dropped_as_marker_surfaces_as_row() {
579        let key = RowKey::new(b"k".to_vec());
580        let live = Value::Text("synthetic-fallback".to_string());
581
582        // Wrapped as a Marker (the pre-fix producer): dropped entirely.
583        assert!(
584            build_row_from_scan(key.clone(), ScanRow::Marker(live.clone()), &[], None).is_none(),
585            "a LIVE value mis-wrapped as Marker is dropped — the bug the producers must avoid"
586        );
587
588        // Wrapped as a live Row (the fixed producer): surfaces under its cell name.
589        let row = build_row_from_scan(
590            key,
591            ScanRow::Row(vec![(Arc::from("data"), live.clone())]),
592            &[],
593            None,
594        )
595        .expect("a live Row must surface");
