1//! Issue #1577 (Epic D / D1): LIMIT/OFFSET pushdown into the materializing scan.
2//!
3//! The materializing executor ([`SelectExecutor::execute`]) historically
4//! decoded EVERY row of a table into a per-row `HashMap`, then threw all but the
5//! first `LIMIT` away in the post-hoc `Limit` step. This module supplies the two
6//! pieces that let it stop early WITHOUT changing a single result:
7//!
8//! - [`scan_pushdown_cap`] decides — from the plan alone — whether the scan may
9//!   be bounded, and to how many ACCEPTED rows.
10//! - [`SelectExecutor::capped_fallback_scan`] performs the bounded full-scan
11//!   fallback by consuming the lazy `scan_stream` (definitionally in lockstep
12//!   with `scan`) and dropping it once the cap is reached, which closes the
13//!   channel and stops the producer decoding the tail. A stream that ends short
14//!   of the cap is reconciled against the authoritative materializing `scan`, so
15//!   a `scan_stream`/`scan` divergence can never drop a row.
16//!
17//! Correctness is the law here: the cap is applied ONLY when the pipeline
18//! between the scan and the query-wide `LIMIT` neither reorders nor drops rows,
19//! and the cap counts rows the executor actually ACCEPTS (post marker
20//! suppression, post predicate) — never raw scan rows — so a suppressed row
21//! tombstone / null-row marker or a predicate miss can never under-deliver.
22

29/// In-flight row bound for the capped streaming fallback scan (issue #1577).
30///
31/// The per-call buffer is `cap + 1` clamped to this ceiling: a small `LIMIT`
32/// decodes only a few rows past the limit before the producer parks (so the
33/// decode work — and the `PARTITION_HEADER_TRY_PARSES` / row-decode counters —
34/// stays `O(limit)`, not `O(table)`), while a large `LIMIT` keeps the same
35/// bounded footprint the streaming path already uses (issue #790).
36const CAPPED_SCAN_STREAM_BUFFER: usize = 1024;
37
38/// Compute the LIMIT/OFFSET pushdown cap for the materializing scan (issue #1577).
39///
40/// Returns `Some(limit + offset)` — the number of ACCEPTED (post-predicate,
41/// post-marker) rows the scan must produce before the downstream `Limit` step
42/// can slice the final `offset..offset+count` window — but ONLY when the plan is
43/// pushdown-safe:
44///
45/// * there IS a `Limit` step, and
46/// * there is NO step between the scan and that `Limit` that reorders, collapses,
47///   or otherwise reduces the row multiset — `Sort`, `Aggregate`,
48///   `PerPartitionLimit` — and NO residual `Filter` step (a `Filter` drops rows
49///   the scan already yielded, so a raw scan cap could under-deliver), and
50/// * the query is not `DISTINCT` (which may dedup rows after the scan).
51///
52/// Returns `None` otherwise, leaving the scan unbounded exactly as before.
53///
54/// # Why the cap preserves results
55///
56/// When none of those steps are present, the only step that may sit between the
57/// scan and the `Limit` is `Project`, which neither reorders nor drops rows. So
58/// the row sequence feeding `Limit` is byte-identical, in the same order, to the
59/// unbounded scan's — just possibly shorter. `execute_limit` takes
60/// `offset..offset+count`, a slice that depends solely on the first
61/// `limit + offset` rows. Producing exactly that many ACCEPTED rows therefore
62/// yields the identical final rows and ordering.
63///
64/// Pushed-down `SSTablePredicate`s are compatible: the caller counts rows it
65/// ACCEPTS (those that pass the predicate), not raw scan rows, so the cap still
66/// means "enough rows for the window".
67pub(super) fn scan_pushdown_cap(
68    steps: &[ExecutionStep],
69    select_clause: &SelectClause,
70) -> Option<usize> {
71    // DISTINCT may dedup rows after the scan; never bound the scan ahead of a
72    // potential row-reducing dedup.
73    if matches!(select_clause, SelectClause::Distinct(_)) {
74        return None;
75    }
76
77    let mut cap: Option<usize> = None;
78    for step in steps {
79        match step {
80            // Any step that REORDERS or REDUCES the row multiset between the scan
81            // and the query-wide LIMIT makes a raw scan bound unsafe: the final
82            // window could then be drawn from rows the bounded scan never emitted.
83            ExecutionStep::Sort { .. }
84            | ExecutionStep::Aggregate { .. }
85            | ExecutionStep::PerPartitionLimit { .. }
86            | ExecutionStep::Filter { .. } => return None,
87            ExecutionStep::Limit { count, offset } => {
88                let count = usize::try_from(*count).unwrap_or(usize::MAX);
89                let offset = offset
90                    .map(|o| usize::try_from(o).unwrap_or(usize::MAX))
91                    .unwrap_or(0);
92                cap = Some(count.saturating_add(offset));
93            }
94            // Order- and count-preserving; safe on either side of the cap.
95            ExecutionStep::SSTableScan { .. } | ExecutionStep::Project { .. } => {}
96        }
97    }
98    cap
99}

101impl SelectExecutor {
102    /// Full-scan fallback that stops DECODING once `cap` rows have been ACCEPTED
103    /// (issue #1577, D1).
104    ///
105    /// # Pre-materializing stream branches (roborev round-4)
106    ///
107    /// Some [`scan_stream`](crate::storage::StorageEngine::scan_stream) branches
108    /// PRE-MATERIALIZE the whole reconciled result before returning the channel —
109    /// any BTI (`da`) reader (whose trie-walk `bti_scan_with_metadata` decodes the
110    /// whole index-less reconciled table before streaming) and the whole-scan
111    /// `tombstones` build. (The `write-support` cross-generation merge is NOT one:
112    /// since #1579 it streams lazily via `stream_generations_for_read`, so a bounded
113    /// consumer decode-stops there.) For the pre-materializing branches the storage
114    /// layer decodes the ENTIRE table, so
115    /// the lazy per-received-row accounting below would under-report
116    /// `QUERY_ROWS_SCANNED` to ~`cap`. This method asks the storage layer
117    /// ([`scan_stream_materializes`](crate::storage::StorageEngine::scan_stream_materializes),
118    /// which owns the branch condition) up front and, when the stream would
119    /// pre-materialize, routes through the fully-materializing `scan` +
120    /// [`collect_capped_materialized`] (which charges the TRUE decoded count while
121    /// the `cap` bounds `rows_processed` and the returned window). No decode-stop is
122    /// possible in that case, so nothing is lost and results are byte-identical.
123    ///
124    /// The remainder of this method is the GENUINELY-LAZY single-generation path.
125    ///
126    /// The fast path consumes the lazy
127    /// [`scan_stream`](crate::storage::StorageEngine::scan_stream) — definitionally
128    /// in lockstep with the materializing `scan` (same token order, same
129    /// cross-generation reconciliation) — and drops it once `cap` rows are
130    /// accepted. Dropping the receiver closes the channel, so the producer stops
131    /// parsing the tail: on a 1M-row table a `LIMIT 10` decodes on the order of
132    /// `cap + buffer` rows, not a million (verified via the #1618
133    /// `PARTITION_HEADER_TRY_PARSES` counter).
134    ///
135    /// The cap counts rows the executor ACCEPTS: a row suppressed by
136    /// [`build_row_from_scan`] (a row tombstone / null-row `ScanRow::Marker`, Issue
137    /// #191) or rejected by [`evaluate_predicates`] is skipped and never counted,
138    /// so a suppressed marker or a predicate miss can never under-deliver.
139    ///
140    /// # Correctness: authoritative reconciliation
141    ///
142    /// If the stream yields FEWER than `cap` accepted rows before ending, that is
143    /// EITHER a table with fewer than `cap` accepted rows OR a `scan_stream` /
144    /// `scan` divergence (some SSTable formats — e.g. certain write-generated
145    /// uncompressed BIG files — under-produce via the block-streaming path). The
146    /// two are indistinguishable here, so this method DOES NOT trust a short
147    /// stream: it re-runs the AUTHORITATIVE materializing `scan` (the exact rows
148    /// the non-pushdown path returns) and takes its first `cap` accepted rows,
149    /// guaranteeing a byte-identical result. A short stream costs one extra scan
150    /// (of a table already known to hold `< cap` rows, or a divergent format);
151    /// there is no decode-stop win in that branch, but correctness is the law.
152    /// Because that re-run `scan` fully materializes the table, this branch charges
153    /// `QUERY_ROWS_SCANNED` the FULL decoded count via `collect_capped_materialized`
154    /// (not just `cap`), so the scan-work metric reflects the real work even when
155    /// the reconciled table holds more than `cap` accepted rows.
156    ///
157    /// A stream that reaches a FULL `cap` is trusted (returned early): its rows are
158    /// the token-first `cap` rows, identical to `scan` truncated to `cap` — the same
159    /// contract the streaming executor already relies on for every result it emits.
160    /// This single-generation lazy path is the ONLY unsafe one (multi-generation
161    /// returns above through the authoritative `scan`, so it is prefix-correct BY
162    /// CONSTRUCTION), so its trusted prefix is protected by a RELEASE-active,
163    /// cost-bounded `(token, key)`-monotonicity guard ([`prefix_is_token_ordered`]):
164    /// on a detected `scan_stream`/`scan` divergence it logs loudly and falls back
165    /// to the authoritative `scan` rather than ever returning the wrong rows.
166    pub(super) async fn capped_fallback_scan(
167        &self,
168        table: &TableId,
169        predicates: &[SSTablePredicate],
170        projection: &[String],
171        schema_opt: Option<&TableSchema>,
172        cap: usize,
173        context: &mut ExecutionContext,
174    ) -> Result<Vec<QueryRow>> {
175        // A `LIMIT 0` (cap == 0) can never accept a row; do not even open a scan.
176        if cap == 0 {
177            return Ok(Vec::new());
178        }
179
180        // ── Pre-materializing stream branches (issue #1577, roborev round-4) ──────
181        //
182        // The lazy fast path below assumes `scan_stream` is LAZY: it charges
183        // `context.scan_rows` per RECEIVED row and drops the stream at the cap, so
184        // dropping the channel stops the producer decoding the tail. That is TRUE
185        // for the genuinely-streaming single-generation merge AND, since #1579, for
186        // the lazy `write-support` cross-generation merge (which streams via
187        // `generation_merge::stream_generations_for_read`). But some `scan_stream`
188        // branches PRE-MATERIALIZE the entire reconciled result before handing back
189        // the channel — any BTI (`da`) reader (trie-walk `bti_scan_with_metadata`
190        // decodes the whole index-less table before streaming) and the whole-scan
191        // `tombstones` build. For those the storage layer decodes
192        // the ENTIRE table regardless of the cap, so consuming lazily and charging
193        // per-received-row would report only ~`cap` rows to `QUERY_ROWS_SCANNED`
194        // while the true decode work is the full table — a metric regression.
195        //
196        // Ask the storage layer (which owns the branch condition — no duplicated
197        // storage-internal logic here) whether `scan_stream` would pre-materialize
198        // for this table + schema. If so, route through the fully-materializing
199        // `scan` + the shared `collect_capped_materialized` accountant, which
200        // charges the TRUE decoded count (`materialized.len()`) up front while the
201        // `cap` still bounds `rows_processed` and the returned window. There is no
202        // decode-stop to lose here (the storage already decoded everything), and the
203        // authoritative `scan` yields byte-identical rows, so RESULTS and
204        // LIMIT/OFFSET semantics are unchanged — only the accounting path differs.
205        if self
206            .storage
207            .scan_stream_materializes(table, schema_opt)
208            .await
209        {
210            let materialized = self
211                .storage
212                .scan(table, None, None, None, schema_opt)
213                .await?;
214            // Issue #1817: hoist the partition-key decode across a partition's rows.
215            let mut pk_cache = super::PartitionKeyCache::default();
216            return collect_capped_materialized(
217                materialized,
218                Some(cap),
219                predicates,
220                context,
221                |(key, value)| {
222                    build_row_from_scan_cached(key, value, projection, schema_opt, &mut pk_cache)
223                },
224            );
225        }
226
227        // Metric accuracy (issue #1577, SUGGESTION-3): snapshot the scan counters
228        // so the short-stream reconciliation below can charge the AUTHORITATIVE
229        // re-scan exactly ONCE. Without this, the partial stream's per-row
230        // increments PLUS the re-scan's increments both land in
231        // `QUERY_ROWS_SCANNED` for the reconcile path, inflating rows-scanned.
232        // Results are unaffected — this only fixes double-counting.
233        let processed_baseline = context.rows_processed;
234        let scan_rows_baseline = context.scan_rows;
235
236        let buffer = cap.saturating_add(1).min(CAPPED_SCAN_STREAM_BUFFER);
237        let mut scan_stream = self
238            .storage
239            .scan_stream(table, None, None, schema_opt, buffer)
240            .await?;
241
242        let mut results = Vec::new();
243        // Issue #1817: hoist the partition-key decode across a partition's rows.
244        let mut pk_cache = super::PartitionKeyCache::default();
245        while let Some(item) = scan_stream.recv().await {
246            let (key, value) = item?;
247            context.rows_processed += 1;
248            context.scan_rows += 1;
249
250            let Some(row) =
251                build_row_from_scan_cached(key, value, projection, schema_opt, &mut pk_cache)
252            else {
253                continue;
254            };
255
256            if evaluate_predicates(&row, predicates)? {
257                results.push(row);
258                if results.len() >= cap {
259                    // ── Trusted full-cap stream fast path (issue #1577) ──────────
260                    //
261                    // Decode-stop win: dropping the stream closes the channel and
262                    // the producer stops parsing the remaining (unneeded) rows. So
263                    // we return the stream's first `cap` rows WITHOUT re-running the
264                    // authoritative `scan` (which would decode the whole table and
265                    // defeat D1's entire purpose).
266                    //
267                    // INVARIANT — this GENUINELY-LAZY single-generation branch is
268                    // the ONLY unsafe path. Here `scan` (materializing, token-sorted
269                    // via `sort_by_token_order`) and `scan_stream` (a SEPARATE lazy
270                    // pipeline that does NOT sort — it trusts on-disk order == token
271                    // order) are kept in lockstep only by PARITY TESTS, not by
272                    // construction. (By contrast, the multi-generation branch above
273                    // returned early through the authoritative `scan`, so it is
274                    // prefix-correct BY CONSTRUCTION and needs no guard here.)
275                    // Returning the lazy prefix is sound ONLY because every SUPPORTED
276                    // writer emits rows in token order:
277                    //   * Cassandra 5.0 SSTable `Data.db` files are token-ordered on
278                    //     disk;
279                    //   * CQLite's memtable is a token-ordered `BTreeMap`, so every
280                    //     flushed generation is token-ordered;
281                    //   * compaction output is a k-way TOKEN merge of token-ordered
282                    //     inputs.
283                    //
284                    // ── Release-active, cost-bounded guard (issue #1577, owner
285                    //    2026-07-06) ─────────────────────────────────────────────
286                    // A prior version verified this invariant ONLY under a
287                    // `debug_assert`, so a RELEASE build ran with NO check and could
288                    // silently return the WRONG rows if the two pipelines ever
289                    // diverged. We now run a cheap O(cap) check on EVERY build: the
290                    // authoritative `scan` emits rows in ascending `(token, key)`
291                    // order, so if this trusted prefix is NOT itself
292                    // `(token, key)`-monotonic it cannot be that authoritative
293                    // prefix — a divergence. The check costs `cap` murmur3 hashes
294                    // (`cap` == LIMIT+OFFSET, the work already done to receive the
295                    // rows), so it NEVER decodes the tail and preserves the
296                    // decode-stop win.
297                    //
298                    // On a detected divergence we NEVER return the possibly-wrong
299                    // prefix: we log LOUDLY and fall back to the authoritative
300                    // materializing `scan` (the exact rows the non-pushdown path
301                    // returns), rolling the scan counters back to baseline first so
302                    // the reconcile charges `QUERY_ROWS_SCANNED` exactly once. That
303                    // fallback decodes the whole table, but it runs ONLY on the
304                    // (should-be-impossible, for supported writers) divergence — the
305                    // happy path stays decode-bounded.
306                    //
307                    // RESIDUAL RISK (named per the invariant doc): the O(cap) check
308                    // proves the prefix is INTERNALLY token-monotonic, which catches
309                    // any SYSTEMATIC reordering (insertion / clustering / reverse
310                    // order all yield a non-monotonic prefix on realistic data). It
311                    // does NOT catch a divergence that leaves the prefix
312                    // coincidentally monotonic while a lower-token row sits just past
313                    // the cap boundary (a single transposition straddling `cap`) —
314                    // detecting that needs the tail we deliberately skip. The
315                    // debug-only `debug_assert_trusted_prefix` below closes that
316                    // residual gap exhaustively (full authoritative parity) in
317                    // debug/test builds.
318                    if !prefix_is_token_ordered(&results) {
319                        tracing::error!(
320                            "issue #1577 invariant violated in RELEASE-active guard: the trusted \
321                             full-cap `scan_stream` prefix is not in ascending (token, key) order, \
322                             so it cannot be the authoritative token-ordered `scan` prefix — a \
323                             `scan_stream`/`scan` divergence on the single-generation lazy path. \
324                             Every supported writer must emit rows in token order. Falling back to \
325                             the authoritative materializing `scan` to guarantee correct rows."
326                        );
327                        drop(scan_stream);
328                        return self
329                            .reconcile_via_authoritative_scan(
330                                table,
331                                predicates,
332                                projection,
333                                schema_opt,
334                                cap,
335                                context,
336                                processed_baseline,
337                                scan_rows_baseline,
338                            )
339                            .await;
340                    }
341
342                    // Exhaustive debug-only guard: re-runs the authoritative
343                    // token-ordered `scan` and verifies the WHOLE trusted prefix
344                    // (catching the residual-risk case the O(cap) check cannot). It
345                    // is compiled out of release (zero release cost) AND of the
346                    // `work-counters` measurement build, whose sole job is to count
347                    // decode work — the guard's verification `scan` parses the whole
348                    // fixture and would pollute the `PARTITION_HEADER_TRY_PARSES`
349                    // counter the decode-stop test asserts. It still runs in a normal
350                    // debug build (no `work-counters`).
351                    #[cfg(all(debug_assertions, not(feature = "work-counters")))]
352                    self.debug_assert_trusted_prefix(
353                        table, predicates, projection, schema_opt, cap, &results,
354                    )
355                    .await?;
356                    return Ok(results);
357                }
358            }
359        }
360
361        // Short stream: reconcile against the authoritative materializing scan so
362        // a `scan_stream`/`scan` divergence can never drop a row. The shared
363        // `reconcile_via_authoritative_scan` rolls the scan counters back to the
364        // pre-stream baseline first (SUGGESTION-3) so the re-scan is the only work
365        // charged to `QUERY_ROWS_SCANNED` for this path — the partial stream's
366        // examined rows are not double-counted.
367        drop(scan_stream);
368        self.reconcile_via_authoritative_scan(
369            table,
370            predicates,
371            projection,
372            schema_opt,
373            cap,
374            context,
375            processed_baseline,
376            scan_rows_baseline,
377        )
378        .await
379    }
380
381    /// Reconcile a bounded scan against the AUTHORITATIVE materializing `scan`
382    /// (issue #1577) — the single fallback shared by BOTH non-decode-bounded
383    /// branches of [`capped_fallback_scan`](Self::capped_fallback_scan): the
384    /// SHORT-STREAM branch (the stream ended before the cap) and the
385    /// RELEASE-active divergence guard (the trusted prefix was not token-ordered).
386    ///
387    /// It first rolls the scan counters back to the pre-stream baseline so the
388    /// partial stream's per-row increments are not double-counted, then re-runs the
389    /// fully-materializing `scan` (the exact rows the non-pushdown path returns) and
390    /// routes it through the SAME `collect_capped_materialized` accountant the
391    /// materializing `execute_sstable_scan` paths use. That charges
392    /// `context.scan_rows` (→ `QUERY_ROWS_SCANNED`) with the TRUE decoded count
393    /// (`authoritative.len()`) UP FRONT — not just the `cap` rows it examines —
394    /// while the `cap` still bounds `rows_processed` and the returned window, so
395    /// RESULTS and LIMIT/OFFSET semantics are byte-identical to the unbounded scan.
396    /// This branch is NOT decode-bounded (unlike the trusted full-cap fast path,
397    /// which drops its stream at the cap); it fully materializes the table, so it is
398    /// only ever reached when decode-stop is impossible or must be abandoned for
399    /// correctness.
400    #[allow(clippy::too_many_arguments)]
401    async fn reconcile_via_authoritative_scan(
402        &self,
403        table: &TableId,
404        predicates: &[SSTablePredicate],
405        projection: &[String],
406        schema_opt: Option<&TableSchema>,
407        cap: usize,
408        context: &mut ExecutionContext,
409        processed_baseline: u64,
410        scan_rows_baseline: u64,
411    ) -> Result<Vec<QueryRow>> {
412        context.rows_processed = processed_baseline;
413        context.scan_rows = scan_rows_baseline;
414        let authoritative = self
415            .storage
416            .scan(table, None, None, None, schema_opt)
417            .await?;
418        // Issue #1817: hoist the partition-key decode across a partition's rows.
419        let mut pk_cache = super::PartitionKeyCache::default();
420        collect_capped_materialized(
421            authoritative,
422            Some(cap),
423            predicates,
424            context,
425            |(key, value)| {
426                build_row_from_scan_cached(key, value, projection, schema_opt, &mut pk_cache)
427            },
428        )
429    }

431    /// Debug-only guard for the trusted full-cap stream fast path in
432    /// [`capped_fallback_scan`](Self::capped_fallback_scan) (issue #1577).
433    ///
434    /// Re-runs the AUTHORITATIVE materializing `scan` (whose rows are token-sorted
435    /// via `sort_by_token_order`), builds its first-`cap` ACCEPTED row keys, and
436    /// asserts they equal the trusted stream's row keys in order. A supported
437    /// writer emits rows in token order, so the two prefixes match; a future
438    /// writer that violates the invariant trips this `debug_assert` in debug/tests
439    /// instead of silently returning misordered rows.
440    ///
441    /// Compiled only under `debug_assertions`, so it has zero release perf cost.
442    /// Also compiled out under the `work-counters` measurement build: its
443    /// authoritative verification `scan` parses the whole fixture and would
444    /// pollute the decode-work counters that build exists to measure. The guard
445    /// still runs in a normal debug build (no `work-counters`).
446    #[cfg(all(debug_assertions, not(feature = "work-counters")))]
447    async fn debug_assert_trusted_prefix(
448        &self,
449        table: &TableId,
450        predicates: &[SSTablePredicate],
451        projection: &[String],
452        schema_opt: Option<&TableSchema>,
453        cap: usize,
454        trusted: &[QueryRow],
455    ) -> Result<()> {
456        let authoritative = self
457            .storage
458            .scan(table, None, None, None, schema_opt)
459            .await?;
460        use crate::types::RowKey;
461        let mut expected: Vec<RowKey> = Vec::with_capacity(cap);
462        for (key, value) in authoritative {
463            if expected.len() >= cap {
464                break;
465            }
466            if let Some(row) = super::build_row_from_scan(key, value, projection, schema_opt) {
467                if evaluate_predicates(&row, predicates)? {
468                    expected.push(row.key);
469                }
470            }
471        }
472        let got: Vec<RowKey> = trusted.iter().map(|r| r.key.clone()).collect();
473        debug_assert_eq!(
474            got, expected,
475            "issue #1577 invariant violated: the trusted full-cap `scan_stream` prefix \
476             diverged from the authoritative token-ordered `scan` prefix. Every supported \
477             writer must emit rows in token order (Cassandra 5.0 on-disk files are \
478             token-ordered; CQLite's memtable is a token-ordered BTreeMap; compaction \
479             output is k-way token-merged). A writer that violates this must sort before \
480             emit, or the full-cap stream fast path returns rows in the wrong order."
481        );
482        Ok(())
483    }
484}

486/// Cheap O(cap) release-active guard for the trusted full-cap `scan_stream`
487/// prefix (issue #1577, owner 2026-07-06).
488///
489/// The authoritative materializing `scan` emits rows in ascending
490/// `(token, key_bytes)` order (`kway_merge_token_order` / `sort_by_token_order`,
491/// where `token = cassandra_murmur3_token(partition_key)`). A trusted lazy prefix
492/// can only equal that authoritative prefix if it is ITSELF `(token, key)`-
493/// monotonic, so this verifies exactly that — the comparison is NON-strict
494/// (`<=`), because a partition's clustering rows share one `(token, key)` and
495/// arrive contiguously (see `sort_by_token_order`'s stability note). Returns
496/// `false` on the first descending adjacent pair (a `scan_stream`/`scan`
497/// divergence on the single-generation lazy path).
498///
499/// Cost is `cap` murmur3 hashes — proportional to the rows already received, so it
500/// never decodes the tail and preserves D1's decode-stop optimization. It cannot
501/// see rows past the cap, so it catches SYSTEMATIC reordering but not a single
502/// transposition straddling the cap boundary (see `capped_fallback_scan`'s
503/// residual-risk note; the debug-only exhaustive guard closes that gap).
504fn prefix_is_token_ordered(rows: &[QueryRow]) -> bool {
505    use crate::util::cassandra_murmur3::cassandra_murmur3_token;
506    let mut prev: Option<(i64, &[u8])> = None;
507    for row in rows {
508        let key = row.key.as_bytes();
509        let token = cassandra_murmur3_token(key);
510        if let Some(prev) = prev {
511            if prev > (token, key) {
512                return false;
513            }
514        }
515        prev = Some((token, key));
516    }
517    true
518}

520/// Collect the ACCEPTED rows from an ALREADY-MATERIALIZED scan, applying an
521/// optional LIMIT+OFFSET `cap` (issue #1577; roborev metric-accounting fix).
522///
523/// The storage layer decoded EVERY entry before returning `rows`, so this
524/// charges `context.scan_rows` — the sole source of the `QUERY_ROWS_SCANNED`
525/// metric — with the TRUE decoded count (`rows.len()`) UP FRONT. The `cap` only
526/// bounds the per-row BUILD/predicate work (`context.rows_processed`) and the
527/// size of the returned window; it must NOT shrink the scan-work metric, or the
528/// metric would under-report the scan the storage layer actually performed.
529///
530/// The materializing `execute_sstable_scan` metadata / partition-targeted paths
531/// AND the short-stream reconciliation branch of
532/// [`SelectExecutor::capped_fallback_scan`] (which re-runs the fully-materializing
533/// `scan`) all route through here, so the full-decode accounting is applied
534/// uniformly and cannot drift per call site. This is deliberately distinct from
535/// the TRULY decode-bounded trusted full-cap stream fast path in
536/// `capped_fallback_scan`, which drops its bounded stream at the cap and so
537/// legitimately charges `scan_rows` only for the rows it actually decoded.
538///
539/// `build` maps a materialized entry to an optional [`QueryRow`] (`None` = a
540/// suppressed marker / tombstone, per [`build_row_from_scan`]); the metadata
541/// caller uses it to attach per-cell metadata BEFORE predicate evaluation. A row
542/// that `build` suppresses or [`evaluate_predicates`] rejects is skipped and
543/// never counted toward `cap`, so the cap means "enough ACCEPTED rows" and can
544/// never under-deliver a match.
545pub(super) fn collect_capped_materialized<T>(
546    rows: Vec<T>,
547    cap: Option<usize>,
548    predicates: &[SSTablePredicate],
549    context: &mut ExecutionContext,
550    mut build: impl FnMut(T) -> Option<QueryRow>,
551) -> Result<Vec<QueryRow>> {
552    // Metric accuracy (issue #1577): the WHOLE scan was materialized/decoded, so
553    // charge the full decoded count regardless of the downstream cap. Charging
554    // per-iteration inside the capped loop below would stop at the cap and make
555    // `QUERY_ROWS_SCANNED` under-report to at most `LIMIT + OFFSET`.
556    let total = rows.len();
557    context.scan_rows += total as u64;
558
559    let mut out = Vec::with_capacity(cap.map_or(total, |c| c.min(total)));
560    for entry in rows {
561        if let Some(c) = cap {
562            if out.len() >= c {
563                break;
564            }
565        }
566        context.rows_processed += 1;
567        let Some(row) = build(entry) else {
568            continue;
569        };
570        if evaluate_predicates(&row, predicates)? {
571            out.push(row);
572        }
573    }
574    Ok(out)
575}
