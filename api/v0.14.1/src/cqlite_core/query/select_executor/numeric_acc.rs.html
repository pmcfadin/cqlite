1//! Numeric accumulation domain for SUM/AVG (issue #2202).
2//!
3//! Split from `aggregation.rs` (campsite rule, epic #1116) so the accumulator's
4//! type-preserving arithmetic lives in one small, self-contained place.
5
6use crate::schema::CqlType;
7use crate::types::Value;
8
9/// The integral width SUM/AVG accumulates and narrows to at finalize (issue
10/// #2202). Cassandra's `AggregateFcts.java` returns the SAME type as the
11/// argument for every integral column — it does NOT promote `tinyint`/
12/// `smallint` to `int` (its own docs warn of an overflow risk precisely
13/// because the result stays the input's narrow width).
14#[derive(Debug, Clone, Copy, PartialEq, Eq)]
15pub(super) enum IntWidth {
16    /// `tinyint` — narrows to `Value::TinyInt` (i8), wraps at the i8 boundary.
17    Byte,
18    /// `smallint` — narrows to `Value::SmallInt` (i16), wraps at the i16 boundary.
19    Short,
20    /// `int` — narrows to `Value::Integer` (i32), wraps at the i32 boundary.
21    Int,
22    /// `bigint`/`counter` — `Value::BigInt` (i64) directly, no narrowing.
23    Long,
24}
25
26impl IntWidth {
27    /// Narrow a wide (`i64`) accumulator to this width's `Value` variant. `as`
28    /// truncation is Rust's two's-complement narrowing cast — the same
29    /// semantics as Java's narrowing primitive conversion (`(byte) x`/
30    /// `(short) x`/`(int) x`) that Cassandra's own aggregate functions use, so
31    /// this is exact parity, not an approximation.
32    fn narrow(self, sum: i64) -> Value {
33        match self {
34            IntWidth::Byte => Value::TinyInt(sum as i8),
35            IntWidth::Short => Value::SmallInt(sum as i16),
36            IntWidth::Int => Value::Integer(sum as i32),
37            IntWidth::Long => Value::BigInt(sum),
38        }
39    }
40}

42/// Numeric accumulation domain for SUM/AVG (issue #2202). The domain is fixed at
43/// group-init time from the aggregate's resolved RESULT CQL type (the SAME
44/// `select_naming::sum_avg_result_cql_type` the result metadata is built from),
45/// so the emitted value variant can never disagree with the metadata type.
46#[derive(Debug, Clone)]
47pub(super) enum NumericAcc {
48    /// Integral SUM/AVG. `sum` accumulates in `i64` with Cassandra's two's-
49    /// complement WRAPPING overflow (Java integer/long semantics) — never
50    /// saturating or panicking — and narrows to `width` only at finalize.
51    ///
52    /// Wrapping equivalence: modular (two's-complement) addition is associative,
53    /// so accumulating in i64 with `wrapping_add` and truncating to the narrow
54    /// width at the end yields exactly the same result as wrapping at every
55    /// step in that narrow width (`(x mod 2^64) mod 2^n == x mod 2^n`). This
56    /// also matches Cassandra's OWN implementation: its AVG aggregates sum in a
57    /// Java `long` and narrow only at `compute()`, exactly like this. AVG uses
58    /// i64 integer division on the wide sum (truncating toward zero, matching
59    /// Java) before narrowing; an AVG over `bigint` whose i64 sum overflows i64
60    /// can diverge from Cassandra's BigInteger AVG — an extreme edge (summing
61    /// enough near-i64::MAX values) that real data never hits.
62    Integral { sum: i64, width: IntWidth },
63    /// Floating SUM/AVG (`float`/`double`/`decimal`/`varint`/unknown argument) —
64    /// accumulated in `f64` and emitted as `Value::Float` (`double`), matching
65    /// CQLite's pre-#2202 behaviour with no regression.
66    Floating(f64),
67}
68
69impl NumericAcc {
70    /// Initialize the accumulation domain from the aggregate's resolved result
71    /// CQL type. `tinyint`/`smallint`/`int` each get their OWN narrow integral
72    /// width (Cassandra does not promote them); `bigint`/`counter` → wide
73    /// integral; everything else (incl. an unknown/`None` type) → floating.
74    /// Mirrors [`crate::query::select_naming::sum_avg_result_cql_type`].
75    pub(super) fn init(result_type: Option<&CqlType>) -> Self {
76        let width = match result_type {
77            Some(CqlType::TinyInt) => IntWidth::Byte,
78            Some(CqlType::SmallInt) => IntWidth::Short,
79            Some(CqlType::Int) => IntWidth::Int,
80            Some(CqlType::BigInt | CqlType::Counter) => IntWidth::Long,
81            _ => return NumericAcc::Floating(0.0),
82        };
83        NumericAcc::Integral { sum: 0, width }
84    }
85
86    /// Fold one non-null input value into the accumulator, returning whether the
87    /// value was accepted (converted in this domain). Integral domains read
88    /// `as_i64` and wrap; the floating domain reads `as_f64`. A value that does
89    /// not convert (wrong variant for the domain) is ignored and returns `false`,
90    /// exactly as the prior `as_f64`-only path ignored non-numeric inputs. AVG
91    /// increments its count only on an accepted value, so a `0` still counts.
92    pub(super) fn add(&mut self, value: &Value) -> bool {
93        match self {
94            NumericAcc::Integral { sum, .. } => match value.as_i64() {
95                Some(v) => {
96                    *sum = sum.wrapping_add(v);
97                    true
98                }
99                None => false,
100            },
101            NumericAcc::Floating(sum) => match value.as_f64() {
102                Some(v) => {
103                    *sum += v;
104                    true
105                }
106                None => false,
107            },
108        }
109    }
110
111    /// The finalized SUM value: integral domains narrow the wide `i64` sum to
112    /// their own width (`Value::TinyInt`/`SmallInt`/`Integer`/`BigInt`); the
113    /// floating domain emits `Value::Float`.
114    pub(super) fn finalize_sum(&self) -> Value {
115        match self {
116            NumericAcc::Integral { sum, width } => width.narrow(*sum),
117            NumericAcc::Floating(sum) => Value::Float(*sum),
118        }
119    }
120
121    /// The finalized AVG value over `count` accepted inputs. Integral domains
122    /// divide the wide `i64` sum by `count` (integer division, truncating
123    /// toward zero, like Java) and narrow to their own width; the floating
124    /// domain divides in f64.
125    ///
126    /// Precondition: `count > 0` — every caller (`finalize_group`,
127    /// `finalize_empty_global_aggregate`) checks this first, so an empty group
128    /// never reaches here; asserted rather than guarded so a future caller
129    /// cannot silently reintroduce a divide-by-zero panic.
130    pub(super) fn finalize_avg(&self, count: u64) -> Value {
131        debug_assert!(
132            count > 0,
133            "finalize_avg requires at least one accepted input"
134        );
135        match self {
136            NumericAcc::Integral { sum, width } => width.narrow(sum / count as i64),
137            NumericAcc::Floating(sum) => Value::Float(sum / count as f64),
138        }
139    }
140}
