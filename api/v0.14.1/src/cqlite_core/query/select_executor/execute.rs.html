1//! Async pipeline entry points for the SELECT executor (issue #1174).
2//!
3//! This submodule continues the [`SelectExecutor`](super::SelectExecutor) `impl`
4//! with the large `async` pipeline methods that drive query execution:
5//! - [`SelectExecutor::execute`] — the materializing plan runner,
6//! - [`SelectExecutor::execute_sstable_scan`] — the SSTable-scan step.
7//!
8//! (The streaming producer, `execute_streaming_background`, moved to the sibling
9//! `streaming` submodule in issue #1578's file-size split.)
10//!
11//! These were relocated verbatim from `mod.rs` (epic #1116 file-size split); the
12//! per-step helpers they call (`execute_filter`, `execute_sort`, etc.) and the
13//! `ExecutionContext` bookkeeping struct remain in `mod.rs`. As a child module,
14//! this file can reach `mod.rs`'s private items directly — the logic, ordering,
15//! and error handling are unchanged.

31impl SelectExecutor {
32    /// Execute an optimized query plan.
33    ///
34    /// Instrumented as `query.select.plan` (issue #1035): this span covers the
35    /// modern SELECT pipeline — SSTable scan, filtering, projection, aggregation,
36    /// and WRITETIME/TTL metadata extraction — and is the parent under which the
37    /// read-path spans (issue #1034) nest. On completion it emits
38    /// [`catalog::QUERY_ROWS_SCANNED`] (rows the scan step examined) dimensioned by
39    /// the honest access path, so the rows-scanned vs rows-returned gap is
40    /// observable. The bounded access-path attribute is recorded on the span; the
41    /// query text and key values never are.
42    #[tracing::instrument(
43        name = "query.select.plan",
44        skip_all,
45        fields(
46            cqlite.query.access_path = tracing::field::Empty,
47            cqlite.query.rows_scanned = tracing::field::Empty,
48            cqlite.query.rows = tracing::field::Empty,
49        )
50    )]
51    pub async fn execute(&self, plan: OptimizedQueryPlan) -> Result<QueryResult> {
52        // Issue #960: clear the global access-path probe so a stale value from a
53        // previous query cannot satisfy a test assertion against this one.
54        crate::query::access_path::reset();
55
56        let table_id = if let Some(ref from_clause) = plan.statement.from_clause {
57            self.extract_table_id(from_clause)?
58        } else {
59            // For queries without FROM clause (like SELECT 1), use a dummy table ID
60            TableId::new("_dummy_")
61        };
62
63        // Issue #1587 (E5): resolve the table's schema ONCE per query into a shared
64        // `Arc<TableSchema>`. Column-metadata building, the SSTable scan, and the
65        // SELECT-* metadata fallback all borrow this same schema (ref-count bump),
66        // instead of each independently re-locking the registry and deep-cloning a
67        // fresh `TableSchema` (2–4 deep clones per query before this).
68        let query_schema: Option<Arc<TableSchema>> = if plan.statement.from_clause.is_some() {
69            self.resolve_table_schema(&table_id).await
70        } else {
71            None
72        };
73
74        // Issue #692: detect whether any WRITETIME/TTL select items are present
75        // during planning and set the opt-in flag so the reader threads per-cell
76        // metadata. This is the "planning" half of the executor wiring; the
77        // "evaluation" half lives in `evaluate_select_expression`.
78        let projection_flags = ProjectionFlags {
79            include_cell_metadata: select_has_writetime_ttl(&plan.statement),
80        };
81        tracing::debug!(
82            "Query plan: include_cell_metadata={}",
83            projection_flags.include_cell_metadata
84        );
85
86        let mut context = ExecutionContext {
87            table_id,
88            columns: self.get_result_columns(&plan.statement, query_schema.as_deref())?,
89            rows_processed: 0,
90            scan_rows: 0,
91            projection_flags,
92            access_path: None,
93            reverse_served: false,
94        };
95
96        // Handle queries without FROM clause (like SELECT 1)
97        if plan.statement.from_clause.is_none() {
98            let mut result = self.execute_constant_query(&plan.statement, &context)?;
99            // Issue #1582 (roborev): apply the statement's LIMIT/OFFSET to the
100            // constant rows BEFORE the byte + row-count budget check, so the budget
101            // is enforced on the rows ACTUALLY returned (post LIMIT/OFFSET) —
102            // consistent with the table-backed path below. In particular `LIMIT 0`
103            // must return empty, never `ResultTooLarge`; an over-budget constant
104            // SELECT with NO limit still trips the guard on its final rows.
105            let offset = plan.statement.offset.unwrap_or(0) as usize;
106            let limit = plan
107                .statement
108                .limit
109                .as_ref()
110                .map(|l| l.count as usize)
111                .unwrap_or(usize::MAX);
112            result.rows = result.rows.into_iter().skip(offset).take(limit).collect();
113            // Keep the row-count metadata consistent with the returned rows.
114            let returned = result.rows.len() as u64;
115            result.rows_affected = returned;
116            result.metadata.total_rows = Some(returned);
117            enforce_materialized_rows(&result.rows, self.max_result_bytes, self.max_result_rows)?;
118            return Ok(result);
119        }
120
121        // Execute the plan step by step
122        let mut intermediate_results = Vec::new();
123
124        // If no execution steps are provided, add a default table scan
125        let execution_steps = if plan.execution_steps.is_empty() {
126            vec![ExecutionStep::SSTableScan {
127                table: context.table_id.clone(),
128                predicates: vec![],
129                projection: context.columns.iter().map(|c| c.name.clone()).collect(),
130            }]
131        } else {
132            plan.execution_steps.clone()
133        };
134
135        // Issue #1577 (D1): decide ONCE — from the plan alone — whether the
136        // SSTable scan may stop early at `LIMIT + OFFSET` accepted rows. `None`
137        // (any Sort/Aggregate/PerPartitionLimit/residual-Filter, DISTINCT, or no
138        // LIMIT) leaves the scan unbounded, exactly as before. Computed here so it
139        // is available inside the step loop below (unused, harmlessly, when the
140        // #1578 global-aggregate fast path takes the scan's place).
141        let scan_cap = scan_pushdown_cap(&execution_steps, &plan.statement.select_clause);
142
143        // Issue #1578 (D2): fold a GROUP-BY-free aggregate over a full table scan
144        // into an O(1) accumulator instead of buffering the whole table here.
145        // Returns None for any plan shape it does not model (GROUP BY, targeted
146        // lookup, WRITETIME/TTL, or a Sort/Project/Limit step) → the buffered step
147        // loop below runs unchanged.
148        if let Some(rows) = self
149            .try_execute_global_aggregate(&execution_steps, query_schema.as_deref(), &mut context)
150            .await?
151        {
152            intermediate_results = rows;
153        } else {
154            for step in &execution_steps {
155                match step {
156                    ExecutionStep::SSTableScan {
157                        table,
158                        predicates,
159                        projection,
160                        ..
161                    } => {
162                        let rows = self
163                            .execute_sstable_scan(
164                                table,
165                                predicates,
166                                projection,
167                                plan.statement.order_by.as_ref(),
168                                query_schema.as_deref(),
169                                scan_cap,
170                                &mut context,
171                            )
172                            .await?;
173                        intermediate_results = rows;
174                    }
175                    ExecutionStep::Filter { expression, .. } => {
176                        intermediate_results =
177                            self.execute_filter(intermediate_results, expression, &mut context)?;
178                    }
179                    ExecutionStep::Sort { order_by, .. } => {
180                        // Issue #1184: when the BIG reverse partition iterator already
181                        // produced the rows in descending clustering order, skip the
182                        // in-memory sort entirely (it remains the fallback otherwise).
183                        if !context.reverse_served {
184                            intermediate_results =
185                                self.execute_sort(intermediate_results, order_by, &mut context)?;
186                        } else {
187                            // Issue #1307 (hardening): skipping this Sort is sound ONLY
188                            // because `reverse_served` is set exclusively by
189                            // `targeted_partition_rows`, which serves the reverse
190                            // promoted-index iterator precisely when `statement.order_by`
191                            // requests the reverse of the stored clustering order — and
192                            // the planner emits EXACTLY ONE `Sort` step, cloned from that
193                            // same `statement.order_by` (see `select_optimizer.rs`). So
194                            // the reverse scan's ordering matches this step's `order_by`,
195                            // and skipping it drops a redundant sort rather than a
196                            // different ordering. What would break the invariant: a
197                            // multi-table / join plan (or any plan) that reused the flag
198                            // across a Sort NOT derived from the reverse-served scan's
199                            // `statement.order_by` — such a plan must clear
200                            // `reverse_served` before this step. The debug_assert pins
201                            // the property (the skipped Sort's key equals the statement's
202                            // order_by); it is debug-only and never alters release
203                            // behavior.
204                            debug_assert!(
205                                plan.statement.order_by.as_ref() == Some(order_by),
206                                "reverse_served Sort-skip invariant violated: the skipped \
207                             Sort's order_by must be the statement's order_by that drove \
208                             the reverse-served scan (single-table plan); a plan whose \
209                             Sort is not the reverse scan's matching Sort must clear \
210                             reverse_served first",
211                            );
212                        }
213                    }
214                    ExecutionStep::Aggregate { plan: agg_plan, .. } => {
215                        intermediate_results = self.execute_aggregation(
216                            intermediate_results,
217                            agg_plan,
218                            query_schema.as_deref(),
219                            &mut context,
220                        )?;
221                    }
222                    ExecutionStep::PerPartitionLimit { count } => {
223                        intermediate_results =
224                            Self::execute_per_partition_limit(intermediate_results, *count);
225                    }
226                    ExecutionStep::Limit { count, offset } => {
227                        intermediate_results = self.execute_limit(
228                            intermediate_results,
229                            *count,
230                            *offset,
231                            &mut context,
232                        )?;
233                    }
234                    ExecutionStep::Project { columns } => {
235                        // Issue #1952 (round-6 fix): branch on whether the projection
236                        // RESHAPES the row. A plain-column Project only trims the
237                        // #1952-widened helper columns — route it through the
238                        // key-preserving `trim_projection` so the row keeps its real
239                        // RowKey / metadata and a sparse row's absent selected cell is
240                        // omitted rather than erroring. Only a reshaping / computed
241                        // projection (alias, arithmetic, aggregate, function,
242                        // writetime/ttl, collection-access) goes through
243                        // `execute_projection`, whose empty-RowKey + name-derivation is
244                        // correct for a computed row that has no natural stored key.
245                        intermediate_results = if columns.iter().any(project_expr_reshapes_row) {
246                            self.execute_projection(intermediate_results, columns, &mut context)?
247                        } else {
248                            self.trim_projection(intermediate_results, columns)
249                        };
250                    }
251                }
252            }
253        }
254
255        // Issue #1582 (D6, narrow subset): enforce the byte-bounded result budget
256        // (primary) + the row-count safety valve (secondary) with a SINGLE robust
257        // check on the FINAL materialized result — AFTER every execution step
258        // (Limit/Offset/Filter/Sort/Aggregate/Project) has produced the rows that
259        // will actually be returned. Because this sees ONLY the returned rows
260        // (post LIMIT/OFFSET), a `LIMIT 10` query never trips and OFFSET-skipped
261        // rows are never charged. This replaces the earlier during-collection
262        // machinery (LIMIT/OFFSET pushdown, per-row early-stop, storage-layer row limit),
263        // which kept generating correctness edge cases; a single final-result
264        // check has no such edges. Reuses the shared `estimate_value_size`
265        // estimator (via `enforce_materialized_rows`).
266        //
267        // SCOPE (owner-accepted boundaries, tracked on #1582 — NOT bugs to fix in
268        // this narrow subset):
269        //   * Does NOT bound PEAK scan memory: `storage.scan` still materializes
270        //     each reader's matching rows before this point (deferred to #1897).
271        //   * Does NOT cover the LEGACY point-lookup `QueryExecutor` path
272        //     (`WHERE id = ?` short lookups route there, not through this modern
273        //     executor) — deferred to the D6 redesign.
274        // Issue #1578 (D2): demote the row-count valve to a genuine safety valve.
275        // A query with an EXPLICIT `LIMIT` already bounds its own result, so it is
276        // exempt from the crude row-count ceiling (the user accepted the count);
277        // the byte budget still guards memory. Without an explicit LIMIT the valve
278        // remains a real net against unbounded materialization.
279        let effective_max_rows = if plan.statement.limit.is_some() {
280            usize::MAX
281        } else {
282            self.max_result_rows
283        };
284        enforce_materialized_rows(
285            &intermediate_results,
286            self.max_result_bytes,
287            effective_max_rows,
288        )?;
289
290        let total_rows = intermediate_results.len() as u64;
291
292        // CRITICAL FIX (Issue #129/#140): Populate metadata.columns for SELECT *
293        // When SELECT * is used and no schema was found, context.columns is empty.
294        // Fall back to inferring column names from the first row's HashMap keys.
295        // IMPORTANT: Must be sorted alphabetically for deterministic JSON output (Issue #129)!
296        let mut columns = context.columns;
297        if columns.is_empty() && !intermediate_results.is_empty() {
298            // Issue #1587 (E5): reuse the schema resolved once at the top of the
299            // query rather than re-locking the registry + deep-cloning here.
300            let schema_opt = query_schema.as_deref();
301
302            let first_row = &intermediate_results[0];
303            let mut col_names: Vec<_> = first_row.values.keys().collect();
304            col_names.sort(); // Sort alphabetically for deterministic ordering (Issue #129)
305
306            let table_name_for_meta = schema_opt.map(|s| format!("{}.{}", s.keyspace, s.table));
307
308            for (idx, col_name) in col_names.iter().enumerate() {
309                let col_name: &str = col_name;
310                // Look up CQL type from schema; derive flat DataType from it (Issue #674).
311                let col_info = match schema_opt
312                    .and_then(|schema| schema.columns.iter().find(|c| c.name.as_str() == col_name))
313                {
314                    Some(schema_col) => column_info_from_type_str(
315                        col_name.to_string(),
316                        &schema_col.data_type,
317                        idx,
318                        table_name_for_meta.clone(),
319                    ),
320                    None => ColumnInfo {
321                        name: col_name.to_string(),
322                        data_type: crate::types::DataType::Text,
323                        nullable: true,
324                        position: idx,
325                        table_name: table_name_for_meta.clone(),
326                        cql_type: None,
327                    },
328                };
329                columns.push(col_info);
330            }
331        }
332
333        // Observability (issue #1035): the `query.select.plan` span declared
334        // `access_path`/`rows_scanned`/`rows` but never recorded them, and
335        // `QUERY_ROWS_SCANNED` was never emitted. Do both here, sourced from the
336        // honest per-query signal (`context.access_path`, set by the SSTable-scan
337        // step) and the rows the scan examined (`context.rows_processed`). Bounded
338        // attributes only — never the query text or key values.
339        //
340        // Issue #2162: this stays a SINGLE-SHOT emission deliberately. The Flight
341        // `do_get` merge/scan path gained incremental `QUERY_ROWS_SCANNED` deltas
342        // (`cqlite-flight/src/scan_progress.rs`), but that path never runs through
343        // `SelectExecutor` — Flight drives `KWayMerger` directly. Making THIS site
344        // incremental too would additionally have to interact with
345        // `limit_pushdown`'s `scan_rows` rebaseline/decode-stop accounting
346        // (`limit_pushdown/mod.rs`), which re-runs a capped scan and resets
347        // `context.scan_rows` — a materially riskier change with no scenario in
348        // #2162's spec requiring it.
349        {
350            use crate::observability::{self as obs, catalog, AttrValue};
351
352            let access_path_label: &'static str = context
353                .access_path
354                .as_ref()
355                .map(|p| p.label())
356                .unwrap_or("unknown");
357
358            obs::add_counter(
359                catalog::QUERY_ROWS_SCANNED,
360                context.scan_rows,
361                &[(
362                    catalog::attr::ACCESS_PATH,
363                    AttrValue::StaticStr(access_path_label),
364                )],
365            );
366
367            let span = tracing::Span::current();
368            span.record(catalog::attr::ACCESS_PATH, access_path_label);
369            span.record("cqlite.query.rows_scanned", context.scan_rows);
370            span.record("cqlite.query.rows", total_rows);
371        }
372
373        // Issue #1035: carry a bounded plan family on the result so the engine's
374        // single observability chokepoint reports a real plan type for SELECTs
375        // (the modern executor previously always returned `plan_info: None`,
376        // forcing plan_type to "unknown").
377        let plan_info = Self::select_plan_info(&plan, context.access_path.as_ref());
378
379        Ok(QueryResult {
380            rows: intermediate_results,
381            rows_affected: total_rows, // Use actual number of rows returned
382            execution_time_ms: 0,      // Will be set by the engine
383            metadata: crate::query::result::QueryMetadata {
384                columns,
385                total_rows: Some(total_rows),
386                plan_info: Some(plan_info),
387                performance: Default::default(),
388                warnings: vec![],
389                // Issue #960: surface the access path the SSTable-scan step chose
390                // on the result from PER-QUERY state (not the global probe), so a
391                // concurrent SELECT cannot overwrite it between record() and here.
392                access_path: context.access_path.clone(),
393            },
394        })
395    }
396
397    /// Execute SSTable scan with predicate pushdown.
398    ///
399    /// Per-row work (build row, decode partition key, evaluate predicates) is
400    /// handled by the free helpers `build_row_from_scan` and
401    /// `evaluate_predicates`, which are shared with the streaming background
402    /// task to keep the two execution paths in lockstep.
403    ///
404    /// Issue #1582 (D6, narrow subset): the single byte/row budget check is
405    /// applied ONCE by [`SelectExecutor::execute`] on the FINAL result, after the
406    /// whole step pipeline (post LIMIT/OFFSET), never mid-collection here.
407    ///
408    /// Issue #1577 (D1): when the caller determines the plan is LIMIT-pushdown
409    /// safe it passes `scan_cap = Some(limit + offset)` — the number of ACCEPTED
410    /// (post-marker, post-predicate) rows the downstream `Limit` needs. This scan
411    /// then stops after that many accepted rows: the full-scan fallback stops
412    /// DECODING early (via [`capped_fallback_scan`](Self::capped_fallback_scan)'s
413    /// bounded stream), and the partition-targeted paths stop the per-row build
414    /// loop. `scan_cap` is `None` (unbounded, as before) whenever a
415    /// Sort/Aggregate/PerPartitionLimit/residual-Filter step or DISTINCT follows,
416    /// or there is no `LIMIT` — so a `WHERE non_pk = ? LIMIT N` still counts
417    /// ACCEPTED rows, never raw rows, and can never silently drop matches.
418    #[cfg_attr(feature = "tombstones", allow(unused_variables))]
419    pub(super) async fn execute_sstable_scan(
420        &self,
421        table: &TableId,
422        predicates: &[SSTablePredicate],
423        projection: &[String],
424        order_by: Option<&crate::query::select_ast::OrderByClause>,
425        // Issue #1587 (E5): schema resolved ONCE per query by the caller and
426        // shared by reference — no per-scan registry lock + deep clone.
427        schema_opt: Option<&TableSchema>,
428        // Issue #1577 (D1): `Some(limit + offset)` when LIMIT pushdown is safe.
429        scan_cap: Option<usize>,
430        context: &mut ExecutionContext,
431    ) -> Result<Vec<QueryRow>> {
432        // FINDING 2 (Issue #955 follow-up): a `token(...)` predicate is evaluated
433        // by hashing the row's raw partition key, so its argument columns MUST be
434        // the full partition key in declared order or the result is silently
435        // wrong. Reject (Cassandra-style) before scanning/evaluating.
436        validate_token_predicates(predicates, schema_opt)?;
437
438        // Data-safety (issue #1694): log the SHAPE of the scan — predicate count
439        // and the constrained column names — never the predicate literals/values.
440        tracing::debug!(
441            "Executing SSTableScan: table=\"{}\", predicates={} on [{}], include_cell_metadata={}",
442            table,
443            predicates.len(),
444            predicates
445                .iter()
446                .map(|p| p.column.as_str())
447                .collect::<Vec<_>>()
448                .join(", "),
449            context.projection_flags.include_cell_metadata,
450        );
451
452        let (keyspace, table_name) = parse_table_id(table);
453
454        match schema_opt {
455            Some(schema) => tracing::debug!(
456                "Found schema for {}.{} with {} columns",
457                schema.keyspace,
458                schema.table,
459                schema.columns.len()
460            ),
461            None => tracing::debug!(
462                "No schema found for {}.{}, proceeding without schema-aware parsing",
463                keyspace.as_deref().unwrap_or("unknown"),
464                table_name
465            ),
466        }
467
468        // Issue #1750: a SCHEMA-LESS `WHERE pk = <literal>` point read can be served
469        // by a key-byte-targeted seek ONLY when the equality column is a
470        // metadata-CONFIRMED sole partition key (single-component pk, no clustering
471        // keys, column absent from the authoritative non-key column names). Resolve
472        // that authoritative shape from the Statistics.db SerializationHeader (the
473        // metadata a schema-less reader HAS) and classify BY ELIMINATION — never a
474        // pk-name/text guess (#28). `None` (metadata unavailable, or the column is a
475        // non-pk column) keeps the honest full-scan path, which correctly matches
476        // regular-column equalities. Resolved only when there is no CQL schema and no
477        // WRITETIME/TTL metadata projection (the two branches below own those).
478        let schemaless_seek: Option<super::schemaless_point::SchemalessPointSeek> =
479            if schema_opt.is_none() && !context.projection_flags.include_cell_metadata {
480                let shape = self.storage.partition_key_shape(table).await;
481                classify_schemaless_point_lookup(predicates, shape.as_ref())
482            } else {
483                None
484            };
485
486        // Issue #693: When WRITETIME(col) or TTL(col) is in the SELECT, use the
487        // metadata-carrying scan so per-cell timestamps reach the QueryRow.
488        let results = if context.projection_flags.include_cell_metadata {
489            // Issue #962: route a fully-constrained `WHERE pk = ?` WRITETIME/TTL
490            // projection through a partition-targeted metadata lookup that prunes
491            // SSTables (bloom/BTI) before decoding, instead of full-scanning every
492            // SSTable for the table. Reuses the SAME `classify_partition_lookup`
493            // decision the non-metadata path uses (the shared resolved
494            // partition-lookup representation). The per-row predicate evaluation
495            // below is unchanged, so the pk equality itself is still applied as a
496            // correctness backstop and any bloom/BTI over-inclusion is filtered out.
497            let scan_results = match classify_partition_lookup(predicates, schema_opt) {
498                PartitionLookupOutcome::Targeted(pk_bytes) => {
499                    tracing::debug!(
500                        "SSTableScan(metadata): partition-key point lookup (key len={}) for \"{}\"",
501                        pk_bytes.len(),
502                        table
503                    );
504                    // Epic #951 (honest paths): the `tombstones` build's metadata
505                    // lookup is a full metadata scan + retain with NO prune,
506                    // reported via `engaged == false`; claim
507                    // `MetadataPartitionLookup` only when it really pruned, else
508                    // report the honest `TombstonesBuildNoPrune` fallback (the
509                    // rows are byte-identical either way).
510                    let (rows, engaged) = self
511                        .storage
512                        .scan_partition_with_cell_metadata(table, &pk_bytes, schema_opt)
513                        .await?;
514                    let path = honest_targeted_path(AccessPath::MetadataPartitionLookup, engaged);
515                    context.access_path = Some(path.clone());
516                    crate::query::access_path::record(path);
517                    rows
518                }
519                // Issue #962: `WHERE pk IN (...)` on the metadata path is NOT yet
520                // fanned out to N targeted metadata lookups; it still full-scans.
521                // Report that honestly (MetadataScanPath) rather than faking a
522                // targeted path — the IN-metadata fan-out is a documented follow-up.
523                PartitionLookupOutcome::MultiTargeted(_) | PartitionLookupOutcome::Fallback(_) => {
524                    let metadata_path = AccessPath::FallbackFullScan {
525                        reason: FallbackReason::MetadataScanPath,
526                    };
527                    context.access_path = Some(metadata_path.clone());
528                    crate::query::access_path::record(metadata_path);
529                    self.storage
530                        .scan_with_cell_metadata(table, None, None, None, schema_opt)
531                        .await?
532                }
533            };
534
535            tracing::debug!("Scan (with metadata) returned {} rows", scan_results.len());
536
537            // Issue #1577 (D1 + roborev metric-accounting fix): this metadata scan
538            // is ALREADY fully materialized — the storage layer decoded every row
539            // before returning it. `collect_capped_materialized` charges
540            // `context.scan_rows` (→ `QUERY_ROWS_SCANNED`) with the TRUE decoded
541            // count up front, so the metric reflects real scan work, while the
542            // LIMIT/OFFSET `scan_cap` only bounds the per-row build/predicate work.
543            // Counting ACCEPTED rows toward the cap never drops a matching row.
544            // Issue #1817: hoist the partition-key decode across the rows of a
545            // partition (the scan yields a partition's rows consecutively).
546            let mut pk_cache = super::PartitionKeyCache::default();
547            collect_capped_materialized(
548                scan_results,
549                scan_cap,
550                predicates,
551                context,
552                |(key, value, cell_meta)| {
553                    let mut row = build_row_from_scan_cached(
554                        key,
555                        value,
556                        projection,
557                        schema_opt,
558                        &mut pk_cache,
559                    )?;
560                    // Attach per-cell metadata so evaluate_writetime_ttl can read it.
561                    if !cell_meta.is_empty() {
562                        row.set_cell_metadata(cell_meta);
563                    }
564                    Some(row)
565                },
566            )?
567        } else if let Some(seek) = schemaless_seek {
568            // Issue #1750 (regression fix, re-scoped): a SCHEMA-LESS `WHERE pk =
569            // <literal>` point read whose equality column is a metadata-CONFIRMED
570            // sole partition key. With no schema the full-scan path CANNOT
571            // reconstruct the partition-key column, so the shared per-row
572            // `evaluate_predicates` backstop below would reject EVERY row on the pk
573            // equality and return 0 rows — the regression from rerouting this read
574            // off the legacy `QueryExecutor` (which looked up by key bytes, never
575            // re-evaluating the predicate). Serve it here by the SAME
576            // key-byte-targeted seek the schema-aware Targeted arm uses
577            // (`scan_partition`), which is self-verifying: it returns ONLY rows whose
578            // raw partition key equals `seek.bytes`, so a wrong/absent key yields
579            // nothing (never another partition's rows). The key is encoded through
580            // the TYPED single-component codec for the pk's AUTHORITATIVE type (from
581            // the Statistics.db SerializationHeader `keyType`), so a parsed integer
582            // literal (`Value::BigInt`) builds the width Cassandra wrote — an `int`
583            // pk's 4-byte key, not an 8-byte one (roborev 3784 FINDING 2). The
584            // decision is made from AUTHORITATIVE metadata by ELIMINATION — never a
585            // pk-name/substring/token-count text guess (#28). A `WHERE <regular_col>
586            // = <literal>` does NOT reach here (the classifier returns `None` because
587            // the column is one of the authoritative non-key names), so it keeps the
588            // honest full-scan path and correctly matches the regular-column cell.
589            //
590            // Post-seek guard (roborev 3784 FINDING 1): the classifier admits the
591            // predicate column by ELIMINATION, which also admits a
592            // nonexistent/misspelled column whose literal happens to encode to a real
593            // partition's key. `finalize_schemaless_seek_row` reconstructs the sole
594            // pk column under its authoritative type/name and RE-EVALUATES the
595            // predicate, so such a column yields `Unknown` and rejects the row
596            // (correct 0 rows) — the pk-name predicate still matches. The schema-less
597            // `SELECT *` column metadata still comes from the first row in `execute`.
598            let (rows, engaged) = self
599                .storage
600                .scan_partition(table, &seek.bytes, None)
601                .await?;
602            let path = honest_targeted_path(AccessPath::PartitionLookup, engaged);
603            context.access_path = Some(path.clone());
604            crate::query::access_path::record(path);
605            context.scan_rows += rows.len() as u64;
606            context.rows_processed += rows.len() as u64;
607            let mut out = Vec::with_capacity(rows.len());
608            for (key, value) in rows {
609                if let Some(row) = super::schemaless_point::finalize_schemaless_seek_row(
610                    key, value, projection, predicates, &seek,
611                ) {
612                    out.push(row);
613                }
614            }
615            out
616        } else {
617            // Issue #949: a fully-constrained `WHERE pk = ?` is served by a
618            // partition-targeted lookup that prunes SSTables via bloom/BTI and only
619            // parses the candidates, instead of scanning every SSTable for the
620            // table. Falls back to a full scan when the partition key isn't fully
621            // pinned or can't be encoded. The per-row predicate evaluation below is
622            // unchanged, so clustering predicates and the pk equality itself are
623            // still applied (and any over-inclusion is filtered out).
624            let scan_results = match classify_partition_lookup(predicates, schema_opt) {
625                PartitionLookupOutcome::Targeted(pk_bytes) => {
626                    tracing::debug!(
627                        "SSTableScan: partition-key point lookup (key len={}) for \"{}\"",
628                        pk_bytes.len(),
629                        table
630                    );
631                    // Issue #954: when a single-column clustering restriction is
632                    // present, push it down to a within-partition seek so a wide
633                    // partition's slice decodes O(matched rows + index), not the
634                    // whole partition. The seek reports whether the clustering
635                    // narrowing actually engaged; the per-row backstop below applies
636                    // the exact bound so output is byte-identical either way.
637                    //
638                    // Issue #960: report the HONEST access path — `ClusteringSlice`
639                    // only when the seek engaged, else `PartitionLookup`. The
640                    // clustering seek exists only on the default build; the
641                    // `tombstones` build uses the plain partition lookup.
642                    // Issue #954/#960/#1184: forward clustering-slice seek OR (for
643                    // `ORDER BY <ck>` reverse-of-stored) the BIG reverse iterator,
644                    // with the honest access path recorded inside the helper.
645                    #[cfg(not(feature = "tombstones"))]
646                    {
647                        self.targeted_partition_rows(
648                            table, &pk_bytes, predicates, order_by, schema_opt, context,
649                        )
650                        .await?
651                    }
652                    #[cfg(feature = "tombstones")]
653                    {
654                        // Epic #951 (honest paths): the `tombstones` build's
655                        // `scan_partition` is a full scan + retain with NO prune,
656                        // reported via `engaged == false`. Report the honest
657                        // fallback rather than a fake `PartitionLookup`; the rows
658                        // are byte-identical to the pruned build.
659                        let (rows, engaged) = self
660                            .storage
661                            .scan_partition(table, &pk_bytes, schema_opt)
662                            .await?;
663                        let path = honest_targeted_path(AccessPath::PartitionLookup, engaged);
664                        context.access_path = Some(path.clone());
665                        crate::query::access_path::record(path);
666                        rows
667                    }
668                }
669                PartitionLookupOutcome::MultiTargeted(pk_keys) => {
670                    tracing::debug!(
671                        "SSTableScan: multi-partition lookup ({} keys) for \"{}\"",
672                        pk_keys.len(),
673                        table
674                    );
675                    // Issue #955/#960: `WHERE pk IN (...)` over the complete key
676                    // is the union of N independent partition-targeted lookups,
677                    // each of which prunes SSTables. Epic #951 (honest paths): on
678                    // the `tombstones` build each lookup full-scans + retains with
679                    // NO prune (`engaged == false`); report `MultiPartitionLookup`
680                    // only when the lookups actually pruned, else the honest
681                    // `TombstonesBuildNoPrune` fallback. Rows are unchanged.
682                    let mut combined = Vec::new();
683                    let mut all_engaged = true;
684                    for pk_bytes in &pk_keys {
685                        let (rows, engaged) = self
686                            .storage
687                            .scan_partition(table, pk_bytes, schema_opt)
688                            .await?;
689                        all_engaged &= engaged;
690                        combined.extend(rows);
691                    }
692                    let path = honest_targeted_path(AccessPath::MultiPartitionLookup, all_engaged);
693                    context.access_path = Some(path.clone());
694                    crate::query::access_path::record(path);
695                    // Order the union to equal a full scan filtered to these keys:
696                    // partitions are stored token-ordered, so sort the combined
697                    // rows by (partition token, raw key bytes). A *stable* sort
698                    // keeps each partition's clustering order (rows for one key
699                    // arrive contiguously from one `scan_partition`) intact.
700                    sort_rows_by_token(&mut combined);
701                    combined
702                }
703                PartitionLookupOutcome::Fallback(reason) => {
704                    // Issue #960: report the honest reason a full scan was chosen.
705                    context.access_path = Some(AccessPath::FallbackFullScan { reason });
706                    crate::query::access_path::record(AccessPath::FallbackFullScan { reason });
707                    // Issue #1577 (D1): when the plan is LIMIT-pushdown safe, stop
708                    // DECODING the table once `cap` rows are accepted. This streams
709                    // (in lockstep with `scan`) and drops the stream at the cap,
710                    // returning the final rows directly — the shared per-row loop
711                    // below is bypassed for this branch.
712                    if let Some(cap) = scan_cap {
713                        return self
714                            .capped_fallback_scan(
715                                table, predicates, projection, schema_opt, cap, context,
716                            )
717                            .await;
718                    }
719                    // Issue #1582 (D6, narrow subset): unbounded full scan; the sole
720                    // budget check is applied once on the FINAL result in `execute`.
721                    self.storage
722                        .scan(table, None, None, None, schema_opt)
723                        .await?
724                }
725            };
726
727            tracing::debug!("Scan returned {} rows", scan_results.len());
728
729            // Issue #1577 (D1 + roborev metric-accounting fix): these results are
730            // ALREADY materialized — the partition-targeted paths decoded their
731            // (partition-bounded) rows, and the TRULY decode-bounded full-scan
732            // fallback already returned early above via `capped_fallback_scan`.
733            // `collect_capped_materialized` charges `context.scan_rows` with the
734            // full decoded count up front (so `QUERY_ROWS_SCANNED` reflects the
735            // real scan), while the `scan_cap` only bounds per-row build work.
736            // Counting ACCEPTED rows toward the cap (build_row_from_scan returns
737            // None for tombstoned/null rows, Issue #191) never drops a match.
738            // Issue #1817: hoist the partition-key decode across a partition's rows.
739            let mut pk_cache = super::PartitionKeyCache::default();
740            collect_capped_materialized(
741                scan_results,
742                scan_cap,
743                predicates,
744                context,
745                |(key, value)| {
746                    build_row_from_scan_cached(key, value, projection, schema_opt, &mut pk_cache)
747                },
748            )?
749        };
750
751        Ok(results)
752    }
