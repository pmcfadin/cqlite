65        println!("🚀 Starting Cassandra 5+ Performance Benchmarking Suite");

74        };
75
76        println!("🎯 PRD Performance Targets:");
77        println!(
78            "   Parse Speed: ≥{} MB/s",

81        println!("   Memory Limit: ≤{} MB", prd_targets.memory_limit_mb);
82        println!(
83            "   Throughput: ≥{} ops/sec",
84            prd_targets.throughput_ops_per_sec
85        );
86        println!("   Max File Size: {} GB", prd_targets.max_file_size_gb);
87
88        // Run benchmarks in sequence
89        let mut all_results = Vec::new();
90
91        // 1. Memory Usage Benchmarks
92        println!("\n📊 1. Memory Usage Benchmarks");
93        let memory_results = self
94            .run_memory_benchmarks(test_data_dir, &prd_targets)

96        all_results.extend(memory_results);
97
98        // 2. Compression Benchmarks
99        println!("\n🗜️ 2. Compression Performance Benchmarks");
100        let compression_results = self
101            .run_compression_benchmarks(test_data_dir, &prd_targets)
102            .await?;
103        all_results.extend(compression_results);
104
105        // 3. Zero-Copy Benchmarks
106        println!("\n⚡ 3. Zero-Copy Deserialization Benchmarks");
107        let zerocopy_results = self
108            .run_zerocopy_benchmarks(test_data_dir, &prd_targets)
109            .await?;
110        all_results.extend(zerocopy_results);
111
112        // 4. Throughput Benchmarks
113        println!("\n🏃 4. Throughput Benchmarks");
114        let throughput_results = self
115            .run_throughput_benchmarks(test_data_dir, &prd_targets)
116            .await?;
117        all_results.extend(throughput_results);
118
119        // Store results
120        for result in &all_results {
121            self.results
122                .insert(result.benchmark_name.clone(), result.clone());
123        }
124
125        // Generate summary report
126        self.print_performance_summary(&all_results, &prd_targets);

190    /// Print comprehensive performance summary
191    fn print_performance_summary(&self, results: &[BenchmarkResult], targets: &PRDTargets) {
192        println!("\n📊 CASSANDRA 5+ PERFORMANCE SUMMARY");
193        println!("═══════════════════════════════════════════");
194
195        let passed = results.iter().filter(|r| r.meets_prd_target).count();
196        let total = results.len();
197        let pass_rate = if total > 0 {
198            (passed as f64 / total as f64) * 100.0
199        } else {
200            0.0
201        };
202
203        println!(
204            "Overall Results: {}/{} benchmarks passed ({:.1}%)",
205            passed, total, pass_rate
206        );
207
208        // Category breakdown
209        let categories = ["Memory", "Compression", "ZeroCopy", "Throughput"];
210        for category in &categories {
211            let category_results: Vec<_> = results
212                .iter()
213                .filter(|r| r.benchmark_name.contains(category))
214                .collect();
215
216            if !category_results.is_empty() {
217                let category_passed = category_results
218                    .iter()
219                    .filter(|r| r.meets_prd_target)
220                    .count();
221                let category_total = category_results.len();
222                let category_pass_rate = (category_passed as f64 / category_total as f64) * 100.0;
223
224                println!(
225                    "\n{} Benchmarks: {}/{} passed ({:.1}%)",
226                    category, category_passed, category_total, category_pass_rate
227                );
228
229                for result in &category_results {
230                    let status = if result.meets_prd_target {
231                        "✅"
232                    } else {
233                        "❌"
234                    };
235                    println!(
236                        "  {} {}: {:.2} MB/s, {:.1} MB memory",
237                        status,
238                        result.benchmark_name,
239                        result.throughput_mb_per_sec,
240                        result.memory_usage_mb
241                    );
242                }
243            }
244        }
245
246        // Best performers
247        if !results.is_empty() {
248            let fastest = results
249                .iter()
250                .max_by(|a, b| {
251                    a.throughput_mb_per_sec
252                        .partial_cmp(&b.throughput_mb_per_sec)
253                        .unwrap()
254                })
255                .unwrap();
256            let most_efficient = results
257                .iter()
258                .min_by(|a, b| {
259                    a.memory_usage_mb
260                        .partial_cmp(&b.memory_usage_mb)
261                        .unwrap_or(std::cmp::Ordering::Equal)
262                })
263                .unwrap();
264
265            println!("\n🏆 TOP PERFORMERS");
266            println!(
267                "   🚀 Fastest: {} ({:.2} MB/s)",
268                fastest.benchmark_name, fastest.throughput_mb_per_sec
269            );
270            println!(
271                "   💾 Most memory efficient: {} ({:.1} MB)",
272                most_efficient.benchmark_name, most_efficient.memory_usage_mb
273            );
274        }
275
276        // PRD compliance summary
277        println!("\n🎯 PRD COMPLIANCE ANALYSIS");
278        self.analyze_prd_compliance(results, targets);
279
280        // Optimization recommendations
281        println!("\n💡 OPTIMIZATION RECOMMENDATIONS");
282        self.generate_optimization_recommendations(results, targets);
283    }

285    /// Analyze PRD compliance in detail
286    fn analyze_prd_compliance(&self, results: &[BenchmarkResult], targets: &PRDTargets) {
287        // Parse speed compliance
288        let parse_speed_results: Vec<_> = results
289            .iter()
290            .filter(|r| r.throughput_mb_per_sec > 0.0)
291            .collect();
292
293        if !parse_speed_results.is_empty() {
294            let avg_parse_speed: f64 = parse_speed_results
295                .iter()
296                .map(|r| r.throughput_mb_per_sec)
297                .sum::<f64>()
298                / parse_speed_results.len() as f64;
299
300            let speed_compliance = avg_parse_speed >= targets.parse_speed_mb_per_sec;
301            let speed_icon = if speed_compliance { "✅" } else { "❌" };
302
303            println!(
304                "   {} Parse Speed: {:.2} MB/s (target: ≥{} MB/s)",
305                speed_icon, avg_parse_speed, targets.parse_speed_mb_per_sec
306            );
307        }
308
309        // Memory usage compliance
310        let memory_results: Vec<_> = results.iter().filter(|r| r.memory_usage_mb > 0.0).collect();
311
312        if !memory_results.is_empty() {
313            let max_memory = memory_results
314                .iter()
315                .map(|r| r.memory_usage_mb)
316                .fold(0.0, f64::max);
317
318            let memory_compliance = max_memory <= targets.memory_limit_mb;
319            let memory_icon = if memory_compliance { "✅" } else { "❌" };
320
321            println!(
322                "   {} Memory Usage: {:.1} MB peak (target: ≤{} MB)",
323                memory_icon, max_memory, targets.memory_limit_mb
324            );
325        }
326
327        // Throughput compliance
328        let throughput_results: Vec<_> = results
329            .iter()
330            .filter(|r| r.operations_per_second > 0.0)
331            .collect();
332
333        if !throughput_results.is_empty() {
334            let max_throughput = throughput_results
335                .iter()
336                .map(|r| r.operations_per_second)
337                .fold(0.0, f64::max);
338
339            let throughput_compliance = max_throughput >= targets.throughput_ops_per_sec;
340            let throughput_icon = if throughput_compliance { "✅" } else { "❌" };
341
342            println!(
343                "   {} Throughput: {:.0} ops/sec (target: ≥{} ops/sec)",
344                throughput_icon, max_throughput, targets.throughput_ops_per_sec
345            );
346        }
347    }

349    /// Generate optimization recommendations based on benchmark results
350    fn generate_optimization_recommendations(
351        &self,
352        results: &[BenchmarkResult],
353        targets: &PRDTargets,
354    ) {
355        let mut recommendations = Vec::new();
356
357        // Analyze memory usage patterns
358        let high_memory_results: Vec<_> = results
359            .iter()
360            .filter(|r| r.memory_usage_mb > targets.memory_limit_mb * 0.8) // >80% of limit
361            .collect();
362
363        if !high_memory_results.is_empty() {
364            recommendations.push(
365                "🔧 Consider implementing streaming readers for large files to reduce memory usage",
366            );
367            recommendations.push("🔧 Optimize buffer pool sizes to stay within memory limits");
368        }
369
370        // Analyze throughput issues
371        let slow_results: Vec<_> = results
372            .iter()
373            .filter(|r| r.throughput_mb_per_sec < targets.parse_speed_mb_per_sec * 0.5) // <50% of target
374            .collect();
375
376        if !slow_results.is_empty() {
377            recommendations.push("⚡ Investigate I/O bottlenecks in slow parsing operations");
378            recommendations.push("⚡ Consider parallel processing for large SSTable files");
379        }
380
381        // Compression-specific recommendations
382        let compression_results: Vec<_> = results
383            .iter()
384            .filter(|r| r.benchmark_name.contains("Compression"))
385            .collect();
386
387        if !compression_results.is_empty() {
388            let best_compression = compression_results.iter().max_by(|a, b| {
389                a.throughput_mb_per_sec
390                    .partial_cmp(&b.throughput_mb_per_sec)
391                    .unwrap()
392            });
393
394            if let Some(best) = best_compression {
395                let _algorithm_name = best.benchmark_name.replace("Compression", "");
396                recommendations
397                    .push("🗜️ Recommend optimal compression algorithm based on benchmark results");
398            }
399        }
400
401        // General recommendations
402        if results.iter().any(|r| !r.meets_prd_target) {
403            recommendations.push(
404                "📊 Review benchmark methodology and consider hardware-specific optimizations",
405            );
406            recommendations
407                .push("🔍 Profile critical paths for additional optimization opportunities");
408        }
409
410        if recommendations.is_empty() {
411            println!("   🎉 All benchmarks performing well! No immediate optimizations needed.");
412        } else {
413            for (i, rec) in recommendations.iter().enumerate() {
414                println!("   {}. {}", i + 1, rec);
415            }
416        }
417    }

449    /// Measure memory usage during operation
450    pub struct MemoryMonitor {
451        baseline_mb: f64,
452        peak_mb: f64,
453        samples: Vec<f64>,
454    }
455
456    impl Default for MemoryMonitor {
457        fn default() -> Self {
458            Self::new()
459        }
460    }
461
462    impl MemoryMonitor {
463        pub fn new() -> Self {
464            let baseline = get_process_memory_mb();
465            Self {
466                baseline_mb: baseline,
467                peak_mb: baseline,
468                samples: vec![baseline],
469            }
470        }
471
472        pub fn sample(&mut self) {
473            let current = get_process_memory_mb();
474            self.samples.push(current);
475            self.peak_mb = self.peak_mb.max(current);
476        }
477
478        pub fn peak_usage_mb(&self) -> f64 {
479            self.peak_mb - self.baseline_mb
480        }
481
482        pub fn average_usage_mb(&self) -> f64 {
483            let sum: f64 = self.samples.iter().map(|&s| s - self.baseline_mb).sum();
484            sum / self.samples.len() as f64
485        }
486    }

488    /// Get current process memory usage in MB
489    pub fn get_process_memory_mb() -> f64 {
490        #[cfg(target_os = "macos")]
491        {
492            get_memory_usage_macos()
493        }
494        #[cfg(target_os = "linux")]
495        {
496            get_memory_usage_linux()
497        }
498        #[cfg(not(any(target_os = "macos", target_os = "linux")))]
499        {
500            0.0 // Fallback for unsupported platforms
501        }
502    }
503
504    #[cfg(target_os = "macos")]
505    fn get_memory_usage_macos() -> f64 {
506        use std::process::{self, Command, Stdio};
507
508        let pid = process::id();
509        let output = Command::new("ps")
510            .args(["-o", "rss=", "-p", &pid.to_string()])
511            .stdout(Stdio::piped())
512            .output();
513
514        if let Ok(output) = output {
515            let rss_kb = String::from_utf8_lossy(&output.stdout)
516                .trim()
517                .parse::<f64>()
518                .unwrap_or(0.0);
519            rss_kb / 1024.0 // Convert KB to MB
520        } else {
521            0.0
522        }
523    }
524
525    #[cfg(target_os = "linux")]
526    fn get_memory_usage_linux() -> f64 {
527        use std::fs;
528
529        let status = fs::read_to_string("/proc/self/status").unwrap_or_default();
530        for line in status.lines() {
531            if line.starts_with("VmRSS:") {
532                let parts: Vec<&str> = line.split_whitespace().collect();
533                if parts.len() >= 2 {
534                    if let Ok(kb) = parts[1].parse::<f64>() {
535                        return kb / 1024.0; // Convert KB to MB
536                    }
537                }
538            }
539        }
540        0.0
541    }

543    /// Generate test data of specified size with realistic SSTable patterns
544    pub fn generate_test_data(size_mb: f64) -> Vec<u8> {
545        let size_bytes = (size_mb * 1024.0 * 1024.0) as usize;
546        let mut data = Vec::with_capacity(size_bytes);
547
548        // Pattern that resembles SSTable data with some compressibility
549        let patterns = [
550            b"SSTable_row_key_",
551            b"timestamp_value_",
552            b"column_data_cont",
553            b"metadata_info___",
554        ];
555
556        let mut pattern_idx = 0;
557        while data.len() < size_bytes {
558            let pattern = patterns[pattern_idx % patterns.len()];
559            data.extend_from_slice(pattern);
560
561            // Add some varying data
562            let counter = (data.len() / pattern.len()) as u32;
563            data.extend_from_slice(&counter.to_be_bytes());
564
565            pattern_idx += 1;
566        }
567
568        data.truncate(size_bytes);
569        data
570    }
