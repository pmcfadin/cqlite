1//! Data.db writer - writes partition and row data
2//!
3//! Generates the Data.db component with V5CompressedLegacy (NB) format.
4//! Maintains partition ordering by Murmur3 token and clustering ordering.
5//! Tracks file positions for Index.db generation.
6//!
7//! Critical requirements:
8//! - Partition ordering: By Murmur3 token, then key bytes (enforced by caller)
9//! - Clustering ordering: By clustering comparator within partition (enforced by caller)
10//! - Row size measurement: After VInt length bytes (Issue #237)
11//! - Delta encoding: Uses Statistics.db baseline for timestamps/TTL/deletion times
12//!
13//! # V5CompressedLegacy Row Format
14//!
15//! Each row is encoded as:
16//! ```text
17//! [row_flags: u8]
18//! [extended_flags: u8 if ROW_HAS_EXTENDED_FLAGS set]
19//! [clustering_prefix: variable if present]
20//! [row_size: VInt]                       ← Measured from AFTER this VInt
21//! [prev_size: VInt]
22//! [timestamp: VInt if ROW_HAS_TIMESTAMP]   ← Delta from min_timestamp
23//! [ttl: VInt if ROW_HAS_TTL]              ← Delta from min_ttl
24//! [deletion: 2 VInts if ROW_HAS_DELETION] ← local_deletion_time delta + deletion timestamp
25//! [column_bitmap: VUInt bitmask of missing columns if NOT ROW_HAS_ALL_COLUMNS]
26//! [cell_data...]
27//! ```
28//!
29//! ## Row Flags
30//! - `0x04` (HAS_TIMESTAMP): Timestamp delta present
31//! - `0x08` (HAS_TTL): TTL delta present
32//! - `0x10` (HAS_DELETION): Deletion time present (two VInts)
33//! - `0x20` (HAS_ALL_COLUMNS): All columns present (no bitmap)
34//! - `0x40` (HAS_COMPLEX_DELETION): Row contains complex column with deletion
35//! - `0x80` (HAS_EXTENDED_FLAGS): Extended flags byte follows
36//!
37//! ## Cell Format
38//! ```text
39//! [flags: u8]
40//! [timestamp: VInt if NOT USE_ROW_TIMESTAMP]  ← Delta from min_timestamp
41//! [local_deletion_time: VUInt if deleted/expiring and NOT USE_ROW_TTL]
42//! [ttl: VUInt if expiring and NOT USE_ROW_TTL]
43//! [value_length: VInt]
44//! [value_bytes]
45//! ```
46//!
47//! ## Cell Flags
48//! - `0x01` (IS_DELETED): Cell is a tombstone
49//! - `0x02` (IS_EXPIRING): TTL fields follow
50//! - `0x04` (HAS_EMPTY_VALUE): Zero-length value
51//! - `0x08` (USE_ROW_TIMESTAMP): Use row-level timestamp (no timestamp delta)
52//! - `0x10` (USE_ROW_TTL): Use row-level TTL (no TTL delta)
53//!
54//! References:
55//! - Cassandra 5.0: `org.apache.cassandra.db.rows.UnfilteredSerializer`
56//! - Parser: `cqlite-core/src/storage/sstable/reader/parsing/v5_compressed_legacy.rs`
57//! - Format docs: `docs/sstables-definitive-guide/chapters/05-data-db-format.md`

114/// Capacity of the streaming Data.db `BufWriter` (Issue #492).
115///
116/// Large enough that each flushed partition coalesces into a handful of big
117/// `write()` syscalls instead of many small default-8 KB ones, preserving the
118/// throughput of the previous single whole-file write while keeping resident
119/// memory bounded (this buffer plus one partition's scratch).
120const DATA_SINK_BUFFER_BYTES: usize = 1024 * 1024;
121
122/// Data.db component writer
123///
124/// Writes partitions and rows in V5CompressedLegacy format with delta encoding.
125/// Caller must provide partitions in token order and rows in clustering order.
126///
127/// # Memory model (Issue #492)
128///
129/// The writer supports two modes that produce **byte-identical** Data.db output:
130///
131/// * **In-memory mode** (`DataWriter::new`): every partition is appended to the
132///   `buffer` scratch and never flushed, so `finish()` returns the full Data.db
133///   bytes. Used by unit tests that inspect the produced bytes directly.
134///
135/// * **Streaming mode** (`DataWriter::with_sink`): each partition is built in the
136///   `buffer` scratch, written to a `BufWriter<File>` over the Data.db path, and
137///   the scratch is cleared. Peak heap is therefore `O(largest partition)` rather
138///   than `O(file)`, keeping a multi-GB compaction within the 128 MB target.
139///
140/// In both modes the file offset of a partition is `position + buffer.len()`
141/// measured before any bytes are written. In streaming mode `buffer` is empty at
142/// that point (just cleared) so the offset is `position`; in memory mode
143/// `position` is always 0 and `buffer` holds all prior partitions, so the offset
144/// equals the legacy `buffer.len()`. The within-partition size math uses relative
145/// deltas into `buffer`, which are identical regardless of mode.
146#[derive(Debug)]
147pub struct DataWriter {
148    /// Per-partition scratch buffer for Data.db content.
149    ///
150    /// In streaming mode this is cleared at the start of every `write_partition`
151    /// and flushed to `sink` at the end, so only one partition is resident.
152    /// In memory mode it accumulates the entire Data.db output.
153    buffer: Vec<u8>,
154    /// Streaming sink over the Data.db path (streaming mode only).
155    ///
156    /// Lazily opened on the first `write_partition` so that the keyspace/table
157    /// directory exists before the first byte is written. `None` in in-memory
158    /// mode.
159    sink: Option<std::io::BufWriter<std::fs::File>>,
160    /// Data.db output path (streaming mode only); used for lazy sink open.
161    data_path: Option<PathBuf>,
162    /// Bytes already flushed to `sink`. Always 0 in in-memory mode.
163    position: u64,
164    /// Statistics metadata for delta encoding
165    stats: StatisticsMetadata,
166}

206    /// Lazily open the streaming sink (and create the parent directory).
207    ///
208    /// No-op in in-memory mode or once the sink is already open.
209    fn ensure_sink(&mut self) -> Result<()> {
210        if self.sink.is_some() {
211            return Ok(());
212        }
213        if let Some(path) = self.data_path.clone() {
214            if let Some(parent) = path.parent() {
215                std::fs::create_dir_all(parent)?;
216            }
217            let file = std::fs::File::create(&path)?;
218            // Use a large BufWriter so a partition's bytes coalesce into a few
219            // big write() syscalls rather than many 8 KB-default ones, matching
220            // the throughput of the old single whole-file write.
221            self.sink = Some(std::io::BufWriter::with_capacity(
222                DATA_SINK_BUFFER_BYTES,
223                file,
224            ));
225        }
226        Ok(())
227    }

229    /// In streaming mode, flush the current scratch buffer to the sink, advance
230    /// `position`, and clear the scratch so only one partition is ever resident.
231    /// No-op in in-memory mode (the scratch keeps accumulating).
232    fn flush_partition(&mut self) -> Result<()> {
233        if self.data_path.is_none() {
234            // In-memory mode: keep accumulating in `buffer`.
235            return Ok(());
236        }
237        self.ensure_sink()?;
238        if let Some(sink) = self.sink.as_mut() {
239            sink.write_all(&self.buffer)?;
240        }
241        self.position += self.buffer.len() as u64;
242        self.buffer.clear();
243        Ok(())
244    }

255    /// Write a complete partition (partition key + all rows)
256    ///
257    /// # Arguments
258    /// * `key` - Decorated partition key (token + raw bytes)
259    /// * `mutations` - All mutations for this partition (must be in clustering order)
260    /// * `schema` - Table schema for column metadata
261    /// * `partition_tombstone` - Optional partition-level tombstone
262    /// * `range_tombstones` - Range tombstones for this partition (must be in clustering order)
263    ///
264    /// # Returns
265    /// File offset where this partition starts (for Index.db)
266    pub fn write_partition(
267        &mut self,
268        key: &DecoratedKey,
269        mutations: &[Mutation],
270        schema: &TableSchema,
271        partition_tombstone: Option<&PartitionTombstone>,
272        range_tombstones: &[RangeTombstone],
273    ) -> Result<u64> {
274        // File offset of this partition = bytes already flushed (`position`) plus
275        // whatever is currently buffered. In streaming mode `buffer` is empty at
276        // the start of each partition (flushed + cleared by the previous call),
277        // so this is `position`; in in-memory mode `position` is 0 and `buffer`
278        // holds all prior partitions, matching the legacy `buffer.len()`.
279        let partition_offset = self.position + self.buffer.len() as u64;
280
281        // Write partition header (with optional tombstone)
282        let header_start = self.buffer.len();
283        self.write_partition_header(key, partition_tombstone)?;
284        let mut prev_unfiltered_size = (self.buffer.len() - header_start) as u64;
285
286        // SSTables must be internally reconciled: Cassandra's read path and
287        // compaction only reconcile rows against deletions from OTHER sources
288        // (memtables / other sstables) — a row shadowed by a partition or
289        // range tombstone in the SAME sstable is served live. Cassandra's own
290        // flush drops shadowed data, and so must we (Issue #716/#717).
291        // `partition_floor` is the shadow timestamp from the partition
292        // tombstone; per-row floors additionally account for covering range
293        // tombstones.
294        let partition_floor = partition_tombstone.map(|pt| pt.deletion_time);
295
296        // Cassandra's SerializationHeader.hasStatic() returns true whenever the schema
297        // declares any static column — and both the writer and reader unconditionally
298        // emit/consume a static-row prelude in that case.  We must do the same.
299        let schema_has_static = schema.columns.iter().any(|c| c.is_static);
300
301        if schema_has_static {
302            // Collect static-column operations from ALL mutations in this partition,
303            // regardless of whether the mutation also carries a clustering key.
304            // Last-write-wins by timestamp_micros when the same column appears twice.
305            // Static cells shadowed by the partition tombstone are dropped.
306            let merged = collect_static_operations(mutations, schema, partition_floor);
307
308            // Mutations shadowed by the partition tombstone cannot contribute
309            // the static row's liveness timestamp or TTL either.
310            let unshadowed_static = |m: &&Mutation| {
311                partition_floor.is_none_or(|floor| m.timestamp_micros > floor)
312                    && has_static_operation(m, schema)
313            };
314
315            if merged.is_empty() {
316                // Schema declares statics but this partition writes none.
317                // Cassandra still expects the prelude; emit the minimal empty form.
318                prev_unfiltered_size =
319                    self.write_empty_static_row(prev_unfiltered_size, schema)? as u64;
320            } else {
321                // Build a synthetic Mutation carrying the merged static ops.
322                // Use the latest timestamp seen across contributing mutations.
323                // `!merged.is_empty()` implies at least one mutation contributed
324                // an unshadowed static op, so the inner `.max()` is guaranteed `Some`.
325                let latest_ts = mutations
326                    .iter()
327                    .filter(unshadowed_static)
328                    .map(|m| m.timestamp_micros)
329                    .max()
330                    .unwrap_or(mutations.first().map(|m| m.timestamp_micros).unwrap_or(0));
331
332                // Pick the TTL from the mutation with the latest timestamp that
333                // contributed a static op (mirrors Cassandra's last-write-wins).
334                let ttl = mutations
335                    .iter()
336                    .filter(unshadowed_static)
337                    .max_by_key(|m| m.timestamp_micros)
338                    .and_then(|m| m.ttl_seconds);
339
340                let synthetic = Mutation {
341                    table: mutations
342                        .first()
343                        .map(|m| m.table.clone())
344                        .unwrap_or_else(|| TableId {
345                            keyspace: schema.keyspace.clone(),
346                            table: schema.table.clone(),
347                        }),
348                    partition_key: mutations
349                        .first()
350                        .map(|m| m.partition_key.clone())
351                        .unwrap_or_else(|| PartitionKey {
352                            columns: Vec::new(),
353                        }),
354                    clustering_key: None,
355                    operations: merged,
356                    timestamp_micros: latest_ts,
357                    ttl_seconds: ttl,
358                    partition_tombstone: None,
359                    range_tombstones: Vec::new(),
360                };
361
362                prev_unfiltered_size =
363                    self.write_static_row_with_prev_size(&synthetic, schema, prev_unfiltered_size)?
364                        as u64;
365            }
366        }
367
368        // Merge all mutations sharing a clustering key into a single row each
369        // (Issue #716/#717: writing one row per mutation produced duplicate
370        // rows with equal clustering — e.g. an INSERT row plus a phantom
371        // tombstone-carrier row — which is invalid in the OA format). Rows
372        // shadowed by the partition tombstone or a covering range tombstone
373        // are dropped during the merge.
374        let rows = self.merge_clustering_rows(
375            mutations,
376            schema,
377            schema_has_static,
378            partition_floor,
379            range_tombstones,
380        );
381
382        // Interleave merged rows with range tombstone bound markers in
383        // clustering order. Cassandra requires every unfiltered (row or
384        // marker) of a partition to appear in clustering order; with equal
385        // clustering values, inclusive-start/exclusive-end bounds sort before
386        // the row and inclusive-end/exclusive-start bounds sort after it
387        // (ClusteringPrefix.Kind.comparedToClustering).
388        enum PartitionItem<'a> {
389            Row(RowWrite<'a>),
390            Marker {
391                bound: &'a ClusteringBound,
392                is_open: bool,
393                deletion_time: i64,
394                local_deletion_time: i32,
395            },
396        }
397
398        let mut items: Vec<PartitionItem> = rows.into_iter().map(PartitionItem::Row).collect();
399        for rt in range_tombstones {
400            items.push(PartitionItem::Marker {
401                bound: &rt.start,
402                is_open: true,
403                deletion_time: rt.deletion_time,
404                local_deletion_time: rt.local_deletion_time,
405            });
406            items.push(PartitionItem::Marker {
407                bound: &rt.end,
408                is_open: false,
409                deletion_time: rt.deletion_time,
410                local_deletion_time: rt.local_deletion_time,
411            });
412        }
413
414        // Sort key: (partition position class, clustering values, bound weight).
415        // class: -1 = before all rows (Bottom), 0 = positioned by clustering
416        // values, 1 = after all rows (Top).
417        fn sort_class<'a, 'b>(item: &'b PartitionItem<'a>) -> (i8, Option<&'b ClusteringKey>, i8) {
418            match item {
419                PartitionItem::Row(row) => (0, row.clustering_key, 0),
420                PartitionItem::Marker { bound, is_open, .. } => match bound {
421                    ClusteringBound::Inclusive(ck) => (0, Some(ck), if *is_open { -1 } else { 1 }),
422                    ClusteringBound::Exclusive(ck) => (0, Some(ck), if *is_open { 1 } else { -1 }),
423                    ClusteringBound::Bottom => (-1, None, 0),
424                    ClusteringBound::Top => (1, None, 0),
425                },
426            }
427        }
428        items.sort_by(|a, b| {
429            let (class_a, ck_a, weight_a) = sort_class(a);
430            let (class_b, ck_b, weight_b) = sort_class(b);
431            class_a
432                .cmp(&class_b)
433                .then_with(|| match (ck_a, ck_b) {
434                    (Some(x), Some(y)) => x.compare(y, schema).unwrap_or_else(|_| x.cmp(y)),
435                    _ => std::cmp::Ordering::Equal,
436                })
437                .then(weight_a.cmp(&weight_b))
438        });
439
440        for item in items {
441            prev_unfiltered_size = match item {
442                PartitionItem::Row(row) => {
443                    self.write_merged_row_with_prev_size(&row, schema, prev_unfiltered_size)? as u64
444                }
445                PartitionItem::Marker {
446                    bound,
447                    is_open,
448                    deletion_time,
449                    local_deletion_time,
450                } => self.write_range_bound(
451                    bound,
452                    is_open,
453                    deletion_time,
454                    local_deletion_time,
455                    schema,
456                    prev_unfiltered_size,
457                )? as u64,
458            };
459        }
460
461        // Write end-of-partition marker
462        self.buffer.push(END_OF_PARTITION);
463
464        // Streaming mode: flush this partition to disk and clear the scratch so
465        // only one partition is ever resident in memory. No-op in memory mode.
466        self.flush_partition()?;
467
468        Ok(partition_offset)
469    }

471    /// Write an empty static-row prelude.
472    ///
473    /// Required by Cassandra whenever the schema has any static column, even
474    /// when this particular partition writes no static cells.
475    ///
476    /// Binary form:
477    /// ```text
478    /// [0x80]              ← row_flags = ROW_HAS_EXTENDED_FLAGS only
479    /// [0x01]              ← extended_flags = EXTENDED_IS_STATIC
480    /// [row_size: VUInt]   ← size of (prev_size VInt + bitmap)
481    /// [prev_size: VUInt]
482    /// [bitmap: VUInt]     ← all-missing bitmap: (1 << N) - 1 for N static cols
483    ///                       (encoded via write_column_subset with empty present set)
484    /// ```
485    fn write_empty_static_row(&mut self, prev_size: u64, schema: &TableSchema) -> Result<usize> {
486        let start_len = self.buffer.len();
487
488        // flags = only HAS_EXTENDED_FLAGS; no timestamp, no TTL, no deletion,
489        // no HAS_ALL_COLUMNS, no HAS_COMPLEX_DELETION.
490        let flags: u8 = ROW_HAS_EXTENDED_FLAGS;
491        self.buffer.push(flags);
492        self.buffer.push(EXTENDED_IS_STATIC);
493
494        // Build the row body: just prev_size VInt + column bitmap (all missing).
495        let mut body = Vec::new();
496
497        // Column bitmap: "all columns missing" for every static column.
498        // write_column_subset with an empty present_set.
499        let static_columns = self.static_columns(schema);
500        let empty_present: std::collections::HashSet<&str> = std::collections::HashSet::new();
501        self.write_column_subset(&mut body, &static_columns, &empty_present)?;
502
503        let prev_size_vint_len = unsigned_len(prev_size);
504        let row_body_size = prev_size_vint_len as u64 + body.len() as u64;
505
506        let mut row_size_buf = Vec::new();
507        encode_unsigned(row_body_size, &mut row_size_buf);
508        self.buffer.extend_from_slice(&row_size_buf);
509
510        encode_unsigned(prev_size, &mut self.buffer);
511        self.buffer.extend_from_slice(&body);
512
513        Ok(self.buffer.len() - start_len)
514    }

516    /// Finish writing and return the Data.db bytes (in-memory mode).
517    ///
518    /// Only valid for writers created via [`DataWriter::new`]. In streaming mode
519    /// the bytes live on disk; use [`DataWriter::finish_streaming`] instead.
520    pub fn finish(self) -> Result<Vec<u8>> {
521        // Hard guard (not debug_assert!, which compiles out in release): on a
522        // streaming writer the bytes live on disk and `buffer` is empty after each
523        // partition flush, so returning it would silently yield a 0-byte Data.db.
524        if self.data_path.is_some() {
525            return Err(Error::InvalidInput(
526                "DataWriter::finish() called on a streaming writer; use finish_streaming()"
527                    .to_string(),
528            ));
529        }
530        Ok(self.buffer)
531    }

533    /// Finish a streaming writer: flush the sink to disk and return the total
534    /// number of Data.db bytes written (i.e. `data_size`).
535    ///
536    /// Any residual scratch (there is none in normal operation, since
537    /// `write_partition` flushes per partition) is flushed first. Returns an
538    /// error if the writer was created in in-memory mode.
539    pub fn finish_streaming(mut self) -> Result<u64> {
540        if self.data_path.is_none() {
541            return Err(Error::InvalidInput(
542                "finish_streaming() called on an in-memory DataWriter".to_string(),
543            ));
544        }
545        // Flush any residual scratch (normally empty), then flush the sink so all
546        // bytes reach the OS file (the subsequent Digest CRC re-read of the same
547        // file sees them via the page cache). This matches the durability of the
548        // previous `tokio::fs::write`, which did not fsync either.
549        self.flush_partition()?;
550        if let Some(mut sink) = self.sink.take() {
551            sink.flush()?;
552        }
553        Ok(self.position)
554    }

556    /// Write partition header
557    ///
558    /// Format (V5CompressedLegacy / Cassandra BigFormat):
559    /// ```text
560    /// [key_length: u16 BE]           ← Partition key length (2-byte unsigned short)
561    /// [key_bytes]                    ← Raw partition key bytes
562    /// [local_deletion_time: i32 BE]  ← i32::MAX for LIVE (DeletionTime.LIVE)
563    /// [deletion_timestamp: i64 BE]   ← i64::MIN for LIVE (DeletionTime.LIVE)
564    /// ```
565    ///
566    /// Note: Cassandra uses `ByteBufferUtil.writeWithShortLength()` for the key,
567    /// which is a 2-byte BE unsigned short. There is NO separate flags byte.
568    /// DeletionTime.LIVE uses sentinel values (Integer.MAX_VALUE, Long.MIN_VALUE).
569    fn write_partition_header(
570        &mut self,
571        key: &DecoratedKey,
572        tombstone: Option<&PartitionTombstone>,
573    ) -> Result<()> {
574        // Partition key length (u16 BE, matching Cassandra's writeWithShortLength)
575        if key.key.len() > 65535 {
576            return Err(Error::InvalidInput(format!(
577                "Partition key too large: {} bytes (max 65535)",
578                key.key.len()
579            )));
580        }
581        self.buffer
582            .write_all(&(key.key.len() as u16).to_be_bytes())?;
583
584        // Partition key bytes
585        self.buffer.extend_from_slice(&key.key);
586
587        // Partition deletion info
588        if let Some(ts) = tombstone {
589            // Local deletion time (i32 BE, in seconds)
590            self.buffer
591                .write_all(&ts.local_deletion_time.to_be_bytes())?;
592            // Deletion timestamp (i64 BE, in microseconds)
593            self.buffer.write_all(&ts.deletion_time.to_be_bytes())?;
594        } else {
595            // DeletionTime.LIVE: Cassandra uses (Integer.MAX_VALUE, Long.MIN_VALUE)
596            self.buffer.write_all(&i32::MAX.to_be_bytes())?;
597            self.buffer.write_all(&i64::MIN.to_be_bytes())?;
598        }
599
600        Ok(())
601    }

627    /// Group same-clustering mutations of a partition and merge each group
628    /// into a single [`RowWrite`].
629    ///
630    /// Mutations must already be sorted by clustering key (the caller —
631    /// `SSTableWriter::write_partition` — sorts them); grouping is by
632    /// adjacency. Pure-static mutations are excluded (their cells live in the
633    /// static-row prelude), and groups that merge to nothing (e.g. mutations
634    /// that exist only to carry partition/range tombstones) produce no row.
635    fn merge_clustering_rows<'a>(
636        &self,
637        mutations: &'a [Mutation],
638        schema: &TableSchema,
639        skip_static_ops: bool,
640        partition_floor: Option<i64>,
641        range_tombstones: &[RangeTombstone],
642    ) -> Vec<RowWrite<'a>> {
643        let row_mutations: Vec<&'a Mutation> = mutations
644            .iter()
645            .filter(|m| !is_static_row_mutation(m, schema))
646            .collect();
647
648        let mut rows = Vec::new();
649        let mut start = 0;
650        while start < row_mutations.len() {
651            let mut end = start + 1;
652            while end < row_mutations.len()
653                && row_mutations[end].clustering_key == row_mutations[start].clustering_key
654            {
655                end += 1;
656            }
657
658            // Shadow floor for this row: partition tombstone plus any range
659            // tombstone covering the group's clustering key.
660            let clustering_key = row_mutations[start].clustering_key.as_ref();
661            let mut shadow_floor = partition_floor;
662            for rt in range_tombstones {
663                if range_tombstone_covers(rt, clustering_key, schema) {
664                    shadow_floor =
665                        Some(shadow_floor.map_or(rt.deletion_time, |f| f.max(rt.deletion_time)));
666                }
667            }
668
669            if let Some(row) = Self::merge_row_group(
670                &row_mutations[start..end],
671                schema,
672                skip_static_ops,
673                shadow_floor,
674            ) {
675                rows.push(row);
676            }
677            start = end;
678        }
679        rows
680    }

682    /// Merge a group of mutations sharing one clustering key into a single
683    /// row, applying Cassandra reconciliation semantics at write time:
684    ///
685    /// - Row deletion: the newest `DeleteRow` wins; mutations at or before
686    ///   the deletion timestamp are shadowed (`DeletionTime.deletes` uses
687    ///   `timestamp <= markedForDeleteAt`).
688    /// - Cells: last-write-wins per column by timestamp; a tombstone wins a
689    ///   timestamp tie (Cassandra cell reconciliation).
690    /// - Liveness: from the newest surviving mutation that writes cells, or
691    ///   a pure primary-key insert (no ops and no tombstone payload). Pure
692    ///   row tombstones carry NO liveness, matching Cassandra's serializer.
693    ///
694    /// Returns `None` when the group produces no row at all (e.g. a mutation
695    /// that exists only to carry a partition or range tombstone, or a row
696    /// fully shadowed by the partition/range tombstone `shadow_floor`).
697    fn merge_row_group<'a>(
698        group: &[&'a Mutation],
699        schema: &TableSchema,
700        skip_static_ops: bool,
701        shadow_floor: Option<i64>,
702    ) -> Option<RowWrite<'a>> {
703        use crate::storage::write_engine::mutation::CellOperation;
704
705        // Newest row deletion in the group (if any). A row deletion at or
706        // before the shadow floor is redundant (the partition/range tombstone
707        // already covers it) and is dropped.
708        let mut row_deletion: Option<(i64, i32)> = None;
709        for m in group {
710            let has_delete_row = m
711                .operations
712                .iter()
713                .any(|op| matches!(op, CellOperation::DeleteRow));
714            if has_delete_row
715                && shadow_floor.is_none_or(|floor| m.timestamp_micros > floor)
716                && row_deletion.is_none_or(|(ts, _)| m.timestamp_micros >= ts)
717            {
718                row_deletion = Some((m.timestamp_micros, (m.timestamp_micros / 1_000_000) as i32));
719            }
720        }
721        // Cells and liveness are shadowed by the strongest covering deletion:
722        // the row deletion or the partition/range tombstone floor.
723        let deletion_ts = match (row_deletion.map(|(ts, _)| ts), shadow_floor) {
724            (Some(a), Some(b)) => Some(a.max(b)),
725            (a, b) => a.or(b),
726        };
727
728        // Per-column last-write-wins; tombstones win timestamp ties.
729        let mut cells: std::collections::HashMap<&'a str, MergedOp<'a>> =
730            std::collections::HashMap::new();
731        // Liveness: (timestamp, row-level TTL) of the newest contributing mutation
732        let mut liveness: Option<(i64, Option<u32>)> = None;
733
734        for m in group {
735            // Shadowed entirely by the row deletion
736            if deletion_ts.is_some_and(|dts| m.timestamp_micros <= dts) {
737                continue;
738            }
739
740            let mut contributes_liveness = false;
741            for op in &m.operations {
742                let column = match op {
743                    CellOperation::Write { column, .. }
744                    | CellOperation::WriteWithTtl { column, .. }
745                    | CellOperation::Delete { column } => column.as_str(),
746                    CellOperation::DeleteRow => continue,
747                };
748                if skip_static_ops && is_static_operation(op, schema) {
749                    continue;
750                }
751                if matches!(
752                    op,
753                    CellOperation::Write { .. } | CellOperation::WriteWithTtl { .. }
754                ) {
755                    contributes_liveness = true;
756                }
757
758                let candidate = MergedOp {
759                    op,
760                    timestamp_micros: m.timestamp_micros,
761                    row_ttl_seconds: m.ttl_seconds,
762                };
763                match cells.entry(column) {
764                    std::collections::hash_map::Entry::Vacant(entry) => {
765                        entry.insert(candidate);
766                    }
767                    std::collections::hash_map::Entry::Occupied(mut entry) => {
768                        let existing = entry.get();
769                        let candidate_is_tombstone =
770                            matches!(candidate.op, CellOperation::Delete { .. });
771                        let wins = candidate.timestamp_micros > existing.timestamp_micros
772                            || (candidate.timestamp_micros == existing.timestamp_micros
773                                && (candidate_is_tombstone
774                                    || !matches!(existing.op, CellOperation::Delete { .. })));
775                        if wins {
776                            entry.insert(candidate);
777                        }
778                    }
779                }
780            }
781
782            // A mutation with no ops and no tombstone payload is a pure
783            // primary-key insert: it creates row liveness but no cells.
784            let pure_pk_insert = m.operations.is_empty()
785                && m.partition_tombstone.is_none()
786                && m.range_tombstones.is_empty();
787            if (contributes_liveness || pure_pk_insert)
788                && liveness.is_none_or(|(ts, _)| m.timestamp_micros >= ts)
789            {
790                liveness = Some((m.timestamp_micros, m.ttl_seconds));
791            }
792        }
793
794        let ops: Vec<MergedOp<'a>> = cells.into_values().collect();
795        if ops.is_empty() && row_deletion.is_none() && liveness.is_none() {
796            return None;
797        }
798
799        Some(RowWrite {
800            clustering_key: group[0].clustering_key.as_ref(),
801            liveness_ts: liveness.map(|(ts, _)| ts),
802            ttl_seconds: liveness.and_then(|(_, ttl)| ttl),
803            row_deletion,
804            ops,
805        })
806    }

808    /// Write one merged row (flags + clustering prefix + sizes + body).
809    fn write_merged_row_with_prev_size(
810        &mut self,
811        row: &RowWrite<'_>,
812        schema: &TableSchema,
813        prev_size: u64,
814    ) -> Result<usize> {
815        use crate::storage::write_engine::mutation::CellOperation;
816
817        let start_len = self.buffer.len();
818
819        // Build row header flags
820        let mut flags = 0u8;
821
822        if row.row_deletion.is_some() {
823            flags |= ROW_HAS_DELETION; // 0x10
824        }
825        if row.liveness_ts.is_some() {
826            flags |= ROW_HAS_TIMESTAMP;
827            if row.ttl_seconds.is_some() {
828                flags |= ROW_HAS_TTL;
829            }
830        }
831
832        // All columns present if there is no deletion, all surviving ops are
833        // non-NULL writes, and they cover every regular column.
834        if row.row_deletion.is_none() {
835            let all_writes = row.ops.iter().all(|mop| {
836                matches!(
837                    mop.op,
838                    CellOperation::Write { .. } | CellOperation::WriteWithTtl { .. }
839                )
840            });
841            let has_nulls = row.ops.iter().any(|mop| match mop.op {
842                CellOperation::Write { value, .. } | CellOperation::WriteWithTtl { value, .. } => {
843                    matches!(value, Value::Null)
844                }
845                _ => false,
846            });
847            let regular_column_count = self.regular_columns(schema).len();
848            if all_writes && !has_nulls && row.ops.len() == regular_column_count {
849                flags |= ROW_HAS_ALL_COLUMNS;
850            }
851        }
852
853        // Check if any operation targets a complex column (non-frozen collection)
854        let has_complex = row.ops.iter().any(|mop| {
855            let col_name = match mop.op {
856                CellOperation::Write { column, .. }
857                | CellOperation::WriteWithTtl { column, .. }
858                | CellOperation::Delete { column } => Some(column.as_str()),
859                _ => None,
860            };
861            col_name.is_some_and(|name| {
862                schema
863                    .columns
864                    .iter()
865                    .find(|c| c.name == name)
866                    .map(|c| is_complex_column(&c.data_type))
867                    .unwrap_or(false)
868            })
869        });
870        if has_complex {
871            flags |= ROW_HAS_COMPLEX_DELETION;
872        }
873
874        // Write row flags
875        self.buffer.push(flags);
876
877        // Write clustering prefix if present (before row_size)
878        if let Some(clustering_key) = row.clustering_key {
879            self.write_clustering_prefix(clustering_key, schema)?;
880        }
881
882        // Calculate row body size (everything after row_size VInt)
883        let row_body = self.build_merged_row_body(row, schema, flags)?;
884
885        let prev_size_vint_len = unsigned_len(prev_size);
886
887        // Write row_size (VInt) — Cassandra's serializedRowBodySize() includes
888        // the prev_unfiltered_size VInt as part of the row body
889        let row_body_size = prev_size_vint_len as u64 + row_body.len() as u64;
890        let mut row_size_buf = Vec::new();
891        encode_unsigned(row_body_size, &mut row_size_buf);
892        self.buffer.extend_from_slice(&row_size_buf);
893
894        // Write prev_unfiltered_size (VInt, inside the row body)
895        encode_unsigned(prev_size, &mut self.buffer);
896
897        // Write rest of row body
898        self.buffer.extend_from_slice(&row_body);
899
900        Ok(self.buffer.len() - start_len)
901    }

931        mutation: &Mutation,
932        schema: &TableSchema,
933        prev_size: u64,

935        let start_len = self.buffer.len();
936
937        // Build row header flags - always includes HAS_EXTENDED_FLAGS for static rows
938        let mut flags = ROW_HAS_EXTENDED_FLAGS;

940        // Check if this is a row tombstone

943                op,

945            )
946        });
947
948        if is_row_tombstone {
949            flags |= ROW_HAS_DELETION;
950        }
951
952        // Timestamp is always present for static rows
953        flags |= ROW_HAS_TIMESTAMP;
954
955        // TTL if present (not applicable to row tombstones)

957            flags |= ROW_HAS_TTL;
958        }
959
960        // Check if all static columns are present
961        if !is_row_tombstone {

965                    crate::storage::write_engine::mutation::CellOperation::Write { .. }
966                        | crate::storage::write_engine::mutation::CellOperation::WriteWithTtl { .. }
967                )
968            });

970                crate::storage::write_engine::mutation::CellOperation::Write { value, .. }
971                | crate::storage::write_engine::mutation::CellOperation::WriteWithTtl {
972                    value,
973                    ..
974                } => {
975                    matches!(value, Value::Null)
976                }
977                _ => false,
978            });
979            // Count static columns only for static row
980            let static_column_count = schema.columns.iter().filter(|c| c.is_static).count();
981
982            if all_writes && !has_nulls && mutation.operations.len() == static_column_count {
983                flags |= ROW_HAS_ALL_COLUMNS;
984            }
985        }
986
987        // Write row flags
988        self.buffer.push(flags);
989
990        // Write extended flags - always EXTENDED_IS_STATIC for static rows
991        self.buffer.push(EXTENDED_IS_STATIC);
992
993        // NO clustering prefix for static rows (key difference from write_row)
994
995        // Build row body
996        let row_body = self.build_static_row_body(mutation, schema, flags)?;
997
998        let prev_size_vint_len = unsigned_len(prev_size);
999
1000        // Write row_size (VInt) — includes prev_unfiltered_size VInt + rest of body
1001        let row_body_size = prev_size_vint_len as u64 + row_body.len() as u64;
1002        let mut row_size_buf = Vec::new();
1003        encode_unsigned(row_body_size, &mut row_size_buf);
1004        self.buffer.extend_from_slice(&row_size_buf);
1005
1006        // Write prev_unfiltered_size (VInt, inside the row body)
1007        encode_unsigned(prev_size, &mut self.buffer);
1008
1009        // Write rest of row body
1010        self.buffer.extend_from_slice(&row_body);
1011
1012        Ok(self.buffer.len() - start_len)
1013    }

1015    /// Build static row body (everything after row_size VInt)
1016    ///
1017    /// Similar to build_row_body but only processes static columns.
1018    fn build_static_row_body(
1019        &self,
1020        mutation: &Mutation,
1021        schema: &TableSchema,
1022        flags: u8,
1023    ) -> Result<Vec<u8>> {
1024        let mut body = Vec::new();
1025
1026        // Write timestamp delta (if HAS_TIMESTAMP)
1027        //
1028        // Fix #644 (S6): Cassandra writes UNSIGNED VInt for all temporal deltas.
1029        // SerializationHeader.java:167: out.writeUnsignedVInt(timestamp - stats.minTimestamp)
1030        if (flags & ROW_HAS_TIMESTAMP) != 0 {
1031            let timestamp_delta = (mutation.timestamp_micros - self.stats.min_timestamp) as u64;
1032            encode_unsigned(timestamp_delta, &mut body);
1033        }
1034
1035        // Write TTL delta (if HAS_TTL)
1036        //
1037        // Fix #644 (S6): Both TTL and LDT deltas are UNSIGNED VInt.
1038        // SerializationHeader.java:177: out.writeUnsignedVInt32(ttl - stats.minTTL)
1039        // SerializationHeader.java:172: out.writeUnsignedVInt32(ldt - stats.minLocalDeletionTime)
1040        if (flags & ROW_HAS_TTL) != 0 {
1041            if let Some(ttl) = mutation.ttl_seconds {
1042                let ttl_delta = ttl as i64 - self.stats.min_ttl as i64;
1043                if ttl_delta < 0 {
1044                    return Err(Error::InvalidInput(format!(
1045                        "TTL {} is less than min_ttl {}",
1046                        ttl, self.stats.min_ttl
1047                    )));
1048                }
1049                encode_unsigned(ttl_delta as u64, &mut body);
1050
1051                let local_deletion_time = self.expiring_local_deletion_time(ttl)?;
1052                let ldt_delta =
1053                    (local_deletion_time as i64) - (self.stats.min_local_deletion_time as i64);
1054                if ldt_delta < 0 {
1055                    return Err(Error::InvalidInput(format!(
1056                        "Local deletion time {} is less than min_local_deletion_time {}",
1057                        local_deletion_time, self.stats.min_local_deletion_time
1058                    )));
1059                }
1060                encode_unsigned(ldt_delta as u64, &mut body);
1061            }
1062        }
1063
1064        // Write deletion (if HAS_DELETION)
1065        if (flags & ROW_HAS_DELETION) != 0 {
1066            // Row tombstone: Cassandra canonical order (markedForDeleteAt first, then localDeletionTime)
1067            // Per SerializationHeader.writeDeletionTime(): writeTimestamp() then writeLocalDeletionTime()
1068            // Fix #644 (S6): both are UNSIGNED VInt.
1069            let ts_delta = (mutation.timestamp_micros - self.stats.min_timestamp) as u64;
1070            encode_unsigned(ts_delta, &mut body);
1071
1072            let local_deletion_time = (mutation.timestamp_micros / 1_000_000) as i32;
1073            let ldt_delta =
1074                local_deletion_time.wrapping_sub(self.stats.min_local_deletion_time) as u32;
1075            encode_unsigned(ldt_delta as u64, &mut body);
1076
1077            // Issue #717: the columns subset is NOT optional for tombstone rows.
1078            // Cassandra's UnfilteredSerializer always reads it after the deletion
1079            // times whenever HAS_ALL_COLUMNS is unset; omitting it makes the
1080            // reader consume the next row's bytes as a subset bitmask
1081            // ("Invalid Columns subset bytes; too many bits set").
1082            if (flags & ROW_HAS_ALL_COLUMNS) == 0 {
1083                let static_columns = self.static_columns(schema);
1084                let empty_present: std::collections::HashSet<&str> =
1085                    std::collections::HashSet::new();
1086                self.write_column_subset(&mut body, &static_columns, &empty_present)?;
1087            }
1088
1089            // No cells written for row tombstones
1090            return Ok(body);
1091        }
1092
1093        // Write column bitmap (if NOT HAS_ALL_COLUMNS)
1094        // For static rows, bitmap only covers static columns
1095        if (flags & ROW_HAS_ALL_COLUMNS) == 0 {
1096            self.write_static_column_bitmap(&mut body, mutation, schema)?;
1097        }
1098
1099        // Write cell data for static columns only
1100        self.write_static_cells(&mut body, mutation, schema)?;
1101
1102        Ok(body)
1103    }

1137    /// Write cells for static columns only
1138    fn write_static_cells(
1139        &self,
1140        buf: &mut Vec<u8>,
1141        mutation: &Mutation,
1142        schema: &TableSchema,
1143    ) -> Result<()> {
1144        // Get set of static column names for validation
1145        let static_column_names: std::collections::HashSet<_> = schema
1146            .columns
1147            .iter()
1148            .filter(|c| c.is_static)
1149            .map(|c| &c.name)
1150            .collect();
1151
1152        for op in self.sorted_operations(mutation, &self.static_columns(schema)) {
1153            match op {
1154                crate::storage::write_engine::mutation::CellOperation::Write { column, value } => {

1219    /// Build a merged row body (everything after the row_size VInt, excluding
1220    /// the prev_unfiltered_size VInt written by the caller).
1221    ///
1222    /// Field order per Cassandra's `UnfilteredSerializer.serializeRowBody`:
1223    /// liveness timestamp, TTL + expiration LDT, row deletion, columns
1224    /// subset, then cells. Issue #717: the columns subset is written for
1225    /// EVERY row lacking HAS_ALL_COLUMNS — including row tombstones.
1226    fn build_merged_row_body(
1227        &self,
1228        row: &RowWrite<'_>,
1229        schema: &TableSchema,
1230        flags: u8,
1231    ) -> Result<Vec<u8>> {
1232        let mut body = Vec::new();
1233
1234        // Write timestamp delta (if HAS_TIMESTAMP)
1235        //
1236        // Fix #644 (S6): Cassandra writes UNSIGNED VInt for all temporal deltas.
1237        // SerializationHeader.java:167: out.writeUnsignedVInt(timestamp - stats.minTimestamp)
1238        if (flags & ROW_HAS_TIMESTAMP) != 0 {
1239            let liveness_ts = row.liveness_ts.ok_or_else(|| {
1240                Error::InvalidInput(
1241                    "ROW_HAS_TIMESTAMP set but row has no liveness timestamp".to_string(),
1242                )
1243            })?;
1244            let timestamp_delta = (liveness_ts - self.stats.min_timestamp) as u64;
1245            encode_unsigned(timestamp_delta, &mut body);
1246        }
1247
1248        // Write TTL delta (if HAS_TTL)
1249        //
1250        // Fix #644 (S6): Both TTL and LDT deltas are UNSIGNED VInt.
1251        // SerializationHeader.java:177: out.writeUnsignedVInt32(ttl - stats.minTTL)
1252        // SerializationHeader.java:172: out.writeUnsignedVInt32(ldt - stats.minLocalDeletionTime)
1253        if (flags & ROW_HAS_TTL) != 0 {
1254            if let Some(ttl) = row.ttl_seconds {
1255                let ttl_delta = ttl as i64 - self.stats.min_ttl as i64;
1256                if ttl_delta < 0 {
1257                    return Err(Error::InvalidInput(format!(
1258                        "TTL {} is less than min_ttl {}",
1259                        ttl, self.stats.min_ttl
1260                    )));
1261                }
1262                encode_unsigned(ttl_delta as u64, &mut body);
1263
1264                let local_deletion_time = self.expiring_local_deletion_time(ttl)?;
1265                let ldt_delta =
1266                    (local_deletion_time as i64) - (self.stats.min_local_deletion_time as i64);
1267                if ldt_delta < 0 {
1268                    return Err(Error::InvalidInput(format!(
1269                        "Local deletion time {} is less than min_local_deletion_time {}",
1270                        local_deletion_time, self.stats.min_local_deletion_time
1271                    )));
1272                }
1273                encode_unsigned(ldt_delta as u64, &mut body);
1274            }
1275        }
1276
1277        // Write deletion (if HAS_DELETION)
1278        if (flags & ROW_HAS_DELETION) != 0 {
1279            // Row tombstone: Cassandra canonical order (markedForDeleteAt first, then localDeletionTime)
1280            // Per SerializationHeader.writeDeletionTime(): writeTimestamp() then writeLocalDeletionTime()
1281            // Fix #644 (S6): both are UNSIGNED VInt.
1282            let (deletion_ts, local_deletion_time) = row.row_deletion.ok_or_else(|| {
1283                Error::InvalidInput("ROW_HAS_DELETION set but row has no deletion time".to_string())
1284            })?;
1285            let ts_delta = (deletion_ts - self.stats.min_timestamp) as u64;
1286            encode_unsigned(ts_delta, &mut body);
1287
1288            let ldt_delta =
1289                local_deletion_time.wrapping_sub(self.stats.min_local_deletion_time) as u32;
1290            encode_unsigned(ldt_delta as u64, &mut body);
1291        }
1292
1293        // Write column bitmap (if NOT HAS_ALL_COLUMNS).
1294        // Issue #717: this is written even for row tombstones — Cassandra's
1295        // deserializer reads the subset right after the deletion times.
1296        if (flags & ROW_HAS_ALL_COLUMNS) == 0 {
1297            self.write_merged_column_bitmap(&mut body, &row.ops, schema)?;
1298        }
1299
1300        // Write cell data (none survive for pure row tombstones)
1301        self.write_merged_cells(&mut body, row, schema)?;
1302
1303        Ok(body)
1304    }

1307    ///
1308    /// Format:
1309    /// ```text
1310    /// [header: VInt]              ← 2 bits per clustering column (state)
1311    /// [value_1: type-specific]    ← Only if state is PRESENT (00)
1312    /// [value_2: type-specific]

1318        schema: &TableSchema,
1319    ) -> Result<()> {

1321        // 00 = PRESENT, 01 = EMPTY, 10 = NULL, 11 = reserved
1322        let mut header = 0u64;
1323        for (i, (_, value)) in clustering_key.columns.iter().enumerate() {
1324            let state = match value {
1325                Value::Null => 2, // NULL
1326                _ => 0,           // PRESENT
1327            };

1329        }
1330
1331        // Write header as VUInt

1333
1334        // Write values for PRESENT columns
1335        for (i, (_, value)) in clustering_key.columns.iter().enumerate() {
1336            if !matches!(value, Value::Null) {

1338                if i >= schema.clustering_keys.len() {

1342                        schema.clustering_keys.len()
1343                    )));
1344                }
1345                let cluster_col = &schema.clustering_keys[i];
1346                let comparator = ComparatorType::from_data_type(&cluster_col.data_type)?;

1349                let value_bytes = serialize_value_for_clustering(value, &comparator)?;

1685    /// Write per-cell TTL fields for a complex cell.
1686    ///
1687    /// When TTL is present, writes:
1688    /// - flags: CELL_IS_EXPIRING (0x02), NO USE_ROW_TIMESTAMP
1689    /// - timestamp delta (unsigned VInt; fix #644: all temporal deltas are unsigned)
1690    /// - local_deletion_time delta (unsigned VInt)
1691    /// - TTL delta (unsigned VInt)
1692    ///
1693    /// When TTL is absent, writes:
1694    /// - flags: base_flags | CELL_USE_ROW_TIMESTAMP (0x08)
1695    ///
1696    /// Returns the flags byte written (for caller to check HAS_EMPTY_VALUE etc.).
1697    fn write_complex_cell_header(
1698        &self,
1699        buf: &mut Vec<u8>,
1700        base_flags: u8,
1701        timestamp_micros: i64,
1702        ttl_seconds: Option<u32>,
1703    ) -> Result<()> {
1704        match ttl_seconds {
1705            Some(ttl) => {
1706                // Expiring cell: IS_EXPIRING flag, explicit timestamp + LDT + TTL
1707                let flags = base_flags | CELL_IS_EXPIRING;
1708                buf.push(flags);
1709
1710                // Timestamp delta (UNSIGNED VInt, NOT USE_ROW_TIMESTAMP)
1711                // Fix #644 (S6): SerializationHeader.java:167 uses writeUnsignedVInt.
1712                let timestamp_delta = (timestamp_micros - self.stats.min_timestamp) as u64;
1713                encode_unsigned(timestamp_delta, buf);
1714
1715                // local_deletion_time = now + ttl
1716                let now_seconds = std::time::SystemTime::now()
1717                    .duration_since(std::time::UNIX_EPOCH)
1718                    .map_err(|e| Error::Storage(format!("System time error: {}", e)))?
1719                    .as_secs() as i32;
1720                let local_deletion_time = now_seconds.saturating_add(ttl as i32);
1721                let ldt_delta =
1722                    (local_deletion_time as i64) - (self.stats.min_local_deletion_time as i64);

2408        &self,
2409        buf: &mut Vec<u8>,
2410        columns: &[&Column],
2411        present_columns: &std::collections::HashSet<&str>,
2412    ) -> Result<()> {
2413        let mut present_indices = Vec::new();
2414        let mut missing_indices = Vec::new();
2415
2416        for (idx, column) in columns.iter().enumerate() {
2417            if present_columns.contains(column.name.as_str()) {
2418                present_indices.push(idx);
2419            } else {
2420                missing_indices.push(idx);
2421            }
2422        }
2423
2424        if missing_indices.is_empty() {
2425            encode_unsigned(0, buf);
2426            return Ok(());
2427        }
2428
2429        if columns.len() < 64 {
2430            let mut bitmap = 0u64;
2431            for idx in missing_indices {
2432                bitmap |= 1u64 << idx;
2433            }
2434            encode_unsigned(bitmap, buf);
2435            return Ok(());
2436        }
2437
2438        encode_unsigned((columns.len() - present_indices.len()) as u64, buf);
2439
2440        if present_indices.len() < columns.len() / 2 {
2441            for idx in present_indices {
2442                encode_unsigned(idx as u64, buf);
2443            }
2444        } else {
2445            for idx in missing_indices {
2446                encode_unsigned(idx as u64, buf);
2447            }
2448        }
2449
2450        Ok(())
2451    }

2513/// Generate a version-1 TimeUUID for use as a list cell path.
2514///
2515/// List elements in Cassandra use TimeUUIDs as cell paths to maintain insertion order.
2516/// Each call with a different `element_index` produces a monotonically increasing UUID.
2517///
2518/// # Arguments
2519/// * `timestamp_micros` - Mutation timestamp in microseconds since Unix epoch
2520/// * `element_index` - Index of the element within the list (for monotonic ordering)
2521fn generate_list_cell_path_timeuuid(timestamp_micros: i64, element_index: u64) -> [u8; 16] {
2522    // UUID v1 timestamp: 100-nanosecond intervals since UUID epoch (Oct 15, 1582)
2523    // Offset from Unix epoch to UUID epoch in 100-ns units
2524    const UUID_EPOCH_OFFSET: u64 = 0x01B2_1DD2_1381_4000;
2525
2526    let ts_100ns = (timestamp_micros as u64) * 10 + element_index;
2527    let uuid_ts = ts_100ns + UUID_EPOCH_OFFSET;
2528
2529    // Extract time fields per RFC 4122
2530    let time_low = (uuid_ts & 0xFFFF_FFFF) as u32;
2531    let time_mid = ((uuid_ts >> 32) & 0xFFFF) as u16;
2532    let time_hi = ((uuid_ts >> 48) & 0x0FFF) as u16 | 0x1000; // version 1
2533
2534    // Fixed clock_seq and node for deterministic output
2535    let clock_seq: u16 = 0x80; // variant bits (10xx) + seq=0
2536    let node: [u8; 6] = [0x00, 0x00, 0x00, 0x00, 0x00, 0x00];
2537
2538    let mut uuid = [0u8; 16];
2539    uuid[0..4].copy_from_slice(&time_low.to_be_bytes());
2540    uuid[4..6].copy_from_slice(&time_mid.to_be_bytes());
2541    uuid[6..8].copy_from_slice(&time_hi.to_be_bytes());
2542    uuid[8] = (clock_seq >> 8) as u8;
2543    uuid[9] = (clock_seq & 0xFF) as u8;
2544    uuid[10..16].copy_from_slice(&node);
2545
2546    uuid
2547}

2571/// Serialize a Value to bytes for cell storage
2572///
2573/// This follows Cassandra's type-specific serialization rules.
2574fn serialize_value(value: &Value) -> Result<Vec<u8>> {
2575    match value {
2576        Value::Null => Ok(Vec::new()),
2577        Value::Boolean(b) => Ok(vec![if *b { 1 } else { 0 }]),
2578        Value::TinyInt(n) => Ok(vec![*n as u8]),
2579        Value::SmallInt(n) => Ok(n.to_be_bytes().to_vec()),
2580        Value::Integer(n) => Ok(n.to_be_bytes().to_vec()),
2581        Value::BigInt(n) => Ok(n.to_be_bytes().to_vec()),
2582        Value::Counter(n) => Ok(n.to_be_bytes().to_vec()),
2583        Value::Float32(f) => Ok(f.to_bits().to_be_bytes().to_vec()),
2584        Value::Float(f) => Ok(f.to_bits().to_be_bytes().to_vec()),
2585        Value::Text(s) => Ok(s.as_bytes().to_vec()),
2586        Value::Blob(bytes) => Ok(bytes.clone()),
2587        Value::Timestamp(millis) => Ok(millis.to_be_bytes().to_vec()),
2588        Value::Date(days) => {
2589            // Cassandra DATE: stored as unsigned int with Integer.MIN_VALUE offset
2590            let stored = days.wrapping_sub(i32::MIN) as u32;
2591            Ok(stored.to_be_bytes().to_vec())
2592        }
2593        Value::Time(nanos) => Ok(nanos.to_be_bytes().to_vec()),
2594        Value::Uuid(bytes) => Ok(bytes.to_vec()),
2595        Value::Inet(bytes) => Ok(bytes.clone()),
2596        Value::Varint(bytes) => Ok(bytes.clone()),
2597        Value::Decimal { scale, unscaled } => {
2598            let mut result = Vec::new();
2599            result.extend_from_slice(&scale.to_be_bytes());
2600            result.extend_from_slice(unscaled);
2601            Ok(result)
2602        }
2603        Value::Duration {
2604            months,
2605            days,
2606            nanos,
2607        } => {
2608            let mut result = Vec::new();
2609            // Cassandra DurationType stores three signed VInts, not fixed-width ints.
2610            encode_signed(*months as i64, &mut result);
2611            encode_signed(*days as i64, &mut result);
2612            encode_signed(*nanos, &mut result);
2613            Ok(result)
2614        }
2615        Value::Udt(udt_value) => {
2616            // Construct UdtTypeDef from UdtValue fields by inferring types
2617            let mut schema =
2618                UdtTypeDef::new(udt_value.keyspace.clone(), udt_value.type_name.clone());
2619
2620            // Infer field types from values
2621            for field in &udt_value.fields {
2622                let field_type = infer_cql_type_from_value(field.value.as_ref());
2623                schema = schema.with_field(field.name.clone(), field_type, true);
2624            }
2625
2626            let serializer = TypeSerializer::new();
2627            serializer.serialize_udt(value, &schema)
2628        }
2629        Value::List(elements) | Value::Set(elements) => {
2630            let mut buf = Vec::new();
2631            buf.extend_from_slice(&len_as_i32(elements.len())?.to_be_bytes());

2633                let elem_bytes = serialize_collection_element(elem, "Collection")?;
2634                buf.extend_from_slice(&len_as_i32(elem_bytes.len())?.to_be_bytes());
2635                buf.extend_from_slice(&elem_bytes);
2636            }
2637            Ok(buf)
2638        }
2639        Value::Map(entries) => {
2640            let mut buf = Vec::new();
2641            buf.extend_from_slice(&len_as_i32(entries.len())?.to_be_bytes());
2642            for (key, val) in entries {
2643                if matches!(key, Value::Null) {
2644                    return Err(Error::InvalidInput(
2645                        "MAP keys cannot be null (CQL semantics)".to_string(),
2646                    ));
2647                }
2648                let key_bytes = serialize_value(key)?;
2649                buf.extend_from_slice(&len_as_i32(key_bytes.len())?.to_be_bytes());
2650                buf.extend_from_slice(&key_bytes);
2651                let val_bytes = serialize_value(val)?;
2652                buf.extend_from_slice(&len_as_i32(val_bytes.len())?.to_be_bytes());
2653                buf.extend_from_slice(&val_bytes);
2654            }
2655            Ok(buf)
2656        }
2657        Value::Tuple(fields) => {
2658            let mut buf = Vec::new();
2659            for field in fields {
2660                match field {
2661                    Value::Null => buf.extend_from_slice(&(-1i32).to_be_bytes()),
2662                    other => {
2663                        let field_bytes = serialize_value(other)?;
2664                        buf.extend_from_slice(&len_as_i32(field_bytes.len())?.to_be_bytes());
2665                        buf.extend_from_slice(&field_bytes);
2666                    }
2667                }
2668            }
2669            Ok(buf)
2670        }
2671        Value::Frozen(inner) => serialize_value(inner),
2672        _ => Err(Error::InvalidInput(format!(
2673            "Unsupported value type for serialization: {:?}",
2674            value
2675        ))),
2676    }

2812/// Collect and merge static-column operations from all mutations in a partition.
2813///
2814/// Scans every mutation (regardless of whether it has a clustering key) and
2815/// collects operations that target static columns.  Last-write-wins by
2816/// `timestamp_micros` when the same column is written more than once.
2817///
2818/// Mutations at or before `shadow_floor` (the partition tombstone's deletion
2819/// timestamp) are skipped: their static cells are shadowed and an sstable
2820/// must be internally reconciled (see `DataWriter::write_partition`).
2821///
2822/// Returns the merged operations in an unspecified order (the writer will
2823/// sort them by schema column order when building the row body).
2824fn collect_static_operations(
2825    mutations: &[Mutation],
2826    schema: &TableSchema,
2827    shadow_floor: Option<i64>,
2828) -> Vec<crate::storage::write_engine::mutation::CellOperation> {
2829    use std::collections::HashMap;
2830
2831    // Map: column_name → (timestamp, operation)
2832    let mut best: HashMap<String, (i64, crate::storage::write_engine::mutation::CellOperation)> =
2833        HashMap::new();
2834
2835    for mutation in mutations {
2836        if shadow_floor.is_some_and(|floor| mutation.timestamp_micros <= floor) {
2837            continue;
2838        }
2839        for op in &mutation.operations {
2840            if !is_static_operation(op, schema) {
2841                continue;
2842            }
2843            let col_name = match op {
2844                crate::storage::write_engine::mutation::CellOperation::Write { column, .. }
2845                | crate::storage::write_engine::mutation::CellOperation::WriteWithTtl {
2846                    column,
2847                    ..
2848                }
2849                | crate::storage::write_engine::mutation::CellOperation::Delete { column } => {
2850                    column.clone()
2851                }
2852                crate::storage::write_engine::mutation::CellOperation::DeleteRow => continue,
2853            };
2854            let entry = best.entry(col_name).or_insert((i64::MIN, op.clone()));
2855            if mutation.timestamp_micros >= entry.0 {
2856                *entry = (mutation.timestamp_micros, op.clone());
2857            }
2858        }
2859    }
2860
2861    best.into_values().map(|(_, op)| op).collect()
2862}

2864/// Whether a range tombstone's clustering range covers the given clustering key.
2865fn range_tombstone_covers(
2866    rt: &RangeTombstone,
2867    clustering_key: Option<&ClusteringKey>,
2868    schema: &TableSchema,
2869) -> bool {
2870    use std::cmp::Ordering;
2871
2872    let Some(ck) = clustering_key else {
2873        return false;
2874    };
2875    let cmp = |bound: &ClusteringKey| ck.compare(bound, schema).unwrap_or_else(|_| ck.cmp(bound));
2876
2877    let after_start = match &rt.start {
2878        ClusteringBound::Inclusive(b) => cmp(b) != Ordering::Less,
2879        ClusteringBound::Exclusive(b) => cmp(b) == Ordering::Greater,
2880        ClusteringBound::Bottom => true,
2881        ClusteringBound::Top => false,
2882    };
2883    let before_end = match &rt.end {
2884        ClusteringBound::Inclusive(b) => cmp(b) != Ordering::Greater,
2885        ClusteringBound::Exclusive(b) => cmp(b) == Ordering::Less,
2886        ClusteringBound::Top => true,
2887        ClusteringBound::Bottom => false,
2888    };
2889    after_start && before_end
2890}

2893///
2894/// Fixed-width types: raw bytes (no length prefix)
2895/// Variable-width types: VInt length + bytes
2896fn serialize_value_for_clustering(value: &Value, comparator: &ComparatorType) -> Result<Vec<u8>> {
2897    match (value, comparator) {
2898        // Fixed-width types (no length prefix)
2899        (Value::Boolean(b), ComparatorType::Boolean) => Ok(vec![if *b { 1 } else { 0 }]),
2900        (Value::TinyInt(n), ComparatorType::TinyInt) => Ok(n.to_be_bytes().to_vec()),
2901        (Value::SmallInt(n), ComparatorType::SmallInt) => Ok(n.to_be_bytes().to_vec()),
2902        (Value::Integer(n), ComparatorType::Int) => Ok(n.to_be_bytes().to_vec()),
2903        (Value::BigInt(n), ComparatorType::BigInt) => Ok(n.to_be_bytes().to_vec()),
2904        (Value::Float32(f), ComparatorType::Float32) => Ok(f.to_bits().to_be_bytes().to_vec()),
2905        (Value::Float(f), ComparatorType::Float) => Ok(f.to_bits().to_be_bytes().to_vec()),
2906        (Value::Timestamp(millis), ComparatorType::Timestamp) => Ok(millis.to_be_bytes().to_vec()),
2907        (Value::Date(days), ComparatorType::Date) => {
2908            // Cassandra DATE in clustering keys: stored as unsigned int with Integer.MIN_VALUE offset
2909            let stored = days.wrapping_sub(i32::MIN) as u32;
2910            let mut result = Vec::new();
2911            encode_unsigned(4, &mut result);
2912            result.extend_from_slice(&stored.to_be_bytes());
2913            Ok(result)
2914        }
2915        (Value::Uuid(bytes), ComparatorType::Uuid) => Ok(bytes.to_vec()),
2916
2917        // Variable-width types (VInt length + bytes)
2918        (Value::Text(s), ComparatorType::Text) => {
2919            let bytes = s.as_bytes();
2920            let mut result = Vec::new();
2921            encode_unsigned(bytes.len() as u64, &mut result);
2922            result.extend_from_slice(bytes);
2923            Ok(result)
2924        }
2925        (Value::Blob(bytes), ComparatorType::Blob) => {
2926            let mut result = Vec::new();
2927            encode_unsigned(bytes.len() as u64, &mut result);
2928            result.extend_from_slice(bytes);
2929            Ok(result)
2930        }
2931
2932        // Frozen collections as clustering keys: serialize the full collection bytes with VInt length prefix
2933        (Value::Frozen(inner), _) => {
2934            let bytes = serialize_value(inner)?;
2935            let mut result = Vec::new();
2936            encode_unsigned(bytes.len() as u64, &mut result);
2937            result.extend_from_slice(&bytes);
2938            Ok(result)
2939        }
2940
2941        _ => Err(Error::InvalidInput(format!(
2942            "Type mismatch or unsupported clustering type: value={:?}, comparator={:?}",
2943            value, comparator
2944        ))),
2945    }
2946}

5763    /// Parse a `write_complex_column` output buffer and return the flag byte for every cell.
5764    ///
5765    /// The buffer has this deterministic structure:
5766    /// ```text
5767    /// [complex_deletion_ts_delta:  unsigned VInt]  ← 2 VInts, time-derived but fixed per stats
5768    /// [complex_deletion_ldt_delta: unsigned VInt]
5769    /// [cell_count: unsigned VInt]
5770    /// per cell:
5771    ///   [flags: u8]
5772    ///   if IS_EXPIRING (0x02 set):
5773    ///     [ts_delta:  unsigned VInt]
5774    ///     [ldt_delta: unsigned VInt]   ← wall-clock-derived
5775    ///     [ttl_delta: unsigned VInt]
5776    ///   [path_len:  unsigned VInt]
5777    ///   [path_bytes: path_len]
5778    ///   if !HAS_EMPTY_VALUE (0x04 NOT set):
5779    ///     [value_len: unsigned VInt]
5780    ///     [value_bytes: value_len]
5781    /// ```
5782    ///
5783    /// Scanning the raw buffer for a flag byte value is fragile because
5784    /// wall-clock-derived LDT bytes can coincidentally equal the flag byte (~1-2% of CI runs).
5785    /// This helper walks the structure deterministically so each flag byte is read at
5786    /// its exact position.
5787    fn parse_complex_cell_flags(buf: &[u8]) -> Vec<u8> {
5788        /// Read one unsigned VInt from `buf` starting at `*pos`; advance `*pos`.
5789        fn read_uvint(buf: &[u8], pos: &mut usize) -> u64 {
5790            let first = buf[*pos];
5791            *pos += 1;
5792            if first == 0xFF {
5793                // 9-byte form: 0xFF + 8 big-endian bytes
5794                let mut v = 0u64;
5795                for _ in 0..8 {
5796                    v = (v << 8) | buf[*pos] as u64;
5797                    *pos += 1;
5798                }
5799                return v;
5800            }
5801            // Count leading 1-bits in `first` to determine extra bytes
5802            let extra = first.leading_ones() as usize;
5803            // Data bits in first byte: mask off the leading 1s and the 0 separator
5804            let mask = 0xFF_u8.wrapping_shr((extra + 1) as u32);
5805            let mut v = (first & mask) as u64;
5806            for _ in 0..extra {
5807                v = (v << 8) | buf[*pos] as u64;
5808                *pos += 1;
5809            }
5810            v
5811        }
5812
5813        let mut pos = 0usize;
5814        // Skip complex deletion header: 2 unsigned VInts
5815        read_uvint(buf, &mut pos);
5816        read_uvint(buf, &mut pos);
5817
5818        // Cell count
5819        let cell_count = read_uvint(buf, &mut pos) as usize;
5820
5821        let mut flags_out = Vec::with_capacity(cell_count);
5822        for _ in 0..cell_count {
5823            let flags = buf[pos];
5824            pos += 1;
5825            flags_out.push(flags);
5826
5827            if (flags & CELL_IS_EXPIRING) != 0 {
5828                // IS_EXPIRING: ts_delta + ldt_delta + ttl_delta (3 unsigned VInts)
5829                read_uvint(buf, &mut pos);
5830                read_uvint(buf, &mut pos);
5831                read_uvint(buf, &mut pos);
5832            }
5833            // USE_ROW_TIMESTAMP / non-expiring cells: no extra fields before path
5834
5835            // Cell path: path_len VInt + path_len bytes
5836            let path_len = read_uvint(buf, &mut pos) as usize;
5837            pos += path_len;
5838
5839            // Cell value: only present when HAS_EMPTY_VALUE is NOT set
5840            if (flags & CELL_HAS_EMPTY_VALUE) == 0 {
5841                let value_len = read_uvint(buf, &mut pos) as usize;
5842                pos += value_len;
5843            }
5844        }
5845
5846        flags_out
5847    }

6268    /// Bounded-memory evidence (Issue #492): after each `write_partition` the
6269    /// scratch buffer must hold only the most recent partition, while the
6270    /// flushed `position` grows monotonically. This is the proof that peak heap
6271    /// is O(largest partition) rather than O(file).
6272    #[test]
6273    fn test_streaming_writer_bounds_memory_to_one_partition() {
6274        let schema = create_test_schema();
6275        let partitions = streaming_test_partitions();
6276
6277        let dir = tempfile::tempdir().unwrap();
6278        let data_path = dir.path().join("nb-1-big-Data.db");
6279        let mut writer = DataWriter::with_sink(create_test_stats(), data_path);
6280
6281        let mut prev_flushed = 0u64;
6282        // Tracks the largest single-partition flushed size. Because the scratch is
6283        // cleared after every partition (asserted below), peak resident Data.db
6284        // bytes are bounded by this value, not the whole file.
6285        let mut max_partition_size = 0usize;
6286        for (i, (key, mutations)) in partitions.iter().enumerate() {
6287            let flushed_before = writer.flushed_position();
6288            writer
6289                .write_partition(key, mutations, &schema, None, &[])
6290                .unwrap();
6291
6292            // After a partition is written it has been flushed and the scratch
6293            // cleared: the scratch must be empty, never accumulating prior
6294            // partitions.
6295            assert_eq!(
6296                writer.scratch_len(),
6297                0,
6298                "scratch must be cleared after partition {i} (bounded memory)"
6299            );
6300
6301            // Flushed bytes must strictly increase by this partition's size.
6302            let flushed_after = writer.flushed_position();
6303            assert!(
6304                flushed_after > flushed_before,
6305                "flushed position must grow after writing partition {i}"
6306            );
6307            let this_partition_size = (flushed_after - flushed_before) as usize;
6308            max_partition_size = max_partition_size.max(this_partition_size);
6309            assert!(flushed_after > prev_flushed);
6310            prev_flushed = flushed_after;
6311        }
6312
6313        let total = writer.finish_streaming().unwrap();
6314        assert_eq!(
6315            total, prev_flushed,
6316            "total size must equal last flushed pos"
6317        );
6318
6319        // Peak resident bytes were bounded by the largest single partition,
6320        // which is far smaller than the whole file for many partitions.
6321        assert!(
6322            (max_partition_size as u64) < total,
6323            "largest single partition ({max_partition_size}) must be smaller than the full file ({total})"
6324        );
6325    }
