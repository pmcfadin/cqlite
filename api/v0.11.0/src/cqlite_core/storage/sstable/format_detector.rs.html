1//! SSTable format detection and version identification
2//!
3//! This module provides bulletproof detection of SSTable format versions
4//! across all Cassandra versions (2.x, 3.x, 4.x, 5.x) with automatic
5//! format-specific parser selection.
6
7use crate::{Error, Result};
8use std::collections::HashMap;
9use std::path::Path;
10
11/// SSTable format versions supported by CQLite
12#[derive(Debug, Clone, PartialEq, Eq)]
13pub enum SSTableFormat {
14    /// Cassandra 2.x formats (ic, jb, etc.)
15    V2x(String),
16    /// Cassandra 3.x formats (ma, mb, mc, md, me)
17    V3x(String),
18    /// Cassandra 4.x formats (na, nb)
19    V4x(String),
20    /// Cassandra 5.x formats (oa, etc.)
21    V5x(String),
22    /// Unknown or unsupported format
23    Unknown(String),
24}
25
26impl SSTableFormat {
27    /// Get the format version string
28    pub fn version(&self) -> &str {
29        match self {
30            SSTableFormat::V2x(v) => v,
31            SSTableFormat::V3x(v) => v,
32            SSTableFormat::V4x(v) => v,
33            SSTableFormat::V5x(v) => v,
34            SSTableFormat::Unknown(v) => v,
35        }
36    }
37
38    /// Check if format supports compression
39    pub fn supports_compression(&self) -> bool {
40        match self {
41            SSTableFormat::V2x(_) => true,
42            SSTableFormat::V3x(_) => true,
43            SSTableFormat::V4x(_) => true,
44            SSTableFormat::V5x(_) => true,
45            SSTableFormat::Unknown(_) => false,
46        }
47    }
48
49    /// Check if format uses chunk-based compression
50    pub fn uses_chunk_compression(&self) -> bool {
51        match self {
52            SSTableFormat::V2x(_) => true,
53            SSTableFormat::V3x(_) => true,
54            SSTableFormat::V4x(_) => true,
55            SSTableFormat::V5x(_) => true,
56            SSTableFormat::Unknown(_) => false,
57        }
58    }
59
60    /// Get expected compression algorithm
61    pub fn default_compression(&self) -> &'static str {
62        match self {
63            SSTableFormat::V2x(_) => "SnappyCompressor",
64            SSTableFormat::V3x(_) => "LZ4Compressor",
65            SSTableFormat::V4x(_) => "LZ4Compressor",
66            SSTableFormat::V5x(_) => "LZ4Compressor",
67            SSTableFormat::Unknown(_) => "LZ4Compressor",
68        }
69    }
70}

72/// SSTable format detector with comprehensive version support
73pub struct FormatDetector {
74    /// Known format version mappings
75    format_map: HashMap<String, SSTableFormat>,
76}
77
78impl FormatDetector {
79    /// Create a new format detector with all known versions
80    pub fn new() -> Self {
81        let mut format_map = HashMap::new();
82
83        // Cassandra 2.x formats
84        format_map.insert("ic".to_string(), SSTableFormat::V2x("ic".to_string()));
85        format_map.insert("jb".to_string(), SSTableFormat::V2x("jb".to_string()));
86
87        // Cassandra 3.x formats
88        format_map.insert("ma".to_string(), SSTableFormat::V3x("ma".to_string()));
89        format_map.insert("mb".to_string(), SSTableFormat::V3x("mb".to_string()));
90        format_map.insert("mc".to_string(), SSTableFormat::V3x("mc".to_string()));
91        format_map.insert("md".to_string(), SSTableFormat::V3x("md".to_string()));
92        format_map.insert("me".to_string(), SSTableFormat::V3x("me".to_string()));
93
94        // Cassandra 4.x formats
95        format_map.insert("na".to_string(), SSTableFormat::V4x("na".to_string()));
96        format_map.insert("nb".to_string(), SSTableFormat::V4x("nb".to_string()));
97
98        // Cassandra 5.x formats
99        format_map.insert("oa".to_string(), SSTableFormat::V5x("oa".to_string()));
100        // BTI format (BtiFormat.java:287-420): `da` is the only BTI version letter.
101        // Mapped to V5x rather than Unknown so callers can select the BTI read path.
102        // Full BTI routing is completed in VG5; here we ensure `da` is not Unknown.
103        format_map.insert("da".to_string(), SSTableFormat::V5x("da".to_string()));
104
105        Self { format_map }
106    }
107
108    /// Detect SSTable format from file path
109    ///
110    /// SSTable files follow pattern: {version}-{generation}-{size}-{component}.db
111    /// Example: nb-1-big-Data.db
112    pub fn detect_from_path(&self, path: &Path) -> Result<SSTableFormat> {
113        let filename = path
114            .file_name()
115            .and_then(|f| f.to_str())
116            .ok_or_else(|| Error::InvalidPath(format!("Invalid SSTable filename: {:?}", path)))?;
117
118        // Extract format version from filename
119        let parts: Vec<&str> = filename.split('-').collect();
120        if parts.len() < 4 {
121            return Err(Error::InvalidFormat(format!(
122                "Invalid SSTable filename format: {}",
123                filename
124            )));
125        }
126
127        let version = parts[0];
128        self.detect_from_version(version)
129    }
130
131    /// Detect format from version string
132    pub fn detect_from_version(&self, version: &str) -> Result<SSTableFormat> {
133        self.format_map
134            .get(version)
135            .cloned()
136            .or_else(|| Some(SSTableFormat::Unknown(version.to_string())))
137            .ok_or_else(|| {
138                Error::UnsupportedFormat(format!("Unknown SSTable version: {}", version))
139            })
140    }
141
142    /// Detect format from multiple SSTable files in a directory
143    pub fn detect_from_directory(&self, dir: &Path) -> Result<SSTableFormat> {
144        use std::fs;
145
146        let entries = fs::read_dir(dir).map_err(Error::Io)?;
147
148        for entry in entries {
149            let entry = entry.map_err(Error::Io)?;
150            let path = entry.path();
151
152            if path.extension().and_then(|s| s.to_str()) == Some("db") {
153                if let Ok(format) = self.detect_from_path(&path) {
154                    return Ok(format);
155                }
156            }
157        }
158
159        Err(Error::InvalidFormat(
160            "No valid SSTable files found in directory".to_string(),
161        ))
162    }
163
164    /// Get all supported format versions
165    pub fn supported_versions(&self) -> Vec<String> {
166        self.format_map.keys().cloned().collect()
167    }
168
169    /// Check if a format version is supported
170    pub fn is_supported(&self, version: &str) -> bool {
171        self.format_map.contains_key(version)
172    }
173}
174
175impl Default for FormatDetector {
176    fn default() -> Self {
177        Self::new()
178    }
179}

181/// SSTable file components
182#[derive(Debug, Clone, PartialEq, Eq)]
183#[allow(clippy::upper_case_acronyms)]
184pub enum SSTableComponent {
185    Data,
186    Index,
187    Summary,
188    Filter,
189    CompressionInfo,
190    Statistics,
191    Digest,
192    TOC,
193}
194
195impl SSTableComponent {
196    /// Parse component from filename
197    pub fn from_filename(filename: &str) -> Option<Self> {
198        if filename.ends_with("-Data.db") {
199            Some(SSTableComponent::Data)
200        } else if filename.ends_with("-Index.db") {
201            Some(SSTableComponent::Index)
202        } else if filename.ends_with("-Summary.db") {
203            Some(SSTableComponent::Summary)
204        } else if filename.ends_with("-Filter.db") {
205            Some(SSTableComponent::Filter)
206        } else if filename.ends_with("-CompressionInfo.db") {
207            Some(SSTableComponent::CompressionInfo)
208        } else if filename.ends_with("-Statistics.db") {
209            Some(SSTableComponent::Statistics)
210        } else if filename.ends_with("-Digest.crc32") {
211            Some(SSTableComponent::Digest)
212        } else if filename.ends_with("-TOC.txt") {
213            Some(SSTableComponent::TOC)
214        } else {
215            None
216        }
217    }
218
219    /// Get component file suffix
220    pub fn suffix(&self) -> &'static str {
221        match self {
222            SSTableComponent::Data => "Data.db",
223            SSTableComponent::Index => "Index.db",
224            SSTableComponent::Summary => "Summary.db",
225            SSTableComponent::Filter => "Filter.db",
226            SSTableComponent::CompressionInfo => "CompressionInfo.db",
227            SSTableComponent::Statistics => "Statistics.db",
228            SSTableComponent::Digest => "Digest.crc32",
229            SSTableComponent::TOC => "TOC.txt",
230        }
231    }
232}

234/// SSTable file info extracted from path
235#[derive(Debug, Clone)]
236pub struct SSTableInfo {
237    pub format: SSTableFormat,
238    /// Raw SSTable identifier as a string.
239    ///
240    /// Cassandra 5.0 supports two id forms (Descriptor.java:95):
241    /// - Sequential integer: `"1"`, `"2"`, …
242    /// - UUID-based hex string: `"6aa08200a25111f0a3fef1a551383fb9"`
243    ///
244    /// This field stores the raw string so both forms round-trip correctly.
245    /// Use `generation_numeric()` when you need an integer id.
246    pub sstable_id: String,
247    /// The `<format>` name segment (`"big"` or `"bti"`).
248    pub size: String,
249    pub component: SSTableComponent,
250    pub base_name: String,
251}
252
253impl SSTableInfo {
254    /// Return the sequential generation number if the id is a plain integer.
255    ///
256    /// Returns `None` for UUID-based ids.
257    pub fn generation_numeric(&self) -> Option<u64> {
258        self.sstable_id.parse::<u64>().ok()
259    }
260
261    /// Parse SSTable info from file path.
262    ///
263    /// Accepts filenames with both sequential and UUID-based SSTable ids:
264    /// - `nb-1-big-Data.db`
265    /// - `nb-6aa08200a25111f0a3fef1a551383fb9-big-Data.db`
266    ///
267    /// The format segment (`big` / `bti`) is located by scanning right-to-left
268    /// from the end of the parts list, so multi-segment UUID ids do not
269    /// interfere with detection.
270    pub fn from_path(path: &Path) -> Result<Self> {
271        let filename = path
272            .file_name()
273            .and_then(|f| f.to_str())
274            .ok_or_else(|| Error::InvalidPath(format!("Invalid SSTable filename: {:?}", path)))?;
275
276        // Strip extension for splitting.
277        let base = if let Some(b) = filename.strip_suffix(".db") {
278            b
279        } else if let Some(b) = filename.strip_suffix(".txt") {
280            b
281        } else {
282            filename
283        };
284
285        let parts: Vec<&str> = base.split('-').collect();
286        if parts.len() < 4 {
287            return Err(Error::InvalidFormat(format!(
288                "Invalid SSTable filename format: {}",
289                filename
290            )));
291        }
292
293        let version = parts[0];
294
295        // Locate the format segment ("big"/"bti") by scanning right-to-left.
296        // Start at index 2 so we never confuse the version with the format.
297        let format_idx = parts[2..]
298            .iter()
299            .enumerate()
300            .rev()
301            .find(|(_, p)| **p == "big" || **p == "bti")
302            .map(|(i, _)| i + 2);
303
304        let format_idx = format_idx.ok_or_else(|| {
305            Error::InvalidFormat(format!(
306                "No 'big' or 'bti' format segment found in {:?}",
307                filename
308            ))
309        })?;
310
311        let size = parts[format_idx].to_string(); // "big" or "bti"
312
313        // id is everything between version and format
314        let sstable_id = parts[1..format_idx].join("-");
315
316        // component is everything after format, re-joined with original extension
317        let component_base = parts[format_idx + 1..].join("-");
318        let extension = if filename.ends_with(".db") {
319            ".db"
320        } else {
321            ".txt"
322        };
323        let component_filename = format!("-{}{}", component_base, extension);
324        let component = SSTableComponent::from_filename(&component_filename).ok_or_else(|| {
325            Error::InvalidFormat(format!(
326                "Unknown SSTable component {:?} in {:?}",
327                component_base, filename
328            ))
329        })?;
330
331        // base_name is <version>-<id>-<format>  (used to build sibling filenames)
332        let base_name = format!("{}-{}-{}", version, sstable_id, size);
333
334        let detector = FormatDetector::new();
335        let format = detector.detect_from_version(version)?;
336
337        Ok(SSTableInfo {
338            format,
339            sstable_id,
340            size,
341            component,
342            base_name,
343        })
344    }
345
346    /// Get path to companion component file
347    pub fn companion_path(
348        &self,
349        component: SSTableComponent,
350        base_dir: &Path,
351    ) -> std::path::PathBuf {
352        base_dir.join(format!("{}-{}", self.base_name, component.suffix()))
353    }
354}
355
356impl Default for SSTableInfo {
357    fn default() -> Self {
358        Self {
359            format: SSTableFormat::Unknown("unknown".to_string()),
360            sstable_id: "0".to_string(),
361            size: "unknown".to_string(),
362            component: SSTableComponent::Data,
363            base_name: "unknown".to_string(),
364        }
365    }
