1//! Backing store abstraction for SSTable block I/O.
2//!
3//! The SSTable reader accesses Data.db through a single seekable byte source.
4//! Historically this was always a buffered file handle (`BufReader<File>`).
5//! [`BlockSource`] generalizes that to two interchangeable backends:
6//!
7//! - [`BlockSource::Buffered`]: standard buffered async file I/O. Reads go
8//!   through the OS page cache with kernel read-ahead. This is the safe,
9//!   universally-portable default and is used for small files (where mmap
10//!   setup overhead is not worth it) or when mmap is disabled.
11//! - [`BlockSource::Mapped`]: a memory-mapped view of the file. The OS maps
12//!   the file into the process address space and serves reads straight from
13//!   the page cache with no per-block `read` syscall and no extra copy into a
14//!   buffered reader. This is well-suited to local SSTable analysis where the
15//!   same files are scanned repeatedly across queries (the page cache is
16//!   shared and reused), which is also the strategy Cassandra itself uses for
17//!   its read path (`disk_access_mode: mmap`).
18//!
19//! Both variants implement [`tokio::io::AsyncRead`] and
20//! [`tokio::io::AsyncSeek`], so every existing call site
21//! (`seek` / `stream_position` / `read` / `read_exact`) works against either
22//! backend without modification. The mmap backend is purely in-memory, so its
23//! poll methods always complete synchronously (`Poll::Ready`).
24

34/// A seekable byte source backing an [`SSTableReader`](super::types::SSTableReader).
35///
36/// See the module documentation for the trade-offs between the two backends.
37pub(crate) enum BlockSource {
38    /// Buffered file I/O through the OS page cache.
39    Buffered(BufReader<File>),
40    /// Memory-mapped file view served directly from the page cache.
41    Mapped(MmapCursor),
42}
43
44impl BlockSource {
45    /// Create a buffered (non-mmap) source from an open tokio file.
46    pub(crate) fn buffered(file: File) -> Self {
47        BlockSource::Buffered(BufReader::new(file))
48    }
49
50    /// Create a memory-mapped source from a previously mapped file.
51    pub(crate) fn mapped(mmap: Arc<Mmap>) -> Self {
52        BlockSource::Mapped(MmapCursor::new(mmap))
53    }
54
55    /// Returns `true` when this source is backed by a memory map.
56    #[cfg(test)]
57    pub(crate) fn is_mmap(&self) -> bool {
58        matches!(self, BlockSource::Mapped(_))
59    }
60}
61
62impl AsyncRead for BlockSource {
63    fn poll_read(
64        self: Pin<&mut Self>,
65        cx: &mut Context<'_>,
66        buf: &mut ReadBuf<'_>,
67    ) -> Poll<io::Result<()>> {
68        match self.get_mut() {
69            BlockSource::Buffered(r) => Pin::new(r).poll_read(cx, buf),
70            BlockSource::Mapped(c) => Pin::new(c).poll_read(cx, buf),
71        }
72    }
73}
74
75impl AsyncSeek for BlockSource {
76    fn start_seek(self: Pin<&mut Self>, position: SeekFrom) -> io::Result<()> {
77        match self.get_mut() {
78            BlockSource::Buffered(r) => Pin::new(r).start_seek(position),
79            BlockSource::Mapped(c) => Pin::new(c).start_seek(position),
80        }
81    }
82
83    fn poll_complete(self: Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<io::Result<u64>> {
84        match self.get_mut() {
85            BlockSource::Buffered(r) => Pin::new(r).poll_complete(cx),
86            BlockSource::Mapped(c) => Pin::new(c).poll_complete(cx),
87        }
88    }
89}

91/// An in-memory read cursor over a memory-mapped file.
92///
93/// Implements the same `AsyncRead`/`AsyncSeek` contract as a buffered file
94/// reader, but every operation completes immediately since the data is already
95/// resident (modulo lazy page faults handled transparently by the OS).
96pub(crate) struct MmapCursor {
97    mmap: Arc<Mmap>,
98    /// Current read position. May legally point past the end of the map, in
99    /// which case reads yield zero bytes (matching `File` EOF semantics).
100    pos: u64,
101}
102
103impl MmapCursor {
104    fn new(mmap: Arc<Mmap>) -> Self {
105        Self { mmap, pos: 0 }
106    }
107}
108
109impl AsyncRead for MmapCursor {
110    fn poll_read(
111        self: Pin<&mut Self>,
112        _cx: &mut Context<'_>,
113        buf: &mut ReadBuf<'_>,
114    ) -> Poll<io::Result<()>> {
115        let this = self.get_mut();
116        let data: &[u8] = &this.mmap;
117        let len = data.len() as u64;
118        // At or past EOF: read nothing and leave the position untouched. A real
119        // `File` preserves a seeked-past-EOF position across a zero-byte read,
120        // so we must not clamp `pos` back to `len` here (that divergence was the
121        // source-level parity bug). Returning zero bytes still drives
122        // `read_exact` to `UnexpectedEof`, which the block-header readers use to
123        // detect EOF.
124        if this.pos >= len {
125            return Poll::Ready(Ok(()));
126        }
127        let pos = this.pos as usize;
128        let remaining = &data[pos..];
129        let n = remaining.len().min(buf.remaining());
130        buf.put_slice(&remaining[..n]);
131        this.pos += n as u64;
132        Poll::Ready(Ok(()))
133    }
134}

136impl AsyncSeek for MmapCursor {
137    fn start_seek(self: Pin<&mut Self>, position: SeekFrom) -> io::Result<()> {
138        let this = self.get_mut();
139        let len = this.mmap.len() as u64;
140        let new_pos = match position {
141            SeekFrom::Start(offset) => offset,
142            SeekFrom::End(offset) => offset_from(len, offset)?,
143            SeekFrom::Current(offset) => offset_from(this.pos, offset)?,
144        };
145        this.pos = new_pos;
146        Ok(())
147    }
148
149    fn poll_complete(self: Pin<&mut Self>, _cx: &mut Context<'_>) -> Poll<io::Result<u64>> {
150        Poll::Ready(Ok(self.get_mut().pos))
151    }
152}

154/// Apply a signed offset to a base position, rejecting seeks before byte 0.
155///
156/// Seeking past the end is permitted (matching `std`/`tokio` `File` behaviour);
157/// subsequent reads simply return zero bytes.
158fn offset_from(base: u64, offset: i64) -> io::Result<u64> {
159    let result = if offset >= 0 {
160        // Overflowing the u64 address space is rejected rather than clamped, to
161        // mirror `std::fs::File`, which errors on a seek that overflows.
162        base.checked_add(offset as u64).ok_or_else(|| {
163            io::Error::new(
164                io::ErrorKind::InvalidInput,
165                "invalid seek to an overflowing position",
166            )
167        })?
168    } else {
169        base.checked_sub(offset.unsigned_abs()).ok_or_else(|| {
170            io::Error::new(
171                io::ErrorKind::InvalidInput,
172                "invalid seek to a negative position",
173            )
174        })?
175    };
176    Ok(result)
177}

184    fn cursor(bytes: &[u8]) -> MmapCursor {
185        // Build an anonymous memory map and copy bytes into it so tests do not
186        // depend on touching the filesystem.
187        let mut mmap = memmap2::MmapMut::map_anon(bytes.len().max(1)).unwrap();
188        mmap[..bytes.len()].copy_from_slice(bytes);
189        let mmap = mmap.make_read_only().unwrap();
190        MmapCursor::new(Arc::new(mmap))
191    }

198        assert_eq!(&buf, b"hello");

251    #[tokio::test]
252    async fn positive_seek_overflow_errors() {
253        // A positive offset that overflows the u64 address space is rejected
254        // with InvalidInput (matching `File`), not saturated to u64::MAX.
255        let mut c = cursor(b"abc");
256        // Park the cursor at the very top of the address space (seeking past EOF
257        // is legal), then a positive Current offset overflows.
258        c.seek(SeekFrom::Start(u64::MAX)).await.unwrap();
259        let err = c.seek(SeekFrom::Current(1)).await.unwrap_err();
260        assert_eq!(err.kind(), io::ErrorKind::InvalidInput);
261
262        // A large-but-non-overflowing positive offset still succeeds (parity:
263        // seeking past EOF is permitted, only overflow is an error).
264        let mut c2 = cursor(b"abc"); // len 3
265        let landed = c2.seek(SeekFrom::End(i64::MAX)).await.unwrap();
266        assert_eq!(landed, 3u64 + i64::MAX as u64);
267    }

269    #[tokio::test]
270    async fn seek_past_eof_preserves_position_like_file() {
271        // Parity with std/tokio `File`: seeking past EOF and issuing a read
272        // returns zero bytes WITHOUT collapsing the cursor back to EOF.
273        let mut c = cursor(b"abc"); // len 3
274        let landed = c.seek(SeekFrom::Start(10)).await.unwrap();
275        assert_eq!(landed, 10);
276
277        let mut b = [0u8; 4];
278        let n = c.read(&mut b).await.unwrap();
279        assert_eq!(n, 0, "read past EOF yields no bytes");
280
281        // The position must still be 10, exactly as a real File would report,
282        // not clamped down to the file length (3).
283        assert_eq!(c.stream_position().await.unwrap(), 10);
284
285        // Seeking back to a valid offset still reads correctly afterwards.
286        c.seek(SeekFrom::Start(1)).await.unwrap();
287        let mut one = [0u8; 1];
288        c.read_exact(&mut one).await.unwrap();
289        assert_eq!(&one, b"b");
290    }

319    #[tokio::test]
320    async fn multipage_read_across_page_boundary() {
321        // Exercise a map larger than a single OS page and read straddling the
322        // 4096-byte boundary, ensuring slicing/position math holds beyond one
323        // page.
324        let len = 10_000usize;
325        let mut data = vec![0u8; len];
326        for (i, b) in data.iter_mut().enumerate() {
327            *b = (i % 251) as u8; // deterministic, spans the page boundary
328        }
329        let mut c = cursor(&data);
330
331        // Read a window that starts before page 1 and ends after it.
332        c.seek(SeekFrom::Start(4090)).await.unwrap();
333        let mut window = [0u8; 16]; // covers 4090..4106, crossing 4096
334        c.read_exact(&mut window).await.unwrap();
335        for (k, b) in window.iter().enumerate() {
336            assert_eq!(*b, ((4090 + k) % 251) as u8);
337        }
338        assert_eq!(c.stream_position().await.unwrap(), 4106);
339
340        // Read the final bytes right up to EOF.
341        c.seek(SeekFrom::Start((len - 4) as u64)).await.unwrap();
342        let mut tail = [0u8; 4];
343        c.read_exact(&mut tail).await.unwrap();
344        for (k, b) in tail.iter().enumerate() {
345            assert_eq!(*b, ((len - 4 + k) % 251) as u8);
346        }
347        assert_eq!(c.stream_position().await.unwrap(), len as u64);
348    }
