1//! BTI (Big Trie Index) parser implementation
2//!
3//! This module provides parsing capabilities for BTI format components:
4//! - Partitions.db BTI index for partition lookups
5//! - Rows.db BTI index for clustering key lookups within partitions
6//!
7//! ## Node-type encoding (TrieNode.java, Cassandra 5.0)
8//!
9//! The high nibble (bits 7-4) of every node's first byte is a 4-bit ordinal that
10//! selects one of up to 16 concrete trie-node subtypes:
11//!
12//! | Nibble | Java class          | Rust category    | Pointer size |
13//! |--------|---------------------|------------------|--------------|
14//! | 0      | PayloadOnly         | `PayloadOnly`    | —            |
15//! | 1      | SingleNoPayload4    | `Single`         | 4-bit delta  |
16//! | 2      | Single8             | `Single`         | 1 byte       |
17//! | 3      | SingleNoPayload12   | `Single`         | 12-bit delta |
18//! | 4      | Single16            | `Single`         | 2 bytes      |
19//! | 5      | Sparse8             | `Sparse`         | 1 byte       |
20//! | 6      | Sparse12            | `Sparse`         | 12-bit       |
21//! | 7      | Sparse16            | `Sparse`         | 2 bytes      |
22//! | 8      | Sparse24            | `Sparse`         | 3 bytes      |
23//! | 9      | Sparse40            | `Sparse`         | 5 bytes      |
24//! | 10     | Dense12             | `Dense`          | 12-bit       |
25//! | 11     | Dense16             | `Dense`          | 2 bytes      |
26//! | 12     | Dense24             | `Dense`          | 3 bytes      |
27//! | 13     | Dense32             | `Dense`          | 4 bytes      |
28//! | 14     | Dense40             | `Dense`          | 5 bytes      |
29//! | 15     | LongDense           | `Dense`          | 8 bytes      |
30//!
31//! The low nibble (bits 3-0) carries payload flags; a non-zero value means a
32//! payload follows immediately after the node's fixed-size header.
33//!
34//! All transition deltas are stored as *backward* distances (the child always
35//! appears earlier in the file), so the absolute child position is:
36//!   `child_pos = parent_pos - delta`
37//!
38//! The "no-payload" Single variants (nibbles 1 and 3) encode the delta in the
39//! low nibble / low 12 bits of the first two bytes respectively, and cannot
40//! carry a payload (their payload-flag nibble is always 0).

60/// Classify the high nibble of a BTI node's first byte into one of the four
61/// Rust-level node categories.
62///
63/// The mapping follows the 16-entry `Types.values[]` array in `TrieNode.java`:
64/// - nibble 0               → `PayloadOnly`
65/// - nibbles 1-4 (Singles)  → `Single`
66/// - nibbles 5-9 (Sparses)  → `Sparse`
67/// - nibbles 10-15 (Denses) → `Dense`
68fn classify_node_nibble(nibble: u8) -> BtiResult<BtiNodeType> {
69    match nibble {
70        0 => Ok(BtiNodeType::PayloadOnly),
71        1..=4 => Ok(BtiNodeType::Single),
72        5..=9 => Ok(BtiNodeType::Sparse),
73        10..=15 => Ok(BtiNodeType::Dense),
74        other => Err(Error::Parse(format!(
75            "Invalid BTI node type nibble: {}",
76            other
77        ))),
78    }
79}

81/// Return the number of bytes used to store each child pointer for this node,
82/// given the raw high nibble (`ordinal`) from the first byte.
83///
84/// Matches the `bytesPerPointer` field of each `TrieNode` subtype in
85/// `TrieNode.java`.  Fractional (12-bit) encodings return `0` as a sentinel;
86/// callers that need to handle those cases do so explicitly.
87fn pointer_bytes_for_ordinal(ordinal: u8) -> u8 {
88    match ordinal {
89        0 => 0,  // PayloadOnly — no pointers
90        1 => 0,  // SingleNoPayload4  — 4-bit delta in low nibble, handled specially
91        2 => 1,  // Single8
92        3 => 0,  // SingleNoPayload12 — 12-bit delta across first two bytes
93        4 => 2,  // Single16
94        5 => 1,  // Sparse8
95        6 => 0,  // Sparse12 — 12-bit packed
96        7 => 2,  // Sparse16
97        8 => 3,  // Sparse24
98        9 => 5,  // Sparse40
99        10 => 0, // Dense12  — 12-bit packed
100        11 => 2, // Dense16
101        12 => 3, // Dense24
102        13 => 4, // Dense32
103        14 => 5, // Dense40
104        15 => 8, // LongDense
105        _ => 0,
106    }
107}

109/// Parse a BTI trie node from a raw byte slice.
110///
111/// This is the **single shared implementation** used by both
112/// [`PartitionsParser`] and [`RowsParser`].  It correctly dispatches all 16
113/// node-type ordinals defined in `TrieNode.java` and produces the appropriate
114/// [`BtiNode`] variant.
115///
116/// # Arguments
117/// * `data`   – raw bytes starting at the node's first byte
118/// * `offset` – absolute file offset of this node (used to compute child
119///   positions; see the module-level doc for the sign convention)
120///
121/// # Errors
122/// Returns a [`crate::error::Error::Parse`] if the data is too short, the
123/// node-type nibble is out of range, or any other structural invariant is
124/// violated.
125fn parse_bti_node(data: &[u8], offset: u64) -> BtiResult<BtiNode> {
126    if data.is_empty() {
127        return Err(Error::Parse("Empty BTI node data".to_string()));
128    }
129
130    let header_byte = data[0];
131    let ordinal = (header_byte >> 4) & 0x0F;
132    let payload_flags = header_byte & 0x0F;
133    let has_payload = payload_flags != 0;
134    let node_type = classify_node_nibble(ordinal)?;
135
136    match node_type {
137        BtiNodeType::PayloadOnly => {
138            // Layout: [header:1] [payload if has_payload]
139            // PayloadOnly MUST have a payload (it is a leaf node).
140            if !has_payload {
141                return Err(Error::Parse(
142                    "PayloadOnly node has no payload flags set".to_string(),
143                ));
144            }
145            let payload = parse_payload_ref(&data[1..])?;
146            Ok(BtiNode {
147                node_type,
148                level: 0,
149                key_prefix: Vec::new(),
150                data: BtiNodeData::PayloadOnly { payload },
151            })
152        }
153
154        BtiNodeType::Single => {
155            // Three concrete layouts depending on ordinal:
156            //
157            //   ordinal 1 (SingleNoPayload4):
158            //     byte 0: [1|delta_high4]  (delta is low 4 bits of byte 0)
159            //     byte 1: transition byte
160            //     NO payload
161            //
162            //   ordinal 3 (SingleNoPayload12):
163            //     byte 0: [3|delta_high4]
164            //     byte 1: delta_low8        → delta = (low4(byte0) << 8) | byte1
165            //     byte 2: transition byte
166            //     NO payload
167            //
168            //   ordinals 2, 4 (Single8, Single16):
169            //     byte 0: [ordinal|payload_flags]
170            //     byte 1: transition byte
171            //     bytes 2..(2+ptr_bytes): backward delta (unsigned big-endian)
172            //     [payload if has_payload]
173            match ordinal {
174                1 => {
175                    // SingleNoPayload4
176                    if data.len() < 2 {
177                        return Err(Error::Parse(
178                            "SingleNoPayload4 node data too short".to_string(),
179                        ));
180                    }
181                    let delta = (header_byte & 0x0F) as u64;
182                    let transition_byte = data[1];
183                    let child_offset = offset.saturating_sub(delta);
184                    Ok(BtiNode {
185                        node_type,
186                        level: 1,
187                        key_prefix: Vec::new(),
188                        data: BtiNodeData::Single {
189                            transition: Transition::new(
190                                transition_byte,
191                                SizedPointer::new(child_offset),
192                            ),
193                        },
194                    })
195                }
196                3 => {
197                    // SingleNoPayload12
198                    if data.len() < 3 {
199                        return Err(Error::Parse(
200                            "SingleNoPayload12 node data too short".to_string(),
201                        ));
202                    }
203                    let delta = (((header_byte & 0x0F) as u64) << 8) | (data[1] as u64);
204                    let transition_byte = data[2];
205                    let child_offset = offset.saturating_sub(delta);
206                    Ok(BtiNode {
207                        node_type,
208                        level: 1,
209                        key_prefix: Vec::new(),
210                        data: BtiNodeData::Single {
211                            transition: Transition::new(
212                                transition_byte,
213                                SizedPointer::new(child_offset),
214                            ),
215                        },
216                    })
217                }
218                _ => {
219                    // Single8 (ordinal 2) or Single16 (ordinal 4)
220                    let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
221                    let needed = 2 + ptr_bytes;
222                    if data.len() < needed {
223                        return Err(Error::Parse(format!(
224                            "Single node (ordinal {}) data too short: need {} bytes, have {}",
225                            ordinal,
226                            needed,
227                            data.len()
228                        )));
229                    }
230                    let transition_byte = data[1];
231                    let delta = read_be_unsigned(&data[2..2 + ptr_bytes]);
232                    let child_offset = offset.saturating_sub(delta);
233                    let transition =
234                        Transition::new(transition_byte, SizedPointer::new(child_offset));
235                    Ok(BtiNode {
236                        node_type,
237                        level: 1,
238                        key_prefix: Vec::new(),
239                        data: BtiNodeData::Single { transition },
240                    })
241                }
242            }
243        }
244
245        BtiNodeType::Sparse => {
246            // Layout (ordinals 5, 7, 8, 9 — full-byte pointers):
247            //   byte 0: [ordinal|payload_flags]
248            //   byte 1: count (number of transitions, 1-255)
249            //   bytes 2..(2+count): transition bytes (sorted)
250            //   then count × ptr_bytes: backward deltas
251            //   [payload if has_payload]
252            //
253            // ordinal 6 (Sparse12) packs two 12-bit deltas into 3 bytes;
254            // that encoding is rare in practice and we decode it correctly
255            // but store the absolute child offsets in the same SizedPointer.
256            if data.len() < 2 {
257                return Err(Error::Parse("Sparse node data too short".to_string()));
258            }
259            let count = data[1] as usize;
260            if count == 0 {
261                return Err(Error::Parse(
262                    "Sparse node must have at least one transition".to_string(),
263                ));
264            }
265
266            let bytes_start = 2;
267            let pointers_start = bytes_start + count;
268
269            if ordinal == 6 {
270                // Sparse12: each pair of pointers packed into 3 bytes
271                let packed_len = (count * 3).div_ceil(2); // ceil(count * 12 / 8) = ceil(count * 3 / 2)
272                let needed = pointers_start + packed_len;
273                if data.len() < needed {
274                    return Err(Error::Parse(format!(
275                        "Sparse12 node data too short: need {}, have {}",
276                        needed,
277                        data.len()
278                    )));
279                }
280                let mut transitions = Vec::with_capacity(count);
281                for i in 0..count {
282                    let t_byte = data[bytes_start + i];
283                    let delta = read_12bit_packed(&data[pointers_start..], i);
284                    let child_offset = offset.saturating_sub(delta);
285                    transitions.push(Transition::new(t_byte, SizedPointer::new(child_offset)));
286                }
287                Ok(BtiNode {
288                    node_type,
289                    level: 1,
290                    key_prefix: Vec::new(),
291                    data: BtiNodeData::Sparse { transitions },
292                })
293            } else {
294                let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
295                let needed = pointers_start + count * ptr_bytes;
296                if data.len() < needed {
297                    return Err(Error::Parse(format!(
298                        "Sparse node (ordinal {}) data too short: need {}, have {}",
299                        ordinal,
300                        needed,
301                        data.len()
302                    )));
303                }
304                let mut transitions = Vec::with_capacity(count);
305                for i in 0..count {
306                    let t_byte = data[bytes_start + i];
307                    let ptr_off = pointers_start + i * ptr_bytes;
308                    let delta = read_be_unsigned(&data[ptr_off..ptr_off + ptr_bytes]);
309                    let child_offset = offset.saturating_sub(delta);
310                    transitions.push(Transition::new(t_byte, SizedPointer::new(child_offset)));
311                }
312                Ok(BtiNode {
313                    node_type,
314                    level: 1,
315                    key_prefix: Vec::new(),
316                    data: BtiNodeData::Sparse { transitions },
317                })
318            }
319        }
320
321        BtiNodeType::Dense => {
322            // Layout (ordinals 11-14 — full-byte pointers):
323            //   byte 0: [ordinal|payload_flags]
324            //   byte 1: start byte (first transition character)
325            //   byte 2: (end - start), i.e. (range_len - 1)  → range_len = byte2+1
326            //   then range_len × ptr_bytes: backward deltas; 0 means "no transition"
327            //   [payload if has_payload]
328            //
329            // ordinal 10 (Dense12): packed 12-bit deltas, similar to Sparse12.
330            // ordinal 15 (LongDense): 8-byte deltas.
331            if data.len() < 3 {
332                return Err(Error::Parse("Dense node data too short".to_string()));
333            }
334            let start_byte = data[1];
335            let range_len = data[2] as usize + 1; // byte2 is (len-1)
336
337            if ordinal == 10 {
338                // Dense12: ceil(range_len * 12 / 8) = (range_len * 3 + 1) / 2
339                let packed_len = (range_len * 3).div_ceil(2);
340                let needed = 3 + packed_len;
341                if data.len() < needed {
342                    return Err(Error::Parse(format!(
343                        "Dense12 node data too short: need {}, have {}",
344                        needed,
345                        data.len()
346                    )));
347                }
348                let mut children = Vec::with_capacity(range_len);
349                for i in 0..range_len {
350                    let delta = read_12bit_packed(&data[3..], i);
351                    // delta == 0 is the "no transition" sentinel → None.  Any
352                    // other delta is a real child at `offset - delta` (which may
353                    // be absolute offset 0 — a legitimate child, distinct from
354                    // the sentinel).
355                    let child = if delta == 0 {
356                        None
357                    } else {
358                        Some(SizedPointer::new(offset.saturating_sub(delta)))
359                    };
360                    children.push(child);
361                }
362                Ok(BtiNode {
363                    node_type,
364                    level: 1,
365                    key_prefix: Vec::new(),
366                    data: BtiNodeData::Dense {
367                        start_byte,
368                        children,
369                    },
370                })
371            } else {
372                let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
373                let needed = 3 + range_len * ptr_bytes;
374                if data.len() < needed {
375                    return Err(Error::Parse(format!(
376                        "Dense node (ordinal {}) data too short: need {}, have {}",
377                        ordinal,
378                        needed,
379                        data.len()
380                    )));
381                }
382                let mut children = Vec::with_capacity(range_len);
383                for i in 0..range_len {
384                    let ptr_off = 3 + i * ptr_bytes;
385                    let delta = read_be_unsigned(&data[ptr_off..ptr_off + ptr_bytes]);
386                    // delta == 0 is the "no transition" sentinel → None.  Any
387                    // other delta is a real child at `offset - delta` (which may
388                    // be absolute offset 0 — a legitimate child, distinct from
389                    // the sentinel).
390                    let child = if delta == 0 {
391                        None
392                    } else {
393                        Some(SizedPointer::new(offset.saturating_sub(delta)))
394                    };
395                    children.push(child);
396                }
397                Ok(BtiNode {
398                    node_type,
399                    level: 1,
400                    key_prefix: Vec::new(),
401                    data: BtiNodeData::Dense {
402                        start_byte,
403                        children,
404                    },
405                })
406            }
407        }
408    }

411/// Read a big-endian unsigned integer of 0-8 bytes from `data`.
412///
413/// Returns 0 for an empty slice; panics if `data.len() > 8` (caller is
414/// responsible for bounds-checking first).
415fn read_be_unsigned(data: &[u8]) -> u64 {
416    let mut result = 0u64;
417    for &byte in data {
418        result = (result << 8) | (byte as u64);
419    }
420    result
421}

423/// Read a 12-bit value stored in the packed format used by Sparse12 / Dense12.
424///
425/// The packing (from `TrieNode.java#read12Bits`):
426/// ```text
427/// word = getShort(base + (3 * index) / 2)
428/// if (index & 1) == 0:  value = word >> 4
429/// else:                  value = word & 0xFFF
430/// ```
431fn read_12bit_packed(data: &[u8], index: usize) -> u64 {
432    let byte_offset = (3 * index) / 2;
433    if byte_offset + 1 >= data.len() {
434        return 0;
435    }
436    let word = ((data[byte_offset] as u16) << 8) | (data[byte_offset + 1] as u16);
437    let value = if (index & 1) == 0 {
438        word >> 4
439    } else {
440        word & 0x0FFF
441    };
442    value as u64
443}

445/// Parse a [`PayloadRef`] from the bytes immediately following a node header.
446///
447/// Format (matches `RowIndexReader.readPayload` / `PartitionIndex`):
448///   8 bytes big-endian: data file offset
449///   4 bytes big-endian: payload byte length
450fn parse_payload_ref(data: &[u8]) -> BtiResult<PayloadRef> {
451    if data.len() < 12 {
452        return Err(Error::Parse(format!(
453            "PayloadRef data too short: need 12 bytes, have {}",
454            data.len()
455        )));
456    }
457    let offset = u64::from_be_bytes([
458        data[0], data[1], data[2], data[3], data[4], data[5], data[6], data[7],
459    ]);
460    let length = u32::from_be_bytes([data[8], data[9], data[10], data[11]]);
461    Ok(PayloadRef::new(offset, length))
462}

464// ─────────────────────────────────────────────────────────────────────────────
465// Real Cassandra BTI Partitions.db trie-payload decoding
466// ─────────────────────────────────────────────────────────────────────────────
467//
468// Cassandra 5.0 `Partitions.db` has NO header.  The file is a compact trie
469// whose nodes are written bottom-up (children appear at lower offsets than
470// parents), with the root offset stored as the **last 8 bytes** of the file as
471// a big-endian u64.
472//
473// Every leaf node (ordinal 0 = PayloadOnly) carries a *payload* described by
474// the `payloadBits` value encoded in the **low nibble** of the node's header
475// byte.  In Cassandra 5.0 the hash byte is **always present**
476// (`FLAG_HAS_HASH_BYTE = 8`), so `payloadBits >= 8` for every real partition
477// leaf.  The payload layout is:
478//
479//   byte 0            : partition-key filter hash (lowest 8 bits of Murmur3)
480//   bytes 1 .. N      : SizedInts-encoded signed `position`, where
481//                        N = payloadBits − 8 + 1  (= payloadBits − 7)
482//
483// **Sign convention** (mirrors `PartitionIndex.java`):
484//   - If `position < 0` → `data_offset = ~position` (bitwise NOT; integer NOT)
485//   - If `position >= 0` → `position` points into `Rows.db` (wide partition)
486//
487// Reference: `PartitionIndex.java:131–135`, `BtiFormat.md:946–963`,
488//            `SizedInts.java` (local mirror: `sized_ints.rs`).
489// ─────────────────────────────────────────────────────────────────────────────
490
491/// The `FLAG_HAS_HASH_BYTE` bit in `payloadBits` (low nibble of a BTI leaf
492/// node's header byte).  When set, the first payload byte is the partition-key
493/// filter hash; the remaining bytes encode the `position` via `SizedInts`.
494///
495/// In Cassandra 5.0 this bit is **always** set for partition index leaves.
496/// Mirrors `PartitionIndex.java:131`.
497pub const FLAG_HAS_HASH_BYTE: u8 = 8;
498
499/// Result of decoding a BTI partition leaf payload.
500///
501/// Mirrors `PartitionIndex.java` payload interpretation: either a direct
502/// `Data.db` offset (for narrow partitions) or a `Rows.db` offset (for wide
503/// partitions that need row-level indexing).
504#[derive(Debug, Clone, PartialEq, Eq)]
505pub enum BtiPartitionLocation {
506    /// Direct `Data.db` byte offset.  The partition starts here.
507    ///
508    /// Occurs when the trie leaf's encoded position is **negative**:
509    /// `data_offset = ~position` (bitwise NOT).
510    DataOffset(u64),
511
512    /// Offset into `Rows.db` for this partition's row-level trie index.
513    ///
514    /// Occurs when the trie leaf's encoded position is **non-negative**.
515    RowsOffset(u64),
516}

518/// Decode the BTI partition leaf payload at `payload_start` inside `trie_data`.
519///
520/// # Arguments
521/// * `trie_data`      – raw bytes of the trie (i.e. the file **excluding** the
522///   8-byte root-offset footer).
523/// * `payload_start`  – offset within `trie_data` where the payload begins
524///   (immediately after the 1-byte node header).
525/// * `payload_bits`   – the `payloadBits` value from the node header's low nibble.
526///
527/// # Returns
528/// `Ok(BtiPartitionLocation)` on success; a parse error if `payload_bits` is
529/// out of the expected range or the data is too short.
530///
531/// # Reference
532/// Mirrors `PartitionIndex.java` `readPayload` and Cassandra's `SizedInts.read`.
533pub fn decode_bti_partition_payload(
534    trie_data: &[u8],
535    payload_start: usize,
536    payload_bits: u8,
537) -> BtiResult<BtiPartitionLocation> {
538    // Every real Cassandra 5.0 partition leaf must have the hash byte flag set.
539    if payload_bits < FLAG_HAS_HASH_BYTE {
540        return Err(Error::Parse(format!(
541            "BTI payload_bits {payload_bits} < FLAG_HAS_HASH_BYTE (8); \
542             hash-byte-less payloads are not supported in Cassandra 5.0 BTI format"
543        )));
544    }
545    if payload_bits > 16 {
546        // SizedInts stores at most 8 bytes; plus 1 hash byte = 9 bytes max → payloadBits ≤ 16
547        return Err(Error::Parse(format!(
548            "BTI payload_bits {payload_bits} > 16; invalid BTI partition leaf"
549        )));
550    }
551
552    // N = payloadBits − FLAG_HAS_HASH_BYTE + 1
553    // For payloadBits = 8 (most common: 1-byte position): N = 1
554    // For payloadBits = 9: N = 2, etc.
555    let position_bytes = (payload_bits - FLAG_HAS_HASH_BYTE + 1) as usize;
556
557    // Minimum payload = 1 hash byte + position_bytes
558    let needed = 1 + position_bytes;
559    if payload_start + needed > trie_data.len() {
560        return Err(Error::Parse(format!(
561            "BTI partition payload at {payload_start} is too short: \
562             need {needed} bytes, have {}",
563            trie_data.len().saturating_sub(payload_start)
564        )));
565    }
566
567    // Skip hash byte (payload_start + 0), read position starting at payload_start + 1
568    let pos_data = &trie_data[payload_start + 1..payload_start + 1 + position_bytes];
569
570    // Sign-extend the big-endian bytes into a signed i64 (SizedInts semantics)
571    let position: i64 = sized_ints_read_from_slice(pos_data)?;
572
573    // Sign convention: negative → ~position is the Data.db byte offset
574    if position < 0 {
575        let data_offset = !position as u64; // bitwise NOT of the negative value
576        Ok(BtiPartitionLocation::DataOffset(data_offset))
577    } else {
578        Ok(BtiPartitionLocation::RowsOffset(position as u64))
579    }
580}

582/// Read a signed big-endian SizedInts value from a byte slice.
583///
584/// Mirrors `SizedInts.read(DataInputPlus, int)` from Cassandra — sign-extends
585/// the most-significant byte.  Accepts 1–8 bytes.
586///
587/// This is the slice-based analogue of `crate::storage::sstable::bti::sized_ints::read`
588/// (which operates on a `std::io::Read` stream).
589fn sized_ints_read_from_slice(data: &[u8]) -> BtiResult<i64> {
590    match data.len() {
591        0 => Ok(0),
592        1 => Ok(data[0] as i8 as i64),
593        2 => Ok(i16::from_be_bytes([data[0], data[1]]) as i64),
594        3 => {
595            let high = data[0] as i8 as i64;
596            let low = u16::from_be_bytes([data[1], data[2]]) as i64;
597            Ok((high << 16) | low)
598        }
599        4 => Ok(i32::from_be_bytes([data[0], data[1], data[2], data[3]]) as i64),
600        5 => {
601            let high = data[0] as i8 as i64;
602            let low = u32::from_be_bytes([data[1], data[2], data[3], data[4]]) as i64;
603            Ok((high << 32) | low)
604        }
605        6 => {
606            let high = i16::from_be_bytes([data[0], data[1]]) as i64;
607            let low = u32::from_be_bytes([data[2], data[3], data[4], data[5]]) as i64;
608            Ok((high << 32) | low)
609        }
610        7 => {
611            let high1 = data[0] as i8 as i64;
612            let high2 = u16::from_be_bytes([data[1], data[2]]) as i64;
613            let low = u32::from_be_bytes([data[3], data[4], data[5], data[6]]) as i64;
614            Ok((high1 << 48) | (high2 << 32) | low)
615        }
616        8 => Ok(i64::from_be_bytes([
617            data[0], data[1], data[2], data[3], data[4], data[5], data[6], data[7],
618        ])),
619        n => Err(Error::Parse(format!(
620            "SizedInts: invalid byte count {n} (expected 1–8)"
621        ))),
622    }
623}

625/// Parse a BTI trie node from `trie_data` at `node_offset` for **traversal**
626/// purposes (i.e. to follow child pointers).  Unlike [`parse_bti_node`], this
627/// function does NOT require any particular payload format: for PayloadOnly and
628/// for nodes with a non-zero `payloadBits` low nibble the payload bytes are
629/// simply skipped — the caller is responsible for reading the payload via
630/// [`read_node_payload`].
631///
632/// This is important because real Cassandra 5.0 `Partitions.db` uses a
633/// compact SizedInts-based payload (2–10 bytes) that is incompatible with the
634/// legacy 12-byte `PayloadRef` format expected by [`parse_payload_ref`].
635fn parse_bti_node_for_traversal(trie_data: &[u8], node_offset: usize) -> BtiResult<BtiNode> {
636    if node_offset >= trie_data.len() {
637        return Err(Error::Parse(format!(
638            "BTI traversal: node_offset {node_offset} >= trie_data.len {}",
639            trie_data.len()
640        )));
641    }
642
643    let data = &trie_data[node_offset..];
644    let header_byte = data[0];
645    let ordinal = (header_byte >> 4) & 0x0F;
646    let payload_flags = header_byte & 0x0F;
647    let node_type = classify_node_nibble(ordinal)?;
648
649    match node_type {
650        BtiNodeType::PayloadOnly => {
651            // PayloadOnly leaf: no children.  Skip payload (we decode it separately).
652            if payload_flags == 0 {
653                return Err(Error::Parse(
654                    "PayloadOnly node has no payload flags set".to_string(),
655                ));
656            }
657            // Use a stub PayloadRef — the caller uses read_node_payload for the real data
658            let stub = PayloadRef::new(0, 0);
659            Ok(BtiNode {
660                node_type,
661                level: 0,
662                key_prefix: Vec::new(),
663                data: BtiNodeData::PayloadOnly { payload: stub },
664            })
665        }
666        BtiNodeType::Single => {
667            // Reuse the existing full parser — it handles pointers correctly.
668            // Payload (if any) comes after the pointer bytes and is not read here.
669            parse_bti_node(data, node_offset as u64)
670        }
671        BtiNodeType::Sparse => parse_bti_node(data, node_offset as u64),
672        BtiNodeType::Dense => parse_bti_node(data, node_offset as u64),
673    }
674}

676/// Walk one BTI trie node at `node_offset` and return the absolute offset of
677/// the child reachable via `search_byte`, or `None` if no such transition
678/// exists.
679///
680/// Returns `Ok(Some(child_offset))` when the byte is found and the child is an
681/// internal node.  Returns `Ok(None)` when there is no matching transition
682/// (key not in trie).
683///
684/// Separately, if the *current* node (at `node_offset`) has an embedded
685/// payload **and** `key_exhausted` is true, the caller should read that
686/// payload; this function does not return it — see [`read_node_payload`].
687fn find_next_child_offset(
688    trie_data: &[u8],
689    node_offset: usize,
690    search_byte: u8,
691) -> BtiResult<Option<usize>> {
692    if node_offset >= trie_data.len() {
693        return Err(Error::Parse(format!(
694            "BTI node offset {node_offset} out of bounds (trie_data.len={})",
695            trie_data.len()
696        )));
697    }
698
699    let bti_node = parse_bti_node_for_traversal(trie_data, node_offset)?;
700
701    match bti_node.find_child(search_byte) {
702        Some(ptr) => {
703            let child = ptr.distance as usize; // distance stores the absolute child offset
704            Ok(Some(child))
705        }
706        None => Ok(None),
707    }

710/// Read the `BtiPartitionLocation` from the payload attached to the BTI node
711/// at `node_offset`.  Returns `None` if the node has no payload.
712///
713/// For `PayloadOnly` nodes the payload immediately follows the 1-byte header.
714/// For other node types (Single/Sparse/Dense with a non-zero `payloadBits`
715/// low nibble) the payload appears *after* all the pointer/transition bytes;
716/// this function delegates correctly to [`decode_bti_partition_payload`].
717fn read_node_payload(
718    trie_data: &[u8],
719    node_offset: usize,
720) -> BtiResult<Option<BtiPartitionLocation>> {
721    if node_offset >= trie_data.len() {
722        return Err(Error::Parse(format!(
723            "BTI payload read: node_offset {node_offset} out of bounds"
724        )));
725    }
726
727    let header_byte = trie_data[node_offset];
728    let ordinal = (header_byte >> 4) & 0x0F;
729    let payload_flags = header_byte & 0x0F; // aka payloadBits
730
731    // Ordinals 1 (SingleNoPayload4) and 3 (SingleNoPayload12) encode their
732    // backward delta in the low nibble / low 12 bits — that nibble is NOT a
733    // payload flag, and these node types can NEVER carry a payload.  Treating
734    // their low nibble as `payloadBits` was a latent bug exposed by full DFS
735    // (issue #832): e.g. a `0x18` SingleNoPayload4 was misread as having a
736    // payload, producing a spurious entry.
737    if ordinal == 1 || ordinal == 3 {
738        return Ok(None);
739    }
740
741    if ordinal == 0 {
742        // PayloadOnly node: the entire node is just [header][payload…]
743        if payload_flags == 0 {
744            return Err(Error::Parse(
745                "PayloadOnly node has zero payload_flags".to_string(),
746            ));
747        }
748        let payload_start = node_offset + 1; // immediately after the 1-byte header
749        Ok(Some(decode_bti_partition_payload(
750            trie_data,
751            payload_start,
752            payload_flags,
753        )?))
754    } else if payload_flags != 0 {
755        // Non-leaf node with an embedded payload (prefix match for short keys).
756        // We need to skip over the node's own pointer/transition data to reach
757        // the payload.  Reuse `parse_bti_node` to find the payload start.
758        //
759        // NOTE: `parse_bti_node` reads its header from `data[0]`, so it must be
760        // passed the slice STARTING at the node (`&trie_data[node_offset..]`),
761        // while the absolute `node_offset` drives child-position arithmetic.
762        // Passing the whole `trie_data` here was a latent bug (it parsed the
763        // node at offset 0 instead) — only exposed once a non-leaf node carries
764        // an embedded payload at a non-zero offset (issue #832).
765        let node = parse_bti_node(&trie_data[node_offset..], node_offset as u64)?;
766        // Determine where the payload bytes start: after all transition/pointer bytes.
767        // payload_position = node_offset + node_byte_size_without_payload
768        let payload_start = payload_start_in_node(&node, trie_data, node_offset)?;
769        Ok(Some(decode_bti_partition_payload(
770            trie_data,
771            payload_start,
772            payload_flags,
773        )?))
774    } else {
775        Ok(None)
776    }
777}

779/// Compute the byte offset of the payload within `trie_data` for a non-leaf
780/// node that carries an embedded payload (`payloadBits != 0`).
781///
782/// This is the byte position immediately after all the node's transition bytes
783/// and pointer bytes, but before any subsequent node data.
784fn payload_start_in_node(node: &BtiNode, trie_data: &[u8], node_offset: usize) -> BtiResult<usize> {
785    use BtiNodeData::*;
786    let header_byte = trie_data[node_offset];
787    let ordinal = (header_byte >> 4) & 0x0F;
788
789    let payload_offset = match &node.data {
790        PayloadOnly { .. } => {
791            // Header(1) + payload: caller handles this case separately
792            node_offset + 1
793        }
794        Single { .. } => {
795            // Singles with payload: header(1) + transition(1) + ptr_bytes
796            let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
797            // Ordinals 1 and 3 are "NoPayload" variants; they cannot have payloads
798            // (the payload_flags nibble is always 0 for those ordinals, so this
799            // branch is unreachable for them).
800            node_offset + 1 + 1 + ptr_bytes
801        }
802        Sparse { transitions } => {
803            let count = transitions.len();
804            let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
805            // Ordinal 6 (Sparse12): packed 12-bit pointers, ceil(count*3/2) bytes
806            let ptr_area = if ordinal == 6 {
807                (count * 3).div_ceil(2)
808            } else {
809                count * ptr_bytes
810            };
811            node_offset + 1 + 1 + count + ptr_area // header + count + transitions + deltas
812        }
813        Dense { children, .. } => {
814            let range_len = children.len();
815            let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
816            let ptr_area = if ordinal == 10 {
817                // Dense12
818                (range_len * 3).div_ceil(2)
819            } else {
820                range_len * ptr_bytes
821            };
822            node_offset + 1 + 1 + 1 + ptr_area // header + start_byte + len-1 + deltas
823        }
824    };
825    Ok(payload_offset)
826}

828/// Walk the BTI partition trie loaded in `trie_data`, looking for a partition
829/// whose **byte-comparable** encoding exactly matches `encoded_key`.
830///
831/// Returns:
832/// - `Ok(Some(BtiPartitionLocation))` when the key is found.
833/// - `Ok(None)` when the key is not present in the trie.
834/// - `Err(_)` on structural parse errors.
835///
836/// This is the inner engine for [`lookup_partition_in_bti_file`].
837/// Walk the BTI partition trie loaded in `trie_data`, looking for a partition
838/// whose **byte-comparable** encoded key *prefix* routes to a leaf.
839///
840/// Cassandra BTI uses a **path-compressed (Patricia) trie**: once a path leads to
841/// a `PayloadOnly` leaf, the remaining key bytes are stored implicitly (as the
842/// compressed suffix).  Therefore a match is declared as soon as a `PayloadOnly`
843/// leaf is reached — regardless of whether the encoded key has been fully consumed.
844/// The caller is responsible for verifying the actual partition key against the
845/// Data.db bytes at the resolved offset (using the partition's hash byte for a
846/// fast pre-filter and the raw key bytes for definitive confirmation).
847///
848/// Returns:
849/// - `Ok(Some(BtiPartitionLocation))` when a leaf is reached.
850/// - `Ok(None)` when no transition exists for a key byte (key not in trie).
851/// - `Err(_)` on structural parse errors.
852///
853/// This is the inner engine for [`lookup_partition_in_bti_file`].
854fn walk_bti_trie(
855    trie_data: &[u8],
856    root_offset: usize,
857    encoded_key: &[u8],
858) -> BtiResult<Option<BtiPartitionLocation>> {
859    let mut current_offset = root_offset;
860    let mut key_pos = 0;
861
862    loop {
863        if current_offset >= trie_data.len() {
864            return Err(Error::Parse(format!(
865                "BTI trie walk: offset {current_offset} out of bounds (trie size {})",
866                trie_data.len()
867            )));
868        }
869
870        let header_byte = trie_data[current_offset];
871        let ordinal = (header_byte >> 4) & 0x0F;
872        let payload_flags = header_byte & 0x0F;
873        let is_leaf = ordinal == 0; // PayloadOnly
874
875        if is_leaf {
876            // Path-compressed (Patricia) trie: a PayloadOnly leaf represents the
877            // unique key whose path through the trie led here.  The remaining key
878            // bytes are stored implicitly (compressed suffix) and need not be
879            // compared here — the caller verifies the actual partition key in Data.db.
880            //
881            // Note: the original implementation required the key to be fully consumed
882            // at the leaf (return Ok(None) otherwise), which is incorrect for
883            // path-compressed tries.  Issue #755 corrects this.
884            return read_node_payload(trie_data, current_offset);
885        }
886
887        // Non-leaf node: if key exhausted, check for an embedded payload
888        if key_pos >= encoded_key.len() {
889            if payload_flags != 0 {
890                return read_node_payload(trie_data, current_offset);
891            }
892            return Ok(None);
893        }
894
895        // Advance: look for the next key byte's transition
896        let next_byte = encoded_key[key_pos];
897        match find_next_child_offset(trie_data, current_offset, next_byte)? {
898            Some(child_offset) => {
899                current_offset = child_offset;
900                key_pos += 1;
901            }
902            None => {
903                // No transition for this byte → key not in trie
904                return Ok(None);
905            }
906        }
907    }
908}

910/// Look up a partition by its **byte-comparable** encoded key in a real
911/// Cassandra 5.0 `Partitions.db` BTI trie file.
912///
913/// # Format
914/// The file is a compact trie with **no header**:
915/// - Nodes are written bottom-up (children before parents).
916/// - The root node's absolute byte offset is stored as the **last 8 bytes** of
917///   the file (big-endian u64).
918/// - Every partition leaf carries a payload consisting of a 1-byte filter hash
919///   plus a SizedInts-encoded signed `position` (see [`decode_bti_partition_payload`]).
920/// - Negative `position` → `data_offset = ~position` (direct `Data.db` pointer).
921/// - Non-negative `position` → `rows_offset` into `Rows.db` (wide partition).
922///
923/// # Arguments
924/// * `reader`      – positioned at any point; this function seeks as needed.
925/// * `encoded_key` – the byte-comparable encoding of the partition key (e.g.
926///   as produced by [`ByteComparableEncoder`]).
927///
928/// # Returns
929/// * `Ok(Some(BtiPartitionLocation::DataOffset(off)))` – the partition starts
930///   at byte `off` in `Data.db`.  **This is the headline result of issue #755.**
931/// * `Ok(Some(BtiPartitionLocation::RowsOffset(off)))` – the partition has a
932///   row index; consult `Rows.db` at `off` for the row-level trie.
933/// * `Ok(None)` – the key was not found.
934/// * `Err(_)` – structural parse error.
935///
936/// # Reference
937/// Mirrors `PartitionIndex.java` (Cassandra 5.0.8), specifically the
938/// `openBlocking` + `exactCandidate` read path; `SizedInts.read`; and the
939/// payload sign convention at `PartitionIndex.java:131–135`.
940pub fn lookup_partition_in_bti_file<R: Read + Seek>(
941    reader: &mut R,
942    encoded_key: &[u8],
943) -> BtiResult<Option<BtiPartitionLocation>> {
944    // Step 1: determine the file size
945    let file_size = reader.seek(SeekFrom::End(0))?;
946
947    if file_size < 8 {
948        return Err(Error::Parse(format!(
949            "BTI Partitions.db is too small ({file_size} bytes; need at least 8 for footer)"
950        )));
951    }
952
953    // Step 2: read the 8-byte root offset from the end of the file
954    reader.seek(SeekFrom::End(-8))?;
955    let mut footer_buf = [0u8; 8];
956    reader.read_exact(&mut footer_buf)?;
957    let root_offset = u64::from_be_bytes(footer_buf);
958
959    // The trie data is everything before the 8-byte footer
960    let trie_size = file_size - 8;
961
962    if root_offset >= trie_size {
963        return Err(Error::Parse(format!(
964            "BTI Partitions.db: root_offset {root_offset} >= trie_size {trie_size}"
965        )));
966    }
967
968    // Step 3: load the entire trie into memory
969    // For typical Partitions.db files this is small (tens of KB for millions of
970    // partitions because the trie shares prefixes aggressively).
971    reader.seek(SeekFrom::Start(0))?;
972    let mut trie_data = vec![0u8; trie_size as usize];
973    reader.read_exact(&mut trie_data)?;
974
975    // Step 4: walk the trie
976    walk_bti_trie(&trie_data, root_offset as usize, encoded_key)
977}

979// ─────────────────────────────────────────────────────────────────────────────
980// BTI partition key encoding for Murmur3Partitioner (issue #755)
981// ─────────────────────────────────────────────────────────────────────────────
982//
983// Cassandra 5.0 BTI `Partitions.db` trie keys are derived from `DecoratedKey`
984// using `ByteComparable.Version.OSS50`.  For `Murmur3Partitioner` the encoding
985// is a path-compressed representation of:
986//
987//   [type_prefix=0x40] ++ [8 bytes: murmur3_token_bc]
988//
989// where `murmur3_token_bc = (murmur3_token(raw_key_bytes) as u64) XOR 0x8000_0000_0000_0000`
990// (flips the sign bit so signed i64 tokens sort correctly as unsigned big-endian bytes).
991//
992// Because the trie uses path compression (Patricia trie), looking up the first
993// few bytes of this 9-byte key is sufficient to reach the unique leaf for a
994// given partition.  The caller must verify the actual partition key in Data.db
995// to confirm the match (the hash byte in the leaf payload provides a fast pre-filter).
996//
997// This encoding was verified against the real `da-2-bti-Partitions.db` fixture:
998//   UUID 22222222-... → token bc first byte 0x90 → DataOffset(0)   ✓
999//   UUID 11111111-... → token bc first byte 0xBC → DataOffset(63)  ✓
1000//   UUID 33333333-... → token bc first byte 0xF9 → DataOffset(125) ✓
1001// ─────────────────────────────────────────────────────────────────────────────
1002
1003/// Encode a raw partition key (any CQL type) into the BTI trie lookup key for
1004/// `Murmur3Partitioner`.
1005///
1006/// The encoding is:
1007///   `[0x40] ++ [8 bytes big-endian: (murmur3_token(raw_key) as u64) XOR 0x8000_0000_0000_0000]`
1008///
1009/// This 9-byte prefix uniquely identifies the partition in the trie for typical
1010/// Cassandra datasets.  In a path-compressed (Patricia) trie, using only the
1011/// first few bytes is correct because each leaf represents exactly one partition:
1012/// once the traversal reaches a leaf, no further byte matching is required.
1013///
1014/// # Arguments
1015/// * `raw_key_bytes` – the raw serialized partition key bytes as stored on disk
1016///   (e.g. 16 bytes for a UUID, big-endian bytes for int/bigint, UTF-8 for text).
1017///
1018/// # Verified against
1019/// Real `da-2-bti-Partitions.db` fixture — see issue #755 for derivation.
1020pub fn encode_partition_key_for_bti_trie(raw_key_bytes: &[u8]) -> [u8; 9] {
1021    use crate::util::cassandra_murmur3::cassandra_murmur3_token;
1022
1023    let token: i64 = cassandra_murmur3_token(raw_key_bytes);
1024    let bc: u64 = (token as u64) ^ 0x8000_0000_0000_0000u64;
1025    let bc_bytes = bc.to_be_bytes();
1026
1027    let mut key = [0u8; 9];
1028    key[0] = 0x40; // fixed type-discriminator prefix (observed in Partitions.db trie)
1029    key[1..9].copy_from_slice(&bc_bytes);
1030    key
1031}

1089///

1091/// header** — the root node's absolute offset is the last 8 bytes of the file.
1092/// This is the footer-based loader used by the traversal iterators; it does NOT
1093/// rely on the fictional [`BtiHeader`] (whose `parse` would misread byte 0 of a

1095///

1100        return Err(Error::Parse(format!(
1101            "BTI file too small ({file_size} bytes; need at least 8 for footer)"
1102        )));
1103    }
1104
1105    reader.seek(SeekFrom::End(-8))?;
1106    let mut footer = [0u8; 8];
1107    reader.read_exact(&mut footer)?;
1108    let root_offset = u64::from_be_bytes(footer);
1109
1110    let trie_size = file_size - 8;
1111    if root_offset >= trie_size {
1112        return Err(Error::Parse(format!(

1114        )));
1115    }
1116
1117    reader.seek(SeekFrom::Start(0))?;
1118    let mut trie_data = vec![0u8; trie_size as usize];
1119    reader.read_exact(&mut trie_data)?;
1120    Ok((trie_data, root_offset as usize))
1121}

1125///
1126/// Crucially this iterates `Dense` children directly by index (transition byte =
1127/// `start_byte + i`) and **skips** the `None` slots, which are the Dense "no
1128/// transition" sentinels (raw delta `0`).  A `Some(ptr)` slot is a real child
1129/// and is always emitted — *including* one whose absolute offset is `0` (the
1130/// first-written leaf in BTI's bottom-up layout legitimately lives at offset 0).
1131/// ([`BtiNode::get_transitions`] returns an empty Vec for Dense nodes and must
1132/// NOT be used for traversal.)
1133///
1134/// `Sparse` transitions are returned in their stored order; the parser preserves
1135/// the on-disk ascending order, and we sort defensively.
1136fn ordered_children(node: &BtiNode) -> Vec<(u8, usize)> {
1137    match &node.data {
1138        BtiNodeData::PayloadOnly { .. } => Vec::new(),
1139        BtiNodeData::Single { transition } => {
1140            vec![(transition.byte, transition.child.distance as usize)]
1141        }
1142        BtiNodeData::Sparse { transitions } => {
1143            let mut out: Vec<(u8, usize)> = transitions
1144                .iter()
1145                .map(|t| (t.byte, t.child.distance as usize))
1146                .collect();
1147            // Defensive: BTI Sparse transitions are stored ascending; enforce it.
1148            out.sort_by_key(|&(b, _)| b);
1149            out
1150        }
1151        BtiNodeData::Dense {
1152            start_byte,
1153            children,
1154        } => {
1155            let mut out = Vec::new();
1156            for (i, child) in children.iter().enumerate() {
1157                // `None` is the Dense "no transition" sentinel (raw delta 0):
1158                // skip it.  `Some(ptr)` is a real child and is always emitted,
1159                // even when `ptr.distance == 0` — a real child can live at
1160                // absolute trie offset 0 (the first-written leaf).
1161                if let Some(ptr) = child {
1162                    let transition_byte = start_byte.wrapping_add(i as u8);
1163                    out.push((transition_byte, ptr.distance as usize));
1164                }
1165            }
1166            out
1167        }
1168    }
1169}

1171/// Generic in-order DFS over a BTI trie, decoding each node payload with
1172/// `decode_payload` and pushing `(reconstructed_key, decoded_payload)` onto the
1173/// result Vec in byte-comparable order.
1174///
1175/// The traversal is iterative (explicit stack), depth-capped at
1176/// [`DFS_MAX_DEPTH`], and bounds-checks every offset — corrupt or cyclic tries
1177/// produce an error rather than an infinite loop or panic.
1178fn dfs_collect_in_order<T, F>(
1179    trie_data: &[u8],
1180    root_offset: usize,
1181    mut decode_payload: F,
1182) -> BtiResult<Vec<(Vec<u8>, T)>>
1183where
1184    F: FnMut(&[u8], usize) -> BtiResult<Option<T>>,
1185{
1186    let mut results: Vec<(Vec<u8>, T)> = Vec::new();
1187
1188    // Stack frame: (node_offset, accumulated_key_bytes).
1189    // We use an explicit stack and push children in *reverse* order so the
1190    // smallest transition byte is processed first (LIFO).
1191    let mut stack: Vec<(usize, Vec<u8>)> = vec![(root_offset, Vec::new())];
1192
1193    while let Some((node_offset, key_bytes)) = stack.pop() {
1194        if key_bytes.len() > DFS_MAX_DEPTH {
1195            return Err(Error::Parse(format!(
1196                "BTI DFS exceeded max depth {DFS_MAX_DEPTH} (corrupt or cyclic trie)"
1197            )));
1198        }
1199        if node_offset >= trie_data.len() {
1200            return Err(Error::Parse(format!(
1201                "BTI DFS: node_offset {node_offset} out of bounds (trie size {})",
1202                trie_data.len()
1203            )));
1204        }
1205
1206        // 1) Emit this node's own payload (if any) BEFORE descending — the key
1207        //    terminating here sorts before any continuation.
1208        if let Some(payload) = decode_payload(trie_data, node_offset)? {
1209            results.push((key_bytes.clone(), payload));
1210        }
1211
1212        // 2) Descend children in ASCENDING transition-byte order.  Because the
1213        //    stack is LIFO, push them in DESCENDING order.
1214        let node = parse_bti_node_for_traversal(trie_data, node_offset)?;
1215        let children = ordered_children(&node);
1216        for &(transition_byte, child_offset) in children.iter().rev() {
1217            let mut child_key = key_bytes.clone();
1218            child_key.push(transition_byte);
1219            stack.push((child_offset, child_key));
1220        }
1221    }
1222
1223    Ok(results)
1224}

1258// Rows.db in-trie payload decoding (RowIndexReader / TrieIndexEntry)
1259// ─────────────────────────────────────────────────────────────────────────────
1260//
1261// IMPORTANT: `Rows.db` in-trie payloads are NOT the `Partitions.db` payload
1262// format (which is a hash byte + SizedInts *signed* position).  A `Rows.db`
1263// trie leaf carries a `RowIndexReader.IndexInfo` whose byte layout is defined
1264// authoritatively by `RowIndexReader.readPayload`
1265// (cassandra-5.0.0 `RowIndexReader.java:111-125`):
1266//
1267//   static IndexInfo readPayload(ByteBuffer buf, int ppos, int bits, Version v) {
1268//       if (bits == 0) return null;
1269//       int bytes = bits & ~FLAG_OPEN_MARKER;            // FLAG_OPEN_MARKER = 8
1270//       long offset = SizedInts.read(buf, ppos, bytes);  // SizedInts, NOT a vint
1271//       ppos += bytes;
1272//       DeletionTime del = (bits & FLAG_OPEN_MARKER) != 0
1273//                          ? DeletionTime.deserialize(buf, ppos) : null;
1274//       return new IndexInfo(offset, del);
1275//   }
1276//
1277// So the low nibble of the node header byte (`payloadBits`) splits as:
1278//   - low 3 bits  → the number of `SizedInts` bytes encoding the block offset
1279//   - bit 0x8     → FLAG_OPEN_MARKER: an open-deletion `DeletionTime` follows
1280//
1281// The `offset` field is the block's offset **relative to the partition start**
1282// in `Data.db` (the IndexInfo doc: "where in the data file to start looking for
1283// a given key"), so absolute Data.db position = `entry.data_position + offset`.
1284//
1285// The writer side (`RowIndexWriter.getSerializer.write`,
1286// `RowIndexWriter.java:160-180`) confirms this exactly:
1287//   bytes = SizedInts.nonZeroSize(payload.offset);
1288//   type.serialize(dest, node, bytes | hasOpenMarker, nodePosition);
1289//   SizedInts.write(dest, payload.offset, bytes);
1290//   if (hasOpenMarker) DeletionTime.serialize(payload.openDeletion, dest);
1291//
1292// Earlier this decoder read the offset as an *unsigned vint* and treated the
1293// whole low nibble as a flag — that misparsed every real wide-partition block
1294// (issue #832, Finding A's sibling bug: payloads came out as 64/0/1/… garbage).
1295//
1296// Reference: docs/sstables-definitive-guide chapter 17 (Rows.db footer);
1297//            cassandra-5.0.0 `RowIndexReader.java`, `RowIndexWriter.java`,
1298//            `TrieIndexEntry.java`, `SizedInts.java`.
1299// ─────────────────────────────────────────────────────────────────────────────

1301/// The `FLAG_OPEN_MARKER` bit in a `Rows.db` trie node's `payloadBits`
1302/// (low nibble of the header byte).  When set, an open-deletion `DeletionTime`

1304/// `RowIndexReader.FLAG_OPEN_MARKER`.

1309/// The headline field is [`data_offset`](Self::data_offset): the block's offset
1310/// **relative to the partition start** in `Data.db`.  To obtain the absolute
1311/// `Data.db` byte position, add the partition's data position (see
1312/// [`BtiRowIndexHeader::data_position`]).
1313#[derive(Debug, Clone, PartialEq, Eq)]
1314pub struct BtiRowIndexEntry {
1315    /// Block offset **relative to the partition start**, decoded via
1316    /// `SizedInts.read(buf, ppos, payloadBits & ~FLAG_OPEN_MARKER)`.
1317    pub data_offset: u64,

1323/// Read an unsigned VInt (Cassandra count-leading-ones encoding, **not** ZigZag)
1324/// from `data`, returning `(value, bytes_consumed)`.
1325///
1326/// This is the encoding Cassandra uses for Data.db positions in the row index
1327/// (`DataOutputPlus.writeUnsignedVInt`).  The number of extra bytes equals the
1328/// number of leading 1-bits in the first byte; the value is big-endian across
1329/// the remaining bits.
1330fn read_unsigned_vint_from_slice(data: &[u8]) -> BtiResult<(u64, usize)> {
1331    if data.is_empty() {
1332        return Err(Error::Parse(
1333            "Rows.db payload: unexpected end of data reading unsigned vint".to_string(),
1334        ));
1335    }
1336    let first = data[0];
1337    let extra_bytes = first.leading_ones() as usize;
1338    if extra_bytes > 8 {
1339        return Err(Error::Parse(format!(
1340            "Rows.db payload: invalid unsigned vint first byte 0x{first:02x}"
1341        )));
1342    }
1343    let total = extra_bytes + 1;
1344    if data.len() < total {
1345        return Err(Error::Parse(format!(
1346            "Rows.db payload: unsigned vint needs {total} bytes, have {}",
1347            data.len()
1348        )));
1349    }
1350
1351    // Data bits in the first byte: 8 - extra_bytes - 1 (the separator 0 bit),
1352    // except when extra_bytes == 8 (first byte is all ones, no data bits).
1353    let mut value: u64 = if extra_bytes >= 8 {
1354        0
1355    } else {
1356        let data_bits = 8 - extra_bytes - 1;
1357        let mask = if data_bits == 0 {
1358            0
1359        } else {
1360            (1u16 << data_bits) - 1
1361        };
1362        (first as u16 & mask) as u64
1363    };
1364    for &b in &data[1..total] {
1365        value = (value << 8) | (b as u64);
1366    }
1367    Ok((value, total))
1368}

1370/// The modern (DA/BTI) `DeletionTime` "live" sentinel byte.
1371///
1372/// In the `da`-family on-disk serializer, a `DeletionTime` written by the BTI
1373/// row-index / trie-index path is encoded as a single `0x80` byte when it is
1374/// `DeletionTime.LIVE` (no deletion), and otherwise as the full value (see
1375/// [`decode_da_deletion_time`]).  Mirrors the live/no-deletion fast-path of
1376/// `org.apache.cassandra.db.DeletionTime.Serializer` in the trie-index
1377/// (`cassandra-5.0.0`) format, where the live sentinel avoids writing the
1378/// 12-byte body.
1379const DA_DELETION_TIME_LIVE_SENTINEL: u8 = 0x80;
1380
1381/// Width of a non-live modern (DA) `DeletionTime` body: `i64 markedForDeleteAt`
1382/// followed by `u32 localDeletionTime`.
1383const DA_DELETION_TIME_BODY_LEN: usize = 12;
1384
1385/// Decode a modern (DA/BTI) `DeletionTime` at `data[start..]`, returning
1386/// `(deletion, bytes_consumed)` where `deletion` is `None` for the LIVE
1387/// sentinel (issue #832 Finding 2).
1388///
1389/// Layout (mirrors `org.apache.cassandra.db.DeletionTime.Serializer` in the
1390/// `da`/trie-index format, cassandra-5.0.0):
1391///
1392///   - a single `0x80` byte → `DeletionTime.LIVE` (no deletion); consumes 1 byte.
1393///   - otherwise the body is `[markedForDeleteAt : i64 BE][localDeletionTime :
1394///     u32 BE]` — `markedForDeleteAt` FIRST, then `localDeletionTime`; consumes
1395///     12 bytes.  This differs from the LEGACY layout
1396///     (`[localDeletionTime i32][markedForDeleteAt i64]`) in BOTH field order
1397///     and the width/signedness of `localDeletionTime` (modern: `u32`).
1398///
1399/// Returns the deletion as `(local_deletion_time, marked_for_delete_at)` to
1400/// match [`BtiRowIndexEntry::open_marker`]'s existing tuple ordering, even
1401/// though the modern wire order is the reverse.
1402///
1403/// # Errors
1404/// Returns a parse error if `start` is out of bounds or a non-live value is
1405/// truncated.
1406fn decode_da_deletion_time(data: &[u8], start: usize) -> BtiResult<(Option<(i32, i64)>, usize)> {
1407    if start >= data.len() {
1408        return Err(Error::Parse(format!(
1409            "DA DeletionTime: start {start} beyond buffer size {}",
1410            data.len()
1411        )));
1412    }
1413    if data[start] == DA_DELETION_TIME_LIVE_SENTINEL {
1414        return Ok((None, 1));
1415    }
1416    if start + DA_DELETION_TIME_BODY_LEN > data.len() {
1417        return Err(Error::Parse(format!(
1418            "DA DeletionTime: non-live value needs {DA_DELETION_TIME_BODY_LEN} bytes, have {}",
1419            data.len().saturating_sub(start)
1420        )));
1421    }
1422    let b = &data[start..start + DA_DELETION_TIME_BODY_LEN];
1423    // markedForDeleteAt FIRST (i64), then localDeletionTime (u32).
1424    let marked_for_delete_at = i64::from_be_bytes([b[0], b[1], b[2], b[3], b[4], b[5], b[6], b[7]]);
1425    let local_deletion_time = u32::from_be_bytes([b[8], b[9], b[10], b[11]]) as i32;
1426    Ok((
1427        Some((local_deletion_time, marked_for_delete_at)),
1428        DA_DELETION_TIME_BODY_LEN,
1429    ))
1430}

1432/// Decode a `Rows.db` in-trie payload (`RowIndexReader.IndexInfo`) at
1433/// `payload_start` inside `trie_data`, given the node's `payload_bits` (low
1434/// nibble of the header byte).
1435///
1436/// Layout (mirrors `RowIndexReader.readPayload`, cassandra-5.0.0
1437/// `RowIndexReader.java:111-125`):
1438///   - `bytes = payload_bits & !FLAG_OPEN_MARKER` → block offset is a
1439///     `SizedInts` value of `bytes` bytes (the offset is relative to the
1440///     partition's data position).
1441///   - if `payload_bits & FLAG_OPEN_MARKER`, an open-deletion `DeletionTime`
1442///     follows in the MODERN DA form ([`decode_da_deletion_time`]): a `0x80`
1443///     LIVE sentinel, else `[i64 markedForDeleteAt][u32 localDeletionTime]`.
1444///
1445/// A `payload_bits` of `0` is not a valid leaf payload here (the caller filters
1446/// such nodes out) and yields an error.
1447fn decode_bti_row_payload(
1448    trie_data: &[u8],
1449    payload_start: usize,
1450    payload_bits: u8,
1451) -> BtiResult<BtiRowIndexEntry> {
1452    if payload_start > trie_data.len() {
1453        return Err(Error::Parse(format!(
1454            "Rows.db payload start {payload_start} beyond trie size {}",
1455            trie_data.len()
1456        )));
1457    }
1458
1459    // Low 3 bits = number of SizedInts bytes; bit 0x8 = open-marker flag.
1460    let offset_bytes = (payload_bits & !FLAG_OPEN_MARKER) as usize;
1461    if offset_bytes == 0 || offset_bytes > 7 {
1462        // RowIndexWriter asserts `bytes < 8` ("rows larger than 32 PiB"); a
1463        // 0-byte offset would mean an empty payload, which the trie does not
1464        // emit for a real row-index block.
1465        return Err(Error::Parse(format!(
1466            "Rows.db payload: invalid SizedInts byte count {offset_bytes} \
1467             (payload_bits=0x{payload_bits:02x}); expected 1..=7"
1468        )));
1469    }
1470    if payload_start + offset_bytes > trie_data.len() {
1471        return Err(Error::Parse(format!(
1472            "Rows.db payload: SizedInts offset needs {offset_bytes} bytes, have {}",
1473            trie_data.len().saturating_sub(payload_start)
1474        )));
1475    }
1476
1477    // SizedInts is a signed sign-extended big-endian read (SizedInts.read).
1478    // Block offsets are non-negative in practice, but we decode faithfully.
1479    let raw = sized_ints_read_from_slice(&trie_data[payload_start..payload_start + offset_bytes])?;
1480    let data_offset = raw as u64;
1481
1482    let open_marker = if payload_bits & FLAG_OPEN_MARKER != 0 {
1483        // DA/BTI modern DeletionTime (issue #832 Finding 2): a 0x80 sentinel
1484        // means LIVE (no open deletion); otherwise [i64 markedForDeleteAt][u32
1485        // localDeletionTime].  This is NOT the legacy [i32 ldt][i64 mfda] form.
1486        let dt_start = payload_start + offset_bytes;
1487        let (deletion, _consumed) = decode_da_deletion_time(trie_data, dt_start)?;
1488        deletion
1489    } else {
1490        None
1491    };
1492
1493    Ok(BtiRowIndexEntry {
1494        data_offset,
1495        open_marker,
1496    })
1497}

1587// ─────────────────────────────────────────────────────────────────────────────
1588// Per-partition Rows.db entry resolution — TrieIndexEntry (issue #832 Finding A)
1589// ─────────────────────────────────────────────────────────────────────────────
1590//
1591// The positive `position` stored in a `Partitions.db` leaf payload
1592// (`BtiPartitionLocation::RowsOffset`) does NOT point at a row-index trie root.
1593// It points at this partition's **row-index entry** in `Rows.db`, which must be
1594// deserialized to recover the actual trie root (plus the partition's Data.db
1595// position, block count and partition-level deletion).  Feeding `RowsOffset`
1596// straight to `iterate_rows_in_bti_trie` parses entry metadata as a trie node
1597// and fails (or, worse, returns garbage) on real wide partitions.
1598//
1599// On-disk layout at `RowsOffset` (validated against the `wide_table` fixture —
1600// see `tests/issue_832_bti_traversal.rs`; cassandra-5.0.0 `TrieIndexEntry.java`
1601// `serialize`/`deserialize`, lines 89-120, and `RowIndexWriter.complete`):
1602//
1603//   [u16 key_length][partition key bytes]      ← short-length-prefixed key
1604//   [data file position : unsigned vint]       ← partition start in Data.db
1605//   [trie_root - base    : SIGNED vint]         ← base = RowsOffset + key_length
1606//   [row index block count : unsigned vint32]
1607//   [partition DeletionTime]                    ← delta/compact form; best-effort
1608//
1609// `TrieIndexEntry.deserialize` computes `indexTrieRoot = readVInt() + base`.
1610// Cassandra passes `base` as the file position from which the entry is read;
1611// for this fixture that base is `RowsOffset + key_length` (the partition key

1646/// Read a signed VInt (Cassandra zig-zag, `DataInputPlus.readVInt`) from `data`,
1647/// returning `(value, bytes_consumed)`.
1648fn read_signed_vint_from_slice(data: &[u8]) -> BtiResult<(i64, usize)> {
1649    let (u, n) = read_unsigned_vint_from_slice(data)?;
1650    // ZigZag decode: (u >>> 1) ^ -(u & 1)
1651    let value = ((u >> 1) as i64) ^ -((u & 1) as i64);
1652    Ok((value, n))
1653}

1670/// Resolve a partition's row-index entry in `Rows.db`, given the `RowsOffset`
1671/// from a `Partitions.db` lookup ([`BtiPartitionLocation::RowsOffset`]).
1672///
1673/// This is the fix for issue #832 Finding A: `RowsOffset` is the offset of the
1674/// per-partition `TrieIndexEntry`, NOT a trie root.  This deserializes that
1675/// entry — recovering the partition's Data.db position, the actual row-index
1676/// trie root, the block count and the partition deletion — so traversal can be
1677/// rooted correctly.
1678///
1679/// `rows_db` is the full `Rows.db` file contents; `rows_offset` is the
1680/// `RowsOffset` value.  All reads are bounds-checked.
1681///
1682/// # Errors
1683/// Returns a parse error if `rows_offset` is out of bounds, the key length is
1684/// implausible, the vint fields are truncated, or the recovered trie root falls
1685/// outside `rows_db`.
1686pub fn resolve_rows_db_entry(rows_db: &[u8], rows_offset: usize) -> BtiResult<BtiRowIndexHeader> {
1687    if rows_offset + 2 > rows_db.len() {
1688        return Err(Error::Parse(format!(
1689            "Rows.db entry: rows_offset {rows_offset} + 2 (key length) exceeds file size {}",
1690            rows_db.len()
1691        )));
1692    }
1693
1694    // [u16 key_length][key bytes]
1695    let key_length = u16::from_be_bytes([rows_db[rows_offset], rows_db[rows_offset + 1]]) as usize;
1696    let entry_start = rows_offset + 2 + key_length;
1697    if entry_start > rows_db.len() {
1698        return Err(Error::Parse(format!(
1699            "Rows.db entry: key length {key_length} at offset {rows_offset} overruns file size {}",
1700            rows_db.len()
1701        )));
1702    }
1703
1704    // base for the SIGNED root delta = RowsOffset + key_length (see module note).
1705    let base = rows_offset + key_length;
1706
1707    let mut cur = entry_start;
1708    let (data_position, n) = read_unsigned_vint_from_slice(&rows_db[cur..])?;
1709    cur += n;
1710
1711    let (root_delta, n) = read_signed_vint_from_slice(&rows_db[cur..])?;
1712    cur += n;
1713
1714    // indexTrieRoot = readVInt() + base   (TrieIndexEntry.deserialize)
1715    let trie_root_signed = root_delta + base as i64;
1716    if trie_root_signed < 0 || (trie_root_signed as usize) >= rows_db.len() {
1717        return Err(Error::Parse(format!(
1718            "Rows.db entry: recovered trie root {trie_root_signed} out of bounds \
1719             (base={base}, delta={root_delta}, file size={})",
1720            rows_db.len()
1721        )));
1722    }
1723    let trie_root = trie_root_signed as usize;
1724
1725    let (block_count_u64, n) = read_unsigned_vint_from_slice(&rows_db[cur..])?;
1726    cur += n;
1727    let block_count = u32::try_from(block_count_u64).map_err(|_| {
1728        Error::Parse(format!(
1729            "Rows.db entry: implausible block count {block_count_u64}"
1730        ))
1731    })?;
1732
1733    // Partition DeletionTime: decode the MODERN DA/BTI form (issue #832
1734    // Finding 2) — a 0x80 byte means LIVE (no deletion, the common case when no
1735    // deletes were issued); otherwise [i64 markedForDeleteAt][u32
1736    // localDeletionTime].  Decoded best-effort: if too few trailing bytes
1737    // remain, leave it `None` rather than failing (it is not required for
1738    // traversal correctness).
1739    let partition_deletion = match decode_da_deletion_time(rows_db, cur) {
1740        Ok((deletion, _consumed)) => deletion,
1741        Err(_) => None,
1742    };
1743
1744    Ok(BtiRowIndexHeader {
1745        data_position,
1746        trie_root,
1747        block_count,
1748        partition_deletion,
1749    })
1750}

1752/// Cassandra component separator used between clustering components in the
1753/// OSS50 byte-comparable form.  Mirrors `ByteSource.NEXT_COMPONENT` (value
1754/// `0x40`) emitted by `ClusteringComparator.asByteComparable` /
1755/// `ByteSource.withTerminator`/`ByteSource.of(...)` in
1756/// `org.apache.cassandra.utils.bytecomparable.ByteSource` (cassandra-5.0.0).
1757const OSS50_NEXT_COMPONENT: u8 = 0x40;

1759/// Encode a single clustering-component [`Value`] in Cassandra OSS50
1760/// **byte-comparable** form (the SAME encoding the `Rows.db` row-index trie
1761/// stores its separators in), appending to `out`.
1762///
1763/// This is NOT CQLite's custom [`ByteComparableEncoder`] (which prepends a
1764/// 1-byte type discriminator and so is byte-incompatible with the on-disk
1765/// trie).  It reproduces the per-type `AbstractType.asComparableBytes(...)`
1766/// production used by Cassandra to build the byte-comparable keys:
1767///
1768/// - `Int32Type` → `(v as u32 ^ 0x8000_0000)` big-endian (sign-flip, 4B), per
1769///   `Int32Type.asComparableBytes` / `ByteSource.optionalSignedFixedLengthNumber`.
1770/// - `LongType` → `(v as u64 ^ 0x8000_0000_0000_0000)` big-endian (8B), per
1771///   `LongType.asComparableBytes`.
1772/// - `ShortType`/`ByteType` (`smallint`/`tinyint`) → sign-flip big-endian (2B /
1773///   1B), per `ShortType`/`ByteType.asComparableBytes`.
1774/// - `BooleanType` → single byte `0x00`/`0x01`, per `BooleanType.asComparableBytes`.
1775/// - `TimestampType` (`timestamp`) → `LongType`-style sign-flip 8B (`TimestampType`
1776///   extends `LongType`'s comparable form).
1777/// - `UUIDType`/`TimeUUIDType` → the 16 raw bytes (network order is already
1778///   byte-comparable for the on-disk separator form here).
1779/// - `UTF8Type`/`AsciiType` (`text`/`ascii`) and `BytesType` (`blob`) → the raw
1780///   value bytes, terminated by the OSS50 variable-length component terminator
1781///   (`0x00`) so a shorter value sorts before a longer one sharing its prefix
1782///   (`ByteSource.of(ByteBuffer)` + the terminator convention in
1783///   `ByteComparable.Version.OSS50`).
1784///
1785/// Any clustering type not enumerated here returns an explicit parse error
1786/// (NO silent wrong results — issue #28 no-heuristics mandate).
1787/// OSS50 variable-length byte-comparable encoding for `BytesType`/`UTF8Type`
1788/// (Cassandra `ByteSource`): every literal `0x00` byte is escaped as
1789/// `0x00 0xFE` (`ESCAPE` + `ESCAPED_0_CONT`) and the component is terminated
1790/// with `0x00 0xFF` (`ESCAPE` + `ESCAPED_0_DONE`).  This makes the encoding
1791/// weakly prefix-free, so a shorter value sorts before a longer value that
1792/// shares its prefix — matching the separators stored in a real `Rows.db`
1793/// trie.  (The project's `ByteComparableEncoder` uses a different,
1794/// non-Cassandra escape scheme and must NOT be used for trie-compatible bounds.)
1795fn encode_varlen_oss50(bytes: &[u8], out: &mut Vec<u8>) {
1796    for &b in bytes {
1797        out.push(b);
1798        if b == 0x00 {
1799            out.push(0xFE);
1800        }
1801    }
1802    out.push(0x00);
1803    out.push(0xFF);
1804}

1806fn encode_clustering_component_oss50(value: &Value, out: &mut Vec<u8>) -> BtiResult<()> {
1807    match value {
1808        // int — Int32Type: sign-flip, big-endian (matches `wide_table` separators,
1809        // e.g. ck=8 → 80 00 00 08).
1810        Value::Integer(v) => {
1811            out.extend_from_slice(&((*v as u32) ^ 0x8000_0000).to_be_bytes());
1812            Ok(())
1813        }
1814        // bigint — LongType: sign-flip, big-endian, 8 bytes.
1815        Value::BigInt(v) | Value::Counter(v) => {
1816            out.extend_from_slice(&((*v as u64) ^ 0x8000_0000_0000_0000).to_be_bytes());
1817            Ok(())
1818        }
1819        // smallint — ShortType: sign-flip, big-endian, 2 bytes.
1820        Value::SmallInt(v) => {
1821            out.extend_from_slice(&((*v as u16) ^ 0x8000).to_be_bytes());
1822            Ok(())
1823        }
1824        // tinyint — ByteType: sign-flip, 1 byte.
1825        Value::TinyInt(v) => {
1826            out.push((*v as u8) ^ 0x80);
1827            Ok(())
1828        }
1829        // boolean — BooleanType: single 0x00/0x01 byte.
1830        Value::Boolean(b) => {
1831            out.push(if *b { 0x01 } else { 0x00 });
1832            Ok(())
1833        }
1834        // timestamp — TimestampType shares LongType's comparable form (8-byte
1835        // sign-flip big-endian of the millisecond value).
1836        Value::Timestamp(v) => {
1837            out.extend_from_slice(&((*v as u64) ^ 0x8000_0000_0000_0000).to_be_bytes());
1838            Ok(())
1839        }
1840        // uuid / timeuuid — raw 16 bytes (already byte-comparable in the on-disk
1841        // separator form for this fixed-length type).
1842        Value::Uuid(bytes) => {
1843            out.extend_from_slice(bytes);
1844            Ok(())
1845        }
1846        // text / ascii — OSS50 variable-length byte-comparable encoding.
1847        Value::Text(s) => {
1848            encode_varlen_oss50(s.as_bytes(), out);
1849            Ok(())
1850        }
1851        // blob / inet — OSS50 variable-length byte-comparable encoding.
1852        Value::Blob(b) | Value::Inet(b) => {
1853            encode_varlen_oss50(b, out);
1854            Ok(())
1855        }
1856        other => Err(Error::Parse(format!(
1857            "BTI range_query: byte-comparable encoding not implemented for {:?}",
1858            other.data_type()
1859        ))),
1860    }
1861}

1863/// Encode a multi-component clustering bound (`&[Value]`) in Cassandra OSS50
1864/// byte-comparable form — the SAME encoding the `Rows.db` trie stores.
1865///
1866/// A single-component clustering encodes to the bare component bytes (matching
1867/// the `wide_table` fixture separators, e.g. ck=8 → `80 00 00 08`, NO framing).
1868/// Multi-component clusterings concatenate per-component byte-comparable
1869/// encodings separated by [`OSS50_NEXT_COMPONENT`] (`ByteSource.NEXT_COMPONENT`,
1870/// per `ClusteringComparator.asByteComparable`), with no leading/trailing frame
1871/// so a prefix bound sorts before any longer key sharing it.
1872pub fn encode_clustering_bound_oss50(values: &[Value]) -> BtiResult<Vec<u8>> {
1873    // All-ascending convenience: every component uses its base byte-comparable
1874    // form. Equivalent to `encode_clustering_bound_oss50_with_order` with all
1875    // `is_reversed = false`.
1876    let mut out = Vec::new();
1877    for (i, v) in values.iter().enumerate() {
1878        if i > 0 {
1879            out.push(OSS50_NEXT_COMPONENT);
1880        }
1881        encode_clustering_component_oss50(v, &mut out)?;
1882    }
1883    Ok(out)
1884}

1886/// Encode a multi-component clustering bound applying each column's clustering
1887/// **ORDER** (ASC/DESC), producing Cassandra OSS50 byte-comparable separators
1888/// that sort in the SAME order the rows are physically written.
1889///
1890/// ## Why DESC must invert the component bytes
1891///
1892/// Cassandra wraps a `CLUSTERING ORDER BY (c DESC)` column's type in
1893/// `ReversedType`. `ReversedType.asComparableBytes` delegates to the base type's
1894/// byte-comparable production and then inverts the resulting [`ByteSource`] via
1895/// `ByteSource.invert(...)`, which complements every emitted data byte
1896/// (`b -> 0xFF ^ b`). Complementing every byte of a (weakly prefix-free)
1897/// byte-comparable encoding reverses its lexicographic order, so a DESCENDING
1898/// value order maps to an ASCENDING byte order. This is exactly what a `Rows.db`
1899/// row-index trie needs: separators are always stored in ascending *byte* order,
1900/// while the underlying clustering values run descending for a DESC column.
1901/// (cassandra-5.0.0 `org.apache.cassandra.db.marshal.ReversedType.asComparableBytes`
1902/// + `org.apache.cassandra.utils.bytecomparable.ByteSource.invert`.)
1903///
1904/// The inter-component framing byte ([`OSS50_NEXT_COMPONENT`], `0x40`) is emitted
1905/// by the *comparator* (`ClusteringComparator.asByteComparable`), NOT by a
1906/// component's type, so it is **not** inverted even when neighbouring components
1907/// are DESC — only the per-component byte-comparable bytes are complemented. This
1908/// matches Cassandra, where each `subtype(i)` (possibly a `ReversedType`) emits
1909/// its own (already inverted) byte source and the comparator separates them with
1910/// the un-inverted `NEXT_COMPONENT` byte.
1911///
1912/// `is_reversed[i]` MUST correspond positionally to `values[i]` (i.e. the schema
1913/// clustering-key order). A short `is_reversed` slice treats missing entries as
1914/// ascending (`false`).
1915pub fn encode_clustering_bound_oss50_with_order(
1916    values: &[Value],
1917    is_reversed: &[bool],
1918) -> BtiResult<Vec<u8>> {
1919    let mut out = Vec::new();
1920    for (i, v) in values.iter().enumerate() {
1921        if i > 0 {
1922            out.push(OSS50_NEXT_COMPONENT);
1923        }
1924        if is_reversed.get(i).copied().unwrap_or(false) {
1925            // Encode the base component into a scratch buffer, then complement
1926            // every byte (ReversedType / ByteSource.invert).
1927            let mut scratch = Vec::new();
1928            encode_clustering_component_oss50(v, &mut scratch)?;
1929            for b in &scratch {
1930                out.push(0xFF ^ *b);
1931            }
1932        } else {
1933            encode_clustering_component_oss50(v, &mut out)?;
1934        }
1935    }
1936    Ok(out)
1937}

1939/// Select the row-index blocks that may contain clustering keys in the
1940/// inclusive byte-comparable range `[start, end]`, applying row-index
1941/// **separator** semantics (issue #832 Finding B).
1942///
1943/// ## Why naive `[start, end]` filtering is wrong
1944///
1945/// A `Rows.db` row-index trie stores **separators**, not block start keys.  For
1946/// consecutive blocks the writer (`RowIndexWriter.add`) stores the shortest
1947/// `sep` with `prevMax < sep <= nextBlockFirstKey`, and `complete()` appends a
1948/// trailing separator after the last block.  Consequently the separator `s_i`
1949/// labels the boundary at the START of block `i`'s key range, and block `i`
1950/// covers the half-open key interval `[s_i, s_{i+1})` (the final block runs to
1951/// the trailing separator).  A reader locates the block for a key `K` via the
1952/// trie *floor* of `K` (`RowIndexReader.separatorFloor`), i.e. the block whose
1953/// separator is the greatest `<= K`.
1954///
1955/// Therefore a block `i` overlaps the requested clustering range `[start, end]`
1956/// iff its key interval `[s_i, s_{i+1})` intersects `[start, end]`:
1957///
1958///   `s_i <= end`  AND  `s_{i+1} > start`
1959///
1960/// (For the last block, `s_{i+1}` is treated as +∞.)  Filtering separators by
1961/// `start <= s_i <= end` — as the original implementation did — drops the
1962/// *floor* block whose separator is below `start` but whose interval still
1963/// contains `start`, and mishandles the block straddling `end`.
1964///
1965/// `entries` MUST be the full ascending-order `(separator, block)` list for one
1966/// partition (as produced by [`iterate_rows_in_bti_trie`]).  `start`/`end` are
1967/// byte-comparable clustering bounds in the **same encoding as the trie keys**
1968/// (Cassandra OSS50 clustering byte-comparable order).  Reversed bounds
1969/// (`start > end`) yield an empty result.
1970pub fn select_row_index_blocks_for_range(
1971    entries: &[(Vec<u8>, BtiRowIndexEntry)],
1972    start: &[u8],
1973    end: &[u8],
1974) -> Vec<BtiRowIndexEntry> {
1975    if start > end || entries.is_empty() {
1976        return Vec::new();
1977    }
1978
1979    let mut out = Vec::new();
1980    for (i, (sep_i, block)) in entries.iter().enumerate() {
1981        // s_{i+1}: the next separator, or +∞ for the last block.
1982        let next_is_greater_than_start = match entries.get(i + 1) {
1983            Some((sep_next, _)) => sep_next.as_slice() > start,
1984            None => true, // +∞ > start
1985        };
1986        let overlaps = sep_i.as_slice() <= end && next_is_greater_than_start;
1987        if overlaps {
1988            out.push(block.clone());
1989        }
1990    }
1991    out
1992}

2039/// BTI header structure for index files
2040#[derive(Debug, Clone)]
2041pub struct BtiHeader {
2042    /// BTI format magic number
2043    pub magic: u32,
2044    /// Format version
2045    pub version: u16,
2046    /// Format flags
2047    pub flags: u16,
2048    /// Offset to root node
2049    pub root_offset: u64,
2050    /// Number of entries in the index
2051    pub entry_count: u64,
2052    /// Additional metadata size
2053    pub metadata_size: u32,
2054}
2055
2056impl BtiHeader {
2057    /// BTI magic number
2058    pub const MAGIC: u32 = 0x6461_0000; // 'da\0\0'
2059
2060    /// Current BTI version
2061    pub const VERSION: u16 = 0x0001;
2062
2063    /// A benign placeholder header for files that carry no fictional
2064    /// [`BtiHeader`] (i.e. every real Cassandra `Rows.db`/`Partitions.db`, which
2065    /// are footer-rooted).  Used by [`RowsParser::new`] so the trie-based,
2066    /// whole-file entry points work on real files that legitimately lack the
2067    /// header; the fields are never consulted by those entry points.
2068    pub fn placeholder() -> Self {
2069        BtiHeader {
2070            magic: Self::MAGIC,
2071            version: Self::VERSION,
2072            flags: 0,
2073            root_offset: 0,
2074            entry_count: 0,
2075            metadata_size: 0,
2076        }
2077    }
2078
2079    /// Parse BTI header from bytes
2080    pub fn parse(data: &[u8]) -> BtiResult<(Self, usize)> {
2081        if data.len() < 24 {
2082            return Err(Error::Parse("BTI header too short".to_string()));
2083        }
2084
2085        let magic = u32::from_be_bytes([data[0], data[1], data[2], data[3]]);
2086        if magic != Self::MAGIC {
2087            return Err(Error::Parse(format!(
2088                "Invalid BTI magic: 0x{:08x}, expected 0x{:08x}",
2089                magic,
2090                Self::MAGIC
2091            )));
2092        }
2093
2094        let version = u16::from_be_bytes([data[4], data[5]]);

2633        // An empty Rows.db (e.g. a 0-byte file for SSTables with no row index)
2634        // yields nothing rather than erroring.
2635        let file_size = parser.reader.seek(SeekFrom::End(0))?;
2636        if file_size == 0 {
2637            return Ok(Self {
2638                parser,
2639                entries: Vec::new().into_iter(),
2640                pending_error: None,
2641            });
2642        }
2643
2644        // Finding A: resolve the per-partition TrieIndexEntry at `rows_offset`
2645        // to recover the actual trie root, then traverse from THAT root.
2646        let (entries, pending_error) = match parser.read_full_rows_db().and_then(|trie| {
2647            let header = resolve_rows_db_entry(&trie, rows_offset)?;
2648            iterate_rows_in_bti_trie(&trie, header.trie_root)
2649        }) {
2650            Ok(v) => (v, None),
2651            Err(e) => (Vec::new(), Some(e)),
2652        };
2653        Ok(Self {
2654            parser,
2655            entries: entries.into_iter(),
2656            pending_error,
2657        })
2658    }
2659}
2660
2661impl<'a, R: Read + Seek> Iterator for RowIterator<'a, R> {
2662    type Item = BtiResult<(Vec<u8>, BtiRowIndexEntry)>;
2663
2664    fn next(&mut self) -> Option<Self::Item> {
2665        if let Some(err) = self.pending_error.take() {
2666            return Some(Err(err));
2667        }
2668        self.entries.next().map(Ok)
2669    }
2670}

2781    /// Sparse12 (ordinal 6): packed 12-bit pointers.
2782    ///
2783    /// Layout per TrieNode.java Sparse12.serialize():
2784    ///   [0x60|pf] [count] [count transition bytes]
2785    ///   [ceil(count*3/2) packed-pointer bytes]
2786    ///
2787    /// Packing: for each even/odd pair (p0, p1) → 3 bytes [p0>>4, (p0<<4)|(p1>>8), p1&0xFF].
2788    /// An odd trailing pointer → 2 bytes [(pd << 4) as short big-endian].
2789    ///
2790    /// Total node bytes = 2 + count + ceil(count*3/2).
2791    /// count=1 → 5 bytes; count=2 → 7 bytes.
2792    fn sparse12_node(payload_flags: u8, pairs: &[(u8, u16)]) -> Vec<u8> {
2793        let count = pairs.len();
2794        let mut v = vec![0x60 | (payload_flags & 0x0F), count as u8];
2795        for &(t, _) in pairs {
2796            v.push(t);
2797        }
2798        // Pack pointers: process pairs, then trailing odd entry
2799        let mut i = 0;
2800        while i + 2 <= count {
2801            let p0 = pairs[i].1 as u32;
2802            let p1 = pairs[i + 1].1 as u32;
2803            v.push((p0 >> 4) as u8);
2804            v.push(((p0 << 4) | (p1 >> 8)) as u8);
2805            v.push((p1 & 0xFF) as u8);
2806            i += 2;
2807        }
2808        if i < count {
2809            // Trailing odd pointer: writeShort((short)(pd << 4)) big-endian
2810            let pd = pairs[i].1 as u32;
2811            let s = (pd << 4) as u16;
2812            v.extend_from_slice(&s.to_be_bytes());
2813        }
2814        v
2815    }

2851    /// Dense12 (ordinal 10): packed 12-bit pointers for a contiguous byte range.
2852    ///
2853    /// Layout per TrieNode.java Dense12.serialize():
2854    ///   [0xA0|pf] [start_byte] [range_len - 1] [ceil(range_len*3/2) packed bytes]
2855    ///
2856    /// Packing matches write12Bits(): even index → [val>>4, carry=val<<4]; odd index → [carry|(val>>8), val&0xFF].
2857    fn dense12_node(payload_flags: u8, start: u8, deltas: &[u16]) -> Vec<u8> {
2858        let range_len = deltas.len();
2859        let mut v = vec![0xA0 | (payload_flags & 0x0F), start, (range_len - 1) as u8];
2860        let mut carry: u8 = 0;
2861        for (i, &d) in deltas.iter().enumerate() {
2862            let val = d as u32;
2863            if (i & 1) == 0 {
2864                v.push((val >> 4) as u8);
2865                carry = (val << 4) as u8;
2866            } else {
2867                v.push(carry | (val >> 8) as u8);
2868                v.push((val & 0xFF) as u8);
2869                carry = 0;
2870            }
2871        }
2872        // If odd number of entries, flush carry byte
2873        if (range_len & 1) == 1 {
2874            v.push(carry);
2875        }
2876        v
2877    }

4661    /// when an internal-node payload's key is a STRICT PREFIX of a deeper leaf
4662    /// key.  This exercises the byte-comparable inclusive filter applied on top
4663    /// of the DFS — the same filter `RowsParser::range_query` uses — and proves
4664    /// that prefixes are neither spuriously dropped nor included.
4665    ///
4666    /// Trie layout (bottom-up):
4667    ///   K  = [0x10]        (internal node, carries a payload: pos=1)
4668    ///   K2 = [0x10, 0x20]  (deeper leaf: pos=2)
4669    ///
4670    ///   offset 0: Rows leaf pos=2                         → K2 payload
4671    ///   offset 2: Single8 trans=0x20 delta=2, payload pos=1 (the K node)
4672    ///   offset 5: Single8 root trans=0x10 delta=3, no payload
4673    ///
4674    /// So root --0x10--> K(internal, payload) --0x20--> K2(leaf).
4675    #[test]
4676    fn range_filter_prefix_relationship_is_sound() {
4677        let mut trie = Vec::new();
4678        // offset 0: leaf for K2 = [0x10,0x20], pos=2
4679        trie.extend_from_slice(&row_leaf_no_marker(2));
4680        // offset 2: Single8 with payload — the K = [0x10] node.
4681        //   header 0x21 (ordinal=2 Single8, payloadBits=1), transition=0x20,
4682        //   delta=2 (child = 2 - 2 = 0 → the K2 leaf), then payload pos=1.
4683        let k_off = trie.len() as u64; // 2
4684        trie.push(0x21); // Single8 + payloadBits=1
4685        trie.push(0x20); // transition byte
4686        trie.push(k_off as u8); // delta=2 → child at offset 0
4687        trie.push(0x01); // payload: unsigned vint pos=1
4688
4689        // offset 6: Single8 root, no payload, transition=0x10 → child = k_off (2).
4690        let root = trie.len() as u64; // 6
4691        trie.push(0x20); // Single8, payloadBits=0 (no payload)
4692        trie.push(0x10); // transition byte
4693        trie.push((root - k_off) as u8); // delta=4 → child at k_off=2

4697        assert_eq!(
4698            all,
4699            vec![

4701                    vec![0x10],
4702                    BtiRowIndexEntry {
4703                        data_offset: 1,
4704                        open_marker: None
4705                    }
4706                ),
4707                (
4708                    vec![0x10, 0x20],
4709                    BtiRowIndexEntry {
4710                        data_offset: 2,
4711                        open_marker: None
4712                    }
4713                ),
4714            ],
4715            "K (internal payload) must sort before its descendant K2"
4716        );

4719        let filter = |lo: &[u8], hi: &[u8]| -> Vec<u64> {
4720            if lo > hi {
4721                return Vec::new();

4757    /// Wrap a single-byte-keyed 3-leaf Rows.db trie behind a synthetic
4758    /// per-partition `TrieIndexEntry`, returning `(buffer, rows_offset)` suitable
4759    /// for `RowsParser::range_query` / `range_query_with_order`.
4760    ///
4761    /// Transition bytes `k1<k2<k3` are the (already byte-comparable) single-byte
4762    /// separators — e.g. the OSS50 encodings of `tinyint` clustering values, which
4763    /// are exactly one byte each (`v ^ 0x80`, or `0xFF ^ (v ^ 0x80)` for DESC).
4764    /// The trie is built first, then the `TrieIndexEntry` is appended; the entry's
4765    /// `rootΔ` points back at the trie root, mirroring the real `Rows.db` layout
4766    /// and the `resolve_rows_db_entry` tests above.
4767    fn make_rows_db_with_three(
4768        (k1, p1): (u8, u8),
4769        (k2, p2): (u8, u8),
4770        (k3, p3): (u8, u8),
4771    ) -> (Vec<u8>, usize) {
4772        let (trie, root) = make_rows_trie_three((k1, p1), (k2, p2), (k3, p3));
4773        let mut buf = trie; // bytes [0, root..] are the trie nodes
4774        let rows_offset = buf.len(); // TrieIndexEntry starts right after the trie
4775
4776        // key: length 4, value 0x00000007 (content is irrelevant to range tests).
4777        buf.extend_from_slice(&4u16.to_be_bytes());
4778        buf.extend_from_slice(&[0x00, 0x00, 0x00, 0x07]);
4779        let base = rows_offset + 4; // RowsOffset + key_length
4780
4781        // dataPos = 0 (unsigned vint); block offsets are relative to it.
4782        buf.push(0);
4783        // rootΔ such that trie_root = `root`: Δ = root - base (zigzag + uvint).
4784        let root_delta: i64 = root as i64 - base as i64;
4785        let zig = ((root_delta << 1) ^ (root_delta >> 63)) as u64;
4786        // Emit as a (possibly multi-byte) unsigned vint, MSB-first count-of-ones.
4787        write_uvint(&mut buf, zig);
4788        // blockCount = 3 (unsigned vint).
4789        buf.push(3);
4790        // partition DeletionTime: MODERN DA live sentinel (no deletion).
4791        buf.push(0x80);
4792
4793        (buf, rows_offset)
4794    }

4796    /// Minimal unsigned-vint writer (count-leading-ones form, matching
4797    /// [`read_unsigned_vint_from_slice`]) for test fixtures. Only the 1- and
4798    /// 2-byte forms are needed for the small `rootΔ` magnitudes used here.
4799    fn write_uvint(out: &mut Vec<u8>, v: u64) {
4800        if v < 0x80 {
4801            // 1-byte form: high bit 0, 7 data bits.
4802            out.push(v as u8);
4803            return;
4804        }
4805        // 2-byte form: leading byte `10xxxxxx` (1 leading one, 0 separator, 6
4806        // data bits) + 1 trailing byte = 14 data bits.
4807        assert!(v < 0x4000, "test vint fixture expects <= 14-bit values");
4808        out.push(0x80 | (v >> 8) as u8);
4809        out.push((v & 0xFF) as u8);
4810    }
