1//! Statistics.db writer - writes SSTable metadata
2//!
3//! Generates the Statistics.db component with min/max timestamps, TTL, and
4//! other metadata used for delta encoding in Data.db.
5//!
6//! Critical requirements:
7//! - MUST be written BEFORE Data.db (provides delta encoding baseline)
8//! - Min timestamp, max timestamp
9//! - Min TTL, max TTL
10//! - Min local deletion time, max local deletion time
11//! - Partition count, row count
12//!
13//! # Statistics.db Format (Cassandra 5.0 Compatible)
14//!
15//! This implementation produces a full Cassandra 5.0 nb-format Statistics.db with:
16//! - TOC (Table of Contents) with checksums
17//! - Four metadata components: VALIDATION, COMPACTION, STATS, HEADER
18//! - Per-component CRC32 checksums
19//! - Global CRC32 checksum validation
20//!
21//! ## Format Structure
22//!
23//! ```text
24//! [0-3]   num_components (u32 BE) = 4
25//! [4-7]   CRC32(num_components)
26//! [8-39]  TOC entries (4 components × 8 bytes each):
27//!           [u32 BE] component_type (MetadataType ordinal)
28//!           [u32 BE] component_offset
29//! [40-43] CRC32(num_components + all TOC entries) [cumulative]
30//! [44+]   Component data:
31//!           [N bytes] component_data
32//!           [4 bytes] CRC32(component_data)
33//!           ... (repeated for each component)
34//! ```
35//!
36//! ## MetadataType Component IDs
37//!
38//! From Cassandra's `MetadataType.java` enum (ordinal values):
39//! - 0: VALIDATION (validator class name)
40//! - 1: COMPACTION (compaction metadata)
41//! - 2: STATS (statistics including EncodingStats)
42//! - 3: SERIALIZATION_HEADER (table schema)
43

51/// Maximum number of bins in the tombstone drop-time histogram.
52///
53/// Mirrors Cassandra's `StreamingTombstoneHistogramBuilder.MAX_BIN_SIZE = 100`.
54const TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE: i32 = 100;
55
56/// Streaming tombstone drop-time histogram.
57///
58/// Mirrors Cassandra's `StreamingTombstoneHistogramBuilder` with a fixed maximum
59/// of 100 bins. Tombstone local-deletion-times are accumulated; when the bin count
60/// exceeds the maximum the two nearest bins are merged, keeping the bin count at or
61/// below `MAX_BIN_SIZE`.
62///
63/// Serialisation uses the **legacy** (nb/mc) format:
64/// ```text
65/// i32 BE  maxBinSize  (100 when non-empty, 0 when empty)
66/// i32 BE  size        (actual number of bins)
67/// for each bin:
68///   f64 BE  point  (local-deletion-time as a double)
69///   i64 BE  value  (count of tombstones in this bin)
70/// ```
71#[derive(Debug, Clone, Default)]
72pub struct TombstoneHistogram {
73    /// Bins keyed by `local_deletion_time as i64` (for stable ordering).
74    /// Each value is the count of tombstones that map to this bin.
75    bins: BTreeMap<i64, i64>,
76}
77
78impl TombstoneHistogram {
79    /// Create an empty histogram.
80    pub fn new() -> Self {
81        Self::default()
82    }
83
84    /// Record a tombstone with the given local-deletion-time (Unix seconds).
85    ///
86    /// If adding the new point would exceed `MAX_BIN_SIZE` the two closest
87    /// existing bins are merged first, exactly as Cassandra's builder does.
88    pub fn update(&mut self, local_deletion_time: i32) {
89        let key = local_deletion_time as i64;
90        *self.bins.entry(key).or_insert(0) += 1;
91
92        // Merge bins while over capacity
93        while self.bins.len() as i32 > TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE {
94            self.merge_closest_bins();
95        }
96    }
97
98    /// Return `true` if no tombstone deletion times have been recorded.
99    pub fn is_empty(&self) -> bool {
100        self.bins.is_empty()
101    }
102
103    /// Return the number of populated bins.
104    pub fn size(&self) -> i32 {
105        self.bins.len() as i32
106    }
107
108    /// Merge the two bins whose keys are closest together.
109    ///
110    /// The merged bin's key is the weighted average of the two points;
111    /// its value is the sum of both counts.  This matches Cassandra's
112    /// `mergeNearestBins` approach.
113    fn merge_closest_bins(&mut self) {
114        if self.bins.len() < 2 {
115            return;
116        }
117
118        // Find the pair of adjacent keys with the smallest gap.
119        let keys: Vec<i64> = self.bins.keys().copied().collect();
120        let mut min_gap = i64::MAX;
121        let mut merge_idx = 0usize;
122        for i in 0..keys.len() - 1 {
123            let gap = keys[i + 1] - keys[i];
124            if gap < min_gap {
125                min_gap = gap;
126                merge_idx = i;
127            }
128        }
129
130        let k1 = keys[merge_idx];
131        let k2 = keys[merge_idx + 1];
132        let v1 = self.bins.remove(&k1).unwrap_or(0);
133        let v2 = self.bins.remove(&k2).unwrap_or(0);
134        let total = v1 + v2;
135        // Weighted average key (same formula as Cassandra)
136        let merged_key = if total == 0 {
137            (k1 + k2) / 2
138        } else {
139            (k1 * v1 + k2 * v2) / total
140        };
141        *self.bins.entry(merged_key).or_insert(0) += total;
142    }
143
144    /// Iterate over bins in ascending key order as `(point_as_f64, count)` pairs.
145    fn entries(&self) -> impl Iterator<Item = (f64, i64)> + '_ {
146        self.bins.iter().map(|(&k, &v)| (k as f64, v))
147    }
148}

177    pub min_local_deletion_time: i32,

179    pub max_local_deletion_time: i32,

209    ///
210    /// Serialised as the `da`-format `StatsMetadata.hasPartitionLevelDeletions`

215    /// contributes a `partition_tombstone`. Ignored by the legacy BIG (nb/oa)
216    /// STATS body, which never serialises this field.
217    pub has_partition_level_deletions: bool,
218}

242    /// Create a new empty statistics metadata
243    pub fn new() -> Self {

247    /// Update timestamp range with a new timestamp value.
248    ///
249    /// LIVE deletion markers must never enter the min/max aggregates (issue
250    /// #851, Cassandra `d5bc7fb5`). Cassandra encodes `DeletionTime.LIVE` with
251    /// `markedForDeleteAt = Long.MIN_VALUE`, and a `NO_DELETION` / absent
252    /// liveness timestamp surfaces as `Long.MAX_VALUE`. Folding either sentinel
253    /// into `minTimestamp` / `maxTimestamp` would poison the stats, so they are
254    /// skipped here at the single aggregation chokepoint.
255    pub fn update_timestamp(&mut self, timestamp: i64) {
256        if Self::is_live_timestamp(timestamp) {
257            return;
258        }
259        self.min_timestamp = self.min_timestamp.min(timestamp);
260        self.max_timestamp = self.max_timestamp.max(timestamp);
261    }
262
263    /// Update local deletion time range (for tombstones) and the drop-time histogram.
264    ///
265    /// Call this for every tombstone local-deletion-time encountered while writing:
266    /// cell tombstones, row deletions, range tombstones, and partition tombstones.
267    ///
268    /// A LIVE marker (`DeletionTime.LIVE.localDeletionTime == Integer.MAX_VALUE`,
269    /// the same value as `Cell.NO_DELETION_TIME`) is not a tombstone: it must not
270    /// pull `minLocalDeletionTime` down to the sentinel nor inflate the tombstone
271    /// drop-time histogram (issue #851, Cassandra `d5bc7fb5`).
272    pub fn update_local_deletion_time(&mut self, deletion_time: i32) {
273        if Self::is_live_local_deletion_time(deletion_time) {
274            return;
275        }
276        self.min_local_deletion_time = self.min_local_deletion_time.min(deletion_time);
277        self.max_local_deletion_time = self.max_local_deletion_time.max(deletion_time);
278        self.tombstone_histogram.update(deletion_time);
279    }
280
281    /// True when `timestamp` is a LIVE / `NO_DELETION` marker rather than a real
282    /// deletion timestamp. CQLite writes `DeletionTime.LIVE` as `Long.MIN_VALUE`
283    /// (see `data_writer.rs`); Cassandra's `NO_DELETION` / `NO_TIMESTAMP` sentinel
284    /// is `Long.MAX_VALUE`. Both mean "no deletion" and must be excluded from
285    /// timestamp aggregation.
286    fn is_live_timestamp(timestamp: i64) -> bool {
287        timestamp == i64::MIN || timestamp == i64::MAX
288    }
289
290    /// True when `deletion_time` is a LIVE marker (`Integer.MAX_VALUE`) rather
291    /// than a real tombstone local-deletion-time.
292    fn is_live_local_deletion_time(deletion_time: i32) -> bool {
293        deletion_time == i32::MAX
294    }
295
296    /// Update TTL range
297    pub fn update_ttl(&mut self, ttl: i32) {
298        if ttl > 0 {
299            self.min_ttl = self.min_ttl.min(ttl);
300            self.max_ttl = self.max_ttl.max(ttl);
301        }
302    }
303
304    /// Increment partition count
305    pub fn increment_partition_count(&mut self) {
306        self.partition_count += 1;
307    }
308
309    /// Record a partition key in the SSTable key range.
310    ///
311    /// Partitions are written in ascending token order (enforced by
312    /// `SSTableWriter::write_partition`), so the FIRST key seen is the lowest and
313    /// the LAST is the highest. Used to populate the `da`-format
314    /// `StatsMetadata.firstKey`/`lastKey` (`hasKeyRange`). The clone is bounded by
315    /// the partition-key size (typically tens of bytes), not the partition data.
316    pub fn update_key_range(&mut self, key: &[u8]) {
317        if self.first_key.is_none() {
318            self.first_key = Some(key.to_vec());
319        }
320        self.last_key = Some(key.to_vec());
321    }
322
323    /// Record that a partition-level deletion (partition tombstone) was written.
324    ///
325    /// Once set, this stays `true` for the lifetime of the SSTable and drives the
326    /// `da`-format `StatsMetadata.hasPartitionLevelDeletions` marker. Called by
327    /// `SSTableWriter::write_partition` for any partition carrying a
328    /// `partition_tombstone`.
329    pub fn mark_partition_level_deletion(&mut self) {
330        self.has_partition_level_deletions = true;
331    }
332
333    /// Increment row count
334    pub fn increment_row_count(&mut self) {
335        self.row_count += 1;
336    }
337
338    /// Add to column count
339    pub fn add_column_count(&mut self, count: u64) {
340        self.column_count += count;
341    }
342
343    /// Add to total rows size
344    pub fn add_rows_size(&mut self, size: u64) {
345        self.total_rows_size += size;
346    }
347
348    /// Finalize metadata before writing (normalize sentinel values)
349    pub fn finalize(&mut self) {
350        // If no timestamps were recorded, set to 0
351        if self.min_timestamp == i64::MAX {
352            self.min_timestamp = 0;
353        }
354        if self.max_timestamp == i64::MIN {
355            self.max_timestamp = 0;
356        }
357
358        // If no deletion times were recorded, set to 0
359        if self.min_local_deletion_time == i32::MAX {
360            self.min_local_deletion_time = 0;
361        }
362        if self.max_local_deletion_time == i32::MIN {
363            self.max_local_deletion_time = 0;
364        }
365
366        // If no TTLs were recorded, set min_ttl to 0
367        if self.min_ttl == i32::MAX {
368            self.min_ttl = 0;
369        }
370    }

373/// Convert a CQL type name to Cassandra internal marshal type.
374///
375/// This is the reverse of `convert_marshal_type_to_cql` in enhanced_statistics_parser.rs.
376/// Used when writing the SERIALIZATION_HEADER component of Statistics.db.
377///
378/// Handles:
379/// - Primitive types: text, int, bigint, uuid, etc.
380/// - Collections: list<T>, set<T>, map<K,V>
381/// - Frozen wrappers: frozen<list<T>>, frozen<map<K,V>>
382/// - Tuples: tuple<T1, T2, ...>
383pub(crate) fn cql_type_to_marshal_type(cql_type: &str) -> String {
384    // Already a marshal string (e.g. a UDT column normalized to UserType(...),
385    // or a column type read back from an input SSTable's SerializationHeader):
386    // return it verbatim. The marshal grammar is case-sensitive (UserType,
387    // Int32Type, ...), so this MUST happen before the lowercasing below, and the
388    // original-case string MUST be preserved. Without this, an already-marshaled
389    // type would fall through to BytesType, advertising the wrong type in the
390    // header while Data.db carries the real (e.g. complex UDT) cells (#929).
391    let raw = cql_type.trim();
392    if raw
393        .to_lowercase()
394        .starts_with("org.apache.cassandra.db.marshal.")
395    {
396        return raw.to_string();
397    }
398
399    // Normalize to lowercase for case-insensitive matching.
400    // CQL type names are case-insensitive, and the parser may preserve
401    // original case from CQL files (e.g., "SET<TEXT>" instead of "set<text>").
402    let trimmed = cql_type.trim().to_lowercase();
403    let trimmed = trimmed.as_str();
404    let prefix = "org.apache.cassandra.db.marshal.";
405
406    // Handle parameterized types: list<T>, set<T>, map<K,V>, frozen<T>, tuple<T1,T2>
407    if let Some(inner) = strip_cql_wrapper(trimmed, "list") {
408        return format!("{prefix}ListType({})", cql_type_to_marshal_type(inner));
409    }
410    if let Some(inner) = strip_cql_wrapper(trimmed, "set") {
411        return format!("{prefix}SetType({})", cql_type_to_marshal_type(inner));
412    }
413    if let Some(inner) = strip_cql_wrapper(trimmed, "map") {
414        let args = split_cql_type_args(inner);
415        if args.len() == 2 {
416            return format!(
417                "{prefix}MapType({},{})",
418                cql_type_to_marshal_type(args[0]),
419                cql_type_to_marshal_type(args[1])
420            );
421        }
422        // Malformed map type — fall through to BytesType
423    }
424    if let Some(inner) = strip_cql_wrapper(trimmed, "frozen") {
425        return format!("{prefix}FrozenType({})", cql_type_to_marshal_type(inner));
426    }
427    if let Some(inner) = strip_cql_wrapper(trimmed, "tuple") {
428        let args = split_cql_type_args(inner);
429        let components: Vec<String> = args.iter().map(|a| cql_type_to_marshal_type(a)).collect();
430        return format!("{prefix}TupleType({})", components.join(","));
431    }
432
433    // Primitive types
434    match trimmed {
435        "text" | "varchar" => format!("{prefix}UTF8Type"),
436        "int" => format!("{prefix}Int32Type"),
437        "bigint" => format!("{prefix}LongType"),
438        "smallint" => format!("{prefix}ShortType"),
439        "tinyint" => format!("{prefix}ByteType"),
440        "float" => format!("{prefix}FloatType"),
441        "double" => format!("{prefix}DoubleType"),
442        "boolean" => format!("{prefix}BooleanType"),
443        "blob" => format!("{prefix}BytesType"),
444        "uuid" => format!("{prefix}UUIDType"),
445        "timeuuid" => format!("{prefix}TimeUUIDType"),
446        "timestamp" => format!("{prefix}TimestampType"),
447        "date" => format!("{prefix}SimpleDateType"),
448        "time" => format!("{prefix}TimeType"),
449        "duration" => format!("{prefix}DurationType"),
450        "inet" => format!("{prefix}InetAddressType"),
451        "ascii" => format!("{prefix}AsciiType"),
452        "decimal" => format!("{prefix}DecimalType"),
453        "varint" => format!("{prefix}IntegerType"),
454        "counter" => format!("{prefix}CounterColumnType"),
455        // Fallback: use BytesType for unknown types
456        _ => format!("{prefix}BytesType"),
457    }
458}

460/// Strip a CQL wrapper type like `list<inner>` and return the inner string.
461/// Returns None if `cql_type` does not start with `wrapper<`.
462fn strip_cql_wrapper<'a>(cql_type: &'a str, wrapper: &str) -> Option<&'a str> {
463    let pattern = format!("{}<", wrapper);
464    if let Some(rest) = cql_type.strip_prefix(&pattern) {
465        // Find the matching closing '>' (handling nested angle brackets)
466        let mut depth = 1;
467        for (i, ch) in rest.char_indices() {
468            match ch {
469                '<' => depth += 1,
470                '>' => {
471                    depth -= 1;
472                    if depth == 0 {
473                        return Some(rest[..i].trim());
474                    }
475                }
476                _ => {}
477            }
478        }
479    }
480    None
481}

483/// Split CQL type arguments at top-level commas (respecting nested angle brackets).
484/// E.g. `"int, map<text, int>"` → `["int", "map<text, int>"]`
485fn split_cql_type_args(s: &str) -> Vec<&str> {
486    let mut result = Vec::new();
487    let mut depth = 0;
488    let mut start = 0;
489    for (i, ch) in s.char_indices() {
490        match ch {
491            '<' => depth += 1,
492            '>' => depth -= 1,
493            ',' if depth == 0 => {
494                result.push(s[start..i].trim());
495                start = i + 1;
496            }
497            _ => {}
498        }
499    }
500    let last = s[start..].trim();
501    if !last.is_empty() {
502        result.push(last);
503    }
504    result
505}

507/// Statistics.db component writer
508///
509/// Writes the Statistics.db file with metadata for SSTable delta encoding.
510#[derive(Debug)]
511pub struct StatisticsWriter {
512    /// Path to the Statistics.db file to write
513    path: PathBuf,
514    /// Emit the Cassandra-canonical `da` (BtiFormat) `StatsMetadata` layout
515    /// instead of the legacy `nb` layout.
516    ///
517    /// The `da` STATS component differs from `nb` in the fields gated by the
518    /// BtiFormat version flags (all true for `da` except `hasLegacyMinMax`):
519    /// `hasUIntDeletionTime`, `hasImprovedMinMax` (clusteringTypes + a covered
520    /// `Slice` instead of legacy min/max value lists), `hasIsTransient`,
521    /// `hasOriginatingHostId`, `hasPartitionLevelDeletionsPresenceMarker`,
522    /// `hasKeyRange` (first/last key) and `hasTokenSpaceCoverage`. Cassandra's
523    /// `sstabledump`/`sstablemetadata` deserialize a `da`-descriptor
524    /// Statistics.db with this layout and reject the `nb` layout (a
525    /// `Slice.<init>` assertion fires while reading `coveredClustering`).
526    /// Authority: cassandra-5.0.0 `StatsMetadata.StatsMetadataSerializer` +
527    /// `BtiFormat.BtiVersion` version flags.
528    bti: bool,
529}

531impl StatisticsWriter {
532    /// Create a new Statistics.db writer for the legacy `nb`/`oa` BIG layout.
533    ///
534    /// # Arguments
535    /// * `path` - Path where Statistics.db will be written
536    pub fn new(path: PathBuf) -> Self {
537        Self { path, bti: false }
538    }
539
540    /// Create a new Statistics.db writer that emits the Cassandra-canonical `da`
541    /// (BtiFormat) `StatsMetadata` layout.
542    pub fn new_bti(path: PathBuf) -> Self {
543        Self { path, bti: true }
544    }
545
546    /// Write Statistics.db file with the given metadata
547    ///
548    /// Generates a Cassandra 5.0 compatible Statistics.db file with full TOC structure:
549    /// 1. TOC header with component count and checksums
550    /// 2. VALIDATION component (validator class name)
551    /// 3. COMPACTION component (minimal metadata)
552    /// 4. STATS component (EncodingStats with baselines)
553    /// 5. SERIALIZATION_HEADER component (schema-derived or minimal stub)
554    ///
555    /// Each component is followed by a CRC32 checksum for validation.
556    ///
557    /// # Arguments
558    /// * `metadata` - Statistics metadata to write
559    /// * `schema` - Optional table schema for populating serialization header
560    ///
561    /// # Returns
562    /// `Ok(())` on success, or an error if writing fails
563    pub fn write(&self, metadata: &StatisticsMetadata, schema: Option<&TableSchema>) -> Result<()> {
564        let mut meta = metadata.clone();
565        meta.finalize();
566
567        // Build component data
568        let validation_data = self.build_validation_component()?;
569        let compaction_data = self.build_compaction_component()?;
570        // The STATS body is version-gated: BTI (`da`) emits the BtiFormat
571        // `StatsMetadata` layout (covered-clustering Slice, uint deletion times,
572        // key range, token-space coverage); BIG (`nb`/`oa`) emits the legacy
573        // layout. `schema` is needed for the `da` clustering-type list.
574        let stats_data = if self.bti {
575            self.build_stats_component_da(&meta, schema)?
576        } else {
577            self.build_stats_component(&meta)?
578        };
579        // Use pre-finalize metadata for the SerializationHeader EncodingStats.
580        // The baselines in the header MUST match those used by the DataWriter for
581        // delta encoding. The DataWriter uses the raw (pre-finalize) metadata values.
582        let header_data = self.build_serialization_header_component(schema, metadata)?;
583
584        // Calculate component offsets
585        // TOC structure: 4 (count) + 4 (checksum) + (4*8) TOC entries + 4 (checksum) = 44 bytes
586        let toc_size = 4 + 4 + (NUM_COMPONENTS as usize * 8) + 4;
587        let mut offset = toc_size;
588
589        let validation_offset = offset;
590        offset += validation_data.len() + 4; // +4 for component checksum
591
592        let compaction_offset = offset;
593        offset += compaction_data.len() + 4;
594
595        let stats_offset = offset;
596        offset += stats_data.len() + 4;
597
598        let header_offset = offset;
599        // header_data has its own checksum at the end
600
601        // Verify all offsets fit in u32 (Statistics.db should never exceed 4GB)
602        if offset > u32::MAX as usize {
603            return Err(Error::Storage(format!(
604                "Statistics.db too large: {} bytes exceeds u32::MAX",
605                offset
606            )));
607        }
608
609        // Build the complete file
610        let mut buffer = Vec::new();
611        let mut crc = crc32fast::Hasher::new();
612
613        // Write component count
614        buffer.write_all(&NUM_COMPONENTS.to_be_bytes())?;
615        self.update_checksum_int(&mut crc, NUM_COMPONENTS);
616
617        // Write first checksum (after count)
618        let checksum1 = crc.clone().finalize();
619        buffer.write_all(&checksum1.to_be_bytes())?;
620
621        // Reset CRC for TOC (we'll recompute cumulatively)
622        crc = crc32fast::Hasher::new();
623        self.update_checksum_int(&mut crc, NUM_COMPONENTS);
624
625        // Write TOC entries (type, offset pairs)
626        self.write_toc_entry(
627            &mut buffer,
628            &mut crc,
629            METADATA_TYPE_VALIDATION,
630            validation_offset as u32,
631        )?;
632        self.write_toc_entry(
633            &mut buffer,
634            &mut crc,
635            METADATA_TYPE_COMPACTION,
636            compaction_offset as u32,
637        )?;
638        self.write_toc_entry(
639            &mut buffer,
640            &mut crc,
641            METADATA_TYPE_STATS,
642            stats_offset as u32,
643        )?;
644        self.write_toc_entry(
645            &mut buffer,
646            &mut crc,
647            METADATA_TYPE_SERIALIZATION_HEADER,
648            header_offset as u32,
649        )?;
650
651        // Write TOC checksum (cumulative from count)
652        let toc_checksum = crc.finalize();
653        buffer.write_all(&toc_checksum.to_be_bytes())?;
654
655        // Write components with per-component checksums
656        self.write_component(&mut buffer, &validation_data)?;
657        self.write_component(&mut buffer, &compaction_data)?;
658        self.write_component(&mut buffer, &stats_data)?;
659        self.write_component(&mut buffer, &header_data)?;
660
661        // Write to file
662        std::fs::write(&self.path, buffer).map_err(|e| {
663            Error::Storage(format!(
664                "Failed to write Statistics.db to {}: {}",
665                self.path.display(),
666                e
667            ))
668        })?;
669
670        Ok(())
671    }

733    ///
734    /// Format (CompactionMetadata.java):
735    /// - cardinality estimator (i32 BE length + HyperLogLogPlus bytes)
736    ///
737    /// We write a minimal valid empty HyperLogLogPlus sketch.
738    fn build_compaction_component(&self) -> Result<Vec<u8>> {
739        let mut buffer = Vec::new();
740
741        // Cardinality estimator: ByteArrayUtil.writeWithLength(bytes, out)
742        // Format: i32 BE length + data bytes
743        //
744        // Minimal valid HyperLogLogPlus(p=11, sp=25) in SPARSE format:
745        // - 4 bytes: version (-2 as i32 = 0xFFFFFFFE)
746        // - 1 byte: p = 11 (0x0B)
747        // - 1 byte: sp = 25 (0x19)
748        // - 1 byte: format type = SPARSE (0x01)
749        // - 4 bytes: tempSetSize = 0
750        // - 4 bytes: sparseSetSize = 0

757            0x01, // format = SPARSE
758            0x00, 0x00, 0x00, 0x00, // tempSetSize = 0
759            0x00, 0x00, 0x00, 0x00, // sparseSetSize = 0
760        ];
761
762        // Write length prefix (i32 BE)
763        buffer.write_all(&(HLL_DATA.len() as i32).to_be_bytes())?;

765        // Write HLL data
766        buffer.write_all(&HLL_DATA)?;

868    /// Build the STATS component for the Cassandra-canonical `da` (BtiFormat)
869    /// layout.
870    ///
871    /// Field order and gating follow cassandra-5.0.0
872    /// `StatsMetadata.StatsMetadataSerializer.serialize` evaluated for
873    /// `BtiFormat.BtiVersion` (all version flags `true` except `hasLegacyMinMax`,
874    /// which is `false`):
875    ///
876    /// 1.  estimatedPartitionSize (EstimatedHistogram)
877    /// 2.  estimatedCellPerPartitionCount (EstimatedHistogram)
878    /// 3.  commitLogUpperBound (CommitLogPosition)
879    /// 4.  minTimestamp, maxTimestamp (long)
880    /// 5.  min/maxLocalDeletionTime as **unsigned int** (`hasUIntDeletionTime`);
881    ///     `NO_DELETION_TIME` (Long.MAX) maps to `0xFFFFFFFF`.
882    /// 6.  minTTL, maxTTL (int)
883    /// 7.  compressionRatio (double)
884    /// 8.  estimatedTombstoneDropTime (TombstoneHistogram)
885    /// 9.  sstableLevel (int), repairedAt (long)
886    /// 10. improvedMinMax (`!hasLegacyMinMax && hasImprovedMinMax`):
887    ///     clusteringTypes list + coveredClustering `Slice`. We emit
888    ///     `Slice.ALL` (BOTTOM..TOP) — a valid, conservative covering slice that
889    ///     Cassandra accepts and that matches the `Covered clusterings: [, ]`
890    ///     shown by `sstablemetadata` on the real `da` fixtures.
891    /// 11. hasLegacyCounterShards (bool)
892    /// 12. totalColumnsSet, totalRows (long)
893    /// 13. commitLogLowerBound (CommitLogPosition), commitLogIntervals (IntervalSet)
894    /// 14. pendingRepair (byte 0 = null)
895    /// 15. isTransient (bool)
896    /// 16. originatingHostId (byte 0 = null)
897    /// 17. hasPartitionLevelDeletions (bool)
898    /// 18. firstKey, lastKey (vint-length ByteBuffer) — `hasKeyRange`
899    /// 19. tokenSpaceCoverage (double) — `hasTokenSpaceCoverage`
900    fn build_stats_component_da(
901        &self,
902        metadata: &StatisticsMetadata,
903        schema: Option<&TableSchema>,
904    ) -> Result<Vec<u8>> {
905        let mut buffer = Vec::new();
906
907        // 1-2. EstimatedHistogram estimatedPartitionSize / estimatedCellPerPartitionCount
908        self.write_estimated_histogram(&mut buffer)?;
909        self.write_estimated_histogram(&mut buffer)?;
910
911        // 3. CommitLogPosition commitLogUpperBound (NONE)
912        buffer.write_all(&(-1i64).to_be_bytes())?; // segmentId
913        buffer.write_all(&0i32.to_be_bytes())?; // position
914
915        // 4. minTimestamp, maxTimestamp
916        buffer.write_all(&metadata.min_timestamp.to_be_bytes())?;
917        buffer.write_all(&metadata.max_timestamp.to_be_bytes())?;
918
919        // 5. min/maxLocalDeletionTime — unsigned int (hasUIntDeletionTime).
920        // No tombstones (sentinel) → NO_DELETION_TIME_UNSIGNED_INTEGER = 0xFFFFFFFF;
921        // otherwise the low 32 bits of the (seconds-since-epoch) deletion time
922        // (Cassandra `CassandraUInt.fromLong` == `(int) value`).
923        let min_ldt_uint: u32 = if metadata.min_local_deletion_time == 0 {
924            0xFFFF_FFFF
925        } else {
926            metadata.min_local_deletion_time as u32
927        };
928        let max_ldt_uint: u32 = if metadata.max_local_deletion_time == 0 {
929            0xFFFF_FFFF
930        } else {
931            metadata.max_local_deletion_time as u32
932        };
933        buffer.write_all(&min_ldt_uint.to_be_bytes())?;
934        buffer.write_all(&max_ldt_uint.to_be_bytes())?;
935
936        // 6. minTTL, maxTTL
937        buffer.write_all(&metadata.min_ttl.to_be_bytes())?;
938        buffer.write_all(&metadata.max_ttl.to_be_bytes())?;
939
940        // 7. compressionRatio (-1.0 = unknown)
941        buffer.write_all(&(-1.0f64).to_be_bytes())?;
942
943        // 8. TombstoneHistogram estimatedTombstoneDropTime.
944        //
945        // The `da` (BtiFormat) version resolves `TombstoneHistogram.getSerializer`
946        // to the modern `HistogramSerializer` (long point + int value per bin),
947        // NOT the legacy serializer (double + long) used by older versions. Using
948        // the legacy entry encoding here mis-sizes the body and derails the
949        // subsequent `coveredClustering` Slice deserialization in Cassandra.
950        self.write_tombstone_histogram_modern(&mut buffer, &metadata.tombstone_histogram)?;
951
952        // 9. sstableLevel, repairedAt
953        buffer.write_all(&0i32.to_be_bytes())?;
954        buffer.write_all(&0i64.to_be_bytes())?;
955
956        // 10. improvedMinMax: clusteringTypes list + coveredClustering Slice.
957        //
958        // AbstractTypeSerializer.serializeList: unsigned-VInt count, then each
959        // type as an unsigned-VInt-length-prefixed UTF-8 marshal-class string —
960        // the exact encoding used for the SERIALIZATION_HEADER clusteringTypes.
961        let clustering_types: Vec<String> = schema
962            .map(|s| {
963                s.clustering_keys
964                    .iter()
965                    .map(|ck| cql_type_to_marshal_type(&ck.data_type))
966                    .collect()
967            })
968            .unwrap_or_default();
969        buffer.write_all(&encode_vuint(clustering_types.len() as u64))?;
970        for ty in &clustering_types {
971            buffer.write_all(&encode_vuint(ty.len() as u64))?;
972            buffer.write_all(ty.as_bytes())?;
973        }
974        // coveredClustering = Slice.ALL = (BOTTOM, TOP). A ClusteringBound
975        // serialises as `[byte kind ordinal][short size][values...]`; BOTTOM and
976        // TOP are empty bounds (size 0). Kind ordinals (ClusteringPrefix.Kind):
977        // INCL_START_BOUND = 1 (BOTTOM), INCL_END_BOUND = 6 (TOP).
978        const KIND_INCL_START_BOUND: u8 = 1;
979        const KIND_INCL_END_BOUND: u8 = 6;
980        buffer.write_all(&[KIND_INCL_START_BOUND])?;
981        buffer.write_all(&0u16.to_be_bytes())?; // start size = 0
982        buffer.write_all(&[KIND_INCL_END_BOUND])?;
983        buffer.write_all(&0u16.to_be_bytes())?; // end size = 0
984
985        // 11. hasLegacyCounterShards
986        buffer.write_all(&[0x00])?;
987
988        // 12. totalColumnsSet, totalRows
989        buffer.write_all(&metadata.column_count.to_be_bytes())?;
990        buffer.write_all(&metadata.row_count.to_be_bytes())?;
991
992        // 13. commitLogLowerBound (NONE) + commitLogIntervals (empty IntervalSet)
993        buffer.write_all(&(-1i64).to_be_bytes())?; // segmentId
994        buffer.write_all(&0i32.to_be_bytes())?; // position
995        buffer.write_all(&0i32.to_be_bytes())?; // interval set size = 0
996
997        // 14. pendingRepair (null)
998        buffer.write_all(&[0x00])?;
999
1000        // 15. isTransient
1001        buffer.write_all(&[0x00])?;
1002
1003        // 16. originatingHostId (null)
1004        buffer.write_all(&[0x00])?;
1005
1006        // 17. hasPartitionLevelDeletions
1007        //
1008        // cassandra-5.0.0 StatsMetadata.StatsMetadataSerializer.serialize line 495:
1009        //   out.writeBoolean(component.hasPartitionLevelDeletions)
1010        // DataOutput.writeBoolean emits a single byte: 0x01 for true, 0x00 false.
1011        // Set when any partition written carries a partition-level tombstone.
1012        let has_partition_deletions = if metadata.has_partition_level_deletions {
1013            0x01u8
1014        } else {
1015            0x00u8
1016        };
1017        buffer.write_all(&[has_partition_deletions])?;
1018
1019        // 18. firstKey, lastKey (ByteBufferUtil.writeWithVIntLength:
1020        // unsigned-VInt length then the raw key bytes). Empty when no partitions.
1021        let first = metadata.first_key.as_deref().unwrap_or(&[]);
1022        let last = metadata.last_key.as_deref().unwrap_or(&[]);
1023        buffer.write_all(&encode_vuint(first.len() as u64))?;
1024        buffer.write_all(first)?;
1025        buffer.write_all(&encode_vuint(last.len() as u64))?;
1026        buffer.write_all(last)?;
1027
1028        // 19. tokenSpaceCoverage (double). Cassandra writes NaN when not computed
1029        // (sstablemetadata renders this as "Local token space coverage: NaN").
1030        buffer.write_all(&f64::NAN.to_be_bytes())?;
1031
1032        Ok(buffer)
1033    }

1037    /// Format:
1038    /// - int: bucket count (we use 2 for minimal valid histogram)
1039    /// - for each bucket: long offset + long count
1040    ///
1041    /// Minimal valid: 2 buckets (size-1=1 offset, size=2 counts)

1057    /// Serialise a `TombstoneHistogram` using Cassandra's legacy format (nb/mc).
1058    ///
1059    /// Binary layout (matches `TombstoneHistogram.LegacyHistogramSerializer`):
1060    /// ```text
1061    /// i32 BE  maxBinSize  — capacity: 100 when non-empty, 0 when empty
1062    /// i32 BE  size        — actual number of populated bins
1063    /// for each of the `size` bins (ascending point order):
1064    ///   f64 BE  point  — local-deletion-time bucket centre (as double)
1065    ///   i64 BE  value  — tombstone count for this bucket
1066    /// ```
1067    ///
1068    /// An empty histogram writes 8 bytes (`maxBinSize=0, size=0`), matching
1069    /// what Cassandra writes for SSTables with no tombstones.
1070    fn write_tombstone_histogram(
1071        &self,
1072        buffer: &mut Vec<u8>,
1073        histogram: &TombstoneHistogram,
1074    ) -> Result<()> {
1075        if histogram.is_empty() {
1076            // Empty: maxBinSize=0, size=0 — no entry pairs follow
1077            buffer.write_all(&0i32.to_be_bytes())?; // maxBinSize
1078            buffer.write_all(&0i32.to_be_bytes())?; // size
1079        } else {
1080            // Non-empty: maxBinSize=100 (the capacity constant), then the bins
1081            buffer.write_all(&TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE.to_be_bytes())?; // maxBinSize
1082            buffer.write_all(&histogram.size().to_be_bytes())?; // size
1083            for (point, value) in histogram.entries() {
1084                buffer.write_all(&point.to_be_bytes())?; // f64 BE point
1085                buffer.write_all(&value.to_be_bytes())?; // i64 BE value
1086            }
1087        }
1088        Ok(())
1089    }

1092    /// `HistogramSerializer` format.
1093    ///
1094    /// Identical framing to the legacy serializer (`i32 maxBinSize`, `i32 size`)
1095    /// but each bin entry is `i64 BE point` + `i32 BE value` (12 bytes), where
1096    /// the legacy serializer used `f64 point` + `i64 value` (16 bytes). The
1097    /// version split: `StatsMetadata.StatsMetadataSerializer` resolves
1098    /// `TombstoneHistogram.getSerializer(version)` to the modern
1099    /// `HistogramSerializer` for `oa`+ (incl. `BtiFormat.BtiVersion` `da`) and
1100    /// the `LegacyHistogramSerializer` for older versions.
1101    ///
1102    /// Authority: repo `docs/sstables-definitive-guide/statistics-db-writer-spec.md`
1103    /// (TombstoneHistogram legacy vs modern note) and
1104    /// cassandra-5.0.0 `TombstoneHistogram.java` (`HistogramSerializer` ->
1105    /// `long` point + `int` value).
1106    ///
1107    /// ```text
1108    /// i32 BE  maxBinSize  — capacity: 100 when non-empty, 0 when empty
1109    /// i32 BE  size        — actual number of populated bins
1110    /// for each of the `size` bins (ascending point order):
1111    ///   i64 BE  point  — local-deletion-time bucket centre (as long)
1112    ///   i32 BE  value  — tombstone count for this bucket
1113    /// ```
1114    fn write_tombstone_histogram_modern(
1115        &self,
1116        buffer: &mut Vec<u8>,
1117        histogram: &TombstoneHistogram,
1118    ) -> Result<()> {
1119        if histogram.is_empty() {
1120            buffer.write_all(&0i32.to_be_bytes())?; // maxBinSize
1121            buffer.write_all(&0i32.to_be_bytes())?; // size
1122        } else {
1123            buffer.write_all(&TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE.to_be_bytes())?; // maxBinSize
1124            buffer.write_all(&histogram.size().to_be_bytes())?; // size
1125            for (point, value) in histogram.entries() {
1126                // Modern: long point (truncate the double bucket centre) + int value.
1127                buffer.write_all(&(point as i64).to_be_bytes())?; // i64 BE point
1128                buffer.write_all(&(value as i32).to_be_bytes())?; // i32 BE value
1129            }
1130        }
1131        Ok(())
1132    }
1133
1134    /// Build SERIALIZATION_HEADER component (MetadataType ordinal 3)
1135    ///
1136    /// Format (SerializationHeader.java Serializer, lines 594-603):
1137    /// - EncodingStats: 3 unsigned VInts (minTimestamp, minLocalDeletionTime, minTTL deltas from epochs)
1138    /// - keyType: VInt length + UTF-8 type string
1139    /// - clusteringTypes: unsigned VInt count + list of types
1140    /// - staticColumns: unsigned VInt count + map of (column name, type)
1141    /// - regularColumns: unsigned VInt count + map of (column name, type)
1142    ///
1143    /// When `schema` is Some, populates keyType, clustering types, and column
1144    /// names/types from the actual table schema. When None, falls back to a
1145    /// minimal stub (BytesType, zero columns).
1146    ///
1147    /// # Column-set encoding (>64 columns)
1148    ///
1149    /// The static/regular column sets are encoded exactly as Cassandra's
1150    /// `SerializationHeader.Serializer.writeColumnsWithTypes`
1151    /// (cassandra-5.0.0 `SerializationHeader.java` lines 489-497): an unsigned-VInt
1152    /// column count followed by `count` `(VInt-length name, VInt-length marshal type)`
1153    /// pairs. This path has **no** 64-column limit and never uses a bitmap.
1154    ///
1155    /// The 64-bit bitmap encoding lives in `Columns.serializer.serializeSubset`
1156    /// (`Columns.java` lines 503-531) and is only used to serialise a per-row column
1157    /// subset against a pre-shared superset (Data.db rows / inter-node messaging),
1158    /// where `supersetCount < 64` selects the bitmap and `>= 64` switches to a VInt
1159    /// delta list. It is never used for the SSTable SERIALIZATION_HEADER, so wide
1160    /// tables (>64 columns) round-trip losslessly here.
1161    fn build_serialization_header_component(
1162        &self,
1163        schema: Option<&TableSchema>,
1164        metadata: &StatisticsMetadata,
1165    ) -> Result<Vec<u8>> {
1166        let mut buffer = Vec::new();
1167
1168        // EncodingStats: 3 unsigned VInts representing deltas from epochs.
1169        // These baselines MUST match the values used by DataWriter for delta encoding.
1170        // Cassandra: EncodingStats.Serializer.serialize() writes:
1171        //   writeUnsignedVInt(minTimestamp - TIMESTAMP_EPOCH)
1172        //   writeUnsignedVInt(minLocalDeletionTime - DELETION_TIME_EPOCH)
1173        //   writeUnsignedVInt(minTTL - TTL_EPOCH)
1174
1175        // minTimestamp delta from epoch
1176        let min_ts = if metadata.min_timestamp == i64::MAX {
1177            // No data recorded: use epoch as baseline
1178            TIMESTAMP_EPOCH as u64
1179        } else {
1180            metadata.min_timestamp as u64
1181        };
1182        let min_ts_delta = min_ts.wrapping_sub(TIMESTAMP_EPOCH as u64);
1183        buffer.write_all(&encode_vuint(min_ts_delta))?;
1184
1185        // minLocalDeletionTime delta from epoch.
1186        //
1187        // Cassandra: `EncodingStats.Serializer.serialize` writes the local-deletion
1188        // baseline with `writeUnsignedVInt32(minLocalDeletionTime - DELETION_TIME_EPOCH)`
1189        // (both operands are Java `int`s), and the reader recovers it with
1190        // `readUnsignedVInt32()`, which runs `VIntCoding.checkedCast` and REJECTS any
1191        // decoded value that does not round-trip through a SIGNED 32-bit `int`
1192        // (`(int)value != value`). The on-disk form therefore carries the SIGN-EXTENDED
1193        // delta: for a small LDT like 2, `2 - DELETION_TIME_EPOCH` is the negative int
1194        // `-1442879998`, written as the sign-extended 64-bit VInt `0xFFFFFFFFAA…` which
1195        // `readUnsignedVInt32` accepts and folds back to `2`. Truncating to a bare `u32`
1196        // (e.g. `2852087298`) is OUT OF RANGE for `checkedCast` and makes Cassandra's
1197        // `sstabledump` reject the SSTable (verified live against `cassandra:5.0`).
1198        //
1199        // Casting the i32 delta to `u64` sign-extends in Rust exactly as Java's
1200        // `writeUnsignedVInt32` requires, and it also handles a far-future LDT stored as
1201        // a negative i32 bit pattern identically (the bit pattern IS the signed int the
1202        // reader expects). This mirrors the DataWriter per-row deletion deltas.
1203        let min_ldt = if metadata.min_local_deletion_time == i32::MAX {
1204            // No deletions: use Integer.MAX_VALUE as baseline (DeletionTime.LIVE)
1205            i32::MAX
1206        } else {
1207            metadata.min_local_deletion_time
1208        };
1209        let min_del_delta = (min_ldt.wrapping_sub(DELETION_TIME_EPOCH) as i64) as u64;
1210        buffer.write_all(&encode_vuint(min_del_delta))?;
1211
1212        // minTTL delta from TTL_EPOCH (TTL_EPOCH=0)
1213        let min_ttl = if metadata.min_ttl == i32::MAX {
1214            // No TTL: use 0 as baseline
1215            0u64
1216        } else {
1217            metadata.min_ttl as u64
1218        };
1219        let min_ttl_delta = min_ttl.wrapping_sub(TTL_EPOCH as u64);
1220        buffer.write_all(&encode_vuint(min_ttl_delta))?;
1221
1222        match schema {
1223            Some(s) => {
1224                // keyType: single PK → simple type, composite PK → CompositeType(...)
1225                let key_marshal = if s.partition_keys.len() > 1 {
1226                    let inner: Vec<String> = s
1227                        .partition_keys
1228                        .iter()
1229                        .map(|pk| cql_type_to_marshal_type(&pk.data_type))
1230                        .collect();
1231                    format!(
1232                        "org.apache.cassandra.db.marshal.CompositeType({})",
1233                        inner.join(",")
1234                    )
1235                } else if !s.partition_keys.is_empty() {
1236                    cql_type_to_marshal_type(&s.partition_keys[0].data_type)
1237                } else {
1238                    "org.apache.cassandra.db.marshal.BytesType".to_string()
1239                };
1240                buffer.write_all(&encode_vuint(key_marshal.len() as u64))?;
1241                buffer.write_all(key_marshal.as_bytes())?;
1242
1243                // clusteringTypes: VUInt count + for each CK: VUInt-length-prefixed marshal type
1244                buffer.write_all(&encode_vuint(s.clustering_keys.len() as u64))?;
1245                for ck in &s.clustering_keys {
1246                    let ck_marshal = cql_type_to_marshal_type(&ck.data_type);
1247                    buffer.write_all(&encode_vuint(ck_marshal.len() as u64))?;
1248                    buffer.write_all(ck_marshal.as_bytes())?;
1249                }
1250
1251                // Collect partition key and clustering key names for filtering
1252                let pk_names: std::collections::HashSet<&str> =
1253                    s.partition_keys.iter().map(|k| k.name.as_str()).collect();
1254                let ck_names: std::collections::HashSet<&str> =
1255                    s.clustering_keys.iter().map(|k| k.name.as_str()).collect();
1256
1257                // staticColumns: filter for is_static && not PK/CK, sorted alphabetically
1258                let mut static_cols: Vec<_> = s
1259                    .columns
1260                    .iter()
1261                    .filter(|c| {
1262                        c.is_static
1263                            && !pk_names.contains(c.name.as_str())
1264                            && !ck_names.contains(c.name.as_str())
1265                    })
1266                    .collect();
1267                static_cols.sort_by(|a, b| a.name.cmp(&b.name));
1268                buffer.write_all(&encode_vuint(static_cols.len() as u64))?;
1269                for col in &static_cols {
1270                    // Column name: VUInt length + UTF-8 bytes
1271                    buffer.write_all(&encode_vuint(col.name.len() as u64))?;
1272                    buffer.write_all(col.name.as_bytes())?;
1273                    // Column type: VUInt length + marshal type bytes
1274                    let col_marshal = cql_type_to_marshal_type(&col.data_type);
1275                    buffer.write_all(&encode_vuint(col_marshal.len() as u64))?;
1276                    buffer.write_all(col_marshal.as_bytes())?;
1277                }
1278
1279                // regularColumns: filter for !is_static && not PK/CK, sorted alphabetically
1280                // Cassandra's SerializationHeader stores columns in natural order (alphabetical)
1281                let mut regular_cols: Vec<_> = s
1282                    .columns
1283                    .iter()
1284                    .filter(|c| {
1285                        !c.is_static
1286                            && !pk_names.contains(c.name.as_str())
1287                            && !ck_names.contains(c.name.as_str())
1288                    })
1289                    .collect();
1290                regular_cols.sort_by(|a, b| a.name.cmp(&b.name));
1291                buffer.write_all(&encode_vuint(regular_cols.len() as u64))?;
1292                for col in &regular_cols {
1293                    buffer.write_all(&encode_vuint(col.name.len() as u64))?;
1294                    buffer.write_all(col.name.as_bytes())?;
1295                    let col_marshal = cql_type_to_marshal_type(&col.data_type);
1296                    buffer.write_all(&encode_vuint(col_marshal.len() as u64))?;
1297                    buffer.write_all(col_marshal.as_bytes())?;
1298                }
1299            }
1300            None => {
1301                // Minimal stub: BytesType key, no clustering, no columns
1302                let key_type = b"org.apache.cassandra.db.marshal.BytesType";
1303                buffer.write_all(&encode_vuint(key_type.len() as u64))?;
1304                buffer.write_all(key_type)?;
1305
1306                // clusteringTypes: 0
1307                buffer.write_all(&encode_vuint(0))?;
1308                // staticColumns: 0
1309                buffer.write_all(&encode_vuint(0))?;
1310                // regularColumns: 0
1311                buffer.write_all(&encode_vuint(0))?;
1312            }
1313        }
1314
1315        Ok(buffer)
1316    }
1317}

2329    /// Parse a Cassandra SSTable SERIALIZATION_HEADER column-set the way Cassandra's
2330    /// `SerializationHeader.Serializer.readColumnsWithType` does (cassandra-5.0.0
2331    /// `SerializationHeader.java` lines 510-520):
2332    ///
2333    /// ```text
2334    /// unsigned-vint  count
2335    /// repeat count times:
2336    ///   vint-length-prefixed UTF-8  column name
2337    ///   vint-length-prefixed UTF-8  marshal type
2338    /// ```
2339    ///
2340    /// Returns the parsed `(name, marshal_type)` pairs and the slice remaining after
2341    /// the column set, so chained sets (static then regular) can be parsed.
2342    fn parse_columns_with_types(input: &[u8]) -> (Vec<(String, String)>, &[u8]) {
2343        use crate::parser::vint::parse_vuint;
2344
2345        let (mut rest, count) = parse_vuint(input).expect("column count vint");
2346        let mut cols = Vec::with_capacity(count as usize);
2347        for _ in 0..count {
2348            let (after_name_len, name_len) = parse_vuint(rest).expect("name length vint");
2349            let name =
2350                String::from_utf8(after_name_len[..name_len as usize].to_vec()).expect("name utf8");
2351            let after_name = &after_name_len[name_len as usize..];
2352
2353            let (after_type_len, type_len) = parse_vuint(after_name).expect("type length vint");
2354            let marshal =
2355                String::from_utf8(after_type_len[..type_len as usize].to_vec()).expect("type utf8");
2356            rest = &after_type_len[type_len as usize..];
2357
2358            cols.push((name, marshal));
2359        }
2360        (cols, rest)
2361    }

2504    /// Regression: the EncodingStats `minLocalDeletionTime` delta must be written
2505    /// as the SIGN-EXTENDED `int` delta `minLocalDeletionTime - DELETION_TIME_EPOCH`,
2506    /// because Cassandra recovers it with `readUnsignedVInt32()` →
2507    /// `VIntCoding.checkedCast`, which REJECTS any decoded value that does not
2508    /// round-trip through a signed 32-bit `int` (`(int)value != value`). A bare
2509    /// `u32` truncation of a small LDT (e.g. delta `2852087298` for LDT `2`) is out
2510    /// of range for `checkedCast` and makes `cassandra:5.0` `sstabledump` reject the
2511    /// SSTable (verified live; see tests/issue_911_bti_partition_deletion_stats.rs).
2512    ///
2513    /// A far-future LDT in `[2^31, 2^32)` is stored as a negative i32 bit pattern,
2514    /// which IS the signed int the reader expects, so the same sign-extending path
2515    /// handles it. This pins both: the small-LDT delta is a sign-extended 64-bit
2516    /// VInt, and the decoded value round-trips through a signed i32.
2517    #[test]
2518    fn test_serialization_header_ldt_baseline_sign_extends_for_checked_cast() {
2519        use crate::parser::vint::parse_vuint;
2520
2521        // Helper: decode the minLocalDeletionTime EncodingStats delta for a given
2522        // baseline and assert it round-trips through Cassandra's checkedCast (i.e.
2523        // the decoded u64, reinterpreted as i64, fits a signed i32).
2524        let decode_delta = |min_ldt: i32| -> u64 {
2525            let mut meta = StatisticsMetadata::new();
2526            meta.min_local_deletion_time = min_ldt;
2527            let writer = StatisticsWriter::new(PathBuf::from("test.db"));
2528            let bytes = writer
2529                .build_serialization_header_component(None, &meta)
2530                .expect("build header");
2531            let (rest, _min_ts_delta) = parse_vuint(&bytes).expect("minTimestamp delta");
2532            let (_rest, min_ldt_delta) = parse_vuint(rest).expect("minLocalDeletionTime delta");
2533            min_ldt_delta
2534        };
2535
2536        // checkedCast accepts `value` iff `(int)value == value` — i.e. the decoded
2537        // u64, reinterpreted as i64, equals its own truncation to i32.
2538        let passes_checked_cast = |delta: u64| -> bool {
2539            let v = delta as i64;
2540            (v as i32) as i64 == v
2541        };
2542
2543        // Small LDT (e.g. 2): the regression case. The delta is the negative int
2544        // `2 - DELETION_TIME_EPOCH`, sign-extended.
2545        let small = decode_delta(2);
2546        let expected_small = (2i32.wrapping_sub(DELETION_TIME_EPOCH) as i64) as u64;
2547        assert_eq!(small, expected_small);
2548        assert!(
2549            passes_checked_cast(small),
2550            "small-LDT delta must round-trip through a signed i32 (checkedCast), got {small:#x}"
2551        );
2552
2553        // Far-future LDT in [2^31, 2^32): a negative i32 bit pattern. Same path.
2554        let far_future = ((1u32 << 31) + 5) as i32;
2555        assert!(far_future < 0, "sanity: far-future LDT is a negative i32");
2556        let far = decode_delta(far_future);
2557        let expected_far = (far_future.wrapping_sub(DELETION_TIME_EPOCH) as i64) as u64;
2558        assert_eq!(far, expected_far);
2559        assert!(
2560            passes_checked_cast(far),
2561            "far-future LDT delta must round-trip through a signed i32 (checkedCast), got {far:#x}"
2562        );
2563    }
