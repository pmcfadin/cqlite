1//! Metric catalog — the single source of truth for CQLite metric names + units.
2//!
3//! Downstream observability issues (epic #1031) MUST import the constants from
4//! this module instead of hard-coding metric-name strings. Centralising the
5//! names here keeps the emitted telemetry consistent across subsystems and lets
6//! us evolve names in exactly one place.
7//!
8//! # Naming conventions
9//!
10//! Names follow the OpenTelemetry metric semantic conventions:
11//! - dot-separated, lower-snake namespaces, rooted under `cqlite.`
12//! - units use the UCUM annotations OTel recommends (`{row}`, `s`, `By`, `1`,
13//!   `{error}`, …); see [`unit`].
14//! - counters describe a monotonically increasing total; their name reflects the
15//!   thing being counted (rows, bytes, errors).
16//! - histograms describe a distribution (durations, sizes).
17//! - gauges describe a current value (in-flight, cache occupancy).
18//!
19//! # Attribute cardinality
20//!
21//! Every metric is documented with the *bounded* attribute set it may carry.
22//! NEVER attach unbounded values (raw error messages, partition keys, full
23//! queries) as attributes — those explode cardinality and cost. Bounded
24//! attribute keys live in [`attr`].
25
26/// Recommended UCUM units for the catalog metrics.

37    /// A count of SSTables.
38    pub const SSTABLES: &str = "{sstable}";
39    /// A count of errors.
40    pub const ERRORS: &str = "{error}";
41    /// A count of tombstones.
42    pub const TOMBSTONES: &str = "{tombstone}";
43    /// Bytes.
44    pub const BYTES: &str = "By";
45    /// Seconds (OTel prefers base-unit seconds for durations).
46    pub const SECONDS: &str = "s";
47    /// A count of OS threads (UCUM annotation).
48    pub const THREADS: &str = "{thread}";
49    /// A count of open file descriptors (UCUM annotation, issue #2419).

57/// Bounded attribute keys for catalog metrics.
58///
59/// These are the ONLY attribute keys downstream code should attach to the
60/// catalog metrics. Each is documented with its allowed value space so the
61/// cardinality stays bounded.
62pub mod attr {
63    /// Low-cardinality error category. Values come from
64    /// [`crate::observability::ObsErrorCategory::as_str`] (≈10 distinct values).
65    pub const ERROR_CATEGORY: &str = "cqlite.error.category";
66    /// Subsystem that produced an event, e.g. `"reader"`, `"query"`,
67    /// `"compaction"`. Callers pass a `&'static str`, so the value space is
68    /// bounded by the code itself.
69    pub const SUBSYSTEM: &str = "cqlite.subsystem";
70    /// SSTable on-disk format family, e.g. `"big"` or `"bti"`. Bounded.
71    pub const SSTABLE_FORMAT: &str = "cqlite.sstable.format";
72    /// Compression algorithm, e.g. `"lz4"`, `"snappy"`, `"none"`. Bounded.
73    pub const COMPRESSION: &str = "cqlite.compression";
74    /// Outcome of a lookup/check, exactly `"hit"` or `"miss"`. Bounded to two
75    /// values; used by partition-lookup and bloom-check counters so a single
76    /// metric carries both arms for ratio dashboards.
77    pub const RESULT: &str = "cqlite.result";
78    /// Read-path access route for a partition lookup, e.g. `"index"` (BIG
79    /// Index.db) or `"bti_trie"` (BTI Partitions.db). Bounded by the code.
80    /// Distinct from [`ACCESS_PATH`], which is the query-engine SELECT access
81    /// path (#1035); this is the storage-layer lookup route (#1034).
82    pub const LOOKUP_ROUTE: &str = "cqlite.read.lookup_route";
83    /// Access path a `SELECT` chose for its SSTable-scan step (issue #1035).
84    ///
85    /// Values come from [`crate::query::access_path::AccessPath::label`] — a
86    /// closed set such as `"full_scan"`, `"partition_lookup"`,
87    /// `"multi_partition_lookup"`, `"clustering_slice"`,
88    /// `"fallback_full_scan"`. Bounded by the `AccessPath` enum itself, so it is
89    /// safe as a metric dimension and span attribute. NEVER carries key values.
90    pub const ACCESS_PATH: &str = "cqlite.query.access_path";
91    /// Query plan family chosen by the planner / executor, e.g. `"table_scan"`,
92    /// `"point_lookup"`, `"index_scan"`, `"range_scan"`, `"aggregation"`
93    /// (issue #1035). Bounded by the executor's plan-type taxonomy.
94    pub const PLAN_TYPE: &str = "cqlite.query.plan_type";
95    /// Arrow Flight RPC method name (issue #1041), e.g. `"do_get"`,
96    /// `"get_flight_info"`, `"get_schema"`, `"handshake"`. Bounded by the
97    /// `FlightService` trait's fixed set of methods — never a request payload.
98    pub const RPC_METHOD: &str = "cqlite.rpc.method";
99    /// Arrow Flight RPC outcome (issue #1041), exactly `"ok"` or `"error"`.
100    /// Bounded to two values so a single metric series carries both arms for
101    /// success/error-rate dashboards.
102    pub const RPC_STATUS: &str = "cqlite.rpc.status";
103    /// `do_get` execution phase (issue #2162; `"admission"` added #2420
104    /// roborev-1700; `"validate"` added #2420 roborev-1702). Bounded to the
105    /// closed set of five TOP-LEVEL values `"validate"`, `"admission"`,
106    /// `"resolve"`, `"merge_setup"`, `"stream"` — a `&'static str` from a fixed
107    /// slot table, never a per-query, per-ticket, key, or query-text value. Used
108    /// as the bounded dimension on [`super::RPC_PHASE_DURATION`] so a stalled
109    /// `do_get` localizes to a phase (time piling up in `merge_setup`, queued
110    /// behind the admission semaphore in `admission`, or stuck parsing/validating
111    /// a malformed ticket in `validate`) from metrics alone.
112    ///
113    /// Cardinality (#2819; extended by #3096): on [`super::RPC_PHASE_DURATION`] the
114    /// `do_get` method ALSO carries SIX in-`stream` sub-phase values
115    /// (`"stream_cold_fault"`, `"stream_decompress"`, `"stream_merge"`, `"stream_encode"`,
116    /// `"stream_encode_framing"`, `"stream_grpc_write"`), so its phase.duration value
117    /// set is ELEVEN. Those values are gated to `do_get` + `phase.duration` only —
118    /// they are NOT added to [`super::RPC_PHASE_ACTIVE`], which stays the five
119    /// top-level values per method. Still a closed, static, low-cardinality set.
120    pub const RPC_PHASE: &str = "cqlite.rpc.phase";
121    /// Reason a `SELECT` fell back to a degraded (full-scan) read path
122    /// (issue #2163). Values come from
123    /// [`crate::query::access_path::FallbackReason::label`] — a documented closed
124    /// set (`no_schema`, `partition_key_not_fully_constrained`,
125    /// `partition_key_encoding_failed`, `metadata_scan_path`, `legacy_executor_path`,
126    /// `tombstones_build_no_prune`). Bounded by the enum itself; NEVER carries a
127    /// partition key, predicate value, or query string.
128    pub const FALLBACK_REASON: &str = "cqlite.query.fallback_reason";
129    /// Structural invariant a BTI `Rows.db` row-index ROOT violated, making the
130    /// clustering read fall back to a full-partition decode (issue #3002). Values
131    /// come from
132    /// [`crate::storage::sstable::bti::RowsTrieRootRejectReason::label`] — a closed,
133    /// static set of SEVEN `&'static str`s (`not_below_entry`,
134    /// `payload_incapable_node_type`, `childless_root_without_payload`,
135    /// `truncated_node`, `sparse_node_without_transitions`, `invalid_payload_bits`,
136    /// `extent_not_at_entry`), STAMPED per enum variant and never derived from file
137    /// bytes or a message string. Bounded by the enum itself; NEVER carries an
138    /// offset, key, or path.
139    pub const ROWS_ROOT_REJECT_REASON: &str = "cqlite.read.rows_root_reject_reason";
140    /// Flight warm-handle refresh outcome (issue #2310). Bounded to the closed set
141    /// `"unchanged"`, `"rebuilt_delta"`, `"fail_closed_retained"` — a `&'static str`
142    /// from a fixed slot table, never a ticket, key, or path value.
143    pub const WARM_REFRESH_OUTCOME: &str = "cqlite.warm.refresh_outcome";
144    /// Fine-grained Flight `do_get` abort reason (issue #2681). Bounded to the
145    /// closed set `"superseded_split"`, `"client_cancel"`, `"admission_shed"`,
146    /// `"snapshot_retired"`, `"internal"`, `"ticket_invalid"` — a `&'static str`
147    /// STAMPED at the abort construction site (never inferred from the gRPC code
148    /// or the error message text, no-heuristics #28). Attached ONLY to
149    /// [`super::ERRORS_TOTAL`] for `cqlite.subsystem = "flight"`, so a benign
150    /// abort (a torn-down split, a client hang-up, an admission shed) is
151    /// distinguishable in-field from a genuine internal fault that all previously
152    /// collapsed into the coarse `cqlite.error.category = "other"` bucket. The
153    /// high-cardinality ticket/split identity + snapshot generation live on the
154    /// abort log/trace event only, NEVER on this or any metric label.
155    pub const FLIGHT_ABORT_REASON: &str = "cqlite.flight.abort_reason";
156    /// Repeat-access bucket for the bounded partition access-distribution probe
157    /// (issue #2827). Bounded to the closed set of EXACTLY six labels
158    /// `"1"`, `"2"`, `"3-4"`, `"5-8"`, `"9-16"`, `"17+"` — the number of times a
159    /// distinct partition was accessed inside one measurement window, summarised
160    /// IN-PROCESS into these six buckets before emission.
161    ///
162    /// The bucket label is the ONLY thing that leaves the process about a
163    /// partition's access count: no partition key, key hash, key prefix, key
164    /// length or token is ever attached (the binding constraint at the top of
165    /// this module). Six values × the three [`SIZE_SOURCE`] values is the whole
166    /// cardinality budget of
167    /// [`super::READ_PARTITION_ACCESS_DISTINCT_PARTITIONS`] — eighteen series, and
168    /// 34 across the whole seven-metric `cqlite.read.partition_access.*` family
169    /// (18 + 6 + 6 + four unlabelled scalars).
170    pub const REPEAT_BUCKET: &str = "cqlite.read.repeat_bucket";
171    /// Provenance of the on-disk byte weight recorded for a partition access
172    /// (issue #2827). Bounded to the closed set of EXACTLY three labels:
173    ///
174    /// - `"successor_gap"` — the extent was MEASURED as `[data_offset,
175    ///   successor_offset)`, bounding to the authoritative uncompressed
176    ///   data-section length for the last partition. This is the normal value:
177    ///   NEITHER Cassandra 5.0 index format records a per-partition size (a BIG
178    ///   index entry is `[key][data_offset vint][promoted_index_len vint]
179    ///   [promoted_index]` —
180    ///   `docs/sstables-definitive-guide/chapters/06-index-and-summary.md`, "Index.db
181    ///   Entry Format" — and the BTI trie resolves an offset only), so the extent is
182    ///   measured from index LAYOUT, the same bound the single-partition seek uses
183    ///   to size its decompression window.
184    /// - `"index"` — an SSTable reported a size directly in its index metadata
185    ///   (`PartitionLoc.data_size`). Unreachable for Cassandra-written SSTables per
186    ///   the above; retained so a producer that genuinely knows a size is never
187    ///   forced to report a measured one.
188    /// - `"unavailable"` — no authoritative extent at all, so the access contributes
189    ///   ZERO bytes and is counted here instead. A size is never estimated,
190    ///   interpolated by proportion, or defaulted (no-heuristics #28): the
191    ///   incompleteness is published as a ratio rather than absorbed.
192    ///
193    /// Where an access mixes provenances the WEAKEST is reported — a total is only
194    /// as well-founded as its weakest component.
195    pub const SIZE_SOURCE: &str = "cqlite.read.size_source";
196}

198/// `cqlite.read.rows` — counter `{row}`.
199///
200/// Rows a read DELIVERED (see `ReadOpMeter`). On the Flight k-way merge scan
201/// (issue #2162) the delta is emitted incrementally during a long-running scan,
202/// at a bounded row threshold, so the counter climbs before the scan returns;
203/// the total is unchanged. That merge-scan emission is FORMAT-AGNOSTIC (carries
204/// no attributes): the k-way merge reconciles rows across potentially several
205/// input SSTables — of possibly mixed BIG/BTI format — into one row set before
206/// this counter's grain, so no single format label is honest at the point of
207/// emission without per-input-file tallies threaded through reconciliation (no
208/// consumer needs that split today; a future extension could add it). A direct
209/// single-SSTable read-path caller may still attach [`attr::SSTABLE_FORMAT`]
210/// where the format is known at its own emission site — this metric's attribute
211/// set is therefore [`attr::SSTABLE_FORMAT`] OR no attributes, never a fabricated
212/// format label.
213pub const READ_ROWS: &str = "cqlite.read.rows";
214

239/// `cqlite.sstable.index_parses_total` — counter `1` (issue #2383).
240///
241/// Incremented ONCE each time the full BIG/NB `Index.db` partition index is
242/// parsed end-to-end into partition entries (`parse_all_partition_keys_with_summary`,
243/// the O(entries) `memcmp`/vint loop that dominates opening a reader for a
244/// many-partition SSTable). It is the scale-free, path-independent probe for the
245/// #2383 resolve-phase CPU spin: the field failure re-parsed the SAME 1.58M-entry
246/// Index.db 8× for one logical query (redundant per-generation opens across
247/// per-query snapshot teardown + un-coalesced concurrent splits), so this counter
248/// climbs far past the number of distinct generations. A correct read path parses
249/// each generation's Index.db at most ONCE per query (warm reuse / rebind
250/// thereafter), so `index_parses_total ≈ Σ generations`, not `Σ generations ×
251/// opens`. Emitted from every full-parse site (warm rebuild, aggregate merge,
252/// point-read routing) so no single call path can hide a redundant parse. No
253/// high-cardinality attributes.
254pub const INDEX_PARSES_TOTAL: &str = "cqlite.sstable.index_parses_total";

256/// `cqlite.sstable.index_interval_parses_total` — counter `1` (issue #2412).
257///
258/// Incremented once per **bounded** `Index.db` interval parse performed by the
259/// lazy Summary-guided BIG partition index: a point lookup binary-searches
260/// `Summary.db`, seeks to the covering sample's position, and parses at most one
261/// `min_index_interval` of entries (§B of the #2412 design). This is a DISTINCT
262/// counter from [`INDEX_PARSES_TOTAL`], which continues to count only WHOLE-file

265/// open of K generations yields `index_parses_total += 0` and
266/// `index_interval_parses_total += 0`, then `+= 1` per point lookup — the
267/// scale-free work-probe for #2412. No high-cardinality attributes.
268pub const INDEX_INTERVAL_PARSES_TOTAL: &str = "cqlite.sstable.index_interval_parses_total";

382/// `cqlite.read.partition_access.distinct_partitions` — counter `{partition}`
383/// (issue #2827).
384///
385/// Number of DISTINCT partitions that fell into each repeat-access bucket over
386/// one closed measurement window of the bounded partition access-distribution
387/// probe. Bounded attributes: [`attr::REPEAT_BUCKET`] (six labels) and
388/// [`attr::SIZE_SOURCE`] (three labels) — eighteen series, fixed forever, regardless
389/// of how many partitions the workload touched. Across all SEVEN
390/// `cqlite.read.partition_access.*` metrics the budget is
391/// `6x3 + 6 + 6 + 1 + 1 + 1 + 1 = 34`.
392///
393/// This is the concentration SHAPE of a keyed workload: a distribution
394/// concentrated in `1` is a uniform (cache-hostile) access pattern; mass in
395/// `9-16`/`17+` is a hot set. It is emitted ONCE per closed window and only when
396/// the window recorded at least one access — a `0/0` emission would be a series
397/// with no subject. The probe is OFF by default
398/// (`CQLITE_PARTITION_ACCESS_PROBE`), so this metric is absent unless an
399/// operator turned it on.
400pub const READ_PARTITION_ACCESS_DISTINCT_PARTITIONS: &str =
401    "cqlite.read.partition_access.distinct_partitions";

403/// `cqlite.read.partition_access.accesses` — counter `1` (issue #2827).
404///
405/// Sum of the repeat counts of the partitions in each bucket, over one closed
406/// window. Bounded attribute: [`attr::REPEAT_BUCKET`].
407///
408/// Emitting the access total PER BUCKET (rather than one grand total) removes
409/// the within-bucket mean from any hit-ratio arithmetic: the open-ended `17+`
410/// bucket stops being unbounded and the cache-sizing bound becomes a point value
411/// rather than an interval. Paired with
412/// [`READ_PARTITION_ACCESS_DISTINCT_PARTITIONS`], `accesses − distinct` is the
413/// number of accesses a clairvoyant cache holding that bucket would have served.

474/// `cqlite.read.partition_access.window_dropped_accesses` — gauge `1` (issue #2827).
475///
476/// Accesses the LAST CLOSED window could not seat, reset every window. No
477/// attributes.
478///
479/// Exists because its cumulative sibling [`READ_PARTITION_ACCESS_DROPPED`] cannot
480/// answer "was THIS window clean": a counter that ever incremented reads non-zero
481/// for the life of the process, so an instantaneous read of it can only say "this
482/// process has lost input at some point". The spec requires that a consumer reading
483/// the emitted series ALONE distinguish a lossy or floored window from a clean one,

485/// [`READ_PARTITION_ACCESS_SAMPLING_FLOOR`].

604/// `cqlite.wal.recovery.duration` — histogram `s` (issue #1707).
605///
606/// How long write-ahead-log recovery at engine open took, in seconds. Recovery is
607/// BOTH halves of what open does to the log: the CRC validation scan (plus any
608/// torn-tail trim) and the replay of its entries into the memtable. On a large or
609/// corrupt-tail WAL the validation scan is the dominant half, which is why the
610/// timer spans the whole window — a replay-only timing would understate exactly the
611/// startup an operator is asking about. Recovery happens EXACTLY ONCE per engine
612/// open, so this series normally carries ONE sample per process — read it as a
613/// value, not as a distribution.
614///
615/// **Why a histogram and not a gauge, which is what one-sample-per-process argues
616/// for**: the gauge plane in this crate is `i64` (`Gauge<i64>`, and OTel's gauge
617/// builders here are `i64`/`u64`), and this value is sub-second in the common case.
618/// An `i64` gauge in base-unit SECONDS would report `0` for a 400 ms recovery — a
619/// FABRICATED zero, which is the precise defect class epic #1686 exists to remove,
620/// and worse than the wart of a one-sample distribution. Reporting milliseconds
621/// instead would break the catalog's base-unit rule (see the `read.phase.*`
622/// naming note). So the instrument follows the value's precision, and every other
623/// duration in this catalog is an `f64` histogram in seconds anyway.
624///
625/// **Recorded unconditionally at every engine open, including the 0-entry case**: a
626/// fresh WAL with nothing to recover genuinely took ~0s, and that IS a measurement —
627/// the absence rule forbids inventing a value nobody measured, not reporting a real
628/// one that happens to be small. It is also recorded when replay found CORRUPTION,
629/// before the lossy-recovery branch, because that is exactly when an operator cares
630/// what open cost. Absence of this series therefore means "no engine was opened in
631/// this process", never "recovery was skipped".
632///
633/// **Healthy vs alarming**: near-zero after a clean shutdown; seconds means a crash
634/// left a large log to validate and replay, which directly delays open — pair it
635/// with [`WAL_SIZE`], whose growth is what makes this number grow.
636/// **Attributes**: none.
637pub const WAL_RECOVERY_DURATION: &str = "cqlite.wal.recovery.duration";

876///
877/// Process resident set size in bytes, sampled from the `VmRSS` field of

888/// `cqlite.flight.blocking_tasks_in_use` — gauge `{thread}` (issue #2419, WS2).
889///
890/// Flight-managed `spawn_blocking` tasks currently outstanding, tracked by a
891/// process-wide atomic incremented on entry to a flight `spawn_blocking`
892/// closure and decremented on exit via an RAII guard (so a panic / cancel /
893/// early-return still decrements). Guards EVERY flight-managed blocking closure
894/// (roborev job 1733 fix 3, so the gauge reflects true pool-saturation
895/// pressure, not merge-only): the streaming merge and aggregate-materialize
896/// closures (`streaming.rs`), `do_get`'s resolve-phase closure (producer/schema
897/// construction + `DirSource::resolve` + token-prune, `service.rs`), and the
898/// `table_stats` `gather_table_stats` closure (`service.rs`). An honest,
899/// dependency-free proxy for blocking-pool pressure.
900///
901/// **Scope caveat**: this is FLIGHT-MANAGED-TASKS-IN-FLIGHT, NOT the global
902/// `tokio` blocking-pool queue depth (which needs a build-wide `tokio_unstable`
903/// cfg — out of scope, design open fork O1). It never records a fabricated
904/// global-pool number. **Healthy vs alarming**: rises with concurrent `do_get`
905/// scans and returns to baseline as they finish; a level pinned near the
906/// blocking-pool size (~512 default) with flat `cqlite.rpc.rows` is the
907/// blocking-pool-saturation smell. OS-independent (always emits). DISTINCT from
908/// [`FLIGHT_ADMISSION_IN_USE`] (held admission permits) — the two measure
909/// different resources. No high-cardinality attributes.

912/// `cqlite.flight.tables_discovered` — gauge `{entry}` (issue #2684).
913///
914/// Number of `<keyspace>/<table>` SSTable directories currently VISIBLE under
915/// the server's `--data-dir`, re-sampled on the same ~2s saturation tick as the
916/// `cqlite.proc.*` gauges. Each tick does a readdir-only walk (keyspace dirs →
917/// table dirs), counting a table dir iff it directly contains a `*-Data.db`
918/// entry (name check only — NO stat-for-generation, NO open, NO parse), so the
919/// cold-start invariant (#2385) holds: sampling never increments
920/// [`INDEX_PARSES_TOTAL`]. `snapshots/` and `backups/` subtrees are excluded and
921/// UUID-suffixed table dirs are counted correctly.
922///
923/// **Healthy vs alarming**: RISES when a new table appears on disk (new table, a
924/// fixed mount) and FALLS when a table is dropped/removed; a wrong or empty
925/// `--data-dir` reads **0** immediately, surfacing an inert mount before the
926/// first query errors lazily. **Undersampling caveat (#2661)**: sampled every
927/// ~2s, so a Prometheus scrape at a longer interval can miss a short-lived
928/// transition — same caveat as the rest of the #2419 saturation family. No
929/// high-cardinality attributes (total-only; the per-keyspace breakdown lives in
930/// the one-time startup log line, never as a metric label).
931pub const FLIGHT_TABLES_DISCOVERED: &str = "cqlite.flight.tables_discovered";

933/// `cqlite.flight.warm_tables` — gauge `{entry}` (issue #2684).
934///
935/// The count of tables with a live (non-empty) warm reader set in the flight
936/// `WarmTableRegistry` (a retired table leaves a zero-reader entry until the next
937/// rebuild, so the reading filters those out rather than using the raw map size).
938/// Atomic-backed at the registry's
939/// mutation sites (independent of the sampler cadence): the post-mutation
940/// `.len()` is emitted while the registry lock is held at the `rebuild()` insert
941/// and `evict_to_budget()` removal, so the remove-then-reinsert transient the
942/// rebuild performs never dips the reading.
943///
944/// **Healthy vs alarming**: RISES on the first serve of a previously-unseen
945/// table (a `do_get` rebuild insert) and FALLS on eviction/retirement (budget
946/// LRU or generation turnover). A level pinned at capacity with steady eviction
947/// churn means the warm byte budget is small versus the working set. No
948/// high-cardinality attributes (total-only). DISTINCT from
949/// [`FLIGHT_TABLES_DISCOVERED`] (what is VISIBLE on disk): this is what is
950/// actually OPENED and served warm.
951pub const FLIGHT_WARM_TABLES: &str = "cqlite.flight.warm_tables";

953/// `cqlite.errors.total` — counter `{error}`.
954///
955/// Total errors observed, the canonical error-rate signal (issue #1038).
956/// Bounded attributes: [`attr::ERROR_CATEGORY`] and [`attr::SUBSYSTEM`] ONLY.
957/// The raw error message is never attached.
958///
959/// **Eagerly registered at 0 on startup (issue #2288).** When the
960/// `observability` feature is active, [`crate::observability::init`] emits a
961/// single `add(0)` with an empty attribute set so this counter is present at `0`
962/// in a scrape of a freshly-started server, before any error. This makes "metric
963/// name absent from the backend" unambiguously mean *error counting isn't wired*
964/// (never *no errors occurred yet*), which cost real diagnostic time during the
965/// #2193 round-4 field investigation. Real errors add their own labeled series
966/// alongside the unlabeled baseline.
967///
968/// Limitation (per the #2193 code audit): a peer connection RESET that arrives
969/// *after* the gRPC `END_STREAM` frame is handled entirely inside the h2/tonic
970/// transport and is invisible at this application layer, so it is not counted
971/// here. Such post-END_STREAM resets are an expected, benign transport event, not
972/// an application error.
973pub const ERRORS_TOTAL: &str = "cqlite.errors.total";
