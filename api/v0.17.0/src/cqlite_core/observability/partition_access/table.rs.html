1//! The fixed-footprint counting structure behind the partition access-distribution
2//! probe (issue #2827, design D4).
3//!
4//! One lazily-allocated, open-addressed (linear-probing) table:
5//!
6//! - `SLOTS = 1 << 17` = 131,072
7//! - entry = `key_hash: AtomicU64` (8 B) + `bytes: AtomicU64` (8 B) +
8//!   `count: AtomicU32` (4 B) + `flags: AtomicU8` (1 B) + 3 B padding = **24 B**
9//! - **total = 131,072 × 24 B = 3,145,728 B = exactly 3 MiB, fixed** — independent
10//!   of the partition count, of qps, and of window length.
11//!
12//! # Why overflow is handled by SAMPLING, not eviction
13//!
14//! When occupancy reaches a 0.75 load factor the recorder widens a hash-prefix
15//! admission predicate by one bit and drops every entry that no longer satisfies
16//! it. The predicate is a function of the key hash ALONE, so it is statistically
17//! independent of a key's access frequency: the survivors are a uniform random
18//! sample of the DISTINCT partitions touched in the window, and each survivor's
19//! count stays EXACT (the predicate is monotone — a key admitted at prefix width
20//! `k` is admitted at every `k' < k` — so a downsample only ever removes keys, it
21//! never invalidates a survivor's already-accumulated count).
22//!
23//! Recency- (LRU) or arrival-ordered eviction would do the opposite: hot keys are
24//! accessed often, so they survive preferentially, the singleton bucket is
25//! under-counted and the histogram OVERSTATES concentration — the direction that
26//! makes a decoded-partition cache look better than it is. A go/no-go instrument
27//! must not be biased toward "go", so eviction is disqualified on correctness
28//! grounds, not on cost.
29//!
30//! # Deletion and probe chains
31//!
32//! Linear probing cannot simply blank a slot: later members of the same probe
33//! chain would become unreachable and their counts would silently split across a
34//! second entry. Downsampling therefore uses **backward-shift deletion**
35//! ([`Table::delete_at`]), which restores the probe-chain invariant in place with
36//! no auxiliary allocation — so the 3 MiB bound holds during a downsample too.
37
38use std::sync::atomic::{AtomicU32, AtomicU64, AtomicU8, AtomicUsize, Ordering};
39
40/// log2 of the slot count.
41pub(super) const SLOT_BITS: u32 = 17;
42/// Number of slots in the counting table.
43pub(super) const SLOTS: usize = 1 << SLOT_BITS;
44/// Slot index mask (`SLOTS` is a power of two).
45const SLOT_MASK: usize = SLOTS - 1;
46/// Declared per-entry footprint. Asserted against the real layout below.
47pub(super) const ENTRY_BYTES: usize = 24;
48/// The whole structure's fixed footprint: exactly 3 MiB.
49pub(super) const TABLE_BYTES: usize = SLOTS * ENTRY_BYTES;
50/// Occupancy at which the recorder widens the sampling prefix (load factor 0.75).
51pub(super) const LOAD_FACTOR_LIMIT: usize = SLOTS / 4 * 3;
52/// Longest linear-probe run before an insert reports the table effectively full.
53///
54/// At a 0.75 load factor with a well-mixed hash a run this long is
55/// vanishingly rare; treating it as "full" simply triggers a downsample one
56/// insert early, which is harmless (the scale is published).
57pub(super) const MAX_PROBES: usize = 64;
58
59/// Sticky per-entry flag: at least one access to this partition resolved an
60/// SSTable that reported no authoritative size, so the entry contributes ZERO
61/// bytes and is counted as `size_source = unavailable`.
62pub(super) const FLAG_SIZE_UNAVAILABLE: u8 = 0b0000_0001;
63
64/// Sticky per-entry flag: at least one access to this partition had its extent
65/// MEASURED as a successor gap rather than read from an index-recorded size, so the
66/// entry's bytes are reported with `size_source = successor_gap`.
67pub(super) const FLAG_SIZE_FROM_GAP: u8 = 0b0000_0010;
68
69/// One counting slot. `key_hash == 0` means EMPTY; a hash that would be zero is
70/// remapped to 1 by [`hash_partition`], so the sentinel is unambiguous.
71#[repr(C)]
72pub(super) struct Slot {
73    key_hash: AtomicU64,
74    bytes: AtomicU64,
75    count: AtomicU32,
76    flags: AtomicU8,
77    _pad: [u8; 3],
78}
79
80// The 3 MiB claim in the docs, the catalog and the spec is only true if the entry
81// really is 24 bytes. Assert the layout at compile time rather than trusting the
82// comment.
83const _: () = assert!(std::mem::size_of::<Slot>() == ENTRY_BYTES);
84const _: () = assert!(TABLE_BYTES == 3 * 1024 * 1024);

90            bytes: AtomicU64::new(0),
91            count: AtomicU32::new(0),
92            flags: AtomicU8::new(0),

105/// A live entry read out of the table at window close.
106#[derive(Clone, Copy, Debug)]
107pub(super) struct Entry {
108    /// Number of accesses recorded for this partition inside the window.
109    pub(super) count: u32,
110    /// Largest on-disk byte weight observed for it (meaningless when
111    /// `size_unavailable` is set — such an entry contributes zero bytes).
112    pub(super) bytes: u64,
113    /// Sticky: at least one access could not be priced authoritatively.
114    pub(super) size_unavailable: bool,
115    /// Sticky: at least one priced access had its extent measured as a gap.
116    pub(super) size_from_gap: bool,
117}

119/// Outcome of an insert attempt.
120#[derive(Clone, Copy, Debug, PartialEq, Eq)]
121pub(super) enum Insert {
122    /// The access was recorded (either into a fresh slot or an existing one).
123    Recorded,
124    /// The key is not in the currently-admitted sample; nothing was recorded.
125    NotAdmitted,
126    /// No slot could be claimed within [`MAX_PROBES`]; the caller must downsample.
127    Full,
128}

130/// The fixed 3 MiB counting table.
131pub(super) struct Table {
132    slots: Box<[Slot]>,
133    occupancy: AtomicUsize,
134}

136/// A 64-bit hash identifying one partition WITHIN ONE TABLE.
137///
138/// # Why the table is part of the identity
139///
140/// [`super::global`] is ONE process-wide recorder shared by every table, so an
141/// identity over raw key bytes alone merges the same key in two different tables
142/// into one entry — and an ordinary tenant/user id shared across tables is common,
143/// not a rare collision. The merge is biased in the unsafe direction twice over:
144/// two singletons become a `count = 2` entry, manufacturing a hit where the truth
145/// is none; and the entry keeps the MAXIMUM byte weight rather than the sum, so it
146/// is under-priced and ranks EARLIER by access density. Both flatter the cache,
147/// which is the bias class D2/D4 disqualify other mechanisms for.
148///
149/// Deliberately NOT the Murmur3 token: the BIG point path takes raw key bytes and
150/// never computes a token, so hashing the token would make the probe force work
151/// that path does not otherwise do. This value is used ONLY for slot addressing
152/// and within-window identity — it is never emitted, logged, or attached to a
153/// metric attribute (the whole point of the bucket histogram is that no per-key
154/// value leaves the process).
155///
156/// FNV-1a over the table scope and the key bytes, separated by a byte that cannot
157/// occur in the scope's UTF-8 (so `("ab", "c")` and `("a", "bc")` cannot collide),
158/// finalised with a splitmix64 avalanche so the HIGH bits (used by the sampling
159/// predicate) and the LOW bits (used for slot addressing) are independently
160/// well-mixed. `0` is reserved as the empty-slot sentinel and is remapped to `1`.
161pub(super) fn hash_partition(keyspace: &str, table: &str, key: &[u8]) -> u64 {
162    /// Separator that cannot appear inside a UTF-8 scan of the scope strings.
163    const SEP: u8 = 0xFF;
164    let mut h: u64 = 0xcbf2_9ce4_8422_2325;
165    let mix = |bytes: &[u8], h: &mut u64| {
166        for b in bytes {
167            *h ^= *b as u64;
168            *h = h.wrapping_mul(0x0000_0100_0000_01b3);
169        }
170    };
171    mix(keyspace.as_bytes(), &mut h);
172    mix(&[SEP], &mut h);
173    mix(table.as_bytes(), &mut h);
174    mix(&[SEP], &mut h);
175    for b in key {
176        h ^= *b as u64;
177        h = h.wrapping_mul(0x0000_0100_0000_01b3);
178    }
179    // splitmix64 finaliser.
180    let mut z = h.wrapping_add(0x9e37_79b9_7f4a_7c15);
181    z = (z ^ (z >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
182    z = (z ^ (z >> 27)).wrapping_mul(0x94d0_49bb_1331_11eb);
183    z ^= z >> 31;
184    if z == 0 {
185        1
186    } else {
187        z
188    }
189}

191/// The hash-prefix admission predicate: a key is admitted at prefix width `k`
192/// when its top `k` hash bits are all zero.
193///
194/// Monotone in `k` (admitted at `k` ⇒ admitted at every smaller `k`), which is
195/// what makes a survivor's already-accumulated count remain exact across a
196/// downsample. Uses the HIGH bits so admission is independent of the low bits that
197/// choose the slot.
198#[inline]
199pub(super) fn admitted(hash: u64, prefix_bits: u32) -> bool {
200    if prefix_bits == 0 {
201        return true;
202    }
203    debug_assert!(prefix_bits < 64);
204    (hash >> (64 - prefix_bits)) == 0
205}

207impl Table {
208    /// Allocate the fixed 3 MiB table. Only ever called once the probe is enabled
209    /// and an access actually arrives (lazy allocation — a disabled probe holds no
210    /// memory at all).
211    pub(super) fn new() -> Self {
212        let mut slots = Vec::with_capacity(SLOTS);
213        slots.resize_with(SLOTS, Slot::empty);
214        Self {
215            slots: slots.into_boxed_slice(),
216            occupancy: AtomicUsize::new(0),
217        }
218    }
219
220    /// The structure's footprint in bytes — a constant, by construction.
221    pub(super) fn footprint_bytes(&self) -> usize {
222        self.slots.len() * ENTRY_BYTES
223    }
224
225    /// Number of distinct keys currently held.
226    pub(super) fn occupancy(&self) -> usize {
227        self.occupancy.load(Ordering::Relaxed)
228    }
229
230    /// Record one access to `hash`, weighted by `bytes` (`None` = the access could
231    /// not be priced authoritatively, which sets the sticky unavailable flag and
232    /// contributes zero bytes).
233    ///
234    /// Takes `&self`: the caller holds only a READ lock, so concurrent recorders
235    /// share the table and mutate their own slots with atomics. Slot claiming is a
236    /// CAS on the `key_hash` sentinel, so two threads racing on the same empty slot
237    /// cannot both claim it.
238    #[cfg(test)]
239    pub(super) fn record(&self, hash: u64, prefix_bits: u32, bytes: Option<u64>) -> Insert {
240        self.record_with_flags(hash, prefix_bits, bytes, 0)
241    }
242
243    /// [`Self::record`] with extra sticky provenance flags OR-ed into the slot.
244    pub(super) fn record_with_flags(
245        &self,
246        hash: u64,
247        prefix_bits: u32,
248        bytes: Option<u64>,
249        extra_flags: u8,
250    ) -> Insert {
251        if !admitted(hash, prefix_bits) {
252            return Insert::NotAdmitted;
253        }
254        let mut idx = (hash as usize) & SLOT_MASK;
255        for _ in 0..MAX_PROBES {
256            let slot = &self.slots[idx];
257            let mut observed = slot.key_hash.load(Ordering::Acquire);
258            if observed == 0 {
259                match slot
260                    .key_hash
261                    .compare_exchange(0, hash, Ordering::AcqRel, Ordering::Acquire)
262                {
263                    Ok(_) => {
264                        self.occupancy.fetch_add(1, Ordering::Relaxed);
265                        observed = hash;
266                    }
267                    // Lost the race: re-read what the winner stored and fall
268                    // through — if it is our own hash we share the slot.
269                    Err(actual) => observed = actual,
270                }
271            }
272            if observed == hash {
273                Self::apply(slot, bytes, extra_flags);
274                return Insert::Recorded;
275            }
276            idx = (idx + 1) & SLOT_MASK;
277        }
278        Insert::Full
279    }
280
281    /// Fold one access into an already-claimed slot.
282    fn apply(slot: &Slot, bytes: Option<u64>, extra_flags: u8) {
283        if extra_flags != 0 {
284            slot.flags.fetch_or(extra_flags, Ordering::Relaxed);
285        }
286        // Saturating rather than wrapping: a window that somehow recorded 2^32
287        // accesses to one partition must not wrap back into the `1` bucket.
288        let mut cur = slot.count.load(Ordering::Relaxed);
289        loop {
290            let next = cur.saturating_add(1);
291            match slot
292                .count
293                .compare_exchange_weak(cur, next, Ordering::Relaxed, Ordering::Relaxed)
294            {
295                Ok(_) => break,
296                Err(actual) => cur = actual,
297            }
298        }
299        match bytes {
300            // The working set is defined over DISTINCT partitions, so an entry
301            // retains the MAXIMUM weight observed, never a running sum: ten
302            // accesses to one partition contribute its size once.
303            Some(b) => {
304                let mut cur = slot.bytes.load(Ordering::Relaxed);
305                while b > cur {
306                    match slot.bytes.compare_exchange_weak(
307                        cur,
308                        b,
309                        Ordering::Relaxed,
310                        Ordering::Relaxed,
311                    ) {
312                        Ok(_) => break,
313                        Err(actual) => cur = actual,
314                    }
315                }
316            }
317            // Sticky for the rest of the window: once an access to this partition
318            // could not be priced, the partition's bytes are not trustworthy and it
319            // is reported under `size_source = unavailable` with zero bytes.
320            None => {
321                slot.flags
322                    .fetch_or(FLAG_SIZE_UNAVAILABLE, Ordering::Relaxed);
323            }
324        }
325    }
326
327    /// Visit every live entry. Requires `&mut` (window close holds the write lock).
328    pub(super) fn for_each_entry(&mut self, mut f: impl FnMut(Entry)) {
329        for slot in self.slots.iter() {
330            if slot.key_hash.load(Ordering::Relaxed) == 0 {
331                continue;
332            }
333            f(Entry {
334                count: slot.count.load(Ordering::Relaxed),
335                bytes: slot.bytes.load(Ordering::Relaxed),
336                size_unavailable: slot.flags.load(Ordering::Relaxed) & FLAG_SIZE_UNAVAILABLE != 0,
337                size_from_gap: slot.flags.load(Ordering::Relaxed) & FLAG_SIZE_FROM_GAP != 0,
338            });
339        }
340    }
341
342    /// Empty the table for the next tumbling window.
343    pub(super) fn reset(&mut self) {
344        for slot in self.slots.iter() {
345            slot.clear();
346        }
347        self.occupancy.store(0, Ordering::Relaxed);
348    }
349
350    /// Drop every entry that is not admitted at `prefix_bits`, restoring the
351    /// probe-chain invariant in place. Returns the new occupancy.
352    ///
353    /// Requires `&mut` (the recorder takes the WRITE lock for a downsample), so no
354    /// concurrent recorder can observe a half-shifted chain.
355    pub(super) fn downsample(&mut self, prefix_bits: u32) -> usize {
356        let mut i = 0usize;
357        while i < SLOTS {
358            let h = self.slots[i].key_hash.load(Ordering::Relaxed);
359            if h != 0 && !admitted(h, prefix_bits) {
360                self.delete_at(i);
361                // Do NOT advance: the backward shift may have moved another entry
362                // into slot `i`, and that one may also need dropping.
363                continue;
364            }
365            i += 1;
366        }
367        self.occupancy()
368    }
369
370    /// Backward-shift deletion for linear probing (Knuth 6.4 algorithm R).
371    ///
372    /// Blanking a slot outright would break every probe chain that passes through
373    /// it — later members would become unreachable and a subsequent access to one
374    /// of them would claim a SECOND slot, splitting one partition's count into two
375    /// lower-repeat entries and silently understating concentration. Shifting the
376    /// displaced members back into the hole keeps every survivor findable at its
377    /// exact count, with no auxiliary allocation.
378    fn delete_at(&mut self, at: usize) {
379        self.slots[at].clear();
380        self.occupancy.fetch_sub(1, Ordering::Relaxed);
381
382        let mut hole = at;
383        let mut probe = (at + 1) & SLOT_MASK;
384        // The loop terminates at the first empty slot; a full table is impossible
385        // here because we just created a hole.
386        while self.slots[probe].key_hash.load(Ordering::Relaxed) != 0 {
387            let h = self.slots[probe].key_hash.load(Ordering::Relaxed);
388            let home = (h as usize) & SLOT_MASK;
389            // Move the entry back only when its home does NOT lie strictly inside
390            // the cyclic window `(hole, probe]` — otherwise moving it to `hole`
391            // would place it BEFORE its own home and it could never be found.
392            if !cyclic_in_exclusive_start(hole, probe, home) {
393                self.move_slot(probe, hole);
394                hole = probe;
395            }
396            probe = (probe + 1) & SLOT_MASK;
397            if probe == hole {
398                break;
399            }
400        }
401    }
402
403    fn move_slot(&mut self, from: usize, to: usize) {
404        let (h, b, c, f) = {
405            let s = &self.slots[from];
406            (
407                s.key_hash.load(Ordering::Relaxed),
408                s.bytes.load(Ordering::Relaxed),
409                s.count.load(Ordering::Relaxed),
410                s.flags.load(Ordering::Relaxed),
411            )
412        };
413        let dst = &self.slots[to];
414        dst.bytes.store(b, Ordering::Relaxed);
415        dst.count.store(c, Ordering::Relaxed);
416        dst.flags.store(f, Ordering::Relaxed);
417        dst.key_hash.store(h, Ordering::Relaxed);
418        self.slots[from].clear();
419    }
420}

422/// Is `x` inside the cyclic range `(start, end]`?
423fn cyclic_in_exclusive_start(start: usize, end: usize, x: usize) -> bool {
424    if start < end {
425        x > start && x <= end
426    } else {
427        x > start || x <= end
428    }
429}

447        for hash in [1u64, 0x0000_0000_dead_beef, u64::MAX, 0x7fff_ffff_ffff_ffff] {

461    #[test]
462    fn downsample_keeps_every_survivor_findable_at_its_exact_count() {
463        // The backward-shift deletion regression: after a downsample every
464        // surviving key must still be reachable on its probe chain, so a
465        // subsequent access folds into the SAME slot rather than claiming a
466        // second one (which would split one partition into two lower-repeat
467        // entries and understate concentration).
468        let mut table = Table::new();
469        let n = 40_000u64;
470        for i in 0..n {
471            let h = hash_partition("ks", "t", &i.to_le_bytes());
472            assert_eq!(table.record(h, 0, Some(10)), Insert::Recorded);
473        }
474        let before = table.occupancy();
475        assert_eq!(before, n as usize);
476
477        let survivors = table.downsample(1);
478        assert!(survivors < before, "a downsample must drop entries");
479
480        // Every admitted key folds into its existing slot: occupancy is unchanged
481        // and each survivor's count becomes exactly 2.
482        for i in 0..n {
483            let h = hash_partition("ks", "t", &i.to_le_bytes());
484            match table.record(h, 1, Some(10)) {
485                Insert::Recorded => {}
486                Insert::NotAdmitted => assert!(!admitted(h, 1)),
487                Insert::Full => panic!("table must not be full at {survivors} entries"),
488            }
489        }
490        assert_eq!(
491            table.occupancy(),
492            survivors,
493            "a second pass over the same keys must claim no new slots"
494        );
495        let mut twos = 0usize;
496        let mut others = 0usize;
497        table.for_each_entry(|e| if e.count == 2 { twos += 1 } else { others += 1 });
498        assert_eq!(others, 0, "every survivor must have folded, not split");
499        assert_eq!(twos, survivors);
500    }

550    #[test]
551    fn a_probe_cluster_reports_full_well_below_the_load_factor() {
552        // The premise of the recorder's widen-instead-of-drop path (C4): `Full` is
553        // reachable far below the load factor, because the probe bound is 64 slots
554        // while a cluster can be much longer. Construct the cluster directly —
555        // MAX_PROBES + 1 keys that all hash to the same home slot — so the property
556        // is demonstrated rather than assumed from a statistical argument.
557        let table = Table::new();
558        // Find enough distinct keys sharing one home slot.
559        let mut home: Option<usize> = None;
560        let mut hashes = Vec::new();
561        let mut i = 0u64;
562        while hashes.len() <= MAX_PROBES && i < 50_000_000 {
563            let h = hash_partition("ks", "t", &i.to_le_bytes());
564            let slot = (h as usize) & SLOT_MASK;
565            match home {
566                None => {
567                    home = Some(slot);
568                    hashes.push(h);
569                }
570                Some(target) if slot == target => hashes.push(h),
571                _ => {}
572            }
573            i += 1;
574        }
575        assert!(
576            hashes.len() > MAX_PROBES,
577            "needed {} colliding keys, found {}",
578            MAX_PROBES + 1,
579            hashes.len()
580        );
581
582        for h in hashes.iter().take(MAX_PROBES) {
583            assert_eq!(table.record(*h, 0, Some(1)), Insert::Recorded);
584        }
585        // The table is ~0.05% full, nowhere near LOAD_FACTOR_LIMIT.
586        assert!(table.occupancy() < LOAD_FACTOR_LIMIT / 100);
587        assert_eq!(
588            table.record(hashes[MAX_PROBES], 0, Some(1)),
589            Insert::Full,
590            "a cluster longer than the probe bound reports Full even on an almost \
591             empty table — which is why the recorder must widen rather than drop"
592        );
593    }

595    #[test]
596    fn the_same_key_in_two_tables_is_two_identities() {
597        // F1: one process-wide recorder serves every table, so an identity over raw
598        // key bytes alone would merge a tenant id shared by two tables into a single
599        // entry — manufacturing a repeat where there are two singletons.
600        let key = b"tenant-42";
601        assert_ne!(
602            hash_partition("ks", "users", key),
603            hash_partition("ks", "orders", key)
604        );
605        assert_ne!(
606            hash_partition("ks_a", "t", key),
607            hash_partition("ks_b", "t", key)
608        );
609        // Same table, same key: one identity.
610        assert_eq!(
611            hash_partition("ks", "users", key),
612            hash_partition("ks", "users", key)
613        );
614        // The separator makes the scope unambiguous: ("ab","c") != ("a","bc").
615        assert_ne!(
616            hash_partition("ab", "c", key),
617            hash_partition("a", "bc", key)
618        );
619    }
