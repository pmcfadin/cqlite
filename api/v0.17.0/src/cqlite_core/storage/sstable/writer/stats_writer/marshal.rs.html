20/// Convert a CQL type name to Cassandra internal marshal type.
21///
22/// This is the reverse of `convert_marshal_type_to_cql` in enhanced_statistics_parser.rs.
23/// Used when writing the SERIALIZATION_HEADER component of Statistics.db.
24///
25/// Handles:
26/// - Primitive types: text, int, bigint, uuid, etc.
27/// - Collections: list<T>, set<T>, map<K,V>
28/// - Frozen wrappers: frozen<list<T>>, frozen<map<K,V>>
29/// - Tuples: tuple<T1, T2, ...>
30///
31/// - Vectors: `vector<element, n>` (issue #4158) — see the arm in
32///   [`render_marshal_type`] for the pinned derivation and for the element types
33///   it REFUSES by name.
34///
35/// This is Cassandra's `AbstractType::toString(false)` (issue #4158): the entry
36/// point is always the un-frozen, "print the `FrozenType(...)` wrapper where it
37/// belongs" case, and [`render_marshal_type`] carries the `ignoreFreezing` flag
38/// Cassandra threads through the type tree.
39///
40/// Returns `Err` where a truthful spelling cannot be produced (the vector arm;
41/// issue #4158) and where the string that would be EMITTED is one Cassandra's own
42/// writer cannot print (the post-condition below). Every other type keeps its
43/// pre-existing total behaviour, including the `_ => BytesType` fallback for an
44/// unresolvable bare name (#929) — widening that fallback into a refusal is NOT
45/// this function's job and would change the documented degradation of every
46/// unresolved UDT column.
47///
48/// # THE #4158 INVARIANT IS ENFORCED HERE, ONCE, ON THE STRING ACTUALLY EMITTED
49/// (#4104, roborev job 121)
50///
51/// #4158's guarantee is a property of the OUTPUT — "never emit a `FrozenType(...)`
52/// wrapper Apache Cassandra's writer cannot print" — and it was being enforced
53/// only at the site that BUILDS a wrapper ([`render_marshal_type`]'s `frozen<..>`
54/// arm, via [`takes_frozen_type_wrapper`]). Every other route into the output was
55/// therefore unguarded, and one was live: an ALREADY-MARSHALED input string is
56/// returned verbatim before any validation, and `TableSchema::data_type` is a
57/// `String`, so a supplied schema declaring
58/// `org.apache.cassandra.db.marshal.FrozenType(org.apache.cassandra.db.marshal.Int32Type)`
59/// went straight into `Statistics.db` — the exact spelling #4158 exists to
60/// prevent, arriving by the one path that never asked.
61///
62/// So the check is a POST-CONDITION on the rendered string rather than another
63/// per-arm guard: [`validate_marshal_frozen`] scans EVERY `FrozenType(` in the
64/// final output, at any nesting depth, whichever arm or pass-through produced it.
65/// It is the SAME gate the READ path applies to every header type string it
66/// decodes (`enhanced_statistics_parser::marshal_type`), so the writer and the
67/// reader hold one opinion about what is possible, and a future arm that
68/// assembles a wrapper wrongly is caught without being individually distrusted.
69///
70/// CASE IS PRESERVED: the validation is case-insensitive but non-consuming, and
71/// the string returned is the rendered one, byte for byte. The marshal grammar is
72/// case-sensitive (`TypeParser` resolves class names verbatim), and case-folding
73/// a pass-through here is a defect this file has already had — a lowercased
74/// `usertype(...)` (#4158).
75pub(crate) fn cql_type_to_marshal_type(cql_type: &str) -> Result<String> {
76    let rendered = render_marshal_type(cql_type, false, false)?;
77    if let Err(refusal) = validate_marshal_frozen(&rendered) {
78        return Err(Error::unsupported_format(format!(
79            "refusing to write SerializationHeader type '{rendered}' for declared \
80             type '{cql_type}': {refusal}"
81        )));
82    }
83    Ok(rendered)
84}

86/// The UDT-FIELD-path disposition of [`cql_type_to_marshal_type`]: identical, except
87/// that a refusal degrades to `BytesType` instead of propagating.
88///
89/// Used ONLY by `data_writer::schema_helpers`, whose `UserType(...)` renderers are
90/// infallible by contract and already DECLARE this exact degradation for a field
91/// they cannot fully represent (`render_udt_marshal`'s scope note; issues
92/// #929/#1011): the direct-write value path writes such a field as an opaque
93/// length-prefixed blob, so `BytesType` keeps the header and the value bytes
94/// SELF-CONSISTENT, which is the property that renderer is documented to preserve.
95/// #4158 changes the COLUMN path's contract, not that one.
96///
97/// Two of its three call sites are provably unreachable-with-an-error, and that is
98/// stated rather than assumed: the ONLY refusal `cql_type_to_marshal_type` can
99/// produce is the vector one, and `render_field_marshal` gives `CqlType::Vector` its
100/// own arm, so its `Custom`/scalar tails never see a vector. The reachable site is
101/// `render_udt_marshal` (the registry-LESS renderer), where a `vector<..>` field
102/// whose element resolves to no marshal class takes the declared degradation.
103///
104/// # A THIRD BEHAVIOUR, DECLARED RATHER THAN LEFT TO BE DISCOVERED (#4158 review)
105///
106/// The degradation also reaches a vector's ELEMENT, one level in. Since #4158
107/// `render_field_marshal` renders `CqlType::Vector` itself and recurses for the
108/// element, so an element that resolves to no marshal class — a UDT FIELD declared
109/// `vector<unregistered_udt, 3>`, i.e. a UDT nested in a UDT with no registry entry
110/// — reaches THIS function through the recursion's `Custom` tail and emits
111///
112/// ```text
113/// org.apache.cassandra.db.marshal.VectorType(org.apache.cassandra.db.marshal.BytesType , 3)
114/// ```
115///
116/// a WRONG element type, where the COLUMN path refuses the same shape by name
117/// (`render_marshal_type`'s `refuse_unknown` arm). So THREE dispositions exist for
118/// one unresolvable name — refuse (column), `BytesType` (UDT field), and
119/// `VectorType(BytesType , n)` (UDT field whose type is a vector) — and this is the
120/// third.
121///
122/// It stays a divergence rather than being unified because the UDT-field renderers
123/// are infallible BY SIGNATURE (`-> String`), so they have no channel to carry the
124/// column path's named refusal; making them consistent means giving those renderers
125/// an error type, which is a contract change to `render_udt_marshal`, not a doc fix.
126///
127/// # THE #4158 POST-CONDITION DEGRADES HERE TOO, AND THAT IS THE DECISION (#4104)
128///
129/// A UDT FIELD whose declared type is an already-marshaled string freezing a
130/// non-freezable type (`FrozenType(Int32Type)`) is refused by
131/// [`cql_type_to_marshal_type`] and therefore reaches `BytesType` HERE, while the
132/// COLUMN path refuses the same input by name. That asymmetry is deliberate and
133/// costs nothing the invariant cares about: the invariant is about what is
134/// EMITTED, and `BytesType` is a legal spelling — the impossible wrapper is
135/// DROPPED, never written. It also keeps this renderer's documented property, that
136/// the header and the value bytes stay SELF-CONSISTENT for a field the direct-write
137/// path writes as an opaque length-prefixed blob (#929/#1011). Pinned by
138/// `passthrough_frozen_tests::a_udt_field_freezing_a_scalar_takes_the_declared_bytestype_degradation`.
139/// The on-disk FRAMING is unaffected either way (`VectorType` and `BytesType` are
140/// both `VARIABLE_LENGTH` at the cell boundary), so the cost is a wrong recorded
141/// element type on a nested-UDT field, not an unreadable file. Pinned by
142/// `schema_helpers_tests::an_unresolvable_vector_element_in_a_udt_field_degrades_to_bytestype`.
143pub(crate) fn cql_type_to_marshal_type_or_bytes(cql_type: &str) -> String {
144    cql_type_to_marshal_type(cql_type)
145        .unwrap_or_else(|_| "org.apache.cassandra.db.marshal.BytesType".to_string())
146}

148/// Marshal type names (without the `org.apache.cassandra.db.marshal.` prefix)
149/// whose `toString(boolean)` carries the
150/// `includeFrozenType = !ignoreFreezing && !isMultiCell` branch — i.e. the ONLY
151/// types that can appear as the immediate inner of a `FrozenType(` in a header
152/// Apache Cassandra wrote (pinned `cassandra-5.0.8`: `ListType.java:195-207`,
153/// `SetType.java:185-197`, `MapType.java:310-321`, `UserType.java:436-448`).
154///
155/// Nothing else has that branch: every scalar inherits
156/// `AbstractType.toString() == getClass().getName()` (`AbstractType.java:741`)
157/// with `toString(boolean) == toString()` (`AbstractType.java:466`), and
158/// `TupleType.toString()` is `getClass().getName() + stringifyTypeParameters(types, true)`
159/// (`TupleType.java:557-560`) — a CQL tuple is already frozen and is never
160/// wrapped. `CQL3Type.Raw::freeze()` correspondingly throws for every
161/// non-collection/tuple/UDT/vector (`CQL3Type.java:647-651`).
162const FREEZE_WRAPPED_HEADS: [&str; 4] = ["ListType(", "SetType(", "MapType(", "UserType("];
163
164/// True iff Cassandra would print a `FrozenType(...)` wrapper around the
165/// already-rendered marshal string `rendered`.
166///
167/// Matching is EXACT-CASE on purpose: the marshal grammar is case-sensitive
168/// (`TypeParser` resolves class names verbatim), so a differently-cased inner is
169/// not a type Cassandra can parse and must not be handed a wrapper that would
170/// make the header unparseable in a NEW way.
171pub(crate) fn takes_frozen_type_wrapper(rendered: &str) -> bool {
172    match rendered.strip_prefix("org.apache.cassandra.db.marshal.") {
173        Some(rest) => FREEZE_WRAPPED_HEADS.iter().any(|h| rest.starts_with(h)),
174        None => false,
175    }
176}

178/// Render `cql_type` as a Cassandra marshal string, mirroring
179/// `AbstractType::toString(boolean ignoreFreezing)`.
180///
181/// `ignore_freezing` is Cassandra's flag: a parent passes
182/// `ignoreFreezing || !isMultiCell` to its subtypes, so every child of a FROZEN
183/// (never-multicell) collection/UDT — and every component of a `TupleType`,
184/// which always passes `true` — drops its own `FrozenType(...)` wrapper, while a
185/// child of a MULTICELL collection keeps it. Corpus-attested both ways:
186/// `LIST<FROZEN<address_type>>` → `ListType(FrozenType(UserType(...)))`, but the
187/// `address` field INSIDE `FrozenType(UserType(...,contact_info,...))` is the
188/// bare `UserType(...)`.
189/// `refuse_unknown` is set ONLY inside a vector element (issue #4158) and is
190/// inherited by every nested call. It flips the unknown-bare-name tail from the
191/// documented `BytesType` degradation (#929) to a NAMED refusal, because a vector's
192/// element type is what decides the value's on-disk WIDTH: `VectorType`'s
193/// `valueLengthIfFixed()` is `elementType.valueLengthIfFixed() * dimension` when the
194/// element is fixed and `VARIABLE_LENGTH` otherwise (`VectorType.java:94-96`). So a
195/// wrong element type there is not a lost type name, it is a wrong FRAMING stamped
196/// into `Statistics.db` — the reader derives the vector's layout from exactly this
197/// string (`parser::repair_clustering::resolve_clustering_value_layout`, #4149).
198/// Outside a vector element the flag is `false` and the fallback is untouched.
199fn render_marshal_type(
200    cql_type: &str,
201    ignore_freezing: bool,
202    refuse_unknown: bool,
203) -> Result<String> {
204    // Already a marshal string (e.g. a UDT column normalized to UserType(...),
205    // or a column type read back from an input SSTable's SerializationHeader):
206    // return it verbatim. The marshal grammar is case-sensitive (UserType,
207    // Int32Type, ...), so this MUST happen before the lowercasing below, and the
208    // original-case string MUST be preserved. Without this, an already-marshaled
209    // type would fall through to BytesType, advertising the wrong type in the
210    // header while Data.db carries the real (e.g. complex UDT) cells (#929).
211    //
212    // VERBATIM, AND STILL NOT UNCHECKED: this string reaches the output without
213    // passing any arm that could have judged it. The #4158 invariant is therefore
214    // enforced as a POST-CONDITION on the rendered result in
215    // `cql_type_to_marshal_type` (#4104, roborev job 121) rather than by a check
216    // added here: the invariant is a property of what is EMITTED, so validating
217    // the emitted string covers this arm, every other arm, and any arm added
218    // later, whereas a per-arm check is only ever as complete as the list of arms
219    // someone remembered.
220    let raw = cql_type.trim();
221    if raw
222        .to_lowercase()
223        .starts_with("org.apache.cassandra.db.marshal.")
224    {
225        return Ok(raw.to_string());
226    }
227
228    let prefix = "org.apache.cassandra.db.marshal.";
229
230    // Handle parameterized types: list<T>, set<T>, map<K,V>, frozen<T>, tuple<T1,T2>.
231    //
232    // Every wrapper is stripped from the ORIGINAL-CASE string (CQL type names
233    // are case-insensitive but the marshal names nested inside are NOT — see
234    // `strip_cql_wrapper`), and the inner is re-entered through this same
235    // function so the `ignore_freezing` flag propagates exactly as Cassandra
236    // propagates it.
237    //
238    // A bare `list`/`set`/`map` is MULTICELL, so it never prints a wrapper for
239    // itself (that is the `frozen<..>` arm's job) and passes `ignore_freezing`
240    // through unchanged (`ignoreFreezing || !isMultiCell` with `isMultiCell`).
241    if let Some(inner) = strip_cql_wrapper(raw, "list") {
242        return Ok(format!(
243            "{prefix}ListType({})",
244            render_marshal_type(inner, ignore_freezing, refuse_unknown)?
245        ));
246    }
247    if let Some(inner) = strip_cql_wrapper(raw, "set") {
248        return Ok(format!(
249            "{prefix}SetType({})",
250            render_marshal_type(inner, ignore_freezing, refuse_unknown)?
251        ));
252    }
253    if let Some(inner) = strip_cql_wrapper(raw, "map") {
254        let args = split_cql_type_args(inner);
255        if args.len() == 2 {
256            return Ok(format!(
257                "{prefix}MapType({},{})",
258                render_marshal_type(args[0], ignore_freezing, refuse_unknown)?,
259                render_marshal_type(args[1], ignore_freezing, refuse_unknown)?
260            ));
261        }
262        // Malformed map type — fall through to BytesType
263    }
264    if let Some(inner) = strip_cql_wrapper(raw, "frozen") {
265        // `frozen<T>` is `T.freeze()`: never multicell, so its subtypes are
266        // stringified with `ignoreFreezing = true` (which also makes `freeze()`
267        // idempotent — a nested `frozen<..>` collapses).
268        let rendered = render_marshal_type(inner, true, refuse_unknown)?;
269        // Print the wrapper only where Cassandra's own writer can: not when a
270        // frozen/tuple parent already suppressed it, and never around a type
271        // that has no `includeFrozenType` branch. Emitting
272        // `FrozenType(<scalar>)` — e.g. `FrozenType(BytesType)` for a `frozen<udt>`
273        // whose name no `UdtRegistry` could resolve — advertises a type no
274        // Cassandra writer can produce (#4158). Dropping the wrapper leaves the
275        // legal spelling of the inner, which for an unresolved UDT name is the
276        // same `BytesType` degradation the BARE unresolved name already takes
277        // (#929: no registry => single simple cell).
278        if ignore_freezing || !takes_frozen_type_wrapper(&rendered) {
279            return Ok(rendered);
280        }
281        return Ok(format!("{prefix}FrozenType({rendered})"));
282    }
283    if let Some(inner) = strip_cql_wrapper(raw, "tuple") {
284        // `TupleType.toString()` has no `includeFrozenType` branch and passes
285        // `true` for its components (`TupleType.java:557-560`).
286        let args = split_cql_type_args(inner);
287        let components = args
288            .iter()
289            .map(|a| render_marshal_type(a, true, refuse_unknown))
290            .collect::<Result<Vec<String>>>()?;
291        return Ok(format!("{prefix}TupleType({})", components.join(",")));
292    }
293
294    // ═══ `vector<element, n>` — issue #4158, the FAIL-OPEN this arm removes ═══
295    //
296    // Without it a vector column reached the `_ => BytesType` fallback below and
297    // CQLite stamped `org.apache.cassandra.db.marshal.BytesType` into the
298    // SerializationHeader — the same defect class as the `FrozenType(BytesType)`
299    // bug this file already fixed: a WRONG type recorded in `Statistics.db`, and one
300    // that compiles silently because the fallback is over a `&str`, not a `CqlType`.
301    //
302    // THE SPELLING, DERIVED AT THE PINNED TAG (`cassandra-5.0.8`):
303    //   VectorType.java:338-342
304    //     public String toString(boolean ignoreFreezing) {
305    //         return getClass().getName()
306    //              + TypeParser.stringifyVectorParameters(elementType, ignoreFreezing, dimension);
307    //     }
308    //   TypeParser.java:239-242
309    //     public static String stringifyVectorParameters(AbstractType<?> type, boolean ignoreFreezing, int dimension) {
310    //         return "(" + type.toString(ignoreFreezing) + " , " + dimension + ")";
311    //     }
312    // So: the package-qualified class name, then `(`, the ELEMENT's own
313    // `toString(ignoreFreezing)`, then the literal `" , "` (SPACE comma SPACE — not
314    // the bare `,` that `stringifyTypeParameters` uses for collections), the decimal
315    // dimension, `)`.
316    //
317    // BYTE ORACLE, Cassandra-written, not CQLite output: the header of
318    // `test-data/fixtures/issue_4114/test_vector/vector_exact-*/nb-1-big-Statistics.db`
319    // carries, at a VInt length of 0x59 = 89,
320    //   org.apache.cassandra.db.marshal.VectorType(org.apache.cassandra.db.marshal.FloatType , 3)
321    // and `vector_pk_only-*` carries the same shape at dimension 1 and 384. Pinned in
322    // `issue_4158_vector_tests::cassandra_written_vector_header_is_reproduced` (this
323    // module's `#[cfg(test)] mod issue_4158_vector_tests`, marshal.rs:12).
324    //
325    // NO `FrozenType(` WRAPPER, EVER: `VectorType.toString(boolean)` has no
326    // `includeFrozenType` branch (it is the two lines quoted above), so a vector is
327    // never wrapped — which is why `VectorType(` is absent from
328    // [`FREEZE_WRAPPED_HEADS`] and `frozen<vector<float, 3>>` renders as the bare
329    // `VectorType(FloatType , 3)`. That matches CQL: `RawVector::freeze()` returns
330    // `this` (`CQL3Type.java:915-919`), so `frozen<vector<..>>` and `vector<..>` are
331    // the SAME Cassandra type with one spelling.
332    //
333    // THE PARAMETERS COME FROM THE ONE SHARED PARSER (`schema::vector_type`), never
334    // from a second `<`/`>` walk here: `Ok(None)` is "not a vector" and falls
335    // through, and a MALFORMED vector is its `Err`, propagated — so a bad dimension
336    // or a third parameter is refused BY NAME instead of degrading to `BytesType`.
337    // The ELEMENT is rendered by THIS SAME recursion at the SAME `ignore_freezing`
338    // — `stringifyVectorParameters` passes the flag straight through
339    // (`type.toString(ignoreFreezing)`), unlike `stringifyTypeParameters(types, true)`
340    // which a tuple forces to `true`. For a scalar element the flag provably cannot
341    // matter (`AbstractType.toString(boolean)` is `return this.toString();`,
342    // `AbstractType.java:466-469`), but it is threaded rather than assumed away so a
343    // parametric element behaves like every other subtype in the tree.
344    //
345    // `refuse_unknown = true` for the element and everything under it: see that
346    // parameter's doc. It converts the ONE untruthful outcome — the unknown-bare-name
347    // `BytesType` degradation — into a named refusal, and it is what makes this arm's
348    // claim total rather than "total except when the element is a UDT nobody
349    // registered".
350    if let Some(args) = cql_vector_kind(raw).into_args(raw)? {
351        return Ok(format!(
352            "{prefix}VectorType({} , {})",
353            render_marshal_type(args.element.trim(), ignore_freezing, true)?,
354            args.dimension
355        ));
356    }
357
358    // Primitive types. CQL type names are case-insensitive, and the parser may
359    // preserve original case from CQL files (e.g. "SET<TEXT>"), so match on a
360    // lowercased copy.
361    //
362    // The fallback is UNCHANGED (#929): an unresolvable bare name renders as
363    // `BytesType`, which is the documented degradation for a UDT column with no
364    // `UdtRegistry` in play. Only the VECTOR arm above refuses, and only because a
365    // vector's element type is what decides the value's on-disk WIDTH.
366    match primitive_marshal_name(&raw.to_lowercase()) {
367        Some(name) => Ok(format!("{prefix}{name}")),
368        // Fallback: use BytesType for unknown types — UNLESS this name is (inside)
369        // a vector element, where it would be a wrong WIDTH rather than a lost name.
370        None if refuse_unknown => Err(Error::unsupported_format(format!(
371            "cannot write a truthful SerializationHeader type for a vector whose \
372             element type is or contains '{raw}': this converter resolves no marshal \
373             class for that name (it has no UDT registry), and Cassandra spells a \
374             vector as VectorType(<element>.toString(ignoreFreezing) , <dimension>) \
375             (TypeParser.stringifyVectorParameters, TypeParser.java:239-242, reached \
376             from VectorType.toString, VectorType.java:338-342). Substituting the \
377             BytesType fallback would record a wrong element type — and therefore a \
378             wrong value width, VectorType.java:94-96 — in Statistics.db (issue \
379             #4158), so this is refused by name rather than guessed (issue #28)"
380        ))),
381        None => Ok(format!("{prefix}BytesType")),
382    }
383}

385/// The marshal SIMPLE NAME of a native CQL scalar, or `None` if `lower` (already
386/// lowercased) names no scalar this converter knows.
387///
388/// Split out of [`render_marshal_type`]'s tail so the vector arm can ASK whether an
389/// element is a known scalar without having to recognise the `_ => BytesType`
390/// fallback after the fact — `blob` legitimately renders to `BytesType`, so the
391/// rendered string cannot distinguish "blob" from "unknown". The mapping itself is
392/// unchanged.
393fn primitive_marshal_name(lower: &str) -> Option<&'static str> {
394    Some(match lower {
395        "text" | "varchar" => "UTF8Type",
396        "int" => "Int32Type",
397        "bigint" => "LongType",
398        "smallint" => "ShortType",
399        "tinyint" => "ByteType",
400        "float" => "FloatType",
401        "double" => "DoubleType",
402        "boolean" => "BooleanType",
403        "blob" => "BytesType",
404        "uuid" => "UUIDType",
405        "timeuuid" => "TimeUUIDType",
406        "timestamp" => "TimestampType",
407        "date" => "SimpleDateType",
408        "time" => "TimeType",
409        "duration" => "DurationType",
410        "inet" => "InetAddressType",
411        "ascii" => "AsciiType",
412        "decimal" => "DecimalType",
413        "varint" => "IntegerType",
414        "counter" => "CounterColumnType",
415        _ => return None,
416    })
417}

419/// Strip a CQL wrapper type like `list<inner>` and return the inner string.
420/// Returns None if `cql_type` does not start with `wrapper<`.
421///
422/// The wrapper keyword is matched CASE-INSENSITIVELY (CQL type names are
423/// case-insensitive; a schema may carry `SET<TEXT>`), but the slice returned is
424/// of the ORIGINAL-CASE input. That matters: an inner that is itself an
425/// `org.apache.cassandra.db.marshal.*` string must reach the recursion with its
426/// case intact, because the marshal grammar is case-sensitive — lowercasing
427/// first turned `list<...UserType(...)>` into an unparseable
428/// `ListType(...usertype(...))` (#4158).
429fn strip_cql_wrapper<'a>(cql_type: &'a str, wrapper: &str) -> Option<&'a str> {
430    if !cql_type.is_char_boundary(wrapper.len()) {
431        return None;
432    }
433    let (head, tail) = cql_type.split_at(wrapper.len());
434    if !head.eq_ignore_ascii_case(wrapper) {
435        return None;
436    }
437    let rest = tail.strip_prefix('<')?;
438    // Find the matching closing '>' (handling nested angle brackets)
439    let mut depth = 1;
440    for (i, ch) in rest.char_indices() {
441        match ch {
442            '<' => depth += 1,
443            '>' => {
444                depth -= 1;
445                if depth == 0 {
446                    return Some(rest[..i].trim());
447                }
448            }
449            _ => {}
450        }
451    }
452    None
453}

455/// Split CQL type arguments at top-level commas (respecting nested angle brackets).
456/// E.g. `"int, map<text, int>"` → `["int", "map<text, int>"]`
457fn split_cql_type_args(s: &str) -> Vec<&str> {
458    let mut result = Vec::new();
459    let mut depth = 0;
460    let mut start = 0;
461    for (i, ch) in s.char_indices() {
462        match ch {
463            '<' => depth += 1,
464            '>' => depth -= 1,
465            ',' if depth == 0 => {
466                result.push(s[start..i].trim());
467                start = i + 1;
468            }
469            _ => {}
470        }
471    }
472    let last = s[start..].trim();
473    if !last.is_empty() {
474        result.push(last);
475    }
476    result
477}

559/// Issue #4158 — Cassandra-attested `FrozenType(...)` wrapper eligibility.
560///
561/// ORACLE (pinned `cassandra-5.0.8`, NOT CQLite's own output): only four
562/// marshal types carry the `includeFrozenType = !ignoreFreezing && !isMultiCell`
563/// branch that emits the wrapper, so only these four can EVER appear as the
564/// immediate inner of a `FrozenType(`:
565///
566/// * `ListType.toString(boolean)`  — `ListType.java:195-207`
567/// * `SetType.toString(boolean)`   — `SetType.java:185-197`
568/// * `MapType.toString(boolean)`   — `MapType.java:310-321`
569/// * `UserType.toString(boolean)`  — `UserType.java:436-448`
570///
571/// Everything else cannot be wrapped:
572/// * every scalar inherits `AbstractType.toString() == getClass().getName()`
573///   (`AbstractType.java:741-744`) and `toString(boolean) == toString()`
574///   (`AbstractType.java:466-469`) — no wrapper branch at all;
575/// * `TupleType.toString()` is `getClass().getName() + stringifyTypeParameters(types, true)`
576///   (`TupleType.java:557-560`) — it never emits the wrapper for itself and
577///   always stringifies its components with `ignoreFreezing = true`.
578///
579/// Corroborated by the corpus census (`test-data` / `CQLITE_DATASETS_ROOT`): of
580/// every `FrozenType(` Apache Cassandra wrote across the corpus the immediate
581/// inner head is only ever MapType / ListType / UserType / SetType — never a
582/// scalar and never TupleType. The end-to-end oracle lives in
583/// `cqlite-core/tests/issue_4158_frozen_wrapper_cassandra_oracle.rs`.
584#[cfg(test)]
585mod frozen_wrapper_cassandra_parity {
586    use super::*;
587
588    const Q: &str = "org.apache.cassandra.db.marshal.";
589
590    /// See the sibling helper in `mod tests`: #4158 made the converter fallible.
591    fn marshal(cql: &str) -> String {
592        cql_type_to_marshal_type(cql).expect("a renderable marshal type")
593    }
594
595    /// A `frozen<...>` whose inner is a bare name this string converter cannot
596    /// resolve (a UDT with no `UdtRegistry` in play) fell through to the
597    /// `_ => BytesType` arm and was then WRAPPED, emitting
598    /// `FrozenType(BytesType)` — a spelling no Cassandra writer can produce
599    /// (`BytesType` is `blob`, a scalar; `CQL3Type.Raw::freeze()` throws for
600    /// every non-collection/tuple/UDT/vector — `CQL3Type.java:647-651`).
601    /// The wrapper must be DROPPED, leaving the legal scalar spelling — the
602    /// same documented degradation a BARE unresolved UDT name already takes
603    /// (issue #929: no registry => single simple cell).
604    #[test]
605    fn frozen_unresolvable_name_never_wraps_a_scalar() {
606        assert_eq!(
607            marshal("frozen<person>"),
608            format!("{Q}BytesType"),
609            "frozen<unresolved-udt> must degrade to the same legal spelling as \
610             the bare unresolved name, never FrozenType(<scalar>)"
611        );
612        // The bare-name fallback it must agree with.
613        assert_eq!(
614            marshal("person"),
615            marshal("frozen<person>"),
616            "the frozen and bare unresolved-name fallbacks must agree"
617        );
618    }
619
620    /// `frozen<scalar>` is not even expressible in CQL, but the converter is
621    /// reachable with one (a hand-built schema, an exporter round-trip). It must
622    /// never fabricate the impossible wrapper.
623    #[test]
624    fn frozen_scalar_never_wraps() {
625        for (cql, expected) in [
626            ("frozen<int>", "Int32Type"),
627            ("frozen<text>", "UTF8Type"),
628            ("frozen<blob>", "BytesType"),
629            ("frozen<uuid>", "UUIDType"),
630            ("frozen<timestamp>", "TimestampType"),
631        ] {
632            assert_eq!(
633                marshal(cql),
634                format!("{Q}{expected}"),
635                "{cql} must render as the bare scalar"
636            );
637        }
638    }
639
640    /// `TupleType.toString()` (`TupleType.java:557-560`) has NO
641    /// `includeFrozenType` branch, so Cassandra never wraps a tuple — a CQL
642    /// tuple is already frozen. Corpus-confirmed: every `TupleType(` Cassandra
643    /// wrote in the corpus is unwrapped, and no `FrozenType(TupleType` exists
644    /// anywhere in it.
645    #[test]
646    fn frozen_tuple_never_wraps() {
647        let expected = format!("{Q}TupleType({Q}Int32Type,{Q}UTF8Type)");
648        assert_eq!(marshal("tuple<int, text>"), expected);
649        assert_eq!(
650            marshal("frozen<tuple<int, text>>"),
651            expected,
652            "frozen<tuple<..>> and tuple<..> are the SAME Cassandra type and \
653             share one spelling"
654        );
655    }
656
657    /// `TupleType` stringifies its components with `ignoreFreezing = true`, and
658    /// a frozen collection/UserType propagates `ignoreFreezing || !isMultiCell`
659    /// to ITS subtypes (`stringifyTypeParameters(..., ignoreFreezing || !isMultiCell)`).
660    /// So a frozen type nested inside an already-frozen (or tuple) parent loses
661    /// its own wrapper. Corpus-attested for the UserType case: inside
662    /// `FrozenType(UserType(test_collections,contact_info,...))` the nested
663    /// `address` field is the BARE `UserType(...)`, not `FrozenType(UserType(...))`.
664    #[test]
665    fn ignore_freezing_propagates_into_frozen_and_tuple_children() {
666        assert_eq!(
667            marshal("tuple<int, frozen<list<int>>>"),
668            format!("{Q}TupleType({Q}Int32Type,{Q}ListType({Q}Int32Type))"),
669            "a tuple component is stringified with ignoreFreezing = true"
670        );
671        assert_eq!(
672            marshal("frozen<list<frozen<set<int>>>>"),
673            format!("{Q}FrozenType({Q}ListType({Q}SetType({Q}Int32Type)))"),
674            "a frozen child of a frozen parent drops its own wrapper"
675        );
676    }
677
678    /// The REGRESSION half: every spelling Apache Cassandra itself wrote into
679    /// the corpus must be reproduced byte-for-byte. Each expectation below is
680    /// the exact string extracted from a Cassandra-written `Statistics.db`
681    /// (`test_collections/frozen_collections_table-*`,
682    /// `test_collections/typed_collections_table-*`), never from CQLite output.
683    #[test]
684    fn cassandra_written_spellings_are_unchanged() {
685        // FROZEN<SET<TEXT>> / FROZEN<LIST<INT>> / FROZEN<MAP<TEXT,TEXT>> and a
686        // non-frozen SET<TEXT>, all from frozen_collections_table's header.
687        assert_eq!(
688            marshal("frozen<set<text>>"),
689            format!("{Q}FrozenType({Q}SetType({Q}UTF8Type))")
690        );
691        assert_eq!(
692            marshal("frozen<list<int>>"),
693            format!("{Q}FrozenType({Q}ListType({Q}Int32Type))")
694        );
695        assert_eq!(
696            marshal("frozen<map<text, text>>"),
697            format!("{Q}FrozenType({Q}MapType({Q}UTF8Type,{Q}UTF8Type))")
698        );
699        assert_eq!(marshal("set<text>"), format!("{Q}SetType({Q}UTF8Type)"));
700        // A non-frozen (multicell) collection propagates ignoreFreezing = false,
701        // so a FROZEN element KEEPS its wrapper. Corpus-attested by
702        // collections_with_udts: `LIST<FROZEN<address_type>>` ->
703        // `ListType(FrozenType(UserType(...)))`.
704        assert_eq!(
705            marshal("list<frozen<set<int>>>"),
706            format!("{Q}ListType({Q}FrozenType({Q}SetType({Q}Int32Type)))"),
707            "a frozen element of a MULTICELL collection keeps its wrapper"
708        );
709        // An already-marshaled UserType inner keeps the wrapper (the resolved
710        // frozen-UDT column shape asserted against Cassandra bytes by
711        // `frozen_udt_column_renders_frozentype_usertype_cassandra_parity`).
712        let user_type = format!("{Q}UserType(ks,706572736f6e,6e616d65:{Q}UTF8Type)");
713        assert_eq!(
714            marshal(&format!("frozen<{user_type}>")),
715            format!("{Q}FrozenType({user_type})")
716        );
717    }
