11/// Issue #1640 (K1): [`SlidingPartitionPolicy`] for the streaming-scan /
12/// compaction-read `(TableId, RowKey, ScanRow, row_timestamp)` path
13/// (`parse_block_with_timestamps` family). Read-side shadowing is active only
14/// when the parser was built for a user-facing query read (`read_shadowing`);
15/// physical consumers leave `shadow` `None` and see every on-disk row.
16pub(super) struct TimestampPolicy<'a> {
17    parser: &'a V5CompressedLegacyParser,
18    table_id: TableId,
19    partition_key: RowKey,
20    /// Issue #1741 per-partition read-side shadow; `None` for physical reads.
21    shadow: Option<PartitionShadow>,
22    /// Issue #480 static cells accumulated for merge into each clustering row.
23    /// Issue #1642 (K3): positional `RowCells`, matching the decoder emit.
24    static_cells: RowCells,
25    /// Issue #3095: the static row's own write timestamp, carried so a
26    /// static-only partition's synthesized row reports the SAME authoritative
27    /// timestamp rule (`row_write_timestamp`) every other row does.
28    static_row_ts: i64,
29    /// Issue #3095: whether this partition yielded at least one CLUSTERING row —
30    /// Cassandra's `partition.hasNext()`, which counts clustering rows only
31    /// because the static row is delivered out of band by `partition.staticRow()`
32    /// (`db/rows/BaseRowIterator.java`).
33    emitted_clustering_row: bool,
34}
35
36impl<'a> TimestampPolicy<'a> {
37    pub(super) fn new(parser: &'a V5CompressedLegacyParser) -> Self {
38        Self {
39            parser,
40            table_id: TableId::new(format!("{}.{}", parser.keyspace, parser.table_name)),
41            partition_key: RowKey::new(Vec::new()),
42            shadow: None,
43            static_cells: Vec::new(),
44            static_row_ts: 0,
45            emitted_clustering_row: false,
46        }
47    }
48
49    /// How this policy names the partition in a report. The driver decodes ONE
50    /// partition per call, so the key is the only identifier available here.
51    fn partition(&self) -> String {
52        format!("key 0x{}", hex::encode(self.partition_key.as_bytes()))
53    }
54}

56#[cfg(test)]
57impl TimestampPolicy<'_> {
58    /// Test seam (issue #3095): install a live STATIC row for the current partition
59    /// exactly as `on_data_row`'s `is_static` branch does, so
60    /// [`TimestampPolicy::on_partition_close`]'s `complete` / `read_shadowing` /
61    /// visible-row decisions can be asserted DIRECTLY rather than only end-to-end.
62    pub(super) fn seed_static_for_test(
63        &mut self,
64        partition_key: RowKey,
65        schema: &TableSchema,
66        row_ts: i64,
67    ) {
68        let static_column = schema
69            .columns
70            .iter()
71            .find(|c| c.is_static)
72            .map(|c| c.name.clone())
73            .unwrap_or_else(|| "s".to_string());
74        self.partition_key = partition_key;
75        self.static_cells = vec![(Arc::from(static_column.as_str()), Value::text("static-val"))];
76        self.static_row_ts = row_ts;
77    }
78
79    /// Test seam: mark that the partition produced a VISIBLE clustering row.
80    pub(super) fn note_visible_row_for_test(&mut self) {
81        self.emitted_clustering_row = true;
82    }
83}

85impl SlidingPartitionPolicy for TimestampPolicy<'_> {
86    type Row = (TableId, RowKey, ScanRow, i64);
87
88    fn on_partition_open(
89        &mut self,
90        partition_key: RowKey,
91        partition_deletion: Option<(i64, i32)>,
92        schema: &TableSchema,
93        _pending: &mut Vec<Self::Row>,
94    ) {
95        self.partition_key = partition_key;
96        // Issue #3095 (roborev BLOCKER): RESET every partition-scoped field here.
97        // Not defensive style — required for correctness: the driver calls this
98        // hook again when a mid-partition `NeedMore` makes the caller refill and
99        // RE-PARSE the same partition from its start (its `pending` rows are
100        // discarded), and a policy instance may be reused across partitions. Left
101        // un-reset, `emitted_clustering_row` from an earlier partition/attempt would
102        // permanently suppress a LATER static-only partition's row, and
103        // `static_cells` would leak one partition's static values into the next.
104        self.static_cells = Vec::new();
105        self.static_row_ts = 0;
106        self.emitted_clustering_row = false;
107        // Issue #1741 (F2): reuse the parser's once-per-read `now_secs`, not a
108        // per-block wall-clock sample, so all blocks of one scan share one `now`.
109        self.shadow = self.parser.read_shadowing.then(|| {
110            PartitionShadow::open(
111                self.parser.now_secs,
112                partition_deletion,
113                clustering_reversed_flags(schema),
114            )
115        });
116    }

118    fn on_range_marker(
119        &mut self,
120        data: &[u8],
121        offset: usize,
122        schema: &TableSchema,
123        _pending: &mut Vec<Self::Row>,
124    ) -> MarkerOutcome {
125        // Issue #1741: on the user-facing scan path decode the marker and feed
126        // the range-tombstone FSM so covered rows are shadowed; the physical path
127        // (shadow == None) only advances past it.
128        if let Some(sh) = self.shadow.as_mut() {
129            match self
130                .parser
131                .parse_range_tombstone_marker_full(data, offset, schema)
132            {
133                Ok((bv, bk, dp, ds, next_offset)) => {
134                    // Issue #3721: the marker PARSED (`next_offset` is bound, so the
135                    // partition body continues there) and the FSM refuses only an
136                    // unrepresentable bound kind. Reporting that as `Stop` — the
137                    // framing terminator a marker PARSE failure earns — made the
138                    // driver `flush_and_emitted!(offset, false)`, i.e. `Ok` with the
139                    // tombstone and every later row of the partition gone. The two
140                    // halves stay distinguished: see `MarkerOutcome`.
141                    if let Err(e) = sh.feed_range_marker(bv, bk, dp, ds) {
142                        return MarkerOutcome::Refused(range_marker_error::range_marker_refused(
143                            e,
144                            &self.partition(),
145                            offset,
146                            next_offset,
147                        ));
148                    }
149                    MarkerOutcome::Advanced(next_offset)
150                }
151                // Issue #3721 (roborev job 78): a marker that cannot be PARSED is
152                // NOT a framing terminator, and calling it one made the FINAL
153                // chunk complete the partition SUCCESSFULLY with the tombstone
154                // and every later row gone from a `SELECT`. The cause is
155                // PRESERVED and the FINAL-vs-non-final decision is left to
156                // `drive_partition_sliding`, the only holder of the chunking
157                // state — a non-final chunk may simply have cut the marker body
158                // in half, the one explanation a refill can fix. Nothing here
159                // inspects bytes to guess whether more data exists (issue #28).
160                // Reasoning + authority: `range_marker_error`'s module docs.
161                Err(cause) => MarkerOutcome::Unparseable(
162                    range_marker_error::range_marker_unparseable(cause, &self.partition(), offset),
163                ),
164            }
165        } else {
166            match self
167                .parser
168                .skip_range_tombstone_marker(data, offset, schema)
169            {
170                Ok(next_offset) => MarkerOutcome::Advanced(next_offset),
171                // Same decision on the PHYSICAL path: a marker whose framing
172                // cannot be skipped is corrupt or truncated data, and reporting
173                // it as end-of-partition truncated the read and said `Ok`.
174                Err(cause) => MarkerOutcome::Unparseable(
175                    range_marker_error::range_marker_unparseable(cause, &self.partition(), offset),
176                ),
177            }
178        }
179    }

181    fn on_data_row(
182        &mut self,
183        data: &[u8],
184        offset: usize,
185        schema: &TableSchema,
186        reader: &crate::storage::sstable::reader::types::SSTableReader,
187        resolution: &RowColumnResolution,
188        pending: &mut Vec<Self::Row>,
189    ) -> DataRowOutcome {
190        match self.parser.parse_row_data_with_offset(
191            data,
192            offset,
193            Some(schema),
194            reader,
195            false,
196            resolution,
197            self.shadow.as_ref(),
198        ) {
199            Ok((
200                mut cells,
201                _row_cell_meta,
202                row_header_opt,
203                next_offset,
204                is_static,
205                _complex_meta,
206            )) => {
207                // Issue #932 (K1): the ONE row-write-timestamp coexistence rule.
208                let row_ts = row_write_timestamp(&row_header_opt);
209
210                if is_static {
211                    // Issue #1741 (Finding 1): drop a shadowed/expired static row's
212                    // cells so a surviving clustering row does not resurface stale
213                    // static data. No-op when shadowing is off.
214                    let static_hidden = self.shadow.as_ref().is_some_and(|sh| {
215                        row_header_opt
216                            .as_ref()
217                            .is_some_and(|h| sh.row_hidden(h, &[]))
218                    });
219                    self.static_cells = if static_hidden { Vec::new() } else { cells };
220                    self.static_row_ts = row_ts;
221                } else {
222                    // Issue #1741: hide rows shadowed by a partition/range tombstone
223                    // or expired by TTL (matching a Cassandra SELECT). No-op on the
224                    // physical path (shadow == None).
225                    let hidden = self.shadow.as_ref().is_some_and(|sh| {
226                        row_header_opt.as_ref().is_some_and(|h| {
227                            let clustering = if sh.needs_clustering() {
228                                extract_clustering_values(&cells, schema)
229                            } else {
230                                Vec::new()
231                            };
232                            sh.row_hidden(h, &clustering)
233                        })
234                    });
235
236                    // Issue #505/#932 row-tombstone display rule + issue #480/#1642
237                    // static merge. Issue #3095: on a user-facing SELECT read the
238                    // tombstone decision is taken over the row's OWN cells FIRST, so a
239                    // static value cannot revive a row-tombstoned row; PHYSICAL
240                    // consumers (compaction) keep the historical order and stay
241                    // byte-identical. See `build_display_row_read_path`.
242                    let row_value = if self.shadow.is_some() {
243                        build_display_row_read_path(
244                            cells,
245                            &self.static_cells,
246                            row_header_opt.as_ref(),
247                            schema,
248                        )
249                    } else {
250                        merge_static_cells(&mut cells, &self.static_cells);
251                        build_display_row(cells, row_header_opt.as_ref(), schema)
252                    };
253
254                    if !hidden {
255                        // Issue #3095 (roborev + rust-reviewer BLOCKER): Cassandra's
256                        // `partition.hasNext()` is evaluated over the ALREADY-FILTERED
257                        // `RowIterator` (`UnfilteredRowIterators.filter`), so a purged
258                        // row does NOT count as a row. A `ScanRow::Marker` is a pure
259                        // row tombstone that every user-facing consumer suppresses
260                        // (`filter_tombstone` / `build_row_from_scan`'s `into_cells`),
261                        // strictly AFTER this point — counting it would make a
262                        // partition whose ONLY clustering rows are deleted emit
263                        // nothing, where Cassandra (and the k-way merge arm) return
264                        // the static row. Count only a VISIBLE row.
265                        self.emitted_clustering_row |= row_is_visible(&row_value);
266                        pending.push((
267                            self.table_id.clone(),
268                            self.partition_key.clone(),
269                            row_value,
270                            row_ts,
271                        ));
272                    }
273                }
274                DataRowOutcome::Decoded(next_offset)
275            }
276            // Issue #3782: preserve the decode error instead of collapsing it into
277            // "no row here". The DRIVER decides whether to tolerate it, from
278            // `at_final_chunk`; a policy has no way to know. Classified
279            // `DecodeFailed` and never `Refused` (#3809) because that is exactly
280            // what it is — the row's bytes did not yield a row, which more bytes
281            // may still fix. This policy raises no refusal: the user-facing read
282            // path builds a display row and has no row-BUILD invariant to fail.
283            Err(e) => DataRowOutcome::DecodeFailed(e),
284        }
285    }

287    /// Issue #3095: Cassandra's static-content-on-an-empty-partition rule.
288    ///
289    /// `SelectStatement.processPartition()` (cassandra-5.0.8, L1099-1120): when
290    /// `!partition.hasNext()` — no CLUSTERING rows — and the out-of-band
291    /// `partition.staticRow()` is not empty, the query returns EXACTLY ONE result
292    /// row whose clustering and REGULAR columns are null
293    /// (`default: result.add((ByteBuffer) null)`) and whose STATIC columns carry
294    /// the static row's values. That branch `return`s, so it is mutually exclusive
295    /// with the per-row loop — which is why this fires only when no clustering row
296    /// was emitted for the partition.
297    ///
298    /// Gated on `read_shadowing`, i.e. user-facing SELECT reads only: a PHYSICAL
299    /// consumer (`verify`, `get_all_entries`, compaction, delta-scan) must see
300    /// exactly the on-disk unfiltereds, so it never gets this synthesized row and
301    /// sstabledump/compaction parity is byte-unchanged.
302    ///
303    /// `self.static_cells` is already empty when the static row was shadowed by a
304    /// partition tombstone or expired by its own TTL (#1741 Finding 1), so a stale
305    /// static row can never resurface here. The clustering-restriction half of
306    /// `returnStaticContentOnPartitionWithNoRows()` is enforced downstream: the
307    /// row carries NULL clustering and NULL regular columns, so any restriction on
308    /// one of those (the only way `queriesFullPartitions()` becomes false) rejects
309    /// it under the reader's/producer's three-valued predicate rule, which keeps a
310    /// row only when the predicate is definitely True.
311    fn on_partition_close(
312        &mut self,
313        schema: &TableSchema,
314        pending: &mut Vec<Self::Row>,
315        complete: bool,
316    ) {
317        // `complete` (issue #3095): synthesize ONLY at a STRUCTURALLY-confirmed
318        // partition end. A truncated/corrupt body (buffer exhausted with no
319        // END_OF_PARTITION, an unrepresentable range marker, an unparseable row) was
320        // only PARTLY observed, so "this partition yielded no clustering row" is not
321        // knowable — fail closed rather than invent a row. This is the SAME guard
322        // the two `block_emit*` decode sites apply via their own `partition_complete`,
323        // so all three paths now agree and the hook's doc is literally true.
324        if !complete
325            || !self.parser.read_shadowing
326            || self.emitted_clustering_row
327            || self.static_cells.is_empty()
328        {
329            return;
330        }
331        // Cells are non-empty and hold at least one static (non-key) column, so
332        // the shared display rule yields `ScanRow::Row`; the header is not needed
333        // (a shadowed/tombstoned static row already emptied `static_cells`).
334        let row_value = build_display_row(std::mem::take(&mut self.static_cells), None, schema);
335        pending.push((
336            self.table_id.clone(),
337            self.partition_key.clone(),
338            row_value,
339            self.static_row_ts,
340        ));
341    }
