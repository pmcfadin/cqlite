1//! One place that turns an untrusted VUInt-decoded length into a `usize`
2//! (issue #3848, truncation axis).
3//!
4//! `parse_vuint` yields a `u64` — up to `u64::MAX` from a 9-byte encoding — and
5//! the historical shape at every call site was `let len = len_raw as usize;`
6//! FOLLOWED by a bounds check. That ordering is unsound on any target where
7//! `usize` is narrower than 64 bits (`wasm32-unknown-unknown` is a configured
8//! target of this workspace): `1u64 << 32` casts to `0usize`, so the bounds
9//! check — however carefully written, saturating or not — is handed a value that
10//! is not the length the file declared, and the input is silently misparsed
11//! instead of rejected.
12//!
13//! The rule these helpers enforce is therefore ORDERING, not arithmetic:
14//! **compare in `u64` space FIRST, convert only once the value is proven to fit
15//! in `usize`.** `available` is widened `usize -> u64` (always lossless), so the
16//! comparison is exact on every target, and the surviving `as usize` cast is
17//! provably lossless because the value is `<= available`, itself a `usize`.
18//!
19//! This is the same guarantee `read_frozen_preamble` and the `#1795` guard
20//! `read_vint_length_prefixed_bytes` already get from capping the raw `u64`
21//! against `MAX_CELL_VALUE_LENGTH` before the cast. These helpers are for the
22//! sites where a byte-count bound — not a cell-value ceiling — is the right
23//! limit: framing fields, and value arms whose only bound is "the bytes that
24//! remain".
25//!
26//! ## The narrowing TARGET is a type parameter, so the guards are TESTABLE
27//!
28//! Every gate component and every CI lane builds a 64-bit target, where
29//! `len_raw as usize` is the IDENTITY for a `u64` — so at `usize` width a
30//! cast-first implementation and the guard AGREE on every input, and a test that
31//! only observes "an error came back" pins nothing (#3042: a test invariant to
32//! the defect it claims to pin). Deleting the guard would leave such a case
33//! green.
34//!
35//! The comparison-and-narrowing step is therefore generic over the
36//! [`LengthWidth`] target. Production code calls the `usize` wrappers below and
37//! its behaviour is unchanged; the tests ALSO instantiate the generic at `u32` —
38//! exactly the semantics a 32-bit `usize` has — where a plain cast and the guard
39//! give DIFFERENT verdicts (`((1u64 << 32) + 8) as u32 == 8` is "8 of 8 bytes
40//! available", i.e. ACCEPTED, while the guard rejects). That regression is then
41//! caught on every lane that runs, with no 32-bit host and no
42//! `cfg(target_pointer_width)` case that would execute nowhere.
43
44use crate::parser::vint_narrow::{narrow_vuint_len, LengthWidth};
45use crate::{Error, Result};

47/// The length as a `usize` if it fits within `available` bytes, else `None`.
48///
49/// The comparison is performed in `u64` space (see the module docs): no cast
50/// happens until the value is known to be representable, so a length that would
51/// truncate on a 32-bit target is rejected rather than silently reinterpreted.
52///
53/// Callers that want the standard "need N bytes for X, only M available"
54/// corruption message should use [`checked_vuint_length`]; this variant exists
55/// for sites whose diagnostic has a different shape (e.g. the range-tombstone
56/// `marker_body_size`, a framing size rather than a cell value).
57pub(super) fn vuint_length_within(len_raw: u64, available: usize) -> Option<usize> {
58    vuint_length_within_as::<usize>(len_raw, available)
59}
60
61/// [`vuint_length_within`] at an arbitrary target width (see the module docs).
62///
63/// The comparison happens in `u64` space — `available` is WIDENED, never
64/// `len_raw` narrowed — and the conversion that follows is CHECKED, so neither
65/// step can reinterpret an out-of-range length as its low bits.
66pub(super) fn vuint_length_within_as<T: LengthWidth>(len_raw: u64, available: T) -> Option<T> {
67    if len_raw > available.widen() {
68        return None;
69    }
70    // Provably `Some`: `len_raw <= available`, itself a `T`. Still a CHECKED
71    // conversion rather than a cast, so a future edit to the comparison above
72    // cannot turn this into a truncation.
73    narrow_vuint_len(len_raw)
74}

76/// [`vuint_length_within`] plus the canonical `Error::corruption` diagnostic.
77///
78/// The message is `"<subject> '<name>': need <len_raw> bytes for <what>, only
79/// <available> available"`, reporting the RAW `u64` length so an adversarial
80/// prefix is named in full rather than in its truncated form.
81pub(super) fn checked_vuint_length(
82    len_raw: u64,
83    available: usize,
84    subject: &str,
85    name: &str,
86    what: &str,
87) -> Result<usize> {
88    checked_vuint_length_as::<usize>(len_raw, available, subject, name, what)
89}
90
91/// [`checked_vuint_length`] at an arbitrary target width (see the module docs).
92pub(super) fn checked_vuint_length_as<T: LengthWidth>(
93    len_raw: u64,
94    available: T,
95    subject: &str,
96    name: &str,
97    what: &str,
98) -> Result<T> {
99    vuint_length_within_as(len_raw, available).ok_or_else(|| {
100        Error::corruption(format!(
101            "{} '{}': need {} bytes for {}, only {} available",
102            subject, name, len_raw, what, available
103        ))
104    })
105}

107/// A FIXED-WIDTH field's declared length, validated in `u64` space against the
108/// sizes the on-disk format allows, and narrowed only on success.
109///
110/// # Why an equality check AFTER the cast is not a guard (issue #3848)
111///
112/// The historical shape at these sites was
113///
114/// ```text
115/// let date_len = date_len as usize;      // raw u64 -> usize
116/// if date_len != 4 { return Err(...) }   // "so a truncated value is caught"
117/// ```
118///
119/// That reasoning is WRONG. Truncation is not a randomising operation: an
120/// adversary chooses the length so its LOW BITS are the expected size. On a
121/// 32-bit target `((1u64 << 32) + 4) as usize == 4usize`, which PASSES
122/// `!= 4` — so a corrupt length is silently ACCEPTED as a valid 4-byte field
123/// instead of being rejected, and the following bytes are misparsed. The
124/// equality check provides no protection against truncation whatsoever; it only
125/// looks as if it does.
126///
127/// So the comparison happens in `u64` space. `allowed` holds `usize` sizes,
128/// widened `usize -> u64` (always lossless), which makes the equality exact on
129/// every target; the surviving `as usize` cast is provably lossless because the
130/// value equals one of the `allowed` sizes, each itself a `usize`.
131fn vuint_length_exact_as<T: LengthWidth>(len_raw: u64, allowed: &[T]) -> Option<T> {
132    // Returns the MATCHED `allowed` entry, so nothing is narrowed at all: the
133    // value handed back is one the caller supplied, already at the target width.
134    allowed.iter().copied().find(|size| len_raw == size.widen())
135}

137/// [`vuint_length_exact_as`] plus the canonical `Error::corruption` diagnostic.
138///
139/// The message keeps the two shapes the fixed-width arms historically used, and
140/// reports the RAW `u64` length so an adversarial prefix is named in full rather
141/// than in its truncated form:
142/// - one allowed size: `"<subject> '<name>': expected <what> length 4, got 17"`
143/// - several:          `"<subject> '<name>': invalid <what> length 17, expected 4 or 16"`
144pub(super) fn checked_vuint_exact_length(
145    len_raw: u64,
146    allowed: &[usize],
147    subject: &str,
148    name: &str,
149    what: &str,
150) -> Result<usize> {
151    checked_vuint_exact_length_as::<usize>(len_raw, allowed, subject, name, what)
152}
153
154/// [`checked_vuint_exact_length`] at an arbitrary target width (see the module
155/// docs): the width the tests instantiate to make the truncation guard
156/// discriminating on a 64-bit host.
157pub(super) fn checked_vuint_exact_length_as<T: LengthWidth>(
158    len_raw: u64,
159    allowed: &[T],
160    subject: &str,
161    name: &str,
162    what: &str,
163) -> Result<T> {
164    vuint_length_exact_as(len_raw, allowed).ok_or_else(|| {
165        let sizes = allowed
166            .iter()
167            .map(|size| size.to_string())
168            .collect::<Vec<_>>()
169            .join(" or ");
170        if allowed.len() == 1 {
171            Error::corruption(format!(
172                "{} '{}': expected {} length {}, got {}",
173                subject, name, what, sizes, len_raw
174            ))
175        } else {
176            Error::corruption(format!(
177                "{} '{}': invalid {} length {}, expected {}",
178                subject, name, what, len_raw, sizes
179            ))
180        }
181    })
