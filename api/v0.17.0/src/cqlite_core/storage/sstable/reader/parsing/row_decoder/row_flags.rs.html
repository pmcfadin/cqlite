1//! Cassandra `UnfilteredSerializer` flag bits, split out of `row_decoder/mod.rs`
2//! under the campsite rule (epic #1116) so that module can gain the
3//! `typed_value` declaration without growing past the file-size ratchet.
4//!
5//! Authority: `cassandra-5.0.8:src/java/org/apache/cassandra/db/rows/UnfilteredSerializer.java`
6//! (the `Flags`/`ExtendedFlags` constants at the head of that class).
7//!
8//! Re-exported into `row_decoder`'s namespace, so the sibling modules that read
9//! these through their `use super::*` glob are unaffected by the move.
10
11// Row header flag constants
12pub(super) const ROW_HAS_TIMESTAMP: u8 = 0x04;
13pub(super) const ROW_HAS_TTL: u8 = 0x08;
14pub(super) const ROW_HAS_DELETION: u8 = 0x10;
15pub(super) const ROW_HAS_ALL_COLUMNS: u8 = 0x20;
16/// Issue #221: row contains a complex column with deletion info.
17pub(super) const ROW_HAS_COMPLEX_DELETION: u8 = 0x40;
18pub(super) const ROW_HAS_EXTENDED_FLAGS: u8 = 0x80;
19
20// Unfiltered marker constants (UnfilteredSerializer.java lines 102-109).
21// Issue #229: these markers were being misinterpreted as row data, causing
22// parsing failures.
23/// Signal end of partition — nothing follows this flag byte.
24pub(super) const END_OF_PARTITION: u8 = 0x01;
25/// Range tombstone marker (not a data row).
26pub(super) const IS_MARKER: u8 = 0x02;
27
28// Extended flags (UnfilteredSerializer.java lines 114-122). These live in the
29// SECOND byte, when ROW_HAS_EXTENDED_FLAGS (0x80) is set.
30/// Static row — has NO clustering prefix.
31pub(super) const EXTENDED_IS_STATIC: u8 = 0x01;
32
33// NOTE: the V5CompressedLegacy format has NO trailing field after row data. The
34// next partition/row starts immediately after `row_size` bytes. (A former
35// ROW_TRAILING_FIELD_SIZE constant was removed as part of the Issue #237 fix.)
