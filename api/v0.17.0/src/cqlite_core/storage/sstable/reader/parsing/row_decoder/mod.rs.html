1//! Parser for Cassandra 5.0 V5CompressedLegacy decompressed blocks
2//!
3//! This format uses **u8 length prefixes** (NOT VInt) for partition keys and strings,
4//! and simplified encoding optimized for compression. This differs from the newer
5//! V5_0NewBig and V5_0Bti formats which use pure VInt encoding.
6//!
7//! ## Partition Key Size Constraints
8//!
9//! **Apache Cassandra Specification**: Partition keys can be up to 64KB (65536 bytes).
10//! **V5CompressedLegacy Format Limitation**: Uses u8 for key length field, limiting keys to 255 bytes max.
11//!
12//! This means V5CompressedLegacy format cannot represent partition keys larger than 255 bytes,
13//! even though Cassandra allows keys up to 64KB. Tables with larger keys would use a different format.
14//!
15//! Based on format research in docs/V5_COMPRESSED_LEGACY_FORMAT_SPEC.md
16//!
17//! ## Format Structure
18//!
19//! ```text
20//! Decompressed Block:
21//! ├─ [0x00] Partition flags (u8)
22//! ├─ [0x01] Partition key length (u8, NOT VInt)
23//! ├─ [0x02..] Raw partition key bytes
24//! ├─ [+0] Partition deletion time (i32 big-endian)
25//! ├─ [+4] Unknown 8-byte field
26//! ├─ [+8] Row data begins
27//! │  ├─ Row header (flags, timestamp, row_size)
28//! │  ├─ Cells:
29//! │  │  ├─ Type tag or flags (u8)
30//! │  │  ├─ Column name length (u8)
31//! │  │  ├─ Column name bytes
32//! │  │  ├─ Value length (varies)
33//! │  │  └─ Value bytes
34//! │  └─ Trailing 4-byte field (NOT included in row_size)
35//! └─ [Next partition or end of block]
36//! ```

100/// One on-disk column to decode, in serialization-header order.
101///
102/// `schema` is the resolved supplied-schema `Column` (drives column identity /
103/// emit name) or `None` for a DROPPED / evolved-away column present on disk but
104/// absent from the supplied schema — its bytes are still consumed to keep the
105/// trailing columns byte-aligned (issue #1080 Part 2), but no cell is emitted.
106///
107/// `header_type` is the AUTHORITATIVE on-disk SerializationHeader marshal type
108/// (`ColumnInfo.column_type`), the no-heuristics source of truth (issue #28) used
109/// to decide complex-ness and to decode complex values. It is `None` only on the
110/// header-empty fallback path (synthetic SSTables) where the supplied schema type
111/// is all we have.
112///
113/// Both fields borrow: `schema` from the per-call `TableSchema`, `header_type`
114/// from `reader.header`. The owning [`RowColumnResolution`] therefore borrows
115/// from both for `'a`.
116pub(super) struct ColumnToParse<'a> {
117    pub(super) schema: Option<&'a crate::schema::Column>,
118    pub(super) header_type: Option<&'a str>,
119    /// The column's emit name, interned ONCE per block as a shared `Arc<str>`
120    /// (issue #1334). Populating a decoded cell with its name is then an
121    /// `Arc::clone` refcount bump instead of a per-cell, per-row heap `String`
122    /// allocation. For an emitted column this is the supplied-schema column
123    /// name; for a DROPPED column (never emitted) it mirrors the on-disk header
124    /// name and is unused.
125    pub(super) name: Arc<str>,
126    /// Precomputed value-decode dispatch (Epic J / issue #1635), resolved ONCE per
127    /// block from the AUTHORITATIVE declared type (the supplied-schema type when
128    /// matched, else the on-disk header marshal type for a DROPPED column — exactly
129    /// the type `parse_cell_value_schema_order` decoded against per cell before J1).
130    /// The per-cell decode `match`es on this instead of re-lowercasing + string-
131    /// matching every cell (no per-cell `to_lowercase`, no per-cell allocation).
132    pub(super) kind: CellKind,
133    /// Precomputed complex-ness (Epic J / issue #1635): `true` for a non-frozen
134    /// collection / multicell UDT (routed to `parse_complex_column`), `false` for a
135    /// single-cell scalar / frozen value. Resolved ONCE per block via
136    /// `is_complex_column` on the authoritative complex-ness type (on-disk header
137    /// marshal type preferred, supplied-schema type on the header-empty fallback) —
138    /// the row body reads this instead of calling `is_complex_column` per row.
139    pub(super) is_complex: bool,
140}

142/// Pre-resolved header→schema column ordering for a whole SSTable block.
143///
144/// Issue #1046 (the true hoist): the header→schema-column resolution is CONSTANT
145/// for every row in an SSTable — the serialization header is fixed and the
146/// supplied schema does not change between rows. Building it once per block and
147/// reusing it across every partition/row means the per-row decode performs ZERO
148/// schema-lookup allocations (no per-row `HashMap`, no per-row `String` clone, no
149/// per-row `Vec` of columns). The expensive `O(header_cols × schema_cols)` name
150/// resolution + the `Vec` allocations happen ONCE per block, not once per row.
151///
152/// Two orderings are precomputed because the on-disk column group differs by row
153/// kind (Cassandra's `missing_columns_bitmap` covers only the static columns for
154/// a static row and only the regular columns for a regular row — issue #702):
155///   - `regular`: header columns with `is_static == false`
156///   - `static_`: header columns with `is_static == true`
157///
158/// The per-row missing-columns bitmap filter is applied by iterating the chosen
159/// slice and skipping bitmapped-out indices INLINE (no per-row `Vec`), so the
160/// row body still allocates nothing for column resolution.
161///
162/// Lifetime: borrows the header-type strings from `reader.header` and the
163/// `Column`s from the supplied `schema`; both outlive the block's row loops
164/// (resolution is built at the top of each `parse_block_emit*` / per-partition
165/// driver, where `reader` and `schema` are in scope). This is a per-BLOCK hoist —
166/// allocations scale with block count, not row count.
167pub(in crate::storage::sstable::reader) struct RowColumnResolution<'a> {
168    /// On-disk regular (non-static) columns in serialization-header order.
169    regular: Vec<ColumnToParse<'a>>,
170    /// On-disk static columns in serialization-header order.
171    static_: Vec<ColumnToParse<'a>>,
172    /// Clustering-key names interned once per block as shared `Arc<str>` handles
173    /// (issue #1334), positionally aligned with `schema.clustering_keys`. Reused
174    /// so the per-row clustering-key cell insert is an `Arc::clone`, not a
175    /// `String` allocation.
176    clustering: Vec<Arc<str>>,
177}
178
179impl<'a> RowColumnResolution<'a> {
180    /// Build the resolution ONCE for a block from the on-disk serialization
181    /// header (`reader.header`) and the supplied `schema`.
182    ///
183    /// Resolution is an exact column-name match, keep-FIRST on a duplicate name
184    /// (Cassandra schemas cannot have duplicate column names, so this is purely
185    /// defensive and matches the prior per-row `HashMap`/`iter().find()`
186    /// semantics). On the header-empty fallback path (synthetic SSTables) the
187    /// supplied schema order is used directly, every column schema-present by
188    /// construction.
189    pub(in crate::storage::sstable::reader) fn build(
190        schema: &'a TableSchema,
191        reader: &'a crate::storage::sstable::reader::types::SSTableReader,
192    ) -> Self {
193        if !reader.header.columns.is_empty() {
194            // O(header_cols × schema_cols) name resolution, but ONCE per block via
195            // a borrowed-key map (keys are `&str` slices into `schema.columns[].name`,
196            // no `String` clones; values are `&Column`). Both borrow for `'a`.
197            let mut resolve_lookup: HashMap<&'a str, &'a crate::schema::Column> =
198                HashMap::with_capacity(schema.columns.len());
199            for col in &schema.columns {
200                resolve_lookup.entry(col.name.as_str()).or_insert(col);
201            }
202
203            let build_for = |want_static: bool| -> Vec<ColumnToParse<'a>> {
204                reader
205                    .header
206                    .columns
207                    .iter()
208                    .filter(|col_info| {
209                        !col_info.is_primary_key
210                            && !col_info.is_clustering
211                            && col_info.is_static == want_static
212                    })
213                    .map(|col_info| {
214                        let schema = resolve_lookup.get(col_info.name.as_str()).copied();
215                        // Intern the emit name ONCE per block (issue #1334): the
216                        // supplied-schema column name when matched, else the
217                        // on-disk header name for a DROPPED column (never emitted).
218                        let name: Arc<str> = Arc::from(
219                            schema
220                                .map(|c| c.name.as_str())
221                                .unwrap_or(col_info.name.as_str()),
222                        );
223                        // J1 (issue #1635): resolve dispatch ONCE per column here.
224                        // `value_type` drives the scalar decode dispatch — the
225                        // supplied-schema type when matched, else the on-disk header
226                        // marshal type for a DROPPED column (exactly the type the
227                        // per-cell path decoded against). `is_complex` uses the
228                        // header marshal type (authoritative complex-ness source,
229                        // carries `UserType(...)`).
230                        let value_type = schema
231                            .map(|c| c.data_type.as_str())
232                            .unwrap_or(col_info.column_type.as_str());
233                        ColumnToParse {
234                            schema,
235                            header_type: Some(col_info.column_type.as_str()),
236                            name,
237                            kind: CellKind::from_type(value_type),
238                            is_complex: V5CompressedLegacyParser::is_complex_column(
239                                col_info.column_type.as_str(),
240                            ),
241                        }
242                    })
243                    .collect()
244            };
245
246            RowColumnResolution {
247                regular: build_for(false),
248                static_: build_for(true),
249                clustering: schema
250                    .clustering_keys
251                    .iter()
252                    .map(|k| Arc::from(k.name.as_str()))
253                    .collect(),
254            }
255        } else {
256            // Fallback to schema order when header is empty (shouldn't happen for
257            // real SSTables). Filter out partition/clustering keys (regular columns
258            // only carry cell data) and split by row kind.
259            tracing::warn!("V5CompressedLegacy: reader.header.columns is empty, falling back to schema order (may cause column misalignment)");
260            let partition_key_names: std::collections::HashSet<&str> = schema
261                .partition_keys
262                .iter()
263                .map(|k| k.name.as_str())
264                .collect();
265            let clustering_key_names: std::collections::HashSet<&str> = schema
266                .clustering_keys
267                .iter()
268                .map(|k| k.name.as_str())
269                .collect();
270            let build_for = |want_static: bool| -> Vec<ColumnToParse<'a>> {
271                schema
272                    .columns
273                    .iter()
274                    .filter(|col| {
275                        !partition_key_names.contains(col.name.as_str())
276                            && !clustering_key_names.contains(col.name.as_str())
277                            && col.is_static == want_static
278                    })
279                    .map(|col| ColumnToParse {
280                        schema: Some(col),
281                        header_type: None,
282                        name: Arc::from(col.name.as_str()),
283                        // J1 (issue #1635): header-empty fallback — the supplied
284                        // schema type is the only authoritative source for both the
285                        // scalar dispatch and complex-ness (matches the pre-J1
286                        // `complex_type = header_type.unwrap_or(&column.data_type)`).
287                        kind: CellKind::from_type(&col.data_type),
288                        is_complex: V5CompressedLegacyParser::is_complex_column(&col.data_type),
289                    })
290                    .collect()
291            };
292            RowColumnResolution {
293                regular: build_for(false),
294                static_: build_for(true),
295                clustering: schema
296                    .clustering_keys
297                    .iter()
298                    .map(|k| Arc::from(k.name.as_str()))
299                    .collect(),
300            }
301        }
302    }
303
304    /// The interned clustering-key name at position `i` (issue #1334), or `None`
305    /// when out of range. Cloning the returned handle into the cells map is an
306    /// `Arc::clone`, not a `String` allocation.
307    pub(super) fn clustering_name(&self, i: usize) -> Option<&Arc<str>> {
308        self.clustering.get(i)
309    }
310
311    /// Number of interned clustering-key name handles (issue #1642). Used to
312    /// size the per-row cell `Vec` capacity hint: clustering-key cells are
313    /// pushed FIRST, before the data columns, so a clustered table would
314    /// otherwise reallocate once past the data-column-only hint.
315    pub(super) fn clustering_len(&self) -> usize {
316        self.clustering.len()
317    }
318
319    /// The pre-resolved on-disk column ordering for a row of the given kind.
320    pub(super) fn columns_for(&self, is_static: bool) -> &[ColumnToParse<'a>] {
321        if is_static {
322            &self.static_
323        } else {
324            &self.regular
325        }
326    }
327}

347/// Outcome of [`V5CompressedLegacyParser::parse_one_partition_with_timestamps`].
348///
349/// The sliding-window compaction-read driver (issue #827) feeds the parser a
350/// `data` slice that grows as decompressed chunks are appended and shrinks as
351/// confirmed partitions are drained. Because a single partition can straddle a
352/// compression-chunk boundary, the bounded parser must distinguish "I parsed a
353/// complete partition" from "I ran out of buffer mid-partition and need more
354/// bytes" — conflating the two would silently drop trailing partitions.
355#[derive(Debug, PartialEq, Eq)]
356pub enum ParseStep {
357    /// A full partition was parsed; `usize` bytes of `data` were consumed and
358    /// can be drained from the front of the sliding window.
359    Emitted(usize),
360    /// `data` appears truncated mid-partition. The caller should append the
361    /// next decompressed chunk and retry. Only returned when `!at_final_chunk`.
362    NeedMore,
363    /// Genuine end of partitions in `data` (no more bytes to consume).
364    Done,
365}

367// Struct-size regression guard (issue #1616, Epic H/H3; see
368// docs/reports/parser-performance-audit-2026-07-01.md §Epic H (finding H3)). `ParseStep`
369// is returned once per partition by the bounded compaction-read driver on the
370// scan hot path. Measured 16 bytes today (discriminant + inlined `usize`) on
371// 64-bit targets. Update this pin DELIBERATELY, never silently: any change —
372// growth or shrink — must be a reviewed edit here.
373#[cfg(target_pointer_width = "64")]
374const _: () = assert!(std::mem::size_of::<ParseStep>() == 16);

376/// Row header data extracted from V5CompressedLegacy row
377#[derive(Debug, Clone)]
378struct RowHeader {
379    /// Row-level timestamp (after delta decoding from min_timestamp)
380    timestamp: Option<i64>,
381    /// Row-level TTL (after delta decoding from min_ttl)
382    ttl: Option<i32>,
383    /// Liveness local deletion time in SECONDS: absolute epoch-second expiry for a
384    /// TTL-bearing INSERT row (the `pk_liveness.ttl()` clock, from `HAS_TTL`).
385    /// This is the `expires_at` for the row liveness — distinct from
386    /// `local_deletion_time` which is the GC-grace clock for row tombstones.
387    /// `None` when `HAS_TTL` was not set.
388    ///
389    /// Stored as `i64` (not `i32`): on oa/da (`hasUIntDeletionTime`) a post-2038 expiry
390    /// is decoded as an UNSIGNED 32-bit value in `[2^31, 2^32)`; keeping it `i64`
391    /// preserves the large positive second count so the read-time TTL filter does not
392    /// wrap it negative and hide a still-live row (#1741 F1). See the decode site in
393    /// `row_framing.rs` for the reinterpretation.
394    liveness_expires_at_seconds: Option<i64>,
395    /// Row-level local deletion time in SECONDS (after delta decoding from
396    /// min_local_deletion_time). This is the GC-grace clock, NOT the reconciliation
397    /// timestamp; do not use it for last-write-wins comparisons.
398    local_deletion_time: Option<i32>,
399    /// Row-level `markedForDeleteAt` reconciliation timestamp in MICROSECONDS
400    /// (absolute = min_timestamp + delta). For a row tombstone this is the
401    /// authoritative deletion timestamp used by the compaction merger for
402    /// timestamp-based last-write-wins shadowing (Issue #505). `None` when the
403    /// row has no HAS_DELETION flag.
404    marked_for_delete_at: Option<i64>,
405    /// Number of bytes consumed by the header
406    header_size: usize,
407    /// Length of the row_size VInt in bytes (needed for offset calculation)
408    /// row_size is measured from AFTER this VInt is consumed
409    row_size_vint_len: usize,
410    /// Bitmask of missing columns from Cassandra's Columns.Serializer format.
411    /// bit=1 means column missing, bit=0 means column present.
412    /// None when HAS_ALL_COLUMNS flag is set.
413    missing_columns_bitmap: Option<u64>,
414    /// Issue #1741 read-side shadowing aggregate: max write timestamp (µs) across
415    /// this row's decoded DATA cells (excludes partition/clustering pseudo-cells;
416    /// a cell inheriting the row timestamp contributes the row liveness timestamp,
417    /// and a non-frozen collection contributes its newest element ts). Includes
418    /// shadow/TTL-dropped cells so a fully-reduced row is still recognised as
419    /// shadowed, but NEVER a tombstone cell (#3094 — that rides as the timestamp-less
420    /// `has_deleted_data_cell`). `None` when the row decoded no LIVE data cell.
421    /// Combined with `timestamp` to decide whether a tombstone shadows the WHOLE row.
422    max_data_cell_timestamp: Option<i64>,
423    /// Issue #1741: max effective expiry (epoch seconds) across this row's expiring
424    /// DATA cells (explicit per-cell TTL or a `USE_ROW_TTL` cell inheriting the
425    /// row's expiry; includes expired cells). `None` when none is expiring.
426    max_data_cell_expires_at: Option<i64>,
427    /// Issue #1741: `true` when at least one decoded DATA cell is LIVE (not
428    /// shadowed, not expired) and live-forever (no TTL). Keeps the row visible
429    /// regardless of liveness expiry. A shadowed/expired cell never sets this.
430    has_live_forever_data_cell: bool,
431    /// Issue #3094: `true` when the row decoded at least one TOMBSTONE cell — a
432    /// PRESENCE fact carrying NO timestamp (one could only RAISE the row maximum,
433    /// i.e. UN-hide a row): see [`PartitionShadow::has_shadow_evidence`].
434    has_deleted_data_cell: bool,
435}
436
437impl RowHeader {
438    /// Whether this row header describes a row tombstone (HAS_DELETION was set).
439    ///
440    /// Detected via `local_deletion_time` being present, which is only set when the
441    /// HAS_DELETION (0x10) flag was decoded (Issue #505).
442    fn is_row_tombstone(&self) -> bool {
443        self.local_deletion_time.is_some()
444    }
445
446    /// Build the `Value::Tombstone(RowTombstone)` for this header.
447    ///
448    /// The reconciliation `deletion_time` is `marked_for_delete_at` (the
449    /// `markedForDeleteAt` field, MICROSECONDS — same clock as HAS_TIMESTAMP), NOT
450    /// the row write `timestamp` and NOT `local_deletion_time` (seconds, GC clock).
451    /// For a pure row tombstone HAS_TIMESTAMP is absent, so using the write
452    /// timestamp would yield epoch 0 and lose every merge comparison (Issue #505).
453    ///
454    /// Falls back to `local_deletion_time` promoted to microseconds only if
455    /// `marked_for_delete_at` is somehow absent while a deletion was recorded
456    /// (should not happen given HAS_DELETION always writes both VInts), keeping the
457    /// merger's LWW ordering meaningful rather than defaulting to 0.
458    fn row_tombstone(&self) -> Value {
459        let deletion_time = self.row_tombstone_deletion_time();
460        Value::Tombstone(Box::new(TombstoneInfo {
461            deletion_time,
462            tombstone_type: TombstoneType::RowTombstone,
463            // Carry the on-disk `localDeletionTime` (GC clock, seconds) so the
464            // compaction merge→rewrite path can preserve it (#873). Absent for a
465            // non-tombstone header, hence the `0` fallback.
466            local_deletion_time: self.local_deletion_time.unwrap_or(0) as i64,
467            ttl: None,
468            range_start: None,
469            range_end: None,
470        }))
471    }
472
473    /// Authoritative reconciliation timestamp (microseconds) for a row tombstone.
474    fn row_tombstone_deletion_time(&self) -> i64 {
475        match self.marked_for_delete_at {
476            Some(ts) => ts,
477            // Defensive fallback: promote the seconds-based local deletion time to
478            // microseconds so ordering remains non-zero and monotonic.
479            None => self
480                .local_deletion_time
481                .map(|s| (s as i64).saturating_mul(1_000_000))
482                .unwrap_or(0),
483        }
484    }
485
486    /// Issue #2374/#2789: the primary-key (row-marker) liveness of this row for
487    /// the READ path. `has_marker` = the row carried `HAS_TIMESTAMP` (an INSERT
488    /// liveness marker, not a data-cell-only UPDATE); `expires_at_seconds` = the
489    /// marker's TTL expiry when `HAS_TTL` was set, else `None` (live-forever).
490    /// Carry-only for reads; the compaction write path never consults it.
491    pub(crate) fn row_liveness(
492        &self,
493    ) -> crate::storage::sstable::reader::compaction_row::RowLiveness {
494        crate::storage::sstable::reader::compaction_row::RowLiveness {
495            has_marker: self.timestamp.is_some(),
496            expires_at_seconds: self.liveness_expires_at_seconds,
497            // Issue #2374/#2789: the authoritative marker write timestamp (µs)
498            // from the row header — the last-write-wins key the cross-generation
499            // fold uses. Never inferred (no-heuristics, #28).
500            marker_timestamp: self.timestamp,
501        }
502    }
503
504    /// Issue #1741: max LIVE write ts (µs) — liveness marker vs decoded live cells.
505    fn max_write_timestamp(&self) -> i64 {
506        let m = self.timestamp.unwrap_or(i64::MIN);
507        self.max_data_cell_timestamp.map_or(m, |c| m.max(c))
508    }
509
510    /// Issue #1741: `true` when a deletion at `deleted_at_micros` shadows the WHOLE
511    /// row — i.e. every piece of the row's data is older than (or equal to) the
512    /// deletion. Follows Cassandra `DeletionTime.deletes(ts) = ts <= markedForDeleteAt`.
513    ///
514    /// Fail-safe (#28): a row with NO authoritative timestamp (the `i64::MIN`
515    /// sentinel — no liveness, no decoded live cell) is NOT shadowed. #3094: a decoded
516    /// cell TOMBSTONE defeats that fail-safe WITHOUT contributing a timestamp. Both
517    /// rules live in [`PartitionShadow::has_shadow_evidence`].
518    fn shadowed_by_deletion_at(&self, deleted_at_micros: i64) -> bool {
519        let max_ts = self.max_write_timestamp();
520        PartitionShadow::has_shadow_evidence(max_ts, self.has_deleted_data_cell)
521            && max_ts <= deleted_at_micros
522    }
523
524    /// Issue #1741: read-time TTL expiry. `true` iff the row carries a TTL somewhere
525    /// AND every piece of its data (primary-key liveness + cells) has expired at
526    /// `now_secs`, matching what a Cassandra `SELECT` hides. Returns `false` for any
527    /// row with no TTL at all (never touches the tombstone-free common case).
528    fn row_liveness_expired(&self, now_secs: i64) -> bool {
529        let has_ttl =
530            self.liveness_expires_at_seconds.is_some() || self.max_data_cell_expires_at.is_some();
531        if !has_ttl {
532            return false;
533        }
534        // A live-forever data cell (written with no TTL) keeps the row visible.
535        if self.has_live_forever_data_cell {
536            return false;
537        }
538        // Primary-key liveness still live? Present iff HAS_TIMESTAMP; live-forever
539        // when it carries no TTL expiry, otherwise live until its expiry passes.
540        let liveness_live = self.timestamp.is_some()
541            && self
542                .liveness_expires_at_seconds
543                .is_none_or(|s| s > now_secs);
544        if liveness_live {
545            return false;
546        }
547        // Any expiring data cell still live keeps the row visible.
548        if self.max_data_cell_expires_at.is_some_and(|e| e > now_secs) {
549            return false;
550        }
551        true
552    }
553}

601/// Issue #1741 (per-element filtering): read-side per-element shadow/TTL filter
602/// context threaded into the complex-column element loop. `Some` ONLY on the
603/// user-facing SELECT read path (a covering partition/range deletion and/or the
604/// read clock is active); `None` for every physical consumer (compaction /
605/// delta-scan / unit tests), which stay byte-unchanged (the filter never drops an
606/// element when it is `None`).
607#[derive(Clone, Copy)]
608pub(crate) struct ElementShadow {
609    /// Covering deletion `markedForDeleteAt` (µs) for the row this collection
610    /// belongs to (the partition tombstone folded with the open range tombstone
611    /// when the row falls inside it), or `None` when nothing covers the row.
612    pub cover: Option<i64>,
613    /// Read clock (epoch seconds) for per-element TTL expiry.
614    pub now: i64,
615    /// Row-liveness write timestamp (µs) inherited by `USE_ROW_TIMESTAMP` elements
616    /// (flag 0x08 — the element carries no own timestamp); `None` when the row
617    /// carries no liveness marker, in which case such an element has no
618    /// authoritative write ts and is NEVER shadowed (no-heuristics, issue #28).
619    pub row_ts: Option<i64>,
620    /// Row-liveness expiry (epoch seconds) inherited by `USE_ROW_TTL` elements
621    /// (flag 0x10 — an expiring element that carries NO explicit per-element
622    /// `localDeletionTime`). This is the SAME effective row expiry the scalar
623    /// `USE_ROW_TTL` cell path computes (`liveness_expires_at_seconds`, falling
624    /// back to the row `local_deletion_time`), so scalar and collection-element
625    /// inherited-TTL semantics stay identical (issue #1741). `None` when the row

627    /// authoritative expiry and is never TTL-expired here (no-heuristics, issue #28).
628    pub row_expires_at: Option<i64>,
629    /// Issue #2038 (round 3): row-liveness TTL in SECONDS — `row_header.ttl`,
630    /// paired with `row_expires_at` above to resolve a `USE_ROW_TTL` element's
631    /// EFFECTIVE `CellExpiration` for the per-cell-metadata `TTL()` value (a
632    /// statement-level `INSERT ... USING TTL n` on a non-frozen collection/UDT
633    /// column). `None` when the row carries no liveness TTL (`HAS_TTL` unset),
634    /// in which case such an element's expiry cannot be resolved here
635    /// (no-heuristics, issue #28) — mirrors the scalar `USE_ROW_TTL` cell
636    /// path's `(row_header.ttl, row_expiry)` pairing (row_data.rs ~line 726).
637    pub row_ttl_seconds: Option<i32>,
638}

640/// Extra metadata produced by `parse_complex_column_inner` for delta-scan callers
641/// (Issue #700, DS4: non-frozen collection v1 semantics).
642///
643/// Returned alongside the `Value` and new offset so the emit path can set the correct
644/// `replaced` flag and `writetime` on the resulting `CellDelta`.
645///
646/// Fields are read by `parse_block_emit_delta` which is `#[cfg(feature = "delta-scan")]`;

650pub(crate) struct ComplexColumnMeta {
651    /// `true` when the collection generation carries a collection-level tombstone
652    /// (`s = {...}` overwrite), meaning the consumer must **replace** rather than
653    /// merge the prior collection state.
654    pub has_collection_tombstone: bool,
655    /// Maximum element writetime seen across all cells in this collection, in µs
656    /// since Unix epoch.  `0` when no element had its own explicit timestamp
657    /// (all inherited the row timestamp).
658    pub max_element_writetime: i64,
659    /// Number of element-level tombstones detected (`is_deleted` flag) in this
660    /// collection cell (Issue #493 territory).  v1 does not represent them; callers
661    /// must count and warn.
662    pub element_tombstone_count: u64,
663    /// Real complex deletion `(markedForDeleteAt µs, localDeletionTime s)` for
664    /// this column, or `None` for the `LIVE` sentinel (epic #899). Used by the
665    /// compaction read path to populate `ComplexColumn.complex_deletion`.
666    /// Always `None` on the user-facing read path (where the field is unused).
667    pub complex_deletion: Option<(i64, i32)>,
668    /// Issue #1741 (Finding 3): read-time TTL aggregate over this collection's
669    /// LIVE (non-deleted) elements. `true` when at least one live element carries
670    /// no TTL of any kind (genuinely live-forever) — such an element keeps the
671    /// row visible regardless of the row-liveness TTL. Computed from authoritative
672    /// per-element cell flags (no heuristics), scalar-only (no extra allocation).
673    pub has_live_forever_element: bool,
674    /// Issue #1741 (Finding 3): max effective expiry (epoch seconds) across this
675    /// collection's live elements that carry an EXPLICIT per-element TTL. `None`
676    /// when no live element has an explicit expiry (elements that inherit the row
677    /// TTL via `USE_ROW_TTL` are governed by the row-liveness expiry instead).
678    /// Folded into the row's `max_data_cell_expires_at` so a collection whose
679    /// elements are all expired does not keep an otherwise-expired row alive.
680    pub max_element_expires_at: Option<i64>,
681    /// Issue #2038 (roborev Medium finding): the `CellExpiration` surfaced in
682    /// per-cell metadata for `TTL(non_frozen_collection/UDT)`, IFF every VISIBLE
683    /// (post shadow/TTL-filter, non-tombstone) element shares the IDENTICAL
684    /// explicit `(ttl_seconds, expires_at_seconds)` pair. Decoded from the
685    /// authoritative per-element cell fields (no heuristics, #28) via the
686    /// `ExpiryHomogeneity` tracker.
687    ///
688    /// Deliberately NARROWER than `max_element_expires_at` above: a dropped
689    /// (shadow/TTL-filtered) element never contributes here (it does for
690    /// `max_element_expires_at`, which drives the orthogonal #1741 row-hidden
691    /// decision), and a MIXED collection (elements with different TTLs, or a
692    /// live-forever element mixed with an expiring one) has no single TTL that
693    /// describes it — this is `None` in that case, correctness over
694    /// over-approximating with one element's expiry. This is the complex-cell
695    /// analogue of the scalar #1743 fix, which surfaces a single-cell expiry
696    /// unambiguously.
697    pub visible_uniform_expiration: Option<CellExpiration>,
698    /// Issue #1741 (per-element filtering): number of LIVE (non-tombstone)
699    /// elements DROPPED from the emitted container by the read-side per-element
700    /// shadow/TTL filter — i.e. elements shadowed by the covering deletion (own
701    /// write ts `<= cover`) or TTL-expired at the read clock. Always `0` for every
702    /// physical consumer (the filter is `None`), so their output is byte-unchanged.
703    /// The read call site uses `> 0` (together with an emptied container) to tell a
704    /// collection that the filter reduced to nothing (read as absent/null) apart
705    /// from one that was genuinely empty / all element-tombstones (kept as-is).
706    pub shadow_filtered_element_count: usize,
707}
