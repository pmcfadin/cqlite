1//! Construction and small self-describing predicates of
2//! [`V5CompressedLegacyParser`] — the constructor, the builder-style
3//! configuration setters, and the two `Statistics.db`/version-gate predicates
4//! that describe the sstable rather than decode it.
5//!
6//! Split out of `row_decoder/mod.rs` under the campsite rule (epic #1116); the
7//! `impl` block is unchanged, so every method keeps its path and visibility.
8
9use super::*;
10
11impl V5CompressedLegacyParser {
12    /// Create a new V5CompressedLegacy parser
13    ///
14    /// # Arguments
15    /// * `keyspace` - Keyspace name
16    /// * `table_name` - Table name
17    /// * `min_timestamp` - Minimum timestamp for delta decoding (from Statistics.db)
18    /// * `min_local_deletion_time` - Minimum local deletion time for delta decoding (from Statistics.db)
19    /// * `min_ttl` - Minimum TTL for delta decoding (from Statistics.db)
20    pub fn new(
21        keyspace: String,
22        table_name: String,
23        min_timestamp: i64,
24        min_local_deletion_time: i64,
25        min_ttl: Option<i64>,
26    ) -> Self {
27        // Default to nb-compatible BIG gates when not supplied by the caller.
28        // Use the infallible nb_fallback() constructor (no expect/unwrap in lib code).
29        let version_gates = std::sync::Arc::new(VersionGates::Big(BigVersionGates::nb_fallback()));
30        Self {
31            keyspace,
32            table_name,
33            min_timestamp,
34            min_local_deletion_time,
35            min_ttl,
36            udt_registry: None,
37            version_gates,
38            read_shadowing: false,
39            // Issue #1741 (F2): sample the read clock ONCE per parser (== once per
40            // read/scan operation); every block/partition below reuses this value.
41            now_secs: now_epoch_secs(),
42        }
43    }
44
45    /// Issue #1741: enable read-side SELECT-semantic shadowing on this parser. Call
46    /// with `true` ONLY when building the parser for a user-facing query read; leave
47    /// the default (`false`) for physical/verification/compaction/delta reads.
48    pub fn with_read_shadowing(mut self, on: bool) -> Self {
49        self.read_shadowing = on;
50        self
51    }
52
53    /// Set the version gates for version-sensitive parsing decisions (VG1 plumbing).
54    ///
55    /// Call this after `new()` with the `Arc<VersionGates>` from `SSTableReader`.
56    /// Until VG3 lands, passing gates here has no effect on parsing behaviour —
57    /// the gate values are stored for future use only.
58    pub fn with_version_gates(mut self, gates: std::sync::Arc<VersionGates>) -> Self {
59        self.version_gates = gates;
60        self
61    }
62
63    /// Set the UDT registry for resolving short UDT type names in frozen collections (Issue #238)
64    pub fn with_udt_registry(mut self, registry: UdtRegistry) -> Self {
65        self.udt_registry = Some(registry);
66        self
67    }
68
69    /// Return `true` when the version gates indicate `hasUIntDeletionTime` (oa / da).
70    ///
71    /// Authority: BigFormat.java:409 — `hasUintDeletionTime = version.compareTo("oa") >= 0`
72    #[inline]
73    pub(super) fn has_uint_deletion_time(&self) -> bool {
74        match self.version_gates.as_ref() {
75            VersionGates::Big(g) => g.has_uint_deletion_time,
76            VersionGates::Bti(g) => g.has_uint_deletion_time,
77        }
78    }
79
80    /// Issue #1741 (Finding 2): `true` when this SSTable's authoritative
81    /// EncodingStats prove it carries NO deletions of any kind — hence NO range
82    /// tombstones. A clustering-slice read can then keep the O(slice) row-index
83    /// fast-forward and skip prefix priming entirely (a range tombstone opening
84    /// before the slice is impossible).
85    ///
86    /// `min_local_deletion_time` is `EncodingStats.minLocalDeletionTime`, the MIN
87    /// of every cell's `localDeletionTime`. A live cell contributes the LIVE
88    /// sentinel `Cell.NO_DELETION_TIME == Integer.MAX_VALUE`; a partition/row/
89    /// range/cell tombstone OR an expiring cell contributes a smaller value. So
90    /// the min equals `Integer.MAX_VALUE` iff the SSTable has no deletion and no
91    /// TTL. This OVER-approximates range-tombstone presence (a cell tombstone or
92    /// TTL also trips it), which is safe: priming then runs and stays correct. No
93    /// stats (`min == 0` from the `build_v5_parser` fallback) conservatively primes.
94    /// No heuristics — authoritative metadata only (issue #28).
95    #[inline]
96    pub(super) fn sstable_may_have_range_tombstones(&self) -> bool {
97        // Integer.MAX_VALUE — Cassandra `Cell.NO_DELETION_TIME` LIVE sentinel.
98        const NO_DELETION_TIME: i64 = i32::MAX as i64;
99        self.min_local_deletion_time != NO_DELETION_TIME
100    }

102    /// Whether the bytes at `offset` begin a new partition header, WITHOUT
103    /// consuming them.
104    ///
105    /// This is the NO-HEURISTICS approach: we validate the actual structure
106    /// instead of guessing from byte patterns. Issue #1641 (K2) made it
107    /// non-allocating on the fast-reject paths — it delegates to
108    /// [`peek_partition_boundary`], which shares the structural walk of
109    /// `parse_partition_header_full` (via `scan_partition_header`) but always
110    /// skips the success-path key `to_vec` and the `PARTITION_HEADER_TRY_PARSES`
111    /// counter. The marker pre-check and readiness gate allocate nothing; the
112    /// strict scan on a `Ready` buffer may still build a discarded error string
113    /// on a structural mismatch. The boolean result is identical to the former
114    /// allocating implementation (marker pre-check + full-parse `is_ok`), proved
115    /// by the `peek_matches_full_parse` proptest.
116    ///
117    /// # Arguments
118    /// * `data` - Binary data buffer
119    /// * `offset` - Offset to check
120    ///
121    /// # Returns
122    /// * `true` if a valid partition header can be parsed at this offset
123    /// * `false` if parsing fails (likely a row header or invalid data)
124    ///
125    /// # Visibility
126    /// Exposed for integration testing to validate partition boundary detection
127    ///
128    /// [`peek_partition_boundary`]: Self::peek_partition_boundary
129    #[doc(hidden)]
130    pub fn peek_is_partition_header(&self, data: &[u8], offset: usize) -> bool {
131        matches!(
132            self.peek_partition_boundary(data, offset),
133            BoundaryPeek::Header
134        )
135    }
