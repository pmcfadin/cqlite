1//! Row DISPLAY + static-merge helpers for the `V5CompressedLegacy` row decoder
2//! (split out of `mod.rs` per the campsite rule, epic #1116).
3//!
4//! Three decisions live here, each shared by every decode site so the rule exists
5//! in exactly ONE place:
6//! * `row_has_non_key_cell` — is there any non-primary-key data cell (issue #932);
7//! * `merge_static_cells` — positional, clustering-row-wins static injection (#1642);
8//! * `build_display_row` — the row-tombstone display rule (#505/#932);
9//! * `extract_clustering_values` — clustering identity for range-tombstone coverage.
10
11use super::*;

13/// Issue #932: does the decoded cell map hold any NON-primary-key data cell?
14///
15/// Primary-key (partition + clustering) columns are surfaced into the cell map
16/// as pseudo-cells (#229) so the read-back path can recover the clustering
17/// identity; they are NOT row data. A row carrying `HAS_DELETION` is a PURE row
18/// tombstone only when no such data cell survives — otherwise the row deletion
19/// COEXISTS with surviving (strictly-newer) cells and the row displays as live.
20pub(super) fn row_has_non_key_cell(cells: &[(Arc<str>, Value)], schema: &TableSchema) -> bool {
21    cells.iter().any(|(name, _)| {
22        let name: &str = name;
23        !schema.partition_keys.iter().any(|k| k.name == name)
24            && !schema.clustering_keys.iter().any(|c| c.name == name)
25    })
26}

28/// Issue #1642 (K3): append accumulated static-column cells onto a clustering
29/// row's positional cell vector. This is an unconditional `extend` (O(n_static))
30/// — NOT a per-cell membership scan — because a static column name can NEVER
31/// collide with a name already in the clustering row's cells, so there is no
32/// clustering-row-wins conflict to resolve.
33///
34/// Disjointness proof (this codebase). `static_cells` is the cell vector of an
35/// IS_STATIC row; its names are exactly `RowColumnResolution::columns_for(true)`
36/// — header columns with `is_static == true` AND `!is_primary_key` AND
37/// `!is_clustering`. A static row has no clustering prefix, so it receives zero
38/// clustering-key pseudo-cells (row_data.rs: `clustering_values` is empty when
39/// static). A clustering row's `cells` names are the clustering-key pseudo-cells
40/// (issue #229) PLUS `columns_for(false)` — header columns with `is_static ==
41/// false` AND `!is_primary_key` AND `!is_clustering`. Every column has exactly
42/// one `is_static` value, so the `is_static == want_static` filter makes
43/// `columns_for(true)` and `columns_for(false)` name-disjoint; and
44/// `columns_for(true)` excludes clustering-key columns, so no static cell shares
45/// a clustering-key pseudo-cell name. Hence the two name sets are disjoint and
46/// the former membership guard could never fire.
47///
48/// Appending AFTER the clustering row's own cells keeps the merged order
49/// deterministic-by-construction (never user-visible: the query result is a
50/// name-keyed map, issue #1334).
51pub(super) fn merge_static_cells(cells: &mut RowCells, static_cells: &RowCells) {
52    cells.extend(
53        static_cells
54            .iter()
55            .map(|(name, value)| (Arc::clone(name), value.clone())),
56    );
57}

59/// Issue #932/#1741: build the user-facing `ScanRow` display value for a parsed
60/// clustering row from its decoded `cells` and row header. Shared by every
61/// user-facing emit path so the row-tombstone display rule lives in ONE place:
62/// a `HAS_DELETION` row that carries NO surviving non-key cell displays as a pure
63/// `Tombstone` marker (suppressed downstream by `filter_tombstone`); a row that
64/// still carries surviving cells displays as a live `Row` (the deletion shadows
65/// only already-absent older cells). An empty cell set becomes a null marker.
66pub(super) fn build_display_row(
67    cells: RowCells,
68    row_header_opt: Option<&RowHeader>,
69    schema: &TableSchema,
70) -> ScanRow {
71    let row_tombstone = row_header_opt.filter(|h| h.is_row_tombstone());
72    let has_data_cell = row_has_non_key_cell(&cells, schema);
73    if row_tombstone.is_some() && !has_data_cell {
74        ScanRow::Marker(
75            row_tombstone
76                .map(|h| h.row_tombstone())
77                .unwrap_or(Value::Null),
78        )
79    } else if cells.is_empty() {
80        ScanRow::Marker(Value::Null)
81    } else {
82        // Issue #1642 (K3): the decoder already emits cells positionally, in
83        // serialization-header (schema) column order — determinism comes from
84        // CONSTRUCTION, not a per-row sort. The former per-row `HashMap`
85        // allocation and alphabetical `sort_by` are gone; the interned
86        // `Arc<str>` name handles (#1334) move straight into the carrier.
87        ScanRow::Row(cells)
88    }
89}

91/// Issue #1741: extract a clustering row's clustering-key values (in schema
92/// clustering order) from its decoded cell map. Only called when a range
93/// tombstone is currently OPEN in the partition, so the per-row clone is off the
94/// tombstone-free hot path. Returns fewer values than the clustering arity only
95/// for a malformed/partial row (missing clustering pseudo-cells).
96pub(super) fn extract_clustering_values(
97    cells: &[(Arc<str>, Value)],
98    schema: &TableSchema,
99) -> Vec<Value> {
100    schema
101        .clustering_keys
102        .iter()
103        .filter_map(|ck| {
104            cells
105                .iter()
106                .find(|(name, _)| name.as_ref() == ck.name.as_str())
107                .map(|(_, v)| v.clone())
108        })
109        .collect()
110}

112/// Whether a built display row is VISIBLE to a user-facing `SELECT` (issue #3095).
113///
114/// [`build_display_row`] returns `ScanRow::Marker` for a pure row tombstone / absent
115/// row, and EVERY user-facing consumer suppresses a marker downstream
116/// (`integrity::filter_tombstone`, and `build_row_from_scan`'s `into_cells`, issue
117/// #505) — so a marker is not a row a `SELECT` returns. This is the single predicate
118/// the static-content-on-an-empty-partition rule uses for Cassandra's
119/// `partition.hasNext()`, which is likewise evaluated over the already-FILTERED
120/// `RowIterator`: `UnfilteredRowIterators.filter(iter, nowInSec)` →
121/// `FilteredRows.filter` installs `db/transform/Filter`, whose
122/// `applyToRow(row) = row.purge(PURGE_ALL, nowInSec, enforceStrictLiveness)` drops a
123/// row that purges to `null` (cassandra-5.0.8).
124pub(super) fn row_is_visible(row: &ScanRow) -> bool {
125    matches!(row, ScanRow::Row(_) | ScanRow::RawRow(_))
126}

128/// Build the display row for a CLUSTERING row of a static-bearing partition, on a
129/// user-facing SELECT read (issue #3095).
130///
131/// The ORDER is load-bearing: the row-tombstone display decision is taken over the
132/// row's OWN cells, and static cells are injected only into a row that survives it.
133///
134/// Cassandra authority, VERBATIM from the pinned `cassandra-5.0.8` tag:
135///
136/// * `db/transform/Filter.java` applies the static row and the clustering rows as
137///   SEPARATE transformations — `protected Row applyToStatic(Row row)` vs
138///   `protected Row applyToRow(Row row) { return row.purge(DeletionPurger.PURGE_ALL,
139///   nowInSec, enforceStrictLiveness); }`. A clustering row is purged from ITSELF and
140///   dropped when `purge` returns `null`; the static row never participates.
141/// * `db/rows/BTreeRow.java`'s `purge` → `update(info, deletion, newTree)` returns
142///   `null` when `info.isEmpty() && deletion.isLive() && BTree.isEmpty(newTree)` —
143///   the row's OWN primary-key liveness, OWN deletion and OWN cell btree. A static
144///   cell can never be in a clustering row's btree (the static row is a separate
145///   `Row` reached out of band via `BaseRowIterator.staticRow()`).
146/// * `db/transform/FilteredRows.java` states the separation outright:
147///   `public boolean isEmpty() { return staticRow().isEmpty() && !hasNext(); }` —
148///   `hasNext()` counts clustering rows only, which is why
149///   `SelectStatement.processPartition()` can branch on `!partition.hasNext()` while
150///   still reading `partition.staticRow()`.
151///
152/// So a static cell can NEVER make a row-tombstoned clustering row live, and a
153/// partition whose clustering rows all purge away still returns its static content.
154/// Injecting first (which
155/// [`merge_static_cells`] + [`build_display_row`] do when called in that order)
156/// made `row_has_non_key_cell` true for a PURE row tombstone, so the row surfaced
157/// as a live `ScanRow::Row` carrying the static value — a phantom row on the
158/// single-generation arm that the k-way merge arm correctly suppresses
159/// (`entry_to_row` drops `RowData::Tombstone`). That divergence is what made the
160/// two arms disagree on a static partition holding only deleted rows.
161///
162/// PHYSICAL consumers (compaction, `verify`, delta-scan) keep the historical
163/// inject-then-decide order — their callers must see every on-disk unfiltered and
164/// their output is byte-pinned — so this is used only where `read_shadowing` is on.
165pub(super) fn build_display_row_read_path(
166    cells: RowCells,
167    static_cells: &RowCells,
168    row_header_opt: Option<&RowHeader>,
169    schema: &TableSchema,
170) -> ScanRow {
171    match build_display_row(cells, row_header_opt, schema) {
172        ScanRow::Row(mut kept) => {
173            merge_static_cells(&mut kept, static_cells);
174            ScanRow::Row(kept)
175        }
176        // A pure row tombstone (or an empty row): statics do not revive it.
177        other => other,
178    }
179}
