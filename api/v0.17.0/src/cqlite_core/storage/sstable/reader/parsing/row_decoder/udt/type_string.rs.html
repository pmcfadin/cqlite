1//! Parse a Cassandra **marshal TYPE STRING** into CQLite schema types (issue #3631).
2//!
3//! Split out of `udt.rs` under the campsite rule (epic #1116: that file is more than
4//! 1.5x the 800-line source target) along a responsibility boundary that was already
5//! there: everything here turns a `SerializationHeader` type STRING into a
6//! [`UdtTypeDef`] / [`CqlType`], while `udt.rs` decodes UDT **value bytes**.
7//!
8//! # Format authority — the pinned tag, never CQLite's own tables (#3041)
9//!
10//! The marshal-name -> CQL-type mapping is
11//! `cassandra-5.0.8:src/java/org/apache/cassandra/cql3/CQL3Type.java`'s `Native`
12//! enum, which is the single place Cassandra binds a CQL native type to its
13//! `AbstractType` implementation:
14//!
15//! | CQL native | marshal class     | CQL native | marshal class      |
16//! |------------|-------------------|------------|--------------------|
17//! | ascii      | `AsciiType`       | int        | `Int32Type`        |
18//! | bigint     | `LongType`        | smallint   | `ShortType`        |
19//! | blob       | `BytesType`       | text       | `UTF8Type`         |
20//! | boolean    | `BooleanType`     | time       | `TimeType`         |
21//! | counter    | `CounterColumnType` | timestamp | `TimestampType`   |
22//! | date       | `SimpleDateType`  | timeuuid   | `TimeUUIDType`     |
23//! | decimal    | `DecimalType`     | tinyint    | `ByteType`         |
24//! | double     | `DoubleType`      | uuid       | `UUIDType`         |
25//! | duration   | `DurationType`    | varchar    | `UTF8Type`         |
26//! | float      | `FloatType`       | varint     | `IntegerType`      |
27//! | inet       | `InetAddressType` |            |                    |
28//!
29//! Three names are NOT in that enum and are mapped anyway, each from its own
30//! authority at the same tag:
31//!
32//! * `DateType` -> `timestamp`. `DateType.asCQL3Type()` returns
33//!   `CQL3Type.Native.TIMESTAMP` — it is the LEGACY 8-byte millis type, NOT CQL
34//!   `date`. It is listed here because a suffix match on `DateType` also matches
35//!   `SimpleDateType` (the real `date`), which is how a `date` field could have been
36//!   decoded as an 8-byte timestamp.
37//! * `VarcharType` -> `text`. An alias `TypeParser` resolves to `UTF8Type`.
38//! * `LexicalUUIDType` -> `uuid`. It has no `asCQL3Type` override, but
39//!   `LexicalUUIDType.Serializer extends UUIDSerializer` and
40//!   `valueLengthIfFixed() == 16`: the VALUE layout is a UUID's, and only the
41//!   comparison order differs.
42//!
43//! `EmptyType`, `VectorType(...)`, `CompositeType(...)`, `DynamicCompositeType(...)`
44//! and `PartitionerDefinedOrder` are deliberately NOT mapped — [`CqlType`] has no
45//! variant that can express them, so they stay [`CqlType::Custom`] and the DECODER
46//! refuses them by name (issue #3631 criterion 5) rather than guessing.
47//!
48//! # The PACKAGE is part of the identity (roborev job 76)
49//!
50//! Every name above is matched under the ONE package rule in
51//! [`marshal_name`](super::marshal_name) — bare, or fully qualified under
52//! `org.apache.cassandra.db.marshal`, and nothing else. A third-party class that
53//! merely SHARES a simple name (`com.acme.Int32Type`, `com.acme.TupleType(...)`) is
54//! refused with [`V5CompressedLegacyParser::foreign_marshal_package_error`], never
55//! decoded as the native type it resembles: CQLite knows nothing about that class's
56//! byte layout, and picking one from a name is a heuristic (#28).

73    /// Parse a UDT type string to extract the UDT definition.
74    /// Cassandra encodes UDTs as:
75    /// `UserType(keyspace,hex_name,field1_hex:type1,field2_hex:type2,...)`
76    ///
77    /// Example:
78    /// ```text
79    /// org.apache.cassandra.db.marshal.UserType(
80    ///   test_collections,
81    ///   616464726573735f74797065,    // hex("address_type")
82    ///   737472656574:UTF8Type,        // street:UTF8Type
83    ///   63697479:UTF8Type,            // city:UTF8Type
84    ///   ...
85    /// )
86    /// ```
87    pub(in crate::storage::sstable::reader::parsing::row_decoder) fn parse_udt_type_definition(
88        type_str: &str,
89    ) -> Result<UdtTypeDef> {
90        Self::parse_udt_type_definition_with_depth(type_str, 0)
91    }

93    /// Internal implementation of parse_udt_type_definition with recursion depth tracking.
94    pub(super) fn parse_udt_type_definition_with_depth(
95        type_str: &str,
96        depth: usize,
97    ) -> Result<UdtTypeDef> {
98        // Check recursion depth to prevent stack overflow
99        if depth > MAX_TYPE_NESTING_DEPTH {
100            return Err(Error::schema(format!(
101                "UDT nesting depth {} exceeds maximum {}. Type string: {}",
102                depth,
103                MAX_TYPE_NESTING_DEPTH,
104                type_str.chars().take(100).collect::<String>()
105            )));
106        }
107
108        // Locate + split the marshal `UserType(...)` arguments through the ONE
109        // locator (`marshal_name.rs`), which validates the PACKAGE of the marker's
110        // class name rather than substring-matching the qualified literal — a plain
111        // `find` accepted `my.org.apache.cassandra.db.marshal.UserType(…)`, a
112        // package SUFFIX (roborev job 76). `udt_field_marshal_types` shares it, so
113        // the two `UserType(` consumers can no longer drift.
114        let parts = Self::marshal_user_type_args(type_str)?;
115
116        // First part is keyspace
117        let keyspace = parts[0].trim();
118        if keyspace.is_empty() {
119            return Err(Error::schema("UDT keyspace cannot be empty"));
120        }
121        let keyspace = keyspace.to_string();
122
123        // Second part is hex-encoded type name
124        let udt_name = Self::decode_hex_name(parts[1].trim())?;
125
126        // Remaining parts are field definitions: hex_name:type
127        let mut udt_def = UdtTypeDef::new(keyspace, udt_name);
128        for field_def in parts.iter().skip(2) {
129            let field_def = field_def.trim();
130            if field_def.is_empty() {
131                continue;
132            }
133
134            // Split on first colon (field name is before, type is after)
135            if let Some(colon_idx) = field_def.find(':') {
136                let field_name_hex = &field_def[..colon_idx];
137                let field_type_str = &field_def[colon_idx + 1..];
138
139                let field_name = Self::decode_hex_name(field_name_hex)?;
140                // Use depth-aware version to track recursion through UDT fields
141                let field_type = Self::parse_cassandra_type_with_depth(field_type_str, depth)?;
142
143                udt_def = udt_def.with_field(field_name, field_type, true);
144            } else {
145                return Err(Error::schema(format!(
146                    "Invalid UDT field definition (missing colon): {}",
147                    field_def
148                )));
149            }
150        }
151
152        Ok(udt_def)
153    }

155    /// Split type arguments by comma, respecting nested parentheses.
156    pub(super) fn split_type_args(s: &str) -> Result<Vec<String>> {
157        let mut parts = Vec::new();
158        let mut current = String::new();
159        let mut depth = 0;
160
161        for c in s.chars() {
162            match c {
163                '(' => {
164                    depth += 1;
165                    current.push(c);
166                }
167                ')' => {
168                    depth -= 1;
169                    current.push(c);
170                }
171                ',' if depth == 0 => {
172                    parts.push(current.clone());
173                    current.clear();
174                }
175                _ => current.push(c),
176            }
177        }
178
179        if !current.is_empty() {
180            parts.push(current);
181        }
182
183        Ok(parts)
184    }

186    /// Decode a hex-encoded name (e.g., "616464726573735f74797065" -> "address_type")
187    pub(super) fn decode_hex_name(hex: &str) -> Result<String> {
188        let bytes = hex::decode(hex)
189            .map_err(|e| Error::schema(format!("Invalid hex-encoded UDT name '{}': {}", hex, e)))?;
190        String::from_utf8(bytes)
191            .map_err(|e| Error::schema(format!("Invalid UTF-8 in UDT name '{}': {}", hex, e)))
192    }

194    /// Parse a Cassandra MARSHAL type string into a [`CqlType`].
195    /// Handles: UTF8Type, Int32Type, ListType(...), SetType(...), MapType(...), UserType(...), FrozenType(...)
196    ///
197    /// This is the crate's ONE marshal-string -> [`CqlType`] entry point, and it
198    /// is `pub(crate)` because both the READ and the WRITE side need it: the
199    /// empty-buffer sentinel's admission on a multicell collection's cell path is
200    /// decided against the DECLARED component type — a map's KEY (issue #3805) or
201    /// a set's ELEMENT (issue #4106) — and the column's declared type arrives in
202    /// either spelling, a CQL `map<K,V>`/`set<T>` or this marshal form. Both
203    /// sides reach it through ONE resolver,
204    /// `crate::storage::sstable::cell_path_component::resolve_declared_cell_path_type`.
205    /// It is deliberately THIS parser and not the string->string
206    /// `convert_marshal_type_to_cql`: the table above is derived name-by-name
207    /// from `cql3/CQL3Type.java`'s `Native` enum at `cassandra-5.0.8` and
208    /// enforces the package rule, so `IntegerType` is `varint` (not `int`) and a
209    /// third-party class sharing a native simple name is refused rather than
210    /// decoded as the type it resembles.
211    ///
212    /// # NO `allow(dead_code)`, conditional or otherwise
213    /// It carried `#[cfg_attr(not(feature = "write-support"), allow(dead_code))]`
214    /// while its ONLY caller was the write-side gate. #4106 gave it an
215    /// UNCONDITIONAL caller on the READ path (the set cell-path admission
216    /// resolves from the complete declared type), so the attribute would now
217    /// silence nothing and merely assert something false. Do not restore it: if
218    /// this ever becomes unreachable again, the honest fix is to say so at the
219    /// site that stopped calling it.
220    pub(crate) fn parse_cassandra_type(type_str: &str) -> Result<CqlType> {
221        Self::parse_cassandra_type_with_depth(type_str, 0)
222    }

224    /// Internal implementation of parse_cassandra_type with recursion depth tracking.
225    pub(super) fn parse_cassandra_type_with_depth(type_str: &str, depth: usize) -> Result<CqlType> {
226        // Check recursion depth to prevent stack overflow
227        if depth > MAX_TYPE_NESTING_DEPTH {
228            return Err(Error::schema(format!(
229                "Type nesting depth {} exceeds maximum {}. Type string: {}",
230                depth,
231                MAX_TYPE_NESTING_DEPTH,
232                type_str.chars().take(100).collect::<String>()
233            )));
234        }
235
236        let type_str = type_str.trim();
237
238        // Handle FrozenType wrapper
239        if let Some(args) = Self::marshal_parameterised_inner(type_str, "FrozenType") {
240            let inner = Self::extract_inner_parens(args)?;
241            let inner_type = Self::parse_cassandra_type_with_depth(&inner, depth + 1)?;
242            return Ok(CqlType::Frozen(Box::new(inner_type)));
243        }
244
245        // Handle UserType (nested UDT)
246        if Self::marshal_parameterised_inner(type_str, "UserType").is_some() {
247            let udt_def = Self::parse_udt_type_definition_with_depth(type_str, depth + 1)?;
248            let fields: Vec<(String, CqlType)> = udt_def
249                .fields
250                .into_iter()
251                .map(|f| (f.name, f.field_type))
252                .collect();
253            return Ok(CqlType::Udt(udt_def.name, fields));
254        }
255
256        // Handle collection types
257        if let Some(args) = Self::marshal_parameterised_inner(type_str, "ListType") {
258            let inner = Self::extract_inner_parens(args)?;
259            let elem_type = Self::parse_cassandra_type_with_depth(&inner, depth + 1)?;
260            return Ok(CqlType::List(Box::new(elem_type)));
261        }
262
263        if let Some(args) = Self::marshal_parameterised_inner(type_str, "SetType") {
264            let inner = Self::extract_inner_parens(args)?;
265            let elem_type = Self::parse_cassandra_type_with_depth(&inner, depth + 1)?;
266            return Ok(CqlType::Set(Box::new(elem_type)));
267        }
268
269        if let Some(args) = Self::marshal_parameterised_inner(type_str, "MapType") {
270            let inner = Self::extract_inner_parens(args)?;
271            let parts = Self::split_type_args(&inner)?;
272            if parts.len() != 2 {
273                return Err(Error::schema(format!(
274                    "MapType requires exactly 2 type arguments: {}",
275                    type_str
276                )));
277            }
278            let key_type = Self::parse_cassandra_type_with_depth(&parts[0], depth + 1)?;
279            let val_type = Self::parse_cassandra_type_with_depth(&parts[1], depth + 1)?;
280            return Ok(CqlType::Map(Box::new(key_type), Box::new(val_type)));
281        }
282
283        // TupleType — STRUCTURAL, so it is PARSED and not name-matched: its marshal
284        // string is `getClass().getName() + stringifyTypeParameters(types, true)`
285        // (cassandra-5.0.8 TupleType.java:557), i.e. a parenthesised, comma-separated
286        // component list. `depth + 1` like every other structural arm — a reset here
287        // would make `MAX_TYPE_NESTING_DEPTH` bound nothing across a
288        // tuple-inside-collection-inside-tuple chain.
289        if let Some(args) = Self::marshal_parameterised_inner(type_str, "TupleType") {
290            let inner = Self::extract_inner_parens(args)?;
291            let parts = Self::split_type_args(&inner)?;
292            if parts.is_empty() {
293                return Err(Error::schema(format!(
294                    "TupleType requires at least one component type: {}",
295                    type_str
296                )));
297            }
298            // `parts.len()` is bounded by the length of a type string that came out of
299            // the SerializationHeader, not by a length field inside a value, so this
300            // reserve cannot be inflated by a corrupt count.
301            let mut components = Vec::with_capacity(parts.len());
302            for part in &parts {
303                components.push(Self::parse_cassandra_type_with_depth(part, depth + 1)?);
304            }
305            return Ok(CqlType::Tuple(components));
306        }
307
308        // VectorType — STRUCTURAL, so it is PARSED and not name-matched (issue
309        // #4114). Its marshal string is
310        // `getClass().getName() + stringifyVectorParameters(type, dimension)`
311        // (`cassandra-5.0.8` VectorType.java:339-342 / TypeParser.java:239-242),
312        // i.e. `VectorType(<element> , <n>)`. `depth + 1` like every other
313        // structural arm.
314        //
315        // Before this arm the whole type collapsed to `CqlType::Custom`, which the
316        // decoder refuses by name — honest, but it meant a `vector<float, n>` UDT
317        // FIELD could not be read at all. The dimension is carried into
318        // `CqlType::Vector` because the on-disk value has no element count and no
319        // per-element framing for a fixed-width element: `n` is the only thing that
320        // makes it parseable (#28).
321        if let Some(args) = Self::marshal_parameterised_inner(type_str, "VectorType") {
322            let inner = Self::extract_inner_parens(args)?;
323            let parsed = crate::schema::vector_type::split_vector_args(&inner, type_str)?;
324            let element = Self::parse_cassandra_type_with_depth(parsed.element, depth + 1)?;
325            return Ok(CqlType::Vector(Box::new(element), parsed.dimension));
326        }
327
328        // ReversedType — a COMPARISON wrapper with no layout of its own:
329        // `ReversedType.asCQL3Type()` and `getSerializer()` both delegate to
330        // `baseType` (cassandra-5.0.8 ReversedType.java:138,144), so the value of a
331        // `ReversedType(X)` is byte-for-byte the value of an `X`.
332        if let Some(args) = Self::marshal_parameterised_inner(type_str, "ReversedType") {
333            let inner = Self::extract_inner_parens(args)?;
334            return Self::parse_cassandra_type_with_depth(&inner, depth + 1);
335        }
336
337        // THE PACKAGE RULE, at the ONE point every non-structural name reaches
338        // (roborev job 76). A name qualified outside Cassandra's marshal package is
339        // a third-party `AbstractType`, and it is refused HERE rather than passed
340        // on as `Custom`, for a reason that is about WHERE THE CONTEXT LIVES: this
341        // function is the only place that knows the string is a MARSHAL type
342        // string, where a dotted name is a Java class name. A `CqlType::Custom`
343        // payload has lost that context — it carries a marshal reference OR a
344        // KEYSPACE-QUALIFIED UDT NAME (`test_ks.address`, a real form on the write
345        // path) — so no downstream predicate can tell `acme.Int32Type` from
346        // `ks.address` without guessing (#28). Structural forms reach this line
347        // too: they match no arm above (each requires the qualified marshal
348        // spelling), so `com.acme.TupleType(…)` is refused here by the same rule.
349        if Self::marshal_simple_name(Self::marshal_head(type_str)).is_none() {
350            return Err(Self::foreign_marshal_package_error(type_str));
351        }
352
353        // A native (non-parameterised) marshal type, else `Custom` — which the
354        // DECODER refuses by name (issue #3631 criterion 5). Nothing here guesses
355        // from bytes (#28): the mapping is keyed on the DECLARED marshal name only.
356        Ok(Self::native_marshal_to_cql_type(type_str)
357            .unwrap_or_else(|| CqlType::Custom(Self::canonical_marshal_class_name(type_str))))
358    }

360    /// The name Cassandra itself would resolve `type_str` to:
361    /// `TypeParser.getAbstractType` (cassandra-5.0.8 TypeParser.java:450) loads
362    /// `compareWith.contains(".") ? compareWith : "org.apache.cassandra.db.marshal." +
363    /// compareWith`, so inside a marshal type string an UNQUALIFIED name IS a class in
364    /// the marshal package.
365    ///
366    /// Recording the resolved form is what lets the decoder tell an unmappable marshal
367    /// type from a UDT NAME: both are [`CqlType::Custom`], and only the qualified
368    /// spelling is unambiguous (roborev job 68, finding 1). Every field type in a real
369    /// `SerializationHeader` already arrives qualified, so this is the tolerated
370    /// hand-written / `TypeParser`-legal bare spelling, normalised by Cassandra's own
371    /// rule rather than by a guess about which names are class names.
372    fn canonical_marshal_class_name(type_str: &str) -> String {
373        let name = type_str.trim();
374        if name.contains('.') {
375            return name.to_string();
376        }
377        format!("{}{name}", Self::MARSHAL_PACKAGE)
378    }

380    /// The marshal-class-name -> [`CqlType`] table: the ONE place a Cassandra native
381    /// `AbstractType` is bound to a CQLite type. `None` for a parameterised form (the
382    /// structural arms above own those) and for a native type [`CqlType`] cannot
383    /// express — see this module's header for the per-name authority and for the
384    /// names deliberately left unmapped.
385    ///
386    /// # The name is resolved under the ONE package rule, then matched EXACTLY
387    /// [`Self::marshal_simple_name`] (see `marshal_name.rs` for the pinned
388    /// `TypeParser` authority) accepts a marshal name in EXACTLY TWO spellings — a
389    /// bare simple name, or a fully-qualified class name under
390    /// `org.apache.cassandra.db.marshal` — and yields the simple name, which is
391    /// then the whole identity of the type.
392    ///
393    /// Two matching mistakes this closes, both of which shipped:
394    ///
395    /// * A **suffix** match (`ends_with`) is at once too loose and too tight.
396    ///   `ends_with("DateType")` also matches `SimpleDateType` (the real CQL
397    ///   `date`), so whether a `date` field decoded as an 8-byte `timestamp`
398    ///   depended on arm ORDER.
399    /// * Taking the text after the last `.` and ignoring the **package** decoded a
400    ///   third-party `com.acme.Int32Type` as CQL `int` — an unknown class's bytes
401    ///   read as if the class were known (roborev job 76). The rule now returns
402    ///   `None` for it, and the type-string parser refuses it by name.
403    pub(in crate::storage::sstable::reader::parsing::row_decoder) fn native_marshal_to_cql_type(
404        marshal_type: &str,
405    ) -> Option<CqlType> {
406        let name = marshal_type.trim();
407        // A parameterised form is structural; the arms above are its only decoders.
408        if name.contains('(') {
409            return None;
410        }
411        // THE package rule (`marshal_name.rs`): `Some` only for a bare simple name
412        // or the canonical `org.apache.cassandra.db.marshal.X` spelling, so a
413        // third-party `com.acme.Int32Type` cannot reach the table below by simple
414        // name alone.
415        let simple = Self::marshal_simple_name(name)?;
416        Some(match simple {
417            // -- CQL3Type.Native, one arm per enum constant --------------------
418            "AsciiType" => CqlType::Ascii,
419            "LongType" => CqlType::BigInt,
420            "BytesType" => CqlType::Blob,
421            "BooleanType" => CqlType::Boolean,
422            "CounterColumnType" => CqlType::Counter,
423            "SimpleDateType" => CqlType::Date,
424            "DecimalType" => CqlType::Decimal,
425            "DoubleType" => CqlType::Double,
426            "DurationType" => CqlType::Duration,
427            "FloatType" => CqlType::Float,
428            "InetAddressType" => CqlType::Inet,
429            "Int32Type" => CqlType::Int,
430            "ShortType" => CqlType::SmallInt,
431            "UTF8Type" => CqlType::Text,
432            "TimeType" => CqlType::Time,
433            "TimestampType" => CqlType::Timestamp,
434            "TimeUUIDType" => CqlType::TimeUuid,
435            "ByteType" => CqlType::TinyInt,
436            "UUIDType" => CqlType::Uuid,
437            "IntegerType" => CqlType::Varint,
438            // -- Not in that enum; each mapped from its own authority ----------
439            // `DateType.asCQL3Type()` -> `TIMESTAMP`: the LEGACY 8-byte millis type.
440            "DateType" => CqlType::Timestamp,
441            // An alias `TypeParser` resolves to `UTF8Type`.
442            "VarcharType" => CqlType::Text,
443            // `LexicalUUIDType.Serializer extends UUIDSerializer`, 16 bytes fixed.
444            "LexicalUUIDType" => CqlType::Uuid,
445            // `LegacyTimeUUIDType extends AbstractTimeUUIDType<UUID>`: a timeuuid's
446            // value layout, differing only in comparison.
447            "LegacyTimeUUIDType" => CqlType::TimeUuid,
448            _ => return None,
449        })
450    }

452    /// Whether a [`CqlType::Custom`] payload is a **marshal type REFERENCE** — a
453    /// Cassandra `AbstractType` class name, or a structural marshal fragment — rather
454    /// than the NAME OF A UDT.
455    ///
456    /// `CqlType::Custom` carries both (see `crate::schema::is_udt_identifier`), and
457    /// the typed decoder has to route them differently: a UDT name goes to the
458    /// nested-UDT decoder, while a marshal reference has no field list and never
459    /// will, so reporting it as a UDT with a missing field list misattributes the
460    /// cause (roborev job 68, finding 1).
461    ///
462    /// The qualified package prefix is the discriminator because it is the single
463    /// shape real `SerializationHeader`s carry — the same fact
464    /// `marshal_is_top_level_frozen_udt` and `parse_udt_type_definition_with_depth`
465    /// already key on (roborev jobs 1359/1361). This reads a DECLARED type string's
466    /// own syntax; it infers nothing from value bytes (#28).
467    pub(in crate::storage::sstable::reader::parsing::row_decoder) fn custom_is_marshal_type_reference(
468        name: &str,
469    ) -> bool {
470        const PKG: &str = "org.apache.cassandra.";
471        let name = name.trim();
472        // A structural fragment: no UDT name can carry a parameter list.
473        if name.contains('(') || name.contains(',') || name.contains('<') {
474            return true;
475        }
476        // `str::get` yields `None` on a non-char-boundary index, so this cannot panic
477        // on a multi-byte name.
478        name.get(..PKG.len())
479            .is_some_and(|prefix| prefix.eq_ignore_ascii_case(PKG))
480    }

482    /// Extract the contents inside parentheses, respecting nesting.
483    ///
484    /// Indexed by `char_indices`, i.e. by BYTE offset. This used to collect a
485    /// `Vec<char>` and slice `s[..i]` with the CHARACTER index `i`, which for any
486    /// type string carrying a multi-byte character before the closing paren is
487    /// either the wrong slice or — when the index lands mid-character — a PANIC
488    /// (`extract_inner_parens("é)")` sliced `s[..1]` inside a 2-byte `é`). A
489    /// SerializationHeader type string is attacker-influenced input, so the read
490    /// path must not panic on it.
491    pub(super) fn extract_inner_parens(s: &str) -> Result<String> {
492        let mut depth = 1usize;
493        for (offset, c) in s.char_indices() {
494            match c {
495                '(' => depth += 1,
496                ')' => {
497                    depth -= 1;
498                    if depth == 0 {
499                        return Ok(s[..offset].to_string());
500                    }
501                }
502                _ => {}
503            }
504        }
505        Err(Error::schema(format!(
506            "Unbalanced parentheses in type: {}",
507            s
508        )))
509    }
510}
