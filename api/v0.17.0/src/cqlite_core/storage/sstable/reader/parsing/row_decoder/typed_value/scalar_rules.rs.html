1//! The per-SCALAR decode rules the `CqlType` decoder needs: a scalar's canonical CQL
2//! short form, the widths its serialized value may take, and whether an EMPTY value
3//! is a value or a NULL (issue #3631).
4//!
5//! Split out of `typed_value.rs` under the campsite rule (epic #1116) along a real
6//! responsibility boundary: `typed_value.rs` walks the STRUCTURE of a value
7//! (collections, tuples, UDTs, the exhaustion assert) while these three tables state
8//! what Cassandra's per-type `validate` / `deserialize` implementations accept for a
9//! single scalar. Each is read at the pinned tag, per type — never inferred from
10//! CQLite's own prior output (#3041).
11
12use super::super::*;

14impl V5CompressedLegacyParser {
15    /// The NON-EMPTY serialized widths `short` admits, or `&[]` when the type is
16    /// genuinely variable-width (`text`, `blob`, `varint`, `decimal`, `duration`).
17    /// An EMPTY buffer is governed separately by [`Self::empty_is_a_value`], so it is
18    /// deliberately absent from these lists.
19    ///
20    /// Authority: `cassandra-5.0.8:src/java/org/apache/cassandra/db/marshal/`, whose
21    /// fixed-width `AbstractType.validate` implementations accept the exact width OR
22    /// an EMPTY buffer and reject everything else (e.g. `Int32Type.validate`:
23    /// `if (bytes.remaining() != 4 && bytes.remaining() != 0) throw`). The
24    /// type-string decoder this module delegates to bounds-checks with `<`, so it
25    /// happily reads a 4-byte int out of a 9-byte frame and drops the rest — the same
26    /// silent-discard class the exhaustion assert exists to refuse, one level down.
27    ///
28    /// # Why this is a SET and not one width (roborev job 68, finding 2)
29    /// `inet` has TWO legal widths, so as a single `Option<usize>` it read as
30    /// "variable-width" and skipped the gate entirely — and the delegated `"inet"`
31    /// arm borrows whatever slice it is handed, so a five-byte field surfaced as a
32    /// `Value::Inet` no `InetAddress` can be built from.
33    /// `serializers/InetAddressSerializer.java` `validate` RETURNS EARLY on an empty
34    /// buffer and otherwise delegates to `InetAddress.getByAddress`, throwing
35    /// *"Expected 4 or 16 byte inetaddress"*; `deserialize` is
36    /// `if (accessor.isEmpty(value)) return null` and
37    /// `InetAddressType.isEmptyValueMeaningless()` is `true`, which is exactly what
38    /// `empty_is_a_value("inet") == false` already encodes.
39    ///
40    /// That is ONE rule with one authority, and this repository already states it
41    /// once — `complex_column/cell_path_key.rs`'s `cql_short_allowed_widths` row
42    /// `"inet" => &[0, 4, 16]` (its `0` is the empty case this predicate delegates).
43    /// The two tables answer deliberately different questions — that one follows
44    /// `validate`, which refuses an empty `date`/`time`/`tinyint`/`smallint`, while
45    /// this one follows `deserialize`, which returns NULL for them — so they are not
46    /// merged here; only the widths agree, which is what finding 2 is about.
47    pub(super) fn scalar_allowed_widths(short: &str) -> &'static [usize] {
48        match short {
49            "boolean" | "tinyint" => &[1],
50            "smallint" => &[2],
51            "int" | "float" | "date" => &[4],
52            "bigint" | "counter" | "double" | "time" | "timestamp" => &[8],
53            "uuid" | "timeuuid" => &[16],
54            "inet" => &[4, 16],
55            _ => &[],
56        }
57    }
58
59    /// The width list as the refusal renders it: `"4"`, or `"4 or 16"`. Keeps the
60    /// single-width wording byte-identical to what it always was.
61    pub(super) fn render_allowed_widths(widths: &[usize]) -> String {
62        widths
63            .iter()
64            .map(|w| w.to_string())
65            .collect::<Vec<_>>()
66            .join(" or ")
67    }

69    /// Whether an EMPTY serialized value is a VALUE for `short`, rather than NULL.
70    ///
71    /// # Measured at the pinned tag, serializer by serializer (issue #3631)
72    /// A zero-length UDT field carries `ByteBufferUtil.EMPTY_BYTE_BUFFER`, so the
73    /// question is what Cassandra's `deserialize` returns for it. Read at
74    /// `cassandra-5.0.8:src/java/org/apache/cassandra/serializers/`:
75    ///
76    /// * `Int32Serializer`, `LongSerializer`, `ShortSerializer`, `ByteSerializer`,
77    ///   `BooleanSerializer`, `FloatSerializer`, `DoubleSerializer`, `UUIDSerializer`,
78    ///   `TimeUUIDSerializer`, `TimestampSerializer`, `SimpleDateSerializer`,
79    ///   `TimeSerializer`, `DecimalSerializer`, `IntegerSerializer` (varint),
80    ///   `InetAddressSerializer` and `DurationSerializer` all read
81    ///   `accessor.isEmpty(value) ? null : …` — i.e. **empty means NULL**. Note that
82    ///   this is WIDER than `AbstractType.isEmptyValueMeaningless()`, which
83    ///   `ShortType`/`ByteType`/`SimpleDateType`/`TimeType`/`DurationType` do not
84    ///   override even though their serializers return null; the serializer is the
85    ///   authority for what the bytes MEAN, so it is what this predicate follows.
86    /// * `UTF8Serializer`/`AsciiSerializer` (text/ascii/varchar) and `BytesSerializer`
87    ///   (blob) have no such guard: an empty buffer is the empty string / empty blob,
88    ///   a real value.
89    ///
90    /// Structured types (list/set/map/tuple/udt) are not scalars and never reach here;
91    /// their empty semantics are the empty collection / all-null components, spelled by
92    /// the arms in [`Self::parse_typed_value_reporting`].
93    ///
94    /// This replaced `create_empty_value_for_type`, whose `_ =>` arm was an EMPTY BLOB
95    /// for every type it did not enumerate — so an empty `int`, an empty `tuple` and an
96    /// empty nested UDT all surfaced as `Blob([])`, which is #3631 criterion 5's silent
97    /// degradation in the zero-length case.
98    pub(super) fn empty_is_a_value(short: &str) -> bool {
99        matches!(short, "text" | "ascii" | "varchar" | "blob")
100    }

102    /// A short, allocation-free label for `ty`, for the consumption assert's message.
103    pub(super) fn typed_value_label(ty: &CqlType) -> &'static str {
104        match ty {
105            CqlType::Frozen(_) => "frozen",
106            CqlType::List(_) => "list",
107            CqlType::Set(_) => "set",
108            CqlType::Map(_, _) => "map",
109            CqlType::Tuple(_) => "tuple",
110            CqlType::Udt(_, _) => "udt",
111            // A `Custom` is a UDT NAME or an unmappable marshal reference; label it
112            // the way `parse_typed_value_reporting` ROUTES it, so a consumption
113            // refusal cannot say "udt" about a type that never went down that arm
114            // (roborev job 68, finding 1).
115            CqlType::Custom(name) => {
116                if Self::custom_is_marshal_type_reference(name) {
117                    "unknown"
118                } else {
119                    "udt"
120                }
121            }
122            scalar => Self::cql_scalar_short_form(scalar).unwrap_or("unknown"),
123        }
124    }

126    /// The canonical CQL short form of a SCALAR `CqlType`, or `None` when `ty` is not
127    /// a scalar this decoder can name.
128    ///
129    /// The returned strings are the arm labels of `parse_value_from_raw_bytes`, which
130    /// implements each scalar's on-disk byte layout. Deliberately exhaustive over
131    /// `CqlType` with no wildcard, so a NEW variant is a compile error here rather
132    /// than a silent blob at run time.
133    pub(in crate::storage::sstable::reader::parsing::row_decoder) fn cql_scalar_short_form(
134        ty: &CqlType,
135    ) -> Option<&'static str> {
136        Some(match ty {
137            CqlType::Boolean => "boolean",
138            CqlType::TinyInt => "tinyint",
139            CqlType::SmallInt => "smallint",
140            CqlType::Int => "int",
141            CqlType::BigInt => "bigint",
142            CqlType::Counter => "counter",
143            CqlType::Float => "float",
144            CqlType::Double => "double",
145            CqlType::Decimal => "decimal",
146            CqlType::Text => "text",
147            CqlType::Ascii => "ascii",
148            CqlType::Varchar => "varchar",
149            CqlType::Blob => "blob",
150            CqlType::Timestamp => "timestamp",
151            CqlType::Date => "date",
152            CqlType::Time => "time",
153            CqlType::Uuid => "uuid",
154            CqlType::TimeUuid => "timeuuid",
155            CqlType::Inet => "inet",
156            CqlType::Duration => "duration",
157            CqlType::Varint => "varint",
158            // Structured types are handled by `parse_typed_value`'s own arms and must
159            // never reach the scalar delegation.
160            CqlType::List(_)
161            | CqlType::Set(_)
162            | CqlType::Map(_, _)
163            | CqlType::Tuple(_)
164            | CqlType::Udt(_, _)
165            | CqlType::Frozen(_)
166            // #4114: a vector is decoded by `parse_typed_value`'s own arm from its
167            // declared dimension, so it must never reach the scalar delegation.
168            | CqlType::Vector(_, _)
169            | CqlType::Custom(_) => return None,
170        })
171    }
