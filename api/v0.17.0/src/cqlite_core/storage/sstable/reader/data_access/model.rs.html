15/// A single-column clustering-key restriction pushed down to a within-partition
16/// seek (Issue #954, Epic #951).
17///
18/// Carries the decoded clustering bound `Value`(s) for the FIRST clustering
19/// column (multi-column prefixes are a documented follow-up). Each bound is one
20/// clustering component; an empty `Vec` means that side is OPEN (unbounded). The
21/// `_inclusive` flags distinguish `>=`/`<=` (inclusive) from `>`/`<` (exclusive)
22/// — they are advisory for byte-window selection (block selection is
23/// over-inclusive by block granularity; the post-scan `evaluate_leaf` backstop
24/// applies the exact bound), so a slightly wider byte window never changes the
25/// final result.
26///
27/// `ck = v` is represented as `start == end == [v]`, both inclusive.
28#[derive(Debug, Clone)]
29pub struct ClusteringSlice {
30    /// Lower bound clustering component(s); empty = unbounded below.
31    pub start: Vec<Value>,
32    /// Whether the lower bound is inclusive (`>=`/`=`) vs exclusive (`>`).
33    pub start_inclusive: bool,
34    /// Upper bound clustering component(s); empty = unbounded above.
35    pub end: Vec<Value>,
36    /// Whether the upper bound is inclusive (`<=`/`=`) vs exclusive (`<`).
37    pub end_inclusive: bool,
38}

54/// Length of the all-`0xFF` sentinel used to represent an OPEN upper clustering
55/// bound (+∞) for byte-comparable block selection (Issue #954). Any separator in
56/// `Rows.db` is shorter or sorts below an all-`0xFF` run of this length, so it
57/// reliably selects through the last block. 64 bytes comfortably exceeds any
58/// realistic single-column clustering separator width.
59#[cfg(not(feature = "tombstones"))]
60pub(super) const MAX_OSS50_BOUND_SENTINEL_LEN: usize = 64;
61
62/// `ByteSource.LT_NEXT_COMPONENT` (cassandra-5.0.8
63/// `utils/bytecomparable/ByteSource.java:75`) — the byte-comparable terminator
64/// `ClusteringPrefix.Kind.INCL_START_BOUND` / `EXCL_END_BOUND` emit
65/// (`db/ClusteringPrefix.java:70-71`). It is smaller than `NEXT_COMPONENT`
66/// (`0x40`), so a bound carrying it sorts BELOW every clustering that extends it.
67#[cfg(not(feature = "tombstones"))]
68const OSS50_LT_NEXT_COMPONENT: u8 = 0x20;
69
70/// `ByteSource.GT_NEXT_COMPONENT` (`ByteSource.java:76`) — the terminator
71/// `ClusteringPrefix.Kind.INCL_END_BOUND` / `EXCL_START_BOUND` emit
72/// (`ClusteringPrefix.java:77-79`). It is LARGER than `NEXT_COMPONENT` (`0x40`)
73/// and than the variable-length escape bytes, so a bound carrying it sorts ABOVE
74/// every clustering that extends it.
75#[cfg(not(feature = "tombstones"))]
76const OSS50_GT_NEXT_COMPONENT: u8 = 0x60;

78/// Normalize a CQL [`ClusteringSlice`] into the `(physical_lower, physical_upper)`
79/// byte-comparable bounds that [`select_row_index_blocks_for_range`] consumes
80/// (issue #954 High-severity correctness fix).
81///
82/// [`select_row_index_blocks_for_range`] selects blocks purely in **physical**
83/// (on-disk, byte-comparable) order — the order `Rows.db` separators are stored
84/// in. The encoder ([`encode_clustering_bound_oss50_with_order`]) already inverts
85/// every byte of a DESC component (`ReversedType` / `ByteSource.invert`), so for a
86/// DESC first clustering column an ASCENDING byte order corresponds to a
87/// DESCENDING CQL value order. The consequence:
88///
89/// - **ASC** (`is_reversed[0] == false`): CQL `start` (the `>=`/`>` lower value
90///   bound) IS the physical-lower bound and CQL `end` (the `<=`/`<` upper value
91///   bound) IS the physical-upper bound. No swap.
92/// - **DESC** (`is_reversed[0] == true`): the CQL lower-value bound encodes to the
93///   physically GREATER bytes and the CQL upper-value bound to the physically
94///   SMALLER bytes, so the roles SWAP. A CQL `ck >= v` (lower) selects the rows
95///   with the largest values, which sit at the physical-LOW byte side through
96///   `enc(v)`; a CQL `ck < v` (upper) selects the physical-HIGH side. The open
97///   sentinels swap accordingly: an open CQL lower (`-∞` value) maps to physical
98///   `+∞` and an open CQL upper (`+∞` value) maps to physical `-∞`.
99///
100/// Block selection is over-inclusive by block granularity and the post-scan
101/// `evaluate_leaf` backstop re-applies the exact CQL bound by VALUE, so the
102/// inclusivity of each bound does not need separate handling here — the physical
103/// window only has to be a SUPERSET of the matching rows. (Swapping the roles is
104/// what guarantees the superset for DESC; the previous code built
105/// `[enc(lower), +∞]` for DESC, which excluded the matching low-byte rows and the
106/// backstop could not recover rows that were never decoded.)
107///
108/// Returns:
109/// - `Ok(Some((lower, upper)))` — usable physical bounds.
110/// - `Ok(None)` — a bound's type is not byte-comparable-encodable here, so the
111///   narrowing is unsafe and the caller must decode the whole partition.
112/// - `Err(_)` — never (kept `Result` for call-site symmetry); reserved.
113#[cfg(not(feature = "tombstones"))]
114pub(super) fn physical_byte_bounds_for_slice(
115    slice: &ClusteringSlice,
116    is_reversed: &[bool],
117) -> Result<Option<(Vec<u8>, Vec<u8>)>> {
118    use crate::storage::sstable::bti::encode_clustering_bound_oss50_with_order;
119
120    // The physical-low sentinel: empty sorts before every separator (-∞).
121    let neg_inf = Vec::<u8>::new();
122    // The physical-high sentinel: an all-0xFF run sorts after every separator (+∞).
123    let pos_inf = || vec![0xFFu8; MAX_OSS50_BOUND_SENTINEL_LEN];
124
125    // Encode a closed CQL bound to its physical byte-comparable form; `None`
126    // bubbles up as an un-encodable-bound fallback at the call site.
127    // The safe fallback is retained (an un-encodable bound just widens to the whole
128    // partition), but the encoder's error text is the ONLY diagnosis of WHICH
129    // clustering type is unsupported, so log it rather than dropping it silently.
130    let encode = |values: &[Value]| -> Option<Vec<u8>> {
131        match encode_clustering_bound_oss50_with_order(values, is_reversed) {
132            Ok(bytes) => Some(bytes),
133            Err(e) => {
134                tracing::debug!(
135                    error = %e,
136                    "OSS50 clustering bound not encodable; falling back to a full-partition \
137                     scan for this slice"
138                );
139                None
140            }
141        }
142    };
143
144    // The FIRST clustering column's order decides the value↔byte direction. A
145    // missing entry (no schema / fewer entries) is ascending, matching the encoder.
146    let first_desc = is_reversed.first().copied().unwrap_or(false);
147
148    // CQL lower bound (`>=`/`>` / equality lower) → its physical byte image.
149    let cql_lower_bytes = if slice.start.is_empty() {
150        None // open CQL lower (value -∞)
151    } else {
152        match encode(&slice.start) {
153            Some(b) => Some(b),
154            None => return Ok(None),
155        }
156    };
157    // CQL upper bound (`<=`/`<` / equality upper) → its physical byte image.
158    let cql_upper_bytes = if slice.end.is_empty() {
159        None // open CQL upper (value +∞)
160    } else {
161        match encode(&slice.end) {
162            Some(b) => Some(b),
163            None => return Ok(None),
164        }
165    };
166
167    let (mut phys_lower, mut phys_upper) = if first_desc {
168        // DESC: CQL upper-value bound → physical-lower bytes; CQL lower-value bound
169        // → physical-upper bytes. Open CQL upper → physical -∞; open CQL lower →
170        // physical +∞.
171        let lower = cql_upper_bytes.unwrap_or(neg_inf);
172        let upper = cql_lower_bytes.unwrap_or_else(pos_inf);
173        (lower, upper)
174    } else {
175        // ASC: CQL lower-value bound → physical-lower bytes; CQL upper-value bound
176        // → physical-upper bytes (the original mapping).
177        let lower = cql_lower_bytes.unwrap_or(neg_inf);
178        let upper = cql_upper_bytes.unwrap_or_else(pos_inf);
179        (lower, upper)
180    };
181
182    // Cassandra bound TERMINATORS (`ClusteringPrefix.Kind.asByteComparable`,
183    // cassandra-5.0.8 `db/ClusteringPrefix.java:68-81`).
184    //
185    // A `Rows.db` separator is the byte-comparable image of a full CLUSTERING (all
186    // components), while a slice bound may name only a PROPER PREFIX of the
187    // clustering key — e.g. `WHERE bucket = 'bo'` on `PRIMARY KEY (pk, bucket, seq)`.
188    // Cassandra distinguishes the two sides of such a prefix with a trailing marker:
189    // `LT_NEXT_COMPONENT` (0x20) sorts the bound BELOW every clustering that extends
190    // it, `GT_NEXT_COMPONENT` (0x60) sorts it ABOVE. Without one, a prefix bound is a
191    // bare prefix and therefore always sorts BELOW its extensions — which silently
192    // TRUNCATES an upper bound to the first row of the prefix's range (issue #3032:
193    // `bucket = 'bo'` returned 12 of 60 rows, the first row-index block only).
194    //
195    // The physical-LOW side always takes `LT` and the physical-HIGH side always takes
196    // `GT`. For the INCLUSIVE kinds (`=`, `>=`, `<=` — after the DESC role swap
197    // above) that is byte-exactly what Cassandra emits. For the EXCLUSIVE kinds it is
198    // deliberately one marker "wider" than Cassandra's, which can only ever ADD the
199    // boundary prefix's own rows to the window: block selection is over-inclusive by
200    // block granularity anyway and the post-scan `evaluate_leaf` backstop re-applies
201    // the exact CQL bound BY VALUE, so a wider window never changes the result — while
202    // a narrower one loses rows that were never decoded. Choosing by side rather than
203    // by inclusivity also keeps the DESC swap correct with no second mapping.
204    //
205    // Appending to the open sentinels is harmless: `-∞` is the empty vector (a lone
206    // 0x20 still sorts below every separator, all of which begin with the 0x40
207    // `NEXT_COMPONENT` byte, and at/above the stored `ByteComparable.EMPTY` block-0
208    // separator exactly as the bare empty bound did), and `+∞` is an all-0xFF run.
209    phys_lower.push(OSS50_LT_NEXT_COMPONENT);
210    phys_upper.push(OSS50_GT_NEXT_COMPONENT);
211
212    Ok(Some((phys_lower, phys_upper)))
213}

260/// Compare two table IDs, handling both qualified (keyspace.table) and unqualified (table) formats.
261///
262/// This function allows flexible matching:
263/// - "keyspace.table" matches "keyspace.table" (exact match)
264/// - "table" matches "keyspace.table" (unqualified matches qualified)
265/// - "keyspace.table" matches "table" (qualified matches unqualified)
266/// - "table" matches "table" (exact match)
267///
268/// This is necessary because:
269/// - Dataset mode SSTables store qualified table_ids (e.g., "test_basic.simple_table")
270/// - Queries can use either qualified ("test_basic.simple_table") or unqualified ("simple_table") names
271/// - Production SSTables may use unqualified table_ids
272pub(in crate::storage::sstable::reader) fn table_ids_match(
273    entry_table_id: &TableId,
274    query_table_id: &TableId,
275) -> bool {
276    let entry_name = entry_table_id.name();
277    let query_name = query_table_id.name();
278
279    // Fast path: exact match
280    if entry_name == query_name {
281        return true;
282    }
283
284    // Extract unqualified table names for comparison
285    let entry_unqualified = if let Some(dot_pos) = entry_name.rfind('.') {
286        &entry_name[dot_pos + 1..]
287    } else {
288        entry_name
289    };
290
291    let query_unqualified = if let Some(dot_pos) = query_name.rfind('.') {
292        &query_name[dot_pos + 1..]
293    } else {
294        query_name
295    };
296
297    // Match if unqualified names are the same
298    entry_unqualified == query_unqualified
299}

301/// Stricter table-id match used by the BTI point-lookup guard (issue #831 review).
302///
303/// [`table_ids_match`] matches on the unqualified table name, so it treats
304/// `ks_a.users` and `ks_b.users` as equal — fine for index lookups that are
305/// already scoped to one table, but too permissive as a defensive guard against
306/// a fully-qualified wrong-keyspace query. When BOTH ids are qualified
307/// (`keyspace.table`), this requires exact `keyspace.table` equality; it only
308/// falls back to the permissive unqualified match when one side lacks a
309/// keyspace (preserving qualified-vs-unqualified flexibility).
310pub(super) fn table_ids_match_strict(entry_table_id: &TableId, query_table_id: &TableId) -> bool {
311    let entry_qualified = entry_table_id.name().contains('.');
312    let query_qualified = query_table_id.name().contains('.');
313    if entry_qualified && query_qualified {
314        entry_table_id.name() == query_table_id.name()
315    } else {
316        table_ids_match(entry_table_id, query_table_id)
317    }
318}

320/// Seek-path table-id consistency check (issue #1284, hardened per #1284 review).
321///
322/// The single-partition SEEK (`bti_decompress_and_parse_target_all`) builds
323/// `entry_table_id` from THIS reader's AUTHORITATIVE serialization header
324/// (`{header.keyspace}.{header.table_name}`). `query_table_id` is the query's id.
325///
326/// `fully_qualified_match` records HOW the manager resolved this reader for the
327/// query (`resolve_reader_list`):
328///   - `true`  — the query's fully-qualified `keyspace.table` key matched the
329///     reader map EXACTLY (path-authoritative: this reader genuinely IS the
330///     queried table), or the query was unqualified.
331///   - `false` — the query was fully qualified but reached this reader via the
332///     bare/unqualified NAME fallback (its keyspace did NOT match any map key).
333///
334/// [`table_ids_match_strict`] additionally requires the *keyspace* to match when
335/// BOTH ids are fully qualified — the correct `get()` defensive guard (#831). For
336/// a seek it falsely REJECTS every row when the reader's HEADER keyspace differs
337/// from the query keyspace even though resolution was an exact FQ match (e.g. a
338/// writer-produced header, or a header whose embedded keyspace is not the
339/// path-derived one), silently zeroing the result and forcing a full-scan
340/// fallback (#1284).
341///
342/// The relaxation (accept on a consistent TABLE NAME despite a header-keyspace
343/// divergence) is therefore gated on `fully_qualified_match`:
344///   - Exact FQ match (or unqualified query): the reader is authoritatively the
345///     target table, so a header-keyspace divergence is benign — accept when the
346///     table names agree, reject a different table name (the #831 wrong-table
347///     guard survives via [`table_ids_match`]). This is #1284's actual goal.
348///   - FQ query reached via the bare-name FALLBACK: the keyspace genuinely did
349///     NOT match, so a relaxed name-only check could surface rows from ANOTHER
350///     keyspace whose table name collides — keep STRICT keyspace matching
351///     ([`table_ids_match_strict`]) and REJECT the wrong-keyspace rows.
352///
353/// This is no-heuristic: it relies only on the authoritative header identity, the
354/// query id, and the authoritative resolution mode — never on guessing.
355///
356/// Used by BOTH the single-partition SEEK (`bti_collect_partition_rows`,
357/// `not(tombstones)` only) and the `get()` POINT-LOOKUP decoder
358/// (`bti_decompress_and_parse_target`, all feature builds, issue #1321). The
359/// point-lookup caller exists in every build, so this helper is NOT gated.
360pub(super) fn table_header_consistent_for_seek(
361    entry_table_id: &TableId,
362    query_table_id: &TableId,
363    fully_qualified_match: bool,
364) -> bool {
365    // Only relax to the name-only consistency check for a fully-qualified query
366    // when resolution was an EXACT fully-qualified match (the reader is
367    // path-authoritatively the queried table). A fully-qualified query that
368    // reached this reader via the bare-name fallback keeps STRICT keyspace
369    // matching, so it can never return rows from a different keyspace whose table
370    // name collides (#1284 review correctness fix).
371    if fully_qualified_match && query_table_id.name().contains('.') {
372        table_ids_match(entry_table_id, query_table_id)
373    } else {
374        table_ids_match_strict(entry_table_id, query_table_id)
375    }
376}

378/// What the BTI point-lookup loop may do with a partition-parse `Err` (issue
379/// #3721, roborev job 80) — the sibling of [`bti_lookup_step`] for the state that
380/// decision cannot see: the parse having FAILED.
381///
382/// The loop's catch-all conflated three states — an `Err`, the emit closure never
383/// firing, and a partition decoding to a FOREIGN key — and answered all three with
384/// the straddle retry or, out of bytes, with absence. Only the first can be a
385/// decode failure, and **a decode failure is not absence**: a BTI trie hit is NOT
386/// followed by a scan, so `Ok(None)` tells the caller the key does not exist and
387/// nothing downstream can tell that apart from a genuine miss. The comment this
388/// replaced said so out loud — "could not be parsed" is not "absent".
389///
390/// # The two classes that cannot mean absence
391///
392/// A VARIANT match, never a message test (issue #28); both are `ErrorCategory::Data`:
393///
394/// * [`Error::ColumnDecode`] — a framed row whose column would have been dropped;
395/// * [`Error::Corruption`] — the range-marker refusals reach the same parser
396///   (`parse_block_emit` delegates to `parse_block_emit_windowed`, whose marker
397///   arms propagate since roborev job 78), as does any other structural refusal.
398///
399/// Everything ELSE keeps the caller's prior behaviour, deliberately. The one that
400/// matters is [`Error::Schema`]: a reader with no schema for the queried table
401/// CANNOT serve the key, and that soft-miss is what lets the caller try the next
402/// reader — the same distinction `super::sequential::is_parse_soft_miss` draws on
403/// the BIG point path. Propagating it would hard-fail a multi-reader `get()` that
404/// must fall back.
405///
406/// # Why this takes `more_bytes_may_arrive`
407///
408/// The #1572 chunk-straddle retry is why this is not a bare `matches!`. A row cut
409/// by the window boundary reports `row body exhausted`, which is ITSELF an
410/// `Error::ColumnDecode`, so refusing on sight would hard-fail a PRESENT key whose
411/// body merely straddles a chunk. While chunks remain the failure is therefore only
412/// REMEMBERED and the retry is UNCHANGED; the memory is answered where the bytes
413/// run out, by [`point_read_absence_or_remembered`].
414///
415/// Returns `Err` only when the caller has every byte it will ever have.
416pub(super) fn point_read_remember_or_bail(
417    parse_result: crate::Result<()>,
418    more_bytes_may_arrive: bool,
419    undecodable: &mut Option<crate::Error>,
420) -> crate::Result<()> {
421    let Err(e) = parse_result else {
422        return Ok(());
423    };
424    if !matches!(
425        e,
426        crate::Error::ColumnDecode { .. } | crate::Error::Corruption(_)
427    ) {
428        return Ok(());
429    }
430    if more_bytes_may_arrive {
431        *undecodable = Some(e);
432        return Ok(());
433    }
434    Err(e)
435}

437/// The verdict where a point read runs out of bytes: propagate a REMEMBERED
438/// undecodable-partition failure rather than reporting the key ABSENT (issue #3721,
439/// roborev job 80). With nothing remembered the caller's prior `Ok(None)` stands,
440/// so a genuinely unparseable or absent tail is unchanged.
441pub(super) fn point_read_absence_or_remembered<T>(
442    undecodable: &mut Option<crate::Error>,
443) -> crate::Result<Option<T>> {
444    match undecodable.take() {
445        Some(e) => Err(e),
446        None => Ok(None),
447    }
448}

450/// Per-iteration decision for the BTI chunk-targeted point-lookup loop.
451#[derive(Debug, PartialEq, Eq)]
452pub(super) enum BtiLookupStep {
453    /// The full partition-key prefix is buffered and matches the queried key —
454    /// parse the partition.
455    Parse,
456    /// The header/key prefix straddles a chunk boundary and is not yet fully
457    /// buffered — read the next chunk before parsing (chunk-targeted path only).
458    PullNextChunk,
459    /// Treat the partition as absent: either the buffered key prefix does not
460    /// match, or a whole-section window is structurally too short to grow.
461    Absent,
462}

464/// Decide what the BTI point-lookup loop should do for the current window state.
465///
466/// Pure so the chunk-straddle control flow is unit-testable without a real
467/// multi-chunk BTI fixture (DataOffset partitions are narrow and fit in one
468/// chunk, so a boundary-straddling header cannot be produced by the available
469/// fixtures). Crucially, when the key prefix is not yet buffered on the
470/// chunk-targeted path this returns [`BtiLookupStep::PullNextChunk`] — it must
471/// NOT lead to parsing a truncated header (issue #831 review).
472pub(super) fn bti_lookup_step(
473    key_prefix_available: bool,
474    key_matches: bool,
475    chunk_targeted: bool,
476) -> BtiLookupStep {
477    if key_prefix_available {
478        if key_matches {
479            BtiLookupStep::Parse
480        } else {
481            BtiLookupStep::Absent
482        }
483    } else if chunk_targeted {
484        BtiLookupStep::PullNextChunk
485    } else {
486        BtiLookupStep::Absent
487    }
488}

490/// Sort a result slice in ascending Cassandra token order.
491///
492/// The authoritative ordering for SSTable partitions is ascending Murmur3 token, with
493/// equal-token ties broken by raw key bytes (lexicographic). This matches the on-disk
494/// physical order (spec §5, Appendix B §313) and the write engine's `DecoratedKey::cmp`.
495///
496/// Computes each key's token once to avoid O(n log n) recomputation inside the comparator.
497pub(super) fn sort_by_token_order(results: &mut Vec<(RowKey, ScanRow)>) {
498    // Map to (token, RowKey, ScanRow), sort, then reassemble.
499    let mut tagged: Vec<(i64, RowKey, ScanRow)> = results
500        .drain(..)
501        .map(|(k, v)| {
502            let t = cassandra_murmur3_token(k.as_bytes());
503            (t, k, v)
504        })
505        .collect();
506    tagged.sort_by(|a, b| a.0.cmp(&b.0).then_with(|| a.1.cmp(&b.1)));
507    results.extend(tagged.into_iter().map(|(_, k, v)| (k, v)));
508}

510/// Sort `(RowKey, ScanRow, CellMeta)` triples by Cassandra Murmur3 token order.
511pub(super) fn sort_by_token_order_with_meta(
512    results: &mut Vec<(
513        RowKey,
514        ScanRow,
515        std::collections::HashMap<String, CellWriteMetadata>,
516    )>,
517) {
518    let mut tagged: Vec<(
519        i64,
520        RowKey,
521        ScanRow,
522        std::collections::HashMap<String, CellWriteMetadata>,
523    )> = results
524        .drain(..)
525        .map(|(k, v, m)| {
526            let t = cassandra_murmur3_token(k.as_bytes());
527            (t, k, v, m)
528        })
529        .collect();
530    tagged.sort_by(|a, b| a.0.cmp(&b.0).then_with(|| a.1.cmp(&b.1)));
531    results.extend(tagged.into_iter().map(|(_, k, v, m)| (k, v, m)));
532}
