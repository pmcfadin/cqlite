1//! CRC-validated compressed offset-read window (issue #1773).
2//!
3//! The compressed offset-read path
4//! ([`read_value_at_offset`](super::super::SSTableReader::read_value_at_offset) →
5//! `get_cached_data`, and the Summary-guided compressed scan walk in
6//! `summary_scan.rs`) used to read `size` raw bytes at an offset and LZ4-decompress
7//! them WITHOUT validating the trailing 4-byte inline per-chunk CRC32 — re-introducing
8//! the #1411 CRC bypass on a latent path (unreachable today via `get`, but live the
9//! moment `find_entry` hits for a compressed table). This module carries the helper
10//! that routes that case through the shared CRC-enforcing chunk reader.
11//!
12//! The positional plane is the CALLER's, never hardcoded here (issue #2876): this
13//! helper serves BOTH read intents, so each caller passes the source its intent
14//! selects. The scan-shaped walks (`summary_scan.rs`'s Summary-guided partition
15//! walk, `full_index_scan.rs` / `full_index_stream.rs`'s full-`Index.db`
16//! enumeration, and the windowed scan) pass the reader's UNADVISED
17//! `scan_positional_source`, because they read Data.db largely sequentially and the
18//! advised mapping's readahead suppression is exactly backwards for them (the
19//! #2210 × #1940 cross-path regression). A genuine point lookup — `get_cached_data`
20//! reached from `read_value_at_offset` — passes the dedicated `MADV_RANDOM`
21//! `point_source`, keeping the advice issue #2210 gave it. (`bti_point.rs` /
22//! `big_promoted.rs` read `point_source` directly and do not route through here.)
23
24use std::sync::atomic::Ordering;
25
26use crate::storage::sstable::reader::SSTableReader;
27use crate::{Error, Result};
28
29impl SSTableReader {
30    /// Decompress the uncompressed byte window `[block_offset, block_offset + size)`
31    /// out of a COMPRESSED Data.db, validating each covering chunk's authoritative
32    /// inline per-chunk CRC32 BEFORE decompression (issue #1773).
33    ///
34    /// Reuses [`read_compressed_chunk_at`] — the positional, CRC-enforcing chunk
35    /// reader also used by the BTI point-read path — so the CRC-then-decompress
36    /// ordering (guardrail #1411) is honored on this path too: a bit-flipped chunk
37    /// surfaces the SAME typed `Error::InvalidFormat` (naming the chunk index +
38    /// offset) that scan / `scan_for_key` return, never garbage. No heuristics: the
39    /// authoritative CRC trailer is checked, never inferred.
40    ///
41    /// `source` is the positional plane the CALLER's read intent selects (issue
42    /// #2876) — see the module doc. It is a parameter rather than a field read so
43    /// that this one helper can serve the scan walks and the point offset read
44    /// without either losing its intended mapping advice.
45    ///
46    /// [`read_compressed_chunk_at`]: super::super::block_io::read_compressed_chunk_at
47    pub(in crate::storage::sstable::reader) async fn read_compressed_offset_window(
48        &self,
49        source: &dyn super::super::read_at::ReadAt,
50        comp_info: &crate::storage::sstable::compression_info::CompressionInfo,
51        block_offset: u64,
52        size: u32,
53    ) -> Result<Vec<u8>> {
54        use crate::storage::sstable::compression::Compression;
55
56        let compression = self
57            .compression_reader
58            .as_ref()
59            .map(|r| Compression::new(*r.algorithm()))
60            .transpose()?;
61
62        read_compressed_offset_window_impl(
63            source,
64            comp_info,
65            compression.as_ref(),
66            self.stats.file_size,
67            block_offset,
68            size,
69        )
70    }
71}

73/// Decompression core of [`SSTableReader::read_compressed_offset_window`], factored
74/// out of the reader method so it can be exercised directly against a hand-built
75/// [`CompressionInfo`] + `ReadAt` source (no reader/SQL roundtrip) — see the
76/// `max_compressed_length == 0` regression test below.
77///
78/// [`CompressionInfo`]: crate::storage::sstable::compression_info::CompressionInfo
79pub(super) fn read_compressed_offset_window_impl(
80    positional_source: &dyn super::super::read_at::ReadAt,
81    comp_info: &crate::storage::sstable::compression_info::CompressionInfo,
82    compression: Option<&crate::storage::sstable::compression::Compression>,
83    file_size: u64,
84    block_offset: u64,
85    size: u32,
86) -> Result<Vec<u8>> {
87    use super::super::block_io;
88
89    let chunk_length = comp_info.chunk_length as usize;
90    if chunk_length == 0 {
91        return Err(Error::corruption(
92            "CompressionInfo chunk_length is zero; cannot map a Data.db offset to a \
93             compressed chunk"
94                .to_string(),
95        ));
96    }
97
98    // Fail CLOSED on a corrupt `CompressionInfo` whose `max_compressed_length == 0`
99    // (issue #2524, mirroring the #1869 fix in the sibling `compressed_partition_window`):
100    // otherwise the raw-chunk fallback below (`compressed.len() >= max_compressed_length`)
101    // is ALWAYS true, so EVERY still-LZ4-compressed chunk would be returned verbatim as
102    // "raw/incompressible" plaintext — never decompressed — and the inline CRC32 (computed
103    // over the genuine on-disk bytes) would still pass, silently handing back garbage rows.
104    // A valid Cassandra `CompressionInfo` never records a zero `max_compressed_length`
105    // (it is `i32::MAX` when minCompressRatio=0, the default; CompressionParams.java:186-189).
106    let max_compressed_length = comp_info.max_compressed_length as usize;
107    if max_compressed_length == 0 {
108        return Err(Error::corruption(
109            "compressed offset read: CompressionInfo max_compressed_length is zero; \
110             cannot distinguish compressed chunks from raw/incompressible ones"
111                .to_string(),
112        ));
113    }
114
115    let start = block_offset as usize;
116    let end = start.checked_add(size as usize).ok_or_else(|| {
117        Error::corruption(format!(
118            "compressed offset read range [{start}, +{size}) overflows"
119        ))
120    })?;
121    // `size >= 1` (callers reject size == 0), so `end > start` and `end - 1`
122    // never underflows; `first_chunk <= last_chunk`.
123    let first_chunk = start / chunk_length;
124    let last_chunk = (end - 1) / chunk_length;
125
126    let mut assembled =
127        Vec::with_capacity(last_chunk.saturating_sub(first_chunk).saturating_add(1) * chunk_length);
128    for chunk_idx in first_chunk..=last_chunk {
129        // CRC-validated compressed bytes (guardrail #1411): the trailing inline
130        // per-chunk CRC32 is checked HERE, before decompression. header_offset is
131        // 0 for nb/BTI (CompressionInfo chunk offsets are absolute from Data.db
132        // byte 0), matching the cursor and BTI point-read paths (issue #1573 C2).
133        //
134        // Fail CLOSED (issue #1773 roborev): for a VALID offset+size every chunk
135        // in `first_chunk..=last_chunk` exists, so a `None` (EOF) here means the
136        // requested range does not actually fit in this compressed Data.db — a
137        // corrupt/out-of-range offset or size. Returning the partial `assembled`
138        // bytes as `Ok` would silently truncate; this must surface as a typed,
139        // non-recoverable corruption error instead.
140        // Issue #2819: attribute the synchronous body-chunk page-in (positional
141        // read + CRC) to the `stream_cold_fault` sub-phase on the flight
142        // per-request sink (no-op when no sink is installed — every non-flight
143        // caller). This wraps ONLY the read, on the per-SSTable scan (producer)
144        // thread; it shares no code interval with the egress `stream_grpc_write`
145        // scope, which is measured on the merge/egress thread in
146        // `cqlite-flight`'s `ChannelSink::emit` — so a slow client's send-park can
147        // never inflate cold-fault.
148        let chunk = crate::observability::stream_subphase::timed(
149            crate::observability::StreamSubPhase::ColdFault,
150            || {
151                block_io::read_compressed_chunk_at(
152                    positional_source,
153                    comp_info,
154                    chunk_idx,
155                    file_size,
156                    0,
157                )
158            },
159        )?;
160        let Some(compressed) = chunk else {
161            return Err(Error::corruption(format!(
162                "compressed offset read requires chunk {chunk_idx} past EOF for range \
163                 [{start}, {end}) — corrupt or out-of-range offset/size"
164            )));
165        };
166
167        // Incompressible-chunk fallback (Bug #639, epic #970): Cassandra stores a
168        // chunk RAW (not compressed) when its compressed length would meet or
169        // exceed `max_compressed_length`; those bytes are already plaintext.
170        // Authority: CompressedSequentialWriter.java:160-177.
171        let decompressed = if compressed.len() >= max_compressed_length {
172            // Already plaintext, but still Data.db bytes this read materialised, so
173            // they are counted at the plane's one raw-chunk boundary (issue #1701 F3)
174            // instead of being handed back invisibly.
175            super::super::chunk_source::count_raw_chunk(
176                &compressed,
177                compression.map(|c| c.algorithm()),
178            );
179            compressed
180        } else if let Some(compression) = compression {
181            // Single decode plane (issue #1598, G2): the actual decompress call
182            // resolves inside `ChunkSource::decompress_only`, so this module holds
183            // zero query-path decompress call sites (the architecture test proves
184            // exactly one such module). CRC is already validated above by
185            // `read_compressed_chunk_at` (guardrail #1411), so we never decode bytes
186            // that failed their inline CRC32.
187            // Issue #2819: attribute the LZ4 decompress to the `stream_decompress`
188            // sub-phase (no-op when no sink is installed). Reached only for a
189            // genuinely compressed chunk (past the incompressible-raw fallback
190            // above), so an uncompressed table records NO `stream_decompress`.
191            let out = crate::observability::stream_subphase::timed(
192                crate::observability::StreamSubPhase::Decompress,
193                || {
194                    super::super::chunk_source::ChunkSource::decompress_only(
195                        Some(compression),
196                        compressed,
197                    )
198                },
199            )?;
200            super::model::DECOMPRESS_CALLS.fetch_add(1, Ordering::Relaxed);
201            out
202        } else {
203            // No compression reader (should not happen: this path is only reached
204            // for a compressed Data.db, which always carries one). Mirror
205            // `stitch_all_chunks`' warn! (issue #2167) so an unexpected reach —
206            // returning still-compressed bytes as if plaintext — is visible in logs.
207            tracing::warn!(
208                "read_compressed_offset_window: No compression reader, using raw chunk data"
209            );
210            compressed
211        };
212        assembled.extend_from_slice(&decompressed);
213    }
214
215    // Slice the requested window out of the assembled uncompressed chunk bytes.
216    //
217    // Fail CLOSED (issue #1773 roborev): for a VALID offset+size the assembled
218    // chunks fully cover `[start, end)`. If they do not — a short final chunk, or
219    // `within_start` itself past what decoded — the offset/size is corrupt or
220    // out-of-range; return a typed corruption error rather than truncating to
221    // `Ok(partial)` / `Ok(empty)`.
222    let window_base = first_chunk * chunk_length;
223    let within_start = start - window_base;
224    let requested_end = end - window_base;
225    if requested_end > assembled.len() {
226        return Err(Error::corruption(format!(
227            "compressed offset read range [{start}, {end}) is short by {} bytes after \
228             decompressing chunks {first_chunk}..={last_chunk} (assembled {} bytes) — \
229             corrupt or out-of-range offset/size",
230            requested_end - assembled.len(),
231            assembled.len()
232        )));
233    }
234    Ok(assembled[within_start..requested_end].to_vec())
235}

237#[cfg(test)]
238mod tests {
239    use crate::storage::sstable::compression_info::CompressionInfo;
240    use crate::storage::sstable::reader::read_at::ReadAt;
241    use crate::Error;
242
243    /// Minimal `ReadAt` double. The `max_compressed_length == 0` guard fires BEFORE
244    /// any chunk read, so `read_at` is never reached — it exists only to satisfy the
245    /// signature (and panics if the guard ever regresses to touch the source).
246    struct NeverReadAt;
247    impl ReadAt for NeverReadAt {
248        fn read_at(&self, _offset: u64, _buf: &mut [u8]) -> crate::Result<usize> {
249            panic!(
250                "read_at must not be reached: the max_compressed_length==0 guard fails closed \
251                    before any chunk I/O"
252            );
253        }
254        fn len(&self) -> u64 {
255            0
256        }
257    }
258
259    /// A hand-built `CompressionInfo` with `max_compressed_length` corrupted to `0`.
260    /// One 64-byte chunk; the chunk_offsets/data_length are immaterial because the
261    /// guard rejects the metadata before mapping any offset to a chunk.
262    fn comp_info_zero_max() -> CompressionInfo {
263        CompressionInfo {
264            algorithm: "LZ4Compressor".to_string(),
265            option_pairs: Vec::new(),
266            chunk_length: 64,
267            max_compressed_length: 0, // corrupt: a valid CompressionInfo never records 0
268            data_length: 64,
269            chunk_offsets: vec![0],
270        }
271    }
272
273    /// Issue #2524 — a corrupt/malformed `CompressionInfo` whose `max_compressed_length
274    /// == 0` MUST fail closed with a typed `Error::Corruption`, never silently return
275    /// still-compressed bytes as plaintext. Pre-fix, the raw-chunk fallback test
276    /// `compressed.len() >= max_compressed_length` was `>= 0` — ALWAYS true — so every
277    /// still-LZ4-compressed chunk was returned verbatim as "raw" plaintext, and the
278    /// inline CRC32 (over the genuine on-disk bytes) still matched, failing OPEN.
279    /// Mirrors the #1869 fix + test for the sibling `compressed_partition_window`.
280    #[test]
281    fn zero_max_compressed_length_fails_closed() {
282        let ci = comp_info_zero_max();
283        let src = NeverReadAt;
284
285        let err = super::read_compressed_offset_window_impl(&src, &ci, None, 0, 0, 16).expect_err(
286            "CompressionInfo.max_compressed_length == 0 must fail closed with a typed \
287                 corruption error, not silently return still-compressed bytes as plaintext",
288        );
289        match err {
290            Error::Corruption(m) => assert!(
291                m.contains("max_compressed_length is zero"),
292                "unexpected corruption text: {m}"
293            ),
294            other => panic!("expected Corruption(max_compressed_length zero), got {other:?}"),
295        }
296    }
