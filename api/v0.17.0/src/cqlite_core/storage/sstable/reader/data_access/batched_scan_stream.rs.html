1//! The BATCHED streaming scan (`scan_stream_batched`, issue #1592 Epic F/F2) — the
2//! additive companion to the per-row `scan_stream` whose channel item is a `Vec`
3//! BATCH of `(RowKey, ScanRow)` entries rather than a single entry.
4//!
5//! Split out of `sequential.rs` (campsite rule, epic #1116) alongside its per-row
6//! sibling `per_row_scan_stream.rs`: that file is well over the ~800-line source
7//! target, and the batched streaming scan is the same self-contained responsibility
8//! — one spawned producer task per reader, feeding a bounded channel, with the
9//! format branch (BTI trie walk / windowed stitch / block-by-block) chosen inside it.
10//!
11//! # Issue #3109: the BTI (`da`) dispatch
12//!
13//! This surface shipped WITHOUT the `bti_partitions_db.is_some()` dispatch that
14//! `scan` and `run_scan_stream` have, so a `da` reader fell straight into the
15//! non-stitching block loop and decoded through the `V5UncompressedOA` state
16//! machine — which honours neither `read_shadowing` nor a caller-pinned read clock.
17//! A BTI table streamed here was therefore read UNSHADOWED. Both streaming surfaces
18//! now share ONE dispatch, [`SSTableReader::stream_bti_scan`], so a third divergent
19//! copy of this decision cannot drift again (the #1577 class).
20//!
21//! Included via `#[path = "batched_scan_stream.rs"] pub(super) mod batched_scan_stream;`
22//! in [`super`], so it shares `sequential.rs`'s imports through `use super::*`.
23//!
24//! # Why the module is `pub(super)` (issue #3384)
25//!
26//! [`batched_channel_capacity`] is the SINGLE definition of this surface's channel
27//! sizing, and `summary_scan::query_rows` derives its exported read-ahead bound
28//! ([`QUERY_ROWS_MAX_READ_AHEAD`](crate::storage::sstable::reader::QUERY_ROWS_MAX_READ_AHEAD))
29//! from it rather than restating the arithmetic. `pub(super)` is the narrowest
30//! visibility that reaches that sibling; the prose lives here rather than beside
31//! the `mod` declaration because `sequential.rs` is 1319 lines, already over the
32//! campsite threshold (epic #1116), and this file is not.
33

36/// Publishes detached-scan completion on drop — see the spawn site (issue #3384).
37struct ScanTaskDone;
38
39impl Drop for ScanTaskDone {
40    fn drop(&mut self) {
41        crate::storage::read_path_probe::mark_batched_scan_finished();
42    }
43}

45/// Batches the public BATCHED-scan channel below holds for a given `buffer_size`.
46///
47/// Sizing the channel to `ceil(buffer_size / BATCH_EMIT_ROWS)` batches keeps its
48/// resident-row budget comparable to the per-row surface's `buffer_size` rather
49/// than `buffer_size * BATCH_EMIT_ROWS`.
50///
51/// A `const fn` with ONE definition because the query-row stream's exported
52/// read-ahead bound
53/// ([`QUERY_ROWS_MAX_READ_AHEAD`](crate::storage::sstable::reader::QUERY_ROWS_MAX_READ_AHEAD))
54/// is derived from it. A second copy of this arithmetic could drift from the
55/// channel the constructor actually builds — which is precisely how a documented
56/// read-ahead bound becomes silently wrong (issue #3384).
57pub(crate) const fn batched_channel_capacity(buffer_size: usize) -> usize {
58    let cap = buffer_size.div_ceil(BATCH_EMIT_ROWS);
59    if cap == 0 {
60        1
61    } else {
62        cap
63    }
64}

66impl SSTableReader {
67    /// Batched streaming scan (issue #1592, Epic F/F2): the additive companion to
68    /// [`scan_stream`](Self::scan_stream) whose channel item is a `Vec` BATCH of
69    /// `(RowKey, ScanRow)` entries rather than a single entry. Forwarding one batch
70    /// per channel send collapses the one-async-wake-per-row cost the internal
71    /// windowed pipeline (issue #1143) was designed to amortize but the public
72    /// forwarder then re-flattened away.
73    ///
74    /// Order and content are identical to [`scan_stream`](Self::scan_stream):
75    /// flattening the batches yields exactly the per-row stream. Backpressure is
76    /// preserved — the channel is bounded (in BATCHES) and every send observes it.
77    pub fn scan_stream_batched(
78        self: std::sync::Arc<Self>,
79        table_id: TableId,
80        start_key: Option<RowKey>,
81        end_key: Option<RowKey>,
82        schema: Option<crate::schema::TableSchema>,
83        buffer_size: usize,
84    ) -> BatchedScanStream {
85        self.scan_stream_batched_admitted(
86            table_id,
87            start_key,
88            end_key,
89            schema,
90            buffer_size,
91            ScanAdmission::Acquire,
92            None,
93            crate::storage::sstable::reader::ScanErrorReporting::TopLevel,
94        )
95    }

97    /// [`scan_stream_batched`](Self::scan_stream_batched) with an explicit
98    /// admission context (issue #1594), mirroring
99    /// [`scan_stream_admitted`](Self::scan_stream_admitted).
100    /// `now_secs` (issue #3058): a caller-pinned read-time TTL clock, threaded to
101    /// the read-shadowing decoder so a caller that already captured ONE
102    /// reconciliation instant for the request (the Flight single-source fast
103    /// path) expires TTL cells at exactly that instant instead of an ambient
104    /// wall-clock sample. `None` keeps the ambient sample.
105    pub(crate) fn scan_stream_batched_admitted(
106        self: std::sync::Arc<Self>,
107        table_id: TableId,
108        start_key: Option<RowKey>,
109        end_key: Option<RowKey>,
110        schema: Option<crate::schema::TableSchema>,
111        buffer_size: usize,
112        admission: ScanAdmission,
113        now_secs: Option<i64>,
114        // Issue #1704: whether anything ENCLOSES this stream. NOT derivable from
115        // `admission` — `query_rows` consumes an `Exempt` stream directly.
116        reporting: crate::storage::sstable::reader::ScanErrorReporting,
117    ) -> BatchedScanStream {
118        // The public channel carries BATCHES (see `batched_channel_capacity`,
119        // which is the SINGLE definition of this sizing — the query-row stream's
120        // exported read-ahead bound is derived from the same `const fn`).
121        let cap = batched_channel_capacity(buffer_size);
122        let (tx, rx) = mpsc::channel(cap);
123        // Read-metric grain (issue #1701): identical rule to the per-row surface —
124        // an `Acquire` scan is the top-level read OPERATION and is measured with
125        // this reader's format label; an `Exempt` sub-scan is not (its merge is).
126        // Sampled before `self` moves into the task below.
127        // START the meter BEFORE the spawn (issue #1701, roborev round 7) — see the
128        // per-row sibling: constructed after it, the producer could run before timing
129        // began. The format label is still sampled before `self` moves into the task.
130        let meter = match admission {
131            ScanAdmission::Acquire => Some(crate::observability::read_metrics::ReadOpMeter::start(
132                Some(self.sstable_format_label()),
133            )),
134            ScanAdmission::Exempt => None,
135        };
136
137        // Read-PHASE accumulator (issue #1707) — see the per-row sibling for why it
138        // is propagated explicitly rather than through a thread-local.
139        let phase_sink = meter.as_ref().and_then(|m| m.phase_sink());
140
141        // CAUSAL completion signal for THIS detached task (issue #3384, roborev).
142        // `BatchedScanStream` is dropped without joining its task, so a consumer that
143        // stopped reading cannot otherwise tell "the scan finished" from "the scan is
144        // descheduled between two increments".
145        //
146        // A DROP guard, not a statement at the end of the block, because the common exit
147        // here is CANCELLATION, not completion: the query-row producer owns a
148        // `current_thread` runtime and drops it as it returns, cancelling this future at
149        // its next await — a trailing statement would never run. Measured: it never fired.
150        //
151        // Constructed BEFORE `tokio::spawn` and moved in (roborev): a future dropped
152        // before its FIRST POLL — a runtime shut down between spawn and schedule — would
153        // otherwise never construct the guard, so nothing would publish and every waiter
154        // would time out. Owning it outside the async block makes the guard's lifetime
155        // the TASK's, not the body's.
156        let done = ScanTaskDone;
157        let task = tokio::spawn(async move {
158            let _done = done;
159            if let Err(e) = self
160                .run_scan_stream_batched(
161                    table_id,
162                    start_key,
163                    end_key,
164                    schema,
165                    tx.clone(),
166                    admission,
167                    now_secs,
168                    phase_sink,
169                )
170                .await
171            {
172                let _ = tx.send(Err(e)).await;
173            }
174        });
175        match meter {
176            Some(meter) => BatchedScanStream::new_measured_batches(rx, task, meter),
177            None => BatchedScanStream::unmetered_as(rx, task, reporting),
178        }
179    }

181    async fn run_scan_stream_batched(
182        self: std::sync::Arc<Self>,
183        table_id: TableId,
184        start_key: Option<RowKey>,
185        end_key: Option<RowKey>,
186        schema: Option<crate::schema::TableSchema>,
187        tx: mpsc::Sender<Result<Vec<(RowKey, ScanRow)>>>,
188        admission: ScanAdmission,
189        // Issue #3058: caller-pinned read-time TTL clock (`None` = ambient).
190        now_secs: Option<i64>,
191        // This scan operation's read-phase accumulator (issue #1707), or `None`.
192        phase_sink: Option<std::sync::Arc<crate::observability::ReadPhaseTimings>>,
193    ) -> Result<()> {
194        // Admission control (issue #1594, F4): identical discipline to the per-row
195        // `run_scan_stream` — one permit per top-level scan operation, held via RAII.
196        let _admission = match admission {
197            ScanAdmission::Acquire => Some(scan_admission::admit().await),
198            ScanAdmission::Exempt => None,
199        };
200        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
201
202        let cursor = self.open_batched_scan_cursor().await?;
203
204        // Issue #3109: BTI (`da`) readers take the SAME shared dispatch the per-row
205        // surface takes — the authoritative trie walk, with read shadowing and the
206        // caller's pinned clock applied — instead of the block-by-block decoder
207        // below, whose `V5UncompressedOA` state machine drops BOTH. Gated on the
208        // SAME `bti_partitions_db.is_some()` condition `scan` and `run_scan_stream`
209        // use, so the three surfaces cannot disagree about which readers are BTI.
210        // See [`SSTableReader::stream_bti_scan`] for the full rationale.
211        //
212        // BEHAVIOR DELTA, stated so it is not a surprise: this branch does NOT apply
213        // the `table_ids_match(&entry_table_id, &table_id)` filter the block loop
214        // below applies — the trie walk returns no per-entry `TableId` to match on.
215        // Safe and consistent with the sibling BTI surfaces (`scan`,
216        // `sequential_scan`, `run_scan_stream`), which all skip it for the same
217        // reason: every entry in this walk comes from the single SSTable being
218        // scanned, so the filter can only ever reject rows that DO belong to it
219        // (the parser tags entries from the SSTable header, which may carry a bare
220        // or default keyspace/table name rather than the query's qualified form).
221        //
222        // Placed AFTER `open_batched_scan_cursor` deliberately: that call is this
223        // task's single, format-branch-independent fault checkpoint (issue #3106),
224        // and moving the BTI return above it would make the checkpoint silently
225        // unreachable for `da` readers — the exact "armable but never armed" hazard
226        // its doc comment warns about. The cursor is then unused on this branch
227        // (`bti_scan_with_metadata` mints its own, issue #815); one open(2) is the
228        // price of keeping the checkpoint branch-agnostic.
229        if self.bti_partitions_db.is_some() {
230            drop(cursor);
231            return self
232                .stream_bti_scan(
233                    start_key.as_ref(),
234                    end_key.as_ref(),
235                    schema.as_ref(),
236                    now_secs,
237                    &WindowedOut::Batched(tx.clone()),
238                )
239                .await;
240        }
241
242        let header_size = self.calculate_header_size();
243        {
244            let mut file_guard = cursor.file.lock().await;
245            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
246        }
247
248        if self.requires_chunk_stitching() {
249            // Affirmative branch marker (issue #3384): the detached-scan completion
250            // signal does NOT cover this branch's `spawn_blocking` decode, so a test
251            // relying on that signal must be able to measure that it did not land here.
252            crate::storage::read_path_probe::mark_batched_scan_stitching_path();
253            // Forward the windowed driver's internal batches straight through — no
254            // flatten, no re-batch (issue #1592). Same driver, same order/content
255            // as the per-row path; only the output surface differs.
256            self.run_scan_stream_windowed(
257                table_id,
258                start_key,
259                end_key,
260                schema,
261                &cursor,
262                now_secs,
263                WindowedOut::Batched(tx.clone()),
264                phase_sink,
265            )
266            .await
267        } else {
268            // Non-stitching formats read block-by-block; emit surviving entries
269            // in BATCH_EMIT_ROWS-capped batches (issue #1592). A block with more
270            // than BATCH_EMIT_ROWS surviving entries is split across multiple
271            // batches so every emitted batch respects `batch.len() <=
272            // BATCH_EMIT_ROWS` (the resident-row budget behind the channel cap);
273            // the remainder carries over to the next block so we still wake the
274            // consumer at most once per full batch.
275            //
276            // Confirmed rows from successfully-parsed prior blocks live in `batch`
277            // until it fills. If a LATER block's read/parse errors mid-scan, flush
278            // those confirmed rows to the consumer BEFORE surfacing the terminal
279            // error — matching both the per-row `run_scan_stream` contract (each row
280            // is sent the instant it is parsed) and the stitching path's
281            // `flush_pending` (issue #1143 / #1592, roborev Finding 1). Propagating
282            // the error via `?` here would silently drop up to BATCH_EMIT_ROWS-1
283            // confirmed rows.
284            let mut batch: Vec<(RowKey, ScanRow)> = Vec::with_capacity(BATCH_EMIT_ROWS);
285            // Work-probe (issue #2398, extended by #3058): same "changed partition
286            // key = one more partition body decoded" accounting `sequential_scan`
287            // does, so the single-source `do_get` fast path's full-ring scan is
288            // visible to the scan-work counter instead of silently unrecorded. The
289            // BTI branch above keeps the same accounting inside
290            // `bti_scan_with_metadata` (issue #3109), so routing `da` readers away
291            // from this loop does not make their decode work invisible.
292            let mut prev_partition_key: Option<RowKey> = None;
293            loop {
294                let block = match self.read_next_block(&cursor).await {
295                    Ok(Some(block)) => block,
296                    Ok(None) => break,
297                    Err(e) => {
298                        if !batch.is_empty() {
299                            let _ = tx.send(Ok(std::mem::take(&mut batch))).await;
300                        }
301                        return Err(e);
302                    }
303                };
304                // #1695: observe consumer closure once per BLOCK, not only when a
305                // batch fills. The four `continue` paths below (table-id mismatch,
306                // below `start`, above `end`, tombstone-filtered) never reach a send,
307                // and `BATCH_EMIT_ROWS` gates the send even when rows DO survive — so
308                // a scan whose filters reject everything read and PARSED every block
309                // of the table after the caller already had its `QueryTimeout`.
310                //
311                // This is the BATCHED sibling of `per_row_scan_stream`'s non-stitching
312                // walk and is the surface query execution actually uses, so fixing only
313                // the per-row one (as an earlier revision did) left the common path
314                // exposed. A plain check suffices: the await above is a bounded disk
315                // read that always returns here, not an open-ended wait on another
316                // producer.
317                if tx.is_closed() {
318                    return Ok(());
319                }
320                let entries = match self.parse_batched_block(&block, schema.as_ref(), now_secs) {
321                    Ok(entries) => entries,
322                    Err(e) => {
323                        if !batch.is_empty() {
324                            let _ = tx.send(Ok(std::mem::take(&mut batch))).await;
325                        }
326                        return Err(e);
327                    }
328                };
329                for (entry_table_id, entry_key, entry_value) in entries {
330                    // Counted BEFORE every filter — including the table-id filter —
331                    // exactly like the sibling sites (`sequential_scan`'s two loops
332                    // and the stitched walk): the partition body was DECODED
333                    // regardless of whether its rows survive, and a `Data.db` whose
334                    // entries carry a non-matching `TableId` (path/header keyspace
335                    // mismatch, or `scan_table_id`'s `"default"` fallback) must not
336                    // report 0 bodies where `sequential_scan` reports N (roborev,
337                    // issue #3058).
338                    if prev_partition_key.as_ref() != Some(&entry_key) {
339                        crate::storage::sstable::work_counters::add_stream_walk_partition_parsed();
340                        prev_partition_key = Some(entry_key.clone());
341                    }
342                    if !table_ids_match(&entry_table_id, &table_id) {
343                        continue;
344                    }
345                    if let Some(ref start) = start_key {
346                        if &entry_key < start {
347                            continue;
348                        }
349                    }
350                    if let Some(ref end) = end_key {
351                        if &entry_key > end {
352                            continue;
353                        }
354                    }
355                    if !self.filter_tombstone(&entry_value) {
356                        continue;
357                    }
358                    batch.push((entry_key, entry_value));
359                    if batch.len() >= BATCH_EMIT_ROWS {
360                        if tx.send(Ok(std::mem::take(&mut batch))).await.is_err() {
361                            return Ok(()); // consumer dropped
362                        }
363                        batch.reserve(BATCH_EMIT_ROWS);
364                    }
365                }
366            }
367            if !batch.is_empty() && tx.send(Ok(batch)).await.is_err() {
368                return Ok(()); // consumer dropped
369            }
370            Ok(())
371        }
372    }
