1//! Disk-access backend resolution for the SSTable reader.
2//!
3//! Split out of `reader/mod.rs` under the campsite rule (epic #1116): that file
4//! is far over the 800-line source target, and these functions are one cohesive
5//! responsibility — turning `StorageConfig` plus the `CQLITE_*` environment
6//! overrides plus the file's size into the concrete backend
7//! ([`crate::config::DiskAccessMode`]) and madvise advice the reader uses.
8//!
9//! Every parse/resolve step here is PURE (system memory and availability are
10//! injected), so the decisions are unit-testable without mutating the
11//! process-global environment — which would race concurrent `open()` tests.
12
13use crate::config::{DiskAccessMode, PrefetchMode};

15/// Returns `true` when memory-mapped reads are force-enabled via the
16/// `CQLITE_USE_MMAP` environment variable.
17///
18/// Accepts `1`, `true`, `yes`, `on` (case-insensitive). Any other value — or
19/// an unset variable — leaves the decision to [`Config`]. This is an opt-in
20/// escape hatch for ad-hoc local use without threading a custom config.
21pub(super) fn mmap_enabled_via_env() -> bool {
22    std::env::var("CQLITE_USE_MMAP")
23        .ok()
24        .as_deref()
25        .map(parse_truthy_env)
26        .unwrap_or(false)
27}
28
29/// Parse a truthy environment-variable value (`1`/`true`/`yes`/`on`,
30/// case-insensitive). Split out so it can be unit-tested without mutating the
31/// process-global environment (which would race other `open()` tests).
32pub(super) fn parse_truthy_env(value: &str) -> bool {
33    matches!(
34        value.trim().to_ascii_lowercase().as_str(),
35        "1" | "true" | "yes" | "on"
36    )
37}

39/// Parse a [`DiskAccessMode`] from a string (`auto`/`buffered`/`mmap`/`direct`,
40/// case-insensitive). Returns `None` for unrecognized values so callers can
41/// keep the configured default. Pure for unit-testing without env mutation.
42pub(super) fn parse_disk_access_mode(value: &str) -> Option<DiskAccessMode> {
43    match value.trim().to_ascii_lowercase().as_str() {
44        "auto" => Some(DiskAccessMode::Auto),
45        "buffered" | "buffer" => Some(DiskAccessMode::Buffered),
46        "mmap" | "mapped" => Some(DiskAccessMode::Mmap),
47        "direct" | "directio" | "direct_io" | "o_direct" => Some(DiskAccessMode::Direct),
48        _ => None,
49    }
50}

52/// Parse a [`PrefetchMode`] from a string (`off`/`sequential`/`willneed`/`auto`).
53/// Returns `None` for unrecognized values. Pure for unit-testing.
54pub(super) fn parse_prefetch_mode(value: &str) -> Option<PrefetchMode> {
55    match value.trim().to_ascii_lowercase().as_str() {
56        "off" | "none" | "no" => Some(PrefetchMode::Off),
57        "sequential" | "seq" => Some(PrefetchMode::Sequential),
58        "willneed" | "will_need" | "will-need" => Some(PrefetchMode::WillNeed),
59        "auto" => Some(PrefetchMode::Auto),
60        _ => None,
61    }
62}

80/// Best-effort total physical RAM in bytes, or `None` when it cannot be
81/// determined on this platform. Used by [`DiskAccessMode::Auto`] to decide when
82/// a file is large enough to warrant page-cache-bypassing direct I/O.
83pub(super) fn system_memory_bytes() -> Option<u64> {
84    #[cfg(unix)]
85    {
86        // SAFETY: `sysconf` is a pure query with no pointer arguments.
87        let pages = unsafe { libc::sysconf(libc::_SC_PHYS_PAGES) };
88        let page_size = unsafe { libc::sysconf(libc::_SC_PAGESIZE) };
89        if pages > 0 && page_size > 0 {
90            return Some((pages as u64).saturating_mul(page_size as u64));
91        }
92        None
93    }
94    #[cfg(not(unix))]
95    {
96        None
97    }
98}

100/// Decide which disk-access backend to use for a Data.db file.
101///
102/// Pure function (system memory injected) so the [`DiskAccessMode::Auto`]
103/// heuristic can be unit-tested deterministically. Resolution rules:
104/// - explicit `Buffered`/`Mmap`/`Direct` are returned unchanged (the caller
105///   applies graceful fallback if the OS refuses the backend);
106/// - `Auto` returns `Buffered` for files below `mmap_min_size_bytes`, `Direct`
107///   when the file exceeds `memory_fraction` of `system_memory` (and memory is
108///   known and direct I/O is available on this platform), otherwise `Mmap`.
109///
110/// The deprecated `use_mmap` flag / `CQLITE_USE_MMAP` env is folded in by the
111/// caller (promoting `Buffered` to `Mmap`), not an input here. `memory_fraction`
112/// is CLAMPED only as defense in depth: callers REJECT it instead (#1696 F2).
113pub(super) fn resolve_disk_access_mode(
114    configured: DiskAccessMode,
115    file_size: u64,
116    mmap_min_size_bytes: u64,
117    memory_fraction: f64,
118    system_memory: Option<u64>,
119    direct_io_available: bool,
120) -> DiskAccessMode {
121    // Zero-length files cannot be mapped and have nothing to read directly;
122    // always use buffered I/O for them regardless of the requested mode.
123    if file_size == 0 {
124        return DiskAccessMode::Buffered;
125    }
126    match configured {
127        DiskAccessMode::Buffered => DiskAccessMode::Buffered,
128        DiskAccessMode::Mmap => DiskAccessMode::Mmap,
129        DiskAccessMode::Direct => DiskAccessMode::Direct,
130        DiskAccessMode::Auto => {
131            if file_size < mmap_min_size_bytes {
132                return DiskAccessMode::Buffered;
133            }
134            let fraction = if memory_fraction.is_finite() && memory_fraction > 0.0 {
135                memory_fraction.min(1.0)
136            } else {
137                0.5
138            };
139            if direct_io_available {
140                if let Some(mem) = system_memory {
141                    let threshold = (mem as f64 * fraction) as u64;
142                    if file_size > threshold {
143                        return DiskAccessMode::Direct;
144                    }
145                }
146            }
147            DiskAccessMode::Mmap
148        }
149    }
150}

152/// Whether the direct-I/O backend is compiled in for this platform.
153pub(super) const fn direct_io_available() -> bool {
154    cfg!(all(
155        unix,
156        any(
157            target_os = "linux",
158            target_os = "android",
159            target_os = "macos"
160        )
161    ))
162}

164/// Resolve a [`PrefetchMode`] into the concrete advice the mmap backend should
165/// issue **AT OPEN**, or `None` for "no advice".
166///
167/// OPEN-TIME only, and that is the whole contract (issue #3853): every mode
168/// except [`PrefetchMode::Sequential`] now yields `None` here.
169/// [`PrefetchMode::WillNeed`] used to map to [`memmap2::Advice::WillNeed`] and
170/// be applied at reader open, which meant a reader that was opened and never
171/// scanned paid a full-file read-ahead and never had it withdrawn. That advice
172/// moved to SCAN START, where it is paired with a `MADV_DONTNEED` when the last
173/// in-flight scan on the reader ends — see [`super::scan_lifetime`]. It is NOT a
174/// policy change for any other mode: `Auto` still issues nothing (#1143) and
175/// `Sequential` keeps its open-time advice, because that mode is an explicit
176/// opt-in to read-ahead WITH drop-behind for a one-shot scan, a different
177/// contract from `WillNeed`'s "make it resident while I am scanning".
178///
179/// `memmap2::Advice` / `Mmap::advise` (madvise) are Unix-only, so this and its
180/// single call site are gated to `#[cfg(unix)]`. On non-Unix targets the mmap
181/// backend simply issues no read-ahead advice.
182///
183/// [`PrefetchMode::Auto`] deliberately issues **no** madvise (issue #1143).
184/// `MADV_SEQUENTIAL` couples aggressive read-ahead with *drop-behind*: pages are
185/// evicted from the page cache as soon as the scan moves past them. In isolation
186/// that is fine (mmap scans are ~40% faster than buffered), but under concurrent
187/// write load the page-cache pressure means the just-dropped pages are gone by
188/// the time an overlapping scan needs them again, so re-reads take *synchronous*
189/// major page faults on the tokio worker thread and the read-side p99 tail blows
190/// up (~2x regression). Relying on the kernel's default read-ahead (no
191/// drop-behind) keeps the isolated mmap win while letting the page cache retain
192/// hot pages, which collapses that tail. Callers who genuinely want the
193/// drop-behind behaviour can still request `Sequential` explicitly
194/// (`CQLITE_PREFETCH=sequential` / [`StorageConfig::prefetch`]).
195#[cfg(unix)]
196pub(super) fn mmap_open_advice_for(prefetch: PrefetchMode) -> Option<memmap2::Advice> {
197    match prefetch {
198        // No madvise: rely on the kernel's default read-ahead. Chosen for `Auto`
199        // to avoid `MADV_SEQUENTIAL` drop-behind evicting hot pages under
200        // concurrent write load (issue #1143).
201        PrefetchMode::Off | PrefetchMode::Auto => None,
202        // Explicit opt-in to aggressive read-ahead + drop-behind. Best for a
203        // one-shot full scan that will not be re-read and should not pin the
204        // whole file in the page cache.
205        PrefetchMode::Sequential => Some(memmap2::Advice::Sequential),
206        // Issue #3853: no OPEN-time advice. `MADV_WILLNEED` for this mode is
207        // issued by `scan_lifetime` when the first scan on the reader begins.
208        PrefetchMode::WillNeed => None,
209    }
