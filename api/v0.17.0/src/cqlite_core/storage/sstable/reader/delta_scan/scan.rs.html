1//! Streaming driver for the delta-scan API (Issue #698 `scan_delta`).
2//!
3//! [`scan_delta`] opens one SSTable generation (a single `Data.db` file) and
4//! streams [`DeltaRecord`](super::DeltaRecord)s in on-disk partition/clustering
5//! order via a [`tokio::sync::mpsc`] channel.  It lives on the reader layer and
6//! does **not** route through the query engine (which merges generations and
7//! suppresses tombstones — the opposite contract).
8

10use crate::types::{ColumnId, TombstoneType, Value};
11
12use super::model::{CellDelta, CellMeta, DeltaRecord, RangeBound, RowKeys, ScanSummaryHandle};

18/// Return type of [`scan_delta`]: a channel receiver for [`DeltaRecord`]s and
19/// a [`ScanSummaryHandle`] for collecting aggregate scan statistics.
20pub type ScanDeltaOutput = (
21    tokio::sync::mpsc::Receiver<crate::Result<DeltaRecord>>,
22    ScanSummaryHandle,
23);
24
25/// Stream [`DeltaRecord`]s from a single SSTable generation directory.
26///
27/// Opens the first `Data.db` file found under `sstable_dir` and streams
28/// [`DeltaRecord::Upsert`] and [`DeltaRecord::StaticUpsert`] records in
29/// on-disk partition/clustering order via the returned channel receiver.
30///
31/// ## Contract
32///
33/// - Records stream in SSTable order (partition, then clustering).
34/// - No cross-SSTable merge, no GC-grace filtering.
35/// - Every live cell carries `writetime` and `expires_at`; TTL is never
36///   resolved at scan time (idempotent output).
37/// - Columns absent from a row are absent from `cells` (not null).
38/// - A cell tombstone (`DELETE col FROM …`) appears as
39///   `CellDelta { value: None, writetime: t, … }`.
40/// - INSERT rows carry `liveness: Some(CellMeta { writetime, expires_at })`;
41///   UPDATE rows carry `liveness: None`.
42///
43/// ## Memory
44///
45/// The channel is bounded by `buffer_size` records.  The parse task pauses
46/// when the consumer falls behind.  Individual [`DeltaRecord`]s are bounded in
47/// size and never accumulated into a full-table collection.
48///
49/// **Caveat**: `prepare_delta_scan` → `stitch_all_chunks` fully materialises the
50/// decompressed data section of the SSTable into a single `Vec<u8>` before
51/// streaming begins.  For very large SSTables this may approach the 128 MB
52/// memory budget; callers should be aware that the stitched buffer is resident
53/// for the duration of the scan.
54///
55/// ## Errors
56///
57/// A hard parse error (corrupt SSTable, missing schema, etc.) is forwarded
58/// as an `Err(…)` item in the channel stream and terminates the scan, and is
59/// counted ONCE into `cqlite.errors.total{category, subsystem="reader"}`
60/// (issue #1704), at the single boundary [`drive_delta_scan`]. Counting is a pure
61/// side effect: the `Err` the consumer receives is unchanged.
62///
63/// [`scan_delta`] returns immediately with an error if the SSTable directory
64/// does not exist or contains no `Data.db` file.
65pub fn scan_delta(
66    sstable_dir: std::path::PathBuf,
67    schema: crate::schema::TableSchema,
68    buffer_size: usize,
69) -> ScanDeltaOutput {
70    let (tx, rx) = tokio::sync::mpsc::channel(buffer_size.max(1));
71    let summary = ScanSummaryHandle::new();
72    let summary_for_task = summary.clone();
73    tokio::spawn(async move {
74        if let Err(e) = drive_delta_scan(sstable_dir, schema, tx.clone(), summary_for_task).await {
75            let _ = tx.send(Err(e)).await;
76        }
77    });
78    (rx, summary)
79}

81/// Setup for one delta scan: locate the generation's `Data.db`, build a platform,
82/// open the reader.
83///
84/// The open states [`OpenErrorReporting::DeferredToCaller`] because this whole function runs
85/// INSIDE [`drive_delta_scan`]'s counted region (issue #1704). `SSTableReader::open`
86/// records its own failure, so using it here counted one failed open twice.
87async fn open_delta_scan_reader(
88    sstable_dir: &std::path::Path,
89) -> crate::Result<std::sync::Arc<SSTableReader>> {
90    let data_db = find_data_db(sstable_dir)?;
91
92    let config = crate::Config::default();
93    let platform =
94        std::sync::Arc::new(crate::Platform::new(&config).await.map_err(|e| {
95            crate::Error::corruption(format!("scan_delta: platform init failed: {e}"))
96        })?);
97
98    SSTableReader::open_with_reporting(
99        &data_db,
100        &config,
101        platform,
102        // `drive_delta_scan` below records this scan's failure; see its doc.
103        crate::storage::sstable::reader::OpenErrorReporting::DeferredToCaller,
104    )
105    .await
106    .map(std::sync::Arc::new)
107    .map_err(|e| {
108        // The pre-existing `Error::corruption(..)` wrap is PRESERVED (issue #1704,
109        // roborev F1). This issue's contract is that emission is a pure side effect
110        // and the `Err` propagates IDENTICALLY, so dropping the wrap — which changed
111        // `scan_delta`'s caller-visible variant and message relative to `main` — was
112        // out of scope here, even though it fixed a real mislabelling. With the open
113        // now `DeferredToCaller` it records nothing of its own, so `drive_delta_scan`
114        // records this WRAPPED, caller-visible error exactly once.
115        //
116        // KNOWN AND UNFIXED HERE: an `io` cause (absent/unreadable file) still
117        // surfaces as `corruption` in BOTH the error and the metric. Pre-existing
118        // classification defect, tracked separately — not a #1704 regression.
119        crate::Error::corruption(format!("scan_delta: failed to open {:?}: {e}", data_db))
120    })
121}

123/// [`scan_delta`]'s driver plus its error-counting seam (issue #1704).
124///
125/// ONE boundary counts a failed delta scan: this one. Every step it delegates to —
126/// locating `Data.db`, platform init, the reader open, and the whole
127/// stitch/parse/emit body — is an unrecorded inner step, which is why the open goes
128/// through [`SSTableReader::open_with_reporting`]. `record_result` counts on the `Err`
129/// arm and returns the result UNCHANGED; the category comes from the classifier
130/// (`Error::obs_category`), never from here.
131async fn drive_delta_scan(
132    sstable_dir: std::path::PathBuf,
133    schema: crate::schema::TableSchema,
134    tx: tokio::sync::mpsc::Sender<crate::Result<DeltaRecord>>,
135    summary: ScanSummaryHandle,
136) -> crate::Result<()> {
137    crate::observability::record_result(
138        "reader",
139        async {
140            let reader = open_delta_scan_reader(&sstable_dir).await?;
141            run_scan_delta(reader, schema, tx, summary).await
142        }
143        .await,
144    )
145}

147/// Internal async driver for [`scan_delta`], over an ALREADY-OPEN reader.
148async fn run_scan_delta(
149    reader: std::sync::Arc<SSTableReader>,
150    schema: crate::schema::TableSchema,
151    tx: tokio::sync::mpsc::Sender<crate::Result<DeltaRecord>>,
152    summary: ScanSummaryHandle,
153) -> crate::Result<()> {
154    // Wrap schema in Arc once — both the emit closure and parse call share the
155    // same allocation rather than cloning the struct twice.
156    let schema_arc = std::sync::Arc::new(schema);
157
158    // Stitch + parse using a private per-scan cursor (issue #815): no scan-wide
159    // mutex needed because the cursor owns its own file position / chunk index.
160    // The parsing itself is synchronous and moves into spawn_blocking.
161    let (stitched, parser) = reader.prepare_delta_scan().await?;
162
163    let schema_for_parse = std::sync::Arc::clone(&schema_arc);
164    let reader_arc = std::sync::Arc::clone(&reader);
165
166    // The parse closure is synchronous; run it on a blocking thread so it can
167    // `blocking_send` without stalling the runtime. Delta-scan still materializes
168    // via `prepare_delta_scan`; streaming SELECT moved to a windowed driver (#1143).
169    let parse_result = tokio::task::spawn_blocking(move || -> crate::Result<()> {
170        parser.parse_block_emit_delta(
171            &stitched,
172            Some(&schema_for_parse),
173            &reader_arc,
174            |(
175                partition_key_raw,
176                cells,
177                cell_meta,
178                row_liveness_ts,
179                is_static,
180                is_row_tombstone,
181                marked_for_delete_at,
182                range_info,       // Issue #699: Some((start_vals,start_incl,end_vals,end_incl,del_at)) for range tombstone
183                is_partition_tombstone, // Issue #699: true for partition-level tombstone
184                col_complex_meta, // Issue #700 DS4: per-column ComplexColumnMeta
185                liveness_expires_at_micros, // Issue #702: TTL liveness expiry in µs (epoch-s * 1_000_000)
186            )| {
187                // ----------------------------------------------------------------
188                // Decode partition key from raw bytes.
189                // (Needed for all record types, so decode upfront.)
190                // ----------------------------------------------------------------
191                let pk_columns = crate::storage::partition_key_codec::decode_partition_key_columns(
192                    &partition_key_raw.0,
193                    &schema_arc,
194                )
195                .map_err(|e| crate::Error::corruption(format!(
196                    "scan_delta: failed to decode partition key (raw bytes {:?}): {e}",
197                    partition_key_raw.0
198                )))?;
199                let partition_values: Vec<Value> = pk_columns.into_iter().map(|(_, v)| v).collect();
200
201                // ----------------------------------------------------------------
202                // Issue #699: Partition tombstone
203                // ----------------------------------------------------------------
204                if is_partition_tombstone {
205                    let deleted_at = marked_for_delete_at.ok_or_else(|| {
206                        crate::Error::corruption(format!(
207                            "scan_delta: partition tombstone for pk={:?} has no markedForDeleteAt \
208                             — cannot represent faithfully (no-heuristics, issue #28)",
209                            partition_values
210                        ))
211                    })?;
212                    let record = DeltaRecord::PartitionDelete {
213                        partition_key: RowKeys::partition_only(partition_values),
214                        deleted_at,
215                    };
216                    return match tx.blocking_send(Ok(record)) {
217                        Ok(()) => Ok(std::ops::ControlFlow::Continue(())),
218                        Err(_) => Ok(std::ops::ControlFlow::Break(())),
219                    };
220                }
221
222                // ----------------------------------------------------------------
223                // Issue #699: Range tombstone
224                // ----------------------------------------------------------------
225                if let Some((start_vals, start_incl, end_vals, end_incl, del_at)) = range_info {
226                    let record = DeltaRecord::RangeDelete {
227                        partition_key: RowKeys::partition_only(partition_values),
228                        start: RangeBound::new(start_vals, start_incl),
229                        end: RangeBound::new(end_vals, end_incl),
230                        deleted_at: del_at,
231                    };
232                    return match tx.blocking_send(Ok(record)) {
233                        Ok(()) => Ok(std::ops::ControlFlow::Continue(())),
234                        Err(_) => Ok(std::ops::ControlFlow::Break(())),
235                    };
236                }
237
238                // Build key-column name sets for filtering.
239                let pk_col_names: std::collections::HashSet<&str> = schema_arc
240                    .partition_keys.iter().map(|k| k.name.as_str()).collect();
241                let clustering_col_names: std::collections::HashSet<&str> = schema_arc
242                    .clustering_keys.iter().map(|ck| ck.name.as_str()).collect();
243                let static_col_names: std::collections::HashSet<&str> = schema_arc
244                    .columns.iter().filter(|c| c.is_static).map(|c| c.name.as_str()).collect();
245
246                // Extract clustering values in declaration order.
247                let clustering_values: Vec<Value> = schema_arc
248                    .clustering_keys.iter()
249                    .filter_map(|ck| cells.get(&ck.name).cloned())
250                    .collect();
251
252                // ----------------------------------------------------------------
253                // Issue #699: Row tombstone
254                // ----------------------------------------------------------------
255                if is_row_tombstone {
256                    let deleted_at = marked_for_delete_at.ok_or_else(|| {
257                        crate::Error::corruption(format!(
258                            "scan_delta: row tombstone for pk={:?} ck={:?} has no markedForDeleteAt \
259                             — cannot represent faithfully (no-heuristics, issue #28)",
260                            partition_values, clustering_values
261                        ))
262                    })?;
263                    let record = DeltaRecord::RowDelete {
264                        keys: RowKeys::new(partition_values, clustering_values),
265                        deleted_at,
266                    };
267                    return match tx.blocking_send(Ok(record)) {
268                        Ok(()) => Ok(std::ops::ControlFlow::Continue(())),
269                        Err(_) => Ok(std::ops::ControlFlow::Break(())),
270                    };
271                }
272
273                // ----------------------------------------------------------------
274                // DS4 (Issue #700): Process element tombstone counts from this row's
275                // collection columns and update the scan summary.
276                // ----------------------------------------------------------------
277                {
278                    let mut row_element_tombstones: u64 = 0;
279                    for (col_name, ccm) in &col_complex_meta {
280                        if ccm.element_tombstone_count > 0 {
281                            row_element_tombstones += ccm.element_tombstone_count;
282                            tracing::warn!(
283                                "scan_delta DS4: collection column '{}' has {} element-level tombstone(s) \
284                                 that cannot be represented in v1 delta semantics (Issue #493 follow-up). \
285                                 These removals are counted in the scan summary but not in the emitted records.",
286                                col_name, ccm.element_tombstone_count
287                            );
288                        }
289                    }
290                    if row_element_tombstones > 0 {
291                        summary.add_element_tombstones(row_element_tombstones);
292                    }
293                }
294
295                // ----------------------------------------------------------------
296                // Build CellDelta entries for non-key columns only.
297                // ----------------------------------------------------------------
298                let mut cell_deltas: Vec<(ColumnId, CellDelta)> = Vec::new();
299
300                for (col_name, value) in &cells {
301                    // Skip key columns — they are part of RowKeys, not cell payload.
302                    if pk_col_names.contains(col_name.as_str())
303                        || clustering_col_names.contains(col_name.as_str())
304                    {
305                        continue;
306                    }
307                    // Static upsert rows: only static columns.
308                    // Regular upsert rows: only non-static columns.
309                    if is_static && !static_col_names.contains(col_name.as_str()) {
310                        continue;
311                    }
312                    if !is_static && static_col_names.contains(col_name.as_str()) {
313                        continue;
314                    }
315
316                    let meta = cell_meta.get(col_name.as_str());
317                    let (writetime, expires_at) = match meta {
318                        Some(m) => {
319                            let exp = m.expiration.as_ref().map(|e| {
320                                // expires_at_seconds is epoch-seconds; delta-scan
321                                // contract requires epoch-microseconds.
322                                e.expires_at_seconds.saturating_mul(1_000_000)
323                            });
324                            (m.write_timestamp_micros, exp)
325                        }
326                        // Fallback: row-level liveness timestamp (should not happen
327                        // when want_cell_metadata=true, but be defensive).
328                        None => (row_liveness_ts.unwrap_or(0), None),
329                    };
330
331                    // DS4 (Issue #700): For collection columns, check whether this
332                    // generation carries a collection-level tombstone (overwrite semantics).
333                    // `replaced = true` signals downstream consumers to replace rather than
334                    // merge the prior collection state.  Always `false` for scalar columns.
335                    //
336                    // Also: the normal read-path `cell_meta` stores `row_ts` for collection
337                    // columns (not per-element max) to keep WRITETIME(col) semantics correct.
338                    // For the delta-scan path we override writetime with the max element
339                    // writetime from ComplexColumnMeta when it is non-zero (roborev Finding 1).
340                    let (replaced, writetime) = match col_complex_meta.get(col_name.as_str()) {
341                        Some(ccm) => {
342                            let effective_wt = if ccm.max_element_writetime != 0 {
343                                ccm.max_element_writetime
344                            } else {
345                                writetime
346                            };
347                            (ccm.has_collection_tombstone, effective_wt)
348                        }
349                        None => (false, writetime),
350                    };
351
352                    let cell = match value {
353                        // Cell tombstone: IS_DELETED flag was set on the cell.
354                        Value::Tombstone(info)
355                            if info.tombstone_type == TombstoneType::CellTombstone =>
356                        {
357                            CellDelta {
358                                value: None,
359                                // Use the tombstone's own deletion_time (authoritative),
360                                // NOT the cell_meta write_timestamp (which inherits row ts).
361                                writetime: info.deletion_time,
362                                expires_at: None,
363                                replaced: false,
364                            }
365                        }
366                        _ => CellDelta {
367                            value: Some(value.clone()),
368                            writetime,
369                            expires_at,
370                            replaced,
371                        },
372                    };
373
374                    cell_deltas.push((ColumnId::new(col_name), cell));
375                }
376
377                // ----------------------------------------------------------------
378                // Emit the appropriate DeltaRecord variant (Upsert / StaticUpsert).
379                // ----------------------------------------------------------------
380                let record = if is_static {
381                    DeltaRecord::StaticUpsert {
382                        partition_key: RowKeys::partition_only(partition_values),
383                        cells: cell_deltas,
384                    }
385                } else {
386                    // Build liveness CellMeta, carrying the TTL expiry when present (Issue #702).
387                    let liveness = row_liveness_ts.map(|ts| {
388                        match liveness_expires_at_micros {
389                            Some(exp) => CellMeta::with_ttl(ts, exp),
390                            None => CellMeta::new(ts),
391                        }
392                    });
393                    DeltaRecord::Upsert {
394                        keys: RowKeys::new(partition_values, clustering_values),
395                        liveness,
396                        cells: cell_deltas,
397                    }
398                };
399
400                // Forward to the channel.  `blocking_send` is correct here
401                // because we are inside spawn_blocking (not an async context).
402                match tx.blocking_send(Ok(record)) {
403                    Ok(()) => Ok(std::ops::ControlFlow::Continue(())),
404                    Err(_) => {
405                        // Consumer dropped: stop streaming.
406                        Ok(std::ops::ControlFlow::Break(()))
407                    }
408                }
409            },
410        )
411    })
412    .await;
413
414    match parse_result {
415        Ok(result) => result,
416        Err(join_err) => Err(crate::Error::corruption(format!(
417            "scan_delta: parse task panicked: {join_err}"
418        ))),
419    }
420}

422/// Find the `Data.db` file inside an SSTable directory.
423///
424/// Cassandra names Data.db files with a generation prefix, e.g.:
425/// `nb-1-big-Data.db` or `na-1-big-Data.db`.
426///
427/// Returns an error if no matching file is found.  If more than one `*-Data.db`
428/// file is present (which violates the single-generation contract), a warning is
429/// logged and the lexicographically smallest file name is returned so behaviour
430/// is at least deterministic rather than OS-dependent.
431fn find_data_db(dir: &std::path::Path) -> crate::Result<std::path::PathBuf> {
432    if !dir.exists() {
433        return Err(crate::Error::corruption(format!(
434            "scan_delta: SSTable directory does not exist: {:?}",
435            dir
436        )));
437    }
438
439    let entries = std::fs::read_dir(dir).map_err(|e| {
440        crate::Error::corruption(format!("scan_delta: cannot read directory {:?}: {e}", dir))
441    })?;
442
443    let mut candidates: Vec<std::path::PathBuf> = entries
444        .filter_map(|e| e.ok())
445        .filter(|e| {
446            e.file_name()
447                .to_str()
448                .map(|n| n.ends_with("-Data.db"))
449                .unwrap_or(false)
450        })
451        .map(|e| e.path())
452        .collect();
453
454    if candidates.is_empty() {
455        return Err(crate::Error::corruption(format!(
456            "scan_delta: no Data.db file found in {:?}",
457            dir
458        )));
459    }
460
461    if candidates.len() > 1 {
462        candidates.sort();
463        tracing::warn!(
464            "scan_delta: {:?} contains {} Data.db files (expected 1 per generation); \
465             using lexicographically first: {:?}. Consider compacting before scanning.",
466            dir,
467            candidates.len(),
468            candidates[0]
469        );
470    }
471
472    Ok(candidates.remove(0))
473}
