1//! The `SSTableReader` OPEN constructors and the single place a failed open is
2//! recorded (issues #1037, #1704).
3//!
4//! Split out of `reader/mod.rs` per the campsite rule (#1116): that file is 1721
5//! lines against the ~800-line target, and issue #1704 needed to add a constructor
6//! here. Everything below is a VERBATIM move except [`SSTableReader::open_with_reporting`],
7//! which is new.
8//!
9//! # One recording boundary per failed operation
10//!
11//! Every public constructor funnels into [`SSTableReader::open_with_cache_cancellable`],
12//! which records a failed open into `cqlite.errors.total{category, subsystem="reader"}`.
13//! That is correct when the open IS the operation. It is WRONG when the open is an
14//! inner step of an operation whose own seam also records — the failure is then
15//! counted twice for one user-visible failure. This is the same single-boundary rule
16//! `compaction::finalize_merge_async` already states for its own inner helpers
17//! ("this is an *unrecorded* inner helper ... recording here too would double-count").
18//!
19//! [`SSTableReader::open_with_reporting`] is the entry point for those inner steps.
20

30impl SSTableReader {
31    /// Open an SSTable file for reading.
32    ///
33    /// Instrumented (epic #1031 / #1034): wraps the open in a
34    /// `sstable.reader.open` span, increments the [`SSTABLES_OPEN`] gauge on
35    /// success, and records an error on the `reader` subsystem when open fails.
36    ///
37    /// [`SSTABLES_OPEN`]: crate::observability::catalog::SSTABLES_OPEN
38    pub async fn open(path: &Path, config: &Config, platform: Arc<Platform>) -> Result<Self> {
39        // Back-compat: existing callers and sibling crates get a FRESH per-reader
40        // decompressed-chunk cache sized from config (issue #1567). Production
41        // reads route through `SSTableManager`, which calls `open_with_cache` with
42        // its SHARED instance so all readers of a dataset share one cache.
43        //
44        // Honor the `config.memory.block_cache.enabled` toggle (issue #1568): when
45        // disabled this yields a genuine no-op cache so the direct reader path
46        // bypasses caching identically to the manager path, instead of the toggle
47        // being decorative here. Reuses the shared `build_chunk_cache` helper.
48        let cache = super::super::build_chunk_cache(config);
49        Self::open_with_cache(path, config, platform, cache).await
50    }

52    /// Cancel-aware [`open`](Self::open) (issue #2383 fix C).
53    ///
54    /// Threads a synchronous [`ScanCancel`](crate::storage::scan_cancel::ScanCancel)
55    /// into the Index.db partition-index parse so a client-disconnect cancel
56    /// aborts a 1.58M-entry parse within one poll interval instead of pinning a
57    /// tokio worker to completion. Non-cancellable callers use [`open`](Self::open)
58    /// (a default never-cancel flag). Returns [`Error::Cancelled`] on a mid-parse
59    /// trip.
60    pub async fn open_cancellable(
61        path: &Path,
62        config: &Config,
63        platform: Arc<Platform>,
64        cancel: crate::storage::scan_cancel::ScanCancel,
65    ) -> Result<Self> {
66        let cache = super::super::build_chunk_cache(config);
67        Self::open_with_cache_cancellable(path, config, platform, cache, cancel).await
68    }

70    /// Open an SSTable file for reading, sharing the provided
71    /// [`DecompressedChunkCache`](crate::storage::cache::DecompressedChunkCache).
72    ///
73    /// Identical to [`open`](Self::open) except the reader stores `cache` (an
74    /// `Arc` clone) instead of minting its own, so every reader a manager opens
75    /// for one dataset consults the same bytes-bounded chunk cache (issue #1567).
76    pub async fn open_with_cache(
77        path: &Path,
78        config: &Config,
79        platform: Arc<Platform>,
80        cache: Arc<crate::storage::cache::DecompressedChunkCache>,
81    ) -> Result<Self> {
82        Self::open_with_cache_cancellable(
83            path,
84            config,
85            platform,
86            cache,
87            crate::storage::scan_cancel::ScanCancel::default(),
88        )
89        .await
90    }

92    /// Cancel-aware [`open_with_cache`](Self::open_with_cache) (issue #2383 fix C):
93    /// same as it but threads `cancel` into the Index.db parse. See
94    /// [`open_cancellable`](Self::open_cancellable).
95    pub async fn open_with_cache_cancellable(
96        path: &Path,
97        config: &Config,
98        platform: Arc<Platform>,
99        cache: Arc<crate::storage::cache::DecompressedChunkCache>,
100        cancel: crate::storage::scan_cancel::ScanCancel,

102        Self::open_instrumented(
103            path,
104            config,
105            platform,
106            cache,
107            cancel,
108            OpenErrorReporting::SelfReported,
109        )
110        .await
111    }

113    /// [`open`](Self::open) with the failed-open reporting mode stated BY THE CALLER
114    /// (issue #1704).
115    ///
116    /// # Why this is a parameter and not a second named function
117    ///
118    /// Both mistakes are silent. [`OpenErrorReporting::SelfReported`] under a caller
119    /// that also records counts one failure TWICE;
120    /// [`OpenErrorReporting::DeferredToCaller`] under a caller with no seam records it
121    /// ZERO times, which is worse and is exactly the regression a universally
122    /// non-recording open introduced. Only the caller knows which it is, so it must
123    /// say — an unparameterised `open_unrecorded` encoded a precondition that nothing
124    /// at the call site could check.
125    ///
126    /// # Behavioural identity, which is now real rather than claimed
127    ///
128    /// Exactly ONE action differs from [`open`](Self::open): the `record_error` call on
129    /// the `Err` arm. Both routes go through [`open_instrumented`](Self::open_instrumented),
130    /// so the `sstable.reader.open` span, its `file_size`/`sstable_format` fields, the
131    /// parenting of the nested open-phase spans, and the `SSTABLES_OPEN` gauge are
132    /// literally the same code. The first version of this function called `open_inner`
133    /// directly and silently dropped the span while its doc claimed identity — the
134    /// shared helper is what makes the claim checkable instead of aspirational.
135    #[cfg(any(feature = "delta-scan", feature = "write-support"))]
136    pub(crate) async fn open_with_reporting(
137        path: &Path,
138        config: &Config,
139        platform: Arc<Platform>,
140        reporting: OpenErrorReporting,
141    ) -> Result<Self> {
142        let cache = super::super::build_chunk_cache(config);
143        Self::open_instrumented(
144            path,
145            config,
146            platform,
147            cache,
148            crate::storage::scan_cancel::ScanCancel::default(),
149            reporting,
150        )
151        .await
152    }

154    /// The ONE open implementation: span, success gauge, and — for
155    /// [`OpenErrorReporting::SelfReported`] only — the failed-open increment.
156    async fn open_instrumented(
157        path: &Path,
158        config: &Config,
159        platform: Arc<Platform>,
160        cache: Arc<crate::storage::cache::DecompressedChunkCache>,
161        cancel: crate::storage::scan_cancel::ScanCancel,
162        reporting: OpenErrorReporting,
163    ) -> Result<Self> {
164        use crate::observability::{self as obs, catalog};
165        use tracing::Instrument as _;
166
167        let span = tracing::debug_span!(
168            "sstable.reader.open",
169            file_size = tracing::field::Empty,
170            sstable_format = tracing::field::Empty,
171        );
172
173        // Instrument the future rather than holding an entered guard across the
174        // `.await`: entering a span guard and then awaiting can attach unrelated
175        // async work scheduled on this task to the span. `Instrument` enters the
176        // span only while this specific future is polled.
177        let result = Self::open_inner(path, config, platform, cache, cancel)
178            .instrument(span.clone())
179            .await;
180        match &result {
181            Ok(reader) => {
182                let format = reader.sstable_format_label();
183                span.record("file_size", reader.stats.file_size);
184                span.record("sstable_format", format);
185                // SSTABLES_OPEN is a snapshot gauge of the live reader count;
186                // record the current PER-FORMAT count after this open succeeds so
187                // the format-attributed gauge series stays correct under mixed
188                // BIG/BTI readers. A SUCCESSFUL open is identical in both modes:
189                // the reporting mode is only ever about who counts a FAILURE.
190                let now = sstables_open_count_for(format).fetch_add(1, Ordering::Relaxed) + 1;
191                obs::record_gauge(
192                    catalog::SSTABLES_OPEN,
193                    now,
194                    &[(catalog::attr::SSTABLE_FORMAT, format.into())],
195                );
196            }
197            Err(e) => {
198                // Both arms run WHILE the open span is current: the instrumented
199                // future has already completed, so the span is no longer entered and
200                // `in_scope` re-enters it for the duration of the call.
201                //
202                // Only the COUNTER is deferred (issue #1704). The SPAN is this call's
203                // own and must be marked either way — suppressing all of
204                // `record_error` also suppressed its `mark_span_error`, so a deferred
205                // open returned `Err` under a span that still looked successful. Same
206                // shape as roborev E, where this function dropped the span outright:
207                // an observability change must not quietly lose span fidelity.
208                span.in_scope(|| match reporting {
209                    OpenErrorReporting::SelfReported => obs::record_error(e, "reader"),
210                    OpenErrorReporting::DeferredToCaller => obs::mark_span_error(e),
211                });
212            }
213        }
214        result
215    }
216}

218/// Who counts a failed open into `cqlite.errors.total{subsystem="reader"}`
219/// (issue #1704). The ONLY thing that varies between the two open routes.
220#[derive(Clone, Copy, Debug, PartialEq, Eq)]
221pub(crate) enum OpenErrorReporting {
222    /// This open IS the operation: it records its own failure. The mode every
223    /// PUBLIC entry point uses, so a caller who does not think about it is never
224    /// silently un-instrumented.
225    SelfReported,
226    /// This open is an inner step; the caller's operation seam records instead.
227    ///
228    /// Constructed by `scan_delta` (feature `delta-scan`) and by the cross-generation
229    /// streaming merge (`not(feature = "tombstones")`). A build with `tombstones` on
230    /// and `delta-scan` off therefore has no constructor for it — the variant is still
231    /// the correct half of the distinction, so it is allowed rather than cfg'd away
232    /// into a shape that would differ per feature set.
233    #[allow(dead_code)]
234    DeferredToCaller,
235}
