1//! The cross-generation partition point read: [`SSTableManager::get`] (split out of
2//! `mod.rs` per the campsite rule, epic #1116).
3//!
4//! Two builds, one contract: the default build returns the first matching
5//! generation, the `tombstones` build collects every generation's value and
6//! reconciles them through [`TombstoneMerger`]. Both walk the SAME resolved reader
7//! list (issue #1321) and both are ONE logical read operation.
8//!
9//! # Why the read metrics live at THIS level (issue #1701, roborev B1)
10//!
11//! A logical point read may probe several SSTable generations. Metering each
12//! per-reader lookup made one `get` emit one `cqlite.read.duration` sample PER
13//! CANDIDATE and count a row once per matching generation instead of once per
14//! reconciled result — a metric that overstates both the read rate and the read
15//! count. So the per-reader lookups are unmetered here
16//! ([`SSTableReader::get_with_resolution_unmetered`]) and the whole operation is
17//! metered ONCE, format-agnostically, because the generations of one table need not
18//! share an on-disk format and a fabricated single label would be a lie.
19//!
20//! # Why the meter starts at FUNCTION ENTRY (roborev F4)
21//!
22//! `resolve_reader_snapshot` takes the manager's reader lock and resolves the target
23//! table's generations. Starting the meter after it put LOCK AND RESOLUTION LATENCY
24//! OUTSIDE the reported read duration — and lock contention behind a slow scan
25//! (issue #1591) is precisely what an operator hunts a read-latency metric for, so
26//! excluding it hides the very stall the metric exists to expose. The meter therefore
27//! covers the whole function, including the `reader_list.is_empty()` early return: a
28//! read of a table with no candidate SSTables is a completed operation with zero rows,
29//! and the `tombstones` build used to emit NOTHING there while the default build
30//! emitted a sample — two builds disagreeing about what a read IS.
31
32use super::SSTableManager;
33use crate::observability::read_metrics::ReadOpMeter;
34use crate::types::{ScanRow, TableId};
35use crate::{Result, RowKey};
36
37#[cfg(feature = "tombstones")]
38use super::tombstone_merger::{EntryMetadata, GenerationValue, TombstoneMerger};
39
40impl SSTableManager {
41    /// Get a value by key from all SSTables with proper tombstone merging
42    #[cfg(feature = "tombstones")]
43    pub async fn get(&self, table_id: &TableId, key: &RowKey) -> Result<Option<ScanRow>> {
44        // ONE meter for the WHOLE logical read, started at FUNCTION ENTRY (issue #1701,
45        // roborev B1 + F4). This module's doc records WHY entry and not
46        // post-resolution, and why the empty-reader-list early return must still
47        // report a duration.
48        let mut meter = ReadOpMeter::start(None);
49
50        // Resolve the applicable reader list FIRST, exactly like the non-tombstones
51        // `get()` path (issue #1321). The previous code iterated EVERY reader in
52        // `self.readers` and passed one global relaxed `fully_qualified_match` flag
53        // to all of them, so same-named tables in OTHER keyspaces passed the relaxed
54        // BTI guard and wrongly contributed values/tombstones to the merge — a
55        // cross-keyspace data-bleed bug. `resolve_reader_list` returns precisely the
56        // readers for the resolved target table across generations, so the relaxed
57        // guard can only ever apply to the readers that ARE the target table; a
58        // wrong-keyspace same-named reader is never in the merge set.
59        //
60        // Issue #1591: snapshot the resolved readers + the authoritative
61        // `fully_qualified_match` signal and DROP the read guard before any I/O.
62        let (reader_list, fully_qualified_match) = self.resolve_reader_snapshot(table_id).await;
63        if reader_list.is_empty() {
64            // A read of a table with no candidate SSTables is still a completed read
65            // operation: it consumed the resolution latency above and returned an
66            // answer, so it reports a duration with zero rows (F4). Emitting nothing
67            // here is what made this build diverge from the default one.
68            meter.finish();
69            return Ok(None);
70        }
71
72        let mut all_values = Vec::new();
73
74        // Collect each applicable generation's value (tombstone-merge semantics are
75        // unchanged: still build a `GenerationValue` per reader and resolve via
76        // `TombstoneMerger::merge_generations`). Only the SET of readers being merged
77        // changed — the resolved list instead of every reader globally.
78        for reader in &reader_list {
79            if let Some(value) = reader
80                .get_with_resolution_unmetered(table_id, key, fully_qualified_match)
81                .await?
82            {
83                let generation = reader.generation;
84                let write_time = reader.extract_write_time_from_entry(key, &value);
85
86                let gen_value = GenerationValue {
87                    value,
88                    metadata: EntryMetadata {
89                        write_time,
90                        generation,
91                        ttl: None, // Would be extracted from SSTable metadata
92                    },
93                };
94                all_values.push(gen_value);
95            }
96        }
97
98        // Use tombstone merger to resolve conflicts across generations
99        let merger = TombstoneMerger::new();
100        let reconciled = merger.merge_generations(all_values);
101        // Count the RECONCILED result, never the per-generation matches: a row that
102        // exists in three generations is ONE row of ONE partition to a reader.
103        if let Ok(Some(_)) = &reconciled {
104            meter.record_row(key);
105        }
106        meter.finish();
107        reconciled
108    }
109
110    /// Get a value by key from all SSTables (simple version without tombstone merging)
111    ///
112    /// Uses `table_readers` (keyed by fully-qualified `"keyspace.table"`) so that only the
113    /// SSTables for the requested table are searched (Issue #680).  Same-named tables in
114    /// different keyspaces (e.g. `test_basic.simple_table` and `test_oa.simple_table`) are
115    /// now correctly distinguished.
116    ///
117    /// Lookup order:
118    ///   1. Exact match on the full `table_id` string (e.g. `"test_basic.simple_table"`)
119    ///   2. Unqualified table name (e.g. `"simple_table"`) — for backward compatibility
120    ///      with flat/non-Cassandra directory layouts that have no keyspace parent.
121    #[cfg(not(feature = "tombstones"))]
122    pub async fn get(&self, table_id: &TableId, key: &RowKey) -> Result<Option<ScanRow>> {
123        // ONE meter for the WHOLE logical read, started at FUNCTION ENTRY (issue #1701,
124        // roborev B1 + F4) — see the `tombstones` sibling above and the rationale in
125        // this module's doc.
126        let mut meter = ReadOpMeter::start(None);
127
128        // Issue #1591: snapshot the resolved readers + the authoritative
129        // `fully_qualified_match` signal and DROP the read guard before any I/O,
130        // so a queued writer never FIFO-parks this point read behind a slow scan.
131        //
132        // `fully_qualified_match`: did resolution match the FULLY-QUALIFIED
133        // `keyspace.table` key exactly, or fall back to the bare table name? An
134        // unqualified query is treated as an exact match (no keyspace to mismatch).
135        // This authoritative signal gates the get() point-lookup table-consistency
136        // guard exactly like the seek path (#1284): only an exact FQ match may relax
137        // to a name-only check across a header-keyspace divergence; a fully-qualified
138        // query resolved via the bare-name fallback keeps strict keyspace matching so
139        // get() never returns another keyspace's same-named rows (issue #1321).
140        let (reader_list, fully_qualified_match) = self.resolve_reader_snapshot(table_id).await;
141
142        // Return the first value found across all SSTables for this table
143        for reader in &reader_list {
144            if let Some(value) = reader
145                .get_with_resolution_unmetered(table_id, key, fully_qualified_match)
146                .await?
147            {
148                meter.record_row(key);
149                meter.finish();
150                return Ok(Some(value));
151            }
152        }
153
154        // A read that resolved ABSENCE still reports its latency (0 rows): dropping it
155        // would bias the distribution toward hits.
156        meter.finish();
157        Ok(None)
158    }
159}
