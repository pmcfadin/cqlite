1//! Which COMPONENT of a multicell collection's declared type a CELL PATH holds,
2//! and the ONE resolver that names it — shared by the READ and WRITE sides
3//! (issues #3805, #4106).
4//!
5//! # Why this is one module and not two
6//! Cassandra decides both collection kinds with ONE line. At `cassandra-5.0.8`,
7//! `schema/ColumnMetadata.java:457-467` validates every collection cell path
8//! with `((CollectionType)type).nameComparator().validate(path.get(0))`, and
9//! `nameComparator()` is the KEYS type of a `MapType` (`db/marshal/MapType.java`)
10//! and the ELEMENTS type of a `SetType` (`db/marshal/SetType.java:101-104`). So
11//! "which component does the cell path hold" is one question with two answers,
12//! and both the writer's admission gate
13//! (`writer::data_writer::cell_path::admit_empty_cell_path`) and the reader's
14//! (`reader::parsing::row_decoder::complex_column::cell_path_empty`) need it.
15//!
16//! It lived only on the WRITE side until #4106 roborev job 449, and that
17//! asymmetry was the finding — see [`resolve_declared_cell_path_type`]'s
18//! "THE SEAM" section for the exact divergence and its cause.
19
20use crate::schema::CqlType;
21use crate::storage::sstable::reader::parsing::row_decoder::V5CompressedLegacyParser;

23/// Which component of a multicell collection's declared type a cell path holds.
24///
25/// The two are the same question — Cassandra answers both with
26/// `nameComparator()` (see the module header) — so they share ONE resolver
27/// rather than two near-identical ones able to drift.
28#[derive(Debug, Clone, Copy, PartialEq, Eq)]
29pub(crate) enum CellPathComponent {
30    /// A `map<K,V>`'s KEY (`MapType.nameComparator()` == the keys type).
31    ///
32    /// WRITE-SIDE ONLY, and gated to say so. Its sole constructor is
33    /// `writer::data_writer::cell_path`; the READ path constructs
34    /// [`Self::SetElement`] only, because the map read route keeps its own
35    /// marshal-over-schema precedence rule (#3612 R7/R8, authority
36    /// `SerializationHeader.getType`) rather than resolving from the declared
37    /// container. Under `--no-default-features` the `write-support` default
38    /// vanishes, `pub mod writer` with it, and this variant would be
39    /// constructed by nothing. Gated rather than `allow(dead_code)`-ed: this
40    /// module's sibling records that a conditional allow "would silence
41    /// nothing and merely assert something false"
42    /// (`row_decoder/udt/type_string.rs`).
43    #[cfg(feature = "write-support")]
44    MapKey,
45    /// A `set<T>`'s ELEMENT (`SetType.nameComparator()` == the elements type).
46    SetElement,
47}
48
49impl CellPathComponent {
50    /// The two declared-type spellings this component is resolvable from, for a
51    /// diagnostic — the message tells the caller exactly what would have worked.
52    ///
53    /// Gated to match its callers, NOT `allow(dead_code)`. Its only caller is
54    /// `writer::data_writer::cell_path`, and `pub mod writer`
55    /// (`storage/sstable/mod.rs`) is itself `#[cfg(feature = "write-support")]`.
56    /// `write-support` is a DEFAULT feature, so it survives every ordinary build
57    /// and vanishes only under `--no-default-features` — which is exactly the
58    /// gate's `feature-iso-parquet` lane
59    /// (`--no-default-features --features all-compression,parquet`), where the
60    /// sole caller disappears and this becomes dead code under `-D warnings`.
61    /// An `allow(dead_code)` would silence the signal instead of expressing the
62    /// invariant: this diagnostic exists only where something can write.
63    #[cfg(feature = "write-support")]
64    pub(crate) fn declared_shapes(self) -> &'static str {
65        match self {
66            CellPathComponent::MapKey => {
67                "the CQL spelling `map<K,V>` nor the Cassandra marshal one \
68                 `org.apache.cassandra.db.marshal.MapType(K,V)`"
69            }
70            CellPathComponent::SetElement => {
71                "the CQL spelling `set<T>` nor the Cassandra marshal one \
72                 `org.apache.cassandra.db.marshal.SetType(T)`"
73            }
74        }
75    }
76}

78/// The DECLARED type of the cell-path component of a multicell collection
79/// column, from the column's COMPLETE declared type string in EITHER spelling —
80/// `None` when the string denotes no collection of that kind at all.
81///
82/// # NEITHER resolution is written here (one fact, one parser)
83///  * the CQL spelling is [`CqlType::parse`];
84///  * the marshal spelling is
85///    [`V5CompressedLegacyParser::parse_cassandra_type`], whose name table is
86///    derived arm-by-arm from `cql3/CQL3Type.java`'s `Native` enum at
87///    `cassandra-5.0.8` and which enforces the marshal PACKAGE rule (a
88///    third-party `com.acme.Int32Type` is refused, not read as `int`).
89///
90/// The string->string `convert_marshal_type_to_cql` in
91/// `parser::enhanced_statistics_parser` is deliberately NOT used: it maps
92/// `IntegerType` to `int` where Cassandra binds it to `varint`, and a
93/// one-argument `MapType(V)` to `map<text, V>`, so reusing it would decide an
94/// admission against a component type Cassandra does not agree with.
95///
96/// # THE SEAM this function exists to close — a COMPLETE type, never a COMPONENT
97/// # NAME (issue #4106, roborev job 449 finding B1)
98///
99/// Two reviewers reported adjacent facts that look contradictory and are not.
100/// The marshal TABLE is shared — `native_marshal_to_cql_type` is the crate's one
101/// marshal-name authority and `primitive_marshal_to_cql_short` is a projection of
102/// it — but the ENTRY CONDITION differs, and that is where read and write
103/// diverged:
104///
105///  * every consumer that receives an ISOLATED COMPONENT NAME consults the table
106///    only when the string already `contains("org.apache.cassandra.db.marshal.")`
107///    — `cell_path_key_cql_type` and the value decoder
108///    (`raw_value/reporting.rs`) both spell it that way;
109///  * this function receives the COMPLETE DECLARED TYPE, so it may hand the whole
110///    string to `parse_cassandra_type`, which KNOWS its input is a marshal type
111///    string and therefore applies Cassandra's own `TypeParser` rule
112///    (`TypeParser.java:450`: inside a marshal type string an UNQUALIFIED name IS
113///    a class in the marshal package).
114///
115/// The divergence is CONTEXT, not table membership. `org.apache.cassandra.db.
116/// marshal.SetType(Int32Type)` — package on the OUTER name, bare inner element
117/// name — is a legal `TypeParser` spelling that the WRITER accepted (it resolves
118/// the component from this complete string) while the READER refused: it split
119/// `Int32Type` out first, and by then the string had LOST the fact that it came
120/// from a marshal type string, so the `contains(package)` guard declined and the
121/// empty cell path decoded as an opaque blob instead of `Value::Empty(Int)`.
122///
123/// The fix is to resolve from the COMPLETE type, restoring the context the split
124/// discarded. **Synthesising the package onto a bare component name is NOT the
125/// fix and must never be reintroduced**: #3612 roborev round 9 finding 2 removed
126/// exactly that, because `format!("org.apache.cassandra.db.marshal.{t}")` made a
127/// foreign `com.acme.CustomBytesType` match `BytesType` and suppressed the
128/// diagnostic that exists for unmodelled custom types — and deciding type
129/// identity from a name suffix is name-pattern inference that #28 forbids. That
130/// rule has a committed regression suite
131/// (`row_decoder::udt::regression_3631_marshal_package_rule_tests`).
132///
133/// # What stays refused, on purpose
134/// A `FrozenType(MapType(K,V))` / `FrozenType(SetType(T))` resolves to
135/// [`CqlType::Frozen`] and NOT to [`CqlType::Map`] / [`CqlType::Set`], so it is
136/// refused here: a frozen collection is ONE inline length-prefixed cell with no
137/// CellPath at all, so its empty component is the inline-element case owned by
138/// `require_fixed_width` (#3847/#4071), not this one. A non-collection
139/// declaration, a foreign-package class, and a spelling neither parser models
140/// are all refused for the reason this function exists.
141pub(crate) fn resolve_declared_cell_path_type(
142    declared: &str,
143    component: CellPathComponent,
144) -> Option<CqlType> {
145    fn pick(ty: CqlType, component: CellPathComponent) -> Option<CqlType> {
146        match (ty, component) {
147            #[cfg(feature = "write-support")]
148            (CqlType::Map(key, _), CellPathComponent::MapKey) => Some(*key),
149            (CqlType::Set(element), CellPathComponent::SetElement) => Some(*element),
150            _ => None,
151        }
152    }
153    if let Ok(ty) = CqlType::parse(declared) {
154        if let Some(resolved) = pick(ty, component) {
155            return Some(resolved);
156        }
157    }
158    match V5CompressedLegacyParser::parse_cassandra_type(declared) {
159        Ok(ty) => pick(ty, component),
160        Err(_) => None,
161    }
162}

164#[cfg(test)]
165mod tests {
166    use super::*;
167
168    /// THE B1 CASE (roborev job 449): package on the OUTER name, BARE inner
169    /// element/key name — a legal `TypeParser` spelling
170    /// (`TypeParser.java:450`) that the component-name classifiers cannot
171    /// resolve and this one can, because it still has the marshal CONTEXT.
172    ///
173    /// The MAP half of every case here lives in the `write_support` module
174    /// below: [`CellPathComponent::MapKey`] is write-side only, so under
175    /// `--no-default-features` it does not exist to assert about. The SET half
176    /// is unconditional — it is the read path's own case.
177    #[test]
178    fn a_bare_inner_marshal_name_resolves_from_the_complete_declared_type() {
179        assert_eq!(
180            resolve_declared_cell_path_type(
181                "org.apache.cassandra.db.marshal.SetType(Int32Type)",
182                CellPathComponent::SetElement
183            ),
184            Some(CqlType::Int)
185        );
186    }
187
188    #[test]
189    fn both_spellings_resolve_a_set_element() {
190        for (declared, expected) in [
191            ("set<int>", CqlType::Int),
192            (
193                "org.apache.cassandra.db.marshal.SetType(org.apache.cassandra.db.marshal.Int32Type)",
194                CqlType::Int,
195            ),
196            ("set<decimal>", CqlType::Decimal),
197        ] {
198            assert_eq!(
199                resolve_declared_cell_path_type(declared, CellPathComponent::SetElement),
200                Some(expected.clone()),
201                "{declared} as a SET element"
202            );
203        }
204        // A MAP declaration has no SET element.
205        assert_eq!(
206            resolve_declared_cell_path_type("map<bigint,text>", CellPathComponent::SetElement),
207            None
208        );
209    }
210
211    /// THE PACKAGE RULE SURVIVES the widening: a foreign inner class is still
212    /// refused rather than read as the marshal type it resembles (#3631 roborev
213    /// job 76 / round 9 finding 2, #28).
214    #[test]
215    fn a_foreign_package_inner_name_is_refused_not_resolved() {
216        for declared in [
217            "org.apache.cassandra.db.marshal.SetType(com.acme.Int32Type)",
218            "org.apache.cassandra.db.marshal.SetType(notorg.apache.cassandra.db.marshal.Int32Type)",
219            "com.acme.SetType(org.apache.cassandra.db.marshal.Int32Type)",
220        ] {
221            assert_eq!(
222                resolve_declared_cell_path_type(declared, CellPathComponent::SetElement),
223                None,
224                "{declared} must NOT resolve to a marshal type"
225            );
226        }
227    }
228
229    /// A FROZEN collection has no CellPath at all, so it resolves to no
230    /// component here — its empty component is the inline case owned by
231    /// `require_fixed_width` (#3847/#4071).
232    #[test]
233    fn frozen_and_non_collection_declarations_resolve_no_set_element() {
234        for declared in FROZEN_AND_NON_COLLECTION {
235            assert_eq!(
236                resolve_declared_cell_path_type(declared, CellPathComponent::SetElement),
237                None,
238                "{declared} must resolve no SET element"
239            );
240        }
241    }
242
243    /// Shared by the set-side test above and its map-side sibling, so the two
244    /// halves cannot drift apart on which declarations they cover.
245    const FROZEN_AND_NON_COLLECTION: &[&str] = &[
246        "frozen<set<int>>",
247        "org.apache.cassandra.db.marshal.FrozenType(org.apache.cassandra.db.marshal.SetType(org.apache.cassandra.db.marshal.Int32Type))",
248        "int",
249        "list<int>",
250        "org.apache.cassandra.db.marshal.ListType(org.apache.cassandra.db.marshal.Int32Type)",
251        "",
252    ];

254    /// The MAP-KEY half. [`CellPathComponent::MapKey`] is constructed only by
255    /// `writer::data_writer::cell_path`, so it exists only under
256    /// `write-support` — and these assertions, including the
257    /// kinds-do-not-cross direction, are only meaningful where it does.
258    #[cfg(feature = "write-support")]
259    mod write_support {
260        use super::*;
261
262        #[test]
263        fn a_bare_inner_marshal_name_resolves_a_map_key() {
264            assert_eq!(
265                resolve_declared_cell_path_type(
266                    "org.apache.cassandra.db.marshal.MapType(Int32Type,Int32Type)",
267                    CellPathComponent::MapKey
268                ),
269                Some(CqlType::Int)
270            );
271        }
272
273        #[test]
274        fn the_component_kinds_do_not_cross() {
275            // A SET declaration has no MAP KEY ...
276            for declared in [
277                "set<int>",
278                "org.apache.cassandra.db.marshal.SetType(org.apache.cassandra.db.marshal.Int32Type)",
279                "set<decimal>",
280            ] {
281                assert_eq!(
282                    resolve_declared_cell_path_type(declared, CellPathComponent::MapKey),
283                    None,
284                    "{declared} must not resolve a MAP KEY"
285                );
286            }
287            // ... and a MAP declaration does.
288            assert_eq!(
289                resolve_declared_cell_path_type("map<bigint,text>", CellPathComponent::MapKey),
290                Some(CqlType::BigInt)
291            );
292        }
293
294        #[test]
295        fn frozen_and_non_collection_declarations_resolve_no_map_key() {
296            for declared in FROZEN_AND_NON_COLLECTION {
297                assert_eq!(
298                    resolve_declared_cell_path_type(declared, CellPathComponent::MapKey),
299                    None,
300                    "{declared} must resolve no MAP key"
301                );
302            }
303        }
304    }
