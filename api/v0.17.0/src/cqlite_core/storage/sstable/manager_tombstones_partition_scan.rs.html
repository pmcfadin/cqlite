1//! The `tombstones` build's partition-TARGETED materializing scans:
2//! [`SSTableManager::scan_partition`] and
3//! [`SSTableManager::scan_partition_with_cell_metadata`] (split out of `mod.rs` per
4//! the campsite rule, epic #1116).
5//!
6//! That build uses a structurally different reader map and has no bloom-prune
7//! `scan_partition` path, so a fully-constrained `WHERE pk = ?` is served by scanning
8//! the table and filtering to the partition key. The rows are byte-identical to what
9//! the pruned `not(tombstones)` path returns, which keeps the query executor free of
10//! any `tombstones` cfg branching — but the ACCESS PATH is a full scan, which is why
11//! both functions return `engaged == false` (Epic #951, honest access paths).
12//!
13//! # ONE meter per logical read, owned by the OUTER targeted API (issue #1701, R9-F1)
14//!
15//! These two functions are the read OPERATION; the full scan underneath them is an
16//! implementation detail of it. Letting the inner scan meter itself made a targeted
17//! single-partition read report EVERY row and EVERY partition of the table — a
18//! `WHERE pk = ?` over an 8-partition fixture reported `read.rows = 8` where the read
19//! delivered 1 — so the same metric meant different things in different feature
20//! builds (the `not(tombstones)` `scan_partition_clustering` meters the delivered
21//! partition) and contradicted the "rows a read DELIVERED" contract
22//! [`catalog::READ_ROWS`](crate::observability::catalog::READ_ROWS) states.
23//!
24//! So the inner scan is handed [`ReadOpMeter::inert`] — a CONSTRUCTION-time no-op,
25//! deliberately distinct from `discard()`, which is the RUNTIME decision a declining
26//! boundary makes — and the meter that reports lives here, started at FUNCTION ENTRY
27//! (the entry-at-function rule `manager_point_read`'s module doc records) and
28//! recording the rows AFTER the `retain`, i.e. exactly what the caller receives.
29
30#![cfg(feature = "tombstones")]
31
32use std::collections::HashMap;
33
34use super::SSTableManager;
35use crate::observability::read_metrics::ReadOpMeter;
36use crate::types::{ScanRow, TableId};
37use crate::{Result, RowKey};
38
39impl SSTableManager {
40    /// `tombstones`-build counterpart of [`scan_partition`](Self::scan_partition).
41    ///
42    /// That build uses a structurally different reader map and has no bloom-prune
43    /// `scan_partition` path, so a fully-constrained `WHERE pk = ?` is served by
44    /// scanning and filtering to the partition key. The output is a subset of
45    /// [`scan`](Self::scan) — identical to what the `not(tombstones)`
46    /// `scan_partition` returns — which keeps the query executor free of any
47    /// `tombstones` cfg branching.
48    ///
49    /// Returns `(rows, engaged)` with `engaged == false`: this is a full scan +
50    /// retain with NO SSTable prune, so the caller MUST report an honest fallback
51    /// access path (`FallbackReason::TombstonesBuildNoPrune`) rather than a targeted
52    /// label, even though the rows match the pruned build byte-for-byte (Epic #951).
53    pub async fn scan_partition(
54        &self,
55        table_id: &TableId,
56        partition_key: &[u8],
57        schema: Option<&crate::schema::TableSchema>,
58    ) -> Result<(Vec<(RowKey, ScanRow)>, bool)> {
59        // The meter for THIS logical read (module doc): started at entry so the
60        // scan's own latency is inside the reported duration, inert underneath so the
61        // full scan emits nothing, and recording the DELIVERED rows after `retain`.
62        let mut meter = ReadOpMeter::start(None);
63        let mut rows = self
64            .scan_with_meter(table_id, None, None, None, schema, ReadOpMeter::inert())
65            .await?;
66        rows.retain(|entry| entry.0.as_bytes() == partition_key);
67        meter.record_keys(rows.iter().map(|(k, ..)| k));
68        Ok((rows, false))
69    }
70
71    /// `tombstones`-build counterpart of
72    /// [`scan_partition_with_cell_metadata`](Self::scan_partition_with_cell_metadata).
73    ///
74    /// That build has no bloom-prune metadata path, so a fully-constrained
75    /// `WHERE pk = ?` WRITETIME/TTL read is served by scanning with metadata and
76    /// filtering to the partition key, matching the `not(tombstones)` output while
77    /// keeping the query executor free of `tombstones` cfg branching.
78    ///
79    /// Returns `(rows, engaged)` with `engaged == false`: this is a full metadata
80    /// scan + retain with NO SSTable prune, so the caller MUST report an honest
81    /// fallback access path (`FallbackReason::TombstonesBuildNoPrune`) rather than a
82    /// targeted label, even though the rows are byte-identical to the pruned build
83    /// (Epic #951, honest access paths).
84    pub async fn scan_partition_with_cell_metadata(
85        &self,
86        table_id: &TableId,
87        partition_key: &[u8],
88        schema: Option<&crate::schema::TableSchema>,
89    ) -> Result<(
90        Vec<(
91            RowKey,
92            ScanRow,
93            HashMap<String, crate::types::CellWriteMetadata>,
94        )>,
95        bool,
96    )> {
97        // Same ownership rule as `scan_partition` above (module doc).
98        let mut meter = ReadOpMeter::start(None);
99        let mut rows = self
100            .scan_with_cell_metadata_with_meter(
101                table_id,
102                None,
103                None,
104                None,
105                schema,
106                ReadOpMeter::inert(),
107            )
108            .await?;
109        rows.retain(|entry| entry.0.as_bytes() == partition_key);
110        meter.record_keys(rows.iter().map(|(k, ..)| k));
111        Ok((rows, false))
112    }
113}
