1//! Cassandra OSS50 **byte-comparable** clustering-bound encoding.
2//!
3//! These encoders reproduce the on-disk separator bytes the `Rows.db` row-index
4//! trie stores (the SAME encoding), NOT CQLite's custom
5//! [`ByteComparableEncoder`](crate::storage::sstable::bti::encoder::ByteComparableEncoder)
6//! (which prepends a 1-byte type discriminator and so is byte-incompatible with
7//! the on-disk trie).
8
9use crate::{error::Error, storage::sstable::bti::node::BtiResult, types::Value};

11/// Cassandra component separator used between clustering components in the
12/// OSS50 byte-comparable form.  Mirrors `ByteSource.NEXT_COMPONENT` (value
13/// `0x40`) emitted by `ClusteringComparator.asByteComparable` /
14/// `ByteSource.withTerminator`/`ByteSource.of(...)` in
15/// `org.apache.cassandra.utils.bytecomparable.ByteSource` (cassandra-5.0.0).
16const OSS50_NEXT_COMPONENT: u8 = 0x40;
17
18/// `ByteSource.ESCAPE` (cassandra-5.0.8
19/// `utils/bytecomparable/ByteSource.java:48`): both the escape introducer for a
20/// literal `0x00` in the data AND the component terminator.
21const OSS50_ESCAPE: u8 = 0x00;
22/// `ByteSource.ESCAPED_0_CONT` (`ByteSource.java:51`): emitted for a run of
23/// literal zeros, and as the TERMINATOR of a component whose data ends in `0x00`.
24const OSS50_ESCAPED_0_CONT: u8 = 0xFE;
25/// `ByteSource.ESCAPED_0_DONE` (`ByteSource.java:52`): closes an escape sequence
26/// when the byte following the literal zero(s) is NOT itself a zero.
27const OSS50_ESCAPED_0_DONE: u8 = 0xFF;
28
29/// OSS50 variable-length byte-comparable encoding for `BytesType`/`UTF8Type` —
30/// `ByteSource.AbstractEscaper` in cassandra-5.0.8
31/// (`utils/bytecomparable/ByteSource.java:309-380`, `Version.OSS50`).
32///
33/// The escaper's contract, quoting the pinned source's own doc and worked
34/// examples: "Escapes 0s as ESCAPE + zero or more ESCAPED_0_CONT +
35/// ESCAPED_0_DONE. If the source ends in 0, we use ESCAPED_0_CONT to make sure
36/// that the encoding remains smaller than that source with a further 0 at the
37/// end."
38///
39/// ```text
40/// "A\0\0B" → 41 00 FE FF 42 00
41/// "A\0B\0" → 41 00 FF 42 00 FE
42/// "A\0"    → 41 00 FE
43/// "AB"     → 41 42 00
44/// ```
45///
46/// So the terminator of a component whose data does NOT end in `0x00` is a
47/// SINGLE `ESCAPE` byte — not `ESCAPE + ESCAPED_0_DONE`. That distinction is
48/// load-bearing: the terminator must sort BELOW the `NEXT_COMPONENT` (`0x40`)
49/// byte that introduces the following clustering component, so that a shorter
50/// component sorts before a longer one sharing its prefix ("any permitted
51/// separator byte will be smaller than the byte following the prefix"). A
52/// two-byte `00 FF` terminator sorts ABOVE `00 40`, i.e. above EVERY key in the
53/// subtree it is meant to bound — which silently truncated a `text`-clustering
54/// slice to the tail of its bucket (issue #3032; a real Cassandra 5.0 `Rows.db`
55/// separator for `('bo', 12)` is `40 62 6f 00 40 80 00 00 0c`, terminator `00`).
56///
57/// ## The EMPTY component (issue #3032 roborev N8)
58///
59/// An empty variable-length component encodes to a bare terminator — one `0x00` —
60/// so a full empty `text` clustering component is `40 00`. This is NOT a guess: it
61/// falls straight out of `AbstractEscaper.next()` (`bufpos == limit == 0`, not
62/// `escaped`, so the first call returns `ESCAPE` and the second `END_OF_STREAM`),
63/// and cassandra-5.0.8 `ClusteringComparator.java:271-273` states it outright in its
64/// worked examples: `"", null, Clustering -> 40 00 3F 40` and
65/// `"", 0000, Clustering -> 40 00 40 0000 40` — the empty first component is `40 00`
66/// in both.
67///
68/// `ByteSource.NEXT_COMPONENT_EMPTY` (`0x3F`, the `3F` above) is therefore **NOT**
69/// the encoding of an empty variable-length component. It replaces the leading
70/// `0x40` only when `AbstractType.asComparableBytes` returns Java `null`, which
71/// happens for a type whose production is `optionalSignedFixedLengthNumber` /
72/// `optionalFixedLength` (`ByteSource.java:176-179`, `:740-743`: `null` when
73/// `accessor.isEmpty(data)`) — i.e. an empty BUFFER for a fixed-length numeric.
74/// `ClusteringComparator.java:328-329` says so explicitly: "null values for some
75/// types (e.g. int, varint **but not text**) that are encoded as empty buffers".
76/// Text/blob go through `ByteSource.of(accessor, data, version)`
77/// (`ByteSource.java:91-94`), an `AccessorEscaper` that is never `null`.
78///
79/// That `0x3F` case is unreachable from this encoder by construction: a
80/// [`Value::Integer`]/[`Value::BigInt`]/… always carries a full-width number, so
81/// CQLite's value model cannot express the empty fixed-length buffer that produces
82/// it. If a representable empty-fixed-length value is ever added, this function's
83/// callers must emit `NEXT_COMPONENT_EMPTY` in place of `NEXT_COMPONENT` for it.
84///
85/// (The project's `ByteComparableEncoder` uses a different, non-Cassandra escape
86/// scheme and must NOT be used for trie-compatible bounds.)
87fn encode_varlen_oss50(bytes: &[u8], out: &mut Vec<u8>) {
88    // Mirrors `AbstractEscaper.next()`'s `escaped` state machine exactly.
89    let mut escaped = false;
90    for &b in bytes {
91        if escaped {
92            if b == OSS50_ESCAPE {
93                // A further zero continues the escape run (and is REPLACED by the
94                // continuation byte — the literal zero is not re-emitted).
95                out.push(OSS50_ESCAPED_0_CONT);
96                continue;
97            }
98            // A non-zero byte closes the run, then is emitted unchanged.
99            out.push(OSS50_ESCAPED_0_DONE);
100            escaped = false;
101        }
102        out.push(b);
103        if b == OSS50_ESCAPE {
104            escaped = true;
105        }
106    }
107    // End of data: a still-open escape run terminates with ESCAPED_0_CONT
108    // (keeping the encoding below that of the same value plus a trailing zero);
109    // otherwise the terminator is a single ESCAPE byte.
110    out.push(if escaped {
111        OSS50_ESCAPED_0_CONT
112    } else {
113        OSS50_ESCAPE
114    });
115}

117/// Encode a single clustering-component [`Value`] in Cassandra OSS50
118/// **byte-comparable** form (the SAME encoding the `Rows.db` row-index trie
119/// stores its separators in), appending to `out`.
120///
121/// This reproduces the per-type `AbstractType.asComparableBytes(...)` production
122/// used by Cassandra to build the byte-comparable keys:
123///
124/// - `Int32Type` → `(v as u32 ^ 0x8000_0000)` big-endian (sign-flip, 4B).
125/// - `LongType` → `(v as u64 ^ 0x8000_0000_0000_0000)` big-endian (8B).
126/// - `ShortType`/`ByteType` (`smallint`/`tinyint`) → sign-flip big-endian.
127/// - `BooleanType` → single byte `0x00`/`0x01`.
128/// - `TimestampType` (`timestamp`) → `LongType`-style sign-flip 8B.
129/// - `UUIDType`/`TimeUUIDType` → the 16 raw bytes.
130/// - `UTF8Type`/`AsciiType` and `BytesType` (`blob`) → raw value bytes,
131///   terminated by the OSS50 variable-length component terminator.
132///
133/// Any clustering type not enumerated here returns an explicit parse error
134/// (NO silent wrong results — issue #28 no-heuristics mandate).
135fn encode_clustering_component_oss50(value: &Value, out: &mut Vec<u8>) -> BtiResult<()> {
136    match value {
137        // int — Int32Type: sign-flip, big-endian (matches `wide_table` separators,
138        // e.g. ck=8 → 80 00 00 08).
139        Value::Integer(v) => {
140            out.extend_from_slice(&((*v as u32) ^ 0x8000_0000).to_be_bytes());
141            Ok(())
142        }
143        // bigint — LongType: sign-flip, big-endian, 8 bytes.
144        Value::BigInt(v) | Value::Counter(v) => {
145            out.extend_from_slice(&((*v as u64) ^ 0x8000_0000_0000_0000).to_be_bytes());
146            Ok(())
147        }
148        // smallint — ShortType: sign-flip, big-endian, 2 bytes.
149        Value::SmallInt(v) => {
150            out.extend_from_slice(&((*v as u16) ^ 0x8000).to_be_bytes());
151            Ok(())
152        }
153        // tinyint — ByteType: sign-flip, 1 byte.
154        Value::TinyInt(v) => {
155            out.push((*v as u8) ^ 0x80);
156            Ok(())
157        }
158        // boolean — BooleanType: single 0x00/0x01 byte.
159        Value::Boolean(b) => {
160            out.push(if *b { 0x01 } else { 0x00 });
161            Ok(())
162        }
163        // timestamp — TimestampType shares LongType's comparable form (8-byte
164        // sign-flip big-endian of the millisecond value).
165        Value::Timestamp(v) => {
166            out.extend_from_slice(&((*v as u64) ^ 0x8000_0000_0000_0000).to_be_bytes());
167            Ok(())
168        }
169        // uuid / timeuuid — raw 16 bytes (already byte-comparable in the on-disk
170        // separator form for this fixed-length type).
171        Value::Uuid(bytes) => {
172            out.extend_from_slice(bytes);
173            Ok(())
174        }
175        // text / ascii — OSS50 variable-length byte-comparable encoding.
176        Value::Text(s) => {
177            encode_varlen_oss50(s.as_ref(), out);
178            Ok(())
179        }
180        // blob / inet — OSS50 variable-length byte-comparable encoding.
181        Value::Blob(b) | Value::Inet(b) => {
182            encode_varlen_oss50(b, out);
183            Ok(())
184        }
185        other => Err(Error::Parse(format!(
186            "BTI range_query: byte-comparable encoding not implemented for {:?}",
187            other.data_type()
188        ))),
189    }
190}

192/// Encode a multi-component clustering bound (`&[Value]`) in Cassandra OSS50
193/// byte-comparable form — the SAME encoding the `Rows.db` trie stores.
194///
195/// A [`OSS50_NEXT_COMPONENT`] (`0x40`) byte precedes **EVERY** component,
196/// including the FIRST: `ClusteringComparator.ByteComparableClustering`
197/// (cassandra-5.0.8 `ClusteringComparator.java:257-275`) "adds a NEXT_COMPONENT
198/// byte before each component ... and finishes with a suitable byte for the
199/// clustering kind", e.g. a COMPLETE `("A", 0005)` clustering →
200/// `40 4100 40 0005 38`, whose last byte is the `Kind` byte
201/// `ClusteringPrefix.Kind.CLUSTERING.asByteComparableValue(OSS50)` =
202/// `ByteSource.TERMINATOR` (`0x38`), NOT another `NEXT_COMPONENT`. (Cassandra's own
203/// javadoc renders that example as `... 0005 40` because it is written for
204/// `Version.LEGACY`, where `CLUSTERING` maps to `ByteSource.NEXT_COMPONENT`
205/// instead — `ClusteringPrefix.java:75-76`.) So a
206/// single `int` clustering ck=8 encodes to `40 80 00 00 08`, which is exactly the
207/// separator byte string a real `wide_table` `Rows.db` trie stores (issue #3002 —
208/// the previous "bare component bytes, NO framing" claim was calibrated against a
209/// mis-rooted traversal that never saw the root's `0x40` transition).
210///
211/// No TERMINATOR (`0x38`) is appended: row-index separators are `separatorGt` /
212/// `nudge` PREFIXES of a clustering (`RowIndexWriter.add`/`complete`), not
213/// complete clusterings, so no `ByteSource.TERMINATOR` is present on disk. A
214/// prefix bound therefore still sorts before any longer key sharing it.
215pub fn encode_clustering_bound_oss50(values: &[Value]) -> BtiResult<Vec<u8>> {
216    // All-ascending convenience: every component uses its base byte-comparable
217    // form. Equivalent to `encode_clustering_bound_oss50_with_order` with all
218    // `is_reversed = false`.
219    let mut out = Vec::new();
220    for v in values {
221        out.push(OSS50_NEXT_COMPONENT);
222        encode_clustering_component_oss50(v, &mut out)?;
223    }
224    Ok(out)
225}

227/// Encode a multi-component clustering bound applying each column's clustering
228/// **ORDER** (ASC/DESC), producing Cassandra OSS50 byte-comparable separators
229/// that sort in the SAME order the rows are physically written.
230///
231/// ## Why DESC must invert the component bytes
232///
233/// Cassandra wraps a `CLUSTERING ORDER BY (c DESC)` column's type in
234/// `ReversedType`. `ReversedType.asComparableBytes` delegates to the base type's
235/// byte-comparable production and then inverts the resulting `ByteSource` via
236/// `ByteSource.invert(...)`, which complements every emitted data byte
237/// (`b -> 0xFF ^ b`). Complementing every byte of a (weakly prefix-free)
238/// byte-comparable encoding reverses its lexicographic order, so a DESCENDING
239/// value order maps to an ASCENDING byte order. This is exactly what a `Rows.db`
240/// row-index trie needs: separators are always stored in ascending *byte* order,
241/// while the underlying clustering values run descending for a DESC column.
242/// (cassandra-5.0.0 `org.apache.cassandra.db.marshal.ReversedType.asComparableBytes`
243/// + `org.apache.cassandra.utils.bytecomparable.ByteSource.invert`.)
244///
245/// The framing byte ([`OSS50_NEXT_COMPONENT`], `0x40`) — emitted before EVERY
246/// component including the first, per
247/// `ClusteringComparator.ByteComparableClustering` — comes from the *comparator*,
248/// NOT from a component's type, so it is **not** inverted even when the component
249/// it precedes is DESC; only the per-component byte-comparable bytes are
250/// complemented. This matches Cassandra, where each `subtype(i)` (possibly a
251/// `ReversedType`) emits its own (already inverted) byte source and the comparator
252/// prefixes each with the un-inverted `NEXT_COMPONENT` byte.
253///
254/// `is_reversed[i]` MUST correspond positionally to `values[i]` (i.e. the schema
255/// clustering-key order). A short `is_reversed` slice treats missing entries as
256/// ascending (`false`).
257pub fn encode_clustering_bound_oss50_with_order(
258    values: &[Value],
259    is_reversed: &[bool],
260) -> BtiResult<Vec<u8>> {
261    let mut out = Vec::new();
262    for (i, v) in values.iter().enumerate() {
263        out.push(OSS50_NEXT_COMPONENT);
264        if is_reversed.get(i).copied().unwrap_or(false) {
265            // Encode the base component into a scratch buffer, then complement
266            // every byte (ReversedType / ByteSource.invert).
267            let mut scratch = Vec::new();
268            encode_clustering_component_oss50(v, &mut scratch)?;
269            for b in &scratch {
270                out.push(0xFF ^ *b);
271            }
272        } else {
273            encode_clustering_component_oss50(v, &mut out)?;
274        }
275    }
276    Ok(out)
277}

283    /// The OSS50 byte-comparable clustering encoder (issue #832 Finding 1)
284    /// reproduces the on-disk trie separator bytes — including the leading
285    /// `0x40 NEXT_COMPONENT` byte every component carries (issue #3002; the real
286    /// `wide_table` `Rows.db` separator for ck=8 is `40 80 00 00 08`).
287    #[test]
288    fn oss50_clustering_encoder() {
289        // Single int component — 0x40 NEXT_COMPONENT + sign-flip big-endian.
290        assert_eq!(
291            encode_clustering_bound_oss50(&[Value::Integer(8)]).unwrap(),
292            vec![0x40, 0x80, 0x00, 0x00, 0x08]
293        );
294        assert_eq!(
295            encode_clustering_bound_oss50(&[Value::Integer(-1)]).unwrap(),
296            vec![0x40, 0x7F, 0xFF, 0xFF, 0xFF]
297        );
298        // Ordering is preserved: -1 < 0 < 100 byte-comparably.
299        let neg = encode_clustering_bound_oss50(&[Value::Integer(-1)]).unwrap();
300        let zero = encode_clustering_bound_oss50(&[Value::Integer(0)]).unwrap();
301        let pos = encode_clustering_bound_oss50(&[Value::Integer(100)]).unwrap();
302        assert!(neg < zero && zero < pos);
303
304        // bigint — 8-byte sign-flip BE behind the 0x40 framing byte.
305        assert_eq!(
306            encode_clustering_bound_oss50(&[Value::BigInt(1)]).unwrap(),
307            vec![0x40, 0x80, 0, 0, 0, 0, 0, 0, 0x01]
308        );
309
310        // Multi-component: 0x40 before EACH of int(1) and text("ab") (the
311        // `("A", 0005) -> 40 4100 40 0005` worked example in
312        // `ClusteringComparator.ByteComparableClustering`), the text component
313        // terminated by a SINGLE `ByteSource.ESCAPE` (0x00) byte.
314        assert_eq!(
315            encode_clustering_bound_oss50(&[Value::Integer(1), Value::text("ab".to_string())])
316                .unwrap(),
317            vec![0x40, 0x80, 0x00, 0x00, 0x01, 0x40, b'a', b'b', 0x00]
318        );
319
320        // Variable-length OSS50 — the four worked examples in the pinned
321        // cassandra-5.0.8 `ByteSource.AbstractEscaper` javadoc
322        // (`utils/bytecomparable/ByteSource.java:314-317`), which are the
323        // AUTHORITY for this encoding (issue #3032):
324        //
325        //     "A\0\0B" -> 41 00 FE FF 42 00
326        //     "A\0B\0" -> 41 00 FF 42 00 FE
327        //     "A\0"    -> 41 00 FE
328        //     "AB"     -> 41 42 00
329        for (data, expected) in [
330            (
331                vec![b'A', 0x00, 0x00, b'B'],
332                vec![0x41u8, 0x00, 0xFE, 0xFF, 0x42, 0x00],
333            ),
334            (
335                vec![b'A', 0x00, b'B', 0x00],
336                vec![0x41u8, 0x00, 0xFF, 0x42, 0x00, 0xFE],
337            ),
338            (vec![b'A', 0x00], vec![0x41u8, 0x00, 0xFE]),
339            (vec![b'A', b'B'], vec![0x41u8, 0x42, 0x00]),
340        ] {
341            let mut got = Vec::new();
342            encode_varlen_oss50(&data, &mut got);
343            assert_eq!(
344                got, expected,
345                "AbstractEscaper worked example for {data:02x?}"
346            );
347        }
348
349        // Same, through the public clustering-bound entry point (framing byte
350        // included). A `text` component terminates with a LONE 0x00 — the real
351        // Cassandra 5.0 `Rows.db` separator for the compound clustering
352        // `('bo', 12)` is `40 62 6f 00 40 80 00 00 0c` (issue #3032 fixture
353        // `test_da/multiclustering_table`), which a `00 FF` terminator would sort
354        // ABOVE instead of below.
355        assert_eq!(
356            encode_clustering_bound_oss50(&[Value::text("a".to_string())]).unwrap(),
357            vec![0x40, b'a', 0x00]
358        );
359        assert_eq!(
360            encode_clustering_bound_oss50(&[Value::text("bo".to_string())]).unwrap(),
361            vec![0x40, b'b', b'o', 0x00]
362        );
363        assert_eq!(
364            encode_clustering_bound_oss50(&[Value::blob(vec![0x01, 0x00, 0x02])]).unwrap(),
365            vec![0x40, 0x01, 0x00, 0xFF, 0x02, 0x00]
366        );
367        // Weakly prefix-free ordering: "a" sorts before "ab", and a text bound
368        // sorts BELOW every longer clustering that extends it — the terminator
369        // (0x00) is smaller than the NEXT_COMPONENT byte (0x40) that would
370        // introduce the following component.
371        let a = encode_clustering_bound_oss50(&[Value::text("a".to_string())]).unwrap();

374        let bo = encode_clustering_bound_oss50(&[Value::text("bo".to_string())]).unwrap();
375        let bo_12 =
376            encode_clustering_bound_oss50(&[Value::text("bo".to_string()), Value::Integer(12)])
377                .unwrap();
378        assert!(
379            bo < bo_12,
380            "a prefix bound on the FIRST clustering component must sort below every \

382        );
383        // And a trailing literal zero still sorts above the same value without it.
384        let z0 = encode_clustering_bound_oss50(&[Value::blob(vec![b'A'])]).unwrap();
385        let z1 = encode_clustering_bound_oss50(&[Value::blob(vec![b'A', 0x00])]).unwrap();
386        let z2 = encode_clustering_bound_oss50(&[Value::blob(vec![b'A', 0x00, 0x00])]).unwrap();
387        assert!(z0 < z1 && z1 < z2, "{z0:02x?} < {z1:02x?} < {z2:02x?}");
388
389        // Unsupported clustering type errors out explicitly.
390        assert!(encode_clustering_bound_oss50(&[Value::Float(1.0)]).is_err());

393    /// Order-aware OSS50 encoder (DESC clustering): a `CLUSTERING ORDER BY DESC`
394    /// column wraps in Cassandra `ReversedType`, whose `asComparableBytes`

396    #[test]
397    fn oss50_clustering_encoder_reversed_order() {
398        // DESC int(8): base 80 00 00 08, complemented -> 7F FF FF F7, behind the
399        // UN-inverted 0x40 framing byte (emitted by the comparator, #3002).
400        assert_eq!(
401            encode_clustering_bound_oss50_with_order(&[Value::Integer(8)], &[true]).unwrap(),
402            vec![0x40, 0x7F, 0xFF, 0xFF, 0xF7]

407            encode_clustering_bound_oss50(&[Value::Integer(8)]).unwrap()

411        let enc = |v: i32| {

413        };
414        // Physical write order for DESC: 5,4,3,2,1,0. Their separator bytes must

419                enc(w[0]) < enc(w[1]),

421                w[0],
422                w[1]

424        }
425
426        // Mixed ASC/DESC: first column ASC int, second column DESC int.
427        let mixed = encode_clustering_bound_oss50_with_order(
428            &[Value::Integer(1), Value::Integer(8)],

430        )
431        .unwrap();
432        assert_eq!(
433            mixed,
434            vec![0x40, 0x80, 0x00, 0x00, 0x01, 0x40, 0x7F, 0xFF, 0xFF, 0xF7],
435            "0x40 framing before EACH component (un-inverted), DESC component complemented"
436        );
437
438        // Mixed-order monotonicity.
439        let m = |a: i32, b: i32| {
440            encode_clustering_bound_oss50_with_order(
441                &[Value::Integer(a), Value::Integer(b)],
442                &[false, true],
443            )
444            .unwrap()
445        };
446        // Same first key (1), second key descending 9,8,7 -> ascending bytes.
447        assert!(m(1, 9) < m(1, 8) && m(1, 8) < m(1, 7));
448        // Different first key dominates regardless of second (ASC leading).
449        assert!(m(1, 0) < m(2, 9));

452    /// An EMPTY variable-length component encodes to a bare terminator: `40 00`
453    /// (issue #3032 roborev N8). NOT `ByteSource.NEXT_COMPONENT_EMPTY` (`0x3F`).
454    ///
455    /// Authority (pinned cassandra-5.0.8, never CQLite's own behavior):
456    ///  * `ByteSource.AbstractEscaper.next()` on an empty buffer — `bufpos == limit
457    ///    == 0`, not `escaped` — returns `ESCAPE` (`0x00`) once, then `END_OF_STREAM`.
458    ///  * `ClusteringComparator.java:271-272` states the result in its own worked
459    ///    examples: `"", null, Clustering -> 40 00 3F 40` and
460    ///    `"", 0000, Clustering -> 40 00 40 0000 40`.
461    ///  * `0x3F` there is the SECOND component's marker, emitted only when
462    ///    `asComparableBytes` returns Java `null` — an empty buffer for a
463    ///    fixed-length numeric (`ByteSource.java:176-179`). Text/blob use
464    ///    `AccessorEscaper`, which is never `null`
465    ///    (`ClusteringComparator.java:328-329`: "e.g. int, varint but not text").
466    #[test]
467    fn oss50_empty_varlen_component_is_a_bare_terminator() {
468        // text "" and blob b"" both encode to NEXT_COMPONENT + a lone ESCAPE.
469        assert_eq!(
470            encode_clustering_bound_oss50(&[Value::Text("".into())]).unwrap(),
471            vec![0x40, 0x00]
472        );
473        assert_eq!(
474            encode_clustering_bound_oss50(&[Value::Blob(Vec::new().into())]).unwrap(),
475            vec![0x40, 0x00]
476        );
477
478        // The `"", 0000, Clustering` shape from the Cassandra javadoc: the empty
479        // leading component is `40 00`, then the int component's own `0x40` framing
480        // (`0000` there is schematic; CQLite emits the sign-flipped Int32Type form).
481        assert_eq!(
482            encode_clustering_bound_oss50(&[Value::Text("".into()), Value::Integer(0)]).unwrap(),
483            vec![0x40, 0x00, 0x40, 0x80, 0x00, 0x00, 0x00]
484        );
485
486        // The load-bearing ORDER property: an empty component sorts below every
487        // non-empty one, and below a longer key sharing it as a prefix.
488        let empty = encode_clustering_bound_oss50(&[Value::Text("".into())]).unwrap();
489        let a = encode_clustering_bound_oss50(&[Value::Text("a".into())]).unwrap();
490        let nul = encode_clustering_bound_oss50(&[Value::Text("\u{0}".into())]).unwrap();
491        assert!(empty < a, "empty text must sort before \"a\"");
492        assert!(empty < nul, "empty text must sort before a single NUL");
493        let empty_then_int =
494            encode_clustering_bound_oss50(&[Value::Text("".into()), Value::Integer(0)]).unwrap();
495        assert!(
496            empty < empty_then_int,
497            "the empty component's terminator must sort below the following 0x40"
498        );
499    }
