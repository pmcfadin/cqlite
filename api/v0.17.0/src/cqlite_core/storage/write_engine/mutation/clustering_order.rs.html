1//! Clustering-value ordering for the write engine — `compare_values`.
2//!
3//! Split out of `mutation.rs` under the campsite rule (epic #1116): that file is
4//! far over the 800-line source target, and this comparator carries the long
5//! per-type authority notes (float total order, and the `time`/`timestamp`
6//! asymmetry) that #3935 required.
7//!
8//! A CHILD of `mutation` rather than a sibling under `write_engine`, on purpose:
9//! `write_engine/mod.rs` is itself ~3560 lines, so adding a `mod` line there
10//! would grow an over-threshold file and the `file-size` ratchet would (rightly)
11//! FAIL. Declaring it from `mutation.rs`, which this split leaves ~70 lines
12//! SMALLER than before, costs the ratchet nothing — and `ClusteringKey` is its
13//! only caller anyway.
14//!
15//! This is the comparator BOTH routes to clustering order reach:
16//! [`ClusteringKey::compare`](super::ClusteringKey::compare) (schema-aware,
17//! used by `write_engine::merge` to sort merged rows "for output order") and
18//! `ClusteringKey`'s `Ord` (the memtable `BTreeMap` key order). It therefore
19//! REQUIRES a strict total order — see the per-arm notes.
20
21use std::cmp::Ordering;
22
23use crate::types::Value;
24use crate::{Error, Result};
25
26/// Compare two values for ordering
27pub(super) fn compare_values(a: &Value, b: &Value) -> Result<Ordering> {
28    use Value::*;
29
30    match (a, b) {
31        (Null, Null) => Ok(Ordering::Equal),
32        (Null, _) => Ok(Ordering::Less),
33        (_, Null) => Ok(Ordering::Greater),
34
35        (Boolean(a), Boolean(b)) => Ok(a.cmp(b)),
36        (TinyInt(a), TinyInt(b)) => Ok(a.cmp(b)),
37        (SmallInt(a), SmallInt(b)) => Ok(a.cmp(b)),
38        (Integer(a), Integer(b)) => Ok(a.cmp(b)),
39        (BigInt(a), BigInt(b)) => Ok(a.cmp(b)),
40        (Counter(a), Counter(b)) => Ok(a.cmp(b)),
41        // Cassandra/Java total order (NaN last, -0.0 < +0.0) — NOT IEEE
42        // partial_cmp. This feeds ClusteringKey's `Ord`/`compare` (memtable
43        // BTreeMap key order + compaction merge), which requires a TOTAL order:
44        // a non-total order would let NaN compare Equal to everything
45        // (transitivity violation) and collapse -0.0/+0.0. See float_cmp.rs and
46        // issues #1870/#2010. Must agree with the reader's Value::partial_cmp,
47        // which since #3935 it does for EVERY arm, `Time` included, and with
48        // `types::comparator::custom::compare_time`.
49        //
50        // ONE DECLARED EXCEPTION to that "EVERY arm" claim, recorded because the
51        // sentence above is a claim about code and would otherwise be false as
52        // written (issue #3805, lead ruling Q2): this function does NOT handle
53        // `Value::Empty`, the empty-buffer sentinel, which falls through to the
54        // mismatched-variant `Err` below while `Value::partial_cmp` sorts it
55        // strictly FIRST. Deliberate, not an oversight: `Value::Empty` arises
56        // from a multicell collection's CELL PATH (a map key), and a CLUSTERING
57        // value never decodes to it, so the divergent input cannot occur here --
58        // adding an untested arm to this hot merge path to satisfy an
59        // unreachable invariant is the worse trade. IF a clustering value ever
60        // CAN decode to `Value::Empty` (an empty clustering component is legal
61        // to Cassandra's storage engine -- see `cql3/UpdateParameters.java:88-101`
62        // at `cassandra-5.0.8`, which forbids it in CQL for COMPACT TABLES only),
63        // this function needs an `Empty` arm ordering it FIRST, matching
64        // `Int32Type.compareCustom` (`db/marshal/Int32Type.java:61-71`) and
65        // `Value::partial_cmp`.
66        (Float32(a), Float32(b)) => Ok(crate::float_cmp::cassandra_float_cmp(*a, *b)),
67        (Float(a), Float(b)) => Ok(crate::float_cmp::cassandra_double_cmp(*a, *b)),
68        (Text(a), Text(b)) => Ok(a.cmp(b)),
69        (Blob(a), Blob(b)) => Ok(a.cmp(b)),
70        // `timestamp` is `TimestampType` = ComparisonType.CUSTOM, whose
71        // `compareCustom` delegates to `LongType.compareLongs` — SIGNED.
72        (Timestamp(a), Timestamp(b)) => Ok(a.cmp(b)),
73        (Date(a), Date(b)) => Ok(a.cmp(b)),
74        // `time` is `TimeType` = ComparisonType.BYTE_ORDER (pinned
75        // `cassandra-5.0.8` `db/marshal/TimeType.java`), i.e.
76        // `ByteBufferUtil.compareUnsigned` over the 8-byte big-endian
77        // nanos-of-day. NOT signed `i64::cmp`: the two agree over `time`'s valid
78        // range (`0..=86_399_999_999_999`) and diverge for an out-of-range
79        // NEGATIVE nanos, which Cassandra's own binary `validate` accepts
80        // (see `types::comparator::custom::compare_time` for the canonical
81        // statement, and #3935 for why validation is NOT the fix).
82        //
83        // TOTAL-ORDER SAFETY: this arm feeds `ClusteringKey`'s `Ord`/`compare`,
84        // hence the memtable `BTreeMap` key order, the compaction merge order
85        // and the physical `Data.db` row order — all of which REQUIRE a strict
86        // total order. Unsigned lexicographic comparison of a FIXED 8-byte array
87        // is trivially total (antisymmetric, transitive, no incomparable pair),
88        // so this is safe; the total-order hazard lives in the float arms above.
89        (Time(a), Time(b)) => Ok(a.to_be_bytes().cmp(&b.to_be_bytes())),
90        (Uuid(a), Uuid(b)) => Ok(a.cmp(b)),
91        (Inet(a), Inet(b)) => Ok(a.cmp(b)),
92
93        // Collection types (element-wise lexicographic comparison)
94        (List(a), List(b)) | (Set(a), Set(b)) => {
95            for (elem_a, elem_b) in a.iter().zip(b.iter()) {
96                let ord = compare_values(elem_a, elem_b)?;
97                if ord != Ordering::Equal {
98                    return Ok(ord);
99                }
100            }
101            Ok(a.len().cmp(&b.len()))
102        }
103        (Map(a), Map(b)) => {
104            for ((ka, va), (kb, vb)) in a.iter().zip(b.iter()) {
105                let key_ord = compare_values(ka, kb)?;
106                if key_ord != Ordering::Equal {
107                    return Ok(key_ord);
108                }
109                let val_ord = compare_values(va, vb)?;
110                if val_ord != Ordering::Equal {
111                    return Ok(val_ord);
112                }
113            }
114            Ok(a.len().cmp(&b.len()))
115        }
116        (Tuple(a), Tuple(b)) => {
117            for (fa, fb) in a.iter().zip(b.iter()) {
118                let ord = compare_values(fa, fb)?;
119                if ord != Ordering::Equal {
120                    return Ok(ord);
121                }
122            }
123            Ok(a.len().cmp(&b.len()))
124        }
125
126        // Frozen wrapper: compare inner values
127        (Frozen(a), Frozen(b)) => compare_values(a, b),
128
129        _ => Err(Error::InvalidInput(format!(
130            "Cannot compare values of different types: {:?} vs {:?}",
131            a, b
132        ))),
133    }
134}
