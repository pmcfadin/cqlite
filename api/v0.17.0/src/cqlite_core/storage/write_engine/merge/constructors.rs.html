1//! `KWayMerger`'s opening constructors, and the ONE place a caller states who counts
2//! a failed input open (issues #1037, #1704).
3//!
4//! Split out of `merge/mod.rs` per the campsite rule (#1116): that file is 12,655
5//! lines against the ~800-line target, so the parameter issue #1704 needed could not
6//! be added there. The public constructor below is a VERBATIM move; the
7//! reporting-aware one it delegates to is new.
8//!
9//! # Why the caller states it, rather than the callee assuming it
10//!
11//! Every input is REOPENED inside its producer thread. A recording open
12//! double-counts when the caller's own seam also records (issue #1704, roborev C on
13//! the streaming read path); a NON-recording open loses the signal entirely when the
14//! caller has no seam — and `KWayMerger` is externally reachable public API
15//! (`cqlite_core::storage::write_engine::merge::KWayMerger::new`), so "wrap every
16//! caller" has no complete form: there is always one more call site outside this
17//! crate. The callee therefore cannot assume either way, and the mode is an explicit
18//! parameter of the internal constructor. Everything public keeps
19//! [`OpenErrorReporting::SelfReported`] — today's behaviour — so a new caller that
20//! does not think about it gets a visible extra increment, never silence.
21
22use std::collections::BinaryHeap;
23use std::path::PathBuf;
24
25use super::{egress_budget, KWayMerger, RunReader, SSTableRowIteratorAdapter};
26use crate::schema::TableSchema;
27use crate::storage::sstable::reader::OpenErrorReporting;
28use crate::{Error, Result};

30impl KWayMerger {
31    /// K-way merge constructor that opens the input SSTables under a cooperative
32    /// [`ScanCancel`](crate::storage::scan_cancel::ScanCancel) (issue #2264).
33    ///
34    /// The token is wired onto every per-run reader so the compaction scan each
35    /// run's producer thread drives — which, for an index-less (Summary.db
36    /// absent) SSTable, otherwise fully materialises the whole Data.db in one
37    /// uninterruptible pass — polls it at a bounded interval and abandons the walk
38    /// promptly when a driving Flight `do_get` is cancelled. `new`/`new_with_gc*`
39    /// delegate here with a never-cancelled default token, so non-Flight callers
40    /// are unaffected.
41    ///
42    /// Failed input opens are [`OpenErrorReporting::SelfReported`]: this is public API
43    /// and its callers are not required to have a recording seam (issue #1704).
44    pub fn new_with_gc_and_registry_cancellable(
45        input_paths: Vec<PathBuf>,
46        schema: &TableSchema,
47        gc_before_secs: Option<i64>,
48        now_secs: Option<i64>,
49        udt_registry: Option<crate::schema::UdtRegistry>,
50        scan_cancel: crate::storage::scan_cancel::ScanCancel,
51    ) -> Result<Self> {
52        Self::new_with_open_reporting(
53            input_paths,
54            schema,
55            gc_before_secs,
56            now_secs,
57            udt_registry,
58            scan_cancel,
59            OpenErrorReporting::SelfReported,
60        )
61    }
62
63    /// [`new_with_gc_and_registry_cancellable`](Self::new_with_gc_and_registry_cancellable)
64    /// with the failed-input-open reporting mode stated by the CALLER (issue #1704).
65    ///
66    /// Pass [`OpenErrorReporting::DeferredToCaller`] only from a call site that can
67    /// name the seam which will count the failure instead. See the module doc.
68    pub(crate) fn new_with_open_reporting(
69        input_paths: Vec<PathBuf>,
70        schema: &TableSchema,
71        gc_before_secs: Option<i64>,
72        now_secs: Option<i64>,
73        udt_registry: Option<crate::schema::UdtRegistry>,
74        scan_cancel: crate::storage::scan_cancel::ScanCancel,
75        reporting: OpenErrorReporting,
76    ) -> Result<Self> {
77        if input_paths.is_empty() {
78            return Err(Error::InvalidInput(
79                "K-way merge requires at least one input file".to_string(),
80            ));
81        }
82
83        // Enforce the dropped-column decode contract (#904/#847): every column
84        // named in `dropped_columns` must still be declared in `columns` so its
85        // cells decode and can be purged. Guard here (the authoritative compaction
86        // entry) for programmatically-built schemas that bypass `validate()`.
87        schema.validate_dropped_columns()?;
88
89        // Create run readers for each input SSTable (ordered newest to oldest).
90        // The registry (when supplied) is cloned onto each reader so a frozen UDT
91        // value decodes structurally instead of erroring out and dropping the row
92        // (issue #1234). No producer-side `LIMIT` push-down (issue #2361, roborev
93        // round 2): every producer scans until genuinely cancelled — `LIMIT` is
94        // enforced purely downstream (consumer post-reconciliation break +
95        // cancel-aware Drop teardown). See
96        // [`SSTableReader::stream_all_partitions_cancellable`](crate::storage::sstable::reader::SSTableReader::stream_all_partitions_cancellable)'s
97        // doc for why a partition-granular producer budget cannot be correct.
98        // Issue #2765: register this k-way merge ONCE and snapshot the adaptive
99        // per-channel egress capacity; ALL source channels below share it, so a
100        // solo compaction gets 256 per source regardless of input count.
101        let (channel_capacity, egress_slot) = egress_budget::begin_merge();
102
103        let mut runs = Vec::with_capacity(input_paths.len());
104        for (run_index, path) in input_paths.iter().enumerate() {
105            let adapter = SSTableRowIteratorAdapter::open(
106                path,
107                run_index,
108                schema,
109                udt_registry.clone(),
110                scan_cancel.clone(),
111                channel_capacity,
112                reporting,
113            )?;
114            runs.push(RunReader::new(Box::new(adapter)));
115        }
116
117        // Initialize heap (will be populated on first step)
118        let heap = BinaryHeap::new();
119
120        Ok(Self {
121            runs,
122            heap,
123            current_partition: None,
124            schema: schema.clone(),
125            // Issue #1668, stage 5c-i: `Arc`-wrapped clone of `schema`, used
126            // only by the heap's schema-aware comparator (see the field doc).
127            schema_arc: std::sync::Arc::new(schema.clone()),
128            gc_before_secs,
129            now_secs,
130            // Conservatively unsafe by default (#921 finding 1): purging is
131            // dormant until a caller proves the compaction spans all of the
132            // table's SSTables via `with_purge_safe`.
133            purge_safe: false,
134            // No overlap bound by default (#935): a partial compaction stays
135            // conservative (no purging) until a caller supplies the min outside
136            // timestamp via `with_max_purgeable_timestamp`.
137            max_purgeable_timestamp: None,
138            _egress_slot: Some(egress_slot),
139        })
140    }
141}

143/// Build the cross-generation STREAMING read merge, whose failed input opens are
144/// counted by the measured `JoinedStream` the rows are delivered through — so the
145/// reopen inside each producer thread must NOT count them again (issue #1704).
146///
147/// A named helper rather than an inline call so the one site that defers can state
148/// WHY in one place, and so `generation_merge`'s call stays a single expression.
149///
150/// Gated exactly like its ONE caller, `generation_merge::stream_generations_for_read`
151/// — a `tombstones` build routes that read through the materializing `scan`, so the
152/// streaming merge (and therefore this helper) does not exist there.
153#[cfg(not(feature = "tombstones"))]
154pub(crate) fn merger_deferring_opens(
155    input_paths: Vec<PathBuf>,
156    schema: &TableSchema,
157) -> Result<KWayMerger> {
158    KWayMerger::new_with_open_reporting(
159        input_paths,
160        schema,
161        None,
162        None,
163        None,
164        crate::storage::scan_cancel::ScanCancel::default(),
165        OpenErrorReporting::DeferredToCaller,
166    )
