1//! Reader→merge row/cell conversion helpers for [`SSTableRowIteratorAdapter`].
2//!
3//! PURE CODE MOTION out of `merge/mod.rs` (issue #3139), same rationale as the
4//! sibling [`producer_iter`](super::producer_iter): `mod.rs` is ~13.5k lines,
5//! far over the ~800-line source campsite target (epic #1116). The moved items
6//! are byte-identical to their pre-move form apart from the `pub(super)`
7//! visibility the move requires.
8//!
9//! These are the pure translation steps BOTH producer shapes run per streamed
10//! source entry — the path-based producer thread
11//! ([`producer_iter`](super::producer_iter)) and the shared-reader one
12//! ([`from_readers`](super::from_readers)) both funnel through
13//! [`SSTableRowIteratorAdapter::build_merge_entry`]: `CompactionRow` →
14//! [`MergeEntry`], `CompactionRowData` → [`RowData`], reader `Value` →
15//! [`RowData`], plus clustering-key extraction and range-bound translation.
16//! They touch no channel and no gauge; they are kept as inherent methods on the
17//! adapter (an `impl` block in a sibling file, exactly as
18//! [`from_readers`](super::from_readers) does) so no call site changes.
19
20use super::producer_iter::SSTableRowIteratorAdapter;

25use crate::types::Value;

29    /// Convert one streamed `(RowKey, Value, timestamp)` source entry into a
30    /// [`MergeEntry`] for run `run_index` (issue #827).
31    ///
32    /// Factored out of the producer loop so the streaming emit callback can call
33    /// it inline. Populates the clustering key from the decoded cells so wide-row
34    /// (clustering) partitions reconcile per `(pk, ck)` instead of collapsing
35    /// into one row.
36    pub(super) fn build_merge_entry(
37        run_index: usize,
38        compaction_row: crate::storage::sstable::reader::compaction_row::CompactionRow,
39        schema: &TableSchema,
40    ) -> Result<MergeEntry> {
41        use crate::storage::sstable::reader::compaction_row::CompactionRowData;
42
43        let crate::storage::sstable::reader::compaction_row::CompactionRow {
44            key,
45            row_timestamp,
46            row_data,
47        } = compaction_row;
48        let decorated_key = DecoratedKey::from_key_bytes(key.as_bytes().to_vec())?;
49
50        // Issue #1072: a partition-level tombstone surfaces as a self-contained
51        // carrier `MergeEntry` (empty live row + `partition_deletion`, no
52        // clustering). `merge_partition_rows` extracts these, applies the MAX
53        // partition floor across sources to shadow older cells/rows/ranges, and
54        // re-emits the surviving partition tombstone. Handled BEFORE the
55        // `RangeMarker` block so the partition floor is the outermost shadow.
56        if let CompactionRowData::PartitionDelete {
57            deletion_time,
58            local_deletion_time,
59        } = row_data
60        {
61            // Carry the deletion in `RowData::Tombstone` so the carrier never
62            // surfaces as a phantom live row to consumers iterating the merge step
63            // stream; `partition_deletion` marks it as the partition-level carrier
64            // and `merge_entry_to_mutation` lifts it onto the partition header
65            // (emitting NO clustering-row `DeleteRow`).
66            return Ok(MergeEntry::new(
67                run_index,
68                decorated_key,
69                None,
70                // Use markedForDeleteAt as the entry timestamp so stats baselines
71                // see a real value (not 0) for the carrier.
72                deletion_time,
73                RowData::Tombstone {
74                    deletion_time,
75                    local_deletion_time,
76                },
77            )
78            .with_partition_deletion((deletion_time, local_deletion_time)));
79        }
80
81        // Issue #933: a range-tombstone marker surfaces as a self-contained
82        // carrier `MergeEntry` (empty live row + `range_deletion`), routed into the
83        // partition's `None` clustering bucket so it is never collapsed with data
84        // rows. `merge_partition_rows` extracts these to shadow covered cells and
85        // re-emit the surviving marker to the output SSTable.
86        if let CompactionRowData::RangeMarker {
87            start,
88            end,
89            deletion_time,
90            local_deletion_time,
91        } = row_data
92        {
93            let range = RangeTombstone {
94                start: Self::compaction_bound_to_mutation(start),
95                end: Self::compaction_bound_to_mutation(end),
96                deletion_time,
97                local_deletion_time,
98            };
99            return Ok(MergeEntry::new(
100                run_index,
101                decorated_key,
102                None,
103                // Use the deletion time as the entry timestamp so the writer's
104                // stats baselines see a real value (not 0) for the carrier.
105                deletion_time,
106                RowData::Live { cells: Vec::new() },
107            )
108            .with_range_deletion(range));
109        }
110        // #912: derive the clustering identity from the per-element compaction row
111        // BEFORE collapsing it to `RowData`. A row tombstone carries no cells in
112        // `RowData::Tombstone`, so its clustering key must come from the
113        // tombstone's captured clustering prefix; otherwise it would collapse into
114        // the partition's `None` bucket and mis-reconcile against the static row
115        // and against other clustering-row tombstones.
116        let clustering_key = Self::extract_clustering_key_from_compaction(&row_data, schema);
117        let (row_data, complex_deletions, row_deletion, row_liveness) =
118            Self::compaction_row_data_to_row_data(row_data, row_timestamp);
119        let entry = MergeEntry::new(
120            run_index,
121            decorated_key,
122            clustering_key,
123            row_timestamp,
124            row_data,
125        )
126        // Issue #2374/#2789: carry the row-marker liveness for the read path.
127        .with_row_liveness(row_liveness);
128        let entry = if complex_deletions.is_empty() {
129            entry
130        } else {
131            entry.with_complex_deletions(complex_deletions)
132        };
133        // Issue #932: carry the coexisting row deletion so reconciliation keeps it
134        // alongside the surviving live cells (preventing resurrection of older
135        // cells in non-compacted SSTables).
136        let entry = match row_deletion {
137            Some((deletion_time, ldt)) => entry.with_row_deletion(deletion_time, ldt),
138            None => entry,
139        };
140        Ok(entry)
141    }

143    /// Convert a reader-native [`CompactionBound`] into a write-engine
144    /// [`ClusteringBound`] (issue #933).
145    ///
146    /// [`CompactionBound`]: crate::storage::sstable::reader::compaction_row::CompactionBound
147    /// [`ClusteringBound`]: crate::storage::write_engine::mutation::ClusteringBound
148    fn compaction_bound_to_mutation(
149        bound: crate::storage::sstable::reader::compaction_row::CompactionBound,
150    ) -> crate::storage::write_engine::mutation::ClusteringBound {
151        use crate::storage::sstable::reader::compaction_row::CompactionBound;
152        use crate::storage::write_engine::mutation::ClusteringBound;
153        match bound {
154            CompactionBound::Inclusive(cols) => {
155                ClusteringBound::Inclusive(ClusteringKey { columns: cols })
156            }
157            CompactionBound::Exclusive(cols) => {
158                ClusteringBound::Exclusive(ClusteringKey { columns: cols })
159            }
160            CompactionBound::Bottom => ClusteringBound::Bottom,
161            CompactionBound::Top => ClusteringBound::Top,
162        }
163    }

165    /// Extract a `ClusteringKey` from the row's live cells using the schema.
166    ///
167    /// For each clustering column declared in the schema (in position order),
168    /// look for a cell with that column name in the decoded `RowData::Live`
169    /// cells.  If all clustering columns are found, return `Some(ClusteringKey)`;
170    /// otherwise (including for tombstone entries that have no cells) return
171    /// `None`.
172    ///
173    /// The clustering columns are intentionally left inside the cells so the
174    /// downstream read-back path can still find them.
175    pub(super) fn extract_clustering_key_from_compaction(
176        row_data: &crate::storage::sstable::reader::compaction_row::CompactionRowData,
177        schema: &TableSchema,
178    ) -> Option<ClusteringKey> {
179        use crate::storage::sstable::reader::compaction_row::CompactionRowData;
180
181        if schema.clustering_keys.is_empty() {
182            return None;
183        }
184
185        match row_data {
186            // A live row surfaces its clustering columns as simple cells (#229).
187            CompactionRowData::Live { simple, .. } => {
188                let mut ck_columns: Vec<(String, Value)> =
189                    Vec::with_capacity(schema.clustering_keys.len());
190                for ck_col in &schema.clustering_keys {
191                    // Any missing clustering column ⇒ the WHOLE key is discarded (the `?`
192                    // returns `None`, treating the row as unclustered).
193                    let pair = simple
194                        .iter()
195                        .find(|c| c.column == ck_col.name)
196                        .map(|c| (ck_col.name.clone(), c.value.clone()))?;
197                    ck_columns.push(pair);
198                }
199                Some(ClusteringKey {
200                    columns: ck_columns,
201                })
202            }
203            // #912: a row tombstone carries its own clustering prefix so it
204            // reconciles in its own bucket instead of collapsing into `None`.
205            // An empty `clustering` (unclustered table / partial prefix) keeps the
206            // pre-#912 `None`-bucket behavior.
207            CompactionRowData::Tombstone { clustering, .. } => {
208                if clustering.is_empty() {
209                    return None;
210                }
211                // Reorder defensively into schema order; bail to `None` if any
212                // declared clustering column is absent.
213                let mut ck_columns: Vec<(String, Value)> =
214                    Vec::with_capacity(schema.clustering_keys.len());
215                for ck_col in &schema.clustering_keys {
216                    // Any missing clustering column ⇒ the WHOLE key is discarded (the `?`
217                    // returns `None`, treating the row as unclustered).
218                    let pair = clustering
219                        .iter()
220                        .find(|(name, _)| name == &ck_col.name)
221                        .map(|(name, v)| (name.clone(), v.clone()))?;
222                    ck_columns.push(pair);
223                }
224                Some(ClusteringKey {
225                    columns: ck_columns,
226                })
227            }
228            // Issue #933: a range marker is handled by `build_merge_entry` before
229            // this point and routed into the `None` clustering bucket.
230            CompactionRowData::RangeMarker { .. } => None,
231            // Issue #1072: a partition tombstone is handled by `build_merge_entry`
232            // before this point and routed into the `None` clustering bucket.
233            CompactionRowData::PartitionDelete { .. } => None,
234        }
235    }

237    /// Convert a [`CompactionRowData`] into the merge `RowData` plus the row's
238    /// complex-deletion markers (epic #899, Phase C — the behavioral flip).
239    ///
240    /// Simple columns become one [`CellData`] each (cell-own ts/ttl/ldt). A
241    /// complex (non-frozen collection / UDT) column is NO LONGER collapsed to a
242    /// single whole-column value: each [`ComplexElement`] becomes its OWN
243    /// per-element [`CellData`] carrying the element's authoritative `cell_path`,
244    /// per-element `timestamp`, `ttl`, `local_deletion_time`, `is_deleted`, and
245    /// on-disk `has_empty_value`. `reconcile_cluster` keys winners on
246    /// `(column, cell_path)`, so disjoint elements of the same column written
247    /// across SSTables all survive (Cassandra `Cells#reconcile`), and the
248    /// merge→mutation step emits a [`CellOperation::WriteComplexElement`] per
249    /// element — preserving the per-element on-disk layout byte-for-byte (epic
250    /// #899 north star). The reader's whole-collection `collapsed_value` is no
251    /// longer threaded to the writer (the per-element path is now authoritative);
252    /// it remains on the reader contract for user-facing reads.
253    ///
254    /// The real per-column complex deletion (`markedForDeleteAt` +
255    /// `localDeletionTime`) is surfaced on `complex_deletions` so the writer
256    /// emits a REAL deletion marker (replacing the LIVE sentinel). `reconcile_cluster`
257    /// reduces these to the strict-superseding (max-mfda) deletion per column NAME
258    /// and SHADOWS covered elements (ts <= mfda) before purge (issue #887). gc_grace
259    /// purging of the surviving marker remains future work (#845).
260    ///
261    /// [`ComplexElement`]: crate::storage::sstable::reader::compaction_row::ComplexElement
262    /// [`CellOperation::WriteComplexElement`]: crate::storage::write_engine::mutation::CellOperation::WriteComplexElement
263    #[allow(clippy::type_complexity)]
264    pub(super) fn compaction_row_data_to_row_data(
265        row_data: crate::storage::sstable::reader::compaction_row::CompactionRowData,
266        _row_timestamp: i64,
267    ) -> (
268        RowData,
269        Vec<ComplexDeletion>,
270        Option<(i64, i32)>,
271        crate::storage::sstable::reader::compaction_row::RowLiveness,
272    ) {
273        use crate::storage::sstable::reader::compaction_row::{CompactionRowData, RowLiveness};
274
275        match row_data {
276            CompactionRowData::Tombstone {
277                deletion_time,
278                local_deletion_time,
279                // Clustering identity is consumed by
280                // `extract_clustering_key_from_compaction` before this conversion
281                // (#912); `RowData::Tombstone` itself carries only the timestamps.
282                clustering: _,
283            } => (
284                RowData::Tombstone {
285                    deletion_time,
286                    local_deletion_time,
287                },
288                Vec::new(),
289                None,
290                RowLiveness::default(),
291            ),
292            CompactionRowData::Live {
293                simple,
294                complex,
295                // Issue #932: a coexisting row deletion surfaces here so the
296                // merge entry carries it alongside the live cells.
297                row_deletion,
298                // Issue #2374/#2789: primary-key liveness carried for the read path.
299                row_liveness,
300            } => {
301                let element_count: usize = complex.iter().map(|c| c.elements.len()).sum();
302                let mut cells = Vec::with_capacity(simple.len() + element_count);
303                let mut complex_deletions = Vec::new();
304
305                for sc in simple {
306                    cells.push(CellData {
307                        column: sc.column,
308                        value: sc.value,
309                        timestamp: sc.timestamp,
310                        ttl: sc.ttl,
311                        cell_path: None,
312                        local_deletion_time: sc.local_deletion_time,
313                        is_complex_element: false,
314                        is_deleted: false,
315                        has_empty_value: false,
316                    });
317                }
318
319                for col in complex {
320                    // Surface the REAL complex deletion (replacing the LIVE
321                    // sentinel) so the writer emits a genuine marker (Phase C).
322                    if let Some((marked_for_delete_at, ldt)) = col.complex_deletion {
323                        complex_deletions.push(ComplexDeletion {
324                            column: col.column.clone(),
325                            marked_for_delete_at,
326                            local_deletion_time: ldt,
327                        });
328                    }
329
330                    // Per-element emit: one CellData per ComplexElement, keyed by
331                    // its authoritative cell_path, carrying per-element write
332                    // metadata verbatim. An element tombstone is represented with
333                    // `is_deleted = true` and a `Value::Tombstone(CellTombstone)`
334                    // value so the per-cell reconcile tie-break (tombstone beats
335                    // live at equal ts) still applies; a live element keeps its
336                    // decoded value (the reader stores the SET member decoded from
337                    // the path, but `has_empty_value` records the on-disk emptiness
338                    // so the writer reproduces it byte-for-byte).
339                    for elem in col.elements {
340                        let value = if elem.is_deleted {
341                            Value::Tombstone(Box::new(crate::types::TombstoneInfo {
342                                deletion_time: elem.timestamp,
343                                tombstone_type: crate::types::TombstoneType::CellTombstone,
344                                // Element's on-disk localDeletionTime (GC clock,
345                                // seconds); `0` when not surfaced (#873).
346                                local_deletion_time: elem.local_deletion_time.unwrap_or(0) as i64,
347                                ttl: None,
348                                range_start: None,
349                                range_end: None,
350                            }))
351                        } else {
352                            elem.value.unwrap_or(Value::Null)
353                        };
354                        cells.push(CellData {
355                            column: col.column.clone(),
356                            value,
357                            timestamp: elem.timestamp,
358                            ttl: elem.ttl,
359                            cell_path: Some(elem.cell_path),
360                            local_deletion_time: elem.local_deletion_time,
361                            is_complex_element: true,
362                            is_deleted: elem.is_deleted,
363                            has_empty_value: elem.has_empty_value,
364                        });
365                    }
366                }
367
368                (
369                    RowData::Live { cells },
370                    complex_deletions,
371                    row_deletion,
372                    row_liveness,
373                )
374            }
375            // Issue #933: range markers are intercepted in `build_merge_entry`
376            // (carrier entry with `range_deletion`); they never reach this
377            // conversion. Map defensively to an empty live row.
378            CompactionRowData::RangeMarker { .. } => (
379                RowData::Live { cells: Vec::new() },
380                Vec::new(),
381                None,
382                RowLiveness::default(),
383            ),
384            // Issue #1072: partition tombstones are intercepted in
385            // `build_merge_entry` (carrier entry with `partition_deletion`); they
386            // never reach this conversion. Map defensively to an empty live row.
387            CompactionRowData::PartitionDelete { .. } => (
388                RowData::Live { cells: Vec::new() },
389                Vec::new(),
390                None,
391                RowLiveness::default(),
392            ),
393        }
394    }

396    /// Convert a reader Value to RowData.
397    ///
398    /// `row_timestamp` is the per-row timestamp decoded from the on-disk row
399    /// header (see [`SSTableReader::iterate_all_partitions_for_compaction`]). The
400    /// reader does not surface per-cell timestamps for live cells, so each live
401    /// cell inherits the row timestamp. This is required for per-cell reconcile
402    /// and row-tombstone shadowing to compare cell timestamps correctly
403    /// (Issue #533) — without it live cells would default to 0 and be wrongly
404    /// shadowed by any row tombstone.
405    ///
406    /// Issue #505: `Value::Tombstone(RowTombstone)` is now correctly emitted by
407    /// the V5CompressedLegacy parser for deleted rows, and
408    /// `Value::Tombstone(CellTombstone)` appears inside `Value::Map` entries for
409    /// deleted cells.  Both are surfaced here so the merger can apply shadowing
410    /// semantics.  A cell tombstone keeps its own `deletion_time` so equal-ts
411    /// reconcile still resolves it correctly.
412    ///
413    /// Epic #899: superseded on the production path by
414    /// [`Self::compaction_row_data_to_row_data`] (the reader now surfaces
415    /// per-element [`CompactionRow`]s). Retained for the legacy-collapse merge
416    /// tests that assert the old whole-column collapse behavior.
417    #[cfg(test)]
418    pub(super) fn value_to_row_data(
419        value: &crate::types::Value,
420        row_timestamp: i64,
421    ) -> Result<RowData> {
422        match value {
423            crate::types::Value::Tombstone(info) => Ok(RowData::Tombstone {
424                deletion_time: info.deletion_time,
425                // #873: TombstoneInfo now carries the source localDeletionTime
426                // (GC clock, seconds). `RowData::Tombstone` stores it as i32
427                // (the on-disk width); the value fits an i32 in practice
428                // (epoch-seconds, incl. far-future negative-i32 bit patterns).
429                // NOTE: this legacy adapter is `#[cfg(test)]` only — the
430                // production compaction tombstone path is
431                // `compaction_row_data_to_row_data`, which already threads the
432                // LDT from `CompactionRowData::Tombstone`.
433                local_deletion_time: info.local_deletion_time as i32,
434            }),
435            crate::types::Value::Map(map_entries) => {
436                let mut cells = Vec::with_capacity(map_entries.len());
437                for (key, val) in map_entries {
438                    let column = match key {
439                        crate::types::Value::Text(s) => String::from_utf8_lossy(s).into_owned(),
440                        other => format!("{:?}", other),
441                    };
442                    // Cell tombstones carry their own deletion_time (Issue #505);
443                    // live cells inherit the row timestamp (Issue #533) so per-cell
444                    // shadowing and LWW order them against row tombstones correctly.
445                    let cell_ts = match val {
446                        crate::types::Value::Tombstone(info) => info.deletion_time,
447                        _ => row_timestamp,
448                    };
449                    cells.push(CellData {
450                        column,
451                        value: val.clone(),
452                        timestamp: cell_ts,
453                        // ttl / local_deletion_time / cell_path are threaded for
454                        // the followup behaviors (#844, #848) but the reader's
455                        // `(RowKey, Value, ts)` compaction stream does not yet
456                        // surface per-cell ttl, the cell's local-deletion-time,
457                        // or a complex-column cell-path (the map key here is the
458                        // top-level column name, not a collection element path),
459                        // so they stay `None`. Populating them is part of #844 /
460                        // #848 once the reader is extended (issue #886 plumbing).
461                        ttl: None,
462                        cell_path: None,
463                        local_deletion_time: None,
464                        is_complex_element: false,
465                        is_deleted: false,
466                        has_empty_value: false,
467                    });
468                }
469                Ok(RowData::Live { cells })
470            }
471            // Single value or other formats - wrap as a single cell
472            other => Ok(RowData::Live {
473                cells: vec![CellData {
474                    column: "value".to_string(),
475                    value: other.clone(),
476                    timestamp: row_timestamp,
477                    // Not surfaced by the reader's compaction stream yet; see the
478                    // note on the map-entry path above (issue #886 plumbing).
479                    ttl: None,
480                    cell_path: None,
481                    local_deletion_time: None,
482                    is_complex_element: false,
483                    is_deleted: false,
484                    has_empty_value: false,
485                }],
486            }),
487        }
488    }
