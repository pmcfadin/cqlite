1//! Cooperative cancellation for long-running synchronous scans (issue #2264).
2//!
3//! The compaction streaming read (`stream_all_partitions_for_compaction`) walks
4//! a whole Data.db on a detached producer thread. For an index-less
5//! (Summary.db-absent) SSTable that walk fully materialises every partition in
6//! one uninterruptible pass — so a Flight `do_get` whose client has disconnected
7//! keeps burning CPU until a coarse ~1–2 min backstop reaps it, ignoring the
8//! cancellation the transport already received.
9//!
10//! [`ScanCancel`] is a cheap, cloneable flag the scan polls at a bounded interval
11//! (every N partitions). It is deliberately a bare `AtomicBool` — the scan runs
12//! on a plain `std::thread` with no async runtime, so it cannot poll an
13//! async-wakeable token. The Flight layer bridges its own
14//! `tokio_util::CancellationToken`-backed `CancelFlag` onto one of these so a
15//! single `cancel()` trips both the async channel-race (PR #2282) AND this
16//! synchronous scan poll.
17
18use std::sync::atomic::{AtomicBool, Ordering};
19use std::sync::Arc;
20
21/// A cheap, cloneable cooperative-cancellation flag for synchronous scans.
22///
23/// All clones share one `AtomicBool`, so cancelling any clone is observed by a
24/// scan polling another. A default (never-cancelled) flag makes every threaded
25/// call site a no-op, so non-cancellable callers pass `ScanCancel::default()`.
26#[derive(Clone, Debug, Default)]
27pub struct ScanCancel(Arc<AtomicBool>);
28
29impl ScanCancel {
30    /// Create a fresh, un-cancelled flag.
31    pub fn new() -> Self {
32        Self::default()
33    }
34
35    /// Request cancellation. Idempotent.
36    pub fn cancel(&self) {
37        self.0.store(true, Ordering::Relaxed);
38    }
39
40    /// Whether cancellation has been requested.
41    pub fn is_cancelled(&self) -> bool {
42        self.0.load(Ordering::Relaxed)
43    }
44
45    /// Return `Err(Error::Cancelled)` if cancellation has been requested, else
46    /// `Ok(())`. The polling helper scan loops call at a bounded interval.
47    pub fn check(&self) -> crate::Result<()> {
48        if self.is_cancelled() {
49            Err(crate::Error::Cancelled)
50        } else {
51            Ok(())
52        }
53    }
54
55    /// Cooperative scan CHECKPOINT: poll for cancellation (issues #2264/#2346)
56    /// **and** yield to the async runtime every [`YIELD_STRIDE`]-th call
57    /// (issue #1695).
58    ///
59    /// # Why the yield belongs here
60    ///
61    /// `query.max_execution_time` is enforced by ONE `tokio::time::timeout` at the
62    /// query-engine chokepoint (`query::engine::deadline`). A `timeout` can only
63    /// elapse when the future it wraps becomes `Pending`: the read path does long
64    /// stretches of SYNCHRONOUS work per poll (stitch a data section, parse a
65    /// block, decode a partition), so without a yield point the wrapper could not
66    /// fire until the whole scan had finished — the budget would be a placebo on
67    /// exactly the runaway full scan it exists to bound.
68    ///
69    /// This carries NO deadline knowledge: the scan never reads a clock and never
70    /// learns the budget (the mandate of issue #1695 is one wrapper at the
71    /// chokepoint, never ad-hoc clock checks in the scan loop). It only makes the
72    /// walk interruptible at the SAME audited cadence at which it is already
73    /// cancellable.
74    ///
75    /// `tick` is the loop's own counter (partition/entry/chunk index). BOTH the
76    /// cancellation poll and the runtime yield happen on every `YIELD_STRIDE`-th
77    /// call, and an off-stride call is a deliberate no-op.
78    ///
79    /// Polling on every call would be the more responsive choice, and an earlier
80    /// revision did that — but it silently RAISED the cancellation cadence of
81    /// three callers that had hand-rolled `if tick & 0xFF == 0 { check()? }`,
82    /// adding a relaxed atomic load and an `.await` point to a per-entry decode
83    /// loop. That cadence was chosen deliberately by #2264/#2346, and #1695 has no
84    /// mandate to retune the read hot path, so this reproduces it exactly.
85    /// A loop that genuinely needs per-iteration cancellation uses
86    /// [`Self::checkpoint_now`] instead.
87    pub async fn checkpoint(&self, tick: usize) -> crate::Result<()> {
88        if tick % YIELD_STRIDE == 0 {
89            self.check()?;
90            tokio::task::yield_now().await;
91        }
92        Ok(())
93    }
94
95    /// Cooperative checkpoint that polls cancellation on EVERY call but yields
96    /// only at the [`YIELD_STRIDE`] boundary — the third of three cadences.
97    ///
98    /// For a loop whose iterations are individually coarse enough to want prompt
99    /// cancellation (one real index random read each), but numerous enough that a
100    /// per-iteration runtime reschedule would be a throughput cost. Reaching for
101    /// [`Self::checkpoint_now`] there preserves the cancellation cadence and
102    /// silently ADDS a yield per iteration where the caller previously had none;
103    /// reaching for [`Self::checkpoint`] preserves the yield cadence and silently
104    /// LOWERS cancellation to one poll per 256. Neither is what such a caller
105    /// wants, which is why this exists as its own method.
106    ///
107    /// The three, so the choice is explicit at every call site:
108    ///
109    /// | method | cancellation poll | runtime yield |
110    /// |---|---|---|
111    /// | [`Self::checkpoint`] | every 256th | every 256th |
112    /// | `checkpoint_polled` | EVERY call | every 256th |
113    /// | [`Self::checkpoint_now`] | EVERY call | EVERY call |
114    pub async fn checkpoint_polled(&self, tick: usize) -> crate::Result<()> {
115        self.check()?;
116        if tick % YIELD_STRIDE == 0 {
117            tokio::task::yield_now().await;
118        }
119        Ok(())
120    }
121
122    /// [`Self::checkpoint`] for a loop whose iterations are individually coarse
123    /// (a whole block / data section), where every iteration is a correct yield
124    /// point.
125    pub async fn checkpoint_now(&self) -> crate::Result<()> {
126        self.check()?;
127        tokio::task::yield_now().await;
128        Ok(())
129    }
130}
131
132/// Checkpoint stride for [`ScanCancel::checkpoint`] — the same 256-iteration
133/// cadence the callers' hand-rolled cancellation polls used (#2264/#2346). Both
134/// the poll and the yield sit on that boundary, so cancellation latency is
135/// UNCHANGED from those callers and the added cost is one task reschedule per 256
136/// decoded partitions/entries/chunks. Stated as an equivalence to the prior code
137/// rather than as a throughput claim: no measurement was taken here, and #2403's
138/// release theme is exactly where an unmeasured "no measurable change" would be
139/// the wrong thing to assert.
140pub const YIELD_STRIDE: usize = 256;

142#[cfg(test)]
143mod tests {
144    use super::*;
145
146    #[test]
147    fn starts_uncancelled_and_cancels() {
148        let c = ScanCancel::new();
149        assert!(!c.is_cancelled());
150        assert!(c.check().is_ok());
151        c.cancel();
152        assert!(c.is_cancelled());
153        assert!(matches!(c.check(), Err(crate::Error::Cancelled)));
154    }
155
156    #[test]
157    fn clones_share_state() {
158        let a = ScanCancel::new();
159        let b = a.clone();
160        a.cancel();
161        assert!(b.is_cancelled(), "cancel on one clone is seen by another");
162    }
163
164    /// A checkpoint on an UNCANCELLED flag must return `Ok` and (at a stride
165    /// boundary) hand control back to the runtime — the property the chokepoint
166    /// timeout depends on (issue #1695).
167    #[tokio::test]
168    async fn checkpoint_checks_and_yields_at_the_stride_only() {
169        let c = ScanCancel::new();
170        // Stride boundary: checks + yields, still Ok.
171        assert!(c.checkpoint(0).await.is_ok());
172        // Off-stride: no check, no yield, still Ok.
173        assert!(c.checkpoint(1).await.is_ok());
174        assert!(c.checkpoint_now().await.is_ok());
175
176        c.cancel();
177        // ON stride, cancellation is observed and takes precedence over the yield.
178        for tick in [0usize, YIELD_STRIDE] {
179            assert!(matches!(
180                c.checkpoint(tick).await,
181                Err(crate::Error::Cancelled)
182            ));
183        }
184        // OFF stride it is deliberately NOT observed. This is the contract, not a
185        // gap: it reproduces the `if tick & 0xFF == 0 { check()? }` cadence the
186        // callers had before #1695, so a per-entry decode loop pays nothing between
187        // boundaries. A loop needing per-iteration cancellation uses
188        // `checkpoint_now`, which is asserted below to still report it.
189        for tick in [1usize, YIELD_STRIDE + 1] {
190            assert!(
191                c.checkpoint(tick).await.is_ok(),
192                "an off-stride checkpoint must be a no-op even once cancelled"
193            );
194        }
195        assert!(matches!(
196            c.checkpoint_now().await,
197            Err(crate::Error::Cancelled)
198        ));
199    }

201    /// The checkpoint's yield is what lets a `tokio::time::timeout` around a
202    /// synchronous-looking scan loop actually elapse (issue #1695): without it the
203    /// loop would run to completion inside one poll.
204    #[tokio::test]
205    async fn checkpoint_makes_a_scan_loop_interruptible_by_timeout() {
206        let c = ScanCancel::new();
207        let out = tokio::time::timeout(std::time::Duration::from_millis(1), async {
208            // A loop with NO yield point of its own, exactly like a decode walk.
209            for tick in 0..usize::MAX {
210                c.checkpoint(tick).await?;
211            }
212            Ok::<(), crate::Error>(())
213        })
214        .await;
215        assert!(
216            out.is_err(),
217            "a loop whose only yield point is the checkpoint MUST be interruptible              by an enclosing timeout"
218        );
219    }
220
221    #[test]
222    fn default_is_never_cancelled() {
223        let c = ScanCancel::default();
224        assert!(!c.is_cancelled());
225        assert!(c.check().is_ok());
226    }
