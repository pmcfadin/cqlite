1//! Always-on READ-PATH ARM probes (issue #3058).
2//!
3//! The Flight `do_get` row route can be served by two structurally different
4//! arms: the k-way compaction merge (`KWayMerger` + the compaction reconciler)
5//! and — when exactly ONE post-prune source is involved — the single-generation
6//! query scan. "Which arm ran" is not observable from the RESULT (both arms are
7//! required to return the same rows), so a test that wants to pin the routing
8//! must observe an explicit MARKER. These counters are that marker.
9//!
10//! # Why an explicit marker, not a timing/throughput inference
11//!
12//! Spec `flight-single-sstable-bypass` (issue AC #1) requires the path-taken
13//! assertion to FAIL when the merge path is entered, and forbids inferring the
14//! arm from elapsed time, throughput, or CPU share (a timing assertion is both
15//! host-dependent and, per issue #2877, capable of passing while the work it
16//! claims to have removed is still being done). A counter incremented at the
17//! merge's own construction / reconcile / cell-metadata-allocation sites is a
18//! direct observation of the work, so `== 0` is a proof, not a correlation.
19//!
20//! # Always-on, like `SCAN_FOR_KEY_CALLS`
21//!
22//! These are plain `Relaxed` adds on the MERGE arm only (never on the scan
23//! arm), following the always-on `SCAN_FOR_KEY_CALLS` precedent
24//! (`data_access/model.rs`) rather than the `cfg`-gated `read_work_counters`
25//! pattern: the consumers are integration tests in a DIFFERENT crate
26//! (`cqlite-flight/tests/`), which cannot see `cfg(test)` counters and must not
27//! be forced to rebuild `cqlite-core` under a non-default feature. The merge arm
28//! costs microseconds per row, so one uncontended relaxed increment per merged
29//! row is unmeasurable there; the fast arm pays literally nothing because it
30//! never reaches these sites.
31//!
32//! # Reading them
33//!
34//! They are PROCESS-GLOBAL. A test observes a DELTA around the operation under
35//! test ([`ReadPathProbe::snapshot`] → run → [`ReadPathProbe::delta_since`]) and
36//! must serialize against any sibling test in the same binary that also merges
37//! (one file = one process is the strongest form; a mutex is the minimum).
38
39use std::sync::atomic::{AtomicI64, AtomicU64, Ordering};
40
41static MERGERS_BUILT: AtomicU64 = AtomicU64::new(0);
42static RECONCILE_ENTRIES: AtomicU64 = AtomicU64::new(0);
43static CELL_METADATA_MAPS: AtomicU64 = AtomicU64::new(0);
44
45/// Record one k-way merger construction over an already-open reader set
46/// (`KWayMerger::new_from_readers` — the Flight warm row route's merge arm).
47#[inline]
48pub fn record_merger_built() {
49    MERGERS_BUILT.fetch_add(1, Ordering::Relaxed);
50}
51
52/// Record one entry into the compaction reconciler
53/// (`KWayMerger::reconcile_cluster_with_overlap_counted`).
54#[inline]
55pub fn record_reconcile_entry() {
56    RECONCILE_ENTRIES.fetch_add(1, Ordering::Relaxed);
57}
58
59/// Record one per-row `HashMap<String, CellWriteMetadata>` allocation in the row
60/// decoder (i.e. one row decoded with `want_cell_metadata == true`).
61#[inline]
62pub fn record_cell_metadata_map() {
63    CELL_METADATA_MAPS.fetch_add(1, Ordering::Relaxed);
64}

66/// A point-in-time reading of the three arm counters.
67#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
68pub struct ReadPathProbe {
69    /// k-way mergers constructed over a warm reader set.
70    pub mergers_built: u64,
71    /// Entries into the compaction reconciler.
72    pub reconcile_entries: u64,
73    /// Per-row cell-write-metadata maps allocated by the row decoder.
74    pub cell_metadata_maps: u64,
75}
76
77impl ReadPathProbe {
78    /// Read the current process-global counter values.
79    pub fn snapshot() -> Self {
80        Self {
81            mergers_built: MERGERS_BUILT.load(Ordering::Relaxed),
82            reconcile_entries: RECONCILE_ENTRIES.load(Ordering::Relaxed),
83            cell_metadata_maps: CELL_METADATA_MAPS.load(Ordering::Relaxed),
84        }
85    }
86
87    /// The work recorded between `earlier` and `self`, saturating (a counter can
88    /// only climb, so a saturating subtraction is exact here and can never
89    /// underflow-panic on a racing reader).
90    pub fn delta_since(&self, earlier: &Self) -> Self {
91        Self {
92            mergers_built: self.mergers_built.saturating_sub(earlier.mergers_built),
93            reconcile_entries: self
94                .reconcile_entries
95                .saturating_sub(earlier.reconcile_entries),
96            cell_metadata_maps: self
97                .cell_metadata_maps
98                .saturating_sub(earlier.cell_metadata_maps),
99        }
100    }
101
102    /// Whether ANY merge-arm work was recorded in this (delta) reading.
103    pub fn any_merge_work(&self) -> bool {
104        self.mergers_built > 0 || self.reconcile_entries > 0 || self.cell_metadata_maps > 0
105    }
106}

112/// Producers that have finished decoding, published with RELEASE ordering.
113///
114/// Deliberately NOT a field of [`ReadPathProbe`]: every field there is loaded
115/// `Relaxed`, which is right for counters read after the work is known to be over,
116/// and wrong for this one — its whole job is to TELL you the work is over.
117static QUERY_ROW_PRODUCERS_FINISHED: AtomicU64 = AtomicU64::new(0);
118
119/// Publish that one `QueryRowStream` producer has finished decoding.
120///
121/// Call BEFORE the terminal message is sent, never after (roborev, issue #3384):
122/// a consumer holding the terminal message must be able to conclude that this
123/// producer can no longer publish, or a producer from a PRIOR test case can
124/// increment into a LATER case's freshly-reset counter and the later case will
125/// observe a completion that was never its own.
126pub fn mark_query_row_producer_finished() {
127    QUERY_ROW_PRODUCERS_FINISHED.fetch_add(1, Ordering::Release);
128}
129
130/// `QueryRowStream` producers that have finished decoding since the last
131/// [`reset_query_row_producers_finished`] (issue #3384).
132///
133/// The CAUSAL completion signal for an abandoned walk. A test that instead waits
134/// for a work counter to "stop growing" is sampling another thread's progress: a
135/// producer merely descheduled, or paused between two increments, is
136/// indistinguishable from one that has stopped, so the walk can resume and drain
137/// the table right after the assertion passed.
138///
139/// # What it proves, and what it does NOT
140///
141/// It proves the OUTER query-row producer stopped. It does NOT prove the INNER
142/// batched scan task has (roborev, issue #3384): `drive_full_scan_rows` drops
143/// `BatchedScanStream` on its way out and that task is never joined. That trail is
144/// BOUNDED — the inner loop consults its consumer only when a batch fills, so after
145/// the receiver is dropped it decodes at most one more emit batch before its `send`
146/// fails and it returns — so it cannot run away, but a reader wanting a FINAL work
147/// count should let the trail land rather than read the instant this signal moves.
148/// Joining the inner task on shutdown is tracked on #3428.
149///
150/// The `Acquire` load is load-bearing, not decoration. With `Relaxed` on both
151/// sides, observing a non-zero count would establish NO happens-before edge with
152/// the producer's earlier work-counter increments, so a reader could see
153/// "finished" and then read a STALE row count — precisely the guarantee this
154/// signal exists to give (roborev, issue #3384).
155pub fn query_row_producers_finished() -> u64 {
156    QUERY_ROW_PRODUCERS_FINISHED.load(Ordering::Acquire)
157}

159/// Blocking scan tasks currently running (issue #3384). A GAUGE, not a counter.
160static BLOCKING_SCAN_TASKS_INFLIGHT: AtomicI64 = AtomicI64::new(0);
161
162/// RAII marker for one blocking scan task: increments on construction, decrements on
163/// drop (issue #3384).
164///
165/// The stitching path's parse/feed halves run under `spawn_blocking`, and blocking
166/// tasks are NOT cancellable — dropping their `JoinHandle` DETACHES them. So neither
167/// the outer query-row producer's completion nor the scan future's drop guard proves
168/// decoding has stopped: both can fire while a detached blocking half is still
169/// draining. This gauge is the THIRD and last of the completion signals — the other two
170/// are `query_row_producers_finished` (outer thread) and the scan future's own drop
171/// guard, and neither covers detached blocking work. A gauge rather than a completion counter
172/// because the number of blocking halves is an implementation detail — a reader waits
173/// for it to reach ZERO instead of knowing how many to expect.
174///
175/// Decrement is a `Drop` impl so an unwinding task still clears its slot; a leaked
176/// increment would hang every future reader.
177/// The private field is load-bearing (roborev, issue #3384): as a UNIT struct this
178/// was constructible by name, so a caller could make one WITHOUT the increment and
179/// then decrement the gauge below zero on drop — silently breaking every completion
180/// check that reads it. `pub(crate)` for the same reason, narrowed further: nothing
181/// outside this crate needs to register a blocking task, and the observers
182/// ([`blocking_scan_tasks_inflight`]) stay public.
183pub(crate) struct BlockingScanTaskGuard(());
184
185impl BlockingScanTaskGuard {
186    /// Register one running blocking scan task. The ONLY way to make one.
187    pub(crate) fn new() -> Self {
188        // `AcqRel` for the same chaining reason as the decrement below.
189        BLOCKING_SCAN_TASKS_INFLIGHT.fetch_add(1, Ordering::AcqRel);
190        Self(())
191    }
192}
193
194impl Drop for BlockingScanTaskGuard {
195    fn drop(&mut self) {
196        // `AcqRel`, not `Release` (roborev, issue #3384). A reader's `Acquire` load of
197        // ZERO synchronizes-with the FINAL decrement only. With plain `Release` stores
198        // that leaves the EARLIER tasks' work unordered: if the feed half decrements
199        // last, the reader's subsequent (Relaxed) work-counter load is not guaranteed to
200        // observe the parse half's increments — so the "final count" would not be final,
201        // which is the entire property this gauge exists to provide. `AcqRel` makes each
202        // decrement ACQUIRE the ones before it, chaining happens-before through every
203        // task so the last decrement publishes all of their work.
204        BLOCKING_SCAN_TASKS_INFLIGHT.fetch_sub(1, Ordering::AcqRel);
205    }
206}
207
208/// Blocking scan tasks still running (issue #3384). ZERO means every detached
209/// blocking half has stopped, so `stream_walk_partitions_parsed` is final.
210pub fn blocking_scan_tasks_inflight() -> i64 {
211    BLOCKING_SCAN_TASKS_INFLIGHT.load(Ordering::Acquire)
212}

214/// Times a batched scan took the STITCHING branch (issue #3384).
215static BATCHED_SCAN_STITCHING_PATHS: AtomicU64 = AtomicU64::new(0);
216
217/// Record that a batched scan routed to `run_scan_stream_windowed` — the
218/// STITCHING branch (issue #3384).
219///
220/// Exists so a test can AFFIRMATIVELY measure which branch its fixture took, rather
221/// than infer it. The completion signal below covers only the non-stitching loop, and
222/// the obvious proxies for "am I on that loop" are all wrong: absence of
223/// `CompressionInfo.db` does not imply it (roborev), and neither does the fixture being
224/// uncompressed in the colloquial sense — `requires_chunk_stitching()` is
225/// `data_format() == V5CompressedLegacy && is_nb_format()`, and `V5_0Uncompressed` maps
226/// to the FORMER but not the LATTER. Two proxies, two different wrong answers; the
227/// branch itself is the only thing worth asserting on.
228pub fn mark_batched_scan_stitching_path() {
229    BATCHED_SCAN_STITCHING_PATHS.fetch_add(1, Ordering::Release);
230}
231
232/// Batched scans that took the stitching branch since
233/// [`reset_batched_scan_stitching_paths`] (issue #3384).
234pub fn batched_scan_stitching_paths() -> u64 {
235    BATCHED_SCAN_STITCHING_PATHS.load(Ordering::Acquire)
236}
237
238/// Zero the stitching-branch count (issue #3384).
239pub fn reset_batched_scan_stitching_paths() {
240    BATCHED_SCAN_STITCHING_PATHS.store(0, Ordering::Release);
241}

243/// Detached `BatchedScanStream` tasks that have terminated, published with
244/// RELEASE ordering (issue #3384).
245static BATCHED_SCANS_FINISHED: AtomicU64 = AtomicU64::new(0);
246
247/// Publish that one detached batched-scan task has terminated.
248///
249/// The SECOND of three abandoned-walk completion signals. `mark_query_row_producer_finished`
250/// proves the outer query-row thread stopped pulling; this proves the detached task it
251/// dropped without joining has stopped DECODING. Only all three together make
252/// `stream_walk_partitions_parsed` final — waiting a fixed interval instead merely
253/// makes the race less likely, which is the defect this issue exists to remove.
254///
255/// # Scope, and it is narrower than "the scan has finished" (roborev, issue #3384)
256///
257/// This is published from a DROP guard on the spawned future, so it covers everything
258/// that decodes INSIDE that future — which is the whole of the block-by-block loop
259/// taken by non-stitching (uncompressed) readers.
260///
261/// It does NOT cover the STITCHING path. `requires_chunk_stitching()` (compressed `nb`)
262/// routes to `run_scan_stream_windowed`, which dispatches `spawn_blocking` parse/feed
263/// tasks; blocking tasks are not cancellable, so they can still be decoding and
264/// advancing `stream_walk_partitions_parsed` when this signal fires. On that path the
265/// signal is PREMATURE and must not be read as finality. Joining those tasks on
266/// cancellation is the real fix and is tracked on #3428.
267pub fn mark_batched_scan_finished() {
268    BATCHED_SCANS_FINISHED.fetch_add(1, Ordering::Release);
269}
270
271/// Detached batched-scan tasks that have terminated since
272/// [`reset_batched_scans_finished`] (issue #3384). `Acquire`, for the same
273/// happens-before reason as [`query_row_producers_finished`].
274pub fn batched_scans_finished() -> u64 {
275    BATCHED_SCANS_FINISHED.load(Ordering::Acquire)
276}
277
278/// Zero the detached batched-scan completion count (issue #3384).
279pub fn reset_batched_scans_finished() {
280    BATCHED_SCANS_FINISHED.store(0, Ordering::Release);
281}

283/// Zero the completion count. A test that will wait on
284/// [`query_row_producers_finished`] must call this first, while holding whatever
285/// lock serializes producers in its binary.
286pub fn reset_query_row_producers_finished() {
287    QUERY_ROW_PRODUCERS_FINISHED.store(0, Ordering::Release);
288}

290#[cfg(test)]
291mod tests {
292    use super::*;
293
294    /// A delta must count exactly the work recorded between the two snapshots,
295    /// independent of whatever a sibling test already accumulated.
296    #[test]
297    fn delta_counts_only_work_between_snapshots() {
298        let before = ReadPathProbe::snapshot();
299        record_merger_built();
300        record_reconcile_entry();
301        record_reconcile_entry();
302        record_cell_metadata_map();
303        let delta = ReadPathProbe::snapshot().delta_since(&before);
304        assert!(delta.mergers_built >= 1, "the merger build was recorded");
305        assert!(delta.reconcile_entries >= 2, "both reconciles recorded");
306        assert!(delta.cell_metadata_maps >= 1, "the map alloc was recorded");
307        assert!(delta.any_merge_work());
308    }
309
310    /// A zero delta reports no merge work — the shape the bypass assertion uses.
311    #[test]
312    fn zero_delta_reports_no_merge_work() {
313        let a = ReadPathProbe {
314            mergers_built: 7,
315            reconcile_entries: 9,
316            cell_metadata_maps: 11,
317        };
318        let delta = a.delta_since(&a);
319        assert_eq!(delta, ReadPathProbe::default());
320        assert!(!delta.any_merge_work());
321    }
