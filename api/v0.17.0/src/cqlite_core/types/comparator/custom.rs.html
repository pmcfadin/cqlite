1//! Value ordering for the CQL scalar types that map to
2//! [`ComparatorType::Custom`](super::ComparatorType) (issue #3790).
3//!
4//! `ComparatorType::from_cql_type` represents `CqlType::Time` as
5//! `Custom("time")` and `CqlType::Inet` as `Custom("inet")`. Both are legal
6//! Cassandra clustering-column types, so both MUST order by value; the generic
7//! `Custom(_)` fallback orders by the FORMATTED STRING, which misorders them
8//! (e.g. `"10.0.0.2" < "9.0.0.1"` as text, while `[10,0,0,2] > [9,0,0,1]` as
9//! address bytes).
10//!
11//! Dispatching on the `Custom(name)` string is schema-driven — the name is
12//! derived authoritatively from the declared type, NOT inferred from byte
13//! patterns (no-heuristics mandate, issue #28). Same idiom as
14//! `storage::sstable::reader::parsing::custom_scalar`.
15//!
16//! Format authority — pinned Cassandra 5.0.8 (a CQLite `file:line` is never
17//! format authority, #3041):
18//! * `src/java/org/apache/cassandra/db/marshal/InetAddressType.java`:
19//!   `InetAddressType() { super(ComparisonType.BYTE_ORDER); }`
20//! * `src/java/org/apache/cassandra/db/marshal/TimeType.java`:
21//!   `TimeType() { super(ComparisonType.BYTE_ORDER); }`
22//!
23//! `ComparisonType.BYTE_ORDER` resolves to `ByteBufferUtil.compareUnsigned`:
24//! UNSIGNED lexicographic comparison of the serialized bytes, with the shorter
25//! operand first where it is a prefix of the longer.
26
27use crate::types::Value;
28use crate::{Error, Result};
29use std::cmp::Ordering;
30
31/// Compare two values under `ComparatorType::Custom(name)`.
32///
33/// `inet` and `time` order by value (see the module docs). Every OTHER name is
34/// the RESIDUAL path — an unresolved UDT reference or a genuinely unknown type,
35/// for which no value ordering is defined — and keeps the historical
36/// formatted-string comparison. That string compare is ONLY that residual.
37pub(super) fn compare(name: &str, left: &Value, right: &Value) -> Result<Ordering> {
38    match name {
39        "inet" => compare_inet(left, right),
40        "time" => compare_time(left, right),
41        _ => {
42            let l_str = format!("{}", left);
43            let r_str = format!("{}", right);
44            Ok(l_str.cmp(&r_str))
45        }
46    }
47}
48
49/// True when `Custom(name)` denotes a scalar type that orders by value.
50///
51/// `supports_ordering()` is public API and must not lie: `inet` and `time` are
52/// legal Cassandra clustering-column types. The residual unresolved-UDT /
53/// unknown name does not order.
54pub(super) fn supports_ordering(name: &str) -> bool {
55    name == "inet" || name == "time"
56}
57
58/// `inet`: unsigned bytewise comparison of the RAW serialized address —
59/// `Value::Inet` already holds exactly those bytes (4 for IPv4, 16 for IPv6),
60/// so no deserialization is needed.
61///
62/// Rust's `<[u8] as Ord>::cmp` IS `ByteBufferUtil.compareUnsigned`: unsigned
63/// lexicographic, and where one operand is a prefix of the other the SHORTER
64/// sorts first. That is also what settles the IPv4-vs-IPv6 differing-length
65/// case, so length needs no special handling here.
66fn compare_inet(left: &Value, right: &Value) -> Result<Ordering> {
67    match (left, right) {
68        (Value::Inet(l), Value::Inet(r)) => Ok(l.as_ref().cmp(r.as_ref())),
69        _ => Err(Error::Schema(
70            "Type mismatch: expected inet values".to_string(),
71        )),
72    }
73}

75/// `time`: comparison of the 8-byte BIG-ENDIAN nanos-since-midnight long, as
76/// unsigned bytes — `TimeType`'s `super(ComparisonType.BYTE_ORDER)`.
77///
78/// This is deliberately NOT a plain signed `i64::cmp`. Over `time`'s valid
79/// range (`0..=86_399_999_999_999`) every value is non-negative, so byte order,
80/// unsigned order and signed numeric order all coincide and this IS numeric
81/// order. They diverge only for an out-of-range NEGATIVE nanos, whose sign bit
82/// sets the leading byte to `0xFF`: Cassandra's BYTE_ORDER sorts it ABOVE every
83/// non-negative value, where signed `i64::cmp` would sort it below. Comparing
84/// `to_be_bytes()` is the serialized form verbatim, so THIS COMPARATOR agrees
85/// with Cassandra for all inputs.
86///
87/// CONVERGENCE (#3935): every `time` ordering site in the tree now compares the
88/// 8-byte big-endian serialized form as unsigned bytes, so they agree for ALL
89/// inputs — in-range and out-of-range-negative alike. The enumerated set:
90///   * THIS comparator, `types::comparator::custom::compare_time`, reached by
91///     `ComparatorType::Custom("time")` on the read/clustering path;
92///   * `Value::PartialOrd`'s `Time` arm (`types::value_ord`) — corrected from signed
93///     `i64::partial_cmp` by #3935. The query-side comparator
94///     `select_executor::value_ops::try_compare_values` delegates here for
95///     same-variant operands (`Value::Time` has no `as_f64`, so it never takes
96///     the numeric-coercion branch), so it converged with it;
97///   * `write_engine::clustering_order::compare_values` — corrected from signed
98///     `i64::cmp` by #3935. Both `ClusteringKey::compare` and `ClusteringKey`'s
99///     `Ord` call it, and `write_engine::merge` sorts merged rows with
100///     `ClusteringKey::compare` under the comment "Sort merged rows by
101///     clustering key for output order", so it governs the memtable `BTreeMap`
102///     placement and the merge order, and it is what WOULD decide the PHYSICAL
103///     ROW ORDER written to `Data.db` for a `time` CLUSTERING COLUMN. Read
104///     "would": TODAY `data_writer/encoding::serialize_value_for_clustering`
105///     has no `Custom` arm, so `(Value::Time(_), Custom("time"))` hits its
106///     `_ =>` fail-closed refusal and CQLite cannot write such a column at all
107///     — pinned by
108///     `issue_3935_time_clustering_row_order`'s
109///     `a_time_clustering_column_is_refused_by_the_write_path_declared_gap`.
110///     It is the site that would REACH DISK first once that arm lands, which is
111///     why a partial fix was refused;
112///   * `cqlite-flight`'s `agg::compare_values` — the `Min`/`Max` accumulator's
113///     total order for pushed-down aggregates on the Arrow Flight / Trino
114///     surface, corrected from signed by #3935. It shared ONE signed arm with
115///     `BigInt`/`Counter`/`Timestamp`, and it was missed by this census's first
116///     pass because that grep was scoped to `cqlite-core` — see the
117///     NON-EXHAUSTIVENESS note below;
118///   * the whole-collection AND frozen writer,
119///     `data_writer/collection_order::compare_collection_elements` — called both
120///     by `write_set_complex_cells`/`write_map_complex_cells` (which then emit
121///     cell paths in that order with no re-sort) and by `encoding.rs`'s frozen
122///     `serialize_value` — corrected from signed by #3935;
123///   * the frozen sorted-collection canonicalizer,
124///     `data_writer/marshal_comparator::classify_comparator`, which orders a UDT's
125///     `SetType` element / `MapType` key field and had the SAME defect
126///     independently (`serialize_collection_elements` does not re-sort, so its
127///     order is the on-disk order for a UDT field) — corrected by #3935;
128///   * the per-element writer, `schema_helpers::compare_cell_paths`, and the
129///     merged-read raw-byte fast path
130///     (`merge::read_assembly::comparator_orders_by_raw_cell_path_bytes`), both
131///     already unsigned raw bytes.
132///
133/// The enumeration is a CENSUS, not a proof of closure: it records the sites
134/// found by grepping every `Value::Time`-vs-`Value::Time` comparison arm plus
135/// every `TimeType` classification in the marshal-name comparators, and a NEW
136/// comparator added later would not be in it. NON-EXHAUSTIVENESS IS NOT
137/// HYPOTHETICAL HERE: the first pass ran that grep across `cqlite-core` ONLY and
138/// therefore missed `cqlite-flight`'s `agg::compare_values` entirely, so any
139/// later re-census must sweep EVERY workspace member, not just this crate.
140///
141/// The REMAINING signed comparisons of `time` VALUES in the tree — whether
142/// spelled as a `Value::Time` arm or as `i64::cmp` over the same nanos —
143/// enumerated BY NAME rather than asserted as a count:
144///
145///   * in LIBRARY code — NONE. After #3935 every library site compares the
146///     8-byte big-endian form as unsigned bytes;
147///   * `data_writer::collection_order::tests::time_negative_nanos_sorts_above_every_non_negative`
148///     — a deliberate test-only NEGATIVE CONTROL: it re-sorts the same input
149///     with the OLD signed closure and asserts the two sequences DIFFER, so the
150///     byte-order assertion provably has teeth rather than being satisfiable by
151///     any total order;
152///   * `data_writer::collection_order::tests::in_range_time_order_is_unchanged`
153///     — a deliberate test-only signed REFERENCE ORACLE: over the valid range
154///     `0..=86_399_999_999_999` every value has a `0x00` sign byte, so unsigned
155///     byte order and signed numeric order COINCIDE, and the case requires the
156///     new comparator to EQUAL `i64::cmp` on every in-range pair. That is the
157///     compatibility half of the fix; a signed comparison is the correct oracle
158///     for it;
159///   * `cqlite-flight`'s
160///     `agg::tests::time_min_max_orders_by_byte_order_not_signed` — the same
161///     NEGATIVE-CONTROL shape one crate over: a plain `Vec<i64>::sort()` (signed)
162///     over the identical input, required to DIFFER from the comparator's
163///     sequence.
164///
165/// Two neighbours that are NOT in this set, so the boundary is not guessed at:
166/// `data_writer::collection_order::tests::timestamp_keeps_signed_order` and
167/// `cqlite-flight`'s `agg::tests::timestamp_min_max_keeps_signed_order` are
168/// signed comparisons of `Value::Timestamp`, which is CORRECT and permanent (see
169/// the next paragraph). The second also re-sorts a `time` sequence, but through
170/// the now-unsigned comparator, and asserts the two types order the SAME input
171/// DIFFERENTLY — the pin against re-unifying the arms.
172///
173/// `Value::Timestamp` is deliberately NOT in the converged set and must stay
174/// SIGNED everywhere — `TimestampType` is `ComparisonType.CUSTOM` and its
175/// `compareCustom` delegates to `LongType.compareLongs` — so do not "unify" a
176/// `Time` arm with a `Timestamp` one. The convergence is pinned by
177/// `cqlite-core/tests/issue_3935_collection_time_byte_order.rs`,
178/// `cqlite-core/tests/issue_3935_time_clustering_row_order.rs` and
179/// `cqlite-flight`'s `agg::tests::time_min_max_orders_by_byte_order_not_signed`.
180///
181/// # CANONICAL STATEMENT: range validation would *not* close the class
182///
183/// This is the ONE place this argument is written out; the other `time`-ordering
184/// sites (`data_writer/collection_order`, `data_writer/marshal_comparator`, and
185/// the `issue_3935_*` / `issue_3790_*` test targets) point HERE rather than
186/// restating it, so a future re-pin has one paragraph to correct instead of four.
187///
188/// RANGE VALIDATION WOULD *NOT* CLOSE THE CLASS — an earlier revision of this
189/// comment called it "the fix that would make all the sites agree trivially",
190/// and that claim is FALSIFIED by the pinned source. `TimeType` has NO
191/// `validate` override, and `serializers/TimeSerializer.java:71-75` `validate`
192/// checks the SIZE ONLY:
193/// `if (accessor.size(value) != 8) throw new MarshalException(...)`. The range
194/// check `result < 0 || result >= TimeUnit.DAYS.toNanos(1)` lives ONLY in
195/// `timeStringToLong` (`TimeSerializer.java:50`), the CQL string-literal / JSON
196/// path. So an 8-byte BINARY out-of-range `time` passes Cassandra's own
197/// validation, is stored, and is ordered by `TimeType`'s BYTE_ORDER. Cassandra
198/// does not reject such a value; BYTE_ORDER is simply what the pinned tag
199/// specifies for it, which is why byte-order-exactness — not validation — is the
200/// rule every site has converged on (see CONVERGENCE above). Range validation was
201/// explicitly REFUSED for #3935: if stock Cassandra can write it, CQLite must read
202/// it, so a range check would make CQLite reject data Cassandra created.
203fn compare_time(left: &Value, right: &Value) -> Result<Ordering> {
204    match (left, right) {
205        (Value::Time(l), Value::Time(r)) => Ok(l.to_be_bytes().cmp(&r.to_be_bytes())),
206        _ => Err(Error::Schema(
207            "Type mismatch: expected time values".to_string(),
208        )),
209    }
