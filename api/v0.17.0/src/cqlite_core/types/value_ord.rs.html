1//! Value ordering for [`Value`](super::Value) — the `PartialOrd` impl.
2//!
3//! Split out of `types.rs` under the campsite rule (epic #1116): that file is far
4//! over the 800-line source target, and this comparator's contract needs a long
5//! doc comment (two divergences from the derived `PartialEq`, and the
6//! `time`/`timestamp` asymmetry) rather than a short one.
7
8use crate::types::Value;
9
10/// Ordering for `Value`.
11///
12/// NOTE (contract split, #1870/#2010/#2074): this `PartialOrd` INTENTIONALLY
13/// diverges from the DERIVED `PartialEq`. For `float`/`double` it uses the
14/// Cassandra/Java total order (`-0.0 < +0.0`; every `NaN` sorts last and compares
15/// Equal to every other `NaN`), whereas the derived `PartialEq` keeps IEEE
16/// semantics (`-0.0 == +0.0`, `NaN != NaN`). `partial_cmp` may thus report `Equal`
17/// where `eq` reports `false` (two NaNs) and vice-versa (`-0.0`/`+0.0`). GROUP BY
18/// grouping (issue #2074) does NOT use the derived `PartialEq` for floats: the
19/// aggregation group-key path (`aggregation::group_key_eq` + `hash_group_key`)
20/// routes them through this SAME total order (all NaN → ONE group; `-0.0`/`+0.0`
21/// DISTINCT). Any future `impl Ord for Value` MUST reuse this comparator, never
22/// derive from `PartialEq`.
23///
24/// NOTE (`time` vs `timestamp`, #3935): the `Time` arm compares the 8-byte
25/// BIG-ENDIAN serialized form as UNSIGNED bytes, matching `TimeType`'s
26/// `super(ComparisonType.BYTE_ORDER)` (pinned `cassandra-5.0.8`
27/// `db/marshal/TimeType.java`); `ByteBufferUtil.compareUnsigned` is what
28/// `BYTE_ORDER` resolves to. The `Timestamp` arm stays SIGNED numeric, because
29/// `TimestampType` is `ComparisonType.CUSTOM` and its `compareCustom` delegates
30/// to `LongType.compareLongs`. The two arms therefore diverge BY DESIGN for a
31/// negative long, and unifying them would break one of the two types. This arm
32/// now agrees with `types::comparator::custom::compare_time` and with the write
33/// path's `write_engine::clustering_order::compare_values` for ALL inputs.
34impl PartialOrd for Value {
35    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
36        use crate::float_cmp::{cassandra_double_cmp as dcmp, cassandra_float_cmp as fcmp};
37        use std::cmp::Ordering;
38        match (self, other) {
39            (Value::Null, Value::Null) => Some(Ordering::Equal),
40            (Value::Null, _) => Some(Ordering::Less),
41            (_, Value::Null) => Some(Ordering::Greater),
42
43            // EMPTY-BUFFER SENTINEL (issue #3805). `Int32Type.compareCustom`
44            // (`db/marshal/Int32Type.java:61-71` at `cassandra-5.0.8`) is
45            //
46            //     if (accessorL.isEmpty(left) || accessorR.isEmpty(right))
47            //         return Boolean.compare(accessorR.isEmpty(right), accessorL.isEmpty(left));
48            //
49            // so with only the LEFT empty: `Boolean.compare(false, true) == -1`.
50            // An empty buffer therefore sorts strictly BEFORE every non-empty
51            // value of its type, INCLUDING `Integer.MIN_VALUE`, and two empties
52            // are Equal (both branches empty ⇒ `Boolean.compare(true, true)`
53            // == 0). Measured on real Cassandra-5.0.2 bytes for four independent
54            // key types — the empty key is FIRST in every column
55            // (`docs/round-artifacts/issue-3805-cassandra-oracle.md` §4b.4).
56            //
57            // These arms sit immediately after the `Null` arms so `Null` keeps
58            // its existing first position (`Null` is not a key Cassandra admits
59            // at all, so no Cassandra ordering is being contradicted) and
60            // BEFORE the per-variant arms, so a sentinel can never fall through
61            // to the `to_string()` fallback below.
62            (Value::Empty(a), Value::Empty(b)) => {
63                // Same type ⇒ the same (empty) bytes ⇒ Equal, exactly as
64                // `compareCustom` reports for two empty buffers. Different
65                // declared types is a cross-type comparison, which Cassandra
66                // never performs (a map has ONE key type); ordering by the tag
67                // keeps this total and deterministic rather than arbitrary.
68                // That tag ordering is a TOTALITY requirement of this impl and
69                // NOT a claim that the pair is comparable: the diagnosing
70                // comparator refuses it (`select_executor::value_ops::
71                // try_compare_values` decides `(Empty, Empty)` itself, above
72                // its discriminant test, and errors on mismatched tags —
73                // roborev job 451).
74                Some(a.cmp(b))
75            }
76            (Value::Empty(_), _) => Some(Ordering::Less),
77            (_, Value::Empty(_)) => Some(Ordering::Greater),
78
79            (Value::Boolean(a), Value::Boolean(b)) => a.partial_cmp(b),
80            (Value::Integer(a), Value::Integer(b)) => a.partial_cmp(b),
81            (Value::BigInt(a), Value::BigInt(b)) => a.partial_cmp(b),
82            (Value::Counter(a), Value::Counter(b)) => a.partial_cmp(b),
83            (Value::Float(a), Value::Float(b)) => Some(dcmp(*a, *b)),
84            (Value::Text(a), Value::Text(b)) => a.partial_cmp(b),
85            (Value::Blob(a), Value::Blob(b)) => a.partial_cmp(b),
86            // `timestamp` (`TimestampType`) is ComparisonType.CUSTOM and its
87            // `compareCustom` delegates to `LongType.compareLongs` — SIGNED.
88            // `time` (`TimeType`) is ComparisonType.BYTE_ORDER — UNSIGNED bytes.
89            // The asymmetry is Cassandra's; do NOT "unify" these two arms.
90            (Value::Timestamp(a), Value::Timestamp(b)) => a.partial_cmp(b),
91            (Value::Time(a), Value::Time(b)) => a.to_be_bytes().partial_cmp(&b.to_be_bytes()),
92            (Value::Date(a), Value::Date(b)) => a.partial_cmp(b),
93            (Value::Uuid(a), Value::Uuid(b)) => a.partial_cmp(b),
94            (Value::TinyInt(a), Value::TinyInt(b)) => a.partial_cmp(b),
95            (Value::SmallInt(a), Value::SmallInt(b)) => a.partial_cmp(b),
96            (Value::Float32(a), Value::Float32(b)) => Some(fcmp(*a, *b)),
97            (Value::Inet(a), Value::Inet(b)) => a.partial_cmp(b),
98
99            // For complex types, compare by string representation
100            (a, b) => a.to_string().partial_cmp(&b.to_string()),
101        }
102    }
103}
