1//! Deprecation reporting for configuration keys CQLite has REMOVED (issue #1696).
2//!
3//! # Why a Rust struct is not the whole story
4//!
5//! Deleting a decorative field from [`crate::Config`] is a COMPILE error for an
6//! embedder writing Rust — the loudest signal available. But `Config` derives
7//! `Deserialize`, and serde's default behaviour is to DISCARD unknown fields, so
8//! every non-Rust authoring surface gets the opposite: a pre-change document that
9//! still says
10//!
11//! ```json
12//! { "storage": { "block_size": 65536 }, "query": { "parallel": { "enabled": true } } }
13//! ```
14//!
15//! deserializes SUCCESSFULLY and is silently ignored. The Python bindings' dict /
16//! JSON bridge is exactly that surface (`cqlite.open(path, config={...})`), which
17//! made the crate-wide rule stated in #1696
18//!
19//! > A removed knob must produce a LOUD signal at the layer where it is set,
20//! > never silence.
21//!
22//! false for anyone configuring CQLite through the bindings (#1696 roborev F1).
23//!
24//! # WHERE THE RULE IS ENFORCED — and where it is NOT (#1696 roborev r2 F3)
25//!
26//! Stated exactly, because a rule quoted without its scope reads as universal
27//! coverage and this one does not have it. The rule holds on:
28//!
29//! * the CLI's config-file loader (`cqlite_cli::config::removed_keys`) — a named
30//!   warning on stderr;
31//! * the Python bindings' dict/JSON entry points — a `UserWarning` naming each
32//!   dead path (visible under Python's DEFAULT filters, which is why the category
33//!   is not `DeprecationWarning`);
34//! * Rust FIELD ACCESS — a compile error, and only for callers writing Rust.
35//!
36//! It does NOT hold on a DIRECT serde deserialization of [`crate::Config`].
37//! `Config` derives `Deserialize`, so `serde_json::from_str::<Config>` /
38//! `from_value::<Config>` bypass [`crate::Config::from_json_str`] and
39//! [`crate::Config::from_json_str_reporting_removed`] entirely, and serde
40//! silently DISCARDS the removed keys. An embedder deserializing a `Config`
41//! document themselves therefore still gets silence: a reporting constructor is
42//! OPTIONAL, and an optional constructor enforces nothing at the boundary it sits
43//! beside. That residual is **issue #3520**, pinned by
44//! `direct_serde_deserialization_is_the_unreported_surface` in this module's
45//! tests. It was scoped out of #1696 rather than fixed: closing it needs a custom
46//! `Deserialize` capturing unknown keys across every nested config struct.
47//!
48//! # Why not `deny_unknown_fields`
49//!
50//! Because it would HARD-FAIL a Python caller whose config predates the removal,
51//! with no migration path, for keys that never did anything. That is the opposite
52//! of the parse-and-ignore-PLUS-a-named-warning posture deliberately chosen for
53//! the CLI's file surface, and #1696 requires ONE consistent posture crate-wide.
54//! So this module is the same mechanism, at the deserialization boundary.
55//!
56//! # One mechanism, one table per DOCUMENT SCHEMA
57//!
58//! [`Removed`], [`removed_keys_present`] and [`deprecation_warning`] live here and
59//! are shared with `cqlite_cli::config::removed_keys`, so the wording, the
60//! matching rule and the shape cannot drift between the two surfaces.
61//!
62//! The TABLES are necessarily per-schema and must not be crossed: this crate's
63//! [`REMOVED_KEYS`] describes a `cqlite_core::Config` document, while the CLI's
64//! describes a `cqlite.toml`/`.yaml`/`.json`. Applying this table to a CLI file
65//! would be WRONG, not merely redundant — the CLI's `[performance]` section is
66//! live (`query_timeout_ms`, `memory_limit_mb`, `cache_size_mb`) while the core's
67//! `performance` tree was removed, so a shared table would scold a user for a key
68//! that works.
69

72/// A configuration key removed by an issue, and why it is gone.
73#[derive(Debug, Clone, Copy, PartialEq, Eq)]
74pub struct Removed {
75    /// Dotted path as it appears in the document, e.g. `storage.block_size`.
76    pub path: &'static str,
77    /// Short note shown to the user: what it did, or that it never did anything.
78    pub note: &'static str,
79}

149/// Which of `table`'s removed keys a document still names.
150///
151/// `has_path` answers "does the document contain this dotted path as a mapping
152/// key" for one concrete document type, so the FILTER is written once and every
153/// surface (this crate's JSON, the CLI's TOML/YAML/JSON) shares it.
154pub fn removed_keys_present(table: &[Removed], has_path: impl Fn(&str) -> bool) -> Vec<&Removed> {
155    table
156        .iter()
157        .filter(|removed| has_path(removed.path))
158        .collect()
159}

161/// The operator-facing deprecation warning for a set of still-named removed keys,
162/// or `None` when the document names none.
163///
164/// Returned as a string rather than logged here so the caller picks the sink (the
165/// CLI prints to stderr, the bindings raise a Python `UserWarning`) and a test can
166/// assert the exact text.
167///
168/// # THE TEXT REPORTS ONLY WHICH KEYS ARE DEAD — never the fate of the load
169///
170/// Stated as a rule at the seam, because the alternative was found broken THREE
171/// times (#1696 roborev F3, r2 F3, r5 F1). The text used to add an assurance —
172/// "they are IGNORED, the configuration still loads" — which is a claim about a
173/// LATER stage's outcome. Every fix moved the emission one stage later
174/// (after deserialization; then after the bindings' validation) and the defect
175/// reappeared at the next stage, because there is ALWAYS a next stage: the CLI
176/// deserializes, then maps into `cqlite_core::Config`, then validates
177/// semantically (`memory_limit_mb = 1` with `cache_size_mb = 64` fails there),
178/// then the caller does whatever it does. No placement can make such a promise
179/// safe.
180///
181/// So this warning says nothing about whether the load succeeds. It names the
182/// keys and states that they have no effect — two facts fully known at the point
183/// of the scan. A warning that only reports what it knows cannot be wrong about
184/// anything else, and callers are consequently free to emit it wherever the raw
185/// document is still in hand. DO NOT reintroduce an outcome claim here.
186pub fn deprecation_warning(source: &str, present: &[&'static Removed]) -> Option<String> {
187    if present.is_empty() {
188        return None;
189    }
190    let mut out = format!(
191        "warning: {source} names {} configuration key{} that CQLite has REMOVED. \
192         Each has NO EFFECT:\n",
193        present.len(),
194        if present.len() == 1 { "" } else { "s" }
195    );
196    for removed in present {
197        // Writing into a String is infallible; `let _` keeps this free of an
198        // `unwrap()` in library code.
199        let _ = writeln!(out, "  - {}: {}", removed.path, removed.note);
200    }
201    out.push_str("Delete them to silence this warning.");
202    Some(out)
203}

205/// Whether a parsed JSON document contains `path` as an object key.
206///
207/// Walked segment by segment, so `storage.block_size` matches only a
208/// `block_size` under `storage` — never a top-level one, nor a
209/// `query.block_size`. Only KEY PRESENCE matters: a removed key set to any value
210/// at all, `null` included, is still someone believing they configured something.
211pub fn json_has_path(root: &serde_json::Value, path: &str) -> bool {
212    let mut cursor = root;
213    for segment in path.split('.') {
214        match cursor.as_object().and_then(|object| object.get(segment)) {
215            Some(next) => cursor = next,
216            None => return false,
217        }
218    }
219    true
220}

222/// The deprecation warning a JSON `Config` document should produce, or `None`
223/// when it names no removed key (or is not parseable JSON at all).
224///
225/// Unparseable content yields `None` rather than an error: this scan must never
226/// be the thing that rejects a document. [`crate::Config::from_json_str`] owns
227/// the real parse and the real error, and only reaches this once that parse
228/// SUCCEEDED.
229pub fn warning_for_json(source: &str, content: &str) -> Option<String> {
230    let document = serde_json::from_str::<serde_json::Value>(content).ok()?;
231    let present = removed_keys_present(REMOVED_KEYS, |path| json_has_path(&document, path));
232    deprecation_warning(source, &present)
233}
