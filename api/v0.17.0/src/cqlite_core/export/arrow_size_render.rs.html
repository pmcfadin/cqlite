1//! Rendered-representation bounds for the Arrow payload-byte estimator
2//! (issue #2825).
3//!
4//! Columns with no authoritative CQL type (and the `Tuple`/`Udt`/`Frozen`/
5//! `Tombstone`/`Null` flat arms) are converted by `build_string_array`, which
6//! renders the value through `ValueFormatter::format_value`; the flat
7//! `build_list_array` / `build_map_array` render each ELEMENT the same way. The
8//! constants here each bound one `format_value` arm, and
9//! [`Estimator::charge_rendered`](super::Estimator::charge_rendered) walks a
10//! value against them.
11//!
12//! Split out of `arrow_size.rs` so both files stay under the campsite threshold
13//! (epic #1116). Declared as a CHILD module of `arrow_size`, so it can see that
14//! module's private [`Estimator`](super::Estimator) and [`Shape`](super::Shape).
15
16use super::{Estimator, Shape};
17use crate::types::Value;

19/// `"true"` / `"false"`.
20pub(super) const RENDER_BOOL_BYTES: usize = 8;
21/// Any integral variant rendered as decimal (`i64::MIN` is 20 chars).
22pub(super) const RENDER_INT_BYTES: usize = 24;
23/// `f32`/`f64` via `{}`/`{:e}`, plus `NaN`/`Infinity`.
24pub(super) const RENDER_FLOAT_BYTES: usize = 40;
25/// `"a8f167f0-ebe7-4f20-a386-31ff138bec3b"`.
26pub(super) const RENDER_UUID_BYTES: usize = 40;
27/// `YYYY-MM-DD HH:MM:SS.fff+0000` or `<invalid-timestamp:-9223372036854775808>`.
28pub(super) const RENDER_TIMESTAMP_BYTES: usize = 48;
29/// `YYYY-MM-DD` or the `<invalid-date:…>` fallback.
30pub(super) const RENDER_DATE_BYTES: usize = 48;
31/// `HH:MM:SS.nnnnnnnnn` or the `<invalid-time:…>` fallback.
32pub(super) const RENDER_TIME_BYTES: usize = 48;
33/// Full IPv6 text form, or the invalid-length fallback.
34pub(super) const RENDER_INET_BYTES: usize = 64;
35/// `"{months}mo{days}d{nanos}ns"` at the widest.
36pub(super) const RENDER_DURATION_BYTES: usize = 64;
37/// `"<deleted@{i64}>"`.
38pub(super) const RENDER_TOMBSTONE_BYTES: usize = 40;
39/// `"null"`.
40pub(super) const RENDER_NULL_BYTES: usize = 8;
41/// The brackets/braces a rendered container puts AROUND its children.
42pub(super) const RENDER_CONTAINER_BYTES: usize = 8;
43/// The `", "` (or `": "`) a rendered container puts BETWEEN its children —
44/// charged per child, because the children themselves are charged separately.
45pub(super) const RENDER_SEPARATOR_BYTES: usize = 2;
46/// Decimal digits produced per magnitude byte (`log10(256) < 2.41`), rounded up.
47pub(super) const DECIMAL_DIGITS_PER_BYTE: usize = 3;
48/// `ValueFormatter::format_decimal`'s zero-padding ceiling: past this the render
49/// switches to bounded exponent form, so padding can never exceed it.
50pub(super) const DECIMAL_SCALE_RENDER_CAP: usize = 1_000_001;
51/// Worst-case UTF-8 expansion of one input byte under `String::from_utf8_lossy`:
52/// an invalid byte becomes U+FFFD, three bytes.
53pub(super) const LOSSY_UTF8_EXPANSION: usize = 3;
54/// Worst-case JSON escape of one input byte: `\u00XX`, six characters.
55pub(super) const JSON_ESCAPE_EXPANSION: usize = 6;
56
57impl<'a> Estimator<'a> {
58    /// Charge the `Utf8` rendering of `value` via `ValueFormatter::format_value`.
59    ///
60    /// Content only — the caller has already charged whatever slot overhead the
61    /// rendering lands in. Leaf variants get a constant bound; container
62    /// variants charge their bracket/separator overhead here and push their
63    /// children as further INLINE rendered nodes (they are sub-parts of this one
64    /// string, not array slots of their own).
65    pub(super) fn charge_rendered(&mut self, value: Option<&'a Value>) {
66        let Some(value) = value else {
67            return;
68        };
69        match super::unwrap_frozen_value(value) {
70            Value::Null => self.add(RENDER_NULL_BYTES),
71            // EMPTY-BUFFER SENTINEL (issue #3805): renders as the EMPTY STRING
72            // (`ValueFormatter::format_value` → `""`, matching
73            // `sstabledump`'s `"path" : [ "" ]`), so it contributes ZERO
74            // rendered content bytes. Charging 0 is exact here, not optimistic:
75            // the rendering has no length that can vary.
76            Value::Empty(_) => {}
77            Value::Boolean(_) => self.add(RENDER_BOOL_BYTES),
78            Value::TinyInt(_)
79            | Value::SmallInt(_)
80            | Value::Integer(_)
81            | Value::BigInt(_)
82            | Value::Counter(_) => self.add(RENDER_INT_BYTES),
83            Value::Float(_) | Value::Float32(_) => self.add(RENDER_FLOAT_BYTES),
84            // LOSSY path: `ValueFormatter::format_value` renders text with
85            // `String::from_utf8_lossy`, which expands EACH invalid byte to a
86            // 3-byte U+FFFD. `Value::Text` is UTF-8-validated at construction
87            // (issue #1644), which would make `s.len()` exact — but that
88            // invariant is not enforced by the type, and a `Value::Text` nested
89            // inside a rendered container reaches this arm without passing the
90            // strict builders' `str::from_utf8` check. Charge the lossy worst
91            // case (review B5); the STRICT typed arm keeps `s.len()` because its
92            // builder hard-errors on invalid UTF-8 instead of expanding it.
93            Value::Text(s) => self.add(s.len().saturating_mul(LOSSY_UTF8_EXPANSION)),
94            // `0x`-prefixed lowercase hex: two chars per byte.
95            Value::Blob(b) => self.add(
96                b.len()
97                    .saturating_mul(2)
98                    .saturating_add(RENDER_CONTAINER_BYTES),
99            ),
100            Value::Timestamp(_) => self.add(RENDER_TIMESTAMP_BYTES),
101            Value::Date(_) => self.add(RENDER_DATE_BYTES),
102            Value::Time(_) => self.add(RENDER_TIME_BYTES),
103            Value::Uuid(_) => self.add(RENDER_UUID_BYTES),
104            Value::Varint(b) => self.add(
105                b.len()
106                    .saturating_mul(DECIMAL_DIGITS_PER_BYTE)
107                    .saturating_add(RENDER_CONTAINER_BYTES),
108            ),
109            // Digits, plus `format_decimal`'s bounded zero padding (past
110            // `DECIMAL_SCALE_RENDER_CAP` it switches to exponent form).
111            Value::Decimal { scale, unscaled } => self.add(
112                unscaled
113                    .len()
114                    .saturating_mul(DECIMAL_DIGITS_PER_BYTE)
115                    .saturating_add((scale.unsigned_abs() as usize).min(DECIMAL_SCALE_RENDER_CAP))
116                    .saturating_add(RENDER_CONTAINER_BYTES),
117            ),
118            Value::Duration { .. } => self.add(RENDER_DURATION_BYTES),
119            Value::Inet(_) => self.add(RENDER_INET_BYTES),
120            Value::Tombstone(_) => self.add(RENDER_TOMBSTONE_BYTES),
121            Value::Json(json) => {
122                let bytes = json_render_bytes(json, &mut self.budget);
123                self.add(bytes);
124            }
125            Value::List(items) | Value::Set(items) | Value::Tuple(items) => {
126                self.add(
127                    RENDER_CONTAINER_BYTES
128                        .saturating_add(items.len().saturating_mul(RENDER_SEPARATOR_BYTES)),
129                );
130                self.charge_children(items.iter().map(|v| (Shape::RenderedInline, Some(v))));
131            }
132            Value::Map(pairs) => {
133                self.add(
134                    RENDER_CONTAINER_BYTES.saturating_add(
135                        pairs
136                            .len()
137                            .saturating_mul(2)
138                            .saturating_mul(RENDER_SEPARATOR_BYTES),
139                    ),
140                );
141                self.charge_children(pairs.iter().flat_map(|(k, v)| {
142                    [
143                        (Shape::RenderedInline, Some(k)),
144                        (Shape::RenderedInline, Some(v)),
145                    ]
146                }));
147            }
148            Value::Udt(udt) => {
149                self.add(RENDER_CONTAINER_BYTES);
150                for f in &udt.fields {
151                    if self.total == usize::MAX {
152                        return;
153                    }
154                    // `format_udt` emits `name: value` pairs — the name and both
155                    // separators are charged here, the value as a child node.
156                    self.add(f.name.len().saturating_add(RENDER_CONTAINER_BYTES));
157                    self.charge_child(Shape::RenderedInline, f.value.as_ref());
158                }
159            }
160            // Unreachable after `unwrap_frozen_value`; kept so the match is
161            // exhaustive without a `_` arm.
162            Value::Frozen(_) => self.add(RENDER_CONTAINER_BYTES),
163        }
164    }
165}

167/// Bounded upper bound on `serde_json::Value::to_string().len()`.
168///
169/// Iterative with its own share of the caller's node budget, so a deeply nested
170/// JSON document cannot recurse or spin. Returns `usize::MAX` when the budget is
171/// exhausted (fail closed).
172pub(super) fn json_render_bytes(root: &serde_json::Value, budget: &mut usize) -> usize {
173    let mut total = 0usize;
174    let mut stack = vec![root];
175    while let Some(node) = stack.pop() {
176        if *budget == 0 {
177            return usize::MAX;
178        }
179        *budget -= 1;
180        match node {
181            serde_json::Value::Null => total = total.saturating_add(RENDER_NULL_BYTES),
182            serde_json::Value::Bool(_) => total = total.saturating_add(RENDER_BOOL_BYTES),
183            serde_json::Value::Number(_) => total = total.saturating_add(RENDER_FLOAT_BYTES),
184            // JSON string escaping can expand a byte to `\u00XX` (6 chars).
185            serde_json::Value::String(s) => {
186                total = total.saturating_add(
187                    s.len()
188                        .saturating_mul(JSON_ESCAPE_EXPANSION)
189                        .saturating_add(RENDER_CONTAINER_BYTES),
190                )
191            }
192            serde_json::Value::Array(items) => {
193                total = total.saturating_add(
194                    RENDER_CONTAINER_BYTES
195                        .saturating_add(items.len().saturating_mul(RENDER_SEPARATOR_BYTES)),
196                );
197                if items.len() > *budget {
198                    return usize::MAX;
199                }
200                stack.extend(items.iter());
201            }
202            serde_json::Value::Object(map) => {
203                total = total.saturating_add(RENDER_CONTAINER_BYTES);
204                if map.len() > *budget {
205                    return usize::MAX;
206                }
207                for (k, v) in map {
208                    total = total.saturating_add(
209                        k.len()
210                            .saturating_mul(JSON_ESCAPE_EXPANSION)
211                            .saturating_add(RENDER_CONTAINER_BYTES),
212                    );
213                    stack.push(v);
214                }
215            }
216        }
217    }
218    total
