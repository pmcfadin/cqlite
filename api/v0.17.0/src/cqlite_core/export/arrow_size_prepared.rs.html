1//! Per-ROW charging over a RESOLVED column set (issues #2825 / #3552).
2//!
3//! # The responsibility boundary
4//!
5//! `arrow_size.rs` owns the charging MODEL — the structural constants, the public
6//! entry points, and [`Estimator`](super::Estimator), which charges ONE Arrow
7//! slot and the child slots it implies. This file owns the layer above it: given a
8//! whole ROW's worth of already-resolved cells, walk the projected columns and
9//! charge each one's slot.
10//!
11//! That is one responsibility with two callers, which is exactly why it is its own
12//! file:
13//!
14//! * [`charge_row`] is the loop BOTH accountings share —
15//!   [`estimate_arrow_row_bytes`](super::estimate_arrow_row_bytes), which resolves
16//!   each cell with a `values.get(name)` probe, and the fused push-time accounting
17//!   below, which resolves them once for the Arrow build pass. Sharing this loop is
18//!   what stops the two from drifting into two charging models; they differ ONLY in
19//!   how a cell is resolved, and their equivalence is pinned per row over the shared
20//!   shape corpus by `arrow_row_accumulator`'s
21//!   `fused_width_equals_the_standalone_estimate_over_the_shape_corpus`.
22//! * [`PreparedColumns`] is the fused side's cache: the per-column slot SHAPES,
23//!   resolved once per column set instead of once per row, plus the two cell
24//!   resolutions that map `super::super::arrow_row_accumulator`'s storage layouts
25//!   (a dense row-major staging slot, and a sparse column-major store) onto the
26//!   `(shape, cell)` pairs [`charge_row`] consumes.
27//!
28//! So the seam is per-SLOT charging (parent) versus per-ROW iteration over a
29//! resolved column set (here) — not a line count. Nothing here knows how a slot is
30//! charged, and nothing in the parent knows how a caller stores its cells.
31//!
32//! Split out of `arrow_size.rs` when the fused accounting pushed that file past the
33//! campsite threshold (epic #1116; the file was 657 lines before issue #3552 and
34//! 813 after). Declared as a CHILD module of `arrow_size`, so it can see that
35//! module's private [`Estimator`](super::Estimator) and [`Shape`](super::Shape) —
36//! the same wiring `arrow_size_shape.rs` and `arrow_size_render.rs` use.
37//!
38//! Visibility is unchanged by the move: [`PreparedColumns`] and its methods stay
39//! `pub(in crate::export)` and are re-exported from `arrow_size` under that same
40//! restriction, so the accumulator's import path does not change and no surface
41//! widened to make the split possible.
42

47/// The charging core BOTH accountings share (issue #3552).
48///
49/// Takes one `(slot shape, resolved cell)` pair per projected column, in column
50/// order, and charges the model this file documents. `None` is an ABSENT column
51/// and is charged exactly like a present-but-null one: its validity byte, its
52/// shape's structural overhead and its per-column residual are all still owed,
53/// because the converter materializes a slot for every projected column of every
54/// row whether the row carries the cell or not.
55///
56/// Sharing this is what stops the standalone estimator and the push-time fused
57/// accounting from drifting into two charging models — they differ ONLY in how a
58/// column's cell is resolved, which is the property the fused accounting's
59/// equivalence test pins.
60///
61/// `pub(super)` and no wider: `arrow_size`'s own `estimate_arrow_row_bytes` is the
62/// second caller (a PARENT cannot see a child's private items, unlike the reverse),
63/// and nothing outside `arrow_size` has any business charging a row directly.
64pub(super) fn charge_row<'a, I>(cells: I) -> usize
65where
66    I: IntoIterator<Item = (Shape<'a>, Option<&'a Value>)>,
67{
68    let mut est = Estimator::new();
69    for (shape, cell) in cells {
70        // Both node budgets are per COLUMN (review C2).
71        est.begin_column();
72        // The per-column residual, charged exactly once per row and only for a
73        // column whose builder materializes a childless array node (review C1).
74        est.add(column_slack_bytes(&shape));
75        // A LEAF column is charged in place, so the narrow path never allocates
76        // the worklist at all (`Vec::new` does not allocate until its first
77        // push). Only collection/struct cells reach `drain`.
78        est.charge_child(shape, cell);
79        est.drain();
80        if est.total == usize::MAX {
81            return usize::MAX;
82        }
83    }
84    est.total
85}

87/// One cell out of a ROW-MAJOR staging slot, addressed by canonical index.
88///
89/// A free function (not a closure at the call site) so the `.zip(..)` argument
90/// stays short enough to read — and so the indirection is named once.
91fn staged_cell(cells: &[Option<Value>], c: usize) -> Option<&Value> {
92    cells.get(c).and_then(Option::as_ref)
93}
94
95/// One cell out of a SPARSE column-major cell store, addressed by canonical
96/// index and row: `cells[c]` holds `(row index, value)` for each PRESENT cell,
97/// in ascending row order.
98///
99/// `None` means the row carries no cell for that column — a legitimate ABSENT
100/// cell, charged exactly as `estimate_arrow_row_bytes` charges a failed
101/// `values.get(name)`, never zero. The binary search is `O(log present)` and is
102/// only on the re-derivation path (`recomputed_payload`), which is the O(rows)
103/// recomputation the incremental accumulator exists to avoid in the first place —
104/// the PRODUCTION charge reads the dense staging slot in `row_bytes`.
105fn sparse_cell(cells: &[Vec<(usize, Value)>], c: usize, row: usize) -> Option<&Value> {
106    let store = cells.get(c)?;
107    let at = store.binary_search_by_key(&row, |(r, _)| *r).ok()?;
108    store.get(at).map(|(_, value)| value)
109}

111/// Per-column Arrow slot shapes resolved ONCE for a column set, so the fused
112/// push-time accounting does not re-run `column_shape` per row (issue #3552).
113///
114/// Visible ONLY inside `crate::export`: this is the seam between the estimator and
115/// `super::arrow_row_accumulator`, not a public contract and not a crate-wide one
116/// (issue #3552 review N5). `estimate_arrow_row_bytes` remains the public surface.
117pub(in crate::export) struct PreparedColumns<'a> {
118    shapes: Vec<Shape<'a>>,
119}
120
121impl<'a> PreparedColumns<'a> {
122    /// Resolve every column's Arrow slot shape once.
123    pub(in crate::export) fn new(columns: &'a [ColumnInfo]) -> Self {
124        Self {
125            shapes: columns.iter().map(column_shape).collect(),
126        }
127    }
128
129    /// Number of prepared columns.
130    pub(in crate::export) fn len(&self) -> usize {
131        self.shapes.len()
132    }
133
134    /// Width of one row whose cells are ALREADY resolved, read through a
135    /// `canonical` indirection: column `i`'s cell is `cells[canonical[i]]`.
136    ///
137    /// The indirection is what lets a value be STORED ONCE and charged for every
138    /// output column that names it. Two output columns with the same name
139    /// (`SELECT a, a`) share one canonical slot, so this charges them both —
140    /// exactly as `estimate_arrow_row_bytes` charges both, resolving the name
141    /// twice — without the value existing twice (issue #3552 review B3).
142    /// `canonical[i] == i` for every column with a unique name.
143    ///
144    /// Fails closed to `usize::MAX` on an arity mismatch or an out-of-range
145    /// canonical index. A short `canonical` would otherwise silently drop the
146    /// trailing columns' charges (`zip` truncates) and UNDER-count, the one
147    /// direction the conservatism contract forbids.
148    pub(in crate::export) fn row_bytes(
149        &self,
150        cells: &[Option<Value>],
151        canonical: &[usize],
152    ) -> usize {
153        if canonical.len() != self.shapes.len() {
154            return usize::MAX;
155        }
156        if canonical.iter().any(|&c| c >= cells.len()) {
157            return usize::MAX;
158        }
159        charge_row(
160            self.shapes
161                .iter()
162                .copied()
163                .zip(canonical.iter().map(|&c| staged_cell(cells, c))),
164        )
165    }
166
167    /// Width of row `row` of a SPARSE column-major cell store, read through the
168    /// same `canonical` indirection: column `i`'s cell is the entry for `row` in
169    /// `cells[canonical[i]]`, or absent.
170    ///
171    /// The layout the accumulator commits into, so a buffered row's width can be
172    /// re-derived from the stored cells rather than from a remembered number.
173    ///
174    /// Fails closed to `usize::MAX` on an arity mismatch or an out-of-range
175    /// canonical index, for the same reason as [`Self::row_bytes`]. It does NOT
176    /// fail closed on a row missing from a column's store: in a sparse store that
177    /// is an ABSENT cell, charged as `None` — the same charge
178    /// `estimate_arrow_row_bytes` makes. `row < rows` is the CALLER's invariant
179    /// (`recomputed_payload` iterates `0..rows`); a cell wrongly missing from the
180    /// store would make this total come out SMALLER than the running accumulator,
181    /// which is exactly what `flush_credited`'s debug assertion compares.
182    pub(in crate::export) fn row_bytes_columnar(
183        &self,
184        cells: &[Vec<(usize, Value)>],
185        canonical: &[usize],
186        row: usize,
187    ) -> usize {
188        if canonical.len() != self.shapes.len() {
189            return usize::MAX;
190        }
191        if canonical.iter().any(|&c| c >= cells.len()) {
192            return usize::MAX;
193        }
194        charge_row(
195            self.shapes
196                .iter()
197                .copied()
198                .zip(canonical.iter().map(|&c| sparse_cell(cells, c, row))),
199        )
200    }
201}
