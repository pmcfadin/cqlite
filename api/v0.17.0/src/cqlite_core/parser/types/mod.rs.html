1//! CQL type system parsing and serialization
2//!
3//! This module handles parsing and serialization of CQL data types as they
4//! appear in Cassandra SSTable format. It maps between the binary representation
5//! and the CQLite Value types.
6//!
7//! The implementation is split by responsibility:
8//! - [`primitives`] — scalar types (int/text/uuid/timestamp/decimal/…).
9//! - [`collections`] — list/set/map (legacy, v5, registry-aware, schema-driven).
10//! - [`udt`] — UDT, tuple, and frozen decoding.
11//! - [`tombstones`] — tombstone/deletion decoding.
12//!
13//! This `mod.rs` keeps the shared model ([`CqlTypeId`], type-id ⇄ value mapping,
14//! empty-value construction), the value-dispatch entry points
15//! ([`parse_cql_value`], [`parse_cql_value_raw`]), the schema-driven dispatcher,
16//! buffer-consumption validation helpers, and serialization — and re-exports the
17//! full public surface so callers using `parser::types::*` are unaffected.
18

49/// CQL type identifiers as they appear in the binary format
50#[derive(Debug, Clone, Copy, PartialEq, Eq)]
51#[repr(u8)]
52pub enum CqlTypeId {
53    Custom = 0x00,
54    Ascii = 0x01,
55    BigInt = 0x02,
56    Blob = 0x03,
57    Boolean = 0x04,
58    Counter = 0x05,
59    Decimal = 0x06,
60    Double = 0x07,
61    Float = 0x08,
62    Int = 0x09,
63    Timestamp = 0x0B,
64    Uuid = 0x0C,
65    Varchar = 0x0D,
66    Varint = 0x0E,
67    Timeuuid = 0x0F,
68    Inet = 0x10,
69    Date = 0x11,
70    Time = 0x12,
71    Smallint = 0x13,
72    Tinyint = 0x14,
73    Duration = 0x15,
74    List = 0x20,
75    Map = 0x21,
76    Set = 0x22,
77    Udt = 0x30,
78    Tuple = 0x31,
79    // Custom CQLite extension for tombstones
80    Tombstone = 0xFF,
81}

83impl TryFrom<u8> for CqlTypeId {
84    type Error = Error;
85
86    fn try_from(value: u8) -> Result<Self> {
87        match value {
88            0x00 => Ok(CqlTypeId::Custom),
89            0x01 => Ok(CqlTypeId::Ascii),
90            0x02 => Ok(CqlTypeId::BigInt),
91            0x03 => Ok(CqlTypeId::Blob),
92            0x04 => Ok(CqlTypeId::Boolean),
93            0x05 => Ok(CqlTypeId::Counter),
94            0x06 => Ok(CqlTypeId::Decimal),
95            0x07 => Ok(CqlTypeId::Double),
96            0x08 => Ok(CqlTypeId::Float),
97            0x09 => Ok(CqlTypeId::Int),
98            0x0B => Ok(CqlTypeId::Timestamp),
99            0x0C => Ok(CqlTypeId::Uuid),
100            0x0D => Ok(CqlTypeId::Varchar),
101            0x0E => Ok(CqlTypeId::Varint),
102            0x0F => Ok(CqlTypeId::Timeuuid),
103            0x10 => Ok(CqlTypeId::Inet),
104            0x11 => Ok(CqlTypeId::Date),
105            0x12 => Ok(CqlTypeId::Time),
106            0x13 => Ok(CqlTypeId::Smallint),
107            0x14 => Ok(CqlTypeId::Tinyint),
108            0x15 => Ok(CqlTypeId::Duration),
109            0x20 => Ok(CqlTypeId::List),
110            0x21 => Ok(CqlTypeId::Map),
111            0x22 => Ok(CqlTypeId::Set),
112            0x30 => Ok(CqlTypeId::Udt),
113            0x31 => Ok(CqlTypeId::Tuple),
114            0xFF => Ok(CqlTypeId::Tombstone),
115            _ => Err(Error::corruption(format!(
116                "Unknown CQL type ID: 0x{:02X}",
117                value
118            ))),
119        }
120    }
121}

128/// Parse a CQL value based on its type with enhanced collection support
129pub fn parse_cql_value(input: &[u8], type_id: CqlTypeId) -> IResult<&[u8], Value> {
130    match type_id {
131        CqlTypeId::Boolean => parse_boolean(input),
132        CqlTypeId::Tinyint => parse_tinyint(input),
133        CqlTypeId::Smallint => parse_smallint(input),
134        CqlTypeId::Int => parse_int(input),
135        CqlTypeId::BigInt => parse_bigint(input),
136        CqlTypeId::Counter => parse_counter(input),
137        CqlTypeId::Float => parse_float(input),
138        CqlTypeId::Double => parse_double(input),
139        CqlTypeId::Ascii | CqlTypeId::Varchar => {
140            // In the SSTable cell format the caller has already extracted exactly
141            // the bytes belonging to this cell (length-framing is done at the cell
142            // level before parse_cql_value is invoked).  The entire `input` slice
143            // IS the text value — no additional length prefix to strip.
144            //
145            // For Ascii, Cassandra's AsciiSerializer requires every byte to be in
146            // the 0x00-0x7F range.  For Varchar/Text any valid UTF-8 is accepted.
147            // We accept both under the same validation path here; a stricter
148            // Ascii-only byte check can be added when needed.
149            #[cfg(feature = "legacy-heuristics")]
150            {
151                // Legacy path: try 4-byte big-endian length prefix, then null-terminated,
152                // then raw UTF-8 (kept for pre-5.0 compatibility only).
153                if input.len() >= 4 {
154                    let length =
155                        u32::from_be_bytes([input[0], input[1], input[2], input[3]]) as usize;
156                    if input.len() >= 4 + length {
157                        let text_bytes = &input[4..4 + length];
158                        if let Ok(text) = String::from_utf8(text_bytes.to_vec()) {
159                            return Ok((&input[4 + length..], Value::Text(text.into())));
160                        }
161                    }
162                }
163                if let Some(null_pos) = input.iter().position(|&b| b == 0) {
164                    if let Ok(text) = String::from_utf8(input[..null_pos].to_vec()) {
165                        return Ok((&input[null_pos + 1..], Value::Text(text.into())));
166                    }
167                }
168                if let Ok(text) = String::from_utf8(input.to_vec()) {
169                    return Ok((&[], Value::Text(text.into())));
170                }
171                parse_text(input)
172            }
173            #[cfg(not(feature = "legacy-heuristics"))]
174            {
175                // Deterministic path: treat the entire input as the UTF-8 text value.
176                let text = String::from_utf8(input.to_vec()).map_err(|_| {
177                    nom::Err::Error(nom::error::Error::new(input, nom::error::ErrorKind::Verify))
178                })?;
179                Ok((&[], Value::Text(text.into())))
180            }
181        }
182        CqlTypeId::Blob => {
183            // The caller has already extracted exactly the bytes belonging to
184            // this cell (length-framing is done at the cell level before
185            // parse_cql_value is invoked), so the entire `input` slice IS the
186            // blob value verbatim — mirroring the sibling Ascii/Varchar arm,
187            // parse_cql_value_raw and parse_blob_value. Framing is the caller's
188            // responsibility via this function's contract; the decoder never
189            // sniffs byte patterns to infer a length (no-heuristics mandate,
190            // issues #28 / #1630). The genuinely VInt-framed decode lives in
191            // parse_blob (used by parse_cql_value_with_schema and the write
192            // side's tagged serialization), not here.
193            Ok((&[], Value::blob(input.to_vec())))
194        }
195        CqlTypeId::Uuid | CqlTypeId::Timeuuid => parse_uuid(input),
196        CqlTypeId::Timestamp => parse_timestamp(input),
197        CqlTypeId::Date => parse_date(input),
198        CqlTypeId::Time => parse_time(input),
199        CqlTypeId::Varint => parse_varint(input),
200        CqlTypeId::Decimal => parse_decimal(input),
201        CqlTypeId::Duration => parse_duration(input),
202        CqlTypeId::Inet => parse_inet(input),
203        CqlTypeId::List => parse_list_enhanced(input),
204        CqlTypeId::Set => parse_set_enhanced(input),
205        CqlTypeId::Map => parse_map_enhanced(input),
206        CqlTypeId::Udt => parse_udt_enhanced(input),
207        CqlTypeId::Tuple => parse_tuple(input),
208        CqlTypeId::Tombstone => parse_tombstone(input),
209        CqlTypeId::Custom => {
210            // Custom types require additional metadata, return as blob for now
211            parse_blob(input)
212        }
213    }
214}

216/// Parse a CQL value from raw data (without length prefix)
217/// This is used for collection elements where length is already parsed
218pub fn parse_cql_value_raw(input: &[u8], type_id: CqlTypeId) -> IResult<&[u8], Value> {
219    match type_id {
220        CqlTypeId::Boolean => parse_boolean(input),
221        CqlTypeId::Tinyint => parse_tinyint(input),
222        CqlTypeId::Smallint => parse_smallint(input),
223        CqlTypeId::Int => parse_int(input),
224        CqlTypeId::BigInt => parse_bigint(input),
225        CqlTypeId::Counter => parse_counter(input),
226        CqlTypeId::Float => parse_float(input),
227        CqlTypeId::Double => parse_double(input),
228        CqlTypeId::Ascii | CqlTypeId::Varchar => {
229            // For map/collection contexts, the input is already length-prefixed at the collection level
230            // So we can treat all input as the text content directly
231            let text = String::from_utf8(input.to_vec()).map_err(|_| {
232                nom::Err::Error(nom::error::Error::new(input, nom::error::ErrorKind::Verify))
233            })?;
234            Ok((&[], Value::Text(text.into())))
235        }
236        CqlTypeId::Blob => {
237            // For blob, use all input as blob data
238            Ok((&[], Value::blob(input.to_vec())))
239        }
240        CqlTypeId::Uuid | CqlTypeId::Timeuuid => parse_uuid(input),
241        CqlTypeId::Timestamp => parse_timestamp(input),
242        CqlTypeId::Date => parse_date(input),
243        CqlTypeId::Time => parse_time(input),
244        CqlTypeId::Varint => parse_varint(input),
245        CqlTypeId::Decimal => parse_decimal(input),
246        CqlTypeId::Duration => parse_duration(input),
247        CqlTypeId::Inet => parse_inet(input),
248        // Collections and complex types should not be called with raw parsing
249        CqlTypeId::List | CqlTypeId::Set | CqlTypeId::Map | CqlTypeId::Udt | CqlTypeId::Tuple => {
250            parse_cql_value(input, type_id) // Fallback to normal parsing
251        }
252        CqlTypeId::Tombstone => parse_tombstone(input),
253        CqlTypeId::Custom => {
254            // Custom types require additional metadata, return as blob for now
255            Ok((&[], Value::blob(input.to_vec())))
256        }
257    }
258}

260/// Create empty value for a CQL type
261pub(super) fn create_empty_value_for_cql_type(cql_type: &CqlType) -> Result<Value> {
262    match cql_type {
263        CqlType::Boolean => Ok(Value::Boolean(false)),
264        CqlType::TinyInt => Ok(Value::TinyInt(0)),
265        CqlType::SmallInt => Ok(Value::SmallInt(0)),
266        CqlType::Int => Ok(Value::Integer(0)),
267        CqlType::BigInt => Ok(Value::BigInt(0)),
268        CqlType::Float => Ok(Value::Float32(0.0)),
269        CqlType::Double => Ok(Value::Float(0.0)),
270        CqlType::Text | CqlType::Ascii | CqlType::Varchar => Ok(Value::text(String::new())),
271        CqlType::Blob => Ok(Value::blob(Vec::new())),
272        CqlType::Uuid | CqlType::TimeUuid => Ok(Value::Uuid([0; 16])),
273        CqlType::Timestamp => Ok(Value::Timestamp(0)),
274        CqlType::Date => Ok(Value::Timestamp(0)),
275        CqlType::Time => Ok(Value::Timestamp(0)),
276        CqlType::List(_) => Ok(Value::List(Vec::new())),
277        CqlType::Set(_) => Ok(Value::Set(Vec::new())),
278        CqlType::Map(_, _) => Ok(Value::Map(Vec::new())),
279        CqlType::Tuple(_) => Ok(Value::Tuple(Vec::new())),
280        CqlType::Udt(name, _) => Ok(Value::Udt(Box::new(UdtValue::new(
281            name.clone(),
282            "unknown".to_string(),
283        )))),
284        CqlType::Frozen(inner) => create_empty_value_for_cql_type(inner),
285        // #4114 (roborev job 109): there is NO empty vector to construct. A
286        // zero-length vector value is an ERROR in Cassandra
287        // (`VectorType.java:365-368`, "Invalid empty vector value"), so this arm must
288        // not fall through to `Value::Null` below — that would turn an invalid value
289        // into a legal-looking one. The refusal comes from the ONE framing rule so
290        // the message matches every other vector site. (A genuinely NULL field is
291        // decided by the outer framing — `length == -1` — and never reaches here; the
292        // collection map path's `length < 0` also lands here, and refusing is the
293        // fail-closed reading: Cassandra permits neither a null nor an empty map
294        // key/value.)
295        CqlType::Vector(element, dimension) => {
296            crate::schema::vector_type::vector_value::decode_framed_float_vector(
297                &[],
298                element,
299                *dimension,
300                "empty framed vector value",
301            )
302        }
303        _ => Ok(Value::Null),
304    }
305}

307/// Convert CqlType to CqlTypeId for parsing
308pub(super) fn cql_type_to_type_id(cql_type: &CqlType) -> CqlTypeId {
309    match cql_type {
310        CqlType::Boolean => CqlTypeId::Boolean,
311        CqlType::TinyInt => CqlTypeId::Tinyint,
312        CqlType::SmallInt => CqlTypeId::Smallint,
313        CqlType::Int => CqlTypeId::Int,
314        CqlType::BigInt => CqlTypeId::BigInt,
315        CqlType::Counter => CqlTypeId::Counter,
316        CqlType::Float => CqlTypeId::Float,
317        CqlType::Double => CqlTypeId::Double,
318        CqlType::Text | CqlType::Ascii | CqlType::Varchar => CqlTypeId::Varchar,
319        CqlType::Blob => CqlTypeId::Blob,
320        CqlType::Uuid => CqlTypeId::Uuid,
321        CqlType::TimeUuid => CqlTypeId::Timeuuid,
322        CqlType::Timestamp => CqlTypeId::Timestamp,
323        CqlType::Date => CqlTypeId::Date,
324        CqlType::Time => CqlTypeId::Time,
325        CqlType::Decimal => CqlTypeId::Decimal,
326        CqlType::Duration => CqlTypeId::Duration,
327        CqlType::Inet => CqlTypeId::Inet,
328        CqlType::Varint => CqlTypeId::Varint,
329        CqlType::List(_) => CqlTypeId::List,
330        CqlType::Set(_) => CqlTypeId::Set,
331        CqlType::Map(_, _) => CqlTypeId::Map,
332        CqlType::Tuple(_) => CqlTypeId::Tuple,
333        CqlType::Udt(_, _) => CqlTypeId::Udt,
334        CqlType::Frozen(_) => CqlTypeId::Blob, // Fallback only; callers should handle Frozen explicitly
335        // #4114: a vector has NO native protocol type id (Cassandra carries it as a
336        // custom type), so this conversion is LOSSY — it discards the element type
337        // and the dimension, and `CqlTypeId::Custom` decodes as a vint-framed blob.
338        // Every caller must therefore intercept `CqlType::Vector` BEFORE reaching a
339        // type id, exactly like the `Frozen` arm above. All three do:
340        // `parse_cql_value_for_type`, `parse_cql_value_for_type_with_registry` (the
341        // one that did NOT, roborev job 109) and `parse_cql_value_with_schema`.
342        CqlType::Vector(_, _) => CqlTypeId::Custom,
343        CqlType::Custom(_) => CqlTypeId::Blob, // Custom types as blob
344    }
345}

370/// Serialize a CQL value to bytes
371pub fn serialize_cql_value(value: &Value) -> Result<Vec<u8>> {
372    let mut result = Vec::new();
373
374    match value {
375        Value::Null => {
376            // Null values are represented with a special length marker
377            result.extend_from_slice(&encode_vint(-1));
378        }
379        Value::Boolean(b) => {
380            result.push(CqlTypeId::Boolean as u8);
381            result.push(if *b { 1 } else { 0 });
382        }
383        Value::Integer(i) => {
384            result.push(CqlTypeId::Int as u8);
385            result.extend_from_slice(&i.to_be_bytes());
386        }
387        Value::BigInt(i) => {
388            result.push(CqlTypeId::BigInt as u8);
389            result.extend_from_slice(&i.to_be_bytes());
390        }
391        Value::Counter(c) => {
392            result.push(CqlTypeId::Counter as u8);
393            result.extend_from_slice(&c.to_be_bytes());
394        }
395        Value::Float(f) => {
396            result.push(CqlTypeId::Double as u8);
397            result.extend_from_slice(&f.to_be_bytes());
398        }
399        Value::Text(s) => {
400            result.push(CqlTypeId::Varchar as u8);
401            result.extend_from_slice(&encode_vint(s.len() as i64));
402            result.extend_from_slice(s.as_ref());
403        }
404        Value::Blob(b) => {
405            result.push(CqlTypeId::Blob as u8);
406            result.extend_from_slice(&encode_vint(b.len() as i64));
407            result.extend_from_slice(b);
408        }
409        Value::Timestamp(ts) => {
410            result.push(CqlTypeId::Timestamp as u8);
411            let millis = ts / 1000; // Convert microseconds to milliseconds
412            result.extend_from_slice(&millis.to_be_bytes());
413        }
414        Value::Date(days) => {
415            result.push(CqlTypeId::Date as u8);
416            result.extend_from_slice(&days.to_be_bytes());
417        }
418        Value::Time(nanos) => {
419            result.push(CqlTypeId::Time as u8);
420            result.extend_from_slice(&nanos.to_be_bytes());
421        }
422        Value::Uuid(uuid) => {
423            result.push(CqlTypeId::Uuid as u8);
424            result.extend_from_slice(uuid);
425        }
426        // EMPTY-BUFFER SENTINEL (issue #3805): REFUSED, never encoded.
427        // See `refuse_empty_sentinel` for why this format has no
428        // representation that reads back as the sentinel.
429        Value::Empty(ty) => return Err(refuse_empty_sentinel(*ty)),
430        Value::Json(json) => {
431            // Store JSON as text
432            let json_str = json.to_string();
433            result.push(CqlTypeId::Varchar as u8);
434            result.extend_from_slice(&encode_vint(json_str.len() as i64));
435            result.extend_from_slice(json_str.as_bytes());
436        }
437        Value::List(list) => {
438            result.push(CqlTypeId::List as u8);
439            result.extend_from_slice(&encode_vint(list.len() as i64));
440
441            // For simplicity, assume all elements are the same type
442            if let Some(first) = list.first() {
443                let element_type = map_value_to_cql_type(first);
444                result.push(element_type as u8);
445
446                for element in list {
447                    if let Value::Null = element {
448                        // Null element: length = -1
449                        result.extend_from_slice(&encode_vint(-1));
450                    } else {
451                        let element_data = serialize_value_without_type_prefix(element)?;
452                        result.extend_from_slice(&encode_vint(element_data.len() as i64));
453                        result.extend_from_slice(&element_data);
454                    }
455                }
456            }
457        }
458        Value::Map(map) => {
459            result.push(CqlTypeId::Map as u8);
460            result.extend_from_slice(&encode_vint(map.len() as i64));
461
462            // For simplicity, assume all keys and values are the same type
463            if let Some((first_key, first_value)) = map.first() {
464                let key_type = map_value_to_cql_type(first_key);
465                let value_type = map_value_to_cql_type(first_value);
466                result.push(key_type as u8);
467                result.push(value_type as u8);
468
469                for (key, value) in map {
470                    // Serialize key with length prefix
471                    if let Value::Null = key {
472                        result.extend_from_slice(&encode_vint(-1));
473                    } else {
474                        let key_data = serialize_value_without_type_prefix(key)?;
475                        result.extend_from_slice(&encode_vint(key_data.len() as i64));
476                        result.extend_from_slice(&key_data);
477                    }
478
479                    // Serialize value with length prefix
480                    if let Value::Null = value {
481                        result.extend_from_slice(&encode_vint(-1));
482                    } else {
483                        let value_data = serialize_value_without_type_prefix(value)?;
484                        result.extend_from_slice(&encode_vint(value_data.len() as i64));
485                        result.extend_from_slice(&value_data);
486                    }
487                }
488            }
489        }
490        Value::TinyInt(i) => {
491            result.push(CqlTypeId::Tinyint as u8);
492            result.push(*i as u8);
493        }
494        Value::Inet(bytes) => {
495            result.push(CqlTypeId::Inet as u8);
496            result.extend_from_slice(&encode_vint(bytes.len() as i64));
497            result.extend_from_slice(bytes);
498        }
499        Value::SmallInt(i) => {
500            result.push(CqlTypeId::Smallint as u8);
501            result.extend_from_slice(&i.to_be_bytes());
502        }
503        Value::Float32(f) => {
504            result.push(CqlTypeId::Float as u8);
505            result.extend_from_slice(&f.to_be_bytes());
506        }
507        Value::Set(set) => {
508            result.push(CqlTypeId::Set as u8);
509            result.extend_from_slice(&encode_vint(set.len() as i64));
510
511            if let Some(first) = set.first() {
512                let element_type = map_value_to_cql_type(first);
513                result.push(element_type as u8);
514
515                for element in set {
516                    if let Value::Null = element {
517                        // Null element: length = -1
518                        result.extend_from_slice(&encode_vint(-1));
519                    } else {
520                        let element_data = serialize_value_without_type_prefix(element)?;
521                        result.extend_from_slice(&encode_vint(element_data.len() as i64));
522                        result.extend_from_slice(&element_data);
523                    }
524                }
525            }
526        }
527        Value::Tuple(tuple) => {
528            result.push(CqlTypeId::Tuple as u8);
529            result.extend_from_slice(&encode_vint(tuple.len() as i64));
530
531            // Serialize type information for each field
532            for element in tuple {
533                let element_type = map_value_to_cql_type(element);
534                result.push(element_type as u8);
535            }
536
537            // Serialize field values with proper length prefixes
538            for element in tuple {
539                if let Value::Null = element {
540                    // Null field: length = -1
541                    result.extend_from_slice(&(-1i32).to_be_bytes());
542                } else {
543                    let element_data = serialize_value_without_type_prefix(element)?;
544                    result.extend_from_slice(&(element_data.len() as i32).to_be_bytes());
545                    result.extend_from_slice(&element_data);
546                }
547            }
548        }
549        Value::Udt(udt) => {
550            result.push(CqlTypeId::Udt as u8);
551
552            // Serialize type name
553            result.extend_from_slice(&encode_vint(udt.type_name.len() as i64));
554            result.extend_from_slice(udt.type_name.as_bytes());
555
556            // Serialize field count
557            result.extend_from_slice(&encode_vint(udt.fields.len() as i64));
558
559            // Serialize field definitions
560            for field in &udt.fields {
561                result.extend_from_slice(&encode_vint(field.name.len() as i64));
562                result.extend_from_slice(field.name.as_bytes());
563
564                // Serialize field type (inferred from value or use blob as fallback)
565                let field_type = match &field.value {
566                    Some(value) => map_value_to_cql_type(value),
567                    None => CqlTypeId::Blob, // Null field, use generic type
568                };
569                result.push(field_type as u8);
570            }
571
572            // Serialize field values
573            for field in &udt.fields {
574                match &field.value {
575                    None => {
576                        // Null field: length = -1
577                        result.extend_from_slice(&(-1i32).to_be_bytes());
578                    }
579                    Some(value) => {
580                        let field_data = serialize_value_without_type_prefix(value)?;
581                        result.extend_from_slice(&(field_data.len() as i32).to_be_bytes());
582                        result.extend_from_slice(&field_data);
583                    }
584                }
585            }
586        }
587        Value::Frozen(boxed_value) => {
588            // For frozen values, just serialize the inner value
589            let inner_bytes = serialize_cql_value(boxed_value)?;
590            result.extend_from_slice(&inner_bytes);
591        }
592        Value::Tombstone(info) => {
593            result.push(CqlTypeId::Tombstone as u8);
594            result.extend_from_slice(&info.deletion_time.to_be_bytes());
595
596            let tombstone_type_byte = match info.tombstone_type {
597                TombstoneType::RowTombstone => 0u8,
598                TombstoneType::CellTombstone => 1u8,
599                TombstoneType::RangeTombstone => 2u8,
600                TombstoneType::TtlExpiration => 3u8,
601                // PartitionTombstone is a delta-scan concept that has no encoding
602                // in the legacy binary format.  Silently mapping it to byte 0
603                // (RowTombstone) would cause a silent lossy round-trip, violating
604                // the no-silent-corruption mandate.  Fail loudly instead.
605                TombstoneType::PartitionTombstone => {
606                    return Err(Error::invalid_operation(
607                        "PartitionTombstone cannot be serialized via the legacy CQL value \
608                         binary format; it is a delta-scan concept only",
609                    ));
610                }
611            };
612            result.push(tombstone_type_byte);
613
614            // Add TTL if present
615            if let Some(ttl) = info.ttl {
616                result.extend_from_slice(&ttl.to_be_bytes());
617            }
618
619            // Add range information for range tombstones
620            if info.tombstone_type == TombstoneType::RangeTombstone {
621                if let (Some(start), Some(end)) = (&info.range_start, &info.range_end) {
622                    result.push(1u8); // Has range marker
623                    result.extend_from_slice(&encode_vint(start.len() as i64));
624                    result.extend_from_slice(start.as_bytes());
625                    result.extend_from_slice(&encode_vint(end.len() as i64));
626                    result.extend_from_slice(end.as_bytes());
627                } else {
628                    result.push(0u8); // No range marker
629                }
630            }
631        }
632        Value::Varint(data) => {
633            result.push(CqlTypeId::Varint as u8);
634            result.extend_from_slice(&encode_vint(data.len() as i64));
635            result.extend_from_slice(data);
636        }
637        Value::Decimal { scale, unscaled } => {
638            result.push(CqlTypeId::Decimal as u8);
639            result.extend_from_slice(&scale.to_be_bytes());
640            result.extend_from_slice(&encode_vint(unscaled.len() as i64));
641            result.extend_from_slice(unscaled);
642        }
643        Value::Duration {
644            months,
645            days,
646            nanos,
647        } => {
648            result.push(CqlTypeId::Duration as u8);
649            result.extend_from_slice(&months.to_be_bytes());
650            result.extend_from_slice(&days.to_be_bytes());
651            result.extend_from_slice(&nanos.to_be_bytes());
652        }
653    }
654
655    Ok(result)
656}

658/// Serialize a CQL value without the type prefix byte
659fn serialize_value_without_type_prefix(value: &Value) -> Result<Vec<u8>> {
660    match value {
661        Value::Null => Ok(vec![]), // Null should be handled at higher level
662        Value::Boolean(b) => Ok(vec![if *b { 1 } else { 0 }]),
663        Value::Integer(i) => Ok(i.to_be_bytes().to_vec()),
664        Value::BigInt(i) => Ok(i.to_be_bytes().to_vec()),
665        Value::Float(f) => Ok(f.to_be_bytes().to_vec()),
666        Value::Text(s) => {
667            // For raw serialization without type prefix, just include the string bytes
668            // The length will be handled by the map format itself
669            Ok(s.to_vec())
670        }
671        Value::Blob(b) => {
672            // For raw serialization without type prefix, just include the blob bytes
673            // The length will be handled by the map format itself
674            Ok(b.to_vec())
675        }
676        Value::Timestamp(ts) => {
677            let millis = ts / 1000; // Convert microseconds to milliseconds
678            Ok(millis.to_be_bytes().to_vec())
679        }
680        Value::Date(days) => Ok(days.to_be_bytes().to_vec()),
681        Value::Time(nanos) => Ok(nanos.to_be_bytes().to_vec()),
682        Value::Uuid(uuid) => Ok(uuid.to_vec()),
683        Value::TinyInt(i) => Ok(vec![*i as u8]),
684        Value::Inet(bytes) => Ok(bytes.to_vec()),
685        Value::SmallInt(i) => Ok(i.to_be_bytes().to_vec()),
686        Value::Float32(f) => Ok(f.to_be_bytes().to_vec()),
687        // EMPTY-BUFFER SENTINEL (issue #3805): REFUSED, never encoded. A
688        // collection element written as `len=0` is read back through
689        // `parse_cql_value_raw`, which (measured, all 12 admitted families)
690        // rejects a zero-length payload. See `refuse_empty_sentinel`.
691        Value::Empty(ty) => Err(refuse_empty_sentinel(*ty)),
692        // For complex types, fall back to full serialization and strip type byte
693        _ => {
694            let full_bytes = serialize_cql_value(value)?;
695            Ok(full_bytes[1..].to_vec()) // Skip the type byte
696        }
697    }
698}

700/// The error both serializers in this module return for the empty-buffer
701/// sentinel (issue #3805).
702///
703/// This format's wire shape is `[type byte][payload]`, and `[type byte]` with
704/// no payload does NOT read back as `Value::Empty(_)`. MEASURED against this
705/// module's own readers, over all 12 families the tag admits: BOTH
706/// [`parse_cql_value`] and [`parse_cql_value_raw`] return `Eof` for a
707/// zero-length payload — `inet` and `decimal`/`varint` included, since their
708/// arms here read a length or a scale first. So a bare type byte does not
709/// decode to a different `Value`; it does not decode at all, and an encoder
710/// emitting it would be writing bytes its own reader rejects. That is why this
711/// is an `Err` and not an encoding, and it is pinned rather than asserted by
712/// `cqlite-core/tests/issue_3805_empty_value_sentinel.rs`
713/// (`no_tagged_form_of_the_sentinel_reads_back_as_the_sentinel`).
714///
715/// Inventing a decodable TAGGED form for the sentinel instead would mean
716/// inventing an encoding no Cassandra authority defines, which the
717/// no-heuristics mandate forbids (issue #28); the sentinel's write-side
718/// representation is a declared fail-closed residual, issue #4072.
719///
720/// # The surface that CAN write it — CORRECTED (roborev job 452, then #4106)
721/// It is a multicell COLLECTION's CELL PATH in the SSTable writer — a map's KEY
722/// (#3805) or a set's ELEMENT (#4106), both in `data_writer::cell_path` — and
723/// ONLY that. An earlier revision of this comment named the type-aware
724/// writer (`crate::storage::serialization::types`) on the ground that a declared
725/// column type "supplies the framing this format lacks" — which was WRONG, and
726/// citing it made this rationale point at a site that now refuses. A declared
727/// type says only that an empty buffer would be LEGAL for that type; the framing
728/// question is separate and is what the cell path supplies (an unsigned-VInt
729/// length in the enclosing collection, `db/marshal/CollectionType.java:361-382`,
730/// so a zero-length path is expressible and MEANS an empty key). The type-aware
731/// writer has the type and NOT the framing, so it refuses too — see
732/// `TypeSerializer`'s `refuse_empty_sentinel_cell_value`.
733fn refuse_empty_sentinel(ty: crate::types::EmptyValueType) -> Error {
734    Error::invalid_operation(format!(
735        "Value::Empty({}) cannot be serialized via the legacy tagged CQL value \
736         binary format: no byte string in this format reads back as the \
737         empty-buffer sentinel (issue #4072), and inventing one has no \
738         Cassandra oracle (issue #28)",
739        ty.cql_name()
740    ))
741}

743fn map_value_to_cql_type(value: &Value) -> CqlTypeId {
744    match value {
745        Value::Null => CqlTypeId::Blob, // Generic fallback
746        Value::Boolean(_) => CqlTypeId::Boolean,
747        Value::Integer(_) => CqlTypeId::Int,
748        Value::BigInt(_) => CqlTypeId::BigInt,
749        Value::Counter(_) => CqlTypeId::Counter,
750        Value::Float(_) => CqlTypeId::Double,
751        Value::Text(_) => CqlTypeId::Varchar,
752        Value::Blob(_) => CqlTypeId::Blob,
753        Value::Timestamp(_) => CqlTypeId::Timestamp,
754        Value::Date(_) => CqlTypeId::Date,
755        Value::Time(_) => CqlTypeId::Time,
756        Value::Uuid(_) => CqlTypeId::Uuid,
757        Value::Json(_) => CqlTypeId::Varchar,
758        Value::TinyInt(_) => CqlTypeId::Tinyint,
759        Value::Inet(_) => CqlTypeId::Inet,
760        Value::SmallInt(_) => CqlTypeId::Smallint,
761        Value::Float32(_) => CqlTypeId::Float,
762        Value::List(_) => CqlTypeId::List,
763        Value::Set(_) => CqlTypeId::Set,
764        Value::Map(_) => CqlTypeId::Map,
765        Value::Tuple(_) => CqlTypeId::Tuple,
766        Value::Udt(_) => CqlTypeId::Udt,
767        Value::Frozen(_) => CqlTypeId::Blob, // Frozen is a wrapper, use blob as fallback
768        Value::Tombstone(_) => CqlTypeId::Tombstone,
769        Value::Varint(_) => CqlTypeId::Varint,
770        Value::Decimal { .. } => CqlTypeId::Decimal,
771        Value::Duration { .. } => CqlTypeId::Duration,
772        // The sentinel reports its DECLARED type, not a synthetic one
773        // (issue #3805). This is a TYPE-ID lookup, not an encoding: the value
774        // itself is refused by both serializers above.
775        Value::Empty(ty) => map_empty_value_type_to_cql_type(*ty),
776    }
777}

796        E::Inet => CqlTypeId::Inet,

806/// Assert that all input bytes have been consumed during parsing
807///
808/// This helper enforces complete buffer consumption to prevent silent data
809/// truncation in collection and UDT parsing. Per Issue #61 acceptance criteria,
810/// all parsers must validate full consumption.
811///
812/// # Arguments
813/// * `remaining` - The remaining bytes after parsing
814/// * `context` - Description of parsing context for error messages
815///
816/// # Returns
817/// * `Ok(())` if buffer is fully consumed
818/// * `Err` if bytes remain unconsumed
819#[inline]
820#[allow(dead_code)] // Will be used in collection parsing functions
821pub(crate) fn assert_full_buffer_consumption(remaining: &[u8], context: &str) -> Result<()> {
822    if !remaining.is_empty() {
823        return Err(Error::corruption(format!(
824            "Buffer not fully consumed in {}: {} bytes remaining",
825            context,
826            remaining.len()
827        )));
828    }
829    Ok(())
830}
831
832/// Validate that a parsed element consumed its entire allocated buffer
833///
834/// Used within collection parsing to ensure each element's buffer is fully
835/// consumed before moving to the next element.
836#[inline]
837#[allow(dead_code)] // Will be used in collection parsing functions
838pub(crate) fn validate_element_consumption<'a>(
839    _input: &'a [u8],
840    remaining: &'a [u8],
841    element_index: usize,
842    collection_type: &str,
843) -> Result<()> {
844    if !remaining.is_empty() {
845        return Err(Error::corruption(format!(
846            "{} element {} did not consume full buffer: {} bytes remaining",
847            collection_type,
848            element_index,
849            remaining.len()
850        )));
851    }
852    Ok(())
853}

859/// Parse CQL value using schema (no heuristics)
860///
861/// Schema-driven parser that never falls back to heuristics. Used by
862/// schema-aware collection parsers to ensure deterministic decoding.
863pub(super) fn parse_cql_value_with_schema<'a>(
864    input: &'a [u8],
865    schema: &CqlType,
866) -> IResult<&'a [u8], Value> {
867    match schema {
868        CqlType::Boolean => parse_boolean(input),
869        CqlType::TinyInt => parse_tinyint(input),
870        CqlType::SmallInt => parse_smallint(input),
871        CqlType::Int => parse_int(input),
872        CqlType::BigInt => parse_bigint(input),
873        CqlType::Counter => parse_counter(input),
874        CqlType::Float => parse_float(input),
875        CqlType::Double => parse_double(input),
876        CqlType::Text | CqlType::Ascii | CqlType::Varchar => parse_text(input),
877        CqlType::Blob => parse_blob(input),
878        CqlType::Uuid | CqlType::TimeUuid => parse_uuid(input),
879        CqlType::Timestamp => parse_timestamp(input),
880        CqlType::Date => parse_date(input),
881        CqlType::Time => parse_time(input),
882        CqlType::Duration => parse_duration(input),
883        CqlType::Inet => parse_inet(input),
884        CqlType::Decimal => parse_decimal(input),
885        CqlType::Varint => parse_varint(input),
886        CqlType::List(element_type) => parse_list_with_schema(input, element_type),
887        CqlType::Set(element_type) => {
888            let (remaining, Value::List(elements)) = parse_list_with_schema(input, element_type)?
889            else {
890                return Err(nom::Err::Error(nom::error::Error::new(
891                    input,
892                    nom::error::ErrorKind::Verify,
893                )));
894            };
895            Ok((remaining, Value::Set(elements)))
896        }
897        CqlType::Map(key_type, value_type) => parse_map_with_schema(input, key_type, value_type),
898        CqlType::Tuple(_) => parse_tuple(input),
899        CqlType::Udt(_, _) => parse_udt(input),
900        // #4114: `vector<float, n>` is `4*n` raw big-endian binary32 bytes with NO
901        // length prefix and no per-element framing (`VectorType.java:94-101`,
902        // `:445-460`), so it consumes a FIXED width off the front of `input` — it is
903        // NOT a collection and must not reach `parse_blob`, which would read the
904        // first float's high byte as a vint length (the #4114 defect).
905        CqlType::Vector(element_type, dimension) => {
906            crate::schema::vector_type::vector_value::parse_float_vector_nom(
907                input,
908                element_type,
909                *dimension,
910            )
911        }
912        CqlType::Frozen(inner) => parse_cql_value_with_schema(input, inner),
913        CqlType::Custom(_) => {
914            // Custom types require additional metadata, parse as blob
915            parse_blob(input)
916        }
917    }
918}
