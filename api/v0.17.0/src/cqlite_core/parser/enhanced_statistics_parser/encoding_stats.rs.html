1//! EncodingStats decoding from the SERIALIZATION_HEADER component of Statistics.db.
2//!
3//! EncodingStats are three epoch-relative unsigned VInt deltas (minTimestamp,
4//! minLocalDeletionTime, minTTL) that prefix the SerializationHeader. They are
5//! the baseline values V5CompressedLegacy needs for delta-coded timestamps/TTLs.
6//! This module decodes those three fields and then defers to
7//! [`super::serialization_header`] for the schema that follows.

21/// Epoch constants matching Cassandra's EncodingStats.java (EncodingStats.Serializer)
22/// Used for delta-encoding/decoding EncodingStats fields in Statistics.db SERIALIZATION_HEADER.
23/// Cassandra serializes: writeUnsignedVInt(value - EPOCH)
24/// Cassandra deserializes: readUnsignedVInt() + EPOCH
25const TIMESTAMP_EPOCH: i64 = 1_442_880_000_000_000; // Sept 22, 2015 00:00:00 UTC in microseconds
26const DELETION_TIME_EPOCH: i64 = 1_442_880_000; // Sept 22, 2015 00:00:00 UTC in seconds
27                                                // TTL epoch is 0 in Cassandra, but kept for consistency with the delta-encoding pattern
28const TTL_EPOCH: i64 = 0;

30/// Parse minimal EncodingStats section from nb-format Statistics.db
31///
32/// Returns: (min_timestamp, min_deletion_time, min_ttl, partition_keys, clustering_keys, columns)
33///
34/// # Arguments
35/// * `input` - The data starting at the STATS component
36/// * `full_input` - The complete Statistics.db content (needed for TOC-based HEADER lookup)
37/// * `header_offset` - Optional offset to SerializationHeader from TOC (Issue #216)
38/// * `gates` - Optional VersionGates for VG3 version-sensitive decoding decisions.
39///   Pass `None` from standalone tools/tests to use nb-compatible defaults.
40///
41/// # The error is TYPED, and that is the point (#4104, roborev job 119)
42///
43/// A semantic refusal (`frozen<scalar>`) and a structural failure (truncated,
44/// mispositioned) are two different answers, and a `nom::Err` can carry neither
45/// the distinction nor the refusal's message. Both used to leave here as the same
46/// bare `ErrorKind::Verify`, so a user opening an SSTable whose header spells
47/// `FrozenType(Int32Type)` was told `Corruption: … code: Verify` — indistinguishable
48/// from a garbled file, while the precise column-naming, citation-carrying message
49/// the gate had already built survived only if `RUST_LOG` happened to be on.
50/// [`HeaderSchemaError::Refused`] carries that message to the caller, which turns it
51/// into the user-visible error.
52pub(super) fn parse_minimal_encoding_stats<'a>(
53    input: &'a [u8],
54    full_input: &'a [u8],
55    header_offset: Option<usize>,
56    gates: Option<&VersionGates>,
57) -> Result<(&'a [u8], EncodingStatsResult), HeaderSchemaError<'a>> {
58    // The SERIALIZATION_HEADER component (type 3) starts with EncodingStats:
59    //   [vuint minTimestamp_delta] [vuint minLocalDeletionTime_delta] [vuint minTTL_delta]
60    // These are unsigned VInt deltas from epoch constants (see EncodingStats.Serializer).
61    // Use the TOC-based offset to read from the correct location.
62
63    let Some(offset) = header_offset else {
64        tracing::debug!("No HEADER TOC offset, using fallback EncodingStats parsing");
65        return parse_encoding_stats_fallback(input, gates);
66    };
67
68    if offset >= full_input.len() {
69        tracing::warn!(
70            "TOC offset 0x{:x} exceeds input length {}, using fallback",
71            offset,
72            full_input.len()
73        );
74        return parse_encoding_stats_fallback(input, gates);
75    }
76
77    let header_data = &full_input[offset..];
78    tracing::debug!(
79        "Parsing EncodingStats + SerializationHeader at TOC offset 0x{:x} ({} bytes available)",
80        offset,
81        header_data.len()
82    );
83
84    // Parse EncodingStats (3 unsigned VInts at start of SERIALIZATION_HEADER)
85    let (rest, (min_timestamp, min_deletion_time, min_ttl)) =
86        parse_encoding_stats_vuints(header_data, gates)?;
87
88    tracing::debug!(
89        "EncodingStats from HEADER: min_timestamp={}, min_deletion_time={}, min_ttl={:?}",
90        min_timestamp,
91        min_deletion_time,
92        min_ttl
93    );
94
95    // Parse the rest of the SerializationHeader (schema info).
96    //
97    // The two failure kinds get OPPOSITE treatment (#4104, roborev job 116). A
98    // STRUCTURAL failure says "the header is not here" — WHERE it lives is in
99    // doubt, so the marker search gets a second attempt at locating it (unchanged
100    // behaviour). A SEMANTIC refusal says "the header IS here and declares a type
101    // Cassandra cannot have written" — position is not in doubt, so re-reading the
102    // same file with a marker search cannot produce a better answer, only a
103    // heuristic guess (`parse_serialization_header` ends by returning an EMPTY
104    // schema as `Ok`, so retrying would turn a correct refusal into silent
105    // acceptance). That is fail-open and a no-heuristics violation (#28), and it
106    // would also put this gate at odds with the KEY-type gate immediately below,
107    // which already fails closed.
108    let (partition_types, clustering_types, columns) = match parse_serialization_header_schema(rest)
109    {
110        Ok((_, result)) => result,
111        // SEMANTIC — fail closed. Decided on the VARIANT, never on message text;
112        // the message is only PROPAGATED (#4104 blocker A), so the user is told
113        // which column and which type were refused, and why.
114        Err(refused @ HeaderSchemaError::Refused(_)) => return Err(refused),
115        // STRUCTURAL — the pre-existing marker-search fallback, untouched.
116        Err(HeaderSchemaError::Structural(e)) => {
117            tracing::warn!(
118                "Schema parsing after EncodingStats failed structurally: {:?}, \
119                 falling back to marker search",
120                e
121            );
122            parse_serialization_header(input)?.1
123        }
124    };
125
126    // Gate 2 of #4104: the key comparators reach `build_column_infos` as RAW
127    // marshal strings (the DESC `ReversedType(..)` signal is derived there), so
128    // this is where a frozen-scalar KEY type is refused. Fail-closed.
129    let (partition_key_columns, clustering_key_columns) =
130        match build_column_infos(&partition_types, &clustering_types) {
131            Ok(cols) => cols,
132            // Semantic, exactly like the column gate above: the refusal message
133            // (which names the key column and its type) is carried to the caller
134            // rather than logged and dropped (#4104 blocker A).
135            Err(e) => return Err(HeaderSchemaError::Refused(e)),
136        };
137
138    Ok((
139        input,
140        (
141            min_timestamp,
142            min_deletion_time,
143            min_ttl,
144            partition_key_columns,
145            clustering_key_columns,
146            columns,
147        ),
148    ))
149}

151/// Parse 3 EncodingStats unsigned VInt deltas and convert to absolute values by adding epochs.
152/// Returns (min_timestamp, min_deletion_time, min_ttl).
153///
154/// # VG3 authority note
155///
156/// The `EncodingStats.Serializer` (EncodingStats.java:274-276) uses the SAME unsigned-VInt
157/// + epoch-offset format for **both** `nb` and `oa`:
158///
159/// ```text
160/// out.writeUnsignedVInt(stats.minTimestamp - TIMESTAMP_EPOCH)
161/// out.writeUnsignedVInt32((int)(stats.minLocalDeletionTime - DELETION_TIME_EPOCH))
162/// out.writeUnsignedVInt32(stats.minTTL - TTL_EPOCH)
163/// ```
164///
165/// The `hasUIntDeletionTime` gate (BigFormat.java:409) affects only the **StatsMetadata**
166/// (STATS component in Statistics.db), not the SerializationHeader component where
167/// EncodingStats lives.  The epoch-relative decoding here is correct for both nb and oa.
168/// `gates` is accepted (not consumed) for API completeness; `None` is fine too.
169fn parse_encoding_stats_vuints<'a>(
170    input: &'a [u8],
171    // VG3: gates threaded here for authority completeness.
172    // Authority investigation: the EncodingStats.Serializer (EncodingStats.java:274-276)
173    // uses the SAME unsigned VInt + epoch format for both nb and oa:
174    //   out.writeUnsignedVInt(stats.minTimestamp - TIMESTAMP_EPOCH)
175    //   out.writeUnsignedVInt32((int)(stats.minLocalDeletionTime - DELETION_TIME_EPOCH))
176    //   out.writeUnsignedVInt32(stats.minTTL - TTL_EPOCH)
177    // The `hasUIntDeletionTime` gate (BigFormat.java:409) affects ONLY the
178    // StatsMetadata section (Statistics.db STATS component), NOT the
179    // SerializationHeader component where EncodingStats lives.  No decode
180    // difference applies here.  Gates accepted but not consumed.
181    _gates: Option<&VersionGates>,
182) -> IResult<&'a [u8], (i64, i64, Option<i64>)> {
183    let (rest, min_ts_delta) = parse_vuint(input)?;
184    let (rest, min_ldt_delta) = parse_vuint(rest)?;
185    let (rest, min_ttl_delta) = parse_vuint(rest)?;
186
187    Ok((
188        rest,
189        (
190            min_ts_delta as i64 + TIMESTAMP_EPOCH,
191            // EncodingStats.java:289: `long minLocalDeletionTime = in.readUnsignedVInt32() + DELETION_TIME_EPOCH`
192            // Same formula for nb and oa — DELETION_TIME_EPOCH is always added back.
193            min_ldt_delta as i64 + DELETION_TIME_EPOCH,
194            Some(min_ttl_delta as i64 + TTL_EPOCH),
195        ),
196    ))
197}

199/// Fallback EncodingStats parser for when no TOC HEADER offset is available.
200/// Uses ad-hoc parsing from the data following the file header.
201fn parse_encoding_stats_fallback<'a>(
202    input: &'a [u8],
203    gates: Option<&VersionGates>,
204) -> Result<(&'a [u8], EncodingStatsResult), HeaderSchemaError<'a>> {
205    // Skip metadata_type (u32 BE) at start of data section
206    let (rest, _metadata_type) = be_u32(input)?;
207
208    // Parse data section length (VInt)
209    let (rest, _data_length) = parse_vuint(rest)?;
210
211    // Parse partitioner string length (VInt)
212    let (rest, partitioner_len) = parse_vuint(rest)?;
213
214    // Skip partitioner string. #3848: `take_vuint_length` narrows the raw `u64`
215    // with a checked conversion — `take(partitioner_len as usize)` would accept a
216    // declared 4 GiB length as an empty run on a 32-bit target.
217    let (rest, _) = take_vuint_length(partitioner_len)(rest)?;
218
219    // Skip additional metadata (observed: ~2 VInts before timestamp fields)
220    let (rest, _metadata1) = parse_vuint(rest)?;
221    let (rest, _metadata2) = parse_vuint(rest)?;
222
223    // Parse EncodingStats fields (unsigned VInt deltas from epoch)
224    let (rest, (min_timestamp, min_deletion_time, min_ttl)) =
225        parse_encoding_stats_vuints(rest, gates)?;
226
227    // Fall back to marker-based header search for schema
228    let (_, (partition_types, clustering_types, columns)) = parse_serialization_header(rest)?;
229
230    // Gate 2 of #4104: the key comparators reach `build_column_infos` as RAW
231    // marshal strings (the DESC `ReversedType(..)` signal is derived there), so
232    // this is where a frozen-scalar KEY type is refused. Fail-closed.
233    let (partition_key_columns, clustering_key_columns) =
234        match build_column_infos(&partition_types, &clustering_types) {
235            Ok(cols) => cols,
236            // Semantic, exactly like the column gate above: the refusal message
237            // (which names the key column and its type) is carried to the caller
238            // rather than logged and dropped (#4104 blocker A).
239            Err(e) => return Err(HeaderSchemaError::Refused(e)),
240        };
241
242    Ok((
243        input,
244        (
245            min_timestamp,
246            min_deletion_time,
247            min_ttl,
248            partition_key_columns,
249            clustering_key_columns,
250            columns,
251        ),
252    ))
