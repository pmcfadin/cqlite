1//! Is a marker-search CANDIDATE actually a SerializationHeader? (#4104, roborev
2//! job 120)
3//!
4//! # Why the question exists at all
5//!
6//! A semantic refusal (`FrozenType(<scalar>)` — a type no Cassandra writer can
7//! have recorded) is authoritative only where the header's LOCATION is known. On
8//! the TOC-anchored path it is known, so the refusal is propagated immediately.
9//! The MARKER SEARCH is the opposite situation: it walks arbitrary bytes looking
10//! for a byte run that decodes like a header, and the decoders accept any UTF-8
11//! string as the key type. So an EARLIER FALSE candidate — a cell value, a
12//! min/max clustering value, any run that happens to sit a few bytes before an
13//! `org.apache.cassandra.db.marshal` occurrence — can decode a plausible
14//! `FrozenType(<scalar>)` column and refuse.
15//!
16//! Honouring that refusal would abort an otherwise VALID SSTable before the real
17//! header was ever reached (an over-refusal). Ignoring it would restore the
18//! fail-open the search had before: a correct refusal replaced by a later
19//! heuristic match or by the dispatcher's own empty-schema success. The answer is
20//! neither: CONFIRM the candidate first, and honour the refusal only then.
21//!
22//! # What confirms a candidate — Cassandra's own writer
23//!
24//! `SerializationHeader.Component.serializer` writes `keyType`, every
25//! `clusteringTypes` entry and every column type through `writeType`, which
26//! serialises `AbstractType::toString()` — the FULLY-QUALIFIED marshal class name
27//! (`cassandra-5.0.8:src/java/org/apache/cassandra/db/SerializationHeader.java`;
28//! the field order and the `[VInt len][string]` framing are tabulated in
29//! `docs/sstables-definitive-guide/chapters/08-statistics-db.md`,
30//! "SerializationHeader Component"). Two consequences are used here, and both are
31//! properties of the WRITER, never of CQLite's prior behaviour:
32//!
33//! 1. A header decodes to the END of its own declared clustering/static/regular
34//!    lists. A run of arbitrary bytes that briefly looked header-shaped does not.
35//! 2. Its key type — and each clustering type — is a marshal class SPELLING, not
36//!    merely a string that contains one somewhere. A candidate the search entered
37//!    a few bytes early decodes a key type with junk in front of the class name,
38//!    which is exactly that difference.
39//!
40//! Nothing here decides a SCHEMA: the survey decode's values are consumed and
41//! dropped, and the only output is the boolean. That is what keeps the deferred
42//! refusal of [`super::sequential::SemanticGate::Survey`] from becoming a
43//! heuristic type fallback (#28).
44
45use super::super::schema_refusal::HeaderSchemaError;
46use super::super::SerializationHeaderResult;

48/// The package every SerializationHeader type string is qualified with.
49const MARSHAL_PACKAGE: &str = "org.apache.cassandra.db.marshal.";
50
51/// Whether `survey` — the SAME candidate bytes decoded under
52/// [`super::sequential::SemanticGate::Survey`], so its refusal did not cut the
53/// decode short — shows a complete SerializationHeader identity.
54///
55/// `true` means a semantic refusal from those bytes is authoritative and must
56/// fail closed. `false` means the candidate is a guess that did not pan out, so
57/// the search continues past it.
58///
59/// The survey result is taken BY VALUE and dropped: a caller cannot accidentally
60/// promote an ungated decode into a schema.
61pub(super) fn candidate_confirmed_as_header(
62    survey: Result<(&[u8], SerializationHeaderResult), HeaderSchemaError<'_>>,
63) -> bool {
64    let Ok((_, (key_types, clustering_types, _columns))) = survey else {
65        tracing::debug!(
66            "Marker-search candidate refused a type but does not decode as a complete \
67             header; treating it as a failed candidate and continuing the search"
68        );
69        return false;
70    };
71
72    // `keyType` is ONE field of the header, and the decoders return exactly the
73    // one they read; anything else is not the shape Cassandra wrote.
74    let [key_type] = key_types.as_slice() else {
75        return false;
76    };
77
78    if !is_marshal_type_spelling(key_type) {
79        tracing::debug!(
80            "Marker-search candidate refused a type but its key type is not a marshal \
81             class spelling ({:?}); continuing the search",
82            key_type
83        );
84        return false;
85    }
86
87    clustering_types.iter().all(|clustering_type| {
88        let spelled = is_marshal_type_spelling(clustering_type);
89        if !spelled {
90            tracing::debug!(
91                "Marker-search candidate refused a type but its clustering type is not a \
92                 marshal class spelling ({:?}); continuing the search",
93                clustering_type
94            );
95        }
96        spelled
97    })
98}

100/// Whether `type_string` is a fully-qualified Cassandra marshal class spelling —
101/// what `AbstractType::toString()` produces and therefore the only thing
102/// `SerializationHeader.writeType` can have written.
103///
104/// Deliberately a SPELLING test, not a "contains" test: a candidate the marker
105/// search entered before the true type-length VInt decodes a key type of the form
106/// `<junk>org.apache.cassandra.db.marshal.…`, which contains the package and is
107/// still not a class name.
108///
109/// Shared with [`super::regular_columns`], whose last-resort scanner has no
110/// `keyType` field to confirm against and so judges the spellings of the types it
111/// did recover (#4104, roborev job 121). One spelling rule, one opinion about
112/// what `writeType` can have produced.
113pub(super) fn is_marshal_type_spelling(type_string: &str) -> bool {
114    // The accepted header forms sometimes hand the top-level type over with a
115    // wrapping paren or bracket (`(org.apache…UUIDType`,
116    // `[org.apache…ReversedType(…)`) — the same normalisation
117    // `marshal_type::is_reversed_comparator` performs.
118    let spelling = type_string
119        .trim()
120        .trim_start_matches(['(', '['])
121        .trim_start();
122    let Some(class_name) = spelling.strip_prefix(MARSHAL_PACKAGE) else {
123        return false;
124    };
125    // A Java class name starts with a letter and carries no control characters.
126    class_name.starts_with(|c: char| c.is_ascii_alphabetic())
127        && !spelling.chars().any(char::is_control)
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133
134    /// The survey result a confirmed candidate produces: one properly spelled
135    /// key type, spelled clustering types, and a decode that reached the end.
136    fn survey_ok<'a>(
137        key_types: &[&str],
138        clustering_types: &[&str],
139    ) -> Result<(&'a [u8], SerializationHeaderResult), HeaderSchemaError<'a>> {
140        Ok((
141            &[],
142            (
143                key_types.iter().map(|t| t.to_string()).collect(),
144                clustering_types.iter().map(|t| t.to_string()).collect(),
145                Vec::new(),
146            ),
147        ))
148    }
149
150    const UTF8: &str = "org.apache.cassandra.db.marshal.UTF8Type";
151
152    #[test]
153    fn a_complete_header_identity_confirms_the_candidate() {
154        assert!(candidate_confirmed_as_header(survey_ok(
155            &[UTF8],
156            &["org.apache.cassandra.db.marshal.TimestampType"],
157        )));
158    }
159
160    /// The over-refusal case: the search entered a few bytes EARLY, so the key
161    /// type carries junk in front of the class name. Cassandra's `writeType`
162    /// cannot have written that, so the candidate's refusal is not authoritative.
163    #[test]
164    fn a_key_type_entered_early_does_not_confirm_the_candidate() {
165        assert!(!candidate_confirmed_as_header(survey_ok(
166            &["\u{0}\u{1}org.apache.cassandra.db.marshal."],
167            &[],
168        )));
169    }
170
171    #[test]
172    fn an_unspelled_clustering_type_does_not_confirm_the_candidate() {
173        assert!(!candidate_confirmed_as_header(survey_ok(
174            &[UTF8],
175            &["a value that merely mentions org.apache.cassandra.db.marshal.Int32Type"],
176        )));
177    }
178
179    /// A candidate that does not decode to the end of its own declared lists is
180    /// bytes that briefly looked like a header.
181    #[test]
182    fn an_incomplete_decode_does_not_confirm_the_candidate() {
183        assert!(!candidate_confirmed_as_header(Err(
184            HeaderSchemaError::structural_at(&[])
185        )));
186    }
187
188    /// `keyType` is one field of the header (guide Ch.8); a survey reporting
189    /// anything else is not the shape Cassandra wrote.
190    #[test]
191    fn a_candidate_without_exactly_one_key_type_is_unconfirmed() {
192        assert!(!candidate_confirmed_as_header(survey_ok(&[], &[])));
193        assert!(!candidate_confirmed_as_header(survey_ok(
194            &[UTF8, UTF8],
195            &[]
196        )));
197    }
198
199    #[test]
200    fn marshal_spellings_accepted_include_the_parenthesised_header_forms() {
201        assert!(is_marshal_type_spelling(UTF8));
202        assert!(is_marshal_type_spelling(
203            "(org.apache.cassandra.db.marshal.UUIDType"
204        ));
205        assert!(is_marshal_type_spelling(
206            "org.apache.cassandra.db.marshal.CompositeType(org.apache.cassandra.db.marshal.UTF8Type)"
207        ));
208        assert!(is_marshal_type_spelling(
209            "[org.apache.cassandra.db.marshal.ReversedType(org.apache.cassandra.db.marshal.Int32Type)"
210        ));
211    }
212
213    #[test]
214    fn a_string_that_merely_contains_the_package_is_not_a_spelling() {
215        // Package with no class name after it.
216        assert!(!is_marshal_type_spelling(
217            "org.apache.cassandra.db.marshal."
218        ));
219        // A different package entirely (the partitioner string, which precedes
220        // the header in the no-TOC preamble).
221        assert!(!is_marshal_type_spelling(
222            "org.apache.cassandra.dht.Murmur3Partitioner"
223        ));
224        // A cell value quoting a type name.
225        assert!(!is_marshal_type_spelling(
226            "frozen<int> is org.apache.cassandra.db.marshal.FrozenType(x)"
227        ));
228    }
