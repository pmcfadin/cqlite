1//! Post-EncodingStats SerializationHeader schema parser.
2//!
3//! `parse_serialization_header_schema` is the authoritative sequential decoder
4//! invoked once the TOC offset has positioned us immediately after the three
5//! EncodingStats VInts. It walks keyType, clusteringTypes, staticColumns, and
6//! regularColumns in their on-disk order (see SerializationHeader.java).
7
8use super::super::super::header::ColumnInfo;
9use super::super::super::vint::parse_vuint;
10use super::super::marshal_type::convert_marshal_type_to_cql_checked;
11use super::super::schema_refusal::HeaderSchemaError;
12use super::super::SerializationHeaderResult;
13use nom::bytes::complete::take;
14
15/// Parse the schema portion of a SerializationHeader (after EncodingStats have been consumed).
16///
17/// Format:
18/// 1. keyType (VInt length + UTF-8 type string)
19/// 2. clusteringTypes (VInt count + [VInt type_len + type]*)
20/// 3. staticColumns (VInt count + [VInt name_len + name + VInt type_len + type]*)
21/// 4. regularColumns (VInt count + [VInt name_len + name + VInt type_len + type]*)
22///
23/// # Failure kinds are NOT interchangeable
24///
25/// The error channel is [`HeaderSchemaError`], not `nom::Err`, precisely so the
26/// caller can tell the two apart (issue #4104, roborev job 116):
27///
28/// * [`HeaderSchemaError::Structural`] — the bytes here are not a readable
29///   header (truncated, implausible declared length, non-UTF-8 name/type). WHERE
30///   the header is, is in doubt, so retrying with the marker-search decoder is
31///   legitimate.
32/// * [`HeaderSchemaError::Refused`] — the header decoded and DECLARES a type
33///   Cassandra cannot have written (`frozen<scalar>`). Position is not in doubt,
34///   so no other decoder can improve the answer; the caller MUST fail closed and
35///   must never substitute a marker-search guess (#28).
36pub(in crate::parser::enhanced_statistics_parser) fn parse_serialization_header_schema<'a>(
37    input: &'a [u8],
38) -> Result<(&'a [u8], SerializationHeaderResult), HeaderSchemaError<'a>> {
39    // Parse keyType (partition key type)
40    let (input, pk_type_len) = parse_vuint(input)?;
41    if pk_type_len == 0 || pk_type_len > 5000 {
42        tracing::debug!("Invalid pk_type_len: {}", pk_type_len);
43        return Err(HeaderSchemaError::structural_at(input));
44    }
45    if pk_type_len > 1000 {
46        tracing::warn!(
47            "Unusually long partition key type string: {} bytes (typical <1000)",
48            pk_type_len
49        );
50    }
51
52    let (input, pk_type_bytes) = take(pk_type_len as usize)(input)?;
53    let partition_key_type = match std::str::from_utf8(pk_type_bytes) {
54        // Gate 2 of #4104: a header type that freezes a scalar is not writable by
55        // Cassandra, so refuse the HEADER rather than read a type it cannot have
56        // recorded. This is a SEMANTIC refusal — the bytes decoded, and what they
57        // declare cannot exist — so it leaves this function as
58        // `HeaderSchemaError::Refused`, carrying the message (with its
59        // `CQL3Type.java:647-651` citation) verbatim. The caller must NOT retry it
60        // through the marker-search fallback (roborev job 116).
61        Ok(s) => match convert_marshal_type_to_cql_checked(s) {
62            Ok(t) => t,
63            Err(e) => {
64                return Err(HeaderSchemaError::refused(format!(
65                    "SerializationHeader partition key type is not writable by Cassandra: {e}"
66                )));
67            }
68        },
69        Err(_) => {
70            tracing::debug!("Invalid UTF-8 in partition key type");
71            return Err(HeaderSchemaError::structural_at(input));
72        }
73    };
74
75    tracing::debug!(
76        "HEADER: Partition key type: {} ({} bytes)",
77        partition_key_type,
78        pk_type_len
79    );
80
81    // Step 3: Parse clusteringTypes
82    let (input, clustering_count) = parse_vuint(input)?;
83    // Sanity check: Cassandra tables rarely have >100 clustering keys
84    if clustering_count > 1000 {
85        tracing::warn!(
86            "Suspicious clustering_count={} in SerializationHeader (expected <100)",
87            clustering_count
88        );
89        return Err(HeaderSchemaError::structural_at(input));
90    }
91    tracing::debug!("HEADER: {} clustering key types", clustering_count);
92
93    let mut input = input;
94    let mut clustering_key_types = Vec::with_capacity(clustering_count as usize);
95
96    for i in 0..clustering_count {
97        let (remaining, ck_type_len) = parse_vuint(input)?;
98        if ck_type_len == 0 || ck_type_len > 5000 {
99            tracing::debug!("Invalid clustering key type length: {}", ck_type_len);
100            return Err(HeaderSchemaError::structural_at(input));
101        }
102        if ck_type_len > 1000 {
103            tracing::warn!(
104                "Unusually long clustering key type string: {} bytes (typical <1000)",
105                ck_type_len
106            );
107        }
108
109        let (remaining, ck_type_bytes) = take(ck_type_len as usize)(remaining)?;
110        // Issue #759: preserve the RAW comparator class name (including any
111        // `ReversedType(...)` wrapper) here. `build_clustering_key_columns` is
112        // the single place that converts to a CQL type AND derives clustering
113        // order from the wrapper, so converting eagerly would discard the DESC
114        // signal. Conversion in `build_clustering_key_columns` is idempotent for
115        // already-CQL strings, keeping the other parse paths correct.
116        let ck_type = match std::str::from_utf8(ck_type_bytes) {
117            Ok(s) => s.to_string(),
118            Err(_) => {
119                tracing::debug!("Invalid UTF-8 in clustering key type {}", i);
120                return Err(HeaderSchemaError::structural_at(input));
121            }
122        };
123
124        tracing::debug!(
125            "HEADER: Clustering key {}: {} ({} bytes)",
126            i,
127            ck_type,
128            ck_type_len
129        );
130        clustering_key_types.push(ck_type);
131        input = remaining;
132    }
133
134    // Step 4: Parse staticColumns
135    let (input, static_count) = parse_vuint(input)?;
136    // Sanity check: Cassandra tables rarely have >1000 static columns
137    if static_count > 10000 {
138        tracing::warn!(
139            "Suspicious static_count={} in SerializationHeader (expected <1000)",
140            static_count
141        );
142        return Err(HeaderSchemaError::structural_at(input));
143    }
144    tracing::debug!("HEADER: {} static columns", static_count);
145
146    let mut input = input;
147    let mut static_columns = Vec::with_capacity(static_count as usize);
148
149    for i in 0..static_count {
150        // Column name
151        let (remaining, name_len) = parse_vuint(input)?;
152        if name_len == 0 || name_len > 200 {
153            tracing::debug!("Invalid static column name length: {}", name_len);
154            return Err(HeaderSchemaError::structural_at(input));
155        }
156
157        let (remaining, name_bytes) = take(name_len as usize)(remaining)?;
158        let column_name = match std::str::from_utf8(name_bytes) {
159            Ok(s) => s.to_string(),
160            Err(_) => {
161                tracing::debug!("Invalid UTF-8 in static column name {}", i);
162                return Err(HeaderSchemaError::structural_at(input));
163            }
164        };
165
166        // Column type
167        let (remaining, type_len) = parse_vuint(remaining)?;
168        if type_len == 0 || type_len > 5000 {
169            tracing::debug!("Invalid static column type length: {}", type_len);
170            return Err(HeaderSchemaError::structural_at(input));
171        }
172        if type_len > 1000 {
173            tracing::warn!(
174                "Unusually long static column type string: {} bytes (typical <1000)",
175                type_len
176            );
177        }
178
179        let (remaining, type_bytes) = take(type_len as usize)(remaining)?;
180        let cql_type = match std::str::from_utf8(type_bytes) {
181            // Gate 2 of #4104 — see the partition-key site for the reasoning.
182            // SEMANTIC: propagated as `Refused`, never retried heuristically.
183            Ok(s) => match convert_marshal_type_to_cql_checked(s) {
184                Ok(t) => t,
185                Err(e) => {
186                    return Err(HeaderSchemaError::refused(format!(
187                        "SerializationHeader static column {i} ('{column_name}') type is not \
188                         writable by Cassandra: {e}"
189                    )));
190                }
191            },
192            Err(_) => {
193                tracing::debug!("Invalid UTF-8 in static column type {}", i);
194                return Err(HeaderSchemaError::structural_at(input));
195            }
196        };
197
198        tracing::debug!(
199            "HEADER: Static column '{}': {} ({} bytes)",
200            column_name,
201            cql_type,
202            type_len
203        );
204
205        static_columns.push(ColumnInfo {
206            name: column_name,
207            column_type: cql_type,
208            is_primary_key: false,
209            key_position: None,
210            is_static: true,
211            is_clustering: false,
212            clustering_reversed: false,
213        });
214
215        input = remaining;
216    }
217
218    // Step 5: Parse regularColumns
219    let (input, regular_count) = parse_vuint(input)?;
220    // Sanity check: Cassandra tables rarely have >1000 regular columns
221    if regular_count > 10000 {
222        tracing::warn!(
223            "Suspicious regular_count={} in SerializationHeader (expected <1000)",
224            regular_count
225        );
226        return Err(HeaderSchemaError::structural_at(input));
227    }
228    tracing::debug!("HEADER: {} regular columns", regular_count);
229
230    let mut input = input;
231    let mut regular_columns = Vec::with_capacity(regular_count as usize);
232
233    for i in 0..regular_count {
234        // Column name
235        let (remaining, name_len) = parse_vuint(input)?;
236        if name_len == 0 || name_len > 200 {
237            tracing::debug!("Invalid regular column name length: {}", name_len);
238            return Err(HeaderSchemaError::structural_at(input));
239        }
240
241        let (remaining, name_bytes) = take(name_len as usize)(remaining)?;
242        let column_name = match std::str::from_utf8(name_bytes) {
243            Ok(s) => s.to_string(),
244            Err(_) => {
245                tracing::debug!("Invalid UTF-8 in regular column name {}", i);
246                return Err(HeaderSchemaError::structural_at(input));
247            }
248        };
249
250        // Column type
251        let (remaining, type_len) = parse_vuint(remaining)?;
252        if type_len == 0 || type_len > 5000 {
253            tracing::debug!("Invalid regular column type length: {}", type_len);
254            return Err(HeaderSchemaError::structural_at(input));
255        }
256        if type_len > 1000 {
257            tracing::warn!(
258                "Unusually long regular column type string: {} bytes (typical <1000)",
259                type_len
260            );
261        }
262
263        let (remaining, type_bytes) = take(type_len as usize)(remaining)?;
264        let cql_type = match std::str::from_utf8(type_bytes) {
265            // Gate 2 of #4104 — see the partition-key site for the reasoning.
266            // SEMANTIC: propagated as `Refused`, never retried heuristically.
267            Ok(s) => match convert_marshal_type_to_cql_checked(s) {
268                Ok(t) => t,
269                Err(e) => {
270                    return Err(HeaderSchemaError::refused(format!(
271                        "SerializationHeader regular column {i} ('{column_name}') type is not \
272                         writable by Cassandra: {e}"
273                    )));
274                }
275            },
276            Err(_) => {
277                tracing::debug!("Invalid UTF-8 in regular column type {}", i);
278                return Err(HeaderSchemaError::structural_at(input));
279            }
280        };
281
282        tracing::debug!(
283            "HEADER: Regular column '{}': {} ({} bytes)",
284            column_name,
285            cql_type,
286            type_len
287        );
288
289        regular_columns.push(ColumnInfo {
290            name: column_name,
291            column_type: cql_type,
292            is_primary_key: false,
293            key_position: None,
294            is_static: false,
295            is_clustering: false,
296            clustering_reversed: false,
297        });
298
299        input = remaining;
300    }
301
302    // Combine static and regular columns
303    let mut all_columns = static_columns;
304    all_columns.extend(regular_columns);
305
306    tracing::debug!(
307        "HEADER parsing complete: partition_key='{}', {} clustering keys, {} total columns",
308        partition_key_type,
309        clustering_key_types.len(),
310        all_columns.len()
311    );
312
313    Ok((
314        input,
315        (vec![partition_key_type], clustering_key_types, all_columns),
316    ))
317}
