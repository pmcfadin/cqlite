20/// Configuration for one-shot ingestion
21#[derive(Debug, Clone)]
22pub struct IngestionConfig {
23    /// Schema file paths (.cql or .json) to load
24    pub schema_paths: Vec<PathBuf>,
25
26    /// Root data directory containing SSTables (e.g., /var/lib/cassandra/data)
27    pub data_dir: PathBuf,
28
29    /// Optional Cassandra version hint (e.g., "5.0")
30    pub version_hint: Option<String>,
31
32    /// Core database configuration
33    pub core_config: Config,
34
35    /// Optional filter for table directories: a **SUBSTRING** match over the whole
36    /// directory path (e.g. `/test_basic/`). Only table directories whose path
37    /// CONTAINS this string are loaded.
38    ///
39    /// The substring semantics are stated here because they are load-bearing and
40    /// surprising: a filter of `/<ks>/<table>-<uuid>` also matches a SIBLING whose
41    /// name EXTENDS it (`<table>-<uuid>-backup`, `<table>-backup`), which silently
42    /// adds generations to the ingest. A caller that needs EXACTLY one directory must
43    /// not express it as a filter — use [`TableDirSelection::Exact`] via
44    /// [`ingest_with_selection`], which compares complete path components (issue
45    /// #3234).
46    pub table_directory_filter: Option<String>,
47}
48
49/// Which discovered table directories the `Database` is built from.
50///
51/// Two genuinely different semantics, kept apart so neither can be mistaken for the
52/// other (issue #3234 roborev F1):
53///
54/// - [`TableDirSelection::Filter`] — the historical
55///   [`IngestionConfig::table_directory_filter`] SUBSTRING match. Loose by design
56///   (`/test_basic/` selects a whole keyspace) and **cannot express "exactly this
57///   directory"**: any sibling whose full name extends the filter also matches.
58/// - [`TableDirSelection::Exact`] — exactly the named directories, compared as
59///   complete path components after canonicalization. No substring, prefix or glob
60///   semantics anywhere in the path, so a `<table>-<uuid>-backup` sibling contributes
61///   nothing. This matters beyond tidiness: an extra directory changes the GENERATION
62///   COUNT, and the generation count selects the scan route.
63#[derive(Debug, Clone, Copy)]
64pub enum TableDirSelection<'a> {
65    /// Substring match over the whole path, from `table_directory_filter`.
66    Filter,
67    /// Exactly these directories, by complete-path-component identity.
68    Exact(&'a [PathBuf]),
69}
70
71/// Complete-path-component identity: canonicalize both sides (absolute,
72/// symlink-free, `.`/`..` resolved) and compare. `Path`'s `Eq` is component-wise, so
73/// this is an exact component comparison and NEVER a substring or prefix one — a
74/// canonicalized `<table>-<uuid>` can no longer match `<table>-<uuid>-backup`.
75///
76/// A path that cannot be canonicalized (it does not exist, or is unreadable) matches
77/// nothing: fail-closed, so an unresolvable request can never widen the selection.
78fn is_same_dir(a: &Path, b: &Path) -> bool {
79    match (a.canonicalize(), b.canonicalize()) {
80        (Ok(x), Ok(y)) => x == y,
81        _ => false,
82    }
83}
84
85/// Apply a [`TableDirSelection`] to discovery's table directories.
86fn select_table_dirs(
87    discovered: &[PathBuf],
88    filter: Option<&str>,
89    selection: TableDirSelection<'_>,
90) -> Vec<PathBuf> {
91    match selection {
92        TableDirSelection::Filter => match filter {
93            Some(pattern) => discovered
94                .iter()
95                .filter(|path| path.to_string_lossy().contains(pattern))
96                .cloned()
97                .collect(),
98            None => discovered.to_vec(),
99        },
100        TableDirSelection::Exact(wanted) => discovered
101            .iter()
102            .filter(|path| wanted.iter().any(|w| is_same_dir(path, w)))
103            .cloned()
104            .collect(),
105    }
106}

139/// Main ingestion function for one-shot execution
140///
141/// This function orchestrates:
142/// 1. Schema loading from provided paths (CQL/JSON files)
143/// 2. SSTable discovery from data directory
144/// 3. Database construction with QueryEngine
145///
146/// # Errors
147///
148/// Returns Error::Schema for schema loading failures (exit code 3)
149/// Returns Error::Io for discovery/data-dir failures (exit code 4)
150/// Returns Error::QueryExecution for query engine setup failures (exit code 5)
151pub async fn ingest(config: IngestionConfig) -> Result<IngestionResult> {
152    ingest_with_selection(config, TableDirSelection::Filter).await
153}
154
155/// [`ingest`], with the table-directory selection stated explicitly.
156///
157/// `TableDirSelection::Filter` is exactly what [`ingest`] does. `Exact` is for a
158/// caller whose measurement or correctness claim depends on ingesting EXACTLY one
159/// directory and no sibling that happens to share a prefix (issue #3234 roborev F1).
160pub async fn ingest_with_selection(
161    config: IngestionConfig,
162    selection: TableDirSelection<'_>,
163) -> Result<IngestionResult> {
164    // Step 1: Validate data directory exists
165    if !config.data_dir.exists() {
166        return Err(Error::Io(std::io::Error::new(
167            std::io::ErrorKind::NotFound,
168            format!(
169                "Data directory does not exist: {}",
170                config.data_dir.display()
171            ),
172        )));
173    }
174
175    if !config.data_dir.is_dir() {
176        return Err(Error::Io(std::io::Error::new(
177            std::io::ErrorKind::InvalidInput,
178            format!(
179                "Data directory path is not a directory: {}",
180                config.data_dir.display()
181            ),
182        )));
183    }
184
185    // Step 2: Initialize Platform
186    let platform = Arc::new(Platform::new(&config.core_config).await?);
187
188    // Step 3: Load schemas using SchemaAggregator
189    let registry_config = SchemaRegistryConfig::default();
190    let schema_registry = Arc::new(RwLock::new(
191        SchemaRegistry::new(
192            registry_config,
193            platform.clone(),
194            config.core_config.clone(),
195        )
196        .await
197        .map_err(|e| Error::Schema(format!("Failed to create schema registry: {}", e)))?,
198    ));
199
200    // Use the SchemaRegistry's internal UDT registry so UDTs are available for query execution
201    // This ensures UDTs loaded from CQL files are available when parsing SSTable data (Issue #238)
202    let udt_registry = schema_registry.read().await.get_udt_registry();
203
204    let aggregator_config = AggregatorConfig {
205        graceful_degradation: false, // Fail fast for one-shot execution
206        validate_udt_dependencies: true,
207    };
208
209    let mut aggregator = SchemaAggregator::new(
210        schema_registry.clone(),
211        udt_registry.clone(),
212        aggregator_config,
213    );
214
215    let schema_load_result = if !config.schema_paths.is_empty() {
216        aggregator
217            .load_from_paths(&config.schema_paths)
218            .await
219            .map_err(|e| Error::Schema(format!("Schema loading failed: {}", e)))?
220    } else {
221        // No schema paths provided - return empty result
222        LoadResult {
223            schemas_loaded: 0,
224            udts_loaded: 0,
225            errors: Vec::new(),
226            warnings: Vec::new(),
227        }
228    };
229
230    // Check for schema loading errors (fail fast)
231    if !schema_load_result.errors.is_empty() {
232        let error_messages: Vec<String> = schema_load_result
233            .errors
234            .iter()
235            .map(|e| format!("{:?}: {}", e.error_type, e.message))
236            .collect();
237        return Err(Error::Schema(format!(
238            "Schema loading failed with {} error(s): {}",
239            schema_load_result.errors.len(),
240            error_messages.join("; ")
241        )));
242    }
243
244    // Step 4: Discover SSTables using DiscoveryService
245    let discovery_service = DiscoveryService::with_schema_registry(
246        config.data_dir.clone(),
247        config.version_hint.clone(),
248        schema_registry.clone(),
249    );
250
251    let service_summary = discovery_service.scan().await.map_err(|e| {
252        // Map discovery errors to appropriate error types
253        match e {
254            Error::Io(_) => e,
255            _ => Error::Io(std::io::Error::other(format!(
256                "SSTable discovery failed: {}",
257                e
258            ))),
259        }
260    })?;
261
262    // Step 5: Select the table directories this Database is built from.
263    let filtered_table_dirs = select_table_dirs(
264        &service_summary.table_directories,
265        config.table_directory_filter.as_deref(),
266        selection,
267    );
268
269    // Step 6: Build Database with discovered (and optionally filtered) SSTables
270    // Pass the loaded schema_registry to the Database so schemas are available to the query engine
271    // Storage path is data_dir (for runtime storage), discovered directories from DiscoveryService
272    let database = Database::open_with_discovered_sstables_and_registry(
273        &config.data_dir,
274        filtered_table_dirs.clone(),
275        config.core_config.clone(),
276        Some(schema_registry.clone()),
277    )
278    .await
279    .map_err(|e| {
280        // Map database creation errors appropriately
281        match e {
282            Error::Schema(_) => e,
283            Error::Io(_) => e,
284            #[cfg(feature = "state_machine")]
285            Error::QueryExecution(_) => e,
286            _ => Error::QueryExecution(format!("Database initialization failed: {}", e)),
287        }
288    })?;
289
290    // Convert from discovery module's DiscoverySummary to ingestion's DiscoverySummary
291    // Use the filtered table directories in the summary
292    let discovery_summary = DiscoverySummary {
293        sstables_found: service_summary.sstables_found,
294        keyspaces: service_summary.keyspaces,
295        tables: service_summary.tables,
296        table_directories: filtered_table_dirs,
297        resolved_version: service_summary.resolved_version,
298    };
299
300    Ok(IngestionResult {
301        database,
302        schema_load_result,
303        discovery_summary,
304        schema_registry: schema_registry.clone(),
305    })
306}

357    /// A keyspace tree holding the wanted `<table>-<uuid>` directory plus two
358    /// siblings whose full names EXTEND it. Returns `(root, wanted, siblings)`.
359    fn tree_with_name_extending_siblings() -> (TempDir, PathBuf, Vec<PathBuf>) {
360        let root = TempDir::new().unwrap();
361        let ks = root.path().join("perf_bti");
362        let uuid = "a1b2c3d40000000000000000000000ff";
363        let wanted = ks.join(format!("wide_multiclustering-{uuid}"));
364        // Both of these are what a substring/prefix matcher cannot distinguish from
365        // `wanted`: the first EXTENDS its complete name, the second extends the bare
366        // table name.
367        let siblings = vec![
368            ks.join(format!("wide_multiclustering-{uuid}-backup")),
369            ks.join("wide_multiclustering-backup"),
370        ];
371        for dir in std::iter::once(&wanted).chain(siblings.iter()) {
372            std::fs::create_dir_all(dir).unwrap();
373            std::fs::write(dir.join("da-1-bti-Data.db"), b"x").unwrap();
374        }
375        (root, wanted, siblings)
376    }

378    /// Issue #3234 roborev F1: `Exact` selects the named directory and NOTHING whose
379    /// name merely extends it.
380    #[test]
381    fn exact_selection_excludes_name_extending_siblings() {
382        let (_root, wanted, siblings) = tree_with_name_extending_siblings();
383        let mut discovered = vec![wanted.clone()];
384        discovered.extend(siblings.iter().cloned());
385
386        let selected = select_table_dirs(
387            &discovered,
388            // A filter is present AND would over-match; `Exact` must ignore it
389            // entirely rather than intersect with it.
390            Some("/perf_bti/wide_multiclustering"),
391            TableDirSelection::Exact(std::slice::from_ref(&wanted)),
392        );
393        assert_eq!(
394            selected,
395            vec![wanted],
396            "Exact must select exactly the named directory"
397        );
398    }

400    /// The control for the test above: the SUBSTRING filter really does sweep in the
401    /// name-extending siblings, which is why `Exact` exists and why a filter of
402    /// `/<ks>/<table>-<uuid>` was never an exact scope (issue #3234 roborev F1).
403    #[test]
404    fn substring_filter_matches_name_extending_siblings() {
405        let (_root, wanted, siblings) = tree_with_name_extending_siblings();
406        let mut discovered = vec![wanted.clone()];
407        discovered.extend(siblings.iter().cloned());
408
409        let wanted_name = wanted.file_name().unwrap().to_string_lossy().to_string();
410        let selected = select_table_dirs(
411            &discovered,
412            Some(&format!("/perf_bti/{wanted_name}")),
413            TableDirSelection::Filter,
414        );
415        assert_eq!(
416            selected.len(),
417            2,
418            "the substring filter matches the wanted dir AND the sibling extending its \
419             full name: {selected:?}"
420        );
421        assert!(selected.contains(&siblings[0]));
422    }

424    /// `Exact` is fail-closed on a path that cannot be canonicalized: an unresolvable
425    /// request selects nothing rather than widening the selection.
