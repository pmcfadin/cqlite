1//! Shared per-mutation statistics fold (issue #1668, stage 5c-iv part 2).
2//!
3//! [`SSTableWriter::write_partition`] folds every mutation of a partition into
4//! `StatisticsMetadata` (min/max timestamp, LDT, TTL, the tombstone-drop-time
5//! histogram, the live-LDT sentinel, and the partition-level-deletion flag)
6//! BEFORE writing any bytes for that partition. The incremental streaming path
7//! (`KWayMerger::merge`, `writer/incremental.rs`) sees mutations one at a time
8//! and cannot buffer a whole partition to reproduce that ordering, but the
9//! fold itself is a pure function of ONE mutation plus the running
10//! accumulator — extracted here so both paths call the exact same logic and
11//! can never drift.
12//!
13//! [`SSTableWriter::write_partition`]: super::SSTableWriter::write_partition
14
15use crate::schema::TableSchema;
16use crate::storage::sstable::writer::data_writer::DataWriter;
17use crate::storage::sstable::writer::stats_writer::StatisticsMetadata;
18use crate::storage::write_engine::mutation::{CellOperation, Mutation};

20/// Whether `group` (one or more mutations sharing a clustering key) survives
21/// into a real emitted row under `shadow_floor` — the SAME decision
22/// [`DataWriter::merge_row_group`] (the sole authority for row shadow-drop,
23/// issue #1668) makes when actually writing Data.db.
24///
25/// Exposed as a thin, side-effect-free pre-check (rather than widening
26/// `merge_row_group`'s own `pub(super)` visibility out to
27/// `write_engine::merge`) so BOTH the buffered
28/// ([`super::SSTableWriter::write_partition`]) and incremental
29/// (`compact_sstables`'s streaming merge, `writer/incremental.rs`) paths can
30/// gate their STATISTICS fold on the SAME real emission decision (issue
31/// #4246): a row fully shadow-dropped from Data.db (e.g. a live write
32/// shadowed by a covering range/partition tombstone) must not lower
33/// persisted `StatisticsMetadata` minima — Cassandra's `MetadataCollector`
34/// only ever observes what `SortedTableWriter` actually emits
35/// (`SortedTableWriter.java`/`MetadataCollector.java`, cassandra-5.0.8).
36///
37/// Scope (issue #4246): this gates CLUSTERING-ROW mutations only. A mutation
38/// that also carries a static-column operation is deliberately EXCLUDED from
39/// this gate by callers (see their own doc comments) — static-cell shadowing
40/// by a partition tombstone is a separate, pre-existing question this fix
41/// does not attempt, and skipping such a mutation's fold here could
42/// under-count a surviving static write.
43pub(crate) fn row_group_survives(
44    group: &[&Mutation],
45    schema: &TableSchema,
46    skip_static_ops: bool,
47    shadow_floor: Option<i64>,
48) -> bool {
49    DataWriter::merge_row_group(group, schema, skip_static_ops, shadow_floor).is_some()
50}

52/// Like [`row_group_survives`], but ALSO returns the group's resolved shadow
53/// boundary (`deletion_ts`, `None` when nothing shadows anything) — the SAME
54/// value [`DataWriter::merge_row_group`] shadows individual mutations
55/// against internally.
56///
57/// A row GROUP can survive overall (`Some` — e.g. a newer `DELETE` for the
58/// clustering key still produces a row) while containing an OLDER mutation
59/// in the SAME group (e.g. an earlier `INSERT` for the same clustering key
60/// in one flush batch) that is itself fully shadowed by that later
61/// deletion — roborev finding on issue #4246: gating only at the group level
62/// still folds that older, non-emitted mutation's timestamp. Callers must
63/// additionally check each mutation via
64/// `deletion_ts.is_none_or(|dts| mutation.timestamp_micros > dts)` (mirroring
65/// `merge_row_group`'s own `mutation_shadowed` test) before folding it.
66///
67/// Also returns the group's RAW resolved row deletion (`row_deletion`,
68/// before combining with `shadow_floor`) — the exact tuple
69/// `merge_row_group` emits UNCONDITIONALLY as `RowWrite.row_deletion`
70/// whenever `Some` (`data_writer/rows.rs`: `row.row_deletion` is set from
71/// `resolve_row_deletion`'s result directly, never re-gated against
72/// `deletion_ts`). Callers fold this ONCE per group via
73/// [`fold_row_deletion_marker`] — never per-mutation (issue #4246 roborev
74/// round-2 finding: a per-mutation `carries_row_deletion` exemption in
75/// `fold_row_content_stats` either double-folded the winning deletion
76/// (already `mutation_shadowed` under a uniform check, since `deletion_ts`
77/// is derived from its own timestamp) or wrongly folded a LOSING `DeleteRow`
78/// that `resolve_row_deletion` discarded and Data.db never emits).
79///
80/// PERFORMANCE (issue #4288, roborev round-5 finding): this calls
81/// `merge_row_group` as a pre-check, and every call site ALSO calls it again
82/// (directly or via `feed_row`/`merge_clustering_rows`) to actually emit the
83/// row — `merge_row_group`'s own full per-column LWW reconciliation
84/// (allocating `cells`/`whole_col_liveness`/`complex_element_ops`/`ops`) now
85/// runs 2-3x per row group on the write hot path instead of once.
86/// `merge_row_group` is a pure function of its arguments, so a future
87/// refactor computing the group's `RowWrite` once and deriving both the
88/// stats-fold decision and the emission from it would remove this
89/// duplication; deferred out of #4246's own scope (a correctness fix) to
90/// avoid a broader refactor under time pressure.
91pub(crate) fn row_group_survival(
92    group: &[&Mutation],
93    schema: &TableSchema,
94    skip_static_ops: bool,
95    shadow_floor: Option<i64>,
96) -> (bool, Option<i64>, Option<(i64, i32)>) {
97    // A SINGLE `merge_row_group` call (issue #4246 roborev round-3
98    // performance finding: this used to ALSO call `resolve_row_deletion`
99    // standalone before `merge_row_group`, which re-runs it internally as
100    // part of its own full reconciliation — a redundant group scan on the
101    // write hot path for every row group, on top of the emitter's own
102    // separate `merge_row_group` call to actually build the row). When the
103    // group survives, `row.row_deletion` IS the exact tuple
104    // `resolve_row_deletion` would have returned (see `merge_row_group`'s
105    // own doc comment) — no separate call needed. `combine_deletion_ts` is a
106    // cheap `Option` combine, not a group re-scan.
107    match DataWriter::merge_row_group(group, schema, skip_static_ops, shadow_floor) {
108        Some(row) => {
109            let deletion_ts = DataWriter::combine_deletion_ts(row.row_deletion, shadow_floor);
110            (true, deletion_ts, row.row_deletion)
111        }
112        // A group with ANY resolved row deletion always produces at least a
113        // tombstone row (`merge_row_group`'s own final "produces no row at
114        // all" check is `false` whenever `row_deletion.is_some()`), so
115        // `None` here implies `row_deletion` was ALSO `None` — every caller
116        // only consumes `deletion_ts`/`row_deletion` inside a `survives`
117        // branch, so this is behaviorally identical to the two-call form.
118        None => (false, None, None),
119    }
120}

122/// Fold the row GROUP's resolved row-deletion marker (the winning `DeleteRow`
123/// / issue #932 decoupled `row_tombstone`) exactly ONCE — the third element
124/// of [`row_group_survival`]'s return tuple. This is the ONLY place a row's
125/// own deletion marker reaches persisted stats: `fold_row_content_stats` no
126/// longer special-cases a deletion-carrying mutation (issue #4246 roborev
127/// round-2 finding — see its doc comment), so a group with N mutations
128/// contending for the row deletion (losing `DeleteRow`s included) folds the
129/// winner's `(timestamp, local_deletion_time)` exactly once, matching what
130/// `merge_row_group` actually emits.
131pub(crate) fn fold_row_deletion_marker(
132    stats: &mut StatisticsMetadata,
133    row_deletion: Option<(i64, i32)>,
134) {
135    if let Some((ts, ldt)) = row_deletion {
136        stats.update_timestamp(ts);
137        stats.update_local_deletion_time(ldt);
138    }
139}

141/// The exact fold sequence for a row GROUP consisting of a SINGLE mutation —
142/// the shape both incremental streaming paths use (`KWayMerger::merge`'s
143/// `PartitionEnd` handling and `WriteEngine::maintenance_step`'s buffered
144/// `PartitionEnd` drain): by the time either sees it, a compaction/merge
145/// "cluster group" is already one fully-reconciled `Mutation` per clustering
146/// key (unlike `write_partition`'s buffered multi-mutation-per-group case),
147/// so `row_group_survival`'s `group` argument is always a one-element slice.
148///
149/// Extracted (issue #4246 roborev round-3 finding): the two call sites used
150/// to hand-assemble this identical four-step sequence independently, so a
151/// future edit to one could silently drift from the other with nothing to
152/// catch it — this module's own `#1668` equivalence test
153/// (`single_mutation_row_group_fold_...`) now exercises THIS function
154/// directly, the same one both production paths call.
155pub(crate) fn fold_single_mutation_row_group(
156    stats: &mut StatisticsMetadata,
157    mutation: &Mutation,
158    schema: &TableSchema,
159    schema_has_static: bool,
160    shadow_floor: Option<i64>,
161) {
162    let (survives, deletion_ts, row_deletion) =
163        row_group_survival(std::slice::from_ref(&mutation), schema, false, shadow_floor);
164    fold_row_deletion_marker(stats, row_deletion);
165    let carries_static = schema_has_static
166        && mutation.operations.iter().any(|op| {
167            crate::storage::sstable::writer::data_writer::is_static_operation(op, schema)
168        });
169    if carries_static {
170        // Static-cell shadowing uses a SEPARATE, partition-floor-only
171        // mechanism unrelated to this row-level `deletion_ts` (out of this
172        // fix's verified scope, see `row_group_survives`'s doc comment) —
173        // pass `None` so per-op shadow gating never applies to it, exactly
174        // the prior unconditional-fold behavior.
175        fold_row_content_stats(stats, mutation, None);
176    } else if survives {
177        fold_row_content_stats(stats, mutation, deletion_ts);
178    }
179}

181/// The fold sequence for a STATIC-ROW CARRIER mutation — the
182/// `clustering_key: None` shape all three writer paths classify out of
183/// their clustering-row grouping and handle on their own branch
184/// (`SSTableWriter::write_partition`'s wholly-static loop,
185/// `KWayMerger::merge`, and `WriteEngine::maintenance_step`).
186///
187/// # A static carrier's own row deletion is excluded from BOTH fold paths
188///
189/// The exclusion is deliberate, and it has to be re-derived in TWO
190/// independent places, because there are two separate paths by which a
191/// mutation reaches the SAME persisted `Statistics.db` fields:
192///
193/// 1. the DIRECT fold — THIS function, called by
194///    `SSTableWriter::write_partition`'s wholly-static loop,
195///    `KWayMerger::merge` and `WriteEngine::maintenance_step`, which folds
196///    into `self.stats` (fixed in issue #4246 roborev round 8, below); and
197/// 2. the TWO-PASS PRE-SEED ENCODING BASELINE —
198///    `SSTableWriter::compute_mutations_baseline_stats` ->
199///    `fold_one_mutation_baseline` -> `pre_seed_encoding_baselines`, the
200///    issue #729 flush pre-scan, whose result is assigned VERBATIM into
201///    `self.stats.min_timestamp` / `.min_local_deletion_time` BEFORE any
202///    partition is written (fixed in roborev round 10).
203///
204/// Path 2 is not a refinement of path 1: `write_partition`'s own fold runs
205/// strictly AFTER the pre-seed assignment and is a `.min()`, so it can only
206/// LOWER the pre-seeded value further, never raise a phantom-contaminated
207/// one back. Gating path 1 alone therefore left the phantom fully intact in
208/// the persisted file — invisible to the `estimatedTombstoneDropTime`
209/// histogram, since pre-seeding is a field assignment rather than an
210/// `update_local_deletion_time` call. That is how round 8's fix read as
211/// complete for two further rounds.
212///
213/// Having to answer "was this actually emitted?" twice, in two hand-written
214/// re-derivations that nothing forces to agree, is the structural problem
215/// issue **#4320** proposes to remove (derive stats from the emitted
216/// artifact). Until then, a change to either path must be mirrored in the
217/// other; `cqlite-core/tests/issue_4246_static_carrier_deletion_phantom.rs`
218/// pins both.
219///
220/// ## The adjudication (round 8)
221///
222/// Issue #4246 roborev rounds 6/7 added a `fold_row_deletion_marker(stats,
223/// DataWriter::resolve_row_deletion(&[mutation], None))` call here, on the
224/// theory that a static carrier's `CellOperation::DeleteRow` / #932
225/// `row_tombstone` was otherwise "folded nowhere". Round 8 overturned it and
226/// the fold was REMOVED: it was a PHANTOM marker — a `Statistics.db` entry
227/// (persisted minima plus an `estimatedTombstoneDropTime` histogram
228/// increment) for a deletion that has no corresponding bytes in `Data.db`.
229/// That is precisely the "persisted stats do not match emitted data" defect
230/// class issue #4246 exists to eliminate, so the fold was an instance of the
231/// bug, not a fix for one.
232///
233/// FORMAT AUTHORITY (pinned `cassandra-5.0.8`, never CQLite's own code).
234/// Cassandra derives a static row's stats from THE SAME `Row` object it just
235/// serialized, in adjacent statements —
236/// `io/sstable/format/SortedTableWriter.java::addStaticRow`:
237/// ```java
238/// partitionWriter.addStaticRow(row);
239/// if (!row.isEmpty())
240///     Rows.collectStats(row, metadataCollector);
241/// ```
242/// and `db/rows/Rows.java::collectStats` reads `row.deletion()` — the very
243/// field `db/rows/UnfilteredSerializer.java::serialize` turns into the
244/// `HAS_DELETION` (`0x10`) flag. Emission and collection therefore cannot
245/// diverge by construction: Cassandra counts a static-row deletion IF AND
246/// ONLY IF it wrote one. Any CQLite fold that counts a deletion the emitter
247/// will not write violates that invariant.
248///
249/// WHAT CQLITE'S EMITTER ACTUALLY DOES. No production path can emit a
250/// static-row deletion at all:
251///   * `data_writer/encoding.rs::is_static_operation` returns `false` for
252///     `CellOperation::DeleteRow`, and `data_writer/static_ops.rs`'s
253///     `StaticOpsTracker::feed` additionally `continue`s on that variant —
254///     so `collect_static_operations`/`StaticOpsTracker::finish` can never
255///     place a `DeleteRow` into the merged static-op set.
256///   * `data_writer/static_rows.rs::write_static_row_with_prev_size` sets
257///     `ROW_HAS_DELETION` only when its `static_ops` slice contains a
258///     `DeleteRow`, and never consults `Mutation::row_tombstone` at all.
259///   * Every production static-row emission passes a merged set into that
260///     function (`partition.rs`, `streaming_partition.rs`,
261///     `incremental_partition.rs`, `incremental.rs::feed_streaming_static_row`).
262///     The one entry point that maps `mutation.operations` UNFILTERED —
263///     `DataWriter::write_static_row` — has NO production caller; it is
264///     reached only from tests.
265///
266/// Verified empirically on the flush path: a static carrier whose only
267/// operation is `DeleteRow` (and, separately, one carrying a #932
268/// `row_tombstone`) emits a static row whose flags byte is `0x80`/`0xa0` —
269/// `ROW_HAS_DELETION` clear — while the pre-removal fold still persisted a
270/// `tombstone_drop_times` bucket for it. Both halves are pinned by
271/// `cqlite-core/tests/issue_4246_static_carrier_deletion_phantom.rs`, which
272/// is also the tripwire: if the emitter is ever taught to write a static-row
273/// deletion, its byte assertion FAILS and this fold must be revisited.
274///
275/// WHY ISSUE #1721 DOES NOT APPLY. Rounds 6/7 justified the fold by citing
276/// #1721 ("without the LDT contribution `min_local_deletion_time` stays
277/// `i32::MAX` and `data_writer/rows.rs`'s below-baseline guard REJECTS the
278/// row"). That is a MISCITATION. Commit `e638bf369`'s regression
279/// (`tests/issue_1385_gc_grace_boundary.rs::
280/// write_decoupled_row_tombstone_with_survivor`) builds a mutation with
281/// `clustering_key: Some(ck)` against a schema whose columns are ALL
282/// `is_static: false` — a CLUSTERING row in a table with no static columns,
283/// which `is_static_row_mutation` rejects twice over. #1721's deletion IS
284/// emitted (by `merge_row_group`, as a `ROW_HAS_DELETION` row), and today it
285/// is folded at the GROUP level by [`fold_row_deletion_marker`] on the
286/// clustering-row path. Nothing in #1721 concerns a static carrier, and the
287/// below-baseline guard cannot fire for a deletion that is never written.
288///
289/// # What this helper does fold
290///
291/// The carrier's row CONTENT only, via [`fold_row_content_stats`] with a
292/// `None` shadow boundary: static-cell shadowing uses a separate,
293/// partition-floor-only mechanism outside this fix's verified scope (see
294/// [`row_group_survives`]'s doc comment), so the carrier keeps its prior
295/// unconditional-fold behavior exactly.
296///
297/// [`fold_row_content_stats`] ALONE, with no accompanying
298/// [`fold_marker_stats`] call (issue #4246 roborev finding; the test-only
299/// `fold_mutation_stats` composes both). The only markers that second fold
300/// could add are a `partition_tombstone`/`range_tombstones` field on this
301/// SAME mutation object — and on a static carrier such a field is
302/// REACHABLE-BUT-DEAD: a carrier reaches this helper only with non-empty
303/// `operations`, whereas those fields are populated exclusively by the
304/// mutually-exclusive partition-only/range-only branches, which require
305/// EMPTY `operations`. So nothing here is double-counted, and nothing that
306/// Data.db actually carries is left unfolded.
307///
308/// Kept as a NAMED helper rather than inlined at the three call sites even
309/// though its body is now a single delegation: it is the one place this
310/// invariant is stated, and the production composition
311/// `fold_marker_stats` + `fold_single_mutation_row_group` +
312/// `fold_static_carrier_stats` is what the fold-equivalence tests assert
313/// against.
314///
315/// KNOWN RESIDUAL, out of scope here (pre-existing, predates #4246): the
316/// delegation below still folds `mutation.timestamp_micros` for a carrier
317/// that contributes NO static cell — where Cassandra's `addStaticRow` skips
318/// collection entirely for an empty row (`if (!row.isEmpty())`). Closing
319/// that requires folding from the merged static-op set the emitter actually
320/// writes, which is the broader "derive stats from the emitted artifact"
321/// refactor `row_group_survival`'s doc comment already defers.
322pub(crate) fn fold_static_carrier_stats(stats: &mut StatisticsMetadata, mutation: &Mutation) {
323    fold_row_content_stats(stats, mutation, None);
324}

326/// Fold ONLY the partition/range tombstone MARKER fields of `mutation` (issue
327/// #4246 roborev finding). A tombstone marker is never itself row-shadowed —
328/// it IS the deletion — so it always contributes, independent of whatever
329/// `fold_row_content_stats` decides about the row content sharing the same
330/// mutation object. Split out from `fold_mutation_stats` so a caller that
331/// folds markers unconditionally (once per mutation, regardless of
332/// [`row_group_survives`]/[`row_group_survival`]) and row content
333/// conditionally (only for surviving mutations) never double-folds a
334/// tombstone's local-deletion-time into the tombstone-drop-time histogram —
335/// `StatisticsMetadata::update_local_deletion_time` is NOT idempotent (it
336/// increments a histogram bucket), so folding the same marker twice inflates
337/// the persisted `estimatedTombstoneDropTime` Cassandra derives compaction
338/// scheduling from.
339pub(crate) fn fold_marker_stats(stats: &mut StatisticsMetadata, mutation: &Mutation) {
340    if let Some(pt) = &mutation.partition_tombstone {
341        stats.update_timestamp(pt.deletion_time);
342        stats.update_local_deletion_time(pt.local_deletion_time);
343        stats.mark_partition_level_deletion();
344    }
345    for rt in &mutation.range_tombstones {
346        stats.update_timestamp(rt.deletion_time);
347        stats.update_local_deletion_time(rt.local_deletion_time);
348    }
349}

351/// Fold one mutation's timestamp/TTL/local-deletion-time/tombstone information
352/// into `stats`. Mirrors the per-mutation loop body that used to live inline in
353/// [`super::SSTableWriter::write_partition`] verbatim — same chokepoints
354/// (`update_timestamp`, `update_local_deletion_time`, `update_ttl`,
355/// `note_live_local_deletion_time`, `mark_partition_level_deletion`), same
356/// per-`CellOperation` handling, so folding every mutation of a partition
357/// through this function (in any order — the chokepoints are commutative
358/// min/max folds, issue #1668 stage 5a) reproduces `write_partition`'s final
359/// `StatisticsMetadata` byte-for-byte.
360///
361/// `#[cfg(test)]` (issue #4246 roborev round 2): every production caller now
362/// needs shadow-aware row content
363/// (`fold_row_content_stats(stats, mutation, shadow_boundary)`), markers
364/// folded separately from row content (`fold_marker_stats`, so a mutation
365/// carrying both never double-counts a tombstone into the drop-time
366/// histogram), and a row's own deletion folded once at the GROUP level
367/// (`fold_row_deletion_marker`, never per-mutation) — this convenience
368/// "fold everything for one mutation treated as its own trivial group"
369/// wrapper has no remaining non-test caller. A TEST-ONLY CONVENIENCE
370/// WRAPPER, not a regression proof (roborev round-2 finding: the earlier
371/// wording claimed it reproduces "exactly what folding a mutation
372/// unconditionally always did", which is circular once its own definition
373/// IS that composition — treats `mutation` as a one-element group via the
374/// real `DataWriter::resolve_row_deletion`/`fold_row_deletion_marker` path
375/// rather than duplicating that logic, so a future drift between the two
376/// would still surface here).
377#[cfg(test)]
378pub(crate) fn fold_mutation_stats(stats: &mut StatisticsMetadata, mutation: &Mutation) {
379    fold_row_content_stats(stats, mutation, None);
380    fold_marker_stats(stats, mutation);
381    let row_deletion = DataWriter::resolve_row_deletion(&[mutation], None);
382    fold_row_deletion_marker(stats, row_deletion);
383}

385/// Everything `fold_mutation_stats` folds EXCEPT the partition/range
386/// tombstone marker fields (see [`fold_marker_stats`]'s doc for why the split
387/// exists, issue #4246 roborev finding). Used by a caller that folds markers
388/// separately/unconditionally so a mutation carrying both row content and a
389/// tombstone is never double-counted into the tombstone-drop-time histogram.
390///
391/// `shadow_boundary` is the row GROUP's resolved `deletion_ts` (see
392/// [`row_group_survival`]), or `None` when the caller does not need
393/// per-mutation shadow awareness (every existing caller before issue #4246,
394/// and the wholly-static/carries-static callers today, which pass `None` to
395/// preserve their prior unconditional-fold behavior exactly).
396///
397/// When `Some(dts)`, the mutation's SIMPLE content (its own leading
398/// timestamp, row-level TTL, and `WriteWithTtl`/`Delete`/`DeleteRow`/`Write`
399/// contributions) is shadow-gated on `mutation.timestamp_micros <= dts` —
400/// matching `merge_row_group`'s own `mutation_shadowed` test EXACTLY, with
401/// NO exemption for a mutation that itself carries a `DeleteRow` op or a
402/// `#932` `row_tombstone` (issue #4246 roborev round-2 finding: an earlier
403/// version exempted such a mutation, reasoning that "`dts` is derived from
404/// its own timestamp when it is the winning deletion" — but `dts` is
405/// ALWAYS `>=` any row-deletion-carrying mutation's own timestamp in the
406/// group by construction, whether it WON or LOST that contention, so the
407/// exemption let a LOSING `DeleteRow`'s never-emitted timestamp/LDT lower
408/// persisted minima — precisely the #4246 defect this fix exists to close).
409/// This makes the `Delete { .. } | DeleteRow` match arm below structurally
410/// unreachable for the `DeleteRow` case specifically (it is always
411/// `mutation_shadowed`) — correct, since the group's row deletion is now
412/// folded exactly once at the GROUP level via
413/// [`fold_row_deletion_marker`]/[`row_group_survival`], never here.
414///
415/// A SECOND, per-cell filter mirrors `merge_row_group`'s own
416/// (`data_writer/rows.rs`, issue #1018 roborev HIGH / issue #4246 roborev
417/// round-4 Medium finding): even when the mutation's ROW timestamp survives
418/// (`!mutation_shadowed`), an individual `Write`/`WriteWithTtl`/`Delete` cell
419/// may carry its OWN (lower) per-cell timestamp in
420/// `Mutation::cell_write_timestamps` (populated by the compaction
421/// merge→mutation path) that is itself `<= dts` — covered by the
422/// partition/range/row tombstone and never emitted, even though the row
423/// survives via other content. Each of those three arms resolves its own
424/// `cell_ts` via `Mutation::cell_write_timestamp` (falling back to
425/// `mutation.timestamp_micros` when no per-cell override exists, a no-op
426/// for the common single-writetime row) and gates on it independently of
427/// `mutation_shadowed`.
428///
429/// `ComplexDeletion`/`WriteComplexElement` ops are gated PER-OP instead,
430/// each against its OWN independent timestamp (`marked_for_delete_at` /
431/// `timestamp_micros`) rather than the mutation's row timestamp — mirroring
432/// `merge_row_group`'s rescue branch (`data_writer/rows.rs`'s
433/// `mutation_shadowed` handling), because such an op can independently
434/// survive and be EMITTED even when the rest of a shadowed mutation is dead
435/// (issue #887/#921's whole point, and issue #4246 roborev finding: a
436/// per-mutation gate that skips these too under-counts an actually-emitted
437/// value).
438pub(crate) fn fold_row_content_stats(
439    stats: &mut StatisticsMetadata,
440    mutation: &Mutation,
441    shadow_boundary: Option<i64>,
442) {
443    let mutation_shadowed = shadow_boundary.is_some_and(|dts| mutation.timestamp_micros <= dts);
444
445    if !mutation_shadowed {
446        stats.update_timestamp(mutation.timestamp_micros);
447        if let Some(ttl) = mutation.ttl_seconds {
448            stats.update_ttl(ttl as i32);
449            let now_seconds = std::time::SystemTime::now()
450                .duration_since(std::time::UNIX_EPOCH)
451                .map(|d| d.as_secs() as i32)
452                .unwrap_or(0);
453            let local_deletion_time = now_seconds.saturating_add(ttl as i32);
454            stats.update_local_deletion_time(local_deletion_time);
455        }
456    }
457    // Track local deletion times for tombstones and TTL cells. Issue #764:
458    // row/cell tombstones use the caller-supplied `local_deletion_time` when
459    // present, else the timestamp-derived value.
460    for op in &mutation.operations {
461        match op {
462            CellOperation::WriteWithTtl {
463                column,
464                ttl_seconds,
465                local_deletion_time,
466                ..
467            } => {
468                if mutation_shadowed {
469                    continue;
470                }
471                // Issue #1018 (roborev round-4 Medium finding, mirroring
472                // `merge_row_group`'s identical per-cell filter,
473                // `data_writer/rows.rs`): a mutation whose ROW timestamp
474                // survives can still carry an individual cell whose OWN
475                // resolved timestamp (`Mutation::cell_write_timestamps`,
476                // populated by the compaction merge→mutation path) is
477                // `<= shadow_boundary` — covered by the partition/range/row
478                // tombstone and never emitted, even though the row itself
479                // survives via other content. `cell_write_timestamp` falls
480                // back to `mutation.timestamp_micros` when no per-cell
481                // override exists, so this is a no-op for the common
482                // single-writetime row.
483                let cell_ts = mutation.cell_write_timestamp(column);
484                if shadow_boundary.is_some_and(|dts| cell_ts <= dts) {
485                    continue;
486                }
487                stats.update_timestamp(cell_ts);
488                stats.update_ttl(*ttl_seconds as i32);
489                // Issue #1538: honor the authoritative per-cell LDT VERBATIM
490                // when present (a surviving expiring cell preserved through
491                // compaction); `None` keeps the historical `now + ttl`
492                // derivation.
493                let local_deletion_time = match local_deletion_time {
494                    Some(ldt) => *ldt,
495                    None => {
496                        let now_seconds = std::time::SystemTime::now()
497                            .duration_since(std::time::UNIX_EPOCH)
498                            .map(|d| d.as_secs() as i32)
499                            .unwrap_or(0);
500                        now_seconds.saturating_add(*ttl_seconds as i32)
501                    }
502                };
503                stats.update_local_deletion_time(local_deletion_time);
504            }
505            op @ CellOperation::Delete { column, .. } => {
506                if mutation_shadowed {
507                    continue;
508                }
509                // Per-cell shadow filter (roborev round-4 Medium finding) —
510                // see the identical comment on `WriteWithTtl` above.
511                let cell_ts = mutation.cell_write_timestamp(column);
512                if shadow_boundary.is_some_and(|dts| cell_ts <= dts) {
513                    continue;
514                }
515                stats.update_timestamp(cell_ts);
516                // Issue #764 / #921 finding 2: record the EXACT LDT the
517                // tombstone is emitted with, via the same helper the emit path
518                // uses, so stats and Data.db bytes agree exactly.
519                let local_deletion_time =
520                    crate::storage::sstable::writer::data_writer::op_cell_local_deletion_time(
521                        op, mutation,
522                    );
523                stats.update_local_deletion_time(local_deletion_time);
524            }
525            // `DeleteRow` is NEVER folded here (issue #4246 roborev round-3
526            // finding): the row's own deletion — whether contributed by a
527            // `DeleteRow` op or a #932 decoupled `row_tombstone` — is folded
528            // EXACTLY ONCE at the GROUP level via `fold_row_deletion_marker`
529            // (see `row_group_survival`'s doc comment), regardless of
530            // `shadow_boundary`. This arm used to double-fold the LDT into
531            // the tombstone-drop-time histogram specifically for a
532            // `carries_static` mutation (whose caller always passes
533            // `shadow_boundary = None`, so `mutation_shadowed` was always
534            // `false` here and this body unconditionally re-ran after the
535            // group-level fold already counted it once) —
536            // `update_local_deletion_time` increments a histogram bucket and
537            // is NOT idempotent, exactly the defect class
538            // `mixed_row_and_partition_tombstone_mutation_folds_tombstone_once`
539            // guards markers against.
540            CellOperation::DeleteRow => {}
541            // Issue #887: a `ComplexDeletion` marker is physically written with
542            // its OWN `marked_for_delete_at` / `local_deletion_time`, which may
543            // fall outside the row's own timestamp/LDT range. INDEPENDENT of
544            // `mutation_shadowed` (issue #4246 roborev finding): gated on its
545            // OWN mfda against `shadow_boundary`, mirroring
546            // `merge_row_group`'s rescue branch — this marker can survive and
547            // be emitted even when the rest of a shadowed mutation is dead.
548            CellOperation::ComplexDeletion {
549                marked_for_delete_at,
550                local_deletion_time,
551                ..
552            } => {
553                if shadow_boundary.is_some_and(|dts| *marked_for_delete_at <= dts) {
554                    continue;
555                }
556                stats.update_timestamp(*marked_for_delete_at);
557                stats.update_local_deletion_time(*local_deletion_time);
558            }
559            // Issue #887: a per-element complex cell carries its OWN explicit
560            // timestamp/ttl/local_deletion_time — INDEPENDENT of
561            // `mutation_shadowed` for the same reason as `ComplexDeletion`
562            // above (issue #4246 roborev finding).
563            CellOperation::WriteComplexElement {
564                timestamp_micros,
565                ttl_seconds,
566                local_deletion_time,
567                is_deleted,
568                ..
569            } => {
570                if shadow_boundary.is_some_and(|dts| *timestamp_micros <= dts) {
571                    continue;
572                }
573                stats.update_timestamp(*timestamp_micros);
574                if let Some(ttl) = ttl_seconds {
575                    stats.update_ttl(*ttl as i32);
576                }
577                if let Some(ldt) = local_deletion_time {
578                    stats.update_local_deletion_time(*ldt);
579                }
580                // Issue #1728 (roborev finding 2): a LIVE complex element
581                // carries Cassandra's `NO_DELETION_TIME` sentinel just like a
582                // live simple `Write`.
583                if !*is_deleted && ttl_seconds.is_none() && local_deletion_time.is_none() {
584                    stats.note_live_local_deletion_time();
585                }
586            }
587            // Issue #1728: a live, non-TTL `Write` cell carries Cassandra's
588            // `Cell.NO_DELETION_TIME` sentinel as its localDeletionTime.
589            CellOperation::Write { column, value, .. } => {
590                if mutation_shadowed {
591                    continue;
592                }
593                // Per-cell shadow filter (roborev round-4 Medium finding) —
594                // see the identical comment on `WriteWithTtl` above.
595                let cell_ts = mutation.cell_write_timestamp(column);
596                if shadow_boundary.is_some_and(|dts| cell_ts <= dts) {
597                    continue;
598                }
599                stats.update_timestamp(cell_ts);
600                if mutation.ttl_seconds.is_none() && !matches!(value, crate::types::Value::Null) {
601                    stats.note_live_local_deletion_time();
602                }
603            }
604        }
605    }
606    // Issue #1721 / #932: a decoupled row tombstone
607    // (`Mutation::row_tombstone = Some((deletion_time, ldt))`) is NOT folded
608    // here (issue #4246 roborev round-2 finding — removed the earlier
609    // unconditional fold). `resolve_row_deletion` already considers
610    // `mutation.row_tombstone` as a candidate for the group's winning row
611    // deletion, on equal footing with a `DeleteRow` op, so folding it again
612    // here would either double-count the winner or wrongly count a losing
613    // `row_tombstone` Data.db never emits. The group's actual winning
614    // `(deletion_time, ldt)` — from a `DeleteRow` op OR a `row_tombstone`,
615    // whichever `resolve_row_deletion` picked — is folded exactly once at
616    // the GROUP level via [`fold_row_deletion_marker`].
617}

619/// Fold `from`'s accumulated range/flags into `into` (issue #1668 stage
620/// 5c-iv part 2). Used to merge a partition-scoped fold (accumulated while
621/// streaming a partition's mutations one at a time) into the SSTable-wide
622/// running `StatisticsMetadata` at partition end, once the incremental
623/// session's exclusive borrow of the writer has ended.
624///
625/// Min/max fields are commutative folds (stage 5a), so feeding both of
626/// `from`'s min and max back through the same chokepoints on `into`
627/// reproduces exactly what folding every mutation directly into `into` would
628/// have produced — PROVIDED `from`'s own untouched-default sentinels are
629/// excluded first. `update_timestamp` already self-filters its untouched
630/// defaults (`i64::MAX`/`i64::MIN`, both treated as the LIVE/NO_DELETION
631/// marker, issue #851), so timestamps are safe to feed unconditionally. LDT
632/// and TTL are NOT symmetric:
633///   * `update_local_deletion_time` filters `i32::MAX` (live sentinel) but
634///     NOT `i32::MIN` (`from.max_local_deletion_time`'s untouched default
635///     when no tombstone was ever folded) — guarded explicitly below, else an
636///     empty `from` would drag `into.min_local_deletion_time` down to
637///     `i32::MIN`.
638///   * `update_ttl` only filters non-positive values, so `from.min_ttl`'s
639///     untouched default (`i32::MAX`, itself positive) must be excluded
640///     explicitly, else it would corrupt `into.max_ttl` to `i32::MAX`.
641///
642/// The live-LDT sentinel itself is NOT re-derived through
643/// `update_local_deletion_time` (which filters `i32::MAX` OUT) —
644/// `from.max_local_deletion_time == i32::MAX` is used as the equivalent "saw
645/// a live cell" signal instead, since `note_live_local_deletion_time` is the
646/// ONLY setter that can assign `i32::MAX` to `max_local_deletion_time` (a
647/// real tombstone LDT is always filtered before it can reach the max).
648pub(crate) fn merge_stats_fold(into: &mut StatisticsMetadata, from: &StatisticsMetadata) {
649    into.update_timestamp(from.min_timestamp);
650    into.update_timestamp(from.max_timestamp);
651
652    into.update_local_deletion_time(from.min_local_deletion_time);
653    if from.max_local_deletion_time > i32::MIN {
654        into.update_local_deletion_time(from.max_local_deletion_time);
655    }
656    if from.max_local_deletion_time == i32::MAX {
657        into.note_live_local_deletion_time();
658    }
659
660    if from.min_ttl != i32::MAX {
661        into.update_ttl(from.min_ttl);
662    }
663    into.update_ttl(from.max_ttl);
664
665    if from.has_partition_level_deletions {
666        into.mark_partition_level_deletion();
667    }
668}
