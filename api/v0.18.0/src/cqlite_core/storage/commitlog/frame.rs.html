1//! Per-record framing + sync-marker walking for a Cassandra CommitLog segment
2//! (issue #2389).
3//!
4//! After the [descriptor header](super::descriptor), a segment body is a chain
5//! of **sync-marker-delimited sections**. Each section begins with an 8-byte
6//! sync marker and holds a run of length-and-CRC-framed mutation records.
7//!
8//! All layouts below are verified byte-for-byte against a real Cassandra 5.0.2
9//! segment (the fixtures under `test-data/datasets/commitlog/`).
10//!
11//! ```text
12//! sync marker (8 bytes, big-endian):
13//!   int32 nextMarker   file offset of the *next* sync marker (= end of section)
14//!   int32 crc          CRC32 over: id-low(BE i32), id-high(BE i32), markerPos(BE i32)
15//!
16//! record (per mutation):
17//!   int32 size         serialized mutation length
18//!   int32 sizeCrc      CRC32 over the 4 size bytes (BE)
19//!   bytes body         `size` mutation bytes
20//!   int32 bodyCrc      CRC32 continuing over size bytes ++ body bytes
21//! ```
22//!
23//! **Truncation vs corruption.** Missing record or marker bytes, a nonzero
24//! marker with a bad CRC, and a valid forward marker beyond end-of-file are
25//! treated as a torn tail: prior records are kept and the walk stops with
26//! [`FrameStep::Truncated`]. A fully present record with a bad CRC, or a
27//! valid-CRC marker with a negative or backward next-marker offset, is
28//! corruption and returns [`Error::CorruptCommitLogFrame`]. Zeroed preallocated
29//! space is a clean end. This mirrors Cassandra's `CommitLogReader`
30//! `tolerateTruncation` behavior without accepting structurally invalid
31//! backward pointers.
32
33use crate::{Error, Result};
34
35/// Size of a sync marker in bytes (`int32 nextMarker` + `int32 crc`).
36pub const SYNC_MARKER_SIZE: usize = 8;
37
38/// Per-record overhead: `int32 size` + `int32 sizeCrc` + `int32 bodyCrc`.
39pub const RECORD_OVERHEAD: usize = 12;
40
41/// One step of a frame walk.
42#[derive(Debug)]
43pub enum FrameStep<'a> {
44    /// A validated mutation record body (the serialized `Mutation` bytes).
45    Record(&'a [u8]),
46    /// Clean end of the written segment (padding / zeroed pre-allocation).
47    End,
48    /// Torn tail: a record or marker ran past end-of-file. Records already
49    /// yielded before this step are valid and complete.
50    Truncated,
51}
52
53/// Lazy cursor over a segment's sync sections and records.
54///
55/// Holds only a borrow of the segment bytes plus a couple of cursor integers —
56/// it never materializes the decoded record set, satisfying the streaming
57/// memory budget (design decision D4 / `oom-audit`).
58pub struct FrameWalker<'a> {
59    bytes: &'a [u8],
60    segment_id: i64,
61    /// Position of the sync marker that opens the current section.
62    marker_pos: usize,
63    /// Cursor within the current section (position of the next record).
64    cursor: usize,
65    /// End offset of the current section (the next marker's position).
66    section_end: usize,
67    /// Whether we are positioned inside an open section.
68    in_section: bool,
69    /// Set when the walk ended on a torn tail rather than clean padding.
70    truncated_end: bool,
71    /// Terminal flag once [`FrameStep::End`]/[`FrameStep::Truncated`] is reached.
72    done: bool,
73}
74
75impl<'a> FrameWalker<'a> {
76    /// Start a walk at the segment body (immediately after the descriptor).
77    ///
78    /// `body_start` is [`super::descriptor::CommitLogDescriptor::header_len`];
79    /// `segment_id` is the descriptor's authoritative segment id (used to verify
80    /// every sync marker's CRC).
81    pub fn new(bytes: &'a [u8], segment_id: i64, body_start: usize) -> Self {
82        Self {
83            bytes,
84            segment_id,
85            marker_pos: body_start,
86            cursor: 0,
87            section_end: 0,
88            in_section: false,
89            truncated_end: false,
90            done: false,
91        }
92    }
93
94    /// Advance to the next record, end, or torn tail.
95    ///
96    /// # Errors
97    /// [`Error::CorruptCommitLogFrame`] when a fully-present record fails its
98    /// size/body CRC, its framing overruns a section, or a valid-CRC sync marker
99    /// has a negative or backward next-marker offset. Invalid marker CRCs and
100    /// forward markers past EOF are reported as [`FrameStep::Truncated`].
101    pub fn next_frame(&mut self) -> Result<FrameStep<'a>> {
102        if self.done {
103            return Ok(FrameStep::End);
104        }
105        loop {
106            if !self.in_section {
107                match self.open_section()? {
108                    Some(()) => {}
109                    None => {
110                        self.done = true;
111                        return Ok(if self.truncated_end {
112                            FrameStep::Truncated
113                        } else {
114                            FrameStep::End
115                        });
116                    }
117                }
118            }
119
120            // Within the open section, read the next record.
121            match self.read_record()? {
122                RecordOutcome::Body(b) => return Ok(FrameStep::Record(b)),
123                RecordOutcome::SectionDone => {
124                    // Advance to the next marker.
125                    self.in_section = false;
126                    self.marker_pos = self.section_end;
127                    continue;
128                }
129                RecordOutcome::CleanEnd => {
130                    self.done = true;
131                    // A torn section (opened past-EOF marker) sets truncated_end
132                    // at open time; honor it here too, not just in open_section's
133                    // own None path — otherwise hitting padding inside a torn
134                    // section's remaining bytes silently downgrades Truncated to
135                    // a clean End (roborev finding, review-first pass).
136                    return Ok(if self.truncated_end {
137                        FrameStep::Truncated
138                    } else {
139                        FrameStep::End
140                    });
141                }
142                RecordOutcome::Truncated => {
143                    self.done = true;
144                    return Ok(FrameStep::Truncated);
145                }
146            }
147        }
148    }
149
150    /// Open the section at `marker_pos`. Returns `Ok(None)` on clean end or torn
151    /// tail (with `truncated_end` set appropriately).
152    fn open_section(&mut self) -> Result<Option<()>> {
153        let pos = self.marker_pos;
154        // A marker whose bytes are missing → torn tail.
155        if pos + SYNC_MARKER_SIZE > self.bytes.len() {
156            // Distinguish a clean end (nothing but absence) from a torn write:
157            // any leftover non-marker bytes present but < 8 is a torn tail.
158            // `|=`, not `=`: a prior torn section may have already set this
159            // true, and this branch reaching `pos < len` false (e.g. a
160            // section whose records happen to fill the buffer exactly) must
161            // not silently downgrade it back to a clean end — the exact bug
162            // this fix's neighboring CleanEnd-arm change was meant to close,
163            // just at a second assignment site (roborev finding, review-
164            // first pass).
165            self.truncated_end |= pos < self.bytes.len();
166            return Ok(None);
167        }
168        let marker = &self.bytes[pos..pos + SYNC_MARKER_SIZE];
169        // Only a fully zeroed marker is clean preallocated space. Do not clear
170        // truncated_end: once the walk has observed a torn tail, later padding
171        // cannot turn it back into a clean segment.
172        if marker.iter().all(|&byte| byte == 0) {
173            return Ok(None);
174        }
175        let next_marker = read_i32_be(self.bytes, pos);
176        // A zero next-marker value with nonzero CRC bytes is not the zeroed
177        // preallocation sentinel; it must still pass CRC and pointer checks.
178        // This preserves the distinction between a torn marker and a valid-CRC
179        // backward pointer.
180        let stored_crc = read_u32_be(self.bytes, pos + 4);
181        let computed = marker_crc(self.segment_id, pos);
182        if computed != stored_crc {
183            // A complete, nonzero marker with a bad CRC is tolerated as a torn
184            // tail. It cannot make a previously observed truncation disappear.
185            self.truncated_end = true;
186            return Ok(None);
187        }
188        // The marker offset is signed on disk. Reject negative values before
189        // any conversion so they cannot wrap into a huge positive `usize` and
190        // masquerade as a torn forward section.
191        if next_marker < 0 {
192            return Err(Error::CorruptCommitLogFrame(format!(
193                "sync marker at offset {pos} has negative next-marker offset {next_marker}"
194            )));
195        }
196        // On targets where usize is narrower than i32, an unrepresentable
197        // positive offset is necessarily beyond this in-memory file and has
198        // the same torn-tail meaning as any other forward offset past EOF.
199        let next = match usize::try_from(next_marker) {
200            Ok(next) => next,
201            Err(_) => {
202                self.open_torn_section(pos);
203                return Ok(Some(()));
204            }
205        };
206        // Marker is valid but points past EOF → the section body is torn. This
207        // section necessarily runs to true end-of-file (nothing can follow a
208        // marker that already overruns the buffer), so the walk is guaranteed
209        // to terminate truncated — set that now, not just on the eventual
210        // "no more marker bytes" path, so a CleanEnd/SectionDone reached while
211        // still inside this torn section doesn't get reported as a clean end
212        // (roborev finding, review-first pass).
213        if next > self.bytes.len() {
214            self.open_torn_section(pos);
215            return Ok(Some(()));
216        }
217        // A marker must point forward past its own 8 bytes. The CRC covers
218        // id/position, not next_marker; it cannot establish how a bad offset
219        // arose. Cassandra 5.0.8 CommitLogSegmentReader.readSyncMarker still
220        // classifies a backward offset with a matching CRC as corruption,
221        // including zero at a real (positive) marker position. Preserve that
222        // policy instead of guessing that the offset resulted from a torn write.
223        if next < pos + SYNC_MARKER_SIZE {
224            return Err(Error::CorruptCommitLogFrame(format!(
225                "sync marker at offset {pos} points backward to {next}"
226            )));
227        }
228        self.section_end = next;
229        self.cursor = pos + SYNC_MARKER_SIZE;
230        self.in_section = true;
231        Ok(Some(()))
232    }
233
234    /// Continue reading the final section to EOF after a valid forward marker
235    /// points beyond the available bytes.
236    fn open_torn_section(&mut self, marker_pos: usize) {
237        self.section_end = self.bytes.len();
238        self.cursor = marker_pos + SYNC_MARKER_SIZE;
239        self.in_section = true;
240        self.truncated_end = true;
241    }
242
243    /// Read one record from within the open section.
244    fn read_record(&mut self) -> Result<RecordOutcome<'a>> {
245        let cur = self.cursor;
246        let len = self.bytes.len();
247
248        // No room for a record header within the section → section finished.
249        if cur >= self.section_end {
250            return Ok(RecordOutcome::SectionDone);
251        }
252        // `is_final_section` distinguishes "this section runs to true EOF" from
253        // "more sections follow at self.section_end". Cassandra pads the tail of
254        // EVERY non-final section up to its next sync marker — a zeroed/short
255        // record header there means "done with THIS section, continue at the
256        // next marker" (SectionDone), not "the whole segment ends here"
257        // (CleanEnd/Truncated). Treating it as segment-end silently drops every
258        // mutation in all subsequent sections (roborev finding, review-first
259        // pass).
260        let is_final_section = self.section_end >= len;
261        if cur + 4 > self.section_end {
262            return Ok(if is_final_section {
263                RecordOutcome::Truncated
264            } else {
265                RecordOutcome::SectionDone
266            });
267        }
268        let size = read_i32_be(self.bytes, cur);
269        // Zeroed padding: true clean end of the written segment only when this
270        // is the final section; otherwise it's just this section's padding.
271        if size <= 0 {
272            return Ok(if is_final_section {
273                RecordOutcome::CleanEnd
274            } else {
275                RecordOutcome::SectionDone
276            });
277        }
278        let size = size as usize;
279        let end = cur + 4 + 4 + size + 4;
280        // Record extends past EOF → torn tail.
281        if end > len {
282            return Ok(RecordOutcome::Truncated);
283        }
284        // Record extends past the section boundary but is within the file: the
285        // marker/section framing disagrees with the record length — corrupt.
286        // No `&& !self.torn_section` guard here: in a torn section
287        // section_end == bytes.len() == len, so `end > self.section_end`
288        // would imply `end > len`, which the EOF check above already caught
289        // and returned Truncated for — this branch is unreachable with
290        // torn_section true regardless, so an explicit exemption clause only
291        // obscured that invariant and invited a future edit to make it live
292        // and wrong (roborev finding, review-first pass).
293        if end > self.section_end {
294            return Err(Error::CorruptCommitLogFrame(format!(
295                "record at offset {cur} (len {size}) overruns section end {}",
296                self.section_end
297            )));
298        }
299
300        let size_crc = read_u32_be(self.bytes, cur + 4);
301        let size_bytes = &self.bytes[cur..cur + 4];
302        let mut hasher = crc32fast::Hasher::new();
303        hasher.update(size_bytes);
304        let computed_size_crc = hasher.finalize();
305        if computed_size_crc != size_crc {
306            return Err(Error::CorruptCommitLogFrame(format!(
307                "record size CRC mismatch at offset {cur}: stored={size_crc:#010x} \
308                 computed={computed_size_crc:#010x}"
309            )));
310        }
311
312        let body_start = cur + 8;
313        let body = &self.bytes[body_start..body_start + size];
314        let body_crc = read_u32_be(self.bytes, body_start + size);
315        // Continuous CRC over the size bytes followed by the body bytes.
316        let mut hasher = crc32fast::Hasher::new();
317        hasher.update(size_bytes);
318        hasher.update(body);
319        let computed_body_crc = hasher.finalize();
320        if computed_body_crc != body_crc {
321            return Err(Error::CorruptCommitLogFrame(format!(
322                "record body CRC mismatch at offset {cur}: stored={body_crc:#010x} \
323                 computed={computed_body_crc:#010x}"
324            )));
325        }
326
327        self.cursor = end;
328        Ok(RecordOutcome::Body(body))
329    }
330}

332enum RecordOutcome<'a> {
333    Body(&'a [u8]),
334    SectionDone,
335    CleanEnd,
336    Truncated,
337}

339/// Compute a sync-marker CRC the way Cassandra's `writeSyncMarker` does:
340/// CRC32 over id-low (BE i32), id-high (BE i32), markerPos (BE i32).
341pub fn marker_crc(segment_id: i64, marker_pos: usize) -> u32 {
342    let id = segment_id as u64;
343    let mut hasher = crc32fast::Hasher::new();
344    hasher.update(&((id & 0xFFFF_FFFF) as u32).to_be_bytes());
345    hasher.update(&((id >> 32) as u32).to_be_bytes());
346    hasher.update(&(marker_pos as u32).to_be_bytes());
347    hasher.finalize()
348}

350#[inline]
351fn read_i32_be(b: &[u8], off: usize) -> i32 {
352    i32::from_be_bytes([b[off], b[off + 1], b[off + 2], b[off + 3]])
353}
354#[inline]
355fn read_u32_be(b: &[u8], off: usize) -> u32 {
356    u32::from_be_bytes([b[off], b[off + 1], b[off + 2], b[off + 3]])
357}

365    fn push_marker(buf: &mut Vec<u8>, pos: usize, next_marker: i32) {
366        buf.extend_from_slice(&next_marker.to_be_bytes());
367        buf.extend_from_slice(&marker_crc(SEGMENT_ID, pos).to_be_bytes());

370    fn push_record(buf: &mut Vec<u8>, body: &[u8]) {
371        let size = body.len() as i32;
372        let size_bytes = size.to_be_bytes();
373        let mut h = crc32fast::Hasher::new();
374        h.update(&size_bytes);
375        let size_crc = h.finalize();
376        buf.extend_from_slice(&size_bytes);
377        buf.extend_from_slice(&size_crc.to_be_bytes());
378        buf.extend_from_slice(body);
379        let mut h = crc32fast::Hasher::new();
380        h.update(&size_bytes);
381        h.update(body);
382        buf.extend_from_slice(&h.finalize().to_be_bytes());
383    }

385    /// Regression for the roborev finding that the `SectionDone` fix (a
386    /// non-final section's padding must finish THAT section, not the whole
387    /// segment) had zero test coverage — every committed fixture is
388    /// single-section. Builds a real two-section segment byte-for-byte per
389    /// this module's own documented layout: section 1 (one record + zero
390    /// padding to the next marker), section 2 (one record, then a clean
391    /// natural end at true EOF).
392    #[test]
393    fn walks_records_across_multiple_sync_sections() {
394        let mut buf = Vec::new();
395        // Section 1: marker at 0, one record, 8 bytes of zero padding.
396        let record1 = b"section-one-record";
397        push_marker(&mut buf, 0, 0 /* placeholder, patched below */);
398        push_record(&mut buf, record1);
399        buf.extend_from_slice(&[0u8; 8]); // padding to the next marker
400        let section2_marker_pos = buf.len();
401        // Patch section 1's marker to point at section 2's marker (its own
402        // section_end), now that we know the offset.
403        let next1 = section2_marker_pos as i32;
404        buf[0..4].copy_from_slice(&next1.to_be_bytes());
405        buf[4..8].copy_from_slice(&marker_crc(SEGMENT_ID, 0).to_be_bytes());
406
407        // Section 2: marker, one record, then true EOF (no trailing padding
408        // needed — section_end == bytes.len() makes this the final section).
409        let record2 = b"section-two-record";
410        let record2_start = section2_marker_pos + SYNC_MARKER_SIZE;
411        let final_len = record2_start + 4 + 4 + record2.len() + 4;
412        push_marker(&mut buf, section2_marker_pos, final_len as i32);
413        push_record(&mut buf, record2);
414        assert_eq!(buf.len(), final_len, "test construction sanity check");
415
416        let mut walker = FrameWalker::new(&buf, SEGMENT_ID, 0);
417        let mut records = Vec::new();
418        loop {
419            match walker.next_frame().expect("no corruption in this fixture") {
420                FrameStep::Record(b) => records.push(b.to_vec()),
421                FrameStep::End => break,
422                FrameStep::Truncated => panic!("expected a clean end, not Truncated"),
423            }
424        }
425        assert_eq!(
426            records,
427            vec![record1.to_vec(), record2.to_vec()],
428            "both sections' records must be yielded — a regression that treats \
429             section 1's padding as segment-end would silently drop record2"
430        );
431    }

433    /// Regression: `truncated_end` was assigned (`=`) rather than OR-ed
434    /// (`|=`) at two sites in `open_section`, so it could be silently
435    /// downgraded from true back to false after a torn section had already
436    /// set it. Trips exactly when a torn section's one record happens to end
437    /// precisely at EOF: the section finishes via `SectionDone` (not
438    /// `CleanEnd`), re-enters `open_section` at `marker_pos == bytes.len()`,
439    /// and that branch's `pos < len` is false — with a plain `=` this wiped
440    /// truncated_end back to false and reported a clean End for a genuinely
441    /// torn segment (roborev finding, review-first pass).
442    #[test]
443    fn torn_section_whose_record_ends_exactly_at_eof_still_reports_truncated() {
444        // Marker's own CRC covers only (segment_id, marker_pos) — NOT the
445        // next_marker value — so next_marker can point arbitrarily far past
446        // EOF while the marker itself still validates, exactly like a real
447        // segment whose section was truncated mid-write.
448        let mut buf = Vec::new();
449        push_marker(&mut buf, 0, 999_999); // far past EOF -> torn section
450        push_record(&mut buf, b"last-record-fills-the-buffer-exactly");
451
452        let mut walker = FrameWalker::new(&buf, SEGMENT_ID, 0);
453        match walker.next_frame().expect("valid record") {
454            FrameStep::Record(_) => {}
455            other => panic!("expected the one record, got {other:?}"),
456        }
457        match walker.next_frame().expect("no corruption") {
458            FrameStep::Truncated => {}
459            other => panic!(
460                "expected Truncated (torn section, record ends exactly at EOF), got {other:?}"
461            ),
462        }
463    }
