1//! Async pipeline entry points for the SELECT executor (issue #1174).
2//!
3//! This submodule continues the [`SelectExecutor`](super::SelectExecutor) `impl`
4//! with the large `async` pipeline methods that drive query execution:
5//! - [`SelectExecutor::execute`] — the materializing plan runner,
6//! - [`SelectExecutor::execute_sstable_scan`] — the SSTable-scan step.
7//!
8//! (The streaming producer, `execute_streaming_background`, moved to the sibling
9//! `streaming` submodule in issue #1578's file-size split.)
10//!
11//! These were relocated verbatim from `mod.rs` (epic #1116 file-size split); the
12//! per-step helpers they call (`execute_filter`, `execute_sort`, etc.) and the
13//! `ExecutionContext` bookkeeping struct remain in `mod.rs`. As a child module,
14//! this file can reach `mod.rs`'s private items directly — the logic, ordering,
15//! and error handling are unchanged.

33impl SelectExecutor {
34    /// Execute an optimized query plan.
35    ///
36    /// Instrumented as `query.select.plan` (issue #1035): this span covers the
37    /// modern SELECT pipeline — SSTable scan, filtering, projection, aggregation,
38    /// and WRITETIME/TTL metadata extraction — and is the parent under which the
39    /// read-path spans (issue #1034) nest. On completion it emits
40    /// [`catalog::QUERY_ROWS_SCANNED`] (rows the scan step examined) dimensioned by
41    /// the honest access path, so the rows-scanned vs rows-returned gap is
42    /// observable. The bounded access-path attribute is recorded on the span; the
43    /// query text and key values never are.
44    #[tracing::instrument(
45        name = "query.select.plan",
46        skip_all,
47        fields(
48            cqlite.query.access_path = tracing::field::Empty,
49            cqlite.query.rows_scanned = tracing::field::Empty,
50            cqlite.query.rows = tracing::field::Empty,
51        )
52    )]
53    pub async fn execute(&self, plan: OptimizedQueryPlan) -> Result<QueryResult> {
54        // Issue #960: clear the global access-path probe so a stale value from a
55        // previous query cannot satisfy a test assertion against this one.
56        crate::query::access_path::reset();
57
58        let table_id = if let Some(ref from_clause) = plan.statement.from_clause {
59            self.extract_table_id(from_clause)?
60        } else {
61            // For queries without FROM clause (like SELECT 1), use a dummy table ID
62            TableId::new("_dummy_")
63        };
64
65        // Issue #4222 (design.md D6): a `<keyspace>.<table>_raw_sstable_data`
66        // FROM-clause reference is intercepted HERE, before schema resolution
67        // or any execution step runs, and routed entirely to the raw view's
68        // own producer — never through `StorageEngine::scan`/`scan_partition`,
69        // which RECONCILE across generations. `raw_view_base_name` prefers a
70        // REAL table literally named with the suffix, if one is registered
71        // (roborev finding: unconditional stripping made such a table
72        // unreachable) — only when no literal schema resolves does the
73        // suffix count as the naming convention (D1).
74        if let Some(base_name) = self.raw_view_base_name(&table_id).await {
75            return self
76                .execute_raw_sstable_view(&plan, &table_id, &base_name)
77                .await;
78        }
79
80        // Issue #1587 (E5): resolve the table's schema ONCE per query into a shared
81        // `Arc<TableSchema>`. Column-metadata building, the SSTable scan, and the
82        // SELECT-* metadata fallback all borrow this same schema (ref-count bump),
83        // instead of each independently re-locking the registry and deep-cloning a
84        // fresh `TableSchema` (2–4 deep clones per query before this).
85        let query_schema: Option<Arc<TableSchema>> = if plan.statement.from_clause.is_some() {
86            self.resolve_table_schema(&table_id).await
87        } else {
88            None
89        };
90
91        // Issue #692: detect whether any WRITETIME/TTL select items are present
92        // during planning and set the opt-in flag so the reader threads per-cell
93        // metadata. This is the "planning" half of the executor wiring; the
94        // "evaluation" half lives in `evaluate_select_expression`.
95        let projection_flags = ProjectionFlags {
96            include_cell_metadata: select_has_writetime_ttl(&plan.statement),
97        };
98        tracing::debug!(
99            "Query plan: include_cell_metadata={}",
100            projection_flags.include_cell_metadata
101        );
102
103        let mut context = ExecutionContext {
104            table_id,
105            columns: self.get_result_columns(&plan.statement, query_schema.as_deref())?,
106            rows_processed: 0,
107            scan_rows: 0,
108            projection_flags,
109            access_path: None,
110            reverse_served: false,
111        };
112
113        // Handle queries without FROM clause (like SELECT 1)
114        if plan.statement.from_clause.is_none() {
115            let mut result = self.execute_constant_query(&plan.statement, &context)?;
116            // Issue #1582 (roborev): apply the statement's LIMIT/OFFSET to the
117            // constant rows BEFORE the byte + row-count budget check, so the budget
118            // is enforced on the rows ACTUALLY returned (post LIMIT/OFFSET) —
119            // consistent with the table-backed path below. In particular `LIMIT 0`
120            // must return empty, never `ResultTooLarge`; an over-budget constant
121            // SELECT with NO limit still trips the guard on its final rows.
122            let offset = plan.statement.offset.unwrap_or(0) as usize;
123            let limit = plan
124                .statement
125                .limit
126                .as_ref()
127                .map(|l| l.count as usize)
128                .unwrap_or(usize::MAX);
129            result.rows = result.rows.into_iter().skip(offset).take(limit).collect();
130            // Keep the row-count metadata consistent with the returned rows.
131            let returned = result.rows.len() as u64;
132            result.rows_affected = returned;
133            result.metadata.total_rows = Some(returned);
134            enforce_materialized_rows(&result.rows, self.max_result_bytes, self.max_result_rows)?;
135            return Ok(result);
136        }
137
138        // Execute the plan step by step
139        let mut intermediate_results = Vec::new();
140
141        // If no execution steps are provided, add a default table scan
142        let execution_steps = if plan.execution_steps.is_empty() {
143            vec![ExecutionStep::SSTableScan {
144                table: context.table_id.clone(),
145                predicates: vec![],
146                projection: context.columns.iter().map(|c| c.name.clone()).collect(),
147            }]
148        } else {
149            plan.execution_steps.clone()
150        };
151
152        // Issue #1577 (D1): decide ONCE — from the plan alone — whether the
153        // SSTable scan may stop early at `LIMIT + OFFSET` accepted rows. `None`
154        // (any Sort/Aggregate/PerPartitionLimit/residual-Filter, DISTINCT, or no
155        // LIMIT) leaves the scan unbounded, exactly as before. Computed here so it
156        // is available inside the step loop below (unused, harmlessly, when the
157        // #1578 global-aggregate fast path takes the scan's place).
158        let scan_cap = scan_pushdown_cap(&execution_steps, &plan.statement.select_clause);
159
160        // Issue #1578 (D2): fold a GROUP-BY-free aggregate over a full table scan
161        // into an O(1) accumulator instead of buffering the whole table here.
162        // Returns None for any plan shape it does not model (GROUP BY, targeted
163        // lookup, WRITETIME/TTL, or a Sort/Project/Limit step) → the buffered step
164        // loop below runs unchanged.
165        if let Some(rows) = self
166            .try_execute_global_aggregate(&execution_steps, query_schema.as_deref(), &mut context)
167            .await?
168        {
169            intermediate_results = rows;
170        } else {
171            for step in &execution_steps {
172                match step {
173                    ExecutionStep::SSTableScan {
174                        table,
175                        predicates,
176                        projection,
177                        ..
178                    } => {
179                        let rows = self
180                            .execute_sstable_scan(
181                                table,
182                                predicates,
183                                projection,
184                                plan.statement.order_by.as_ref(),
185                                query_schema.as_deref(),
186                                scan_cap,
187                                &mut context,
188                            )
189                            .await?;
190                        intermediate_results = rows;
191                    }
192                    ExecutionStep::Filter { expression, .. } => {
193                        intermediate_results =
194                            self.execute_filter(intermediate_results, expression, &mut context)?;
195                    }
196                    ExecutionStep::Sort { order_by, .. } => {
197                        // Issue #1184: when the BIG reverse partition iterator already
198                        // produced the rows in descending clustering order, skip the
199                        // in-memory sort entirely (it remains the fallback otherwise).
200                        if !context.reverse_served {
201                            intermediate_results =
202                                self.execute_sort(intermediate_results, order_by, &mut context)?;
203                        } else {
204                            // Issue #1307 (hardening): skipping this Sort is sound ONLY
205                            // because `reverse_served` is set exclusively by
206                            // `targeted_partition_rows`, which serves the reverse
207                            // promoted-index iterator precisely when `statement.order_by`
208                            // requests the reverse of the stored clustering order — and
209                            // the planner emits EXACTLY ONE `Sort` step, cloned from that
210                            // same `statement.order_by` (see `select_optimizer.rs`). So
211                            // the reverse scan's ordering matches this step's `order_by`,
212                            // and skipping it drops a redundant sort rather than a
213                            // different ordering. What would break the invariant: a
214                            // multi-table / join plan (or any plan) that reused the flag
215                            // across a Sort NOT derived from the reverse-served scan's
216                            // `statement.order_by` — such a plan must clear
217                            // `reverse_served` before this step. The debug_assert pins
218                            // the property (the skipped Sort's key equals the statement's
219                            // order_by); it is debug-only and never alters release
220                            // behavior.
221                            debug_assert!(
222                                plan.statement.order_by.as_ref() == Some(order_by),
223                                "reverse_served Sort-skip invariant violated: the skipped \
224                             Sort's order_by must be the statement's order_by that drove \
225                             the reverse-served scan (single-table plan); a plan whose \
226                             Sort is not the reverse scan's matching Sort must clear \
227                             reverse_served first",
228                            );
229                        }
230                    }
231                    ExecutionStep::Aggregate { plan: agg_plan, .. } => {
232                        intermediate_results = self.execute_aggregation(
233                            intermediate_results,
234                            agg_plan,
235                            query_schema.as_deref(),
236                            &mut context,
237                        )?;
238                    }
239                    ExecutionStep::PerPartitionLimit { count } => {
240                        intermediate_results =
241                            Self::execute_per_partition_limit(intermediate_results, *count);
242                    }
243                    ExecutionStep::Limit { count, offset } => {
244                        intermediate_results = self.execute_limit(
245                            intermediate_results,
246                            *count,
247                            *offset,
248                            &mut context,
249                        )?;
250                    }
251                    ExecutionStep::Project { columns } => {
252                        // Issue #1952 (round-6 fix): branch on whether the projection
253                        // RESHAPES the row. A plain-column Project only trims the
254                        // #1952-widened helper columns — route it through the
255                        // key-preserving `trim_projection` so the row keeps its real
256                        // RowKey / metadata and a sparse row's absent selected cell is
257                        // omitted rather than erroring. Only a reshaping / computed
258                        // projection (alias, arithmetic, aggregate, function,
259                        // writetime/ttl, collection-access) goes through
260                        // `execute_projection`, whose empty-RowKey + name-derivation is
261                        // correct for a computed row that has no natural stored key.
262                        intermediate_results = if columns.iter().any(project_expr_reshapes_row) {
263                            self.execute_projection(intermediate_results, columns, &mut context)?
264                        } else {
265                            self.trim_projection(intermediate_results, columns)
266                        };
267                    }
268                }
269            }
270        }
271
272        // Issue #1582 (D6, narrow subset): enforce the byte-bounded result budget
273        // (primary) + the row-count safety valve (secondary) with a SINGLE robust
274        // check on the FINAL materialized result — AFTER every execution step
275        // (Limit/Offset/Filter/Sort/Aggregate/Project) has produced the rows that
276        // will actually be returned. Because this sees ONLY the returned rows
277        // (post LIMIT/OFFSET), a `LIMIT 10` query never trips and OFFSET-skipped
278        // rows are never charged. This replaces the earlier during-collection
279        // machinery (LIMIT/OFFSET pushdown, per-row early-stop, storage-layer row limit),
280        // which kept generating correctness edge cases; a single final-result
281        // check has no such edges. Reuses the shared `estimate_value_size`
282        // estimator (via `enforce_materialized_rows`).
283        //
284        // SCOPE (owner-accepted boundaries, tracked on #1582 — NOT bugs to fix in
285        // this narrow subset):
286        //   * Does NOT bound PEAK scan memory: `storage.scan` still materializes
287        //     each reader's matching rows before this point (deferred to #1897).
288        //   * Does NOT cover the LEGACY point-lookup `QueryExecutor` path
289        //     (`WHERE id = ?` short lookups route there, not through this modern
290        //     executor) — deferred to the D6 redesign.
291        // Issue #1578 (D2): demote the row-count valve to a genuine safety valve.
292        // A query with an EXPLICIT `LIMIT` already bounds its own result, so it is
293        // exempt from the crude row-count ceiling (the user accepted the count);
294        // the byte budget still guards memory. Without an explicit LIMIT the valve
295        // remains a real net against unbounded materialization.
296        let effective_max_rows = if plan.statement.limit.is_some() {
297            usize::MAX
298        } else {
299            self.max_result_rows
300        };
301        enforce_materialized_rows(
302            &intermediate_results,
303            self.max_result_bytes,
304            effective_max_rows,
305        )?;
306
307        let total_rows = intermediate_results.len() as u64;
308
309        // CRITICAL FIX (Issue #129/#140): Populate metadata.columns for SELECT *
310        // When SELECT * is used and no schema was found, context.columns is empty.
311        // Fall back to inferring column names from the first row's HashMap keys.
312        // IMPORTANT: Must be sorted alphabetically for deterministic JSON output (Issue #129)!
313        let mut columns = context.columns;
314        if columns.is_empty() && !intermediate_results.is_empty() {
315            // Issue #1587 (E5): reuse the schema resolved once at the top of the
316            // query rather than re-locking the registry + deep-cloning here.
317            let schema_opt = query_schema.as_deref();
318
319            let first_row = &intermediate_results[0];
320            let mut col_names: Vec<_> = first_row.values.keys().collect();
321            col_names.sort(); // Sort alphabetically for deterministic ordering (Issue #129)
322
323            let table_name_for_meta = schema_opt.map(|s| format!("{}.{}", s.keyspace, s.table));
324
325            for (idx, col_name) in col_names.iter().enumerate() {
326                let col_name: &str = col_name;
327                // Look up CQL type from schema; derive flat DataType from it (Issue #674).
328                let col_info = match schema_opt
329                    .and_then(|schema| schema.columns.iter().find(|c| c.name.as_str() == col_name))
330                {
331                    Some(schema_col) => column_info_from_type_str(
332                        col_name.to_string(),
333                        &schema_col.data_type,
334                        idx,
335                        table_name_for_meta.clone(),
336                    ),
337                    None => ColumnInfo {
338                        name: col_name.to_string(),
339                        data_type: crate::types::DataType::Text,
340                        nullable: true,
341                        position: idx,
342                        table_name: table_name_for_meta.clone(),
343                        cql_type: None,
344                    },
345                };
346                columns.push(col_info);
347            }
348        }
349
350        // Observability (issue #1035): the `query.select.plan` span declared
351        // `access_path`/`rows_scanned`/`rows` but never recorded them, and
352        // `QUERY_ROWS_SCANNED` was never emitted. Do both here, sourced from the
353        // honest per-query signal (`context.access_path`, set by the SSTable-scan
354        // step) and the rows the scan examined (`context.rows_processed`). Bounded
355        // attributes only — never the query text or key values.
356        //
357        // Issue #2162: this stays a SINGLE-SHOT emission deliberately. The Flight
358        // `do_get` merge/scan path gained incremental `QUERY_ROWS_SCANNED` deltas
359        // (`cqlite-flight/src/scan_progress.rs`), but that path never runs through
360        // `SelectExecutor` — Flight drives `KWayMerger` directly. Making THIS site
361        // incremental too would additionally have to interact with
362        // `limit_pushdown`'s `scan_rows` rebaseline/decode-stop accounting
363        // (`limit_pushdown/mod.rs`), which re-runs a capped scan and resets
364        // `context.scan_rows` — a materially riskier change with no scenario in
365        // #2162's spec requiring it.
366        {
367            use crate::observability::{self as obs, catalog, AttrValue};
368
369            let access_path_label: &'static str = context
370                .access_path
371                .as_ref()
372                .map(|p| p.label())
373                .unwrap_or("unknown");
374
375            obs::add_counter(
376                catalog::QUERY_ROWS_SCANNED,
377                context.scan_rows,
378                &[(
379                    catalog::attr::ACCESS_PATH,
380                    AttrValue::StaticStr(access_path_label),
381                )],
382            );
383
384            let span = tracing::Span::current();
385            span.record(catalog::attr::ACCESS_PATH, access_path_label);
386            span.record("cqlite.query.rows_scanned", context.scan_rows);
387            span.record("cqlite.query.rows", total_rows);
388        }
389
390        // Issue #1035: carry a bounded plan family on the result so the engine's
391        // single observability chokepoint reports a real plan type for SELECTs
392        // (the modern executor previously always returned `plan_info: None`,
393        // forcing plan_type to "unknown").
394        let plan_info = Self::select_plan_info(&plan, context.access_path.as_ref());
395
396        Ok(QueryResult {
397            rows: intermediate_results,
398            rows_affected: total_rows, // Use actual number of rows returned
399            execution_time_ms: 0,      // Will be set by the engine
400            metadata: crate::query::result::QueryMetadata {
401                columns,
402                total_rows: Some(total_rows),
403                plan_info: Some(plan_info),
404                performance: Default::default(),
405                warnings: vec![],
406                // Issue #960: surface the access path the SSTable-scan step chose
407                // on the result from PER-QUERY state (not the global probe), so a
408                // concurrent SELECT cannot overwrite it between record() and here.
409                access_path: context.access_path.clone(),
410            },
411        })
412    }

414    /// Execute SSTable scan with predicate pushdown.
415    ///
416    /// Per-row work (build row, decode partition key, evaluate predicates) is
417    /// handled by the free helpers `build_row_from_scan` and
418    /// `evaluate_predicates`, which are shared with the streaming background
419    /// task to keep the two execution paths in lockstep.
420    ///
421    /// Issue #1582 (D6, narrow subset): the single byte/row budget check is
422    /// applied ONCE by [`SelectExecutor::execute`] on the FINAL result, after the
423    /// whole step pipeline (post LIMIT/OFFSET), never mid-collection here.
424    ///
425    /// Issue #1577 (D1): when the caller determines the plan is LIMIT-pushdown
426    /// safe it passes `scan_cap = Some(limit + offset)` — the number of ACCEPTED
427    /// (post-marker, post-predicate) rows the downstream `Limit` needs. This scan
428    /// then stops after that many accepted rows: the full-scan fallback stops
429    /// DECODING early (via [`capped_fallback_scan`](Self::capped_fallback_scan)'s
430    /// bounded stream), and the partition-targeted paths stop the per-row build
431    /// loop. `scan_cap` is `None` (unbounded, as before) whenever a
432    /// Sort/Aggregate/PerPartitionLimit/residual-Filter step or DISTINCT follows,
433    /// or there is no `LIMIT` — so a `WHERE non_pk = ? LIMIT N` still counts
434    /// ACCEPTED rows, never raw rows, and can never silently drop matches.
435    #[cfg_attr(feature = "tombstones", allow(unused_variables))]
436    pub(super) async fn execute_sstable_scan(
437        &self,
438        table: &TableId,
439        predicates: &[SSTablePredicate],
440        projection: &[String],
441        order_by: Option<&crate::query::select_ast::OrderByClause>,
442        // Issue #1587 (E5): schema resolved ONCE per query by the caller and
443        // shared by reference — no per-scan registry lock + deep clone.
444        schema_opt: Option<&TableSchema>,
445        // Issue #1577 (D1): `Some(limit + offset)` when LIMIT pushdown is safe.
446        scan_cap: Option<usize>,
447        context: &mut ExecutionContext,
448    ) -> Result<Vec<QueryRow>> {
449        // FINDING 2 (Issue #955 follow-up): a `token(...)` predicate is evaluated
450        // by hashing the row's raw partition key, so its argument columns MUST be
451        // the full partition key in declared order or the result is silently
452        // wrong. Reject (Cassandra-style) before scanning/evaluating.
453        validate_token_predicates(predicates, schema_opt)?;
454
455        // Issue #1918: resolve the read-path forcing mode ONCE for this scan step
456        // (config over env over auto). An invalid env value fails the query loudly
457        // here rather than silently running under auto.
458        let mode = self.resolved_read_path_mode()?;
459
460        // Data-safety (issue #1694): log the SHAPE of the scan — predicate count
461        // and the constrained column names — never the predicate literals/values.
462        tracing::debug!(
463            "Executing SSTableScan: table=\"{}\", predicates={} on [{}], include_cell_metadata={}",
464            table,
465            predicates.len(),
466            predicates
467                .iter()
468                .map(|p| p.column.as_str())
469                .collect::<Vec<_>>()
470                .join(", "),
471            context.projection_flags.include_cell_metadata,
472        );
473
474        let (keyspace, table_name) = parse_table_id(table);
475
476        match schema_opt {
477            Some(schema) => tracing::debug!(
478                "Found schema for {}.{} with {} columns",
479                schema.keyspace,
480                schema.table,
481                schema.columns.len()
482            ),
483            None => tracing::debug!(
484                "No schema found for {}.{}, proceeding without schema-aware parsing",
485                keyspace.as_deref().unwrap_or("unknown"),
486                table_name
487            ),
488        }
489
490        // Issue #1750: a SCHEMA-LESS `WHERE pk = <literal>` point read can be served
491        // by a key-byte-targeted seek ONLY when the equality column is a
492        // metadata-CONFIRMED sole partition key (single-component pk, no clustering
493        // keys, column absent from the authoritative non-key column names). Resolve
494        // that authoritative shape from the Statistics.db SerializationHeader (the
495        // metadata a schema-less reader HAS) and classify BY ELIMINATION — never a
496        // pk-name/text guess (#28). `None` (metadata unavailable, or the column is a
497        // non-pk column) keeps the honest full-scan path, which correctly matches
498        // regular-column equalities. Resolved only when there is no CQL schema and no
499        // WRITETIME/TTL metadata projection (the two branches below own those).
500        // Issue #1918: `point`/`auto` keep the schema-less sole-pk targeted seek
501        // (#1750). Under forced `full` the query cannot take the general full-scan
502        // path for THIS shape: with no schema the per-row predicate backstop in
503        // `collect_capped_materialized` cannot reconstruct the pk column to match
504        // the literal, so a full scan would silently return 0 rows instead of the
505        // row `auto` returns — violating the spec's "identical to `auto`" SHALL.
506        // Fail closed with a clear error (mirror of `point_forbids_fallback`: full
507        // forbidding a shape only the specialized non-full seek can serve).
508        let schemaless_seek: Option<super::schemaless_point::SchemalessPointSeek> =
509            if schema_opt.is_none() && !context.projection_flags.include_cell_metadata {
510                let shape = self.storage.partition_key_shape(table).await;
511                let seek = classify_schemaless_point_lookup(predicates, shape.as_ref());
512                full_forbids_schemaless_seek(mode, seek.is_some())?;
513                // Under `auto`/`point` keep the seek; under `full` `seek` is `None`
514                // here only when the query did not qualify (the qualifying case
515                // errored above), so it correctly falls through to the full scan.
516                if mode == crate::config::ReadPathMode::Full {
517                    None
518                } else {
519                    seek
520                }
521            } else {
522                None
523            };
524
525        // Issue #693: When WRITETIME(col) or TTL(col) is in the SELECT, use the
526        // metadata-carrying scan so per-cell timestamps reach the QueryRow.
527        let results = if context.projection_flags.include_cell_metadata {
528            // Issue #962: route a fully-constrained `WHERE pk = ?` WRITETIME/TTL
529            // projection through a partition-targeted metadata lookup that prunes
530            // SSTables (bloom/BTI) before decoding, instead of full-scanning every
531            // SSTable for the table. Reuses the SAME `classify_partition_lookup`
532            // decision the non-metadata path uses (the shared resolved
533            // partition-lookup representation). The per-row predicate evaluation
534            // below is unchanged, so the pk equality itself is still applied as a
535            // correctness backstop and any bloom/BTI over-inclusion is filtered out.
536            // Issue #1918: the single forcing gate wraps the classifier outcome.
537            let outcome = classify_partition_lookup(predicates, schema_opt);
538            let scan_results = match apply_forcing(outcome, mode)? {
539                // Forced `full`: run the same full metadata scan the organic
540                // fallback uses, recorded with the distinct forced reason.
541                ForcedPlan::ForceFullScan => {
542                    let path = AccessPath::FallbackFullScan {
543                        reason: FallbackReason::ForcedFullScan,
544                    };
545                    context.access_path = Some(path.clone());
546                    crate::query::access_path::record(path);
547                    self.storage
548                        .scan_with_cell_metadata(table, None, None, None, schema_opt)
549                        .await?
550                }
551                ForcedPlan::Proceed(PartitionLookupOutcome::Targeted(pk_bytes)) => {
552                    tracing::debug!(
553                        "SSTableScan(metadata): partition-key point lookup (key len={}) for \"{}\"",
554                        pk_bytes.len(),
555                        table
556                    );
557                    // Epic #951 (honest paths): the `tombstones` build's metadata
558                    // lookup is a full metadata scan + retain with NO prune,
559                    // reported via `engaged == false`; claim
560                    // `MetadataPartitionLookup` only when it really pruned, else
561                    // report the honest `TombstonesBuildNoPrune` fallback (the
562                    // rows are byte-identical either way).
563                    let (rows, engaged) = self
564                        .storage
565                        .scan_partition_with_cell_metadata(table, &pk_bytes, schema_opt)
566                        .await?;
567                    // Issue #1918: under `point` a post-call no-prune (`engaged ==
568                    // false`, tombstones build) is a non-targeted execution → fail
569                    // closed rather than silently full-scanning.
570                    point_requires_engaged(mode, engaged, FallbackReason::TombstonesBuildNoPrune)?;
571                    let path = honest_targeted_path(AccessPath::MetadataPartitionLookup, engaged);
572                    context.access_path = Some(path.clone());
573                    crate::query::access_path::record(path);
574                    rows
575                }
576                // Issue #1916: `WHERE pk IN (...)` on the metadata path is the union
577                // of N independent partition-targeted metadata lookups, each of which
578                // prunes SSTables (bloom/BTI) before decoding — mirroring the plain
579                // `MultiTargeted` fan-out (see the non-metadata arm below). #962
580                // shipped the single-key metadata fast path and left this fan-out as a
581                // documented follow-up; this arm is that follow-up. Each per-key call
582                // goes through the SAME single/multi-generation reconciliation the
583                // single-key `Targeted` arm uses (#1741), so the merged WRITETIME/TTL
584                // values are byte-identical to the old full-scan result.
585                //
586                // Epic #951 (honest paths): on the `tombstones` build each lookup
587                // full-scans + retains with NO prune (`engaged == false`); report
588                // `MultiPartitionLookup` only when the lookups actually pruned, else
589                // the honest `TombstonesBuildNoPrune` fallback. Rows are unchanged.
590                ForcedPlan::Proceed(PartitionLookupOutcome::MultiTargeted(pk_keys)) => {
591                    tracing::debug!(
592                        "SSTableScan(metadata): multi-partition lookup ({} keys) for \"{}\"",
593                        pk_keys.len(),
594                        table
595                    );
596                    let mut combined = Vec::new();
597                    let mut all_engaged = true;
598                    for pk_bytes in &pk_keys {
599                        let (rows, engaged) = self
600                            .storage
601                            .scan_partition_with_cell_metadata(table, pk_bytes, schema_opt)
602                            .await?;
603                        all_engaged &= engaged;
604                        combined.extend(rows);
605                    }
606                    // Issue #1918: `point` fails closed when the fan-out did not prune
607                    // (tombstones build) rather than silently full-scanning.
608                    point_requires_engaged(
609                        mode,
610                        all_engaged,
611                        FallbackReason::TombstonesBuildNoPrune,
612                    )?;
613                    let path = honest_targeted_path(AccessPath::MultiPartitionLookup, all_engaged);
614                    context.access_path = Some(path.clone());
615                    crate::query::access_path::record(path);
616                    // Order the union with the IDENTICAL token-then-raw-bytes rule the
617                    // plain `MultiTargeted` arm uses (`sort_rows_by_token`), so the
618                    // metadata IN result equals a full scan filtered to these keys.
619                    sort_metadata_rows_by_token(&mut combined);
620                    combined
621                }
622                // A genuinely unclassifiable metadata projection (no usable
623                // restriction) still full-scans; report that honestly
624                // (MetadataScanPath) rather than faking a targeted path.
625                // Issue #1918: under `point` a classification `Fallback` already
626                // errored up front in `apply_forcing`, so this arm is reached only
627                // under `auto`/`full`; `point_forbids_fallback` keeps the fail-closed
628                // contract explicit here.
629                ForcedPlan::Proceed(PartitionLookupOutcome::Fallback(_)) => {
630                    point_forbids_fallback(mode, FallbackReason::MetadataScanPath)?;
631                    let metadata_path = AccessPath::FallbackFullScan {
632                        reason: FallbackReason::MetadataScanPath,
633                    };
634                    context.access_path = Some(metadata_path.clone());
635                    crate::query::access_path::record(metadata_path);
636                    self.storage
637                        .scan_with_cell_metadata(table, None, None, None, schema_opt)
638                        .await?
639                }
640            };
641
642            tracing::debug!("Scan (with metadata) returned {} rows", scan_results.len());
643
644            // Issue #1577 (D1 + roborev metric-accounting fix): this metadata scan
645            // is ALREADY fully materialized — the storage layer decoded every row
646            // before returning it. `collect_capped_materialized` charges
647            // `context.scan_rows` (→ `QUERY_ROWS_SCANNED`) with the TRUE decoded
648            // count up front, so the metric reflects real scan work, while the
649            // LIMIT/OFFSET `scan_cap` only bounds the per-row build/predicate work.
650            // Counting ACCEPTED rows toward the cap never drops a matching row.
651            // Issue #1817: hoist the partition-key decode across the rows of a
652            // partition (the scan yields a partition's rows consecutively).
653            let mut pk_cache = super::PartitionKeyCache::default();
654            collect_capped_materialized(
655                scan_results,
656                scan_cap,
657                predicates,
658                context,
659                |(key, value, cell_meta)| {
660                    let mut row = build_row_from_scan_cached(
661                        key,
662                        value,
663                        projection,
664                        schema_opt,
665                        &mut pk_cache,
666                    )?;
667                    // Attach per-cell metadata so evaluate_writetime_ttl can read it.
668                    if !cell_meta.is_empty() {
669                        row.set_cell_metadata(cell_meta);
670                    }
671                    Some(row)
672                },
673            )?
674        } else if let Some(seek) = schemaless_seek {
675            // Issue #1750 (regression fix, re-scoped): a SCHEMA-LESS `WHERE pk =
676            // <literal>` point read whose equality column is a metadata-CONFIRMED
677            // sole partition key. With no schema the full-scan path CANNOT
678            // reconstruct the partition-key column, so the shared per-row
679            // `evaluate_predicates` backstop below would reject EVERY row on the pk
680            // equality and return 0 rows — the regression from rerouting this read
681            // off the legacy `QueryExecutor` (which looked up by key bytes, never
682            // re-evaluating the predicate). Serve it here by the SAME
683            // key-byte-targeted seek the schema-aware Targeted arm uses
684            // (`scan_partition`), which is self-verifying: it returns ONLY rows whose
685            // raw partition key equals `seek.bytes`, so a wrong/absent key yields
686            // nothing (never another partition's rows). The key is encoded through
687            // the TYPED single-component codec for the pk's AUTHORITATIVE type (from
688            // the Statistics.db SerializationHeader `keyType`), so a parsed integer
689            // literal (`Value::BigInt`) builds the width Cassandra wrote — an `int`
690            // pk's 4-byte key, not an 8-byte one (roborev 3784 FINDING 2). The
691            // decision is made from AUTHORITATIVE metadata by ELIMINATION — never a
692            // pk-name/substring/token-count text guess (#28). A `WHERE <regular_col>
693            // = <literal>` does NOT reach here (the classifier returns `None` because
694            // the column is one of the authoritative non-key names), so it keeps the
695            // honest full-scan path and correctly matches the regular-column cell.
696            //
697            // Post-seek guard (roborev 3784 FINDING 1): the classifier admits the
698            // predicate column by ELIMINATION, which also admits a
699            // nonexistent/misspelled column whose literal happens to encode to a real
700            // partition's key. `finalize_schemaless_seek_row` reconstructs the sole
701            // pk column under its authoritative type/name and RE-EVALUATES the
702            // predicate, so such a column yields `Unknown` and rejects the row
703            // (correct 0 rows) — the pk-name predicate still matches. The schema-less
704            // `SELECT *` column metadata still comes from the first row in `execute`.
705            let (rows, engaged) = self
706                .storage
707                .scan_partition(table, &seek.bytes, None)
708                .await?;
709            // Issue #1918: `point` fails closed on a post-call no-prune. (`full`
710            // never reaches here — the seek is skipped above under forced `full`.)
711            point_requires_engaged(mode, engaged, FallbackReason::TombstonesBuildNoPrune)?;
712            let path = honest_targeted_path(AccessPath::PartitionLookup, engaged);
713            context.access_path = Some(path.clone());
714            crate::query::access_path::record(path);
715            context.scan_rows += rows.len() as u64;
716            context.rows_processed += rows.len() as u64;
717            let mut out = Vec::with_capacity(rows.len());
718            for (key, value) in rows {
719                if let Some(row) = super::schemaless_point::finalize_schemaless_seek_row(
720                    key, value, projection, predicates, &seek,
721                ) {
722                    out.push(row);
723                }
724            }
725            out
726        } else {
727            // Issue #949: a fully-constrained `WHERE pk = ?` is served by a
728            // partition-targeted lookup that prunes SSTables via bloom/BTI and only
729            // parses the candidates, instead of scanning every SSTable for the
730            // table. Falls back to a full scan when the partition key isn't fully
731            // pinned or can't be encoded. The per-row predicate evaluation below is
732            // unchanged, so clustering predicates and the pk equality itself are
733            // still applied (and any over-inclusion is filtered out).
734            // Issue #1918: the single forcing gate wraps the classifier outcome.
735            let outcome = classify_partition_lookup(predicates, schema_opt);
736            let scan_results = match apply_forcing(outcome, mode)? {
737                // Forced `full`: run the SAME full-scan + reconciliation code the
738                // organic fallback uses (so rows/order match `auto`), recorded with
739                // the distinct `ForcedFullScan` reason so it is never mistaken for
740                // an organic fallback.
741                ForcedPlan::ForceFullScan => {
742                    let path = AccessPath::FallbackFullScan {
743                        reason: FallbackReason::ForcedFullScan,
744                    };
745                    context.access_path = Some(path.clone());
746                    crate::query::access_path::record(path);
747                    if let Some(cap) = scan_cap {
748                        return self
749                            .capped_fallback_scan(
750                                table, predicates, projection, schema_opt, cap, context,
751                            )
752                            .await;
753                    }
754                    self.storage
755                        .scan(table, None, None, None, schema_opt)
756                        .await?
757                }
758                ForcedPlan::Proceed(PartitionLookupOutcome::Targeted(pk_bytes)) => {
759                    tracing::debug!(
760                        "SSTableScan: partition-key point lookup (key len={}) for \"{}\"",
761                        pk_bytes.len(),
762                        table
763                    );
764                    // Issue #954: when a single-column clustering restriction is
765                    // present, push it down to a within-partition seek so a wide
766                    // partition's slice decodes O(matched rows + index), not the
767                    // whole partition. The seek reports whether the clustering
768                    // narrowing actually engaged; the per-row backstop below applies
769                    // the exact bound so output is byte-identical either way.
770                    //
771                    // Issue #960: report the HONEST access path — `ClusteringSlice`
772                    // only when the seek engaged, else `PartitionLookup`. The
773                    // clustering seek exists only on the default build; the
774                    // `tombstones` build uses the plain partition lookup.
775                    // Issue #954/#960/#1184: forward clustering-slice seek OR (for
776                    // `ORDER BY <ck>` reverse-of-stored) the BIG reverse iterator,
777                    // with the honest access path recorded inside the helper.
778                    #[cfg(not(feature = "tombstones"))]
779                    {
780                        self.targeted_partition_rows(
781                            table, &pk_bytes, predicates, order_by, schema_opt, context,
782                        )
783                        .await?
784                    }
785                    #[cfg(feature = "tombstones")]
786                    {
787                        // Epic #951 (honest paths): the `tombstones` build's
788                        // `scan_partition` is a full scan + retain with NO prune,
789                        // reported via `engaged == false`. Report the honest
790                        // fallback rather than a fake `PartitionLookup`; the rows
791                        // are byte-identical to the pruned build.
792                        let (rows, engaged) = self
793                            .storage
794                            .scan_partition(table, &pk_bytes, schema_opt)
795                            .await?;
796                        // Issue #1918: `point` fails closed on the tombstones-build
797                        // no-prune rather than silently full-scanning.
798                        point_requires_engaged(
799                            mode,
800                            engaged,
801                            FallbackReason::TombstonesBuildNoPrune,
802                        )?;
803                        let path = honest_targeted_path(AccessPath::PartitionLookup, engaged);
804                        context.access_path = Some(path.clone());
805                        crate::query::access_path::record(path);
806                        rows
807                    }
808                }
809                ForcedPlan::Proceed(PartitionLookupOutcome::MultiTargeted(pk_keys)) => {
810                    tracing::debug!(
811                        "SSTableScan: multi-partition lookup ({} keys) for \"{}\"",
812                        pk_keys.len(),
813                        table
814                    );
815                    // Issue #955/#960: `WHERE pk IN (...)` over the complete key
816                    // is the union of N independent partition-targeted lookups,
817                    // each of which prunes SSTables. Epic #951 (honest paths): on
818                    // the `tombstones` build each lookup full-scans + retains with
819                    // NO prune (`engaged == false`); report `MultiPartitionLookup`
820                    // only when the lookups actually pruned, else the honest
821                    // `TombstonesBuildNoPrune` fallback. Rows are unchanged.
822                    let mut combined = Vec::new();
823                    let mut all_engaged = true;
824                    for pk_bytes in &pk_keys {
825                        let (rows, engaged) = self
826                            .storage
827                            .scan_partition(table, pk_bytes, schema_opt)
828                            .await?;
829                        all_engaged &= engaged;
830                        combined.extend(rows);
831                    }
832                    // Issue #1918: `point` fails closed when the fan-out did not
833                    // prune (tombstones build) rather than silently full-scanning.
834                    point_requires_engaged(
835                        mode,
836                        all_engaged,
837                        FallbackReason::TombstonesBuildNoPrune,
838                    )?;
839                    let path = honest_targeted_path(AccessPath::MultiPartitionLookup, all_engaged);
840                    context.access_path = Some(path.clone());
841                    crate::query::access_path::record(path);
842                    // Order the union to equal a full scan filtered to these keys:
843                    // partitions are stored token-ordered, so sort the combined
844                    // rows by (partition token, raw key bytes). A *stable* sort
845                    // keeps each partition's clustering order (rows for one key
846                    // arrive contiguously from one `scan_partition`) intact.
847                    sort_rows_by_token(&mut combined);
848                    combined
849                }
850                ForcedPlan::Proceed(PartitionLookupOutcome::Fallback(reason)) => {
851                    // Issue #960: report the honest reason a full scan was chosen.
852                    // (Under `point` a `Fallback` already errored in `apply_forcing`,
853                    // so this arm is reached only under `auto`.)
854                    context.access_path = Some(AccessPath::FallbackFullScan { reason });
855                    crate::query::access_path::record(AccessPath::FallbackFullScan { reason });
856                    // Issue #1577 (D1): when the plan is LIMIT-pushdown safe, stop
857                    // DECODING the table once `cap` rows are accepted. This streams
858                    // (in lockstep with `scan`) and drops the stream at the cap,
859                    // returning the final rows directly — the shared per-row loop
860                    // below is bypassed for this branch.
861                    if let Some(cap) = scan_cap {
862                        return self
863                            .capped_fallback_scan(
864                                table, predicates, projection, schema_opt, cap, context,
865                            )
866                            .await;
867                    }
868                    // Issue #1582 (D6, narrow subset): unbounded full scan; the sole
869                    // budget check is applied once on the FINAL result in `execute`.
870                    self.storage
871                        .scan(table, None, None, None, schema_opt)
872                        .await?
873                }
874            };
875
876            tracing::debug!("Scan returned {} rows", scan_results.len());
877
878            // Issue #1577 (D1 + roborev metric-accounting fix): these results are
879            // ALREADY materialized — the partition-targeted paths decoded their
880            // (partition-bounded) rows, and the TRULY decode-bounded full-scan
881            // fallback already returned early above via `capped_fallback_scan`.
882            // `collect_capped_materialized` charges `context.scan_rows` with the
883            // full decoded count up front (so `QUERY_ROWS_SCANNED` reflects the
884            // real scan), while the `scan_cap` only bounds per-row build work.
885            // Counting ACCEPTED rows toward the cap (build_row_from_scan returns
886            // None for tombstoned/null rows, Issue #191) never drops a match.
887            // Issue #1817: hoist the partition-key decode across a partition's rows.
888            let mut pk_cache = super::PartitionKeyCache::default();
889            collect_capped_materialized(
890                scan_results,
891                scan_cap,
892                predicates,
893                context,
894                |(key, value)| {
895                    build_row_from_scan_cached(key, value, projection, schema_opt, &mut pk_cache)
896                },
897            )?
898        };
899
900        Ok(results)
901    }
