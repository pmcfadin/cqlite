11/// Suffix that marks a table reference as the raw-SSTable-view name for its
12/// base table (design.md D1 — a naming convention over the existing flat
13/// `keyspace.table` namespace, never a new grammar). The single source of
14/// truth lives in [`crate::query::raw_view_naming`] — shared with
15/// `Database::has_schema_for_table`/`schema_status`, so the CLI's pre-flight
16/// schema check (issue #199) recognizes a raw-view name too, instead of
17/// rejecting every raw-view query before `SelectExecutor` can intercept it.
18pub(in crate::query::select_executor) use crate::query::raw_view_naming::RAW_SSTABLE_VIEW_SUFFIX as RAW_VIEW_SUFFIX;

20/// `Some(true)` for a CQL type whose cells are individually addressable (a
21/// non-frozen list/set/map/UDT) — the column gets ONLY the
22/// `_complex_deletion` trio (design.md D7), never the single-cell
23/// `_timestamp`/`_ttl`/`_local_deletion_time`/`_tombstone` quad: a collection
24/// has no ONE cell timestamp to report, and declaring those four columns for
25/// a complex type would always render NULL — indistinguishable from "no
26/// timestamp exists" (roborev finding, issue #4222). `Frozen(..)`
27/// collections/UDTs are single-cell and excluded on purpose: Cassandra never
28/// gives them a per-column complex-deletion marker, so they keep the plain
29/// quad instead.
30///
31/// `None` — the caller must FAIL CLOSED, never default to "simple" — for
32/// ANY `CqlType::Custom(_)`, not just a non-`udt:`-prefixed one (roborev
33/// finding, issue #4222 — round 5, correcting the FIRST fix's own gap): a
34/// bare (non-frozen) UDT name parses to `Custom("udt:<name>")` via
35/// `cql_type_parser.rs`'s explicit UDT-shaped branch — matched below — but
36/// an all-LOWERCASE, non-primitive type name ALSO reaches `Custom(_)`, via
37/// that same parser's final fallback arm (`cql_type_parser.rs:274`), with
38/// NO prefix at all. Cassandra identifiers default to lowercase unless
39/// quoted, so a real UDT named e.g. `address` or `person` parses to exactly
40/// `Custom("address")` — indistinguishable, at this call site with no UDT
41/// registry to consult, from a genuinely unrecognized custom type.
42/// Defaulting the non-`udt:`-prefixed case to "simple" (the FIRST fix's
43/// remaining gap) therefore silently misclassified every real
44/// lowercase-named UDT column as single-cell, assigning it the WRONG
45/// metadata shape rather than refusing to guess (issue #28 no-heuristics).
46fn try_is_complex_cql_type(t: &CqlType) -> Option<bool> {
47    match t {
48        CqlType::List(_) | CqlType::Set(_) | CqlType::Map(_, _) | CqlType::Udt(_, _) => Some(true),
49        CqlType::Custom(_) => None,
50        _ => Some(false),
51    }
52}

54/// Build the raw view's full column contract (design.md D7) from the BASE
55/// table's schema — a pinned public surface (spec's column-snapshot
56/// requirement).
57///
58/// Order: partition-key columns, clustering-key columns (plain data columns:
59/// every physical row, including a range-tombstone bound row, carries them),
60/// then per-non-key-column metadata (the base value plus, for a simple
61/// column, its `_timestamp`/`_ttl`/`_local_deletion_time`/`_tombstone` quad,
62/// or for a collection/UDT column, its `_complex_deletion` trio only), then
63/// the row-level, partition-level, row-kind/range-tombstone, and source
64/// columns.
65///
66/// Fails closed (design.md D8) with `Error::Schema` when a synthesized
67/// metadata-column name COLLIDES with a real base-table column name (e.g. a
68/// base table with its own `generation`/`sstable`/`row_kind` column, or a
69/// `val` column alongside a `val_timestamp` column) — silently overwriting
70/// one of the two would be a no-heuristics-violating silent data loss
71/// (roborev finding, issue #4222), never something this view may do quietly.
72///
73/// Returns the synthesized column set ALONGSIDE the set of names that are
74/// METADATA DERIVATIVES (the per-cell `_timestamp`/`_ttl`/
75/// `_local_deletion_time`/`_tombstone`/`_complex_deletion*` names and the
76/// row-level `row_timestamp`/`row_ttl`/`row_liveness_expires_at`/
77/// `row_local_deletion_time`/`row_tombstone`/`row_deletion_timestamp` set) — roborev finding,
78/// issue #4222 — round 9, correcting round 8's own gap: that fix classified
79/// a predicate's column by NAME SUFFIX (`predicates.rs`'s
80/// `is_metadata_derivative_column`), which misclassifies a REAL base-table
81/// column that happens to be named e.g. `event_timestamp` with no sibling
82/// `event` column to trip the collision guard (which only fires when a
83/// SIBLING column's synthesized name collides, not when an UNRELATED base
84/// column merely LOOKS like one). Returning the set THIS FUNCTION ACTUALLY
85/// SYNTHESIZED — never re-derived from a name pattern — makes that
86/// misclassification structurally impossible: a real base column can never
87/// end up in this set, because it is populated only at the exact call
88/// sites that push a synthesized quad/trio/row-level metadata name.
89pub(in crate::query::select_executor) fn raw_view_columns(
90    base: &TableSchema,
91) -> crate::Result<(Vec<ColumnInfo>, HashSet<String>)> {
92    let key_names: HashSet<&str> = base
93        .partition_keys
94        .iter()
95        .map(|k| k.name.as_str())
96        .chain(base.clustering_keys.iter().map(|k| k.name.as_str()))
97        .collect();
98
99    let mut columns: Vec<ColumnInfo> = Vec::new();
100    let mut metadata_names: HashSet<String> = HashSet::new();
101    let mut seen: HashSet<String> = HashSet::new();
102    let mut push = |columns: &mut Vec<ColumnInfo>,
103                    metadata_names: &mut HashSet<String>,
104                    name: String,
105                    type_str: &str,
106                    is_metadata: bool| {
107        if !seen.insert(name.clone()) {
108            return Err(Error::Schema(format!(
109                "raw SSTable view: base table '{}.{}' has a column named '{name}' that \
110                 collides with a synthesized metadata column name — the raw view's column \
111                 contract (design.md D7) cannot be built without silently shadowing one of \
112                 the two",
113                base.keyspace, base.table
114            )));
115        }
116        if is_metadata {
117            metadata_names.insert(name.clone());
118        }
119        let position = columns.len();
120        columns.push(column_info_from_type_str(name, type_str, position, None));
121        Ok(())
122    };
123
124    for pk in &base.partition_keys {
125        push(
126            &mut columns,
127            &mut metadata_names,
128            pk.name.clone(),
129            &pk.data_type,
130            false,
131        )?;
132    }
133    for ck in &base.clustering_keys {
134        push(
135            &mut columns,
136            &mut metadata_names,
137            ck.name.clone(),
138            &ck.data_type,
139            false,
140        )?;
141    }
142
143    // `TableSchema::columns` carries EVERY declared column, key columns
144    // included (issue #4222 research pass) — skip the ones already emitted
145    // above so a key column is never duplicated as a "non-key" metadata quad.
146    for col in base
147        .columns
148        .iter()
149        .filter(|c| !key_names.contains(c.name.as_str()))
150    {
151        push(
152            &mut columns,
153            &mut metadata_names,
154            col.name.clone(),
155            &col.data_type,
156            false,
157        )?;
158
159        // Fail closed (design.md D8) on an unparseable declared type rather
160        // than defaulting to "simple" (roborev finding, issue #4222): a type
161        // this parser cannot classify might be complex, and silently
162        // declaring the wrong column shape is the same class of defect the
163        // collision check above exists to prevent.
164        // [`CqlType::parse`] — the parser the doc comments above cite
165        // (`cql_type_parser.rs`) — and NEVER `row_build.rs`'s
166        // `parse_cql_type_str` (roborev finding, issue #4222 — round 11):
167        // the latter is `parser::complex_types::ComplexTypeParser`, which
168        // is materially weaker. It has NO `varint` arm at all (so `varint`
169        // fell through to `Custom("varint")` and was refused as an
170        // "ambiguous UDT"), and its primitive alternation lists `time`
171        // BEFORE `timeuuid`, so `"timeuuid"` matched `Time` with a
172        // trailing `"uuid"` and was rejected outright as malformed. Both
173        // outcomes are hard errors here, so the raw view was UNUSABLE for
174        // any table declaring a `timeuuid`/`varint`/`vector<..>`/UDT
175        // column — real committed fixtures do (`basic-types.cql`'s
176        // `session_id TIMEUUID`, `issue-4114-vector-float.cql`'s
177        // `vector<float, n>`) — and the message misdiagnosed a primitive
178        // parser gap as UDT ambiguity. `CqlType::parse` resolves all of
179        // those and reserves `Custom(_)` for the genuinely-ambiguous UDT
180        // shape the fail-closed arm below is actually about.
181        let cql_type = CqlType::parse(&col.data_type).map_err(|e| {
182            Error::Schema(format!(
183                "raw SSTable view: base table '{}.{}' column '{}' has a declared type \
184                 ('{}') this parser cannot classify as simple or complex ({e}) — refusing \
185                 to guess the column contract shape (issue #28 no-heuristics)",
186                base.keyspace, base.table, col.name, col.data_type
187            ))
188        })?;
189        let is_complex = try_is_complex_cql_type(&cql_type).ok_or_else(|| {
190            Error::Schema(format!(
191                "raw SSTable view: base table '{}.{}' column '{}' has declared type '{}', \
192                 which parses to an unclassifiable custom type — this parser cannot tell \
193                 whether it is a single-cell type or a multi-cell UDT/collection without a \
194                 UDT registry (a bare lowercase UDT name parses to this exact shape) — \
195                 refusing to guess the column contract shape (issue #28 no-heuristics)",
196                base.keyspace, base.table, col.name, col.data_type
197            ))
198        })?;
199        if is_complex {
200            push(
201                &mut columns,
202                &mut metadata_names,
203                format!("{}_complex_deletion", col.name),
204                "boolean",
205                true,
206            )?;
207            // `bigint`, never `int` (roborev finding, issue #4222 — round
208            // 8, matching `partition_deletion_time`/`range_deletion_time`'s
209            // round-7 fix for the same defect class): a far-future LDT is
210            // carried as the wrapped `as u32 as i32` on-disk bit pattern
211            // (`compaction_row.rs`'s module-header invariant on
212            // `ComplexColumn::complex_deletion`'s `i32` element), and an
213            // `int` column can only render its SIGN-EXTENDED (fabricated
214            // negative) form.
215            push(
216                &mut columns,
217                &mut metadata_names,
218                format!("{}_complex_deletion_time", col.name),
219                "bigint",
220                true,
221            )?;
222            push(
223                &mut columns,
224                &mut metadata_names,
225                format!("{}_complex_deletion_timestamp", col.name),
226                "bigint",
227                true,
228            )?;
229        } else {
230            push(
231                &mut columns,
232                &mut metadata_names,
233                format!("{}_timestamp", col.name),
234                "bigint",
235                true,
236            )?;
237            push(
238                &mut columns,
239                &mut metadata_names,
240                format!("{}_ttl", col.name),
241                "int",
242                true,
243            )?;
244            // `bigint`, never `int` (roborev finding, issue #4222 — round
245            // 8): see the identical `_complex_deletion_time` note above —
246            // `SimpleCell::local_deletion_time`/`TombstoneInfo::local_deletion_time`
247            // both feed this column and either can carry a value an `int`
248            // cannot render honestly.
249            push(
250                &mut columns,
251                &mut metadata_names,
252                format!("{}_local_deletion_time", col.name),
253                "bigint",
254                true,
255            )?;
256            push(
257                &mut columns,
258                &mut metadata_names,
259                format!("{}_tombstone", col.name),
260                "text",
261                true,
262            )?;
263        }
264    }
265
266    // The row-level metadata QUINTET (roborev finding, issue #4222 — round
267    // 9): every one of these is a metadata derivative — populated ONLY on
268    // a plain `row_kind = 'row'` row (`row_map.rs`'s `Live`/`Tombstone`
269    // arms) — so `is_metadata = true` for each.
270    push(
271        &mut columns,
272        &mut metadata_names,
273        "row_timestamp".to_string(),
274        "bigint",
275        true,
276    )?;
277    push(
278        &mut columns,
279        &mut metadata_names,
280        "row_ttl".to_string(),
281        "int",
282        true,
283    )?;
284    // The liveness marker's OWN expiry (epoch seconds), distinct from the row
285    // TOMBSTONE's GC clock below (roborev finding, issue #4222 — round 11).
286    // The two COEXIST on a single physical row (`row_deletion` alongside a
287    // TTL'd `row_liveness`, issue #932), and folding both into
288    // `row_local_deletion_time` meant the tombstone's LDT silently
289    // OVERWROTE — and discarded — the marker's expiry. Separate facts,
290    // separate columns.
291    push(
292        &mut columns,
293        &mut metadata_names,
294        "row_liveness_expires_at".to_string(),
295        "bigint",
296        true,
297    )?;
298    // The row TOMBSTONE's GC clock, and ONLY that (round 11, above).
299    // `bigint`, never `int` (roborev finding, issue #4222 — round 8): see
300    // the identical `<col>_local_deletion_time` note above.
301    push(
302        &mut columns,
303        &mut metadata_names,
304        "row_local_deletion_time".to_string(),
305        "bigint",
306        true,
307    )?;
308    push(
309        &mut columns,
310        &mut metadata_names,
311        "row_tombstone".to_string(),
312        "text",
313        true,
314    )?;
315    // The row tombstone's own `markedForDeleteAt` — distinct from
316    // `row_local_deletion_time` (the GC-clock seconds), mirroring the
317    // partition/range pairs below (roborev finding, issue #4222: this was
318    // previously discarded, making a row tombstone's writetime unrecoverable
319    // from this view).
320    push(
321        &mut columns,
322        &mut metadata_names,
323        "row_deletion_timestamp".to_string(),
324        "bigint",
325        true,
326    )?;
327
328    // The remaining columns are all ALWAYS-APPLICABLE (source-identity/
329    // row_kind/partition-and-range-deletion — `predicates.rs`'s
330    // `always_applicable_column_names`), never a metadata DERIVATIVE of a
331    // base column — `is_metadata = false` for each.
332    push(
333        &mut columns,
334        &mut metadata_names,
335        "partition_deletion_time".to_string(),
336        "bigint",
337        false,
338    )?;
339    push(
340        &mut columns,
341        &mut metadata_names,
342        "partition_deletion_timestamp".to_string(),
343        "bigint",
344        false,
345    )?;
346
347    push(
348        &mut columns,
349        &mut metadata_names,
350        "row_kind".to_string(),
351        "text",
352        false,
353    )?;
354    push(
355        &mut columns,
356        &mut metadata_names,
357        "bound_inclusive".to_string(),
358        "boolean",
359        false,
360    )?;
361    push(
362        &mut columns,
363        &mut metadata_names,
364        "range_deletion_time".to_string(),
365        "bigint",
366        false,
367    )?;
368    push(
369        &mut columns,
370        &mut metadata_names,
371        "range_deletion_timestamp".to_string(),
372        "bigint",
373        false,
374    )?;
375
376    push(
377        &mut columns,
378        &mut metadata_names,
379        "sstable".to_string(),
380        "text",
381        false,
382    )?;
383    // `bigint`, not `int` (roborev finding, issue #4222): `SSTableReader::generation`
384    // is a `u64`; narrowing it to `i32` would SATURATE (and thus collapse
385    // two distinct generations to the same fabricated value) for any real
386    // corpus whose generation identifiers exceed `i32::MAX` — a
387    // no-heuristics violation in a view whose whole purpose is authoritative
388    // per-generation source identity.
389    push(
390        &mut columns,
391        &mut metadata_names,
392        "generation".to_string(),
393        "bigint",
394        false,
395    )?;
396    push(
397        &mut columns,
398        &mut metadata_names,
399        "format".to_string(),
400        "text",
401        false,
402    )?;
403    // KNOWN, DELIBERATE scope limitation (roborev finding, issue #4222 —
404    // round 8): `position`'s PROJECTED value diverges by internal access
405    // path, and nothing distinguishes the two cases from the value alone.
406    // `point.rs::resolve_position` resolves a real byte offset via a
407    // second index lookup (best-effort — a lookup miss/error also yields
408    // `Null`, per that function's own doc); `scan.rs`'s full-scan producer
409    // NEVER consults an index per row and always reports `Value::Null`.
410    // So `SELECT pk, position FROM t_raw_sstable_data WHERE pk = 1` (point
411    // path) yields a real offset while the SAME projection with no WHERE
412    // (full-scan path) yields `NULL` for every row — a caller cannot tell
413    // "not measured on this access path" from "genuinely no offset" from
414    // the value alone. A `position` PREDICATE is rejected outright for
415    // exactly this reason (see the check in `execute_raw_sstable_view`);
416    // the PROJECTED value is NOT rejected, because a `SELECT *` (this
417    // view's primary, already-tested shape) must keep working over BOTH
418    // access paths, and a bare `NULL` is at least an honest "not measured"
419    // signal, never a fabricated one. Resolving `position` on the
420    // full-scan path too (a per-row index lookup) would undermine that
421    // producer's whole "bounded full scan" cost model; deferred rather
422    // than fixed here.
423    push(
424        &mut columns,
425        &mut metadata_names,
426        "position".to_string(),
427        "bigint",
428        false,
429    )?;
430
431    Ok((columns, metadata_names))
432}
