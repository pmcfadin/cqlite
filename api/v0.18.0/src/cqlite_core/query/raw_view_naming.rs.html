1//! The `_raw_sstable_data` table-name suffix (issue #4222) — a tiny,
2//! crate-shared naming convention (design.md D1) between the raw view's own
3//! interception (`select_executor::raw_view`) and every OTHER place in the
4//! engine that resolves "does this table have a schema" from a table-name
5//! STRING before a query ever reaches the executor.
6//!
7//! The concrete defect this module exists to fix: `Database::has_schema_for_table`
8//! / `schema_status` (used by the CLI's issue #199 pre-flight schema check,
9//! `cqlite-cli/src/main.rs`) resolve the LITERAL table name a query names —
10//! for `SELECT * FROM ks.t_raw_sstable_data`, that is the SUFFIXED name, which
11//! is never itself a registered schema (only `ks.t` is). Without this shared
12//! constant the CLI rejected every raw-view query before `SelectExecutor`
13//! ever got a chance to intercept it (design.md D6).
14
15/// Suffix that marks a table reference as the raw-SSTable-view name for its
16/// base table (design.md D1). The single source of truth for the literal
17/// string — [`select_executor::raw_view`](crate::query::select_executor)'s
18/// own copy is a `pub(in ...)` alias of this constant, not a second literal.
19pub const RAW_SSTABLE_VIEW_SUFFIX: &str = "_raw_sstable_data";
20
21/// Strip [`RAW_SSTABLE_VIEW_SUFFIX`] from a BARE table name (no keyspace
22/// segment — callers needing the qualified form pair this with their own
23/// keyspace handling, as `select_executor::raw_view` does). `None` when the
24/// suffix is absent, or stripping it would leave an empty base name.
25pub fn strip_raw_view_suffix(table_name: &str) -> Option<&str> {
26    table_name
27        .strip_suffix(RAW_SSTABLE_VIEW_SUFFIX)
28        .filter(|base| !base.is_empty())
29}
30
31#[cfg(test)]
32mod tests {
33    use super::*;
34
35    #[test]
36    fn strips_bare_suffix_only() {
37        assert_eq!(strip_raw_view_suffix("t_raw_sstable_data"), Some("t"));
38        assert_eq!(strip_raw_view_suffix("t"), None);
39        assert_eq!(strip_raw_view_suffix("_raw_sstable_data"), None);
40    }
