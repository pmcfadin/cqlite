1//! GROUP BY key hashing + equality (issue #1587 hash index, #2074 Cassandra
2//! float comparator).
3//!
4//! Split out of `aggregation.rs` (campsite rule, epic #1116) — these are pure,
5//! self-contained functions with no `AggregationState`/executor dependencies.
6//! [`hash_group_key`] and [`group_key_eq`] MUST stay mutually consistent (equal
7//! keys hash equal) — see each doc comment.
8
9use crate::types::Value;
10use std::hash::{Hash, Hasher};
11
12/// Hash a group key consistently with [`group_key_eq`]: if two keys are equal
13/// under that predicate, they hash identically. Floats follow Cassandra's
14/// total-order comparator (issue #2074), so NaN is canonicalized to a single bit
15/// pattern (all NaN keys hash-collide into ONE group) and signed-zero bits are
16/// kept DISTINCT (`-0.0` and `+0.0` hash apart into two groups) — at ALL nesting
17/// depths, recursing into `List`/`Set`/`Map`/`Tuple`/`Frozen`. Exotic variants
18/// that are not cheap to hash structurally fall back to a discriminant-only
19/// contribution — still consistent (equal values agree on discriminant), merely
20/// less selective, and only for key types that never appear in practice.
21pub(super) fn hash_group_key(key: &[Value]) -> u64 {
22    let mut hasher = std::collections::hash_map::DefaultHasher::new();
23    key.len().hash(&mut hasher);
24    for v in key {
25        hash_one_value(v, &mut hasher);
26    }
27    hasher.finish()
28}
29
30fn hash_one_value<H: Hasher>(v: &Value, h: &mut H) {
31    // Tag by discriminant so different variants never alias.
32    std::mem::discriminant(v).hash(h);
33    match v {
34        Value::Null => {}
35        Value::Boolean(b) => b.hash(h),
36        Value::Integer(x) => x.hash(h),
37        Value::BigInt(x) | Value::Counter(x) | Value::Timestamp(x) | Value::Time(x) => x.hash(h),
38        Value::Date(x) => x.hash(h),
39        Value::TinyInt(x) => x.hash(h),
40        Value::SmallInt(x) => x.hash(h),
41        // Issue #2074 (Cassandra comparator, consistent with `group_key_eq`):
42        // canonicalize NaN to ONE bit pattern so all NaN keys hash-collide into one
43        // group; keep signed-zero bits DISTINCT so `-0.0`/`+0.0` hash into two.
44        Value::Float(f) => (if f.is_nan() { f64::NAN } else { *f }).to_bits().hash(h),
45        Value::Float32(f) => (if f.is_nan() { f32::NAN } else { *f }).to_bits().hash(h),
46        Value::Text(s) => s.hash(h),
47        Value::Blob(b) | Value::Varint(b) | Value::Inet(b) => b.hash(h),
48        Value::Uuid(u) => u.hash(h),
49        Value::Decimal { scale, unscaled } => {
50            scale.hash(h);
51            unscaled.hash(h);
52        }
53        Value::Duration {
54            months,
55            days,
56            nanos,
57        } => {
58            months.hash(h);
59            days.hash(h);
60            nanos.hash(h);
61        }
62        Value::List(items) | Value::Set(items) | Value::Tuple(items) => {
63            for item in items {
64                hash_one_value(item, h);
65            }
66        }
67        Value::Map(pairs) => {
68            for (k, val) in pairs {
69                hash_one_value(k, h);
70                hash_one_value(val, h);
71            }
72        }
73        Value::Frozen(inner) => hash_one_value(inner, h),
74        // Exotic / rarely-grouped variants: discriminant-only (still consistent).
75        Value::Json(_) | Value::Udt(_) | Value::Tombstone(_) => {}
76    }
77}

79/// Group-key VALUE equality with Cassandra float-comparator semantics (issue
80/// #2074), at ALL nesting depths: `float`/`double` — top-level OR nested inside a
81/// `List`/`Set`/`Map`/`Tuple`/`Frozen` group key — route through `float_cmp` so
82/// all NaN bit-patterns are EQUAL and `-0.0`/`+0.0` are DISTINCT (Cassandra's
83/// comparator); every other leaf uses derived `==`. Recurses through the SAME
84/// variants [`hash_one_value`] does, so it stays consistent with
85/// [`hash_group_key`] at every depth. Cross-variant keys stay distinct (the
86/// `_ => a == b` fallback) — never `compare_values_ordering`'s cross-coercion.
87fn group_value_eq(a: &Value, b: &Value) -> bool {
88    match (a, b) {
89        (Value::Float(p), Value::Float(q)) => {
90            crate::float_cmp::cassandra_double_cmp(*p, *q).is_eq()
91        }
92        (Value::Float32(p), Value::Float32(q)) => {
93            crate::float_cmp::cassandra_float_cmp(*p, *q).is_eq()
94        }
95        (Value::List(x), Value::List(y))
96        | (Value::Set(x), Value::Set(y))
97        | (Value::Tuple(x), Value::Tuple(y)) => {
98            x.len() == y.len() && x.iter().zip(y).all(|(m, n)| group_value_eq(m, n))
99        }
100        (Value::Map(x), Value::Map(y)) => {
101            x.len() == y.len()
102                && x.iter()
103                    .zip(y)
104                    .all(|((k1, v1), (k2, v2))| group_value_eq(k1, k2) && group_value_eq(v1, v2))
105        }
106        (Value::Frozen(x), Value::Frozen(y)) => group_value_eq(x, y),
107        _ => a == b,
108    }
109}
110
111/// Group-key equality: see [`group_value_eq`] for the per-element (recursive,
112/// float-comparator-aware) semantics.
113pub(super) fn group_key_eq(a: &[Value], b: &[Value]) -> bool {
114    a.len() == b.len() && a.iter().zip(b).all(|(x, y)| group_value_eq(x, y))
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120
121    /// Issue #2074: hash/eq consistency at every nesting depth — the specific
122    /// property the review round flagged. For every pair below, `group_key_eq`
123    /// agreeing (or disagreeing) MUST match `hash_group_key` agreeing (or
124    /// disagreeing), or the hash index would silently split/merge groups the
125    /// exact-match confirmation disagrees with.
126    #[test]
127    fn hash_and_eq_agree_for_nested_floats() {
128        let other_nan = f64::from_bits(0xFFF8_0000_0000_0001);
129        assert!(other_nan.is_nan());
130        let cases: &[(Value, Value, bool)] = &[
131            (
132                Value::Tuple(vec![Value::Float(0.0)]),
133                Value::Tuple(vec![Value::Float(-0.0)]),
134                false, // -0.0/+0.0 distinct even nested
135            ),
136            (
137                Value::Tuple(vec![Value::Float(f64::NAN)]),
138                Value::Tuple(vec![Value::Float(other_nan)]),
139                true, // differing NaN bit patterns merge even nested
140            ),
141            (
142                Value::List(vec![Value::Float(0.0)]),
143                Value::List(vec![Value::Float(-0.0)]),
144                false,
145            ),
146            (
147                Value::Map(vec![(Value::Integer(1), Value::Float(0.0))]),
148                Value::Map(vec![(Value::Integer(1), Value::Float(-0.0))]),
149                false,
150            ),
151            (
152                Value::Frozen(Box::new(Value::Float(f64::NAN))),
153                Value::Frozen(Box::new(Value::Float(other_nan))),
154                true,
155            ),
156        ];
157        for (a, b, expect_eq) in cases {
158            let ka = [a.clone()];
159            let kb = [b.clone()];
160            let eq = group_key_eq(&ka, &kb);
161            assert_eq!(
162                eq, *expect_eq,
163                "group_key_eq({a:?}, {b:?}) expected {expect_eq}"
164            );
165            let hash_eq = hash_group_key(&ka) == hash_group_key(&kb);
166            assert!(
167                hash_eq == eq || !eq,
168                "hash/eq inconsistency for {a:?} vs {b:?}: eq={eq} but hashes {}",
169                if hash_eq { "match" } else { "differ" }
170            );
171            // When eq() says equal, hash MUST also agree (the formal contract);
172            // when eq() says unequal, colliding hashes are merely a (harmless)
173            // collision, not a bug, so we only assert the equal-implies-equal-hash
174            // direction strictly.
175            if eq {
176                assert!(
177                    hash_eq,
178                    "eq() reports equal but hashes differ for {a:?} vs {b:?} — \
179                     violates the hash/eq contract"
180                );
181            }
182        }
183    }
