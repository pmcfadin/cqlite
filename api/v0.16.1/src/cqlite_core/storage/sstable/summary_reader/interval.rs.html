1//! Token-ordered `Summary.db` find-by-key + the lazy `Index.db` interval accessor
2//! for the Summary-guided BIG partition index (issue #2412, design §A/§B).
3//!
4//! Two primitives that make BIG open Cassandra-lazy:
5//!
6//! - [`find_interval_for_key`] binary-searches the token-ordered `Summary.db` samples
7//!   for the half-open `Index.db` byte range [`SummaryInterval`] that must contain a
8//!   query key's partition entry (the core new search; [`super::SummaryReader::find_by_key`]
9//!   is the method wrapper).
10//! - [`lookup_key_in_interval`] reads **only that one interval** from disk and
11//!   resolves the exact key, recording exactly one **interval** parse
12//!   (`cqlite.sstable.index_interval_parses_total`) — never a full parse, so a
13//!   lazy-open regression stays visible on `index_parses_total` (design §F). An
14//!   END-BOUNDED interval is scanned in full (its `[start, end)` byte range between
15//!   two real samples is the bound, not an entry count — Cassandra's `Summary.db`
16//!   downsampling can widen it well past `min_index_interval`); only the read-to-EOF
17//!   LAST interval keeps a downsampling-adjusted entry cap (roborev job 1709).
18//!
19//! No-heuristics (issue #28): every seek offset and interval boundary is an
20//! authoritative `Summary.db` sample position; entry framing is the on-disk `Index.db`
21//! structure. Nothing is inferred from value bytes.
22

32/// A `Summary.db`-derived half-open byte range `[start_position, end_position)` into
33/// `Index.db` that must contain a query key's partition entry (issue #2412).
34///
35/// Produced by [`super::SummaryReader::find_by_key`]. `end_position == None` means
36/// "scan to EOF" (the covering sample is the last one). Both bounds are authoritative
37/// summary sample positions — the interval is structural, never a guessed boundary (#28).
38#[derive(Debug, Clone, Copy, PartialEq, Eq)]
39pub struct SummaryInterval {
40    /// Inclusive start byte offset into `Index.db` (the covering sample's position).
41    pub start_position: u64,
42    /// Exclusive end byte offset into `Index.db` (the next sample's position), or
43    /// `None` to read to EOF when the covering sample is the last one.
44    pub end_position: Option<u64>,
45    /// Index of the covering (floor) sample within [`super::SummaryReader::get_entries`].
46    pub sample_index: usize,
47}
48
49/// Token-ordered binary search over `Summary.db` samples for the `Index.db` interval
50/// covering `key` (issue #2412). Pure core of [`super::SummaryReader::find_by_key`],
51/// factored out so it is unit-testable without a `Platform`/file open.
52///
53/// The samples are in Cassandra **token order** (murmur3 of the raw partition key,
54/// ties broken by raw bytes), NOT lexicographic byte order, so the search compares
55/// with `cmp_partition_keys_by_token` — comparing raw bytes would be wrong for any
56/// non-degenerate ring. Returns the half-open `[start, end)`: `start` is the greatest
57/// sample whose key is `<= key` (Cassandra's `getPosition` floor); `end` is the next
58/// sample's position or `None` (EOF) for the last sample. A key below the first sample
59/// clamps the floor to sample 0 (Cassandra walks from the index start); the C5 range
60/// short-circuit already answers genuinely out-of-`[first_key,last_key]` reads upstream
61/// with zero probe work. Returns `None` only for an empty summary.
62pub(crate) fn find_interval_for_key(
63    entries: &[SummaryEntry],
64    key: &[u8],
65) -> Option<SummaryInterval> {
66    use crate::util::cassandra_murmur3::cmp_partition_keys_by_token;
67    use std::cmp::Ordering;
68
69    if entries.is_empty() {
70        return None;
71    }
72
73    // Count of samples whose key is <= `key` in token order (monotone prefix).
74    let le_count = entries.partition_point(|e| {
75        cmp_partition_keys_by_token(&e.partition_key, key) != Ordering::Greater
76    });
77    // Floor index: greatest sample <= key, clamped to 0 when key sorts below all.
78    let floor = le_count.saturating_sub(1);
79
80    Some(SummaryInterval {
81        start_position: entries[floor].position,
82        end_position: entries.get(floor + 1).map(|e| e.position),
83        sample_index: floor,
84    })
85}

87/// Result of resolving a key within one Summary-bounded `Index.db` interval.
88#[derive(Debug)]
89pub struct IntervalLookup {
90    /// The matched partition entry, or `None` when the key is genuinely absent from
91    /// this interval (an authoritative absence between two summary samples — the
92    /// whole-file `scan_for_key` oracle is NOT needed for the common case, design §B).
93    pub entry: Option<PartitionIndexEntry>,
94    /// Number of `Index.db` entries actually parsed/touched, bounded by ONE summary
95    /// interval's actual entry count (an end-bounded interval's `[start, end)` byte
96    /// range; a downsampling-adjusted cap for the read-to-EOF last interval). The
97    /// scale-free work-probe for Requirement 2 — bounded by the interval, never by
98    /// the SSTable's total partition count.
99    pub entries_touched: usize,
100}
101
102/// Cassandra's `IndexSummary` base (non-downsampled) sampling level — sample
103/// spacing is `min_index_interval` partitions ONLY at this level; a lower
104/// `sampling_level` widens spacing proportionally (`IndexSummary.java`,
105/// `BASE_SAMPLING_LEVEL = 128`). Duplicated from the (feature-gated,
106/// write-support-only) writer-side constant of the same name and value so this
107/// read-path module never depends on `write-support` being enabled.
108const BASE_SAMPLING_LEVEL: u32 = 128;
109
110/// Cap on entries scanned in the LAST (read-to-EOF) `Index.db` interval —
111/// roborev job 1709 (High, data-loss class): the cap must be DOWNSAMPLING-AWARE.
112///
113/// Cassandra downsamples `Summary.db` under index-summary memory pressure
114/// (`sampling_level < BASE_SAMPLING_LEVEL`), which widens EVERY interval —
115/// including the last — up to `min_index_interval * BASE_SAMPLING_LEVEL /
116/// sampling_level` partitions, not `min_index_interval`. A cap fixed at
117/// `min_index_interval + 1` (the pre-fix behavior) would cut a downsampled last
118/// interval's scan short before reaching a present key that legitimately sits
119/// beyond the un-downsampled baseline. This is the ONLY cap that survives this
120/// fix (see [`scan_interval_bytes`]'s caller): a `Some(end_position)` interval
121/// is bounded by its authoritative `[start, end)` byte range instead, so no
122/// entry-count cap is needed there at all.
123///
124/// `sampling_level` is clamped to Cassandra's documented `[1,
125/// BASE_SAMPLING_LEVEL]` range (`IndexSummary.java`) before dividing: `0`
126/// (corrupt/never-should-happen) clamps to `1` — the MAXIMUM legitimate
127/// widening, the fail-safe direction for a cap (a too-generous cap only costs
128/// extra scan work; a too-small one risks exactly this issue's false-negative).
129/// A value `> BASE_SAMPLING_LEVEL` (equally never-should-happen) clamps to
130/// `BASE_SAMPLING_LEVEL`, i.e. the un-downsampled baseline — never a guess,
131/// always derived from the already-parsed header fields (no-heuristics, #28).
132fn last_interval_entry_cap(min_index_interval: u32, sampling_level: u32) -> usize {
133    let effective_sampling_level = sampling_level.clamp(1, BASE_SAMPLING_LEVEL);
134    let effective_interval = (min_index_interval as u64).saturating_mul(BASE_SAMPLING_LEVEL as u64)
135        / (effective_sampling_level as u64);
136    // +1 covers the boundary entry that sits exactly at EOF.
137    usize::try_from(effective_interval.saturating_add(1)).unwrap_or(usize::MAX)
138}

140/// Scan a slice of `Index.db` interval bytes for the entry whose key equals `key`.
141///
142/// Pure core of [`lookup_key_in_interval`] (unit-testable without a file). Parses
143/// forward with [`parse_big_index_entry`] up to `entry_cap` entries, stopping at the
144/// first exact key match, at buffer exhaustion, at the entry cap, or at the first
145/// unparseable entry (a truncated boundary tail — treated as end-of-interval).
146pub(crate) fn scan_interval_bytes(bytes: &[u8], key: &[u8], entry_cap: usize) -> IntervalLookup {
147    let mut remaining = bytes;
148    let mut entries_touched = 0usize;
149
150    while !remaining.is_empty() && entries_touched < entry_cap {
151        match parse_big_index_entry(remaining) {
152            Ok((rest, entry)) => {
153                // Forward-progress guard mirrors the full parser's debug_assert.
154                if rest.len() >= remaining.len() {
155                    break;
156                }
157                entries_touched += 1;
158                if entry.key_digest.as_ref() == key {
159                    return IntervalLookup {
160                        entry: Some(entry),
161                        entries_touched,
162                    };
163                }
164                remaining = rest;
165            }
166            // Unparseable tail at the interval boundary — end of this interval's whole
167            // entries (never a heuristic: the boundary is an authoritative summary sample).
168            Err(_) => break,
169        }
170    }
171
172    IntervalLookup {
173        entry: None,
174        entries_touched,
175    }
176}

178/// Read one Summary-bounded `Index.db` interval from disk and resolve `key` within it
179/// (issue #2412, design §B).
180///
181/// Seeks to `interval.start_position`, reads the interval bytes (`[start, end)`, or to
182/// EOF when `interval.end_position` is `None`), and scans forward for the exact key.
183/// Increments `cqlite.sstable.index_interval_parses_total` exactly once (one bounded
184/// interval parse), never `index_parses_total` (design §F).
185///
186/// Entry-count cap (roborev job 1709, High — downsampling false-negative fix):
187/// an END-BOUNDED interval (`end_position.is_some()`) is NOT entry-capped at all —
188/// its authoritative `[start, end)` byte range, delimited by two REAL adjacent
189/// `Summary.db` samples, already bounds the read; every whole entry the buffer
190/// holds is scanned regardless of Cassandra's downsampling (which can widen an
191/// interval to `min_index_interval * 128 / sampling_level` partitions — a fixed
192/// `min_index_interval`-derived cap could otherwise stop short of a present key
193/// and be (wrongly) treated as an authoritative absence by the caller,
194/// `big_get_with_resolution`'s `covering_interval_is_end_bounded` gate). The
195/// read-to-EOF LAST interval keeps a cap — [`last_interval_entry_cap`], derived
196/// from the SAME downsampling-adjusted formula — because `covering_interval_is_end_bounded`
197/// is `false` for it, so a capped miss there is never promoted to an authoritative
198/// absence (the caller falls back to `scan_for_key` instead, preserving #1572).
199///
200/// Corrupt-input hardening (roborev job 1709 follow-up, fuzz doctrine #1614: a
201/// parser must never panic/abort/OOM on arbitrary bytes — and through the
202/// bindings, an allocation-failure abort kills the WHOLE host process, not just
203/// this request):
204/// - `end_position` is validated against the `Index.db` file's ACTUAL length
205///   BEFORE it ever sizes an allocation. An unvalidated, corrupt `Summary.db`
206///   sample claiming an `end` far beyond EOF would otherwise drive
207///   `Vec::resize` to attempt a huge (attacker/corruption-controlled)
208///   allocation — which aborts the process on failure — before `read_exact`
209///   ever gets a chance to fail cleanly. A too-large `end` now returns a clean
210///   `Err` (fail-closed), never a huge alloc.
211/// - Non-monotone samples (`end <= start`) are ALSO fail-closed `Err`, not
212///   silently routed into the read-to-EOF branch: `end_position` staying
213///   `Some(_)` there would leave `covering_interval_is_end_bounded` reporting
214///   `true`, so a corrupt/degenerate interval's capped miss could otherwise be
215///   wrongly promoted to an AUTHORITATIVE absence (no `scan_for_key`
216///   fallback) — the exact false-negative class this job's primary fix closes,
217///   re-opened through a different corrupt-input door.
218pub async fn lookup_key_in_interval(
219    index_path: &Path,
220    interval: SummaryInterval,
221    key: &[u8],
222    min_index_interval: u32,
223    sampling_level: u32,
224) -> Result<IntervalLookup> {
225    let mut file = File::open(index_path).await?;
226
227    // Validate BEFORE seeking/reading/allocating (no-heuristics #28: derived from
228    // the file's authoritative metadata, never a guess).
229    let file_len = file.metadata().await?.len();
230    match interval.end_position {
231        Some(end) if end <= interval.start_position => {
232            return Err(Error::corruption(format!(
233                "lookup_key_in_interval: interval end_position {end} does not exceed \
234                 start_position {} — non-monotone Summary.db sample ordering (corrupt input)",
235                interval.start_position
236            )));
237        }
238        Some(end) if end > file_len => {
239            return Err(Error::corruption(format!(
240                "lookup_key_in_interval: interval end_position {end} exceeds Index.db file \
241                 length {file_len} (corrupt Summary.db sample position)"
242            )));
243        }
244        _ => {}
245    }
246
247    file.seek(std::io::SeekFrom::Start(interval.start_position))
248        .await?;
249
250    let mut buffer = Vec::new();
251    let entry_cap = match interval.end_position {
252        Some(end) => {
253            // Bounded read of exactly the interval's whole entries (validated
254            // above: `end > start` and `end <= file_len`). The byte range itself
255            // is the bound — no entry-count cap needed or applied.
256            let len = (end - interval.start_position) as usize;
257            buffer.resize(len, 0);
258            file.read_exact(&mut buffer).await?;
259            usize::MAX
260        }
261        // Last interval: read to EOF. The scan is bounded by the
262        // downsampling-adjusted `last_interval_entry_cap`, so this can never
263        // become a whole-file walk for a many-partition SSTable.
264        None => {
265            file.read_to_end(&mut buffer).await?;
266            last_interval_entry_cap(min_index_interval, sampling_level)
267        }
268    };
269
270    // One bounded interval parse (distinct counter — never a full parse, design §F).
271    crate::observability::add_counter(
272        crate::observability::catalog::INDEX_INTERVAL_PARSES_TOTAL,
273        1,
274        &[],
275    );
276
277    Ok(scan_interval_bytes(&buffer, key, entry_cap))
278}

286    /// Build token-ordered summary samples from arbitrary raw keys (issue #2412). The
287    /// `Summary.db` on disk is written in token order; a test mirrors that by sorting
288    /// with the same comparator `find_interval_for_key` searches with and assigning
289    /// strictly increasing `Index.db` positions.
290    fn token_ordered_samples(keys: &[&[u8]]) -> Vec<SummaryEntry> {
291        use crate::util::cassandra_murmur3::cmp_partition_keys_by_token;
292        let mut keys: Vec<Vec<u8>> = keys.iter().map(|k| k.to_vec()).collect();
293        keys.sort_by(|a, b| cmp_partition_keys_by_token(a, b));
294        keys.into_iter()
295            .enumerate()
296            .map(|(i, partition_key)| SummaryEntry {
297                partition_key,
298                position: (i as u64) * 128,
299            })
300            .collect()
301    }

379    /// Encode one BIG `Index.db` entry: `[key_len u16 BE][key][data_offset vint][promoted_len vint=0]`.
380    fn encode_entry(key: &[u8], data_offset: u64) -> Vec<u8> {
381        let mut out = Vec::new();
382        out.extend_from_slice(&(key.len() as u16).to_be_bytes());
383        out.extend_from_slice(key);
384        out.extend_from_slice(&crate::parser::vint::encode_vuint(data_offset));
385        out.extend_from_slice(&crate::parser::vint::encode_vuint(0));
386        out
387    }

389    fn build_interval(keys_offsets: &[(&[u8], u64)]) -> Vec<u8> {
390        let mut buf = Vec::new();
391        for (k, off) in keys_offsets {
392            buf.extend_from_slice(&encode_entry(k, *off));
393        }
394        buf
395    }

502    /// Write `n` synthetic entries at strictly increasing offsets, returning the
503    /// bytes plus the byte offset the `target_index`'th entry starts at (so a
504    /// caller can carve out a `[start, target_end)` sub-slice as an end-bounded
505    /// interval whose LAST entry is the target — the exact shape
506    /// `find_interval_for_key` would hand a caller: the interval's `end_position`
507    /// sits at the position of the NEXT summary sample, which is the position
508    /// immediately after the entries this interval covers).
509    fn build_n_entries(n: usize, target_index: usize, target_key: &[u8]) -> (Vec<u8>, u64) {
510        let mut buf = Vec::new();
511        let mut target_end = 0u64;
512        for i in 0..n {
513            let key: Vec<u8> = if i == target_index {
514                target_key.to_vec()
515            } else {
516                format!("k{i:08}").into_bytes()
517            };
518            buf.extend_from_slice(&encode_entry(&key, i as u64));
519            if i == target_index {
520                target_end = buf.len() as u64;
521            }
522        }
523        (buf, target_end)
524    }

630    /// Last-interval (read-to-EOF) generous-cap bound test: `last_interval_entry_cap`
631    /// must derive the SAME downsampling-adjusted width formula
632    /// (`min_index_interval * 128 / sampling_level + 1`) the module doc promises —
633    /// pinned directly (pure function, no I/O) so the formula itself is load-bearing,
634    /// not just its end-to-end effect.
635    #[test]
636    fn last_interval_entry_cap_is_downsampling_adjusted() {
637        // Full sampling (128): cap is the pre-fix baseline, min_index_interval + 1.
638        assert_eq!(last_interval_entry_cap(128, 128), 129);
639        // 4x downsampled (sampling_level = 32): width widens to 128*128/32 = 512.
640        assert_eq!(last_interval_entry_cap(128, 32), 513);
641        // Maximum downsampling (sampling_level = 1): width widens to 128*128 = 16384.
642        assert_eq!(last_interval_entry_cap(128, 1), 16385);
643        // Corrupt/never-should-happen sampling_level = 0 clamps to 1 (max legitimate
644        // widening — the fail-safe direction for a cap, never a divide-by-zero).
645        assert_eq!(last_interval_entry_cap(128, 0), 16385);
646        // Corrupt/never-should-happen sampling_level > BASE_SAMPLING_LEVEL clamps to
647        // 128 (the un-downsampled baseline).
648        assert_eq!(last_interval_entry_cap(128, 9999), 129);
649    }

685    /// An `end_position` far beyond the `Index.db` file's actual length (a corrupt
686    /// `Summary.db` sample) must return a clean `Err`, NEVER attempt a huge
687    /// allocation. The synthetic file here is tiny (a handful of bytes); a
688    /// too-large `HOSTILE_END` would otherwise drive `buffer.resize` to zero-fill
689    /// a huge buffer from a few real bytes on disk before `read_exact` ever gets a
690    /// chance to fail.
691    ///
692    /// Asserts on the SPECIFIC validation error message (`"exceeds Index.db file
693    /// length"`), not merely `is_err()`: a generic `Err` is also reachable via
694    /// `read_exact`'s own `UnexpectedEof` AFTER a (successful, just slow) huge
695    /// allocation — that path is NOT what this test is pinning, and asserting only
696    /// `is_err()` would pass on EITHER path, silently accepting the huge-alloc
697    /// route. Matching the specific message proves the FAIL-FAST metadata-length
698    /// check fired BEFORE any allocation was attempted (never depends on actually
699    /// forcing a multi-GiB allocation inside a unit test, which would itself be
700    /// slow/host-memory-dependent and defeat the point of a fast, safe pin).
701    #[tokio::test]
702    async fn end_position_beyond_file_length_is_a_clean_error_not_a_huge_alloc() {
703        let bytes = build_interval(&[(b"alpha", 0), (b"bravo", 100)]);
704        let path = write_temp_index("beyond-eof", &bytes).await;
705
706        // A corrupt sample claiming an `end` ~1 GiB past this tiny file's real length.
707        const HOSTILE_END: u64 = 1 << 30;
708        let iv = SummaryInterval {
709            start_position: 0,
710            end_position: Some(HOSTILE_END),
711            sample_index: 0,
712        };
713        let res = lookup_key_in_interval(&path, iv, b"alpha", 128, 128).await;
714        let err = res.unwrap_err();
715        let msg = err.to_string();
716        assert!(
717            msg.contains("exceeds Index.db file length"),
718            "an end_position ({HOSTILE_END}) far beyond the file's real length ({}) must \
719             fail via the metadata-length pre-check (fast, no huge alloc attempted), got: {msg}",
720            bytes.len()
721        );
722
723        let _ = std::fs::remove_dir_all(path.parent().unwrap());
724    }
