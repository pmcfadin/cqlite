1//! Partition lookup and index-based access methods for SSTableReader
2//!
3//! This module contains methods for efficient partition lookup using Index.db,
4//! Summary.db, and Statistics.db readers.
5

12impl SSTableReader {
13    /// Consult the process-global key→partition-offset cache (issue #2059) with this
14    /// reader's authoritative generation identity. A `None` identity (a stat failure
15    /// at open) BYPASSES the cache — a structural miss, never a fabricated identity
16    /// (no-heuristics #28). Fail-closed on identity mismatch is enforced inside the
17    /// cache: an entry keyed on a different generation is a miss.
18    pub(crate) fn key_cache_get(&self, partition_key: &[u8]) -> Option<PartitionLoc> {
19        let identity = self.generation_identity?;
20        self.key_offset_cache.get(identity, partition_key)
21    }
22
23    /// Populate the process-global key→partition-offset cache (issue #2059) with this
24    /// reader's generation identity. A `None` identity is a no-op (the cache is
25    /// bypassed for this reader rather than fabricating an identity).
26    pub(crate) fn key_cache_insert(&self, partition_key: &[u8], loc: PartitionLoc) {
27        if let Some(identity) = self.generation_identity {
28            self.key_offset_cache.insert(identity, partition_key, loc);
29        }
30    }

32    /// Drop ALL of this reader's generation's entries from the process-global key
33    /// cache (issue #2059 §C). Called when the generation is removed / compacted
34    /// away / evicted from the warm registry, so its locations are reclaimed
35    /// promptly and recorded on the distinct `invalidations` counter (never the
36    /// budget-driven `evictions`). Returns the number of entries dropped. A
37    /// #2383 rebind does NOT call this — the identity is unchanged across a rebind,
38    /// so entries survive. No-op when the identity is `None`.
39    ///
40    /// `pub` so the flight `WarmTableRegistry` can invalidate on warm eviction
41    /// (issue #2059 §C) — the warm registry is the process's largest reader pin, so
42    /// its evictions are the primary reclamation trigger for the global cache.
43    pub fn invalidate_key_cache_entries(&self) -> u64 {
44        match self.generation_identity {
45            Some(identity) => self.key_offset_cache.invalidate(identity),
46            None => 0,
47        }
48    }

50    /// Enhanced partition lookup using Index.db reader with promoted index support.
51    ///
52    /// `partition_key` must be the raw partition-key bytes as produced by
53    /// [`PartitionKey::to_bytes`](crate::storage::write_engine::mutation::PartitionKey::to_bytes):
54    ///
55    /// - **Single-component keys** — raw value bytes (UUID = 16 bytes, int = 4 BE bytes, etc.).
56    /// - **Multi-component (composite) keys** — `[len: u16 BE][value bytes][0x00]` per component,
57    ///   including a trailing `0x00` after the final component.
58    ///
59    /// The Index.db key_lookup map is keyed on these exact raw bytes (set when the BIG-format
60    /// parser was fixed in Issue #552).  The old digest-based path (which caused every lookup
61    /// to miss) has been removed.  On a miss the function returns `Ok(None)` so callers can
62    /// fall through to their existing sequential-scan fallback.
63    pub async fn lookup_partition_with_index(
64        &self,
65        partition_key: &[u8],
66    ) -> Result<Option<(u64, u32)>> {
67        use crate::observability::{self as obs, catalog};
68
69        let format = self.sstable_format_label();
70
71        // B4 key→partition-offset cache (issue #1570): a repeated point read of a
72        // present key returns the cached `(offset,size)` without re-probing Index.db
73        // (`INDEX_PROBES` stays 0 on the hit). Positive-only. Checked FIRST — before
74        // any materialize / interval read — so the hit skips ALL probe work on both
75        // the Summary-guided and resident-map paths (issue #2412 §B).
76        if let Some(loc) = self.key_cache_get(partition_key) {
77            debug!(
78                "B4 key-cache hit for partition (raw key len={}): offset={}, size={}",
79                partition_key.len(),
80                loc.data_offset,
81                loc.data_size
82            );
83            obs::add_counter(
84                catalog::READ_PARTITION_LOOKUP,
85                1,
86                &[
87                    (catalog::attr::RESULT, "hit".into()),
88                    (catalog::attr::LOOKUP_ROUTE, "index".into()),
89                    (catalog::attr::SSTABLE_FORMAT, format.into()),
90                ],
91            );
92            return Ok(Some((loc.data_offset, loc.data_size)));
93        }
94
95        // Stage 3 (issue #2412 §B): a usable `Summary.db` + a not-yet-materialized
96        // lazy `Index.db` → resolve via ONE bounded Summary-guided interval instead of
97        // materializing the whole map. See `reader::summary_point` for the
98        // authoritative-absence rules the BIG point path applies to a resulting `None`.
99        if self.should_use_summary_interval() {
100            return self
101                .lookup_partition_via_summary_interval(partition_key)
102                .await;
103        }
104
105        // FellBack / already-materialized path: ensure the full resident map, then
106        // probe it. `ensure_materialized` MUST run BEFORE the tracing span — the
107        // non-`Send` `EnteredSpan` held across an await makes callers non-`Send`. A
108        // no-op after the first call, or for an eagerly-opened (FellBack) reader.
109        if let Some(index_reader) = &self.index_reader {
110            index_reader.ensure_materialized(&self.scan_cancel).await?;
111        }
112
113        let _span = tracing::debug_span!("sstable.partition_lookup.index").entered();
114
115        if let Some(index_reader) = &self.index_reader {
116            // A5/B4 read-work counter (`INDEX_PROBES`; consumer B4): one per real
117            // `Index.db` probe. A B4 cache hit above records nothing. No-op in release.
118            crate::storage::sstable::read_work_counters::record_index_probe();
119            // Direct raw-key lookup — O(1) HashMap lookup.
120            // Index.db entries are keyed on the raw partition key bytes since #552;
121            // no Murmur3 digest computation is needed or correct here.
122            if let Some(entry) = index_reader.lookup_partition(partition_key) {
123                debug!(
124                    "Found partition via Index.db raw-key lookup: offset={}, size={}",
125                    entry.data_offset, entry.data_size
126                );
127                obs::add_counter(
128                    catalog::READ_PARTITION_LOOKUP,
129                    1,
130                    &[
131                        (catalog::attr::RESULT, "hit".into()),
132                        (catalog::attr::LOOKUP_ROUTE, "index".into()),
133                        (catalog::attr::SSTABLE_FORMAT, format.into()),
134                    ],
135                );
136                // Cache this present-key resolution so the next point read of the
137                // same key skips the Index.db probe (issue #1570/#2059).
138                self.key_cache_insert(
139                    partition_key,
140                    PartitionLoc::new(entry.data_offset, entry.data_size),
141                );
142                return Ok(Some((entry.data_offset, entry.data_size)));
143            } else {
144                debug!(
145                    "Partition not found in Index.db for raw key (len={})",
146                    partition_key.len()
147                );
148            }
149        } else {
150            debug!("No Index.db reader available for partition lookup");
151        }
152        obs::add_counter(
153            catalog::READ_PARTITION_LOOKUP,
154            1,
155            &[
156                (catalog::attr::RESULT, "miss".into()),
157                (catalog::attr::LOOKUP_ROUTE, "index".into()),
158                (catalog::attr::SSTABLE_FORMAT, format.into()),
159            ],
160        );
161        Ok(None)
162    }

164    /// BTI ("da") partition point lookup via the in-memory Partitions.db trie
165    /// (issue #831, building on the verified #755 primitive).
166    ///
167    /// Encodes `partition_key` into the BTI byte-comparable trie key and walks the
168    /// trie to resolve the partition's location. Returns:
169    ///
170    /// - `Ok(Some(offset))` — the UNCOMPRESSED Data.db byte offset of the
171    ///   partition. INVARIANT: this offset indexes into the DECOMPRESSED data
172    ///   section, never raw file bytes. The offset is resolved in one of two ways:
173    ///     - **NARROW** partition (`BtiPartitionLocation::DataOffset`) — the trie
174    ///       returns the Data.db offset directly.
175    ///     - **WIDE** partition (`BtiPartitionLocation::RowsOffset`) — the trie
176    ///       returns a positive offset into `Rows.db`; we deserialize that
177    ///       partition's `TrieIndexEntry` via [`resolve_rows_db_entry`] and use its
178    ///       recovered `data_position` (issue #909/#910). Both forms share the same
179    ///       uncompressed Data.db offset domain, so the caller treats them
180    ///       identically.
181    /// - `Ok(None)` — the reader is not BTI, or the key has no trie path (the
182    ///   partition is definitely absent from this SSTable).
183    /// - `Err(_)` — a structural trie parse error, or a `RowsOffset` was returned
184    ///   without an accompanying `Rows.db` (a structurally invalid BTI SSTable).
185    ///
186    /// Because the BTI trie uses path compression, a returned offset may be a
187    /// candidate for a *prefix-colliding* key. The caller (`bti_point_lookup`)
188    /// MUST verify the partition-key bytes at the resolved offset equal the
189    /// queried key before returning rows.
190    ///
191    /// [`resolve_rows_db_entry`]: crate::storage::sstable::bti::resolve_rows_db_entry
192    pub fn lookup_partition_via_bti_trie(&self, partition_key: &[u8]) -> Result<Option<u64>> {
193        use crate::storage::sstable::bti::encode_partition_key_for_bti_trie;
194
195        // Not a BTI reader — no trie to consult (no descent, no encode, no count).
196        if self.bti_partitions_db.is_none() {
197            return Ok(None);
198        }
199
200        // B4 key→partition-offset cache (issue #1570) — the OUTERMOST layer, checked
201        // BEFORE the C3 memo and BEFORE the encode, so a repeated point read of the
202        // same present key returns the cached offset WITHOUT a trie descent
203        // (`TRIE_WALKS` stays 0) AND WITHOUT re-encoding the key (no `KEY_HASH_CALLS`
204        // bump — the encode below is skipped on a hit, preserving C4's
205        // one-hash-per-read property). Unlike the single-entry C3 memo, the multi-key
206        // B4 cache serves a hit even across interleaved keys. Trie-hit-only
207        // (prefix-collision candidates included; the caller re-verifies the key bytes
208        // at the resolved offset), never a trie MISS: the cached offset is the exact
209        // offset a fresh descent resolves, so re-verification reaches the identical
210        // conclusion (correctness guardrail; mirrors the C3 memo). Emit the presence
211        // counters (found = true) so a cache-served lookup records the same presence
212        // decision a fresh descent does. A disabled cache (`block_cache.enabled=false`)
213        // is a genuine no-op `get`, so this falls through and the read re-walks.
214        if let Some(loc) = self.key_cache_get(partition_key) {
215            Self::emit_bti_presence_counters(true);
216            return Ok(Some(loc.data_offset));
217        }
218
219        // Issue #1574 (C3) single walk: reuse the prune's resolution for the seek —
220        // a pure function of the immutable trie + key. Checked BEFORE the encode so
221        // a memo hit costs neither a trie descent NOR a Murmur3 key hash. A
222        // stale/absent slot simply misses and re-walks. The presence-oracle
223        // observability is preserved: a memo hit still records the presence decision.
224        if let Some(resolved) = self.bti_lookup_memo_get(partition_key) {
225            Self::emit_bti_presence_counters(resolved.is_some());
226            return Ok(resolved);
227        }
228
229        // Issue #1575 (C4): encode ONCE here (Murmur3 hash + byte-comparable
230        // encoding, `KEY_HASH_CALLS`). The multi-candidate prune path instead calls
231        // `lookup_partition_via_bti_trie_encoded` with a single hoisted encoding so N
232        // candidate generations share ONE hash rather than re-hashing per candidate.
233        let encoded = encode_partition_key_for_bti_trie(partition_key);
234        self.bti_trie_resolve(partition_key, &encoded)
235    }

237    /// Resolve a BTI partition using a PRE-ENCODED byte-comparable key, so the
238    /// Murmur3 hash + encoding is computed once per read and reused across every
239    /// candidate SSTable's prune instead of being recomputed per candidate
240    /// (issue #1575 / C4). `encoded` MUST equal
241    /// `encode_partition_key_for_bti_trie(partition_key)`; the caller
242    /// (`SSTableManager`'s candidate prune) computes it exactly once. Semantics are
243    /// otherwise identical to [`lookup_partition_via_bti_trie`]: same same-key memo,
244    /// same presence-oracle observability, same resolved offset. A non-BTI reader
245    /// returns `Ok(None)` (the `encoded` argument is simply unused).
246    pub fn lookup_partition_via_bti_trie_encoded(
247        &self,
248        partition_key: &[u8],
249        encoded: &[u8; 9],
250    ) -> Result<Option<u64>> {
251        if self.bti_partitions_db.is_none() {
252            return Ok(None);
253        }
254        // B4 key→partition-offset cache (issue #1570) — the OUTERMOST layer on the
255        // candidate-prune path too, checked BEFORE the C3 memo and the trie descent.
256        // The caller (`prune_candidates`) already hoisted the single encode into
257        // `encoded` (C4 / `KEY_HASH_CALLS == 1` per read); a cache hit here simply
258        // skips this candidate's trie descent (`TRIE_WALKS` stays 0 on the hit) — the
259        // multi-key cache covers interleaved keys the single-entry memo cannot. Same
260        // trie-hit-only correctness guardrail as the raw-key entry: the cached offset
261        // is the exact offset a fresh descent resolves. A disabled cache is a no-op
262        // `get`, so this falls through and the candidate re-walks.
263        if let Some(loc) = self.key_cache_get(partition_key) {
264            Self::emit_bti_presence_counters(true);
265            return Ok(Some(loc.data_offset));
266        }
267        if let Some(resolved) = self.bti_lookup_memo_get(partition_key) {
268            Self::emit_bti_presence_counters(resolved.is_some());
269            return Ok(resolved);
270        }
271        self.bti_trie_resolve(partition_key, encoded)
272    }

274    /// Shared BTI trie descent for a pre-encoded key — the body behind both
275    /// [`lookup_partition_via_bti_trie`] and
276    /// [`lookup_partition_via_bti_trie_encoded`]. The caller has already confirmed
277    /// this is a BTI reader and consulted the same-key memo; this records the single
278    /// `TRIE_WALKS`, walks the resident trie buffer with the pre-encoded key, emits
279    /// the presence-oracle counters, and stores the memo. It never re-encodes the
280    /// key (no `KEY_HASH_CALLS`), which is the property C4 hoists.
281    fn bti_trie_resolve(&self, partition_key: &[u8], encoded: &[u8; 9]) -> Result<Option<u64>> {
282        use crate::observability::{self as obs, catalog};
283        use crate::storage::sstable::bti::{
284            lookup_partition_in_bti_slice, resolve_rows_db_entry, BtiPartitionLocation,
285        };
286
287        let span = tracing::debug_span!(
288            "sstable.partition_lookup.bti_trie",
289            partition_shape = tracing::field::Empty,
290        );
291        let _entered = span.enter();
292
293        let Some(partitions_db) = &self.bti_partitions_db else {
294            // Not a BTI reader — no trie to descend (defensive; caller pre-checks).
295            return Ok(None);
296        };
297
298        // A5 read-work counter (TRIE_WALKS; consumers C3/C4): one per real trie
299        // descent — counted only once we have a `Partitions.db` to descend, so a
300        // non-BTI reader that returned above never bumps it. No-op in release
301        // (design.md Decision 1/2).
302        crate::storage::sstable::read_work_counters::record_trie_walk();
303
304        // Issue #1574 (C3): walk the resident `Arc<Vec<u8>>` trie buffer in place —
305        // no per-lookup whole-file copy. Issue #1575 (C4): with the PRE-ENCODED key,
306        // so the encode is not repeated per candidate.
307        let lookup =
308            lookup_partition_in_bti_slice(partitions_db.as_slice(), encoded).map_err(|e| {
309                Error::corruption(format!(
310                    "BTI Partitions.db trie lookup failed for partition key (len={}): {}",
311                    partition_key.len(),
312                    e
313                ))
314            });
315        let lookup = match lookup {
316            Ok(v) => v,
317            Err(e) => {
318                obs::record_error(&e, "reader");
319                return Err(e);
320            }
321        };
322
323        // The BTI Partitions.db trie IS the presence oracle for a BTI SSTable
324        // (BTI has no bloom filter). Emit READ_BLOOM_CHECKS exactly ONCE here — the
325        // single common path every BTI presence/point lookup funnels through
326        // (get → locate → bti_point_lookup, might_contain_partition, the candidate
327        // prune). `cqlite.result` is hit for a trie HIT
328        // (maybe-present/found: a DataOffset/RowsOffset, possibly a prefix-collision
329        // candidate the caller re-verifies) and miss for a trie MISS (definitive
330        // absence). A trie parse error returns above without an outcome and is
331        // counted as an error, not a bloom check, so this never double counts.
332        // Emitting before the per-arm handling (which can still error on a missing
333        // Rows.db for a wide partition) guarantees a single emission per call.
334        obs::add_counter(
335            catalog::READ_BLOOM_CHECKS,
336            1,
337            &[
338                (
339                    catalog::attr::RESULT,
340                    if lookup.is_some() { "hit" } else { "miss" }.into(),
341                ),
342                (catalog::attr::SSTABLE_FORMAT, "bti".into()),
343            ],
344        );
345
346        match lookup {
347            Some(BtiPartitionLocation::DataOffset(off)) => {
348                span.record("partition_shape", "narrow");
349                obs::add_counter(
350                    catalog::READ_PARTITION_LOOKUP,
351                    1,
352                    &[
353                        (catalog::attr::RESULT, "hit".into()),
354                        (catalog::attr::LOOKUP_ROUTE, "bti_trie".into()),
355                        (catalog::attr::SSTABLE_FORMAT, "bti".into()),
356                    ],
357                );
358                debug!(
359                    "BTI trie resolved NARROW partition (key len={}) to Data.db offset {}",
360                    partition_key.len(),
361                    off
362                );
363                self.bti_lookup_memo_store(partition_key, Some(off));
364                // Cache this trie-hit resolution (a prefix-collision candidate,
365                // re-verified by the caller downstream) so a later interleaved read of
366                // the same key skips the trie descent (issue #1570/#2059).
367                self.key_cache_insert(partition_key, PartitionLoc::offset_only(off));
368                Ok(Some(off))
369            }
370            Some(BtiPartitionLocation::RowsOffset(rows_offset)) => {
371                // WIDE partition: the trie pointed at this partition's
372                // TrieIndexEntry inside Rows.db. Deserialize it to recover the
373                // partition's Data.db start position (`data_position`), which lives
374                // in the SAME uncompressed-offset domain the narrow path uses, so
375                // the caller can decode the partition identically (#909/#910).
376                span.record("partition_shape", "wide");
377                let rows_db = match self.bti_rows_db.as_ref().ok_or_else(|| {
378                    Error::corruption(format!(
379                        "BTI Partitions.db trie returned RowsOffset({}) for partition key \
380                         (len={}) but this reader has no Rows.db loaded; the SSTable is \
381                         structurally invalid (Rows.db is required for wide partitions).",
382                        rows_offset,
383                        partition_key.len()
384                    ))
385                }) {
386                    Ok(v) => v,
387                    Err(e) => {
388                        obs::record_error(&e, "reader");
389                        return Err(e);
390                    }
391                };
392
393                let header = match resolve_rows_db_entry(rows_db.as_slice(), rows_offset as usize)
394                    .map_err(|e| {
395                        Error::corruption(format!(
396                            "BTI Rows.db row-index entry at RowsOffset({}) is unreadable for \
397                             partition key (len={}): {}",
398                            rows_offset,
399                            partition_key.len(),
400                            e
401                        ))
402                    }) {
403                    Ok(v) => v,
404                    Err(e) => {
405                        obs::record_error(&e, "reader");
406                        return Err(e);
407                    }
408                };
409
410                obs::add_counter(
411                    catalog::READ_PARTITION_LOOKUP,
412                    1,
413                    &[
414                        (catalog::attr::RESULT, "hit".into()),
415                        (catalog::attr::LOOKUP_ROUTE, "bti_trie".into()),
416                        (catalog::attr::SSTABLE_FORMAT, "bti".into()),
417                    ],
418                );
419                debug!(
420                    "BTI trie resolved WIDE partition (key len={}) via RowsOffset {} -> Data.db \
421                     position {} ({} row-index blocks)",
422                    partition_key.len(),
423                    rows_offset,
424                    header.data_position,
425                    header.block_count
426                );
427                self.bti_lookup_memo_store(partition_key, Some(header.data_position));
428                // Cache this present WIDE-partition resolution (issue #1570/#2059).
429                self.key_cache_insert(
430                    partition_key,
431                    PartitionLoc::offset_only(header.data_position),
432                );
433                Ok(Some(header.data_position))
434            }
435            None => {
436                obs::add_counter(
437                    catalog::READ_PARTITION_LOOKUP,
438                    1,
439                    &[
440                        (catalog::attr::RESULT, "miss".into()),
441                        (catalog::attr::LOOKUP_ROUTE, "bti_trie".into()),
442                        (catalog::attr::SSTABLE_FORMAT, "bti".into()),
443                    ],
444                );
445                self.bti_lookup_memo_store(partition_key, None);
446                Ok(None)
447            }
448        }
449    }

451    /// Cheap presence oracle: can this SSTable possibly contain `partition_key`?
452    ///
453    /// Used to prune SSTables before a partition-targeted scan (the query engine's
454    /// `WHERE pk = ?` fast path). Returning `false` MUST be definitive — the SSTable
455    /// is then skipped entirely without being parsed — so this only ever returns
456    /// `false` for an authoritative "absent" signal:
457    ///
458    /// - **BTI ("da")** readers have no bloom filter; the Partitions.db trie is the
459    ///   authoritative present/absent oracle. A trie miss (`Ok(None)`) is definitive
460    ///   absence. A trie hit may be a prefix-collision candidate, which is a safe
461    ///   *false positive* here (the partition scan re-verifies the key). Any trie
462    ///   parse error is treated conservatively as "maybe present".
463    /// - **BIG-format** readers consult the bloom filter, which never reports false
464    ///   negatives: `might_contain == false` is definitive absence. With no bloom
465    ///   filter loaded we cannot prune, so we conservatively return `true`.
466    ///
467    /// `partition_key` must be the raw partition-key bytes (same encoding the bloom
468    /// filter and Index.db/BTI trie are keyed on).
469    pub fn might_contain_partition(&self, partition_key: &[u8]) -> bool {
470        use crate::observability::{self as obs, catalog};
471
472        if self.bti_partitions_db.is_some() {
473            // Issue #2163 (roborev r7): restore the C5 range short-circuit
474            // (`locate`'s Step 1) ahead of the trie probe. The r2 refactor
475            // removed `locate_encoded` (replaced by routing the candidate-prune
476            // path through these `might_contain_partition[_encoded]` helpers)
477            // and dropped this step for BTI, so an out-of-range key paid a
478            // pointless trie descent and never recorded
479            // `read_work_counters::record_range_short_circuit`. A no-op for a
480            // genuine BTI reader today (no Summary.db, so
481            // `partition_key_out_of_range` always returns `false`), but this
482            // restores exact step-ordering parity with `locate` for any
483            // degraded case and keeps the work-counter accounting consistent.
484            if self.partition_key_out_of_range(partition_key) {
485                crate::storage::sstable::read_work_counters::record_range_short_circuit();
486                return false;
487            }
488            // BTI: trie miss is authoritative absence; any error is conservative.
489            // `lookup_partition_via_bti_trie` is the single common path that emits
490            // READ_BLOOM_CHECKS for the BTI presence check — do NOT emit again here
491            // or the metric would be double counted.
492            let present = matches!(
493                self.lookup_partition_via_bti_trie(partition_key),
494                Ok(Some(_)) | Err(_)
495            );
496            if !present {
497                // Definitive trie miss → this SSTable is pruned from the read
498                // (issue #2163).
499                self.emit_sstable_pruned();
500            }
501            return present;
502        }
503        // BIG: only record READ_BLOOM_CHECKS when a bloom filter actually exists.
504        // With no filter loaded we cannot prune, so we conservatively return `true`
505        // WITHOUT recording a check (a no-filter "hit" is not a real bloom check and
506        // would inflate the metric).
507        match &self.bloom_filter {
508            Some(bloom) => {
509                let present = bloom.might_contain(partition_key);
510                obs::add_counter(
511                    catalog::READ_BLOOM_CHECKS,
512                    1,
513                    &[
514                        (
515                            catalog::attr::RESULT,
516                            if present { "hit" } else { "miss" }.into(),
517                        ),
518                        (
519                            catalog::attr::SSTABLE_FORMAT,
520                            self.sstable_format_label().into(),
521                        ),
522                    ],
523                );
524                if !present {
525                    // Bloom `might_contain == false` is a definitive negative →
526                    // this SSTable is pruned from the read (issue #2163).
527                    self.emit_sstable_pruned();
528                }
529                present
530            }
531            None => true,
532        }
533    }

562    /// Candidate-prune presence check using a PRE-ENCODED BTI byte-comparable key,
563    /// so a multi-generation `WHERE pk = ?` read hashes+encodes the key ONCE (the
564    /// encoding is identical for every candidate) instead of once per candidate
565    /// (issue #1575 / C4).
566    ///
567    /// `encoded` MUST equal `encode_partition_key_for_bti_trie(partition_key)`,
568    /// computed once by the caller. For a BTI reader this consults the trie with the
569    /// pre-encoded key (no re-hash); for a BIG reader (no `Partitions.db`) it falls
570    /// back to the raw-key [`might_contain_partition`] path (the bloom filter is
571    /// keyed on the raw key — there is no BTI encoding to hoist), so a mixed or
572    /// non-BTI candidate set stays correct. Semantics match
573    /// [`might_contain_partition`]: `false` is definitive absence; a BTI trie hit may
574    /// be a safe prefix-collision false positive re-verified by the partition scan;
575    /// any trie parse error is treated conservatively as "maybe present".
576    pub fn might_contain_partition_encoded(&self, partition_key: &[u8], encoded: &[u8; 9]) -> bool {
577        if self.bti_partitions_db.is_some() {
578            // Issue #2163 (roborev r7): restore the C5 range short-circuit
579            // (`locate`'s Step 1) ahead of the trie probe — see the identical
580            // restoration + rationale in `might_contain_partition`'s BTI branch.
581            // This is the candidate-prune path's OWN entry point (called from
582            // `SSTableManager::prune_candidates` for every BTI candidate), so
583            // restoring it here is what actually re-covers the multi-generation
584            // prune loop the old `locate_encoded` used to gate.
585            if self.partition_key_out_of_range(partition_key) {
586                crate::storage::sstable::read_work_counters::record_range_short_circuit();
587                return false;
588            }
589            let present = matches!(
590                self.lookup_partition_via_bti_trie_encoded(partition_key, encoded),
591                Ok(Some(_)) | Err(_)
592            );
593            if !present {
594                // Definitive trie miss on the candidate-prune path → SSTable pruned
595                // (issue #2163). The BIG branch below delegates to
596                // `might_contain_partition`, which emits its own prune, so there is
597                // no double count.
598                self.emit_sstable_pruned();
599            }
600            return present;
601        }
602        // BIG: no BTI encoding to hoist — the bloom filter hashes the raw key.
603        self.might_contain_partition(partition_key)
604    }

606    /// Enhanced partition iteration using Summary.db reader
607    ///
608    /// Note: Token-based range queries are not directly supported because Summary.db
609    /// does not store token values (Issue #218). Instead, this iterates all summary
610    /// entries and returns all partition data.
611    ///
612    /// For token-based filtering, compute tokens from partition keys after retrieval.
613    ///
614    /// ## Issue #500: Sequential-scan fallback for writer-produced SSTables
615    ///
616    /// The Summary.db → Index.db → Data.db lookup path depends on Index.db format
617    /// compatibility between writer and reader (digest format vs. raw-key format).
618    /// Locally written SSTables emit raw-key Index.db entries that the reader's
619    /// digest-based parser cannot resolve, so the lookup loop returns 0 entries
620    /// even though Summary.db enumerates the partitions correctly.
621    ///
622    /// When that happens we fall back to `sequential_scan`, which walks Data.db
623    /// directly. For V5CompressedLegacy NB SSTables (the format the writer emits),
624    /// `sequential_scan` uses the chunk-stitching path and returns every partition.
625    pub async fn iterate_all_partitions(&self) -> Result<Vec<(RowKey, ScanRow)>> {
626        // Delegates to the per-call-token sibling with the reader's own field —
627        // byte-identical to the pre-#2346 behaviour of this method (every
628        // pre-existing caller keeps its exact cancellation semantics).
629        self.iterate_all_partitions_cancellable(&self.scan_cancel)
630            .await
631    }
632
633    /// Like [`Self::iterate_all_partitions`], but takes an explicit PER-CALL
634    /// cancellation token instead of always reading the reader's own
635    /// [`SSTableReader::scan_cancel`] field (issue #2346).
636    ///
637    /// A cached/shared `Arc<SSTableReader>` (a future warm-handle registry) has
638    /// no `&mut self` available to set a per-request cancel flag, so this seam
639    /// lets the compaction streaming path
640    /// ([`SSTableReader::stream_all_partitions_for_compaction`]'s non-stitching
641    /// fallback) drive two concurrent enumerations with two INDEPENDENT tokens.
642    /// `iterate_all_partitions` itself is UNCHANGED — it delegates here with
643    /// `&self.scan_cancel`, so every existing caller keeps identical behaviour.
644    pub(crate) async fn iterate_all_partitions_cancellable(
645        &self,
646        scan_cancel: &crate::storage::scan_cancel::ScanCancel,
647    ) -> Result<Vec<(RowKey, ScanRow)>> {
648        // Index-random-read path (issue #2302): when a `Index.db` is present on a
649        // BIG SSTable, enumerate EVERY partition via the full Index.db
650        // partition-offset table instead of the sparse `Summary.db` samples.
651        //
652        // The historical loop walked `Summary.db` (only ~1-in-128 partitions) AND
653        // passed `data_size = 0` to `parse_partition_at_offset` (Index.db never
654        // stores partition size), so it read zero bytes per entry, resolved zero
655        // partitions, and SILENTLY fell back to a full `sequential_scan` on EVERY
656        // read — even with complete, valid components. Each partition's span is
657        // bounded instead by the successor entry's offset (the last by the
658        // data-section end), authoritative on-disk structure rather than a size
659        // guess (issue #28). Index.db offsets are in the UNCOMPRESSED data-section
660        // domain: for an uncompressed reader that equals the raw file domain; for a
661        // compressed reader the per-partition slice is mapped to its covering
662        // compression chunk(s) and decompressed (`read_compressed_offset_window`).
663        if self.index_reader.is_some() && self.bti_partitions_db.is_none() {
664            match self
665                .iterate_all_partitions_via_full_index(scan_cancel)
666                .await?
667            {
668                Some(results) => return Ok(results),
669                None => {
670                    // Present components that did NOT fully resolve: never silent.
671                    tracing::warn!(
672                        "SSTable Index.db is present but iterate_all_partitions could not \
673                         resolve every partition through the index-random-read path; \
674                         falling back to a full sequential scan of Data.db (issue #2302). \
675                         This should not happen for a well-formed BIG SSTable — the emitted \
676                         Index.db/Summary.db may be malformed or the data-section length is \
677                         inconsistent with the index offsets."
678                    );
679                }
680            }
681        } else if self.bti_partitions_db.is_none() {
682            // BIG SSTable with NO usable `index_reader` (this branch is only reached
683            // when `index_reader.is_none()`): the full-index random-read path cannot
684            // run, so every iteration falls into the sequential Data.db scan below — a
685            // real per-read perf cliff. The WARN trigger keys on Index.db availability,
686            // NOT Summary.db presence (roborev job 1612): a present Summary.db only
687            // samples ~1-in-128 partitions and never gates the full-index path, so a
688            // present-Summary/missing-Index pair is the SAME silent degradation this
689            // issue exists to kill and must be named LOUD, never dropped silently into
690            // `sequential_scan`. The sub-cause (failed-to-open vs genuinely absent)
691            // only shapes the message.
692            if self.index_present_but_unloadable {
693                // The sibling Index.db EXISTS on disk but failed to open/parse
694                // (issue #2302, roborev job 1606): distinct from a genuinely absent
695                // file — the component is malformed/truncated.
696                tracing::warn!(
697                    "SSTable Index.db is present on disk but failed to open/parse \
698                     (iterate_all_partitions cannot use the index-random-read path) \
699                     and falls back to a full sequential scan of Data.db (a per-read \
700                     perf cliff, issue #2302). The Index.db component may be malformed \
701                     or truncated — regenerate the SSTable or restore an intact Index.db."
702                );
703            } else {
704                // Index.db genuinely absent. Whether or not Summary.db loaded, the
705                // full-index path is unavailable, so this is still the degraded
706                // full-scan cliff (issues #2295/#2302). Name the absent components so
707                // an operator can spot an incomplete snapshot directory served with
708                // only Data.db (or Data.db + Summary.db) instead of silently paying
709                // the full-scan cost on every read.
710                let missing = if self.summary_reader.is_none() {
711                    "Index.db and Summary.db"
712                } else {
713                    "Index.db"
714                };
715                tracing::warn!(
716                    "SSTable has no usable random-access partition index ({missing} \
717                     absent): iterate_all_partitions falls back to a full sequential \
718                     scan of Data.db, materializing and sorting every partition on \
719                     each read (a per-read perf cliff, issues #2295/#2302). For \
720                     snapshot reads, ensure the snapshot directory holds the full \
721                     SSTable component set (Index.db + Summary.db), not just Data.db."
722                );
723            }
724        }
725
726        // Fallback path: sequential walk of Data.db.
727        // Used when Summary.db is absent OR when the Index.db lookup loop returned
728        // no entries (Issue #500: writer-produced SSTables emit a raw-key Index.db
729        // format the reader's digest-based parser does not resolve).
730        let table_id = self.scan_table_id();
731        let schema = self.schema.as_deref();
732        self.sequential_scan(&table_id, None, None, None, schema, scan_cancel)
733            .await
734    }

736    /// Whether this reader has a **usable random-access partition index** — a
737    /// loaded `Index.db` (BIG format; issue #2302 resolves CQLite-written Index.db
738    /// too) or a `Partitions.db` trie (BTI format) — that lets
739    /// [`iterate_all_partitions`](Self::iterate_all_partitions) and point lookups
740    /// avoid a full sequential Data.db scan.
741    ///
742    /// This is a **capability probe**: it must report `true` only for the exact
743    /// states that route to the non-sequential path. Both point lookups
744    /// (`index_reader.lookup_partition` / the BTI trie) and `iterate_all_partitions`
745    /// (the full-`Index.db` walk / BTI) gate on `index_reader`/`bti_partitions_db`
746    /// — never on `summary_reader`.
747    ///
748    /// Issue #2302 (roborev job 1613): `summary_reader` is DELIBERATELY excluded. A
749    /// `Summary.db` only samples ~1-in-128 partitions and never gates the
750    /// random-access route, so a present-Summary / absent-Index BIG reader is
751    /// degraded — every read falls into `sequential_scan` and WARNs. Including
752    /// `summary_reader` here made the probe report "fast path available" for that
753    /// degraded reader (a fidelity lie). Narrowing to the truly-supporting state
754    /// keeps the probe honest.
755    ///
756    /// Snapshot-completeness probe (issue #2295): a directory served with only
757    /// `Data.db` (its index siblings absent) still opens but reports `false`; a
758    /// complete component set loads `Index.db` (→ `index_reader`) and reports
759    /// `true`. All current callers (this module's own tests + the #2295 snapshot
760    /// test) use this probe to mean "does the fast/index path apply," so a single
761    /// narrowed predicate covers every caller without a split API.
762    pub fn has_partition_index(&self) -> bool {
763        self.index_reader.is_some() || self.bti_partitions_db.is_some()
764    }

766    /// Build the TableId used for fallback `sequential_scan` from header metadata.
767    ///
768    /// The reader populates `header.keyspace` / `header.table_name` from either the
769    /// SSTable header or the parent directory path. When the V5CompressedLegacy
770    /// stitching path is used, table_id matching is skipped, so any non-empty value
771    /// is accepted; for other formats this returns the qualified `keyspace.table`
772    /// form so the scan filter matches.
773    pub(in crate::storage::sstable::reader) fn scan_table_id(&self) -> TableId {
774        let keyspace = &self.header.keyspace;
775        let table_name = &self.header.table_name;
776        if !keyspace.is_empty() && !table_name.is_empty() {
777            TableId::from(format!("{}.{}", keyspace, table_name))
778        } else if !table_name.is_empty() {
779            TableId::from(table_name.as_str())
780        } else {
781            TableId::from("default")
782        }
783    }
