1//! Summary-guided streaming partition enumeration (issue #2412 §C / #2413
2//! Option A, Stage 4).
3//!
4//! The #2361 streaming walk ([`full_index_stream`](super::full_index_stream))
5//! emits each partition as the index resolves it, but sources its entries from
6//! the resident `Vec<PartitionIndexEntry>` — which requires the whole `Index.db`
7//! to be materialised first ([`IndexReader::ensure_materialized`](crate::storage::sstable::index_reader::IndexReader::ensure_materialized),
8//! the ~500MB-resident structure #2385 retired at open). This module sources the
9//! SAME walk from a FORWARD-STREAMED `Index.db`
10//! ([`IndexEntryStream`](crate::storage::sstable::index_reader::IndexEntryStream)),
11//! so a warm BIG reader never materialises the full map (spec R3/R4).
12//!
13//! Two extra wins over the materialising walk:
14//! - **No resident `Vec`.** Entry memory is one refill window, partition-count
15//!   independent — the property the warm registry (Stage 5) relies on.
16//! - **Token pushdown (#2413).** A token-range split begins the walk at the
17//!   `Summary.db` sample covering the range start
18//!   ([`SummaryReader::scan_start_position_for_token`](crate::storage::sstable::summary_reader::SummaryReader::scan_start_position_for_token))
19//!   and stops once the walk passes the range end, so out-of-range partition
20//!   bodies are never read/decoded. A compaction consumer passes no range and
21//!   walks the full ring.
22//!
23//! Walk contract preserved from #2361: the `(token, key)` order guard
24//! ([`check_token_order`](super::full_index_stream::check_token_order)),
25//! per-partition structural coverage
26//! ([`partition_slice_fully_consumed`](SSTableReader::partition_slice_fully_consumed),
27//! Signal B), cancel-aware polling, and the `stream_walk_partitions_parsed`
28//! work-probe. The completeness signal (Signal A, mid-entry truncation) moves to
29//! the streaming terminus ([`IndexEntryStream::truncated_tail`](crate::storage::sstable::index_reader::IndexEntryStream)).
30//!
31//! No-heuristics (issue #28): the start offset is an authoritative `Summary.db`
32//! sample position; every entry is authoritative `Index.db` framing; each
33//! partition body is bounded by the SUCCESSOR entry's offset (last by the
34//! data-section end) — never a guessed size.

45/// Half-open `(start_excl, end_incl]` token bound pushed into the per-SSTable walk
46/// (issue #2413 Option A). Mirrors the flight `TokenFilter` half-open semantics
47/// exactly (including the `start == end` FULL-ring convention, #2228); the flight
48/// crate constructs one of these from its `TokenFilter` and a grid test pins that
49/// the two agree, so the membership rule lives in ONE place.
50#[derive(Debug, Clone, Copy)]
51pub struct ScanTokenBound {
52    /// Exclusive lower bound.
53    pub start_excl: i64,
54    /// Inclusive upper bound.
55    pub end_incl: i64,
56    /// Ring-wraparound segment (`start > end`): keep `token > start || token <= end`.
57    pub wraparound: bool,
58}
59
60impl ScanTokenBound {
61    /// Whether `token` is inside this half-open `(start, end]` range.
62    pub fn contains(&self, token: i64) -> bool {
63        // #2228: equal endpoints denote the FULL ring, not the empty set.
64        if self.start_excl == self.end_incl {
65            return true;
66        }
67        if self.wraparound {
68            token > self.start_excl || token <= self.end_incl
69        } else {
70            token > self.start_excl && token <= self.end_incl
71        }
72    }
73
74    /// Whether every remaining (token-ascending) partition is guaranteed to be
75    /// ABOVE this range, so a forward walk can stop. Only sound for a
76    /// non-wraparound range (a wraparound range has in-range tokens at both ends
77    /// of the ring, so a forward walk cannot early-stop).
78    fn can_stop_past(&self, token: i64) -> bool {
79        !self.wraparound && self.start_excl != self.end_incl && token > self.end_incl
80    }
81}

83impl SSTableReader {
84    /// The `Index.db` path SIBLING of this reader's CURRENT `Data.db` path (issue
85    /// #2412 §C, #2383-aware). Derived from [`Self::file_path`] (the ArcSwap that a
86    /// #2383 inode-rebind updates) by swapping the `-Data.db` component suffix, so a
87    /// fresh streaming `Index.db` open follows a snapshot rebind rather than
88    /// re-opening a torn-down path. Falls back to the reader's recorded open-time
89    /// index path when the `Data.db` name is not the expected `*-Data.db` shape.
90    fn current_index_db_path(&self) -> std::path::PathBuf {
91        let data = self.file_path();
92        // Same `-Data.db` → `-Index.db` sibling rule the #2383 rebind uses to repoint
93        // the lazy IndexReader, so both stay on the rebound generation (issue #2356).
94        crate::storage::sstable::reader::index_db_sibling(&data).unwrap_or_else(|| {
95            self.index_reader
96                .as_ref()
97                .map(|ir| ir.index_path())
98                .unwrap_or(data)
99        })
100    }

102    /// Shared Summary-guided FORWARD `Index.db` walk (issue #2412 §C / #2413
103    /// Option A): stream in-range partition SLICES from a `Summary.db`-guided start
104    /// offset WITHOUT materialising the resident partition map, invoking `decode`
105    /// on each in-range partition's exact `Data.db` byte slice.
106    ///
107    /// Both emit shapes are built on this: the read-shadowing ScanRow decoder
108    /// ([`stream_partitions_summary_guided`](Self::stream_partitions_summary_guided),
109    /// the non-stitching `V5_0Uncompressed` model) and the full-fidelity
110    /// CompactionRow decoder
111    /// ([`stream_partitions_summary_guided_compaction`](Self::stream_partitions_summary_guided_compaction),
112    /// the chunk-stitching `nb` merge model). Factoring the walk here means the
113    /// order guard, coverage check, work-probe, and range/terminus logic are
114    /// defined ONCE — the two decoders differ only in how a proven-complete slice
115    /// becomes rows.
116    ///
117    /// `token_bound`: `None` = full ring from offset 0; `Some(bound)` = begin at
118    /// the sample covering `start_excl` and stop once past `end_incl`.
119    ///
120    /// Returns [`FullIndexStreamOutcome::FellBack`] (nothing emitted) when there is
121    /// no usable summary/index to stream from. Any inconsistency AFTER the first
122    /// emit is a hard [`Error`] (fail-closed): a streaming walk cannot fall back.
123    async fn walk_in_range_partition_slices<D>(
124        &self,
125        scan_cancel: &ScanCancel,
126        token_bound: Option<ScanTokenBound>,
127        parser: &crate::storage::sstable::reader::parsing::V5CompressedLegacyParser,
128        schema: Option<&crate::schema::TableSchema>,
129        decode: &mut D,
130    ) -> Result<FullIndexStreamOutcome>
131    where
132        D: FnMut(&[u8]) -> Result<ControlFlow<()>>,
133    {
134        let (Some(_index_reader), Some(summary)) =
135            (self.index_reader.as_ref(), self.summary_reader.as_ref())
136        else {
137            return Ok(FullIndexStreamOutcome::FellBack);
138        };
139        if summary.get_entries().is_empty() {
140            return Ok(FullIndexStreamOutcome::FellBack);
141        }
142
143        // Exclusive end of the last partition (uncompressed data-section domain),
144        // identical to the materialising walk's derivation.
145        let data_section_end = match self.compression_info.as_deref() {
146            Some(ci) => ci.data_length,
147            None => self
148                .stats
149                .file_size
150                .saturating_sub(self.actual_header_size as u64),
151        };
152        if data_section_end == 0 {
153            return Ok(FullIndexStreamOutcome::FellBack);
154        }
155
156        // Summary-guided start offset: the covering sample for a token range's
157        // exclusive lower bound, else the index start for a full scan.
158        //
159        // WRAPAROUND (roborev endgame finding, High — silent data loss): a
160        // wraparound range (`start_excl > end_incl`, the ring segment crossing the
161        // min-token boundary) is `(start_excl, MAX] ∪ [MIN, end_incl]` — its
162        // in-range partitions live in TWO disjoint physical regions of the
163        // token-ordered `Index.db`: a HIGH-token tail (`token > start_excl`) and a
164        // LOW-token head (`token <= end_incl`). This walk is a single FORWARD pass
165        // to EOF (`Index.db` is not circular), so starting at
166        // `scan_start_position_for_token(start_excl)` (the HIGH segment's start)
167        // can NEVER reach the LOW segment's entries — they physically precede that
168        // offset and are silently skipped entirely. For a wraparound range the walk
169        // MUST start at the index's true beginning (offset 0) so BOTH segments are
170        // reachable; `ScanTokenBound::contains` (the per-entry filter below) already
171        // selects exactly the two segments, and `can_stop_past` is unconditionally
172        // `false` for `wraparound` (verified: the walk never early-stops before
173        // EOF), so the pair is coherent — a full walk, filtered.
174        let start_position = match token_bound {
175            Some(bound) if bound.wraparound => 0,
176            Some(bound) => summary.scan_start_position_for_token(bound.start_excl),
177            None => 0,
178        };
179
180        // Derive the `Index.db` path from the reader's CURRENT (possibly
181        // #2383-rebound) `Data.db` path, NOT the index_reader's recorded open-time
182        // path: a warm reader rebound across a snapshot teardown keeps its already
183        // -open `Data.db` FD (valid even after the snapshot dir is cleared) and
184        // swaps only `file_path`, but the streaming index read does a FRESH
185        // `File::open` — so it must follow the rebind or ENOENT on the dead
186        // snapshot path (#2352). Snapshot components are same-inode hardlinks, so
187        // the rebound path's `Index.db` is byte-identical to the open-time one.
188        let index_db_path = self.current_index_db_path();
189        let mut stream = crate::storage::sstable::index_reader::IndexEntryStream::open(
190            &index_db_path,
191            start_position,
192        )
193        .await?;
194
195        // One-entry lookahead so each partition body is bounded by its SUCCESSOR's
196        // offset (the last by the data-section end), exactly as the materialising
197        // walk bounds `entries[i]` by `entries[i+1]`.
198        let mut prev_key: Option<(i64, Vec<u8>)> = None;
199        let mut index = 0usize;
200        let mut emitted_any = false;
201        let Some(mut current) = stream.next().await? else {
202            // Zero entries from the start offset: nothing in range (or empty).
203            return Ok(FullIndexStreamOutcome::Streamed);
204        };
205
206        loop {
207            scan_cancel.check()?;
208
209            let cur_key: Vec<u8> = current
210                .raw_key
211                .as_deref()
212                .map(|k| k.to_vec())
213                .unwrap_or_default();
214            // Empty partition keys round-trip corrupt through BOTH read paths
215            // (issue #2302/#2325): before any emit, fall back safely; after an
216            // emit, fail closed rather than feed the merger a corrupt shape.
217            if cur_key.is_empty() {
218                if emitted_any {
219                    return Err(Error::corruption(
220                        "walk_in_range_partition_slices: empty partition key mid-walk \
221                         (issue #2302/#2325)"
222                            .to_string(),
223                    ));
224                }
225                return Ok(FullIndexStreamOutcome::FellBack);
226            }
227
228            let token = cassandra_murmur3_token(&cur_key);
229
230            // (token, key) order guard — the merge-input contract (issue #2361).
231            check_token_order(
232                prev_key.as_ref().map(|(t, k)| (*t, k.as_slice())),
233                token,
234                &cur_key,
235                index,
236            )?;
237
238            // Read the successor to bound the current partition (and to decide the
239            // range stop). The successor of the LAST entry is the data-section end.
240            let next = stream.next().await?;
241            let end = next
242                .as_ref()
243                .map(|n| n.data_offset)
244                .unwrap_or(data_section_end);
245
246            // Token pushdown (#2413): decode only in-range bodies.
247            let in_range = token_bound.map(|b| b.contains(token)).unwrap_or(true);
248            if in_range {
249                let start = current.data_offset;
250                if end <= start {
251                    return Err(Error::corruption(format!(
252                        "walk_in_range_partition_slices: non-ascending offset at entry \
253                         {index} (offset {start} >= successor {end}, issue #2412)"
254                    )));
255                }
256                let span = end - start;
257                let Ok(size) = u32::try_from(span) else {
258                    return Err(Error::corruption(format!(
259                        "walk_in_range_partition_slices: partition {index} span {span} \
260                         overflows u32 (issue #2412)"
261                    )));
262                };
263
264                let raw = if let Some(ci) = self.compression_info.as_deref() {
265                    self.read_compressed_offset_window(ci, start, size).await?
266                } else {
267                    let absolute_offset = start + self.actual_header_size as u64;
268                    self.read_uncompressed_verified(&self.file, absolute_offset, size as usize)
269                        .await?
270                };
271
272                // Structural coverage (Signal B): the slice must decode as exactly
273                // one complete partition. Mid-walk failure = fail-closed.
274                if !self.partition_slice_fully_consumed(parser, &raw, schema)? {
275                    return Err(Error::corruption(format!(
276                        "walk_in_range_partition_slices: partition {index} slice not fully \
277                         consumed (truncated/corrupt body, issue #2412)"
278                    )));
279                }
280
281                // Work-probe (issue #2398): one partition body decoded. A narrow
282                // token range must keep this near its in-range slice, not O(all).
283                crate::storage::sstable::work_counters::add_stream_walk_partition_parsed();
284                emitted_any = true;
285                match decode(&raw)? {
286                    ControlFlow::Continue(()) => {}
287                    ControlFlow::Break(()) => return Ok(FullIndexStreamOutcome::Streamed),
288                }
289            }
290
291            // Advance. Stop early once a non-wraparound range is fully past its end.
292            if let Some(bound) = token_bound {
293                if bound.can_stop_past(token) {
294                    return Ok(FullIndexStreamOutcome::Streamed);
295                }
296            }
297            match next {
298                Some(n) => {
299                    prev_key = Some((token, cur_key));
300                    current = n;
301                    index += 1;
302                }
303                None => break,
304            }
305        }
306
307        // Terminus completeness (Signal A, issue #2302): for a FULL scan a
308        // mid-entry-truncated `Index.db` tail must be detectable — after an emit
309        // the streaming walk cannot fall back, so this is a hard fail-closed error
310        // (never a silent under-enumeration). A token-range scan does not enumerate
311        // the tail, so a truncated tail beyond its range is not its concern.
312        if token_bound.is_none() && stream.truncated_tail() {
313            return Err(Error::corruption(
314                "walk_in_range_partition_slices: Index.db tail truncated mid-entry — \
315                 full-scan enumeration is incomplete (Signal A, issue #2302/#2412)"
316                    .to_string(),
317            ));
318        }
319
320        Ok(FullIndexStreamOutcome::Streamed)
321    }

323    /// Read-shadowing ScanRow decoder over the shared walk (the non-stitching
324    /// `V5_0Uncompressed` model): each in-range partition slice is decoded with the
325    /// read-shadowing parser (`build_v5_parser(true)`) and its surviving rows are
326    /// emitted as `(RowKey, ScanRow)` — byte-identical to
327    /// [`stream_all_partitions_via_full_index`](Self::iterate_all_partitions_via_full_index)'s
328    /// per-partition emit, but streamed (no resident `Vec`) + token-scoped.
329    pub(in crate::storage::sstable::reader) async fn stream_partitions_summary_guided<F>(
330        &self,
331        scan_cancel: &ScanCancel,
332        token_bound: Option<ScanTokenBound>,
333        emit: &mut F,
334    ) -> Result<FullIndexStreamOutcome>
335    where
336        F: FnMut((RowKey, ScanRow)) -> Result<ControlFlow<()>>,
337    {
338        let parser = self.build_v5_parser(true);
339        let reader_schema = self.get_table_schema(None);
340        let schema = reader_schema.as_ref();
341        self.walk_in_range_partition_slices(scan_cancel, token_bound, &parser, schema, &mut |raw| {
342            let parsed = parser.parse_block(raw, schema, self)?;
343            for (_table_id, row_key, value) in parsed {
344                if self.filter_tombstone(&value) {
345                    match emit((row_key, value))? {
346                        ControlFlow::Continue(()) => {}
347                        ControlFlow::Break(()) => return Ok(ControlFlow::Break(())),
348                    }
349                }
350            }
351            Ok(ControlFlow::Continue(()))
352        })
353        .await
354    }

356    /// Full-fidelity CompactionRow decoder over the shared walk (the chunk-stitching
357    /// `nb` merge model): each in-range partition slice is decoded with the
358    /// compaction parser (`build_v5_parser(false)`) via
359    /// `parse_one_partition_for_compaction` — byte-identical to
360    /// [`drain_compaction_window`](Self::stream_all_partitions_for_compaction)'s
361    /// per-partition [`CompactionRow`](super::super::compaction_row::CompactionRow)
362    /// emit (preserving cell timestamps / tombstone markers the k-way merger's LWW
363    /// reconciliation needs), but streamed + token-scoped. `at_final_chunk = true`
364    /// because `raw` is exactly one complete, coverage-proven partition slice.
365    async fn stream_partitions_summary_guided_compaction<F>(
366        &self,
367        scan_cancel: &ScanCancel,
368        token_bound: Option<ScanTokenBound>,
369        schema: Option<&crate::schema::TableSchema>,
370        emit: &mut F,
371    ) -> Result<FullIndexStreamOutcome>
372    where
373        F: FnMut(super::super::compaction_row::CompactionRow) -> Result<ControlFlow<()>>,
374    {
375        use crate::storage::sstable::reader::parsing::ParseStep;
376        let parser = self.build_v5_parser(false);
377        let owned_schema = schema.cloned().or_else(|| self.get_table_schema(None));
378        let sch = owned_schema.as_ref();
379        self.walk_in_range_partition_slices(scan_cancel, token_bound, &parser, sch, &mut |raw| {
380            let mut broke = false;
381            let step = parser.parse_one_partition_for_compaction(
382                raw,
383                sch,
384                self,
385                true,
386                &mut |row: super::super::compaction_row::CompactionRow| match emit(row)? {
387                    ControlFlow::Continue(()) => Ok(ControlFlow::Continue(())),
388                    ControlFlow::Break(()) => {
389                        broke = true;
390                        Ok(ControlFlow::Break(()))
391                    }
392                },
393            )?;
394            // The coverage check already proved `raw` decodes as exactly one
395            // complete partition, so a non-`Emitted` step here is a structural
396            // inconsistency — fail closed rather than silently drop the partition.
397            if !matches!(step, ParseStep::Emitted(_)) {
398                return Err(Error::corruption(
399                    "stream_partitions_summary_guided_compaction: coverage-proven slice \
400                     did not decode to a terminated partition (issue #2412)"
401                        .to_string(),
402                ));
403            }
404            if broke {
405                return Ok(ControlFlow::Break(()));
406            }
407            Ok(ControlFlow::Continue(()))
408        })
409        .await
410    }

412    /// Query-serve partition enumeration for the WARM reader-based merge (issue
413    /// #2412 §C / #2413 Option A) — the analogue of
414    /// [`stream_all_partitions_for_compaction`](Self::stream_all_partitions_for_compaction)
415    /// that the flight `do_get` warm path drives (`from_readers::drive_query_stream`).
416    ///
417    /// Routing MIRRORS `stream_all_partitions_for_compaction`'s per-mechanism emit
418    /// shape so the merger reconciles identically, but STREAMS the index (no
419    /// resident `Vec`, spec R4) and pushes the token range into the walk (#2413):
420    /// - Chunk-stitching (`nb`) readers use the full-fidelity CompactionRow decoder
421    ///   ([`stream_partitions_summary_guided_compaction`](Self::stream_partitions_summary_guided_compaction)) —
422    ///   byte-identical to `drain_compaction_window`'s emit, so cross-generation
423    ///   LWW reconciliation is preserved.
424    /// - Non-stitching (`V5_0Uncompressed`) readers use the read-shadowing ScanRow
425    ///   decoder ([`stream_partitions_summary_guided`](Self::stream_partitions_summary_guided)) —
426    ///   byte-identical to `stream_all_partitions_cancellable`'s emit.
427    ///
428    /// A `FellBack` (no usable summary/index) routes on to the full-ring
429    /// `stream_all_partitions_for_compaction`. Compaction consumers do NOT call
430    /// this (they use the path-based stream directly, no token range), so they keep
431    /// full-ring byte-parity walks unchanged (spec R3).
432    pub async fn stream_all_partitions_for_query<F>(
433        &self,
434        schema: Option<&crate::schema::TableSchema>,
435        scan_cancel: &ScanCancel,
436        token_bound: Option<ScanTokenBound>,
437        mut emit: F,
438    ) -> Result<()>
439    where
440        F: FnMut(super::super::compaction_row::CompactionRow) -> Result<ControlFlow<()>>,
441    {
442        let summary_usable = self
443            .summary_reader
444            .as_ref()
445            .map(|s| !s.get_entries().is_empty())
446            .unwrap_or(false);
447        if self.index_reader.is_some() && self.bti_partitions_db.is_none() && summary_usable {
448            let outcome = if self.requires_chunk_stitching() {
449                // `nb` merge model: full-fidelity CompactionRows (cell timestamps /
450                // tombstone markers preserved for the merger's LWW reconciliation).
451                self.stream_partitions_summary_guided_compaction(
452                    scan_cancel,
453                    token_bound,
454                    schema,
455                    &mut emit,
456                )
457                .await?
458            } else {
459                // `V5_0Uncompressed` model: read-shadowing rows adapted to
460                // CompactionRow (row_timestamp 0), matching the non-stitching
461                // compaction stream's emit exactly.
462                self.stream_partitions_summary_guided(scan_cancel, token_bound, &mut |(k, v)| {
463                    let row =
464                        super::super::compaction_row::CompactionRow::from_legacy_value(k, v, 0);
465                    emit(row)
466                })
467                .await?
468            };
469            if matches!(outcome, FullIndexStreamOutcome::Streamed) {
470                return Ok(());
471            }
472            tracing::warn!(
473                "stream_all_partitions_for_query: Summary-guided streaming fell back \
474                 (no usable Summary.db/Index.db to stream); using the full-ring \
475                 compaction stream (issue #2412)."
476            );
477        }
478        // Full-ring fallback (no usable summary/index): the downstream token filter
479        // still bounds the result set, so correctness holds without pushdown.
480        self.stream_all_partitions_for_compaction(schema, scan_cancel, emit)
481            .await
482    }
483}

489    #[test]
490    fn contains_non_wraparound_half_open() {
491        let b = ScanTokenBound {
492            start_excl: 10,
493            end_incl: 20,
494            wraparound: false,
495        };
496        assert!(!b.contains(10), "start is exclusive");
497        assert!(b.contains(11));
498        assert!(b.contains(20), "end is inclusive");

506            end_incl: 5,
507            wraparound: false,
508        };
509        for t in [i64::MIN, -1, 0, 5, 6, i64::MAX] {

531            wraparound: false,

535        let wrap = ScanTokenBound {
