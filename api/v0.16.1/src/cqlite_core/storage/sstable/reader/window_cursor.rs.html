1//! Sliding decompressed-byte window helpers (issue #1589; enum-backed
2//! borrow API added issue #1644 K5 stage 2, design D1).
3//!
4//! Both the user-facing windowed scan (`scan_stream_windowed`) and the streaming
5//! compaction driver (`data_access::compaction`) keep a window of decompressed
6//! bytes, parse CONFIRMED partitions from the FRONT, and refill from the tail as
7//! new chunks arrive. This module hosts the test-only byte-movement [`probe`]
8//! (issue #1589), the [`WindowCursor`] cursor that bounds per-window memmove to
9//! Θ(W), and (issue #1644) the [`Bytes`]-native borrow path that lets a value
10//! fully inside one chunk be materialized as a refcounted view instead of a copy.
11
12use bytes::Bytes;
13use std::ops::Range;
14
15/// A payload materialized through the window borrow API (issue #1644, D1).
16///
17/// Both variants carry a [`Bytes`] payload; the distinction exists so the
18/// window-borrow work-probe can record how many bytes were physically COPIED
19/// (`Owned`) vs BORROWED (zero-copy refcount view, `Borrowed`) — the #1644
20/// acceptance proof (scan-window-borrow spec, "bytes copied into values").
21#[derive(Debug, Clone)]
22pub(crate) enum ValueBytes {
23    /// A refcounted view of the parent chunk's `Bytes` — no allocation.
24    Borrowed(Bytes),
25    /// A standalone copy: either the range straddled a chunk boundary (no
26    /// single parent `Bytes` to slice — correctness over borrow, D1), or the
27    /// window had no active `Bytes` backing (e.g. the `Vec<u8>`-based
28    /// compaction driver, which never adopted the #1940 `Bytes` substrate).
29    Owned(Bytes),
30}
31
32impl ValueBytes {
33    /// Take the payload, discarding the borrowed/owned distinction.
34    pub(crate) fn into_bytes(self) -> Bytes {
35        match self {
36            ValueBytes::Borrowed(b) | ValueBytes::Owned(b) => b,
37        }
38    }
39}

41/// Shared pointer-range check (issue #1644, D1): is `sub` a subslice of
42/// `base`? If so, calls `on_borrow(offset, len)` to produce a zero-copy
43/// [`ValueBytes::Borrowed`]; otherwise falls back to an owned copy of `sub`.
44///
45/// Used by BOTH [`WindowCursor::borrow_subslice`] (against `self.as_slice()`,
46/// when the window's backing is `Backing::Borrowed`) and
47/// [`super::value_borrow::borrow_active`] (against the active window's raw
48/// `Bytes`, installed via a thread-local for the streaming decode call graph)
49/// — the ONE implementation of the pointer-range check, per D1's "localize the
50/// borrow/copy decision to one place". The check is a pointer-VALUE
51/// comparison only (`as usize` casts, never a dereference outside `sub`'s own
52/// valid range), so it can only ever choose borrow-vs-copy, never affect
53/// memory safety: the copy fallback dereferences only `sub` itself, a slice
54/// the caller already legitimately holds.
55pub(crate) fn borrow_bytes_subslice(
56    base: &[u8],
57    sub: &[u8],
58    on_borrow: impl FnOnce(usize, usize) -> ValueBytes,
59) -> ValueBytes {
60    if sub.is_empty() {
61        return ValueBytes::Owned(Bytes::new());
62    }
63    let base_start = base.as_ptr() as usize;
64    let base_end = base_start + base.len();
65    let sub_start = sub.as_ptr() as usize;
66    let sub_end = sub_start.saturating_add(sub.len());
67    if sub_start >= base_start && sub_end <= base_end {
68        on_borrow(sub_start - base_start, sub.len())
69    } else {
70        #[cfg(feature = "scan-offload-probe")]
71        probe::note_bytes_copied_into_value(sub.len());
72        ValueBytes::Owned(Bytes::copy_from_slice(sub))
73    }
74}

76/// The window's backing storage (issue #1644, D1).
77enum Backing {
78    /// The logical window IS exactly one chunk's `Bytes` (a cursor over it,
79    /// `start` tracks consumption). The steady state: the previous chunk was
80    /// fully consumed before the next arrived, so `refill_owned` REPLACED the
81    /// backing with the incoming `Bytes` by move + refcount — no copy.
82    Borrowed(Bytes),
83    /// A partition straddled a chunk boundary, leaving a residual unconsumed
84    /// tail that had to be concatenated with the next chunk into an owned
85    /// buffer — the pre-#1644 copy path. Also the ONLY state the `Vec<u8>`-
86    /// sourced compaction driver ever uses (it has no `Bytes` to move).
87    Stitched(Vec<u8>),
88}
89
90/// A decompressed-byte window with a FRONT CURSOR (issue #1589) and a
91/// [`Bytes`]-native borrow path (issue #1644).
92///
93/// Invariant: `start <= backing.len()`. The logical window (the bytes not yet
94/// consumed) is `backing[start..]`. [`consume`](Self::consume) advances `start`
95/// with NO memmove; a refill ([`refill`](Self::refill) /
96/// [`refill_owned`](Self::refill_owned)) reclaims the consumed prefix with a
97/// SINGLE compaction — once per refill, not once per confirmed partition —
98/// then appends the new chunk. Each byte is therefore physically moved at most
99/// once per refill it survives.
100///
101/// [`as_slice`](Self::as_slice) yields exactly the same unconsumed bytes
102/// regardless of backing — the parser's scanning/boundary logic is UNCHANGED;
103/// only value-materialization call sites use [`borrow`](Self::borrow) /
104/// [`borrow_subslice`](Self::borrow_subslice).
105pub(crate) struct WindowCursor {
106    backing: Backing,
107    start: usize,
108}
109
110impl WindowCursor {
111    /// A new, empty window.
112    pub(crate) fn new() -> Self {
113        Self {
114            backing: Backing::Stitched(Vec::new()),
115            start: 0,
116        }
117    }
118
119    /// A cheap, immutable SNAPSHOT view over `bytes` alone — used by
120    /// [`super::value_borrow::ActiveWindowGuard`] to install the current
121    /// window's active `Bytes` (issue #1644, D1) as a small standalone
122    /// `WindowCursor` a decode site can call [`borrow_subslice`](Self::borrow_subslice)
123    /// on, without holding a live reference to the real scanning window (whose
124    /// `start` cursor keeps advancing as decode proceeds — the snapshot's own
125    /// `start` stays 0, matching `bytes`'s full extent).
126    pub(crate) fn from_borrowed_bytes(bytes: Bytes) -> Self {
127        Self {
128            backing: Backing::Borrowed(bytes),
129            start: 0,
130        }
131    }
132
133    #[inline]
134    fn backing_slice(&self) -> &[u8] {
135        match &self.backing {
136            Backing::Borrowed(b) => b,
137            Backing::Stitched(v) => v,
138        }
139    }
140
141    /// The unconsumed bytes — the logical window the parser reads. Cursor-relative.
142    #[inline]
143    pub(crate) fn as_slice(&self) -> &[u8] {
144        &self.backing_slice()[self.start..]
145    }
146
147    /// Number of unconsumed bytes remaining in the logical window.
148    #[inline]
149    pub(crate) fn len(&self) -> usize {
150        self.backing_slice().len() - self.start
151    }
152
153    /// Whether the logical window is empty (all appended bytes consumed).
154    #[inline]
155    pub(crate) fn is_empty(&self) -> bool {
156        self.start >= self.backing_slice().len()
157    }
158
159    /// Consume `n` bytes from the FRONT of the logical window by advancing the
160    /// cursor — NO memmove. `n` is clamped to the unconsumed length, exactly
161    /// matching the prior `drain(0..n.min(len))` semantics (a parser reporting
162    /// `consumed > remaining` cannot advance the cursor past the window end).
163    #[inline]
164    pub(crate) fn consume(&mut self, n: usize) {
165        let capped = n.min(self.len());
166        self.start += capped;
167    }
168
169    /// The current window's `Bytes` backing when it is a single, un-stitched
170    /// chunk (issue #1644, D1) — `None` when the window is
171    /// [`Backing::Stitched`] (a straddle occurred, or this window has never
172    /// received a `Bytes`-sourced refill, e.g. the compaction driver). A clone
173    /// is a refcount bump, not a copy: cheap to call once per partition-parse
174    /// to install the active borrow source (`value_borrow`).
175    pub(crate) fn active_bytes(&self) -> Option<Bytes> {
176        match &self.backing {
177            Backing::Borrowed(b) => Some(b.clone()),
178            Backing::Stitched(_) => None,
179        }
180    }
181
182    /// Borrow the byte range `range` (relative to the LOGICAL window — the same
183    /// coordinate space as `as_slice()`) as a [`ValueBytes`] (issue #1644, D1).
184    ///
185    /// - [`Backing::Borrowed`] → a zero-copy [`Bytes::slice`] view (refcount
186    ///   bump only).
187    /// - [`Backing::Stitched`] → an owned copy: a straddling value has no
188    ///   single parent `Bytes` to slice from, so correctness takes the copy
189    ///   (D1 — never sacrifice correctness to avoid a copy).
190    pub(crate) fn borrow(&self, range: Range<usize>) -> ValueBytes {
191        let abs_start = self.start + range.start;
192        let abs_end = self.start + range.end;
193        match &self.backing {
194            Backing::Borrowed(b) => ValueBytes::Borrowed(b.slice(abs_start..abs_end)),
195            Backing::Stitched(v) => {
196                #[cfg(feature = "scan-offload-probe")]
197                probe::note_bytes_copied_into_value(abs_end.saturating_sub(abs_start));
198                ValueBytes::Owned(Bytes::copy_from_slice(&v[abs_start..abs_end]))
199            }
200        }
201    }
202
203    /// Borrow `sub` — a byte slice a caller holds that is EXPECTED to be a
204    /// subslice of `self.as_slice()` — as a [`ValueBytes`] (issue #1644, D1).
205    ///
206    /// Delegates to [`borrow_bytes_subslice`], the shared pointer-range check
207    /// also used by [`super::value_borrow`] (the active-window decode-borrow
208    /// source), against `self.as_slice()`.
209    pub(crate) fn borrow_subslice(&self, sub: &[u8]) -> ValueBytes {
210        borrow_bytes_subslice(self.as_slice(), sub, |offset, len| {
211            self.borrow(offset..offset + len)
212        })
213    }
214
215    /// Reclaim the consumed prefix (single compaction) so a fresh chunk can be
216    /// appended at offset 0, returning an owned buffer holding ONLY the
217    /// unconsumed residual at offset 0. Shared by both refill entry points below.
218    ///
219    /// When the backing is already an owned [`Backing::Stitched`] `Vec`, the
220    /// residual is compacted IN PLACE (`copy_within` + `truncate`), REUSING the
221    /// existing allocation — no fresh `Vec` per refill (the pre-#1644
222    /// allocation pattern). A [`Backing::Borrowed`] chunk has no reusable owned
223    /// `Vec`, so its residual is copied into a fresh one.
224    fn take_residual_buf(&mut self) -> Vec<u8> {
225        let start = self.start;
226        let residual_len = self.len();
227        #[cfg(feature = "scan-offload-probe")]
228        if start > 0 {
229            probe::note_bytes_memmoved(residual_len);
230        }
231        match std::mem::replace(&mut self.backing, Backing::Stitched(Vec::new())) {
232            Backing::Stitched(mut buf) => {
233                if start > 0 {
234                    buf.copy_within(start.., 0);
235                }
236                buf.truncate(residual_len);
237                buf
238            }
239            Backing::Borrowed(b) => {
240                let mut buf = Vec::with_capacity(residual_len);
241                buf.extend_from_slice(&b[start..]);
242                buf
243            }
244        }
245    }
246
247    /// Append a freshly decompressed `chunk` sourced as an OWNED, refcounted
248    /// [`Bytes`] (issue #1644 — the #1940 substrate; `scan_stream_windowed`'s
249    /// Bytes-sourced IO half).
250    ///
251    /// When the window is fully consumed (`self.is_empty()`, the steady state:
252    /// the previous chunk was drained before this one arrived), REPLACES the
253    /// backing directly with `chunk` — a move + refcount bump, NOT a copy —
254    /// switching to [`Backing::Borrowed`] so subsequent `borrow`/`borrow_subslice`
255    /// calls for this chunk are zero-copy. Otherwise (a residual straddling
256    /// tail survives) stitches: copies the residual + `chunk` into one owned
257    /// `Vec<u8>` ([`Backing::Stitched`]) — a straddling value has no single
258    /// parent `Bytes`, so correctness requires the copy fallback (D1).
259    pub(crate) fn refill_owned(&mut self, chunk: Bytes) {
260        #[cfg(feature = "scan-offload-probe")]
261        probe::note_bytes_appended(chunk.len());
262        if self.is_empty() {
263            self.backing = Backing::Borrowed(chunk);
264            self.start = 0;
265            return;
266        }
267        let mut residual = self.take_residual_buf();
268        residual.reserve(chunk.len());
269        residual.extend_from_slice(&chunk);
270        self.backing = Backing::Stitched(residual);
271        self.start = 0;
272    }
273
274    /// Append a freshly decompressed `chunk` sourced as a borrowed `&[u8]`
275    /// (the `Vec<u8>`-based compaction driver, which never adopted the #1940
276    /// `Bytes` substrate). ALWAYS copies — there is no `Bytes` to move — so the
277    /// window ends up [`Backing::Stitched`] regardless of prior state. Like the
278    /// pre-#1644 path it compacts the residual IN PLACE and appends into the
279    /// reused buffer (see [`take_residual_buf`](Self::take_residual_buf)) — no
280    /// fresh allocation per refill. See [`refill_owned`](Self::refill_owned) for
281    /// the Bytes-sourced zero-copy-refill fast path.
282    pub(crate) fn refill(&mut self, chunk: &[u8]) {
283        #[cfg(feature = "scan-offload-probe")]
284        probe::note_bytes_appended(chunk.len());
285        let mut residual = self.take_residual_buf();
286        residual.reserve(chunk.len());
287        residual.extend_from_slice(chunk);
288        self.backing = Backing::Stitched(residual);
289        self.start = 0;
290    }
291}

293/// Test-only probe (issue #1589) for the sliding window's byte-movement cost;
294/// extended (issue #1644) with a bytes-copied-into-VALUES counter — the
295/// scan-window-borrow acceptance proof (bytes-copied-into-values ≈ 0 on the
296/// non-straddling borrow path).
297///
298/// Records the total bytes physically memmoved by the window's consume/compact
299/// path, the total decompressed bytes appended, and the total bytes copied
300/// (rather than borrowed) into a materialized value. Compiled ONLY under the
301/// non-default `scan-offload-probe` feature, so it adds zero cost and no public
302/// surface in normal/release builds.
303#[cfg(feature = "scan-offload-probe")]
304#[doc(hidden)]
305pub mod probe {
306    use std::sync::atomic::{AtomicBool, AtomicUsize, Ordering};
307
308    static ARMED: AtomicBool = AtomicBool::new(false);
309    static BYTES_MEMMOVED: AtomicUsize = AtomicUsize::new(0);
310    static BYTES_APPENDED: AtomicUsize = AtomicUsize::new(0);
311    static BYTES_COPIED_INTO_VALUES: AtomicUsize = AtomicUsize::new(0);
312
313    /// Arm the probe and reset the counters. Call before driving a scan/compaction.
314    pub fn arm() {
315        BYTES_MEMMOVED.store(0, Ordering::SeqCst);
316        BYTES_APPENDED.store(0, Ordering::SeqCst);
317        BYTES_COPIED_INTO_VALUES.store(0, Ordering::SeqCst);
318        ARMED.store(true, Ordering::SeqCst);
319    }
320
321    /// Disarm the probe (restores the production no-op state).
322    pub fn disarm() {
323        ARMED.store(false, Ordering::SeqCst);
324    }
325
326    /// Record `n` bytes physically memmoved while consuming/compacting the window.
327    pub(crate) fn note_bytes_memmoved(n: usize) {
328        if ARMED.load(Ordering::Relaxed) {
329            BYTES_MEMMOVED.fetch_add(n, Ordering::Relaxed);
330        }
331    }
332
333    /// Record `n` bytes appended (decompressed) into the window, if armed.
334    pub(crate) fn note_bytes_appended(n: usize) {
335        if ARMED.load(Ordering::Relaxed) {
336            BYTES_APPENDED.fetch_add(n, Ordering::Relaxed);
337        }
338    }
339
340    /// Record `n` bytes COPIED (not borrowed) into a materialized value's
341    /// `Bytes` payload — the #1644 window-borrow work-probe.
342    pub(crate) fn note_bytes_copied_into_value(n: usize) {
343        if ARMED.load(Ordering::Relaxed) {
344            BYTES_COPIED_INTO_VALUES.fetch_add(n, Ordering::Relaxed);
345        }
346    }
347
348    /// Total bytes physically memmoved by the window during the armed run.
349    pub fn recorded_bytes_memmoved() -> usize {
350        BYTES_MEMMOVED.load(Ordering::Relaxed)
351    }
352
353    /// Total decompressed bytes appended into the window during the armed run.
354    pub fn recorded_bytes_appended() -> usize {
355        BYTES_APPENDED.load(Ordering::Relaxed)
356    }
357
358    /// Total bytes copied (rather than borrowed) into a materialized value
359    /// during the armed run — should stay ≈0 for a non-straddling text-heavy
360    /// scan (issue #1644 acceptance proof).
361    pub fn recorded_bytes_copied_into_values() -> usize {
362        BYTES_COPIED_INTO_VALUES.load(Ordering::Relaxed)
363    }
364}

371    fn consume_advances_cursor_without_memmove() {
372        let mut w = WindowCursor::new();
373        w.refill(b"abcdefgh");
374        assert_eq!(w.as_slice(), b"abcdefgh");
375        assert_eq!(w.len(), 8);
376        assert!(!w.is_empty());
377        w.consume(3);
378        assert_eq!(w.as_slice(), b"defgh");
379        assert_eq!(w.len(), 5);
380        w.consume(5);
381        assert!(w.is_empty());
382        assert_eq!(w.len(), 0);
383        assert_eq!(w.as_slice(), b"");
384    }
