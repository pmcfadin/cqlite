1//! Row-granular resumable within-partition compaction streaming (issue #2299).
2//!
3//! Split out of the sibling `compaction` module (campsite rule, epic #1116):
4//! [`V5CompressedLegacyParser::stream_partition_body_incremental`] and its
5//! cross-call [`CompactionPartitionState`] / per-structure [`PartitionStreamStep`]
6//! types. Reuses the SAME `CompactionPolicy` decode as the buffered
7//! `parse_one_partition_for_compaction`, so tombstone / timestamp / range-marker
8//! semantics are byte-identical — only the buffering GRANULARITY differs (this
9//! path drains one confirmed structure at a time so the sliding-window compaction
10//! driver never has to keep a WIDE partition fully resident).
11

20impl V5CompressedLegacyParser {
21    /// Row-granular resumable partition streaming for the compaction read path
22    /// (issue #2299).
23    ///
24    /// [`Self::parse_one_partition_for_compaction`] buffers a WHOLE partition
25    /// into the driver-owned `pending` vec and only reports its consumed byte
26    /// count (and forwards its rows) once the partition is CONFIRMED complete —
27    /// i.e. at the END_OF_PARTITION marker or the next partition header. For a
28    /// WIDE partition (a single partition holding the bulk of an SSTable, e.g. a
29    /// time-series row split by clustering-key range across the merge inputs)
30    /// that means the sliding-window driver in
31    /// [`SSTableReader::stream_all_partitions_for_compaction`] cannot advance its
32    /// window cursor until the whole partition is resident — so peak memory
33    /// scales with `max_partition_size`, not `one row + one chunk`. With four
34    /// concurrent producers each holding a ~40 MiB wide partition (window bytes
35    /// PLUS the buffered rows), the compaction blows the 128 MiB budget.
36    ///
37    /// This method streams the partition BODY one confirmed structure at a time,
38    /// invoking `emit` per row/marker and reporting via [`PartitionStreamStep`]
39    /// exactly how many bytes of `data` are now safe to drain from the FRONT of
40    /// the caller's sliding window — WITHOUT waiting for END_OF_PARTITION. The
41    /// caller drives it as a resumable cursor: it calls this repeatedly on the
42    /// (growing/shrinking) window, advancing the front cursor by the returned
43    /// `consumed` after every call.
44    ///
45    /// `state` carries the cross-call partition context (the partition key, the
46    /// header-parsed flag, and the in-flight range-tombstone start bound) so a
47    /// partition that straddles a chunk boundary resumes correctly after a
48    /// refill. Tombstone / timestamp / range-marker semantics are byte-identical
49    /// to [`Self::parse_one_partition_for_compaction`] — this is the SAME
50    /// `CompactionPolicy` decode, only the buffering granularity differs.
51    ///
52    /// `at_final_chunk` flips a mid-structure end-of-buffer between a refill
53    /// request ([`PartitionStreamStep::NeedMore`]) and terminal completion
54    /// ([`PartitionStreamStep::PartitionDone`]), exactly as the sliding driver's
55    /// `NeedMore`/`Done` distinction does.
56    ///
57    /// `resolution` is the column resolution built ONCE per scan by the caller
58    /// ([`SSTableReader::stream_all_partitions_for_compaction`]) and threaded in
59    /// (the same way `schema` / `state` are). It is derived purely from the
60    /// SSTable serialization header (`reader.header`) + `schema`, both INVARIANT
61    /// across every partition of a single SSTable, so a single build is
62    /// semantically identical to the buffered `drive_partition_sliding`'s
63    /// per-partition build (`partition_driver.rs:179`) — only without that path's
64    /// per-partition (and, for this per-structure driver, once-PER-ROW) allocation
65    /// churn: `RowColumnResolution::build` allocates a `HashMap` over
66    /// `schema.columns` plus a fresh `Arc<str>` per header/clustering column, so
67    /// rebuilding it per drain step turned an O(partitions) cost into
68    /// O(rows × header_cols) on exactly the wide-partition workload issue #2299
69    /// optimizes. Threading the once-built resolution restores per-scan
70    /// (≤ per-partition) allocation. `None` only when `schema` is `None`, which
71    /// this method rejects before any row decode.
72    ///
73    /// [`SSTableReader::stream_all_partitions_for_compaction`]: crate::storage::sstable::SSTableReader
74    pub fn stream_partition_body_incremental<F>(
75        &self,
76        data: &[u8],
77        schema: Option<&TableSchema>,
78        reader: &crate::storage::sstable::reader::types::SSTableReader,
79        resolution: Option<&RowColumnResolution>,
80        at_final_chunk: bool,
81        state: &mut CompactionPartitionState,
82        emit: &mut F,
83    ) -> Result<PartitionStreamStep>
84    where
85        F: FnMut(
86            crate::storage::sstable::reader::compaction_row::CompactionRow,
87        ) -> Result<std::ops::ControlFlow<()>>,
88    {
89        if data.is_empty() {
90            return Ok(if at_final_chunk {
91                PartitionStreamStep::AllDone
92            } else {
93                PartitionStreamStep::NeedMore
94            });
95        }
96
97        let schema = schema.ok_or_else(|| {
98            Error::schema(format!(
99                "V5CompressedLegacy (compaction) format requires schema for {}.{}",
100                self.keyspace, self.table_name
101            ))
102        })?;
103
104        // The column resolution is built ONCE per scan by the caller and threaded
105        // in (see the doc comment). It is required whenever `schema` is present
106        // (the caller builds it from the same schema); a missing resolution here
107        // is an internal wiring bug, not a data condition.
108        let resolution = resolution.ok_or_else(|| {
109            Error::schema(format!(
110                "V5CompressedLegacy (compaction) streaming requires a prebuilt column \
111                 resolution for {}.{}",
112                self.keyspace, self.table_name
113            ))
114        })?;
115
116        // (1) Parse the partition header ONCE per partition. When the header has
117        //     already been parsed for this partition (a resumed body call after a
118        //     refill) the window front is the body start (offset 0).
119        if !state.header_parsed {
120            match self.partition_header_readiness(data) {
121                PartitionHeaderReadiness::Malformed => {
122                    // Skip one byte to resynchronise (matches the buffered driver's
123                    // `Emitted(1)` malformed-header behaviour).
124                    return Ok(PartitionStreamStep::Consumed(1));
125                }
126                PartitionHeaderReadiness::Incomplete => {
127                    return Ok(if at_final_chunk {
128                        PartitionStreamStep::AllDone
129                    } else {
130                        PartitionStreamStep::NeedMore
131                    });
132                }
133                PartitionHeaderReadiness::Ready => {}
134            }
135
136            let (partition_key, after_header, partition_deletion) =
137                match self.parse_partition_header_full(data, 0) {
138                    Ok(v) => v,
139                    Err(_) => {
140                        if !at_final_chunk
141                            && self.partition_header_readiness(data)
142                                == PartitionHeaderReadiness::Incomplete
143                        {
144                            return Ok(PartitionStreamStep::NeedMore);
145                        }
146                        return Ok(PartitionStreamStep::Consumed(1));
147                    }
148                };
149
150            // Emit a partition-level tombstone carrier immediately (issue #1072),
151            // via the shared CompactionPolicy hook — same synthetic carrier the
152            // buffered path pushes into `pending`. The policy's transient state
153            // (partition key + in-flight range-start bound) is written back into
154            // `state` so a resumed body call rebuilds an equivalent policy.
155            let mut policy = CompactionPolicy::new(self);
156            let mut carriers: Vec<crate::storage::sstable::reader::compaction_row::CompactionRow> =
157                Vec::new();
158            policy.on_partition_open(partition_key, partition_deletion, schema, &mut carriers);
159            state.partition_key = policy.partition_key().clone();
160            state.pending_range_start = policy.pending_range_start().clone();
161            state.header_parsed = true;
162            // The header bytes are confirmed; drain them from the window front so a
163            // refill re-parse never re-reads the header. The next call sees the body
164            // at offset 0.
165            for row in carriers {
166                match emit(row)? {
167                    std::ops::ControlFlow::Continue(()) => {}
168                    std::ops::ControlFlow::Break(()) => {
169                        return Ok(PartitionStreamStep::Break(after_header));
170                    }
171                }
172            }
173            return Ok(PartitionStreamStep::Consumed(after_header));
174        }
175
176        // (2) Body: parse ONE confirmed structure (row / range marker /
177        //     end-of-partition) from the FRONT of the window (offset 0 — this
178        //     driver resumes from 0 each call, reporting its consumed byte count so
179        //     the caller advances the window cursor per structure), emit it, and
180        //     report its consumed bytes. Rebuild the policy from `state` (it only
181        //     carries owned pieces: the partition key and the in-flight
182        //     range-tombstone start bound). `data` is non-empty here (early return
183        //     at the top), so the front byte `data[0]` is always in bounds.
184
185        // END_OF_PARTITION (0x01): partition complete; consume the marker byte and
186        // signal the caller to reset for the next partition.
187        if Self::is_end_of_partition(data[0]) {
188            state.reset();
189            return Ok(PartitionStreamStep::PartitionDone(1));
190        }
191
192        let mut policy = CompactionPolicy::new(self);
193        policy.set_partition_key(state.partition_key.clone());
194        policy.set_pending_range_start(state.pending_range_start.clone());
195
196        // A range-tombstone marker: pair it via the policy (the surviving marker
197        // row, if any, is emitted here). A truncated marker on a non-final chunk
198        // requests more bytes.
199        if Self::is_range_tombstone_marker(data[0]) {
200            let mut emitted: Vec<crate::storage::sstable::reader::compaction_row::CompactionRow> =
201                Vec::new();
202            match policy.on_range_marker(data, 0, schema, &mut emitted) {
203                MarkerOutcome::Advanced(next_offset) => {
204                    // Confirm the marker is fully framed within the window.
205                    if next_offset > data.len() {
206                        return Ok(if at_final_chunk {
207                            PartitionStreamStep::AllDone
208                        } else {
209                            PartitionStreamStep::NeedMore
210                        });
211                    }
212                    state.pending_range_start = policy.pending_range_start().clone();
213                    for row in emitted {
214                        match emit(row)? {
215                            std::ops::ControlFlow::Continue(()) => {}
216                            std::ops::ControlFlow::Break(()) => {
217                                return Ok(PartitionStreamStep::Break(next_offset));
218                            }
219                        }
220                    }
221                    return Ok(PartitionStreamStep::Consumed(next_offset));
222                }
223                MarkerOutcome::Stop => {
224                    if at_final_chunk {
225                        state.reset();
226                        return Ok(PartitionStreamStep::PartitionDone(0));
227                    }
228                    return Ok(PartitionStreamStep::NeedMore);
229                }
230            }
231        }
232
233        // A data row: decode exactly one, emit it, and report its consumed bytes.
234        let mut emitted: Vec<crate::storage::sstable::reader::compaction_row::CompactionRow> =
235            Vec::new();
236        match policy.on_data_row(data, 0, schema, reader, resolution, &mut emitted) {
237            Some(next_offset) => {
238                // Confirm the row is fully framed WITHIN the window: a next_offset
239                // STRICTLY PAST the buffer end means we decoded a truncated row on a
240                // non-final chunk — request more bytes rather than emit a partial
241                // row. NB: `>` not `>=` (the buffered `drive_partition_sliding` uses
242                // `>=` on line 240 of partition_driver.rs). The difference is
243                // intentional and load-bearing for this resume-from-0 driver: this
244                // path parses ONE structure from the FRONT (offset 0) and reports
245                // its consumed byte count so the caller advances the window cursor
246                // per-structure, so a row that ends EXACTLY at `data.len()`
247                // (`next_offset == data.len()`) is a fully-framed row whose bytes we
248                // must consume and drain — treating `==` as truncation would stall
249                // the cursor forever (re-parsing the same trailing row every refill).
250                // The buffered driver's `>=` is correct for ITS loop (it keeps
251                // advancing `offset` within one call and uses `offset >= data.len()`
252                // to decide whether to look for the next structure), not for a
253                // per-structure resume cursor. Do not "simplify" `>` to `>=`.
254                if next_offset > data.len() {
255                    return Ok(if at_final_chunk {
256                        PartitionStreamStep::AllDone
257                    } else {
258                        PartitionStreamStep::NeedMore
259                    });
260                }
261                state.pending_range_start = policy.pending_range_start().clone();
262                for row in emitted {
263                    match emit(row)? {
264                        std::ops::ControlFlow::Continue(()) => {}
265                        std::ops::ControlFlow::Break(()) => {
266                            return Ok(PartitionStreamStep::Break(next_offset));
267                        }
268                    }
269                }
270                // Partition-boundary defense-in-depth (roborev): mirror the buffered
271                // `drive_partition_sliding` EXACTLY. If the very next bytes are a
272                // partition header (a body NOT terminated by an explicit
273                // END_OF_PARTITION 0x01 — e.g. malformed/corrupt input, or a
274                // producer that elides the marker), the current partition is
275                // complete. Feeding the next partition's header into `on_data_row`
276                // would mis-attribute/desync it. Split here (reset for the next
277                // partition) so the streaming path and the buffered path can never
278                // diverge. Only meaningful when the header is fully framed within
279                // the window; a truncated header on a non-final chunk falls through
280                // to the normal per-structure refill on the next call.
281                if next_offset < data.len() && self.peek_is_partition_header(data, next_offset) {
282                    state.reset();
283                    return Ok(PartitionStreamStep::PartitionDone(next_offset));
284                }
285                Ok(PartitionStreamStep::Consumed(next_offset))
286            }
287            None => {
288                // A row failed to parse. Mid-stream that may be a row straddling the
289                // chunk boundary, so request more bytes unless this is the final
290                // chunk (where it is end-of-partition).
291                if at_final_chunk {
292                    state.reset();
293                    Ok(PartitionStreamStep::PartitionDone(0))
294                } else {
295                    Ok(PartitionStreamStep::NeedMore)
296                }
297            }
298        }
299    }
300}

302/// Cross-call state for [`V5CompressedLegacyParser::stream_partition_body_incremental`]
303/// (issue #2299). Carries the in-flight partition's decode context so a partition
304/// that straddles a sliding-window chunk boundary resumes correctly after a refill.
305pub struct CompactionPartitionState {
306    /// Whether the partition header (key + partition-level deletion) has been
307    /// parsed for the CURRENT partition. Reset to `false` at each partition
308    /// boundary so the next partition re-parses its own header.
309    header_parsed: bool,
310    /// The current partition's key, decoded once from the header and reused for
311    /// every row this partition emits across resumed calls.
312    partition_key: RowKey,
313    /// Issue #933: the in-flight range-tombstone start bound
314    /// `(bound, markedForDeleteAt µs, localDeletionTime s)`, carried across
315    /// resumed calls so a range whose CLOSE bound arrives in a later chunk still
316    /// pairs correctly (the incremental driver never re-parses from the partition
317    /// start, so this cross-call carry replaces the buffered driver's re-derive).
318    pending_range_start: Option<(
319        crate::storage::sstable::reader::compaction_row::CompactionBound,
320        i64,
321        i32,
322    )>,
323}
324
325impl Default for CompactionPartitionState {
326    fn default() -> Self {
327        Self::new()
328    }
329}
330
331impl CompactionPartitionState {
332    /// A fresh state positioned at the start of a (not-yet-parsed) partition.
333    pub fn new() -> Self {
334        Self {
335            header_parsed: false,
336            partition_key: RowKey::new(Vec::new()),
337            pending_range_start: None,
338        }
339    }
340
341    /// Reset to await the next partition header (after an END_OF_PARTITION or a
342    /// terminal trailing partition).
343    fn reset(&mut self) {
344        self.header_parsed = false;
345        self.partition_key = RowKey::new(Vec::new());
346        self.pending_range_start = None;
347    }
348}

350/// Outcome of one [`V5CompressedLegacyParser::stream_partition_body_incremental`]
351/// call (issue #2299). Reports how many bytes of the window front are now safe to
352/// drain, WITHOUT waiting for a whole partition.
353#[derive(Debug, PartialEq, Eq)]
354pub enum PartitionStreamStep {
355    /// One structure (header prefix, row, or range marker) was confirmed and its
356    /// rows emitted; `usize` bytes may be drained from the window front. The
357    /// current partition CONTINUES — call again.
358    Consumed(usize),
359    /// The partition ended (END_OF_PARTITION, or a terminal trailing partition on
360    /// the final chunk); `usize` bytes may be drained. State is reset for the next
361    /// partition.
362    PartitionDone(usize),
363    /// The window is (possibly) truncated mid-structure and `!at_final_chunk`:
364    /// the caller must append the next chunk and retry (nothing to drain). Never
365    /// returned when `at_final_chunk`.
366    NeedMore,
367    /// The window is empty at the final chunk — genuine end of data. Terminal.
368    AllDone,
369    /// `emit` returned `Break` (consumer dropped); `usize` bytes were consumed up
370    /// to and including the breaking structure. The caller stops the scan.
371    Break(usize),
372}
