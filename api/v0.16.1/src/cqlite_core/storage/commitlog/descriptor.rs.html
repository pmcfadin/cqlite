1//! Cassandra CommitLog segment **descriptor** parsing (issue #2389).
2//!
3//! Every CommitLog segment begins with a `CommitLogDescriptor` header that
4//! Cassandra writes via `CommitLogDescriptor.writeHeader` (Cassandra 5.0
5//! `org.apache.cassandra.db.commitlog.CommitLogDescriptor`). The on-disk layout,
6//! all big-endian, is:
7//!
8//! ```text
9//! int32   version          descriptor/messaging version (VERSION_40 = 7 for 5.0)
10//! int64   id               segment id (wall-clock-derived at segment creation)
11//! int16   paramsLen        length of the UTF-8 parameters JSON
12//! bytes   params           UTF-8 JSON: {} when uncompressed/unencrypted
13//! int32   crc              CRC32 (java.util.zip.CRC32 / IEEE) over the header
14//! ```
15//!
16//! The CRC covers, in order (each int fed big-endian, matching Cassandra's
17//! `FBUtilities.updateChecksumInt`): `version`, the low 32 bits of `id`, the
18//! high 32 bits of `id`, `paramsLen` as a 32-bit int, then the raw params bytes.
19//!
20//! Version and format are read **authoritatively from these header bytes**,
21//! never inferred from the filename or file size (no-heuristics, issue #28).
22
23use crate::{Error, Result};
24
25/// The single Cassandra 5.0-era CommitLog descriptor version CQLite supports.
26///
27/// `CommitLogDescriptor.VERSION_40 == 7`; Cassandra 4.0 and 5.0 both write it as
28/// `current_version`. Mirrors the exact-allowlist posture of
29/// [`crate::storage::sstable::version_gate::BigVersionGates`].
30pub const SUPPORTED_COMMITLOG_VERSION: i32 = 7;
31
32/// Version gate for a Cassandra CommitLog segment.
33///
34/// Constructed only from the descriptor's authoritative `version` field. Rejects
35/// any version outside the supported Cassandra 5.0-era range via a typed
36/// [`Error::UnsupportedCommitLogVersion`], the same pattern as
37/// `BigVersionGates`/`BtiVersionGates` for SSTables.
38#[derive(Debug, Clone, Copy, PartialEq, Eq)]
39pub struct CommitLogVersionGates {
40    /// The descriptor version this gate set was computed from.
41    pub version: i32,
42}
43
44impl CommitLogVersionGates {
45    /// Validate a CommitLog descriptor version.
46    ///
47    /// # Errors
48    /// Returns [`Error::UnsupportedCommitLogVersion`] for any version other than
49    /// [`SUPPORTED_COMMITLOG_VERSION`].
50    pub fn from_version(version: i32) -> Result<Self> {
51        if version != SUPPORTED_COMMITLOG_VERSION {
52            return Err(Error::UnsupportedCommitLogVersion {
53                version,
54                floor: SUPPORTED_COMMITLOG_VERSION,
55                ceiling: SUPPORTED_COMMITLOG_VERSION,
56            });
57        }
58        Ok(Self { version })
59    }
60}

62/// Parsed CommitLog segment descriptor header.
63#[derive(Debug, Clone, PartialEq, Eq)]
64pub struct CommitLogDescriptor {
65    /// Descriptor/messaging version (authoritative, from the header).
66    pub version: i32,
67    /// Segment id.
68    pub id: i64,
69    /// Raw parameters JSON string (`{}` when uncompressed/unencrypted).
70    pub params_json: String,
71    /// Compressor class name if the segment declares commitlog compression.
72    ///
73    /// `Some(_)` means the mutation stream is compressed — CQLite v1 fails
74    /// closed on these (design decision D3); it never decodes compressed bytes
75    /// as if uncompressed.
76    pub compression_class: Option<String>,
77    /// Whether the descriptor declares an encryption context (also unsupported).
78    pub encrypted: bool,
79    /// Number of header bytes consumed — the offset at which the segment body
80    /// (first sync marker) begins.
81    pub header_len: usize,
82}
83
84impl CommitLogDescriptor {
85    /// Parse a descriptor from the start of a segment's bytes.
86    ///
87    /// Validates the header CRC and the version gate before returning. Does not
88    /// touch the mutation stream.
89    ///
90    /// # Errors
91    /// - [`Error::CorruptCommitLogFrame`] on a short/malformed header or a CRC
92    ///   mismatch.
93    /// - [`Error::UnsupportedCommitLogVersion`] for an out-of-range version.
94    pub fn parse(bytes: &[u8]) -> Result<Self> {
95        // version(4) + id(8) + paramsLen(2) + crc(4) = 18 byte minimum.
96        if bytes.len() < 18 {
97            return Err(Error::CorruptCommitLogFrame(format!(
98                "segment too short for a descriptor header: {} bytes",
99                bytes.len()
100            )));
101        }
102
103        let version = read_i32_be(bytes, 0);
104        // Gate the version FIRST (authoritative) — but only after confirming the
105        // header is structurally readable, so we never parse a bogus version off
106        // a truncated file as if meaningful.
107        let gates = CommitLogVersionGates::from_version(version)?;
108
109        let id = read_i64_be(bytes, 4);
110        let params_len = read_u16_be(bytes, 12) as usize;
111
112        let params_start = 14;
113        let params_end = params_start + params_len;
114        let crc_end = params_end + 4;
115        if bytes.len() < crc_end {
116            return Err(Error::CorruptCommitLogFrame(format!(
117                "descriptor header claims {} param bytes but segment is only {} bytes",
118                params_len,
119                bytes.len()
120            )));
121        }
122
123        let params_bytes = &bytes[params_start..params_end];
124        let stored_crc = read_u32_be(bytes, params_end);
125
126        // CRC over version, id-low, id-high, paramsLen(int32), params bytes.
127        let mut hasher = crc32fast::Hasher::new();
128        hasher.update(&version.to_be_bytes());
129        let id_u = id as u64;
130        hasher.update(&((id_u & 0xFFFF_FFFF) as u32).to_be_bytes());
131        hasher.update(&((id_u >> 32) as u32).to_be_bytes());
132        hasher.update(&(params_len as u32).to_be_bytes());
133        hasher.update(params_bytes);
134        let computed_crc = hasher.finalize();
135
136        if computed_crc != stored_crc {
137            return Err(Error::CorruptCommitLogFrame(format!(
138                "descriptor CRC mismatch: stored={stored_crc:#010x} computed={computed_crc:#010x}"
139            )));
140        }
141
142        let params_json = String::from_utf8_lossy(params_bytes).into_owned();
143        let (compression_class, encrypted) = parse_params(&params_json);
144
145        // Keep the gate's version (validated) as the authoritative value.
146        Ok(Self {
147            version: gates.version,
148            id,
149            params_json,
150            compression_class,
151            encrypted,
152            header_len: crc_end,
153        })
154    }
155
156    /// Whether the segment declares an unsupported (compressed or encrypted)
157    /// payload. When true, [`crate::storage::commitlog::CommitLogReader`] fails
158    /// closed rather than misdecoding the stream.
159    pub fn is_unsupported_payload(&self) -> bool {
160        self.compression_class.is_some() || self.encrypted
161    }
162}

164/// Extract the compressor class and encryption flag from the params JSON.
165///
166/// The params object is Cassandra's own authoritative descriptor metadata; we
167/// parse it as JSON (never sniff the payload). An unparseable/empty params
168/// string is treated as uncompressed+unencrypted (the `{}` baseline).
169///
170/// The key names are the ones Cassandra 5.0.2 actually writes in
171/// `CommitLogDescriptor.constructParametersString` /
172/// `EncryptionContext.toHeaderParameters`:
173/// - `compressionClass` (`COMPRESSION_CLASS_KEY`) — a plain STRING value (the
174///   compressor class name), NOT a nested `{"class": ...}` object.
175///   (`compressionParameters` also exists but is informational and unneeded
176///   for unsupported-payload detection.)
177/// - `encCipher` / `encKeyAlias` / `encIV` — the encryption-context keys; there
178///   is no `encryptionContext`/`encryption` key on the wire.
179fn parse_params(json: &str) -> (Option<String>, bool) {
180    let value: serde_json::Value = match serde_json::from_str(json) {
181        Ok(v) => v,
182        Err(_) => return (None, false),
183    };
184    let obj = match value.as_object() {
185        Some(o) => o,
186        None => return (None, false),
187    };
188    // `compressionClass` is a plain string. Some configurations write it as an
189    // explicit JSON `null` (key present, value null) when compression is unset,
190    // rather than omitting the key; a present-but-null value must parse as "no
191    // compression" (None), same as an absent key — never fail-closed on a valid
192    // uncompressed segment. An empty string is likewise treated as no
193    // compression. Only a present, non-null, non-empty string names a
194    // compressor the reader must refuse.
195    let compression_class = obj
196        .get("compressionClass")
197        .and_then(|c| c.as_str())
198        .filter(|s| !s.is_empty())
199        .map(|s| s.to_string());
200    // Encryption is declared by any of the real encryption-context keys being
201    // present with a non-null value. Same present-but-null hazard as
202    // compression: `{"encCipher": null}` is NOT encrypted.
203    let encrypted = ["encCipher", "encKeyAlias", "encIV"]
204        .iter()
205        .any(|k| obj.get(*k).is_some_and(|v| !v.is_null()));
206    (compression_class, encrypted)
207}

209#[inline]
210fn read_u16_be(b: &[u8], off: usize) -> u16 {
211    u16::from_be_bytes([b[off], b[off + 1]])
212}
213#[inline]
214fn read_i32_be(b: &[u8], off: usize) -> i32 {
215    i32::from_be_bytes([b[off], b[off + 1], b[off + 2], b[off + 3]])
216}
217#[inline]
218fn read_u32_be(b: &[u8], off: usize) -> u32 {
219    u32::from_be_bytes([b[off], b[off + 1], b[off + 2], b[off + 3]])
220}
221#[inline]
222fn read_i64_be(b: &[u8], off: usize) -> i64 {
223    let mut a = [0u8; 8];
224    a.copy_from_slice(&b[off..off + 8]);
225    i64::from_be_bytes(a)
226}

232    /// Build a descriptor header the same way Cassandra's `writeHeader` does,
233    /// so the parser's CRC and field offsets are exercised in isolation. This
234    /// is a *unit* fixture for offset math only; the authoritative correctness
235    /// oracle is the real Cassandra-produced segment (see the integration test).
236    ///
237    /// `pub(crate)` so the [`super::super::reader`] public-surface tests can
238    /// build a Cassandra-shaped descriptor header without duplicating the CRC.
239    pub(crate) fn build_header(version: i32, id: i64, params: &str) -> Vec<u8> {
240        let params_bytes = params.as_bytes();
241        let mut out = Vec::new();
242        out.extend_from_slice(&version.to_be_bytes());
243        out.extend_from_slice(&id.to_be_bytes());
244        out.extend_from_slice(&(params_bytes.len() as u16).to_be_bytes());
245        out.extend_from_slice(params_bytes);
246        let mut hasher = crc32fast::Hasher::new();
247        hasher.update(&version.to_be_bytes());
248        let id_u = id as u64;
249        hasher.update(&((id_u & 0xFFFF_FFFF) as u32).to_be_bytes());
250        hasher.update(&((id_u >> 32) as u32).to_be_bytes());
251        hasher.update(&(params_bytes.len() as u32).to_be_bytes());
252        hasher.update(params_bytes);
253        out.extend_from_slice(&hasher.finalize().to_be_bytes());
254        out
255    }

270        let id = 1_689_012_345_678_i64;
271        let bytes = build_header(7, id, "{}");
272        let desc = CommitLogDescriptor::parse(&bytes).expect("parse");
273        assert_eq!(desc.version, 7);
274        assert_eq!(desc.id, id);
275        assert_eq!(desc.header_len, bytes.len());
