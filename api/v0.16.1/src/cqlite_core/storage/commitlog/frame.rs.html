1//! Per-record framing + sync-marker walking for a Cassandra CommitLog segment
2//! (issue #2389).
3//!
4//! After the [descriptor header](super::descriptor), a segment body is a chain
5//! of **sync-marker-delimited sections**. Each section begins with an 8-byte
6//! sync marker and holds a run of length-and-CRC-framed mutation records.
7//!
8//! All layouts below are verified byte-for-byte against a real Cassandra 5.0.2
9//! segment (the fixtures under `test-data/datasets/commitlog/`).
10//!
11//! ```text
12//! sync marker (8 bytes, big-endian):
13//!   int32 nextMarker   file offset of the *next* sync marker (= end of section)
14//!   int32 crc          CRC32 over: id-low(BE i32), id-high(BE i32), markerPos(BE i32)
15//!
16//! record (per mutation):
17//!   int32 size         serialized mutation length
18//!   int32 sizeCrc      CRC32 over the 4 size bytes (BE)
19//!   bytes body         `size` mutation bytes
20//!   int32 bodyCrc      CRC32 continuing over size bytes ++ body bytes
21//! ```
22//!
23//! **Truncation vs corruption.** A record or marker whose bytes are *missing*
24//! (extend past end-of-file) is a torn tail — tolerated: prior records are kept
25//! and the walk stops with [`FrameStep::Truncated`]. A record that is *fully
26//! present* but whose CRC does not match is genuine corruption — surfaced as
27//! [`Error::CorruptCommitLogFrame`]. This mirrors Cassandra's
28//! `CommitLogReader` `tolerateTruncation` behavior.
29
30use crate::{Error, Result};
31
32/// Size of a sync marker in bytes (`int32 nextMarker` + `int32 crc`).
33pub const SYNC_MARKER_SIZE: usize = 8;
34
35/// Per-record overhead: `int32 size` + `int32 sizeCrc` + `int32 bodyCrc`.
36pub const RECORD_OVERHEAD: usize = 12;
37
38/// One step of a frame walk.
39#[derive(Debug)]
40pub enum FrameStep<'a> {
41    /// A validated mutation record body (the serialized `Mutation` bytes).
42    Record(&'a [u8]),
43    /// Clean end of the written segment (padding / zeroed pre-allocation).
44    End,
45    /// Torn tail: a record or marker ran past end-of-file. Records already
46    /// yielded before this step are valid and complete.
47    Truncated,
48}
49
50/// Lazy cursor over a segment's sync sections and records.
51///
52/// Holds only a borrow of the segment bytes plus a couple of cursor integers —
53/// it never materializes the decoded record set, satisfying the streaming
54/// memory budget (design decision D4 / `oom-audit`).
55pub struct FrameWalker<'a> {
56    bytes: &'a [u8],
57    segment_id: i64,
58    /// Position of the sync marker that opens the current section.
59    marker_pos: usize,
60    /// Cursor within the current section (position of the next record).
61    cursor: usize,
62    /// End offset of the current section (the next marker's position).
63    section_end: usize,
64    /// Whether we are positioned inside an open section.
65    in_section: bool,
66    /// Set when the walk ended on a torn tail rather than clean padding.
67    truncated_end: bool,
68    /// Terminal flag once [`FrameStep::End`]/[`FrameStep::Truncated`] is reached.
69    done: bool,
70}
71
72impl<'a> FrameWalker<'a> {
73    /// Start a walk at the segment body (immediately after the descriptor).
74    ///
75    /// `body_start` is [`super::descriptor::CommitLogDescriptor::header_len`];
76    /// `segment_id` is the descriptor's authoritative segment id (used to verify
77    /// every sync marker's CRC).
78    pub fn new(bytes: &'a [u8], segment_id: i64, body_start: usize) -> Self {
79        Self {
80            bytes,
81            segment_id,
82            marker_pos: body_start,
83            cursor: 0,
84            section_end: 0,
85            in_section: false,
86            truncated_end: false,
87            done: false,
88        }
89    }
90
91    /// Advance to the next record, end, or torn tail.
92    ///
93    /// # Errors
94    /// [`Error::CorruptCommitLogFrame`] when a fully-present record fails its
95    /// size/body CRC, or a marker is internally inconsistent.
96    pub fn next_frame(&mut self) -> Result<FrameStep<'a>> {
97        if self.done {
98            return Ok(FrameStep::End);
99        }
100        loop {
101            if !self.in_section {
102                match self.open_section()? {
103                    Some(()) => {}
104                    None => {
105                        self.done = true;
106                        return Ok(if self.truncated_end {
107                            FrameStep::Truncated
108                        } else {
109                            FrameStep::End
110                        });
111                    }
112                }
113            }
114
115            // Within the open section, read the next record.
116            match self.read_record()? {
117                RecordOutcome::Body(b) => return Ok(FrameStep::Record(b)),
118                RecordOutcome::SectionDone => {
119                    // Advance to the next marker.
120                    self.in_section = false;
121                    self.marker_pos = self.section_end;
122                    continue;
123                }
124                RecordOutcome::CleanEnd => {
125                    self.done = true;
126                    // A torn section (opened past-EOF marker) sets truncated_end
127                    // at open time; honor it here too, not just in open_section's
128                    // own None path — otherwise hitting padding inside a torn
129                    // section's remaining bytes silently downgrades Truncated to
130                    // a clean End (roborev finding, review-first pass).
131                    return Ok(if self.truncated_end {
132                        FrameStep::Truncated
133                    } else {
134                        FrameStep::End
135                    });
136                }
137                RecordOutcome::Truncated => {
138                    self.done = true;
139                    return Ok(FrameStep::Truncated);
140                }
141            }
142        }
143    }
144
145    /// Open the section at `marker_pos`. Returns `Ok(None)` on clean end or torn
146    /// tail (with `truncated_end` set appropriately).
147    fn open_section(&mut self) -> Result<Option<()>> {
148        let pos = self.marker_pos;
149        // A marker whose bytes are missing → torn tail.
150        if pos + SYNC_MARKER_SIZE > self.bytes.len() {
151            // Distinguish a clean end (nothing but absence) from a torn write:
152            // any leftover non-marker bytes present but < 8 is a torn tail.
153            // `|=`, not `=`: a prior torn section may have already set this
154            // true, and this branch reaching `pos < len` false (e.g. a
155            // section whose records happen to fill the buffer exactly) must
156            // not silently downgrade it back to a clean end — the exact bug
157            // this fix's neighboring CleanEnd-arm change was meant to close,
158            // just at a second assignment site (roborev finding, review-
159            // first pass).
160            self.truncated_end |= pos < self.bytes.len();
161            return Ok(None);
162        }
163        let next_marker = read_i32_be(self.bytes, pos);
164        // Zeroed pre-allocation → clean end of the written region.
165        if next_marker == 0 {
166            self.truncated_end = false;
167            return Ok(None);
168        }
169        // Validate the marker CRC (id-low, id-high, markerPos).
170        let stored_crc = read_u32_be(self.bytes, pos + 4);
171        let computed = marker_crc(self.segment_id, pos);
172        if computed != stored_crc {
173            // An invalid marker over an all-zero region is the normal clean end;
174            // otherwise it is a torn/garbage marker (tolerate as truncation).
175            let all_zero = self.bytes[pos..pos + SYNC_MARKER_SIZE]
176                .iter()
177                .all(|&b| b == 0);
178            // `|=` for the same monotonicity reason as above.
179            self.truncated_end |= !all_zero;
180            return Ok(None);
181        }
182        let next = next_marker as usize;
183        // Marker is valid but points past EOF → the section body is torn. This
184        // section necessarily runs to true end-of-file (nothing can follow a
185        // marker that already overruns the buffer), so the walk is guaranteed
186        // to terminate truncated — set that now, not just on the eventual
187        // "no more marker bytes" path, so a CleanEnd/SectionDone reached while
188        // still inside this torn section doesn't get reported as a clean end
189        // (roborev finding, review-first pass).
190        if next > self.bytes.len() {
191            self.section_end = self.bytes.len();
192            self.cursor = pos + SYNC_MARKER_SIZE;
193            self.in_section = true;
194            self.truncated_end = true;
195            return Ok(Some(()));
196        }
197        // A marker must point strictly forward past its own 8 bytes.
198        if next < pos + SYNC_MARKER_SIZE {
199            return Err(Error::CorruptCommitLogFrame(format!(
200                "sync marker at offset {pos} points backward to {next}"
201            )));
202        }
203        self.section_end = next;
204        self.cursor = pos + SYNC_MARKER_SIZE;
205        self.in_section = true;
206        Ok(Some(()))
207    }
208
209    /// Read one record from within the open section.
210    fn read_record(&mut self) -> Result<RecordOutcome<'a>> {
211        let cur = self.cursor;
212        let len = self.bytes.len();
213
214        // No room for a record header within the section → section finished.
215        if cur >= self.section_end {
216            return Ok(RecordOutcome::SectionDone);
217        }
218        // `is_final_section` distinguishes "this section runs to true EOF" from
219        // "more sections follow at self.section_end". Cassandra pads the tail of
220        // EVERY non-final section up to its next sync marker — a zeroed/short
221        // record header there means "done with THIS section, continue at the
222        // next marker" (SectionDone), not "the whole segment ends here"
223        // (CleanEnd/Truncated). Treating it as segment-end silently drops every
224        // mutation in all subsequent sections (roborev finding, review-first
225        // pass).
226        let is_final_section = self.section_end >= len;
227        if cur + 4 > self.section_end {
228            return Ok(if is_final_section {
229                RecordOutcome::Truncated
230            } else {
231                RecordOutcome::SectionDone
232            });
233        }
234        let size = read_i32_be(self.bytes, cur);
235        // Zeroed padding: true clean end of the written segment only when this
236        // is the final section; otherwise it's just this section's padding.
237        if size <= 0 {
238            return Ok(if is_final_section {
239                RecordOutcome::CleanEnd
240            } else {
241                RecordOutcome::SectionDone
242            });
243        }
244        let size = size as usize;
245        let end = cur + 4 + 4 + size + 4;
246        // Record extends past EOF → torn tail.
247        if end > len {
248            return Ok(RecordOutcome::Truncated);
249        }
250        // Record extends past the section boundary but is within the file: the
251        // marker/section framing disagrees with the record length — corrupt.
252        // No `&& !self.torn_section` guard here: in a torn section
253        // section_end == bytes.len() == len, so `end > self.section_end`
254        // would imply `end > len`, which the EOF check above already caught
255        // and returned Truncated for — this branch is unreachable with
256        // torn_section true regardless, so an explicit exemption clause only
257        // obscured that invariant and invited a future edit to make it live
258        // and wrong (roborev finding, review-first pass).
259        if end > self.section_end {
260            return Err(Error::CorruptCommitLogFrame(format!(
261                "record at offset {cur} (len {size}) overruns section end {}",
262                self.section_end
263            )));
264        }
265
266        let size_crc = read_u32_be(self.bytes, cur + 4);
267        let size_bytes = &self.bytes[cur..cur + 4];
268        let mut hasher = crc32fast::Hasher::new();
269        hasher.update(size_bytes);
270        let computed_size_crc = hasher.finalize();
271        if computed_size_crc != size_crc {
272            return Err(Error::CorruptCommitLogFrame(format!(
273                "record size CRC mismatch at offset {cur}: stored={size_crc:#010x} \
274                 computed={computed_size_crc:#010x}"
275            )));
276        }
277
278        let body_start = cur + 8;
279        let body = &self.bytes[body_start..body_start + size];
280        let body_crc = read_u32_be(self.bytes, body_start + size);
281        // Continuous CRC over the size bytes followed by the body bytes.
282        let mut hasher = crc32fast::Hasher::new();
283        hasher.update(size_bytes);
284        hasher.update(body);
285        let computed_body_crc = hasher.finalize();
286        if computed_body_crc != body_crc {
287            return Err(Error::CorruptCommitLogFrame(format!(
288                "record body CRC mismatch at offset {cur}: stored={body_crc:#010x} \
289                 computed={computed_body_crc:#010x}"
290            )));
291        }
292
293        self.cursor = end;
294        Ok(RecordOutcome::Body(body))
295    }
296}
297
298enum RecordOutcome<'a> {
299    Body(&'a [u8]),
300    SectionDone,
301    CleanEnd,
302    Truncated,
303}
304
305/// Compute a sync-marker CRC the way Cassandra's `writeSyncMarker` does:
306/// CRC32 over id-low (BE i32), id-high (BE i32), markerPos (BE i32).
307pub fn marker_crc(segment_id: i64, marker_pos: usize) -> u32 {
308    let id = segment_id as u64;
309    let mut hasher = crc32fast::Hasher::new();
310    hasher.update(&((id & 0xFFFF_FFFF) as u32).to_be_bytes());
311    hasher.update(&((id >> 32) as u32).to_be_bytes());
312    hasher.update(&(marker_pos as u32).to_be_bytes());
313    hasher.finalize()
314}
315
316#[inline]
317fn read_i32_be(b: &[u8], off: usize) -> i32 {
318    i32::from_be_bytes([b[off], b[off + 1], b[off + 2], b[off + 3]])
319}
320#[inline]
321fn read_u32_be(b: &[u8], off: usize) -> u32 {
322    u32::from_be_bytes([b[off], b[off + 1], b[off + 2], b[off + 3]])
323}

331    fn push_marker(buf: &mut Vec<u8>, pos: usize, next_marker: i32) {
332        buf.extend_from_slice(&next_marker.to_be_bytes());
333        buf.extend_from_slice(&marker_crc(SEGMENT_ID, pos).to_be_bytes());
334    }
335
336    fn push_record(buf: &mut Vec<u8>, body: &[u8]) {
337        let size = body.len() as i32;
338        let size_bytes = size.to_be_bytes();
339        let mut h = crc32fast::Hasher::new();
340        h.update(&size_bytes);
341        let size_crc = h.finalize();
342        buf.extend_from_slice(&size_bytes);
343        buf.extend_from_slice(&size_crc.to_be_bytes());
344        buf.extend_from_slice(body);
345        let mut h = crc32fast::Hasher::new();
346        h.update(&size_bytes);
347        h.update(body);
348        buf.extend_from_slice(&h.finalize().to_be_bytes());
349    }

351    /// Regression for the roborev finding that the `SectionDone` fix (a
352    /// non-final section's padding must finish THAT section, not the whole
353    /// segment) had zero test coverage — every committed fixture is
354    /// single-section. Builds a real two-section segment byte-for-byte per
355    /// this module's own documented layout: section 1 (one record + zero
356    /// padding to the next marker), section 2 (one record, then a clean
357    /// natural end at true EOF).
358    #[test]
359    fn walks_records_across_multiple_sync_sections() {
360        let mut buf = Vec::new();
361        // Section 1: marker at 0, one record, 8 bytes of zero padding.
362        let record1 = b"section-one-record";
363        push_marker(&mut buf, 0, 0 /* placeholder, patched below */);
364        push_record(&mut buf, record1);
365        buf.extend_from_slice(&[0u8; 8]); // padding to the next marker
366        let section2_marker_pos = buf.len();
367        // Patch section 1's marker to point at section 2's marker (its own
368        // section_end), now that we know the offset.
369        let next1 = section2_marker_pos as i32;
370        buf[0..4].copy_from_slice(&next1.to_be_bytes());
371        buf[4..8].copy_from_slice(&marker_crc(SEGMENT_ID, 0).to_be_bytes());
372
373        // Section 2: marker, one record, then true EOF (no trailing padding
374        // needed — section_end == bytes.len() makes this the final section).
375        let record2 = b"section-two-record";
376        let record2_start = section2_marker_pos + SYNC_MARKER_SIZE;
377        let final_len = record2_start + 4 + 4 + record2.len() + 4;
378        push_marker(&mut buf, section2_marker_pos, final_len as i32);
379        push_record(&mut buf, record2);
380        assert_eq!(buf.len(), final_len, "test construction sanity check");
381
382        let mut walker = FrameWalker::new(&buf, SEGMENT_ID, 0);
383        let mut records = Vec::new();
384        loop {
385            match walker.next_frame().expect("no corruption in this fixture") {
386                FrameStep::Record(b) => records.push(b.to_vec()),
387                FrameStep::End => break,
388                FrameStep::Truncated => panic!("expected a clean end, not Truncated"),
389            }
390        }
391        assert_eq!(
392            records,
393            vec![record1.to_vec(), record2.to_vec()],
394            "both sections' records must be yielded — a regression that treats \
395             section 1's padding as segment-end would silently drop record2"
396        );
397    }

399    /// Regression: `truncated_end` was assigned (`=`) rather than OR-ed
400    /// (`|=`) at two sites in `open_section`, so it could be silently
401    /// downgraded from true back to false after a torn section had already
402    /// set it. Trips exactly when a torn section's one record happens to end
403    /// precisely at EOF: the section finishes via `SectionDone` (not
404    /// `CleanEnd`), re-enters `open_section` at `marker_pos == bytes.len()`,
405    /// and that branch's `pos < len` is false — with a plain `=` this wiped
406    /// truncated_end back to false and reported a clean End for a genuinely
407    /// torn segment (roborev finding, review-first pass).
408    #[test]
409    fn torn_section_whose_record_ends_exactly_at_eof_still_reports_truncated() {
410        // Marker's own CRC covers only (segment_id, marker_pos) — NOT the
411        // next_marker value — so next_marker can point arbitrarily far past
412        // EOF while the marker itself still validates, exactly like a real
413        // segment whose section was truncated mid-write.
414        let mut buf = Vec::new();
415        push_marker(&mut buf, 0, 999_999); // far past EOF -> torn section
416        push_record(&mut buf, b"last-record-fills-the-buffer-exactly");
417
418        let mut walker = FrameWalker::new(&buf, SEGMENT_ID, 0);
419        match walker.next_frame().expect("valid record") {
420            FrameStep::Record(_) => {}
421            other => panic!("expected the one record, got {other:?}"),
422        }
423        match walker.next_frame().expect("no corruption") {
424            FrameStep::Truncated => {}
425            other => panic!(
426                "expected Truncated (torn section, record ends exactly at EOF), got {other:?}"
427            ),
428        }
429    }
