1//! Egress-channel-depth gauge for the k-way merge (issue #2419, WS2).
2//!
3//! Backs the [`crate::observability::catalog::MERGE_EGRESS_CHANNEL_DEPTH`] gauge
4//! with a process-global live count of merged DATA entries currently buffered in
5//! the bounded producer→consumer `sync_channel` (capacity
6//! `STREAMING_CHANNEL_CAPACITY` = 256, `merge/mod.rs`). `std::sync::mpsc`'s
7//! `sync_channel` exposes no `len()`, so occupancy is tracked explicitly, exactly
8//! as the #2316 producer-thread gauge tracks live producer threads:
9//!
10//! * [`sent`] increments the count after a successful DATA-entry `send`
11//!   (`from_readers::forward_row`).
12//! * [`received`] decrements it when the consumer pulls that entry off the
13//!   channel (`SSTableRowIteratorAdapter::next`).
14//! * [`reconcile_residual`] (issue #2419 roborev job 1733) subtracts, in ONE
15//!   atomic op, any entries a producer sent but its consumer never received — a
16//!   cancelled/disconnected merge whose channel was torn down while entries were
17//!   still buffered (or racing the teardown). Called ONLY from
18//!   `SSTableRowIteratorAdapter::drop`, AFTER its producer thread has been
19//!   joined: reading a per-adapter sent/received delta BEFORE the join (or
20//!   trying to drain-then-decrement before dropping the receiver) races a
21//!   concurrently-running producer thread and can leak the gauge upward
22//!   permanently — see that `Drop` impl's doc for the full derivation.
23//!
24//! Only DATA entries (`Ok(MergeEntry)`) are tracked on BOTH sides; the rare
25//! terminal error message (`Err(MergeProducerError)`) is untracked on send and
26//! on receive, so it never unbalances the level. The `max(0)` floor matches
27//! `RpcMetrics::finish`, guarding against any unexpected residual imbalance.
28//!
29//! The gauge is OS-independent — it always emits on every platform (unlike the
30//! `/proc`-derived `cqlite.proc.*` saturation gauges).
31
32use std::sync::atomic::{AtomicI64, Ordering};
33
34use crate::observability;
35
36static DEPTH: AtomicI64 = AtomicI64::new(0);
37
38fn record(level: i64) {
39    observability::record_gauge(
40        observability::catalog::MERGE_EGRESS_CHANNEL_DEPTH,
41        // Floor at 0 so an unexpected imbalance never records a negative gauge
42        // (matches `RpcMetrics::finish`).
43        level.max(0),
44        &[],
45    );
46}
47
48/// Apply `delta` to `atomic` and return the resulting level — the exact
49/// arithmetic [`sent`], [`received`], and [`reconcile_residual`] apply to the
50/// shared [`DEPTH`]. Parameterized over the atomic (issue #2419 roborev job
51/// 1733, the #2451 flake class) so a test can pin this SAME logic against a
52/// private, per-test atomic instead of racing every other concurrently-running
53/// test that also drives a real merge egress channel through the shared global.
54fn adjust(atomic: &AtomicI64, delta: i64) -> i64 {
55    atomic.fetch_add(delta, Ordering::SeqCst) + delta
56}
57
58/// Account one DATA entry that was just successfully sent into the bounded
59/// egress channel. Balanced by exactly one [`received`] or, for entries a
60/// consumer never pulled, by [`reconcile_residual`] post-join.
61pub(super) fn sent() {
62    record(adjust(&DEPTH, 1));
63}
64
65/// Account one DATA entry that was just received (consumed) from the egress
66/// channel, decrementing the live occupancy.
67pub(super) fn received() {
68    record(adjust(&DEPTH, -1));
69}
70
71/// Subtract `residual` DATA entries from the shared depth in ONE atomic op
72/// (issue #2419 roborev job 1733) — the post-join reconcile for entries a
73/// producer sent but its consumer never received. A no-op for `residual <= 0`
74/// (the common case: every send was received). See the module doc for why this
75/// must run only after the producer thread has been joined.
76pub(super) fn reconcile_residual(residual: i64) {
77    if residual > 0 {
78        record(adjust(&DEPTH, -residual));
79    }
80}

82#[cfg(test)]
83mod tests {
84    use super::*;

86    /// Producer-fast / consumer-slow harness against the tracked
87    /// send/recv/reconcile arithmetic (issue #2419, spec scenario "Depth rises
88    /// when the consumer is slower than the producer"): the depth climbs above
89    /// zero while sends outrun receives, stays bounded by the notional channel
90    /// capacity, and returns to exactly zero once every send is balanced by a
91    /// receive.
92    ///
93    /// Roborev job 1733 (the #2451 flake class): this now runs against a
94    /// PRIVATE, per-test `AtomicI64` via [`adjust`] — the exact same arithmetic
95    /// [`sent`]/[`received`] apply to the shared [`DEPTH`] — rather than the
96    /// process-global atomic. The prior version drove `DEPTH` directly, so any
97    /// concurrently-running test in this binary that ALSO exercises a real merge
98    /// egress channel (elsewhere in the crate) could perturb the shared level
99    /// mid-window, flaking this test's exact-equality assertions. A local atomic
100    /// makes the pin fully deterministic: no other test can ever touch it.
101    #[test]
102    fn depth_rises_while_backed_up_and_returns_to_baseline() {
103        let local = AtomicI64::new(0);
104        const CAP: i64 = 256;
105
106        // Producer races ahead: fill toward the bounded capacity.
107        for _ in 0..CAP {
108            adjust(&local, 1);
109        }
110        let backed_up = local.load(Ordering::SeqCst);
111        assert!(
112            backed_up > 0,
113            "depth must rise above zero while the producer outruns the consumer \
114             (backed_up={backed_up})"
115        );
116        assert!(
117            backed_up <= CAP,
118            "tracked depth must stay bounded by the channel capacity \
119             (backed_up={backed_up}, cap={CAP})"
120        );
121
122        // Consumer drains every entry.
123        for _ in 0..CAP {
124            adjust(&local, -1);
125        }
126        assert_eq!(
127            local.load(Ordering::SeqCst),
128            0,
129            "after draining every entry the depth returns to exactly zero \
130             (every tracked send balanced by exactly one tracked receive)"
131        );
132    }
133
134    /// [`reconcile_residual`] subtracts a positive residual in one op and is a
135    /// no-op for `<= 0` — pinned against a private atomic for the same
136    /// determinism reason as the scenario test above.
137    #[test]
138    fn reconcile_residual_subtracts_positive_and_ignores_non_positive() {
139        let local = AtomicI64::new(10);
140        // Mirror `reconcile_residual`'s logic directly against the local atomic
141        // (the public function only targets the shared `DEPTH`).
142        let residual = 4;
143        if residual > 0 {
144            adjust(&local, -residual);
145        }
146        assert_eq!(local.load(Ordering::SeqCst), 6);
147
148        let before = local.load(Ordering::SeqCst);
149        for non_positive in [0, -3] {
150            if non_positive > 0 {
151                adjust(&local, -non_positive);
152            }
153        }
154        assert_eq!(
155            local.load(Ordering::SeqCst),
156            before,
157            "a non-positive residual must never adjust the level"
158        );
159    }

161    /// Pins the ACTUAL fix (issue #2419 roborev job 1733): a send that races
162    /// teardown — landing AFTER the point a pre-fix drain-until-`Empty` loop
163    /// would already have stopped looking — is still fully reconciled, because
164    /// the reconcile now runs from an authoritative post-join sent/received
165    /// DELTA computed once the producer thread is provably done, never a live
166    /// drain-then-drop race. Modeled deterministically against a private atomic
167    /// (never the shared `DEPTH`, for the same reason as the tests above): the
168    /// arithmetic mirrors exactly what `sent`/`received`/`reconcile_residual`
169    /// apply, and what `SSTableRowIteratorAdapter::drop` does with its own
170    /// `sent_count`/`received_count` fields.
171    #[test]
172    fn reconcile_after_join_absorbs_a_send_that_raced_teardown() {
173        let depth = AtomicI64::new(0);
174        let baseline = depth.load(Ordering::SeqCst);
175
176        // Normal traffic: three entries sent and received, balanced.
177        let mut sent_count: i64 = 0;
178        let mut received_count: i64 = 0;
179        for _ in 0..3 {
180            adjust(&depth, 1);
181            sent_count += 1;
182            adjust(&depth, -1);
183            received_count += 1;
184        }
185        assert_eq!(depth.load(Ordering::SeqCst), baseline);
186
187        // Teardown begins (cancel tripped, receiver about to be torn down).
188        // Model the exact race this fix addresses: the producer thread, still
189        // alive, gets ONE more send through — precisely the send a
190        // drain-until-`Empty`-then-drop loop could miss because it lands AFTER
191        // that loop's last `Empty` observation. The consumer never receives it.
192        adjust(&depth, 1);
193        sent_count += 1;
194
195        // The producer thread is now joined (guaranteed no more sends
196        // possible). Reconcile using the authoritative post-join delta, exactly
197        // as `SSTableRowIteratorAdapter::drop` does.
198        let residual = sent_count - received_count;
199        assert_eq!(residual, 1, "exactly the one entry that raced teardown");
200        if residual > 0 {
201            adjust(&depth, -residual);
202        }
203
204        assert_eq!(
205            depth.load(Ordering::SeqCst),
206            baseline,
207            "the post-join reconcile must absorb a send that raced teardown, \
208             returning the gauge to its pre-scenario baseline — the #2419 \
209             roborev job 1733 regression this fix eliminates"
210        );
211    }
