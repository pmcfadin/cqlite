1//! Data.db writer - writes partition and row data
2//!
3//! Generates the Data.db component with V5CompressedLegacy (NB) format.
4//! Maintains partition ordering by Murmur3 token and clustering ordering.
5//! Tracks file positions for Index.db generation.
6//!
7//! Critical requirements:
8//! - Partition ordering: By Murmur3 token, then key bytes (enforced by caller)
9//! - Clustering ordering: By clustering comparator within partition (enforced by caller)
10//! - Row size measurement: After VInt length bytes (Issue #237)
11//! - Delta encoding: Uses Statistics.db baseline for timestamps/TTL/deletion times
12//!
13//! # V5CompressedLegacy Row Format
14//!
15//! Each row is encoded as:
16//! ```text
17//! [row_flags: u8]
18//! [extended_flags: u8 if ROW_HAS_EXTENDED_FLAGS set]
19//! [clustering_prefix: variable if present]
20//! [row_size: VInt]                       ← Measured from AFTER this VInt
21//! [prev_size: VInt]
22//! [timestamp: VInt if ROW_HAS_TIMESTAMP]   ← Delta from min_timestamp
23//! [ttl: VInt if ROW_HAS_TTL]              ← Delta from min_ttl
24//! [deletion: 2 VInts if ROW_HAS_DELETION] ← local_deletion_time delta + deletion timestamp
25//! [column_bitmap: VUInt bitmask of missing columns if NOT ROW_HAS_ALL_COLUMNS]
26//! [cell_data...]
27//! ```
28//!
29//! ## Row Flags
30//! - `0x04` (HAS_TIMESTAMP): Timestamp delta present
31//! - `0x08` (HAS_TTL): TTL delta present
32//! - `0x10` (HAS_DELETION): Deletion time present (two VInts)
33//! - `0x20` (HAS_ALL_COLUMNS): All columns present (no bitmap)
34//! - `0x40` (HAS_COMPLEX_DELETION): Row contains complex column with deletion
35//! - `0x80` (HAS_EXTENDED_FLAGS): Extended flags byte follows
36//!
37//! ## Cell Format
38//! ```text
39//! [flags: u8]
40//! [timestamp: VInt if NOT USE_ROW_TIMESTAMP]  ← Delta from min_timestamp
41//! [local_deletion_time: VUInt if deleted/expiring and NOT USE_ROW_TTL]
42//! [ttl: VUInt if expiring and NOT USE_ROW_TTL]
43//! [value_length: VInt]
44//! [value_bytes]
45//! ```
46//!
47//! ## Cell Flags
48//! - `0x01` (IS_DELETED): Cell is a tombstone
49//! - `0x02` (IS_EXPIRING): TTL fields follow
50//! - `0x04` (HAS_EMPTY_VALUE): Zero-length value
51//! - `0x08` (USE_ROW_TIMESTAMP): Use row-level timestamp (no timestamp delta)
52//! - `0x10` (USE_ROW_TTL): Use row-level TTL (no TTL delta)
53//!
54//! References:
55//! - Cassandra 5.0: `org.apache.cassandra.db.rows.UnfilteredSerializer`
56//! - Parser: `cqlite-core/src/storage/sstable/reader/parsing/row_decoder.rs`
57//! - Format docs: `docs/sstables-definitive-guide/chapters/05-data-db-format.md`

103// Range tombstone bound kinds.
104//
105// These are the ordinals of Cassandra's `ClusteringPrefix.Kind` enum
106// (ClusteringPrefix.java) — the byte written on disk by
107// `ClusteringBoundOrBoundary.Serializer.serialize()`:
108//   0 = EXCL_END_BOUND, 1 = INCL_START_BOUND,
109//   2 = EXCL_END_INCL_START_BOUNDARY, 3 = STATIC_CLUSTERING,
110//   4 = CLUSTERING, 5 = INCL_END_EXCL_START_BOUNDARY,
111//   6 = INCL_END_BOUND, 7 = EXCL_START_BOUND.
112// (Issue #717: the writer previously used a private 0..5 numbering that no
113// Cassandra reader understands.)
114const EXCL_END_BOUND: u8 = 0;
115const INCL_START_BOUND: u8 = 1;
116const INCL_END_BOUND: u8 = 6;
117const EXCL_START_BOUND: u8 = 7;
118// The CLUSTERING ordinal (a regular row's clustering name). Used by
119// `partition::sort_class` as the row's Kind tiebreak so co-located bounds order
120// against rows exactly as Cassandra's comparator does (issue #1220).
121const CLUSTERING: u8 = 4;
122
123// Range tombstone BOUNDARY kinds (issue #1220). A boundary closes one range and
124// opens the next at the SAME clustering point, carrying TWO deletion-time pairs
125// (primary = end/close of the previous range, secondary = start/open of the next
126// range). Cassandra emits these whenever two adjacent ranges share a boundary
127// point with complementary inclusivity (`ClusteringBoundOrBoundary.Serializer`):
128//   2 = EXCL_END_INCL_START_BOUNDARY (close exclusive, open inclusive),
129//   5 = INCL_END_EXCL_START_BOUNDARY (close inclusive, open exclusive).
130const EXCL_END_INCL_START_BOUNDARY: u8 = 2;
131const INCL_END_EXCL_START_BOUNDARY: u8 = 5;

136/// Capacity of the streaming Data.db `BufWriter` (Issue #492).
137///
138/// Large enough that each flushed partition coalesces into a handful of big
139/// `write()` syscalls instead of many small default-8 KB ones, preserving the
140/// throughput of the previous single whole-file write while keeping resident
141/// memory bounded (this buffer plus one partition's scratch).
142const DATA_SINK_BUFFER_BYTES: usize = 1024 * 1024;

144/// Data.db component writer
145///
146/// Writes partitions and rows in V5CompressedLegacy format with delta encoding.
147/// Caller must provide partitions in token order and rows in clustering order.
148///
149/// # Memory model (Issue #492)
150///
151/// The writer supports two modes that produce **byte-identical** Data.db output:
152///
153/// * **In-memory mode** (`DataWriter::new`): every partition is appended to the
154///   `buffer` scratch and never flushed, so `finish()` returns the full Data.db
155///   bytes. Used by unit tests that inspect the produced bytes directly.
156///
157/// * **Streaming mode** (`DataWriter::with_sink`): each partition is built in the
158///   `buffer` scratch, written to a `BufWriter<File>` over the Data.db path, and
159///   the scratch is cleared. Peak heap is therefore `O(largest partition)` rather
160///   than `O(file)`, keeping a multi-GB compaction within the 128 MB target.
161///
162/// In both modes the file offset of a partition is `position + buffer.len()`
163/// measured before any bytes are written. In streaming mode `buffer` is empty at
164/// that point (just cleared) so the offset is `position`; in memory mode
165/// `position` is always 0 and `buffer` holds all prior partitions, so the offset
166/// equals the legacy `buffer.len()`. The within-partition size math uses relative
167/// deltas into `buffer`, which are identical regardless of mode.
168#[derive(Debug)]
169pub struct DataWriter {
170    /// Per-partition scratch buffer for Data.db content.
171    ///
172    /// In streaming mode this is cleared at the start of every `write_partition`
173    /// and flushed to `sink` at the end, so only one partition is resident.
174    /// In memory mode it accumulates the entire Data.db output.
175    buffer: Vec<u8>,
176    /// Reusable scratch buffer for one row's serialized body (issue #1673, R2).
177    ///
178    /// Each row's body (timestamp/TTL/deletion/column-bitmap/cells, everything
179    /// after the row_size VInt) is built into this buffer, then appended to
180    /// `buffer`. `build_merged_row_body` / `build_static_row_body` `clear()` it
181    /// at the start of every row — `clear()` retains capacity, so after warmup no
182    /// per-row body allocation occurs (it previously allocated a fresh throwaway
183    /// `Vec` per row). Distinct field from `buffer` so the two can be borrowed
184    /// disjointly (`buffer.extend_from_slice(&row_scratch)`).
185    row_scratch: Vec<u8>,
186    /// Streaming sink over the Data.db path (streaming mode only).
187    ///
188    /// Lazily opened on the first `write_partition` so that the keyspace/table
189    /// directory exists before the first byte is written. `None` in in-memory
190    /// mode.
191    sink: Option<std::io::BufWriter<std::fs::File>>,
192    /// Data.db output path (streaming mode only); used for lazy sink open.
193    data_path: Option<PathBuf>,
194    /// Bytes already flushed to `sink`. Always 0 in in-memory mode.
195    position: u64,
196    /// Incremental whole-file (`Digest.crc32`) + per-chunk (`CRC.db`) checksum
197    /// accumulator, fed every flushed byte in write order (issue #1663).
198    ///
199    /// Streaming mode only: [`flush_partition`](Self::flush_partition) feeds each
200    /// flushed scratch buffer through it, so [`finish_streaming`](Self::finish_streaming)
201    /// can return both checksums without re-reading the finished `Data.db`. Left
202    /// untouched (empty) in in-memory mode, which never calls `finish_streaming`.
203    crc: StreamingCrc,
204    /// Encoding baselines used for delta encoding.
205    stats: EncodingStatsBaselines,
206    /// Issue #1741: emit the partition-level `DeletionTime` in the oa/`da`
207    /// serialization (1-byte `0x80` LIVE sentinel; `markedForDeleteAt`(i64) +
208    /// `localDeletionTime`(u32) when deleted) rather than the legacy na/`nb`
209    /// layout (`localDeletionTime`(i32) + `markedForDeleteAt`(i64), LIVE encoded as
210    /// `i32::MAX`+`i64::MIN`). `true` ONLY for `da` (BTI) SSTables, whose reader
211    /// applies `hasUIntDeletionTime` (oa) decoding; a `da` file written with the
212    /// legacy layout has its live-partition sentinel misread as a tombstone (which
213    /// the read-side shadowing then treats as a partition delete). Default `false`
214    /// preserves byte-identical `nb` output.
215    oa_partition_deletion: bool,
216    /// Schema-constant ordered column lists + per-column complexity, computed
217    /// exactly once per writer (issue #1674, R3).
218    ///
219    /// The schema is fixed for a writer's lifetime, yet `regular_columns` /
220    /// `static_columns` were re-filtered and re-sorted up to 3× per row, and the
221    /// sort comparator (`column_order_key` → `is_complex_column`) allocated a
222    /// lowercased `String` on every comparison — `O(R·C·log C)` allocations for R
223    /// rows × C columns. This memoizes the ordered column INDICES (Cassandra
224    /// serialization-header order: `(is_complex, name)` key) plus a per-column
225    /// `is_complex` flag, so the per-row hot path reads cached data and
226    /// `to_lowercase` never runs per row. Stored as indices (not `&Column`) to
227    /// avoid entangling the long-lived cache with the per-call `schema` borrow;
228    /// callers resolve `schema.columns[idx]` at use. Filled lazily on first use
229    /// via [`OnceCell`](std::cell::OnceCell) (interior mutability keeps the
230    /// existing `&self` accessors), which also guarantees single computation.
231    column_cache: std::cell::OnceCell<OrderedCols>,
232}

325///
326/// Carries the total Data.db byte size and the checksums accumulated during the

370    /// Create a streaming Data.db writer that flushes each partition to `data_path`.
371    ///
372    /// The file is opened lazily on the first `write_partition` (creating the

390            column_cache: std::cell::OnceCell::new(),
391        }

394    /// Lazily open the streaming sink (and create the parent directory).
395    ///
396    /// No-op in in-memory mode or once the sink is already open.
397    pub(super) fn ensure_sink(&mut self) -> Result<()> {
398        if self.sink.is_some() {
399            return Ok(());
400        }
401        if let Some(path) = self.data_path.clone() {
402            if let Some(parent) = path.parent() {
403                // Create the keyspace/table tree. The flush durability barrier
404                // fsyncs the full leaf→data-root chain unconditionally before
405                // the WAL truncate, so this creation need not track which
406                // ancestors it made (issue #1392).
407                crate::storage::write_engine::durability::create_dir_all(parent)?;
408            }
409            let file = std::fs::File::create(&path)?;
410            // Use a large BufWriter so a partition's bytes coalesce into a few
411            // big write() syscalls rather than many 8 KB-default ones, matching
412            // the throughput of the old single whole-file write.
413            self.sink = Some(std::io::BufWriter::with_capacity(
414                DATA_SINK_BUFFER_BYTES,
415                file,
416            ));
417        }
418        Ok(())
419    }

421    /// In streaming mode, flush the current scratch buffer to the sink, advance
422    /// `position`, and clear the scratch so only one partition is ever resident.
423    /// No-op in in-memory mode (the scratch keeps accumulating).
424    pub(super) fn flush_partition(&mut self) -> Result<()> {
425        if self.data_path.is_none() {
426            // In-memory mode: keep accumulating in `buffer`.
427            return Ok(());
428        }
429        self.ensure_sink()?;
430        if let Some(sink) = self.sink.as_mut() {
431            sink.write_all(&self.buffer)?;
432        }
433        // Feed the just-written bytes through the incremental checksum accumulator
434        // (issue #1663) so `finish_streaming` never re-reads the finished Data.db.
435        // Done here (after the sink write, before the clear) so every byte written
436        // to disk is checksummed exactly once, in write order — chunks straddle
437        // partition boundaries just as the re-read oracle chunks the raw file.
438        self.crc.update(&self.buffer);
439        self.position += self.buffer.len() as u64;
440        self.buffer.clear();
441        Ok(())
442    }

444    /// Flush the current scratch `buffer` to the streaming sink MID-PARTITION
445    /// (issue #2299), advancing `position` and clearing the scratch.
446    ///
447    /// Mechanically identical to [`Self::flush_partition`] (feed the CRC/`CRC.db`
448    /// accumulator in write order, advance `position`, clear `buffer`), but named
449    /// distinctly to document the caller's contract: it is safe to call between
450    /// two whole promoted-index blocks of an IN-PROGRESS partition ONLY because the
451    /// streaming session tracks every partition offset as flush-invariant absolute
452    /// math (`writer.position() - partition_offset`), not as a `buffer`-relative
453    /// index. Calling it mid-BLOCK would be equally correct for the on-disk bytes
454    /// (they are append-only), but the session only calls it at block boundaries so
455    /// its bounded-scratch guarantee is exactly one promoted-index block. No-op in
456    /// in-memory mode (the scratch keeps accumulating, matching `flush_partition`).
457    pub(super) fn flush_buffered_partition_scratch(&mut self) -> Result<()> {
458        // Same body as `flush_partition`: the two differ only in intent/documentation.
459        self.flush_partition()
460    }

479    /// Finish writing and return the Data.db bytes (in-memory mode).
480    ///
481    /// Only valid for writers created via [`DataWriter::new`]. In streaming mode
482    /// the bytes live on disk; use [`DataWriter::finish_streaming`] instead.
483    pub fn finish(self) -> Result<Vec<u8>> {
484        // Hard guard (not debug_assert!, which compiles out in release): on a
485        // streaming writer the bytes live on disk and `buffer` is empty after each
486        // partition flush, so returning it would silently yield a 0-byte Data.db.
487        if self.data_path.is_some() {
488            return Err(Error::InvalidInput(
489                "DataWriter::finish() called on a streaming writer; use finish_streaming()"
490                    .to_string(),
491            ));
492        }
493        Ok(self.buffer)
494    }

496    /// Finish a streaming writer: flush the sink to disk and return the total
497    /// byte size plus the checksums accumulated during the write (issue #1663).
498    ///
499    /// Any residual scratch (there is none in normal operation, since
500    /// `write_partition` flushes per partition) is flushed first. Returns an
501    /// error if the writer was created in in-memory mode.
502    ///
503    /// The returned [`StreamFinish`] carries the whole-file `Digest.crc32` value
504    /// and the per-chunk `CRC.db` values that were accumulated as the data
505    /// streamed to disk, so `SSTableWriter::finish` computes neither by
506    /// re-reading the finished `Data.db`.
507    pub fn finish_streaming(mut self) -> Result<StreamFinish> {
508        if self.data_path.is_none() {
509            return Err(Error::InvalidInput(
510                "finish_streaming() called on an in-memory DataWriter".to_string(),
511            ));
512        }
513        // Flush any residual scratch (normally empty), then flush the BufWriter
514        // so all bytes reach the OS file, and fsync the file so its *contents*
515        // are durable on the storage device (issue #1392). A plain `flush()`
516        // only pushes bytes into the page cache; without the fsync a crash after
517        // the WAL is truncated could lose the Data.db contents even though the
518        // directory entry was persisted. This fsync (plus the per-component
519        // fsyncs in `SSTableWriter::finish`) is the durability guarantee for
520        // both the flush and compaction write paths.
521        self.flush_partition()?;
522        if let Some(mut sink) = self.sink.take() {
523            sink.flush()?;
524            sink.get_ref()
525                .sync_all()
526                .map_err(|e| Error::Storage(format!("Failed to fsync Data.db contents: {e}")))?;
527        }
528        // Finalize the incremental checksums accumulated in `flush_partition`
529        // (issue #1663). `finalize` closes any trailing short chunk.
530        let data_size = self.position;
531        let (digest_crc32, chunk_crcs) = self.crc.finalize();
532        Ok(StreamFinish {
533            data_size,
534            digest_crc32,
535            chunk_crcs,
536        })
537    }
538
539    /// Get current file position (for Index.db offset tracking).
540    ///
541    /// This is the total number of Data.db bytes produced so far: bytes already
542    /// flushed to the sink (`position`) plus bytes currently buffered. Identical
543    /// in both streaming and in-memory modes.
544    pub fn position(&self) -> u64 {
545        self.position + self.buffer.len() as u64
546    }
547
548    /// Length of the per-partition scratch buffer.
549    ///
550    /// In streaming mode this reflects only the most recently written partition
551    /// (the scratch is cleared after each flush), which is the basis of the
552    /// bounded-memory guarantee. Test-only accessor.
553    #[cfg(test)]
554    pub(crate) fn scratch_len(&self) -> usize {
555        self.buffer.len()
556    }
557
558    /// Number of bytes already flushed to the streaming sink. Test-only accessor.
559    #[cfg(test)]
560    pub(crate) fn flushed_position(&self) -> u64 {
561        self.position
562    }
