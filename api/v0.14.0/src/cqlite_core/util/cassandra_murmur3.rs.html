1//! Cassandra-compatible Murmur3 hash implementation
2//!
3//! This is an exact port of Cassandra's `MurmurHash.hash3_x64_128` from
4//! `org.apache.cassandra.utils.MurmurHash`. It preserves Cassandra's
5//! sign-extension bug in tail byte processing, which causes it to diverge
6//! from the standard Murmur3 algorithm for inputs containing bytes >= 0x80.
7//!
8//! Used for:
9//! - Murmur3Partitioner token computation
10//! - Bloom filter (Filter.db) hash computation
11
12const C1: u64 = 0x87c3_7b91_1142_53d5;
13const C2: u64 = 0x4cf5_ad43_2745_937f;
14
15// Body mix constants (per-block additive constants for h1 and h2).
16const H1_ADD: u64 = 0x52dc_e729;
17const H2_ADD: u64 = 0x3849_5ab5;
18
19// Finalization mix multipliers used by `fmix64`.
20const FMIX_C1: u64 = 0xff51_afd7_ed55_8ccd;
21const FMIX_C2: u64 = 0xc4ce_b9fe_1a85_ec53;
22
23/// Compute Cassandra's `MurmurHash.hash3_x64_128` with seed 0.
24///
25/// Returns `(h1, h2)` matching `result[0]` and `result[1]` from the Java implementation.
26///
27/// This differs from standard Murmur3 in that tail bytes are sign-extended
28/// (Java's `byte` is signed), not zero-extended.
29pub fn cassandra_murmur3_x64_128(data: &[u8]) -> (i64, i64) {
30    let mut h1: u64 = 0;
31    let mut h2: u64 = 0;
32
33    // Body: process 16-byte blocks.
34    // Cassandra's getBlock reads bytes as unsigned (& 0xff) in little-endian order,
35    // which is equivalent to u64::from_le_bytes.
36    let mut chunks = data.chunks_exact(16);
37    for block in &mut chunks {
38        let mut k1 = u64::from_le_bytes(block[0..8].try_into().unwrap());
39        let mut k2 = u64::from_le_bytes(block[8..16].try_into().unwrap());
40
41        k1 = k1.wrapping_mul(C1);
42        k1 = k1.rotate_left(31);
43        k1 = k1.wrapping_mul(C2);
44        h1 ^= k1;
45        h1 = h1.rotate_left(27);
46        h1 = h1.wrapping_add(h2);
47        h1 = h1.wrapping_mul(5).wrapping_add(H1_ADD);
48
49        k2 = k2.wrapping_mul(C2);
50        k2 = k2.rotate_left(33);
51        k2 = k2.wrapping_mul(C1);
52        h2 ^= k2;
53        h2 = h2.rotate_left(31);
54        h2 = h2.wrapping_add(h1);
55        h2 = h2.wrapping_mul(5).wrapping_add(H2_ADD);
56    }
57
58    // Tail: process remaining bytes.
59    // Cassandra's bug: `(long) key.get(offset+N)` sign-extends because Java's byte is signed.
60    // We replicate this with: byte as i8 as i64 as u64.
61    let tail = chunks.remainder();
62    let mut k1: u64 = 0;
63    let mut k2: u64 = 0;
64
65    // Sign-extend a byte the way Java does: byte → signed byte → long.
66    let signed = |byte: u8| -> u64 { byte as i8 as i64 as u64 };
67
68    // Shift schedule matches the Java fall-through switch:
69    // positions 0..8 feed k1 (shifts 0, 8, 16, 24, 32, 40, 48, 56);
70    // positions 8..15 feed k2 (shifts 0, 8, 16, 24, 32, 40, 48).
71    for (pos, &byte) in tail.iter().enumerate() {
72        let shift = ((pos % 8) * 8) as u32;
73        if pos < 8 {
74            k1 ^= signed(byte) << shift;
75        } else {
76            k2 ^= signed(byte) << shift;
77        }
78    }
79
80    if tail.len() >= 9 {
81        k2 = k2.wrapping_mul(C2);
82        k2 = k2.rotate_left(33);
83        k2 = k2.wrapping_mul(C1);
84        h2 ^= k2;
85    }
86    if !tail.is_empty() {
87        k1 = k1.wrapping_mul(C1);
88        k1 = k1.rotate_left(31);
89        k1 = k1.wrapping_mul(C2);
90        h1 ^= k1;
91    }
92
93    // Finalization
94    let len = data.len() as u64;
95    h1 ^= len;
96    h2 ^= len;
97
98    h1 = h1.wrapping_add(h2);
99    h2 = h2.wrapping_add(h1);
100
101    h1 = fmix64(h1);
102    h2 = fmix64(h2);
103
104    h1 = h1.wrapping_add(h2);
105    h2 = h2.wrapping_add(h1);
106
107    (h1 as i64, h2 as i64)
108}

110/// Compute a Murmur3 partition token matching Cassandra's Murmur3Partitioner.
111///
112/// Takes h1 from the 128-bit hash and applies `normalize`:
113/// maps `i64::MIN` to `i64::MAX` (Cassandra excludes the minimum token value).
114pub fn cassandra_murmur3_token(data: &[u8]) -> i64 {
115    let (h1, _) = cassandra_murmur3_x64_128(data);
116    normalize(h1)
117}

119/// Compare two raw partition keys by Cassandra's canonical global (token-ring)
120/// order: Murmur3 token first, raw key bytes as a deterministic tiebreak.
121///
122/// This is THE cross-partition order Cassandra presents rows in, and the single
123/// authoritative definition shared by every access path that concatenates rows
124/// from more than one source:
125/// - the streaming k-way merge (`sstable/mod.rs`, #1580),
126/// - the `WHERE pk IN (...)` fan-out (`select_executor::lookup::sort_rows_by_token`,
127///   #955), and
128/// - the multi-candidate `scan_partition` / metadata concat fallbacks (#1917).
129///
130/// Raw-byte order alone is the WRONG global order (it differs from token order for
131/// almost all key sets), so all of the above route through this comparator to keep
132/// row order identical regardless of which path served the query.
133pub fn cmp_partition_keys_by_token(a: &[u8], b: &[u8]) -> std::cmp::Ordering {
134    cassandra_murmur3_token(a)
135        .cmp(&cassandra_murmur3_token(b))
136        .then_with(|| a.cmp(b))
137}

139/// Compute `DecoratedKey.filterHashLowerBits()` for a raw partition key, matching
140/// Cassandra 5.0's `org.apache.cassandra.db.DecoratedKey`.
141///
142/// Cassandra computes the full 128-bit Murmur3 hash of the partition-key bytes and
143/// uses the two 64-bit words distinctly:
144/// - `hash[0]` (`h1`, after `normalize`) becomes the `Murmur3Partitioner` token.
145/// - `hash[1]` (`h2`, **un-normalized**) is returned by `filterHashLowerBits()` and
146///   is the value whose **low 8 bits** the BTI `Partitions.db` partition index writes
147///   as each leaf's "hash byte" (fast mismatch pre-filter).
148///
149/// Reference: `DecoratedKey.filterHashLowerBits()` returns `hash[1]`
150/// (`org.apache.cassandra.db.DecoratedKey`, Cassandra 5.0.8); the BTI partition index
151/// stores `(byte) decoratedKey.filterHashLowerBits()`
152/// (`org.apache.cassandra.io.sstable.format.bti.PartitionIndex`).
153///
154/// Verified against the real `da-2-bti-Partitions.db` fixture
155/// (`test-data/datasets/sstables/test_da/simple_table-*`): the three UUID partitions
156/// store hash bytes `0x24`, `0x22`, `0xf4`, which are exactly the low bytes of `h2`
157/// for those keys.
158///
159/// Note this is `h2`, NOT the token's `h1`; the placeholder it replaces incorrectly
160/// derived the byte from the byte-comparable `h1` token.
161pub fn cassandra_partition_filter_hash_lower_bits(data: &[u8]) -> i64 {
162    let (_, h2) = cassandra_murmur3_x64_128(data);
163    h2
164}

166/// Cassandra's normalize: `Long.MIN_VALUE` maps to `Long.MAX_VALUE`.
167fn normalize(v: i64) -> i64 {
168    if v == i64::MIN {
169        i64::MAX
170    } else {
171        v
172    }
173}

175fn fmix64(mut value: u64) -> u64 {
176    value ^= value >> 33;
177    value = value.wrapping_mul(FMIX_C1);
178    value ^= value >> 33;
179    value = value.wrapping_mul(FMIX_C2);
180    value ^= value >> 33;
181    value
182}

198    /// A deterministic key set whose raw-byte order inverts token order, so the
199    /// two comparators cannot be conflated. Keys are simple ASCII partition keys.
200    fn inverting_key_set() -> Vec<Vec<u8>> {
201        (0u32..24)
202            .map(|i| format!("pk{i:03}").into_bytes())
203            .collect()
204    }

224    fn cmp_by_token_is_a_total_order_consistent_with_sorting() {

229        keys.reverse();

234            assert!(

238                w[1],

243    #[test]
244    fn token_order_differs_from_raw_byte_order() {
245        // Regression guard for #1917: token order is NOT raw-byte order. The old
246        // concat sites re-sorted by `a.0.cmp(&b.0)` (raw bytes); this asserts that
247        // order genuinely differs from the token order Cassandra presents, so the
248        // swap is observable, not cosmetic.
249        let keys = inverting_key_set();
250
251        let mut by_bytes = keys.clone();
252        by_bytes.sort();
253
254        let mut by_token = keys.clone();
255        by_token.sort_by(|a, b| cmp_partition_keys_by_token(a, b));
256
257        assert_ne!(
258            by_bytes,
259            by_token,
260            "expected raw-byte order to differ from token order for {} keys",
261            keys.len()
262        );
263
264        // And prove at least one adjacent pair actually inverts.
265        let inverts = keys.iter().enumerate().any(|(i, a)| {
266            keys.iter().skip(i + 1).any(|b| {
267                let token_ord = cmp_partition_keys_by_token(a, b);
268                let byte_ord = a.as_slice().cmp(b.as_slice());
269                token_ord != Ordering::Equal && byte_ord != Ordering::Equal && token_ord != byte_ord
270            })
271        });
272        assert!(
273            inverts,
274            "no inverting pair found — key set is a poor regression guard"
275        );
276    }

495    /// Validate the byte-comparable token encoding for the three UUID keys
496    /// used in the test_da/simple_table BTI fixture.  These values were verified
497    /// against the real Partitions.db trie structure (issue #755).
498    ///
499    /// The byte-comparable form of a Murmur3 token is:
500    ///   (token as u64) XOR 0x8000_0000_0000_0000  (flips sign bit → unsigned sort order)
501    ///
502    /// The first byte of each bc token matches the Sparse8 transition byte in the
503    /// real Partitions.db trie (see da-2-bti-Partitions.db hex analysis in #755).
504    #[test]
505    fn uuid_murmur3_token_bc_matches_bti_trie_transitions() {
506        // These expected values were verified by parsing the real
507        // da-2-bti-Partitions.db fixture and reading the Sparse8 transition bytes.
508        // Token order determines Data.db write order: 22 < 11 < 33.
509        struct Case {
510            uuid_byte: u8,
511            expected_token: i64,
512            expected_bc_first_byte: u8,
513        }
514        let cases = [
515            Case {
516                uuid_byte: 0x22,
517                expected_token: 1213057064512856170,
518                expected_bc_first_byte: 0x90,
519            },
520            Case {
521                uuid_byte: 0x11,
522                expected_token: 4360155383588533346,
523                expected_bc_first_byte: 0xBC,
524            },
525            Case {
526                uuid_byte: 0x33,
527                expected_token: 8780122315263850168u64 as i64,
528                expected_bc_first_byte: 0xF9,
529            },
530        ];
531        for c in &cases {
532            let uuid = [c.uuid_byte; 16];
533            let token = cassandra_murmur3_token(&uuid);
534            assert_eq!(
535                token, c.expected_token,
536                "token mismatch for UUID {:02X}*16",
537                c.uuid_byte
538            );
539            let bc = (token as u64) ^ 0x8000_0000_0000_0000u64;
540            assert_eq!(
541                (bc >> 56) as u8,
542                c.expected_bc_first_byte,
543                "bc first byte mismatch for UUID {:02X}*16",
544                c.uuid_byte
545            );
546        }
