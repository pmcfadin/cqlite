23    /// Get comprehensive reader health and performance metrics
24    pub async fn get_health_metrics(&self) -> Result<SSTableReaderHealthMetrics> {
25        let stats = self.stats().await?;
26
27        // Cache health is sourced from the shared B1 decompressed-chunk cache
28        // (issue #1568): the dead per-reader block cache and its always-empty
29        // memory summation were deleted, so these numbers now reflect the real
30        // cache rather than a structural zero.
31        let cache = self.chunk_cache();
32        let (hits, misses) = (cache.hit_count(), cache.miss_count());
33        let cache_hit_rate = if hits + misses > 0 {
34            hits as f64 / (hits + misses) as f64
35        } else {
36            0.0
37        };
38        let memory_usage = std::mem::size_of::<Self>() + cache.resident_bytes();
39
40        let file_path = self.file_path();
41        Ok(SSTableReaderHealthMetrics {
42            file_accessible: file_path.exists(),
43            file_path,
44            header_version: self.header.cassandra_version,
45            total_file_size: stats.file_size,
46            estimated_memory_usage: memory_usage,
47            block_cache_entries: cache.len(),
48            block_cache_hit_rate: cache_hit_rate,
49            compression_enabled: self.compression_reader.is_some(),
50            compression_algorithm: self.header.compression.algorithm.clone(),
51            bloom_filter_enabled: self.bloom_filter.is_some(),
52            index_available: self.index.is_some(),
53            generation: self.generation,
54            last_error: None,
55        })
56    }

58    /// Perform an integrity check on the SSTable.
59    ///
60    /// Issue #1283: this is a THIN PROJECTION over `verify::verify_sstable` — the
61    /// single source of truth for SSTable integrity — not an independent check
62    /// pipeline. The legacy implementation walked only `Data.db` blocks, so a
63    /// corrupt `Index.db` / `Digest.crc32` / `Summary.db` / `Filter.db` or
64    /// out-of-order keys (all of which the verifier FAILs) read back `Healthy`
65    /// here — a divergent verdict. We now run the authoritative verifier in
66    /// `Full` mode over the reader's EXACT generation (`self.file_path`, not merely
67    /// its parent directory — roborev #1283) and map its `VerifyReport` onto the
68    /// legacy `IntegrityCheckResult` shape the (test-only) consumers expect.
69    pub async fn perform_integrity_check(&self) -> Result<IntegrityCheckResult> {
70        let file_path = self.file_path();
71        debug!("Starting integrity check for {:?}", file_path);
72
73        // Delegate to the authoritative engine, verifying the EXACT generation this
74        // reader is opened on (issue #1283, roborev). The directory may hold several
75        // generations; `verify_sstable` resolves the lexicographically-first
76        // `*-Data.db`, which would report the wrong SSTable's integrity here.
77        // `verify_sstable_generation` verifies precisely `self.file_path`'s
78        // generation. We use the SAME Config/Platform the reader was opened with.
79        // Data corruption is reported as findings inside an Ok(report); only
80        // environmental problems return Err.
81        let report = verify::verify_sstable_generation(
82            &file_path,
83            VerifyMode::Full,
84            &self.open_config,
85            self.platform.clone(),
86        )
87        .await?;
88
89        // Project VerifyReport -> IntegrityCheckResult.
90        //  - any finding  => Corrupted; none => Healthy (no Degraded — issue #1283).
91        //  - rows_scanned => total_entries.
92        //  - findings' rendered strings => parsing_errors.
93        //  - per-block indices (corrupted_blocks/unreadable_blocks/total_blocks_checked)
94        //    are not produced by the verifier and no production consumer reads them,
95        //    so they stay best-effort empty/zero.
96        let parsing_errors: Vec<String> = report.findings.iter().map(|f| f.to_string()).collect();
97        let overall_status = if report.findings.is_empty() {
98            IntegrityStatus::Healthy
99        } else {
100            IntegrityStatus::Corrupted
101        };
102
103        // `checksum_mismatches` is a deprecated, always-0 compatibility field
104        // (issue #1283): the projection never populates it. The narrow allow keeps
105        // clippy `-D warnings` clean while the dead computation stays removed.
106        #[allow(deprecated)]
107        let result = IntegrityCheckResult {
108            file_path: file_path.clone(),
109            total_blocks_checked: 0,
110            corrupted_blocks: Vec::new(),
111            checksum_mismatches: 0,
112            unreadable_blocks: 0,
113            total_entries: report.rows_scanned.unwrap_or(0),
114            parsing_errors,
115            overall_status,
116        };
117
118        info!(
119            "Integrity check completed for {:?}: {:?}, {} findings, {} rows scanned",
120            file_path,
121            result.overall_status,
122            result.parsing_errors.len(),
123            result.total_entries
124        );
125
126        Ok(result)
127    }

129    /// Enhanced tombstone filtering using TombstoneMerger
130    #[cfg(feature = "tombstones")]
131    pub(super) fn filter_tombstone(&self, row: &ScanRow) -> bool {
132        // Issue #1334: a live row (`ScanRow::Row`) is always kept here — row-level
133        // tombstone suppression only applies to markers. (Cell tombstones inside a
134        // live row are preserved for callers to inspect.) Only a marker carries a
135        // `Value` whose tombstone/TTL semantics this filter evaluates.
136        let value = match row {
137            // A live row (decoded or raw undecoded fallback) is always kept.
138            ScanRow::Row(_) | ScanRow::RawRow(_) => return true,
139            ScanRow::Marker(v) => v,
140        };
141        // Use the fast tombstone check for performance
142        let write_time = self.extract_write_time_from_value(value);
143
144        if self
145            .tombstone_merger
146            .fast_tombstone_check(value, write_time)
147        {
148            // Value is deleted by tombstone
149            return false;
150        }
151
152        // Check for TTL expiration on regular values
153        if let Some(ttl) = self.extract_ttl_from_value(value) {
154            let current_time = std::time::SystemTime::now()
155                .duration_since(std::time::UNIX_EPOCH)
156                .map(|d| d.as_micros() as i64)
157                .unwrap_or_else(|e| {
158                    warn!("Failed to get system time: {}; using fallback value 0", e);
159                    0
160                });
161
162            if current_time > write_time + ttl {
163                // Value has expired
164                return false;
165            }
166        }
167
168        true // Keep valid, non-deleted values
169    }

171    /// Simple tombstone filtering (fallback when tombstones feature is disabled).
172    ///
173    /// Row tombstones (`Value::Tombstone(RowTombstone)`) are always filtered out of
174    /// user-facing scan/get results, regardless of the `tombstones` feature flag.
175    /// This prevents deleted rows that are still present on disk (either from a live
176    /// SSTable that contains a tombstone entry, or from a post-compaction SSTable
177    /// that preserved tombstone rows for GC purposes) from appearing in query results.
178    ///
179    /// Cell tombstones (`Value::Tombstone(CellTombstone)`) within a Map are NOT
180    /// filtered here — they are preserved so callers can inspect them.  If a caller
181    /// needs to suppress null-cell entries, it should do so at the query layer.
182    ///
183    /// (Issue #505)
184    #[cfg(not(feature = "tombstones"))]
185    pub(super) fn filter_tombstone(&self, row: &ScanRow) -> bool {
186        use crate::types::TombstoneType;
187        // Issue #1334: a live row (`ScanRow::Row`) is always kept; filter out only a
188        // row-level tombstone marker.
189        !matches!(
190            row,
191            ScanRow::Marker(Value::Tombstone(info))
192                if info.tombstone_type == TombstoneType::RowTombstone
193        )
194    }

196    /// Enhanced multi-generation tombstone filtering for compaction
197    #[cfg(feature = "tombstones")]
198    pub async fn filter_with_multi_generation_merge(
199        &self,
200        table_id: &TableId,
201        entries: Vec<(RowKey, Vec<GenerationValue>)>,
202    ) -> Result<Vec<(RowKey, ScanRow)>> {
203        let mut results = Vec::new();
204
205        tracing::debug!(
206            "Processing {} key groups for multi-generation merge",
207            entries.len()
208        );
209
210        // Use batch processing for better performance
211        const BATCH_SIZE: usize = 1000;
212
213        let batches: Vec<_> = entries.chunks(BATCH_SIZE).collect();
214
215        for (batch_idx, batch) in batches.iter().enumerate() {
216            tracing::debug!(
217                "Processing batch {}/{} with {} entries",
218                batch_idx + 1,
219                batches.len(),
220                batch.len()
221            );
222
223            let batch_entries = batch.to_vec();
224            let merged_results = self
225                .tombstone_merger
226                .batch_merge_with_tombstones(batch_entries, BATCH_SIZE)?;
227
228            for (key, merged_value) in merged_results {
229                if let Some(value) = merged_value {
230                    if self.should_include_value_after_merge(&value, table_id, &key)? {
231                        results.push((key, value));
232                    }
233                } else {
234                    // Value was completely tombstoned
235                    tracing::debug!("Value for key {:?} was completely tombstoned", key);
236                }
237            }
238        }
239
240        tracing::debug!(
241            "Multi-generation merge completed: {} final results from {} input groups",
242            results.len(),
243            entries.len()
244        );
245
246        Ok(results)
247    }

249    /// Enhanced filtering logic for post-merge values including collection validation
250    #[cfg(feature = "tombstones")]
251    fn should_include_value_after_merge(
252        &self,
253        row: &ScanRow,
254        _table_id: &TableId,
255        _key: &RowKey,
256    ) -> Result<bool> {
257        // Issue #1334: the merge now yields whole rows. A live row with at least one
258        // cell is included; a marker (row tombstone / null row) or an empty row is
259        // suppressed.
260        match row {
261            ScanRow::Row(cells) => Ok(!cells.is_empty()),
262            // A raw undecoded fallback row carries live bytes → included.
263            ScanRow::RawRow(bytes) => Ok(!bytes.is_empty()),
264            ScanRow::Marker(_) => Ok(false),
265        }
266    }

268    /// Extract TTL from value metadata
269    #[cfg(feature = "tombstones")]
270    fn extract_ttl_from_value(&self, value: &Value) -> Option<i64> {
271        match value {
272            Value::Tombstone(info) => info.ttl,
273            _ => None, // Regular values would have TTL in SSTable metadata
274        }
275    }

277    /// Extract write time from value
278    #[cfg(feature = "tombstones")]
279    fn extract_write_time_from_value(&self, value: &Value) -> i64 {
280        match value {
281            Value::Tombstone(info) => info.deletion_time,
282            _ => std::time::SystemTime::now()
283                .duration_since(std::time::UNIX_EPOCH)
284                .map(|d| d.as_micros() as i64)
285                .unwrap_or_else(|e| {
286                    warn!("Failed to get system time: {}; using fallback value 0", e);
287                    0
288                }),
289        }
290    }
