1//! Opt-in, default-OFF presence-oracle false-negative verification switch
2//! (issue #2163).
3//!
4//! A presence oracle (the BIG bloom filter / BTI Partitions.db trie) must NEVER
5//! report a false negative: a "definitely absent" verdict for a key that is
6//! actually present would silently drop data. This switch turns on an
7//! AUTHORITATIVE confirmation scan of an SSTable whenever its oracle reports a
8//! key absent; a contradiction increments `cqlite.read.bloom.false_negatives`
9//! (see [`crate::observability::catalog::READ_BLOOM_FALSE_NEGATIVES`]). Under a
10//! correct oracle the counter stays 0 — a non-zero value is a corruption alarm.
11//!
12//! It is **off by default** and gated by an explicit runtime switch, because the
13//! confirmation scan is the one presence-oracle counter that costs real work. The
14//! switch is read once from the `CQLITE_VERIFY_PRESENCE_ORACLE` environment
15//! variable (booleans: `1/0`, `true/false`, `yes/no`, `on/off`), then cached. The
16//! `observability-testing`-gated [`set_enabled_for_testing`] lets the correctness
17//! tests toggle it deterministically without touching process env.
18//!
19//! # Config plumbing + precedence (issue #2163, roborev r4)
20//!
21//! [`crate::observability::ObservabilityConfig::verify_presence_oracle`] is a
22//! PROGRAMMATIC way to set this switch (e.g. `ObservabilityConfig::builder()
23//! .verify_presence_oracle(true).build()`), for callers that construct config in
24//! code rather than through the environment. [`crate::observability::init`] calls
25//! [`apply_config`] with that field so the config value actually reaches this
26//! switch — it is not merely decorative. Precedence (conventional
27//! env-overrides-config, matching every other `ObservabilityConfig` field's own
28//! `from_env` behaviour): if [`ENV_VAR`] is explicitly set to a parseable boolean,
29//! it ALWAYS wins over the config value; otherwise the config value applies.
30
31use std::sync::atomic::{AtomicU8, Ordering};
32
33const UNINIT: u8 = 0;
34const OFF: u8 = 1;
35const ON: u8 = 2;
36
37/// Tri-state cache: `UNINIT` until first resolved from the environment, then
38/// pinned to `OFF`/`ON`. Test overrides store `OFF`/`ON` directly.
39static STATE: AtomicU8 = AtomicU8::new(UNINIT);
40
41/// Environment variable that enables the opt-in verification when truthy.
42pub const ENV_VAR: &str = "CQLITE_VERIFY_PRESENCE_ORACLE";
43
44fn parse_bool(s: &str) -> Option<bool> {
45    match s.trim().to_ascii_lowercase().as_str() {
46        "1" | "true" | "yes" | "on" => Some(true),
47        "0" | "false" | "no" | "off" => Some(false),
48        _ => None,
49    }
50}
51
52/// Whether presence-oracle false-negative verification is enabled.
53///
54/// Resolves lazily from [`ENV_VAR`] on first call (default `false` when unset or
55/// unparseable), caching the result so subsequent reads are a single relaxed
56/// atomic load. Always `false` unless explicitly turned on — so the default
57/// production read path performs no confirmation scan and costs nothing beyond
58/// the existing presence check.
59pub fn enabled() -> bool {
60    match STATE.load(Ordering::Acquire) {
61        OFF => false,
62        ON => true,
63        _ => {
64            let value = std::env::var(ENV_VAR)
65                .ok()
66                .and_then(|v| parse_bool(&v))
67                .unwrap_or(false);
68            let encoded = if value { ON } else { OFF };
69            // Only the first resolver wins; a concurrent test override that already
70            // pinned the state is respected (compare_exchange fails, we re-read).
71            match STATE.compare_exchange(UNINIT, encoded, Ordering::AcqRel, Ordering::Acquire) {
72                Ok(_) => value,
73                Err(existing) => existing == ON,
74            }
75        }
76    }
77}

79/// Apply an [`ObservabilityConfig`](crate::observability::ObservabilityConfig)-
80/// sourced enablement value to this runtime switch (issue #2163, roborev r4 —
81/// closes the "config field is decorative" gap).
82///
83/// [`ENV_VAR`], when explicitly set to a parseable boolean, ALWAYS takes
84/// precedence over `config_value` — the conventional env-overrides-config rule
85/// [`crate::observability::ObservabilityConfig::from_env`] already applies to
86/// every other field. Otherwise `config_value` becomes the switch's state. Called
87/// from [`crate::observability::init`] (both the `observability`-feature-enabled
88/// and inert builds) so a programmatic
89/// `ObservabilityConfig::builder().verify_presence_oracle(true).build()` actually
90/// flips the switch the read path consults. Idempotent and safe to call multiple
91/// times (e.g. across repeated `init` calls in a test process) — it always
92/// recomputes from scratch and FORCES the result into the shared state, so a
93/// later `init` call can still correct an earlier one.
94pub(crate) fn apply_config(config_value: bool) {
95    let env_value = std::env::var(ENV_VAR).ok().and_then(|v| parse_bool(&v));
96    let effective = env_value.unwrap_or(config_value);
97    STATE.store(if effective { ON } else { OFF }, Ordering::Release);
98}

100/// Force the switch on/off for tests (issue #2163). Gated behind
101/// `observability-testing` so production builds never expose a runtime mutator;
102/// integration tests compile the library with that feature and toggle the switch
103/// deterministically around a flow.
104#[cfg(feature = "observability-testing")]
105pub fn set_enabled_for_testing(enabled: bool) {
106    STATE.store(if enabled { ON } else { OFF }, Ordering::Release);
107}

109#[cfg(test)]
110mod tests {
111    use super::*;
112
113    #[test]
114    fn parse_bool_recognizes_bounded_forms() {
115        for t in ["1", "true", "TRUE", "yes", "on"] {
116            assert_eq!(parse_bool(t), Some(true), "{t}");
117        }
118        for f in ["0", "false", "no", "off"] {
119            assert_eq!(parse_bool(f), Some(false), "{f}");
120        }
121        assert_eq!(parse_bool("maybe"), None);
122    }
123
124    #[cfg(feature = "observability-testing")]
125    #[test]
126    fn test_override_toggles_state() {
127        set_enabled_for_testing(true);
128        assert!(enabled());
129        set_enabled_for_testing(false);
130        assert!(!enabled());
131    }
