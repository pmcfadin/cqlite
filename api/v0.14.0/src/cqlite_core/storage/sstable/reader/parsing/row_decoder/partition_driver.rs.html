1//! Issue #1640 (Epic K, finding **K1**): the single V5CompressedLegacy
2//! partition/row emit driver.
3//!
4//! The parser performance audit (`docs/reports/parser-performance-audit-2026-07-01.md`)
5//! found the partition/row emit skeleton — partition-header parse →
6//! END_OF_PARTITION / range-tombstone-marker checks → row-body decode →
7//! boundary peek → `ParseStep` advance / `flush_and_emitted!` — duplicated
8//! across five emit functions, with the subtle issue-#932 row-write-timestamp
9//! coexistence decision hand-copied into the two sliding-window loops. "This is
10//! how parity regressions are manufactured": a future tombstone fix must land in
11//! N places or the paths silently diverge.
12//!
13//! This module owns:
14//! - [`row_write_timestamp`] — the ONE issue-#932 row-write-timestamp decision.
15//! - [`SlidingPartitionPolicy`] / [`MarkerOutcome`] /
16//!   [`V5CompressedLegacyParser::drive_partition_sliding`] — the ONE bounded
17//!   sliding-window partition skeleton, driven by policy hooks that capture the
18//!   per-consumer differences (streaming-scan timestamps vs per-element
19//!   compaction rows).
20//!
21//! K1 is the HEAD of the K-emit chain: K2 (#1641 non-allocating boundary peek),
22//! K3 (#1642 positional row emit), K4 (#1643 `Arc` key handles) are separate
23//! lanes and are NOT folded in here.

27/// Issue #932 (K1): the single row-write-timestamp coexistence decision.
28///
29/// A `HAS_DELETION` row may ALSO carry a liveness timestamp (surviving cells
30/// written strictly after the row deletion). Prefer that liveness timestamp as
31/// the row timestamp so those cells inherit it and are NOT shadowed by the older
32/// row deletion during reconcile; fall back to `markedForDeleteAt` only for a
33/// PURE row tombstone (which has no `HAS_TIMESTAMP`). `0` when the row carries
34/// neither a liveness timestamp nor a deletion.
35///
36/// Formerly hand-copied into `parse_one_partition_with_timestamps`
37/// (`block_emit_windowed.rs`) and `parse_one_partition_for_compaction`
38/// (`compaction.rs`); centralized here so the decision lives in exactly one
39/// place (audit finding K1).
40pub(super) fn row_write_timestamp(row_header_opt: &Option<RowHeader>) -> i64 {
41    let row_tombstone = row_header_opt.as_ref().filter(|h| h.is_row_tombstone());
42    row_header_opt
43        .as_ref()
44        .and_then(|h| h.timestamp)
45        .or_else(|| row_tombstone.map(|h| h.row_tombstone_deletion_time()))
46        .unwrap_or(0)
47}

49/// How the driver should advance after a policy handled a range-tombstone marker.
50pub(super) enum MarkerOutcome {
51    /// The marker was consumed; continue the row loop at this offset.
52    Advanced(usize),
53    /// The marker could not be represented/parsed faithfully — terminate the
54    /// partition (the driver flushes buffered rows on the final chunk, else
55    /// returns `NeedMore`), mirroring the pre-K1 `break`/`NeedMore` behaviour.
56    Stop,
57}

59/// Per-consumer policy for the bounded sliding-window partition skeleton
60/// (issue #1640, K1). The driver owns the framing; each policy owns exactly the
61/// three behaviours that differ between the streaming-scan timestamps path and
62/// the per-element compaction path.
63///
64/// Both policies buffer emitted rows into the driver-owned `pending` vec (never
65/// forwarding directly), so a mid-partition `NeedMore` re-parse cannot double-
66/// emit (issue #827): `pending` is flushed only when the partition is confirmed
67/// complete.
68pub(super) trait SlidingPartitionPolicy {
69    /// The row carrier this policy emits.
70    type Row;
71
72    /// Called once, after the partition header is parsed, before the row loop.
73    /// `partition_deletion` is the header's `Option<(markedForDeleteAt µs,
74    /// localDeletionTime s)>`. The timestamps policy opens a read-side shadow
75    /// here (when shadowing); the compaction policy pushes a synthetic
76    /// partition-delete row for a partition tombstone (issue #1072).
77    fn on_partition_open(
78        &mut self,
79        partition_key: RowKey,
80        partition_deletion: Option<(i64, i32)>,
81        schema: &TableSchema,
82        pending: &mut Vec<Self::Row>,
83    );
84
85    /// Handle a range-tombstone marker at `offset`. The timestamps policy feeds
86    /// the range-tombstone FSM (or skips on a physical read); the compaction
87    /// policy pairs bound markers into `RangeMarker` rows (issue #933).
88    fn on_range_marker(
89        &mut self,
90        data: &[u8],
91        offset: usize,
92        schema: &TableSchema,
93        pending: &mut Vec<Self::Row>,
94    ) -> MarkerOutcome;
95
96    /// Decode and handle one data row at `offset`, pushing any emitted row into
97    /// `pending`. Returns `Some(next_offset)` on success, or `None` when the row
98    /// could not be parsed (the driver treats that as end-of-partition on the
99    /// final chunk, else `NeedMore`).
100    fn on_data_row(
101        &mut self,
102        data: &[u8],
103        offset: usize,
104        schema: &TableSchema,
105        reader: &crate::storage::sstable::reader::types::SSTableReader,
106        resolution: &RowColumnResolution,
107        pending: &mut Vec<Self::Row>,
108    ) -> Option<usize>;
109}

111impl V5CompressedLegacyParser {
112    /// The single bounded sliding-window partition skeleton (issue #1640, K1).
113    ///
114    /// Parses exactly ONE partition from the front of `data`, driving the
115    /// per-consumer `policy` hooks, and reports how the parse terminated via
116    /// [`ParseStep`] — identical `Emitted`/`NeedMore`/`Done` semantics to the
117    /// pre-K1 `parse_one_partition_with_timestamps` /
118    /// `parse_one_partition_for_compaction` bodies this replaces.
119    ///
120    /// `at_final_chunk` flips a mid-partition parse failure between a refill
121    /// request (`NeedMore`) and a terminal flush, exactly as before.
122    pub(super) fn drive_partition_sliding<P, F>(
123        &self,
124        data: &[u8],
125        schema: &TableSchema,
126        reader: &crate::storage::sstable::reader::types::SSTableReader,
127        at_final_chunk: bool,
128        policy: &mut P,
129        mut emit: F,
130    ) -> Result<ParseStep>
131    where
132        P: SlidingPartitionPolicy,
133        F: FnMut(P::Row) -> Result<std::ops::ControlFlow<()>>,
134    {
135        if data.is_empty() {
136            return Ok(ParseStep::Done);
137        }
138
139        // #1741 (roborev HIGH): size the header need-more decision correctly for
140        // the oa/da DeletionTime form via the authoritative discriminator peek,
141        // so a deleted header split across a NON-FINAL chunk returns `NeedMore`
142        // instead of being mis-parsed and skipped (which desynced the scan and,
143        // on compaction, dropped a partition tombstone).
144        match self.partition_header_readiness(data) {
145            PartitionHeaderReadiness::Malformed => return Ok(ParseStep::Emitted(1)),
146            PartitionHeaderReadiness::Incomplete => {
147                return Ok(if at_final_chunk {
148                    ParseStep::Done
149                } else {
150                    ParseStep::NeedMore
151                });
152            }
153            PartitionHeaderReadiness::Ready => {}
154        }
155
156        let (partition_key, mut offset, partition_deletion) = match self
157            .parse_partition_header_full(data, 0)
158        {
159            Ok(v) => v,
160            // Defense-in-depth: `Ready` guarantees the DeletionTime is fully
161            // present, so a parse failure here cannot be truncation. On a
162            // non-final chunk only re-request bytes if the header is still
163            // incomplete; otherwise skip a byte to resynchronise (NeedMore on
164            // a complete buffer would loop forever). Under `Ready` this stays
165            // the legacy skip-a-byte resync.
166            Err(_) => {
167                if !at_final_chunk
168                    && self.partition_header_readiness(data) == PartitionHeaderReadiness::Incomplete
169                {
170                    return Ok(ParseStep::NeedMore);
171                }
172                return Ok(ParseStep::Emitted(1));
173            }
174        };
175
176        // Issue #1046: per-PARTITION resolution build (this driver is re-entered
177        // once per partition by the sliding-window caller; allocations scale with
178        // partition count, not row count). Borrows header strings + schema.
179        let resolution = RowColumnResolution::build(schema, reader);
180
181        // Finding 1 (#827): buffer this partition's rows locally and forward them
182        // to the external `emit` only once the partition is CONFIRMED complete (an
183        // `Emitted` return). A mid-partition `NeedMore` discards `pending` so the
184        // caller can refill and re-parse from the partition start without
185        // double-emitting already-buffered rows. Bounded by ONE partition's rows.
186        let mut pending: Vec<P::Row> = Vec::new();
187
188        policy.on_partition_open(partition_key, partition_deletion, schema, &mut pending);
189
190        // Flush the buffered rows, honouring an early `Break`, and report the
191        // bytes consumed for this (complete) partition so the caller drains
192        // correctly.
193        macro_rules! flush_and_emitted {
194            ($consumed:expr) => {{
195                for row in pending.drain(..) {
196                    match emit(row)? {
197                        std::ops::ControlFlow::Continue(()) => {}
198                        std::ops::ControlFlow::Break(()) => break,
199                    }
200                }
201                Ok(ParseStep::Emitted($consumed))
202            }};
203        }
204
205        loop {
206            // END_OF_PARTITION (0x01): partition complete, consume the marker.
207            if offset < data.len() && Self::is_end_of_partition(data[offset]) {
208                offset += 1;
209                return flush_and_emitted!(offset);
210            }
211
212            // Consumed everything but never saw END_OF_PARTITION: the partition
213            // may continue in the next chunk. On NeedMore flush NOTHING (drop the
214            // buffered rows) so the caller can refill and re-parse from the start.
215            if offset >= data.len() {
216                if at_final_chunk {
217                    return flush_and_emitted!(offset);
218                }
219                return Ok(ParseStep::NeedMore);
220            }
221
222            if Self::is_range_tombstone_marker(data[offset]) {
223                match policy.on_range_marker(data, offset, schema, &mut pending) {
224                    MarkerOutcome::Advanced(next_offset) => {
225                        offset = next_offset;
226                        continue;
227                    }
228                    MarkerOutcome::Stop => {
229                        if at_final_chunk {
230                            return flush_and_emitted!(offset);
231                        }
232                        return Ok(ParseStep::NeedMore);
233                    }
234                }
235            }
236
237            match policy.on_data_row(data, offset, schema, reader, &resolution, &mut pending) {
238                Some(next_offset) => {
239                    offset = next_offset;
240                    if offset >= data.len() {
241                        // End of the buffer without an explicit END_OF_PARTITION:
242                        // the partition may continue in the next chunk.
243                        if at_final_chunk {
244                            return flush_and_emitted!(offset);
245                        }
246                        return Ok(ParseStep::NeedMore);
247                    }
248                    if self.peek_is_partition_header(data, offset) {
249                        // Next partition starts here — current one is complete.
250                        return flush_and_emitted!(offset);
251                    }
252                }
253                None => {
254                    // A row failed to parse. Mid-stream that may be a row
255                    // straddling the chunk boundary, so request more bytes unless
256                    // this is the final chunk (where it is end-of-partition).
257                    if at_final_chunk {
258                        return flush_and_emitted!(offset);
259                    }
260                    return Ok(ParseStep::NeedMore);
261                }
262            }
263        }
264    }
265}

370    /// Test-only [`SlidingPartitionPolicy`] over a synthetic buffer. It exercises
371    /// the driver's framing skeleton WITHOUT any real row decode: each
372    /// [`STUB_ROW_BYTE`] is one row (buffered into `pending`, consuming 1 byte),
373    /// any range-tombstone marker is answered with [`MarkerOutcome::Stop`], and

459    /// Build a synthetic **nb**-format partition: a LIVE header (no partition
460    /// tombstone) followed by `body` bytes. The parser built by
461    /// [`V5CompressedLegacyParser::new`] uses the nb-compatible gates
462    /// (`has_uint_deletion_time == false`), so the header is:
463    /// `flags(1) + key_len(1) + key(1) + nb DeletionTime` where the nb
464    /// DeletionTime is a 4-byte localDeletionTime (`i32::MAX` == LIVE sentinel) +
465    /// 8-byte markedForDeleteAt (0). Fixed bytes only — no wall-clock input.
466    #[cfg(feature = "write-support")]
467    fn synthetic_partition(body: &[u8]) -> Vec<u8> {
468        let mut buf = vec![0x00, 0x01, 0x42]; // flags, key_len=1, key=[0x42]
469        buf.extend_from_slice(&i32::MAX.to_be_bytes()); // LIVE localDeletionTime
470        buf.extend_from_slice(&0i64.to_be_bytes()); // markedForDeleteAt
471        buf.extend_from_slice(body);
472        buf
473    }

507    /// (a) Issue-#827 no-double-emit invariant: a truncated partition on a
508    /// NON-final chunk returns `NeedMore` and forwards ZERO rows — *even though*
509    /// the policy already buffered a row into `pending`. Discarding `pending` on a
510    /// mid-partition `NeedMore` is what makes a refill-and-re-parse from the
511    /// partition start safe (a forwarded row here would be duplicated on re-parse).
512    #[cfg(feature = "write-support")]
513    #[tokio::test]
514    async fn truncated_non_final_chunk_buffers_but_emits_zero() {
515        // Header + one row byte, no END_OF_PARTITION: the buffer ends mid-partition.
516        let data = synthetic_partition(&[STUB_ROW_BYTE]);
517        let (step, buffered, collected) = drive(&data, false).await;
518        assert_eq!(
519            step,
520            ParseStep::NeedMore,
521            "a mid-partition end-of-buffer on a non-final chunk must request more bytes"
522        );
523        assert_eq!(
524            buffered, 1,
525            "the row WAS buffered into the driver-owned pending vec"
526        );
527        assert!(
528            collected.is_empty(),
529            "NeedMore must DISCARD pending and forward zero rows so a re-parse cannot \
530             double-emit (issue #827)"
531        );
532    }
