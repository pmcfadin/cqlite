1//! Data access methods for SSTableReader.
2//!
3//! This module contains all methods related to reading data from SSTables,
4//! split by read path:
5//! - [`model`] — shared model types and free helpers (clustering slice,
6//!   table-id matching, token-order sorting, the `scan_for_key` counter).
7//! - [`bti`] — BTI ("da") trie-resolved point lookups, single-partition seeks,
8//!   clustering-slice narrowing, and the whole-Data.db BTI scan.
9//! - [`sequential`] — sequential / index-driven range and full scans, the
10//!   `scan_for_key` fallback, cell-metadata scan, and the streaming scan.
11//! - [`compaction`] — timestamp-preserving compaction iteration and the bounded
12//!   streaming compaction driver.
13//!
14//! `mod.rs` keeps the shared entry points and the I/O / parser infrastructure
15//! every read path builds on (`get`, `build_v5_parser`, `new_scan_cursor`,
16//! `read_next_block`, the offset-read helpers, the chunk-stitch helpers, and the
17//! delta-scan bridge), plus the public-surface re-exports.

150    /// Return `true` when Data.db uses the V5CompressedLegacy NB chunked format and
151    /// therefore requires all chunks to be stitched before parsing.
152    ///
153    /// The correct predicate is:
154    ///   data_format == V5CompressedLegacy  AND  is_nb_format()
155    ///
156    /// Rationale:
157    /// - `V5CompressedLegacy` identifies the row serialization format (u16 length
158    ///   prefixes, legacy encoding) used by all Cassandra 5 'nb' SSTables.
159    /// - `is_nb_format()` identifies the chunked-compression read path. It intentionally
160    ///   EXCLUDES `V5_0Uncompressed`, which uses the same row format but stores data as
161    ///   a single contiguous block (no chunk boundaries, no stitching needed).
162    /// - Using `is_compressed` (compression_reader.is_some()) would be wrong for NB
163    ///   format because the per-chunk decompression is handled inside `stitch_and_parse_all_chunks`,
164    ///   and `is_compressed` may differ from `is_nb_format` for edge-case versions.
165    pub(super) fn requires_chunk_stitching(&self) -> bool {
166        let data_format = self.header.cassandra_version.data_format();
167        matches!(data_format, DataFormat::V5CompressedLegacy)
168            && self.header.cassandra_version.is_nb_format()
169    }

171    /// Test-only: return the fully stitched + decompressed Data.db data section
172    /// for a V5CompressedLegacy ("nb") SSTable, or `None` for any other format.
173    ///
174    /// This exposes the exact on-disk bytes (post-decompression) that the
175    /// production scan path parses, so the Issue #1623 corpus-differential test
176    /// can locate REAL Cassandra `writeUnsignedVInt` length prefixes and route
177    /// those literal bytes through [`crate::parser::vint::parse_vint_length`].
178    /// It replicates the seek + stitch preamble of `sequential_scan` exactly.
179    #[cfg(test)]
180    pub(crate) async fn stitched_data_section_for_tests(&self) -> Result<Option<Vec<u8>>> {
181        if !self.requires_chunk_stitching() {
182            return Ok(None);
183        }
184        let cursor = self.new_scan_cursor().await?;
185        let header_size = self.calculate_header_size();
186        {
187            let mut file_guard = cursor.file.lock().await;
188            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
189        }
190        let stitched = self.stitch_all_chunks(&cursor).await?;
191        Ok(Some(stitched))
192    }

194    /// Get a value by key from the SSTable.
195    ///
196    /// Resolution-mode-agnostic entry point: callers that do not carry the
197    /// manager's `resolve_reader_list` signal (e.g. the per-reader helpers in
198    /// `partition_lookup`, `schema_aware_reader`, and benchmarks) get the STRICT
199    /// table-consistency guard — `fully_qualified_match = false` reproduces exactly
200    /// today's `table_ids_match_strict` behavior on the BTI point-lookup path, so
201    /// this is a behavior-preserving conservative default. The manager's `get()`
202    /// calls [`SSTableReader::get_with_resolution`] with the authoritative signal so
203    /// an exact fully-qualified match can accept rows across a benign header-keyspace
204    /// divergence (issue #1321, mirroring the seek path #1284).
205    pub async fn get(&self, table_id: &TableId, key: &RowKey) -> Result<Option<ScanRow>> {
206        self.get_with_resolution(table_id, key, false).await
207    }

209    /// Get a value by key, threading the authoritative resolution mode
210    /// (`fully_qualified_match`) into the BTI point-lookup guard (issue #1321).
211    ///
212    /// See [`SSTableReader::get`] for the resolution-mode contract. Only the BTI
213    /// ("da") point-lookup path consults `fully_qualified_match`; the bloom/Index.db/
214    /// sequential fallbacks are unaffected by it.
215    pub async fn get_with_resolution(
216        &self,
217        table_id: &TableId,
218        key: &RowKey,
219        fully_qualified_match: bool,
220    ) -> Result<Option<ScanRow>> {
221        // Issue #1576 (C5): O(1) authoritative range short-circuit. If the query key
222        // sorts outside this SSTable's [first_key, last_key] bound (Summary.db, in
223        // Cassandra token order — no heuristics), the partition is definitely absent;
224        // return absence BEFORE any bloom check, Index.db probe, or BTI trie descent.
225        // Inclusive bound (== first/last stays in range), so it never drops a present
226        // partition. A no-op when no authoritative bound exists (BTI/no Summary).
227        if self.partition_key_out_of_range(key.as_bytes()) {
228            crate::storage::sstable::read_work_counters::record_range_short_circuit();
229            return Ok(None);
230        }
231
232        // Issue #831 / #909: BTI ("da") readers resolve partitions via the
233        // Partitions.db trie (O(log n)), never via Index.db (absent for BTI) or
234        // the sequential scan. The trie is the AUTHORITATIVE presence oracle for a
235        // BTI SSTable — it answers present/absent definitively — so we branch here
236        // BEFORE the bloom-filter pre-check. Skipping the bloom filter for BTI is
237        // both correct (the trie is authoritative; bloom is only an optimization)
238        // and necessary: a writer-produced Filter.db whose hashing does not match
239        // the reader's would otherwise cause false negatives and drop live
240        // partitions (the writer→reader roundtrip #909 must read back). It also
241        // guarantees a BTI get() can never fall through to scan_for_key.
242        let (row, oracle_pruned) = if self.bti_partitions_db.is_some() {
243            self.bti_point_lookup(table_id, key, fully_qualified_match)
244                .await?
245        } else {
246            // BIG ("nb"/uncompressed) readers: raw-key Index.db resolve +
247            // covering-chunk seek (issue #1572). The bloom pre-check, the fast
248            // Index.db-resolved chunk-targeted decode, and the index-less
249            // `scan_for_key` fallback all live in `big_point`.
250            self.big_get_with_resolution(table_id, key, fully_qualified_match)
251                .await?
252        };
253
254        // Issue #2163 (roborev r4): `oracle_pruned` is `true` ONLY when the
255        // presence oracle itself (bloom-miss for BIG / trie-miss for BTI) excluded
256        // this SSTable from the read BEFORE any decode or scan — the PRIMARY
257        // single-reader point-read path, which the spec scenario "a partition
258        // point lookup ... through the public read surface" names directly. This
259        // is the SAME emit site `might_contain_partition[_encoded]` use (via
260        // `emit_sstable_pruned`), so a candidate pre-pruned by
261        // `SSTableManager::prune_candidates` (excluded from the candidate list, so
262        // `get()` is never called on it for this read) is never double-counted:
263        // exactly one of {prune-time check, this get-time check} runs per SSTable
264        // per logical read.
265        if oracle_pruned {
266            self.emit_sstable_pruned();
267
268            // Opt-in presence-oracle false-negative verification: when the
269            // default-off switch is enabled, an AUTHORITATIVE confirmation scan
270            // proves this exclusion truthful; a contradiction increments
271            // `cqlite.read.bloom.false_negatives`. Off by default → this whole
272            // block is skipped and the read costs nothing extra. Gated on
273            // `oracle_pruned` (not merely `row.is_none()`) so a `None` reached via
274            // the primary path's OWN authoritative `scan_for_key` — which already
275            // IS the confirming scan — never triggers a REDUNDANT second scan.
276            if super::presence_verification::enabled() {
277                if let Err(e) = self
278                    .verify_presence_oracle_negative(table_id, key.as_bytes())
279                    .await
280                {
281                    // Issue #2163 (roborev r5): the READ stays fail-open — a
282                    // verification-scan failure (e.g. `scan_for_key` erroring on
283                    // corruption or an unreadable SSTable) must NEVER fail (or
284                    // even affect) the actual read this opt-in check is merely
285                    // double-checking; `row` above is returned unchanged either
286                    // way. But a SILENT-MISS DETECTOR that itself fails silently
287                    // defeats its own purpose, so the failure is surfaced LOUDLY
288                    // instead of discarded: an error-level log with context, AND
289                    // a record through the EXISTING error-rate signal
290                    // (`cqlite.errors.total{category,subsystem}`, issue #1038) —
291                    // never a new metric. `record_error` maps `Error::Corruption`
292                    // (the typical `scan_for_key` failure mode) to the bounded
293                    // `Corruption` category.
294                    tracing::error!(
295                        error = %e,
296                        sstable_format = self.sstable_format_label(),
297                        "opt-in presence-oracle false-negative verification scan FAILED — the \
298                         read itself is unaffected (fail-open), but this soundness check could \
299                         not run for this SSTable and needs investigation"
300                    );
301                    crate::observability::record_error(&e, "reader");
302                }
303            }
304        }
305        Ok(row)
306    }

308    /// Stitch all compressed chunks and parse as a single buffer (V5CompressedLegacy)
309    ///
310    /// This helper method extracts the stitching logic from get_all_entries so it can be
311    /// reused by sequential_scan and other methods that need to handle V5CompressedLegacy
312    /// format where partitions can span chunk boundaries.
313    ///
314    /// `read_shadowing` (issue #1741): `true` for user-facing SELECT scans
315    /// (`scan_for_key`, `sequential_scan`), `false` for the physical
316    /// `get_all_entries` (integrity verification / data-manager) so it counts every
317    /// on-disk row.
318    pub(super) async fn stitch_and_parse_all_chunks(
319        &self,
320        cursor: &ScanCursor,
321        schema: Option<&crate::schema::TableSchema>,
322        read_shadowing: bool,
323        scan_cancel: &crate::storage::scan_cancel::ScanCancel,
324    ) -> Result<Vec<(TableId, RowKey, ScanRow)>> {
325        let stitched_buffer = self
326            .stitch_all_chunks_cancellable(cursor, scan_cancel)
327            .await?;
328        let parser = self.build_v5_parser(read_shadowing);
329
330        // Get schema (use provided schema or reader's schema)
331        let reader_schema;
332        let table_schema = if let Some(s) = schema {
333            Some(s)
334        } else {
335            reader_schema = self.get_table_schema(None);
336            reader_schema.as_ref()
337        };
338
339        // Parse the stitched decompressed buffer
340        let entries = parser.parse_block(&stitched_buffer, table_schema, self)?;
341        tracing::debug!(
342            "stitch_and_parse_all_chunks: Parsed {} entries from stitched buffer",
343            entries.len()
344        );
345
346        Ok(entries)
347    }

387    /// Read, decompress, and concatenate every compressed chunk of the data
388    /// section into a single buffer.
389    ///
390    /// V5CompressedLegacy partitions can span chunk boundaries, so the whole
391    /// data section must be stitched before parsing. The returned buffer is
392    /// bounded by the *uncompressed data-section size* — it scales with on-disk
393    /// bytes, not row count (issue #790).
394    ///
395    /// Precondition: the caller has seeked `cursor`'s file to the start of the
396    /// data section (the cursor's chunk index starts at 0 when freshly minted).
397    ///
398    /// Existing callers use the reader's own [`SSTableReader::scan_cancel`]
399    /// field (unchanged pre-#2346 behaviour); this is a thin wrapper over
400    /// [`Self::stitch_all_chunks_cancellable`], which the per-call seam
401    /// (`sequential_scan`) drives with its caller-supplied token instead.
402    pub(super) async fn stitch_all_chunks(&self, cursor: &ScanCursor) -> Result<Vec<u8>> {
403        self.stitch_all_chunks_cancellable(cursor, &self.scan_cancel)
404            .await
405    }

407    /// [`Self::stitch_all_chunks`] with an explicit PER-CALL cancel token
408    /// (issue #2346). `scan_cancel` is polled every 256 chunks so a cancelled
409    /// caller abandons the stitch — the chunk-read/decompress loop is the
410    /// I/O-bound phase of the stitched scan path, matching the compaction
411    /// stream's per-256-chunk cadence (`stream_all_partitions_for_compaction`).
412    pub(super) async fn stitch_all_chunks_cancellable(
413        &self,
414        cursor: &ScanCursor,
415        scan_cancel: &crate::storage::scan_cancel::ScanCancel,
416    ) -> Result<Vec<u8>> {
417        use crate::storage::sstable::compression::Compression;
418
419        // Pre-allocate buffer for ~2.5MB (estimated max size for test data)
420        let mut stitched_buffer = Vec::with_capacity(2_500_000);
421
422        // Incompressible-chunk fallback (Bug #639, epic #970): Cassandra stores a
423        // chunk RAW (not compressed) when its compressed length would meet or
424        // exceed `max_compressed_length`. `ChunkDecompressor::decompress_chunk`
425        // already honours this, but the stitch path did not — it blindly tried to
426        // LZ4/Snappy/etc-decode a raw chunk, which fails on the `incompressible`
427        // fixture. Mirror the writer rule here: when the (CRC-stripped) chunk
428        // length >= max_compressed_length, the bytes are already plaintext.
429        // Authority: CompressedSequentialWriter.java:160-177.
430        let max_compressed_length = self
431            .compression_info
432            .as_ref()
433            .map(|ci| ci.max_compressed_length as usize)
434            .unwrap_or(usize::MAX);
435
436        let mut chunk_count = 0;
437        while let Some(compressed_chunk) = self.read_next_block(cursor).await? {
438            // Cooperative cancellation (issue #2346): poll the per-call token every
439            // 256 chunks so a cancelled stitched scan abandons the I/O/decompress
440            // walk promptly instead of stitching the entire data section first.
441            if chunk_count & 0xFF == 0 {
442                scan_cancel.check()?;
443            }
444            let decompressed_chunk = if compressed_chunk.len() >= max_compressed_length {
445                // Stored uncompressed by Cassandra — pass the raw bytes through.
446                tracing::debug!(
447                    "stitch_all_chunks: chunk {} is incompressible (len={} >= max_compressed_length={}), using raw bytes",
448                    chunk_count,
449                    compressed_chunk.len(),
450                    max_compressed_length
451                );
452                compressed_chunk
453            } else if let Some(compression_reader) = &self.compression_reader {
454                // Single decode plane (issue #1598, G2): route the stitch-path
455                // decompress through ChunkSource so it is the ONLY query-path module
456                // that calls Compression::decompress. Behavior-identical (no cache,
457                // no counter) to the prior inline call.
458                let compression = Compression::new(*compression_reader.algorithm())?;
459                super::chunk_source::ChunkSource::decompress_only(
460                    Some(&compression),
461                    compressed_chunk,
462                )
463                .map_err(|e| {
464                    Error::corruption(format!(
465                        "stitch_all_chunks: Failed to decompress chunk {}: {}",
466                        chunk_count, e
467                    ))
468                })?
469            } else {
470                // No compression (should not happen for V5CompressedLegacy)
471                tracing::warn!("stitch_all_chunks: No compression reader, using raw chunk data");
472                compressed_chunk
473            };
474
475            stitched_buffer.extend_from_slice(&decompressed_chunk);
476            chunk_count += 1;
477        }
478
479        tracing::debug!(
480            "stitch_all_chunks: Stitched {} chunks, total buffer: {} bytes",
481            chunk_count,
482            stitched_buffer.len()
483        );
484
485        Ok(stitched_buffer)
486    }

488    /// Build a [`V5CompressedLegacyParser`] configured from this reader's header,
489    /// statistics (EncodingStats), version gates, and UDT registry.
490    ///
491    /// `read_shadowing` (issue #1741) MUST be `true` for user-facing query reads
492    /// (`scan`/`scan_stream`/`scan_with_cell_metadata`/point `get`) so the emit path
493    /// applies SELECT-semantic partition/range-tombstone shadowing and TTL expiry, and
494    /// `false` for PHYSICAL consumers that must see every on-disk row — integrity
495    /// verification via `get_all_entries`, `sstable_data_manager`, delta-scan, and the
496    /// compaction read path (which reconciles tombstones itself across generations).
497    ///
498    /// [`V5CompressedLegacyParser`]: crate::storage::sstable::reader::parsing::V5CompressedLegacyParser
499    pub(super) fn build_v5_parser(
500        &self,
501        read_shadowing: bool,
502    ) -> crate::storage::sstable::reader::parsing::V5CompressedLegacyParser {
503        let keyspace = self.header.keyspace.clone();
504        let table_name = self.header.table_name.clone();
505
506        // Extract EncodingStats from statistics_reader (if available)
507        let (min_timestamp, min_local_deletion_time, min_ttl) =
508            if let Some(stats_reader) = &self.statistics_reader {
509                let ts_stats = &stats_reader.statistics().timestamp_stats;
510                (
511                    ts_stats.min_timestamp,
512                    ts_stats.min_deletion_time,
513                    ts_stats.min_ttl,
514                )
515            } else {
516                (0, 0, None)
517            };
518
519        let parser = crate::storage::sstable::reader::parsing::V5CompressedLegacyParser::new(
520            keyspace,
521            table_name,
522            min_timestamp,
523            min_local_deletion_time,
524            min_ttl,
525        )
526        // VG1: thread VersionGates from SSTableReader down to row parser so
527        // that VG3 can flip gate-sensitive code paths without re-deriving gates.
528        .with_version_gates(self.version_gates.clone())
529        // Issue #1741: SELECT-semantic read shadowing (see fn docs).
530        .with_read_shadowing(read_shadowing);

532        if let Some(ref registry) = self.udt_registry {

534        } else {

536        }

539    /// Read value at a specific offset with caching
540    pub async fn read_value_at_offset(&self, offset: u64, size: u32) -> Result<Option<ScanRow>> {
541        // Size must be non-zero for offset-based reading
542        if size == 0 {
543            return Err(Error::corruption(format!(
544                "Cannot read value at offset {} with size=0. This should have been caught earlier and handled via sequential scan.",
545                offset
546            )));
547        }
548
549        // Read-time CRC verification for uncompressed BIG SSTables (issue #1396).
550        // The index-based scan and point-lookup paths reach Data.db here (bypassing
551        // read_next_block / read_uncompressed_data_block), so verify the CRC.db
552        // chunk(s) covering [offset, offset+size) BEFORE returning any bytes. A
553        // mismatch is a typed Error::Corruption naming the chunk + offset (never
554        // wrong values / never a silent result). No-op when no CRC.db is present
555        // (compressed tables / BTI / absent-CRC.db warn-and-proceed).
556        self.verify_uncompressed_range(offset, size).await?;
557
558        // Read + decompress in ONE decode plane (issue #1598, G2). `get_cached_data`
559        // returns the final DECODED window: for a compressed Data.db it routes through
560        // the CRC-validated `read_compressed_offset_window` (issue #1773 — per-chunk
561        // inline CRC32 checked BEFORE decompression, whose sole `Compression::decompress`
562        // now resolves inside `ChunkSource`), or the CRC.db-verified raw bytes for an
563        // uncompressed one. There is no second decode here — doing so used to
564        // double-decompress the compressed path (and skip its inline CRC, the #1411 bug).
565        let data = self.get_cached_data(offset, size).await?;
566
567        // Preserve raw data until schema is available. Pre-#1334 this offset-read
568        // placeholder returned a bare `Value::Blob` of the row's raw value bytes,
569        // which the schema-aware decode path then schema-decoded and the no-schema query
570        // layer surfaced as a synthetic single-column "data" row. Carry that RAW
571        // provenance explicitly (issue #1334): a schema-aware consumer decodes the
572        // bytes; a no-schema consumer surfaces a single "data" blob — with no
573        // downstream guessing from the value's shape.
574        let row = ScanRow::RawRow(data.to_vec());
575
576        // Extract write time from value (placeholder - would need to be parsed from SSTable)
577        let _write_time = std::time::SystemTime::now()
578            .duration_since(std::time::UNIX_EPOCH)
579            .map(|d| d.as_micros() as i64)
580            .unwrap_or_else(|e| {
581                warn!("Failed to get system time: {}; using fallback value 0", e);
582                0
583            });
584
585        // Filter out tombstones and expired data
586        if !self.filter_tombstone(&row) {
587            return Ok(None);
588        }
589
590        Ok(Some(row))
591    }

593    /// Read block with caching support and hit/miss tracking.
594    ///
595    /// Wired to the shared [`DecompressedChunkCache`] (issue #1567): the cache is
596    /// consulted BEFORE the file read, so a repeat read of a resident region is a
597    /// refcount-bump hit that touches the file zero times and re-decompresses
598    /// nothing. Keyed by the index-resolved `block_offset` in the BIG-point-read
599    /// namespace ([`NS_BIG_POINT`]), which is disjoint from the chunk-index
600    /// namespaces used by the windowed-scan and BTI sites. The returned buffer is
601    /// already CRC-verified: [`read_value_at_offset`](Self::read_value_at_offset)
602    /// runs `verify_uncompressed_range` before calling this, so a hit returns
603    /// bytes verified when they were first inserted.
604    ///
605    /// [`DecompressedChunkCache`]: crate::storage::cache::DecompressedChunkCache
606    async fn get_cached_data(&self, block_offset: u64, size: u32) -> Result<Vec<u8>> {
607        // The shared B1 cache tracks its own hit/miss counters (issue #1567). The
608        // ranged BIG-point key carries `size` as its aux discriminant so two reads at
609        // the same offset with different sizes can never alias (roborev #1567).
610        let key = crate::storage::cache::ChunkKey::with_aux(
611            self.chunk_cache_id ^ NS_BIG_POINT,
612            block_offset,
613            size as u64,
614        );
615        if let Some(hit) = self.chunk_cache.get(&key) {
616            return Ok(hit.to_vec());
617        }
618
619        // Produce the final DECOMPRESSED, integrity-verified window for this offset.
620        let data = if let Some(comp_info) = self.compression_info.as_deref() {
621            // COMPRESSED offset read (issue #1773): the authoritative inline per-chunk
622            // CRC32 MUST be validated before decompression. `read_compressed_offset_window`
623            // reuses the shared CRC-enforcing chunk reader (multi-chunk assembly,
624            // fail-closed past-EOF) rather than reading `size` raw bytes and blindly
625            // LZ4-decoding them — the latter re-introduced the exact #1411 CRC bypass.
626            // Its single decompress resolves inside `ChunkSource` (issue #1598, G2), so
627            // this is NOT a second decode plane.
628            self.read_compressed_offset_window(comp_info, block_offset, size)
629                .await?
630        } else {
631            // UNCOMPRESSED offset read. Positioned read on the shared point source
632            // (issue #1573, C2): no cursor mutex is held across this I/O, so
633            // concurrent point reads do not convoy. The covering CRC.db chunks were
634            // already verified by `verify_uncompressed_range` before we got here
635            // (issue #1396), so these bytes are integrity-checked too.
636            model::CHUNK_READ_CALLS.fetch_add(1, Ordering::Relaxed);
637            let mut buffer = vec![0u8; size as usize];
638            self.point_source.read_exact_at(block_offset, &mut buffer)?;
639            buffer
640        };
641
642        // Insert into the shared cache (converts the Vec to Arc<[u8]> once) and
643        // return an owned copy (this site's callers consume a Vec).
644        let arc = self.chunk_cache.insert(key, data);
645        Ok(arc.to_vec())
646    }

648    /// Verify the `CRC.db` chunk(s) covering the uncompressed Data.db byte range
649    /// `[offset, offset + size)` against their stored per-chunk CRC32 (issue
650    /// #1396), on the offset-read path used by the index-based scan and point
651    /// lookups.
652    ///
653    /// A partition read touches a sub-range of one or more `chunk_size` blocks; a
654    /// chunk's CRC can only be checked over the WHOLE chunk, so this reads each
655    /// covering chunk (bounded to one `chunk_size` block at a time) and compares.
656    /// Each chunk is verified at most once per reader lifetime (memoized in
657    /// [`SSTableReader::verified_uncompressed_chunks`]), keeping the cost at the
658    /// budgeted one CRC32 pass per chunk even when many partitions share a chunk.
659    ///
660    /// No-op when this reader has no `CRC.db` (compressed tables carry inline
661    /// per-chunk CRCs; BTI ships none; an absent `CRC.db` is warn-and-proceed).
662    async fn verify_uncompressed_range(&self, offset: u64, size: u32) -> Result<()> {
663        let Some(crc) = self.crc_reader.as_deref() else {
664            return Ok(());
665        };
666        if size == 0 {
667            return Ok(());
668        }
669        let file_size = self.stats.file_size;
670        // Fix 3 (issue #1396): a corrupt on-disk offset/size can overflow
671        // `offset + size` (debug panic / wrapped range that misattributes CRC
672        // chunks) or point past EOF. Use checked arithmetic and reject a range
673        // that overflows or exceeds the Data.db length as typed corruption
674        // BEFORE deriving any chunk index. `size >= 1` here (0 handled above),
675        // so `end > offset` and `end - 1` never underflows.
676        let end = offset
677            .checked_add(size as u64)
678            .filter(|end| *end <= file_size)
679            .ok_or_else(|| {
680                Error::corruption(format!(
681                    "uncompressed read range [0x{offset:x}, +{size}) overflows or exceeds the \
682                     Data.db length {file_size}; refusing to verify a corrupt offset"
683                ))
684            })?;
685        // Each covering chunk is CRC'd over its on-disk bytes via a positioned
686        // read on the shared point source (issue #1573, C2): the verifier never
687        // holds a cursor mutex across I/O, so it neither convoys concurrent point
688        // reads nor disturbs any scan cursor's position.
689        self.verify_covering_chunks(crc, offset, end, |lo, hi, chunk| {
690            let mut buf = vec![0u8; (hi - lo) as usize];
691            self.point_source.read_exact_at(lo, &mut buf).map_err(|e| {
692                Error::corruption(format!(
693                    "failed to read uncompressed chunk {chunk} at Data.db offset 0x{lo:x} for CRC verification: {e}"
694                ))
695            })?;
696            Ok(crc32fast::hash(&buf))
697        })
698    }

700    /// In-buffer counterpart of [`Self::verify_uncompressed_range`] for the
701    /// whole-section point-read fallback (issue #1573): the section is ALREADY
702    /// resident in `section`, covering Data.db `[base, base + section.len())`, so
703    /// each fully-contained covering chunk is CRC-checked directly from memory —
704    /// the section is transferred from disk EXACTLY ONCE (the caller's windowed
705    /// read), not a second time for CRC. A chunk that straddles below `base` (only
706    /// possible when `base` is not `CRC.db`-chunk-aligned, e.g. a nonzero header
707    /// that does not fall on a chunk boundary) re-reads just that one chunk so the
708    /// CRC-before-use guarantee holds regardless of header alignment. Same CRC32
709    /// algorithm, chunk layout, memoization, and mismatch/typed-error semantics as
710    /// [`Self::verify_uncompressed_range`].
711    async fn verify_uncompressed_section_in_buffer(&self, base: u64, section: &[u8]) -> Result<()> {
712        let Some(crc) = self.crc_reader.as_deref() else {
713            return Ok(());
714        };
715        if section.is_empty() {
716            return Ok(());
717        }
718        let file_size = self.stats.file_size;
719        let end = base
720            .checked_add(section.len() as u64)
721            .filter(|end| *end <= file_size)
722            .ok_or_else(|| {
723                Error::corruption(format!(
724                    "uncompressed section [0x{base:x}, +{}) overflows or exceeds the Data.db \
725                     length {file_size}; refusing to verify a corrupt section",
726                    section.len()
727                ))
728            })?;
729        self.verify_covering_chunks(crc, base, end, |lo, hi, chunk| {
730            if lo >= base && hi <= end {
731                // Chunk fully resident in the already-read buffer — CRC from
732                // memory, no second I/O.
733                let s = (lo - base) as usize;
734                let e = (hi - base) as usize;
735                Ok(crc32fast::hash(&section[s..e]))
736            } else {
737                // Chunk extends below `base` (unaligned header) — read just it.
738                let mut buf = vec![0u8; (hi - lo) as usize];
739                self.point_source.read_exact_at(lo, &mut buf).map_err(|e| {
740                    Error::corruption(format!(
741                        "failed to read uncompressed chunk {chunk} at Data.db offset 0x{lo:x} for CRC verification: {e}"
742                    ))
743                })?;
744                Ok(crc32fast::hash(&buf))
745            }
746        })
747    }

749    /// Walk the `CRC.db` chunk(s) covering Data.db `[start, end)` (`end` already
750    /// bounds-checked `<= file_size` by the caller, `end > start`), skipping chunks
751    /// already verified for this reader, comparing each covering chunk's CRC32 to
752    /// `CRC.db`, and memoizing successes so a chunk is verified at most once per
753    /// reader lifetime. `compute(lo, hi, chunk)` yields the CRC32 of the chunk's
754    /// on-disk bytes `[lo, hi)`; callers supply either a positioned disk read or an
755    /// in-buffer slice, which is what lets the whole-section fallback read its bytes
756    /// exactly once. This is the single place the chunk geometry, mismatch error,
757    /// and memoization live — the two public verify entry points differ ONLY in how
758    /// they source each chunk's bytes. Module-visible so the windowed scan's
759    /// synchronous uncompressed piece reader shares the same memoized path (#1940).
760    pub(in crate::storage::sstable::reader) fn verify_covering_chunks(
761        &self,
762        crc: &super::crc::CrcDb,
763        start: u64,
764        end: u64,
765        mut compute: impl FnMut(u64, u64, u64) -> Result<u32>,
766    ) -> Result<()> {
767        let cs = crc.chunk_size() as u64;
768        if cs == 0 {
769            return Err(Error::corruption(
770                "CRC.db chunk size is zero; cannot verify uncompressed read",
771            ));
772        }
773        let file_size = self.stats.file_size;
774        let first = start / cs;
775        let last = (end - 1) / cs;
776        for chunk in first..=last {
777            // Skip chunks already verified for this reader.
778            {
779                let seen = self
780                    .verified_uncompressed_chunks
781                    .lock()
782                    .unwrap_or_else(|e| e.into_inner());
783                if seen.contains(&chunk) {
784                    continue;
785                }
786            }
787            let lo = chunk * cs;
788            let hi = ((chunk + 1) * cs).min(file_size);
789            if hi <= lo {
790                break; // range extends past EOF; nothing real to verify
791            }
792            let computed = compute(lo, hi, chunk)?;
793            let expected = crc.crc_for_chunk(chunk as usize)?;
794            if computed != expected {
795                return Err(Error::corruption(format!(
796                    "uncompressed CRC32 mismatch for chunk {} at Data.db offset 0x{:x} \
797                     ({} bytes): expected=0x{:08x}, computed=0x{:08x} (CRC.db)",
798                    chunk,
799                    lo,
800                    hi - lo,
801                    expected,
802                    computed
803                )));
804            }
805            self.verified_uncompressed_chunks
806                .lock()
807                .unwrap_or_else(|e| e.into_inner())
808                .insert(chunk);
809        }
810        Ok(())
811    }

813    /// The SINGLE CRC-checked chokepoint for reading a raw byte range from an
814    /// *uncompressed* Data.db section (issue #1396).
815    ///
816    /// EVERY uncompressed offset read MUST flow through here so that no public
817    /// read path can silently bypass `CRC.db` verification and hand corrupt
818    /// bytes to the parser. It runs the shared, memoized verifier
819    /// [`Self::verify_uncompressed_range`] over the covering chunk(s) BEFORE any
820    /// bytes are returned; a mismatch is the typed [`Error::Corruption`] naming
821    /// the failing chunk + Data.db offset. The verifier is a no-op when this
822    /// reader has no `CRC.db` (compressed tables carry inline per-chunk CRCs,
823    /// BTI ships none, an absent `CRC.db` is warn-and-proceed), so this helper is
824    /// also the correct accessor for the raw-read step of compressed offset reads.
825    ///
826    /// `offset` is an ABSOLUTE Data.db file offset (post-header). `file` is the
827    /// handle to read the range from — the shared point-read handle
828    /// ([`Self::file`]) or a scan-local cursor's private handle (issue #815).
829    /// CRC verification always uses the reader's own handle, so it is independent
830    /// of `file` and never disturbs a scan cursor's position.
831    ///
832    /// Future uncompressed offset reads MUST call this instead of doing their own
833    /// `seek` + `read_exact`, so the CRC check can never be forgotten again.
834    pub(in crate::storage::sstable::reader) async fn read_uncompressed_verified(
835        &self,
836        file: &tokio::sync::Mutex<super::source::BlockSource>,
837        offset: u64,
838        len: usize,
839    ) -> Result<Vec<u8>> {
840        use tokio::io::AsyncReadExt;
841
842        let size = u32::try_from(len).map_err(|_| {
843            Error::corruption(format!(
844                "uncompressed read length {len} exceeds u32 range for CRC verification \
845                 at Data.db offset 0x{offset:x}"
846            ))
847        })?;
848        // Verify the covering CRC.db chunk(s) BEFORE returning any bytes.
849        self.verify_uncompressed_range(offset, size).await?;
850
851        let mut buf = vec![0u8; len];
852        {
853            let mut guard = file.lock().await;
854            guard.seek(SeekFrom::Start(offset)).await?;
855            guard.read_exact(&mut buf).await?;
856        }
857        Ok(buf)
858    }

912    /// Prepare for a delta-scan pass: stitch all compressed chunks of the data
913    /// section and return the decompressed buffer together with a pre-configured
914    /// parser.
915    ///
916    /// Uses its own per-scan cursor (issue #815), so it no longer needs the
917    /// caller to serialize against concurrent reads. This method is gated on the
918    /// `delta-scan` feature and is the only bridge between the SSTableReader
919    /// internals and the `delta_scan` module, which cannot access private
920    /// helpers directly.
921    ///
922    /// The `schema` parameter is not used here — it is threaded through the
923    /// caller's `parse_block_emit_delta` invocation instead.  The parser is
924    /// built via `build_v5_parser()` which handles version-gates and UDT
925    /// registry without needing the schema at construction time.
926    #[cfg(feature = "delta-scan")]
927    pub async fn prepare_delta_scan(
928        &self,
929    ) -> Result<(Vec<u8>, super::parsing::V5CompressedLegacyParser)> {
930        use tokio::io::AsyncSeekExt;
931
932        // Seek the per-scan cursor to the start of the data section.
933        let cursor = self.new_scan_cursor().await?;
934        let header_size = self.calculate_header_size();
935        {
936            let mut file_guard = cursor.file.lock().await;
937            file_guard
938                .seek(std::io::SeekFrom::Start(header_size as u64))
939                .await?;
940        }
941
942        // Stitch all compressed chunks (bounded by uncompressed data-section size).
943        let stitched = self.stitch_all_chunks(&cursor).await?;
944
945        // Build a parser (re-using the existing builder so version-gates and
946        // UDT registry are threaded through correctly).
947        let parser = self.build_v5_parser(false);
948
949        Ok((stitched, parser))
950    }
