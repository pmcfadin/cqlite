1//! BTI ("da") read paths: trie-resolved point lookups, single-partition seeks,
2//! clustering-slice narrowing, and the whole-Data.db BTI scan.
3//!
4//! BTI SSTables carry no Index.db/Summary.db. The Partitions.db trie is the
5//! authoritative present/absent oracle; every method here resolves an
6//! uncompressed Data.db offset via the trie and decodes only the chunk window
7//! that holds the target partition (issue #831 / #909 / #953 / #954).

39    /// Single-partition *seek* for the partition-targeted lookup fast path,
40    /// clustering-slice-aware (Issue #953 + #954, Epic #951).
41    ///
42    /// Where [`scan`](Self::scan) decodes EVERY partition in this SSTable and the
43    /// caller retains one, this resolves the target partition's `Data.db` offset
44    /// from the authoritative index (the BTI Partitions.db trie or the BIG
45    /// `Index.db`) and decodes ONLY that partition — the same per-partition decode
46    /// `scan` runs (`parse_block_emit` over the chunk-targeted decompressed
47    /// window), so its output is byte-for-byte identical to filtering the full
48    /// `scan` result down to `partition_key`.
49    ///
50    /// Offset domains (no-heuristics: authoritative resolved offsets only):
51    /// - **BTI ("da")** — `lookup_partition_via_bti_trie` returns the UNCOMPRESSED
52    ///   `Data.db` offset; a trie miss is authoritative absence.
53    /// - **BIG (`nb`)** — `lookup_partition_with_index` returns the partition's
54    ///   offset into the (uncompressed) data section. A hit is authoritative
55    ///   present; a MISS returns `Ok(None)` (the `Index.db` may be digest-keyed or
56    ///   incomplete, exactly the `get()` fallback rationale at #517) so the caller
57    ///   re-checks via a full scan rather than risk a false negative.
58    ///
59    /// Prefix-collision / wrong-offset guard: the decode
60    /// (`bti_decompress_and_parse_target_all`) re-verifies the decoded partition
61    /// key equals `partition_key` before collecting any row, so a BTI
62    /// prefix-collision candidate or a stale/mismatched index offset decodes to
63    /// nothing and is reported as absent — never a wrong partition. Every
64    /// clustering row of the matched partition is collected (not just the first),
65    /// so a multi-row partition returns all rows.
66    ///
67    /// Compiled only for the default (`not(tombstones)`) build: the manager's
68    /// seek-driven `scan_partition` exists only there, so under `tombstones` this
69    /// would be dead code.
70    ///
71    /// When `clustering` is `Some(slice)` AND this reader is BTI (`da`) with a
72    /// per-partition row index (`Rows.db`), the target partition's authoritative
73    /// row index is consulted to resolve the byte extent of the row-index
74    /// block(s) covering the requested clustering range, and ONLY that byte window
75    /// is decoded — so a `WHERE pk = ? AND ck </>/= ?` slice over a wide partition
76    /// decodes O(matched rows + index block slack) rather than the whole
77    /// partition. The post-scan `evaluate_leaf` backstop trims the
78    /// block-granularity over-read, so the returned rows are a superset of the
79    /// exact slice and the final query output is byte-identical to the
80    /// full-partition decode + post-filter.
81    ///
82    /// Returns `Ok(Some((rows, clustering_seek_engaged)))`:
83    /// - `clustering_seek_engaged == true` only when the clustering row-index
84    ///   narrowing actually bounded the decode (BTI wide partition with a usable
85    ///   row index and an encodable bound). The caller reports
86    ///   [`AccessPath::ClusteringSlice`](crate::query::access_path::AccessPath::ClusteringSlice)
87    ///   in that case.
88    /// - `clustering_seek_engaged == false` when the partition was decoded in full
89    ///   (no clustering slice, a NARROW BTI partition with no row index, the BIG
90    ///   format, or an un-encodable bound). Results are still correct — the caller
91    ///   reports the honest `PartitionLookup` path, NOT a fake clustering slice.
92    ///
93    /// `Ok(None)` mirrors [`scan_single_partition`]: the seek is not applicable
94    /// (no authoritative offset) and the caller must fall back to a full scan +
95    /// retain.
96    ///
97    /// [`scan_single_partition`]: crate::storage::sstable::SSTableReader
98    #[cfg(not(feature = "tombstones"))]
99    pub(crate) async fn scan_single_partition_clustering(
100        &self,
101        table_id: &TableId,
102        partition_key: &[u8],
103        clustering: Option<&ClusteringSlice>,
104        // `true` iff the manager resolved this reader by an EXACT fully-qualified
105        // `keyspace.table` match (or the query is unqualified). `false` means a
106        // fully-qualified query reached this reader via the bare-name fallback, in
107        // which case the seek guard keeps STRICT keyspace matching so it can never
108        // return rows from a different keyspace whose table name collides (#1284).
109        fully_qualified_match: bool,
110        schema: Option<&crate::schema::TableSchema>,
111    ) -> Result<Option<(Vec<(RowKey, ScanRow)>, bool)>> {
112        // 1. Resolve the partition's uncompressed Data.db offset, and record
113        //    whether THIS path's "decoded nothing" is authoritative absence (BTI
114        //    trie) or merely inconclusive (BIG Index.db).
115        let is_bti = self.bti_partitions_db.is_some();
116        // Resolve the target partition's UNCOMPRESSED `Data.db` start offset.
117        let offset = if is_bti {
118            match self.lookup_partition_via_bti_trie(partition_key)? {
119                // Trie hit: candidate uncompressed offset (re-verified on decode).
120                Some(off) => off,
121                // Trie miss is AUTHORITATIVE absence for BTI (no rows, no seek).
122                None => return Ok(Some((Vec::new(), false))),
123            }
124        } else {
125            match self.lookup_partition_with_index(partition_key).await? {
126                // Index.db hit: a candidate offset into the data section. (The
127                // `data_size` is 0 in writer-produced Index.db, so we do NOT use it
128                // as a bound — the successor offset below is the authoritative end.)
129                Some((off, _size)) => off,
130                // No Index.db hit: cannot seek authoritatively (the index may be
131                // digest-keyed / incomplete, exactly the get() #517 rationale).
132                // Fall back to a full scan.
133                None => return Ok(None),
134            }
135        };
136
137        // AUTHORITATIVE end bound (issue #953 / #951 MEDIUM): the target partition
138        // occupies `[offset, end)`, where `end` is the SUCCESSOR partition's start
139        // offset (next trie/index entry). Decompressing exactly the chunks covering
140        // that half-open range materializes every byte of the target partition —
141        // including a row/cell that SPANS multiple compression chunks — without
142        // reading the next partition. This replaces the previous next-partition
143        // *boundary-scan* heuristic (a row-count-stability guard that could falsely
144        // accept a boundary mid-partition); see `bti_decompress_and_parse_target_all`.
145        //
146        // `None` means `offset` is the LAST partition (no successor): the callee
147        // bounds the end with the authoritative data-section length, or falls back
148        // to the safe full-scan path when that length is unknown.
149        let end_bound = self.successor_partition_offset(offset)?.map(|e| e as usize);
150
151        let schema_opt = self.get_table_schema(schema);
152
153        // Issue #954 / #1184: resolve the within-partition row-body byte window for a
154        // single-column clustering slice from the authoritative index (BTI `Rows.db`
155        // trie or BIG promoted `IndexInfo` blocks). The unified resolver lives in
156        // `big_promoted.rs` (campsite: keeps this over-threshold file from growing).
157        let (row_body_window, decode_end_bound, clustering_engaged) = self
158            .resolve_clustering_seek_window(
159                is_bti,
160                partition_key,
161                offset,
162                clustering,
163                schema_opt.as_ref(),
164                end_bound,
165            )?;
166
167        // Issue #1184: an engaged BIG clustering narrowing decodes the selected block
168        // window via `big_promoted.rs` (partition-key-bytes guard, not the BTI strict
169        // table-id match that rejects writer-header SSTables). BTI keeps its decoder.
170        if !is_bti && clustering_engaged {
171            if let Some(rows) = self
172                .big_decode_clustering_window(
173                    partition_key,
174                    offset,
175                    decode_end_bound,
176                    row_body_window,
177                    schema_opt.as_ref(),
178                )
179                .await?
180            {
181                if !rows.is_empty() {
182                    super::super::super::work_counters::add_partition_decoded();
183                }
184                return Ok(Some((rows, true)));
185            }
186        }
187
188        // 2. Decode ONLY the target partition at the resolved offset, using the
189        //    SAME parser the scan path uses. `bti_decompress_and_parse_target_all`
190        //    chunk-targets the decompression (decodes just the chunk window that
191        //    holds the partition) and re-verifies the decoded key, so this is
192        //    O(1) PARTITIONS decoded regardless of the SSTable's partition count.
193        //
194        //    Issue #953 correctness fix: this collects EVERY clustering row of the
195        //    one target partition (bounded by the authoritative successor offset /
196        //    data-section length), not just the first row, so a `WHERE pk = ?`
197        //    over a multi-clustering-row
198        //    partition returns all rows — byte-identical to filtering the full
199        //    scan down to `partition_key`. The single-row `*_target` decoder is
200        //    still used by the `get()` point-lookup path, which returns one Value.
201        //
202        //    Issue #954: when `row_body_window` is set, the parse is bounded to the
203        //    clustering slice's row-index block extent so only O(slice) rows are
204        //    decoded (the post-scan backstop trims the block-granularity slack).
205        let parser = self.build_v5_parser(true);
206        let key = RowKey::from(partition_key.to_vec());
207        let decoded_rows = match self
208            .bti_decompress_and_parse_target_all(
209                offset as usize,
210                decode_end_bound,
211                row_body_window,
212                &key,
213                table_id,
214                fully_qualified_match,
215                schema_opt.as_ref(),
216                &parser,
217            )
218            .await?
219        {
220            // Authoritatively bounded decode (rows may be empty for an absent key).
221            Some(rows) => rows,
222            // The seek could not bound the target partition authoritatively (the
223            // LAST partition with an unknown data-section length): fall back to the
224            // safe full scan + retain for correctness, per the #953 mandate.
225            None => return Ok(None),
226        };
227
228        // 3. Record the per-partition decode (Issue #953 / #958): exactly ONE
229        //    partition is decoded for a hit regardless of how many clustering rows
230        //    it yields — `partitions_decoded` counts partitions, not rows. This is
231        //    the signal that proves the within-SSTable seek (vs a full
232        //    parse-then-retain). A non-empty decode means the partition exists.
233        if !decoded_rows.is_empty() {
234            super::super::super::work_counters::add_partition_decoded();
235            // Tombstone suppression matches the user-facing scan path
236            // (`sequential_scan`/`bti_scan_with_metadata` both apply it),
237            // applied per-row so a row tombstone is dropped while live rows in
238            // the same partition survive.
239            let rows: Vec<(RowKey, ScanRow)> = decoded_rows
240                .into_iter()
241                .filter(|value| self.filter_tombstone(value))
242                .map(|value| (key.clone(), value))
243                .collect();
244            return Ok(Some((rows, clustering_engaged)));
245        }
246
247        // Decoded nothing at the resolved offset. Whether that is authoritative
248        // depends on HOW the offset was resolved (Constraint #4: never return a
249        // wrong/empty result from an unsupported/inconclusive seek):
250        //
251        // - **BTI** — the trie is the authoritative present/absent oracle and the
252        //   decode re-verified the key, so "decoded nothing" means the trie
253        //   candidate was a prefix-collision for an absent key. AUTHORITATIVE
254        //   empty: the caller does NOT fall back.
255        // - **BIG** — the `Index.db` offset is only a candidate position; its
256        //   promoted-index / chunk layout is not as load-bearing as the BTI trie,
257        //   so a failed decode at the resolved offset is INCONCLUSIVE (a partition
258        //   that straddles a chunk boundary, or a stale offset, can fail the
259        //   chunk-targeted decode yet be found by a full parse). Fall back to a
260        //   full scan rather than risk a false negative.
261        if is_bti {
262            // Authoritative absence (prefix-collision candidate for an absent key).
263            // No rows decoded, so report the clustering seek as NOT engaged.
264            Ok(Some((Vec::new(), false)))
265        } else {
266            Ok(None)
267        }
268    }

270    /// Resolve the within-partition row-body byte window covering a single-column
271    /// clustering slice, using the target partition's authoritative BTI row index
272    /// (Issue #954, Epic #951).
273    ///
274    /// For a WIDE BTI partition the `Partitions.db` trie points at a per-partition
275    /// `TrieIndexEntry` in `Rows.db`; that entry's row-index trie maps clustering
276    /// **separators** to row-index BLOCK offsets (relative to the partition
277    /// start). [`select_row_index_blocks_for_range`] applies the authoritative
278    /// separator-floor semantics to pick exactly the blocks whose key interval
279    /// intersects `[start, end]`, so the returned byte window is the smallest
280    /// authoritative extent that can contain the requested clustering range.
281    ///
282    /// Returns `Ok(Some(window))` only when the narrowing is authoritative and
283    /// useful:
284    /// - the reader is BTI with a `Rows.db`,
285    /// - the partition is WIDE (`Partitions.db` returned a `RowsOffset`, i.e. the
286    ///   partition has a row index — a NARROW partition has no per-partition row
287    ///   index to seek within),
288    /// - the clustering bound(s) encode to the OSS50 byte-comparable form, and
289    /// - the selected block set is non-empty.
290    ///
291    /// Returns `Ok(None)` (decode the whole partition, report `PartitionLookup`)
292    /// for every other case — a NARROW partition, an empty `Rows.db`, an
293    /// un-encodable bound, or a slice that selects no block. This is the honest
294    /// fallback: correctness is preserved by decoding the full partition and
295    /// letting the post-scan backstop filter.
296    #[cfg(not(feature = "tombstones"))]
297    pub(super) fn bti_clustering_row_window(
298        &self,
299        partition_key: &[u8],
300        slice: &ClusteringSlice,
301        schema: Option<&crate::schema::TableSchema>,
302    ) -> Result<Option<ClusteringRowWindow>> {
303        use crate::storage::sstable::bti::{
304            encode_partition_key_for_bti_trie, lookup_partition_in_bti_slice,
305            resolve_rows_db_entry, rows_floor_block, rows_strict_ceiling_block,
306            BtiPartitionLocation,
307        };
308
309        let (Some(partitions_db), Some(rows_db)) = (&self.bti_partitions_db, &self.bti_rows_db)
310        else {
311            return Ok(None);
312        };
313
314        // Resolve the partition's location. Only a WIDE partition (RowsOffset) has
315        // a per-partition row index we can seek within; a NARROW partition
316        // (DataOffset) has none, so decode it in full.
317        // Issue #1574 (C3): walk the resident trie buffer in place (no whole-file copy).
318        // Issue #1575 (C4): encode the raw key then walk with the pre-encoded key via
319        // the single `lookup_partition_in_bti_slice` primitive every BTI lookup shares.
320        let encoded = encode_partition_key_for_bti_trie(partition_key);
321        let rows_offset = match lookup_partition_in_bti_slice(partitions_db.as_slice(), &encoded)
322            .map_err(|e| {
323                Error::corruption(format!(
324                    "BTI clustering seek: Partitions.db trie lookup failed (key len={}): {}",
325                    partition_key.len(),
326                    e
327                ))
328            })? {
329            Some(BtiPartitionLocation::RowsOffset(off)) => off as usize,
330            // NARROW partition or absent key: no row index to narrow with.
331            Some(BtiPartitionLocation::DataOffset(_)) | None => return Ok(None),
332        };
333
334        // Resolve the per-partition row-index entry ONCE (issue #1647 / L1: the
335        // pre-L1 path resolved it HERE and again inside `iterate_rows_for_partition`,
336        // discarding the first). `header.trie_root` roots the O(key-length)
337        // floor/ceiling walks below, so a clustering read never materializes every
338        // row-index block.
339        let header = resolve_rows_db_entry(rows_db.as_slice(), rows_offset).map_err(|e| {
340            Error::corruption(format!(
341                "BTI clustering seek: Rows.db entry at RowsOffset({rows_offset}) unreadable: {e}"
342            ))
343        })?;
344        let root = header.trie_root;
345
346        // Per-column reverse order for the FIRST clustering column (single-column
347        // scope per #954). A missing/absent schema treats it as ascending.
348        let is_reversed: Vec<bool> = schema
349            .map(|s| {
350                s.clustering_keys
351                    .iter()
352                    .map(|c| matches!(c.order, crate::schema::ClusteringOrder::Desc))
353                    .collect()
354            })
355            .unwrap_or_default();
356
357        // Encode the CQL bounds into the PHYSICAL byte-comparable order the row
358        // index uses, normalizing for a DESC first clustering column (issue #954
359        // High-severity correctness fix). The floor/ceiling walks operate purely in
360        // physical (on-disk, byte-comparable) order, so the CQL lower/upper bounds
361        // must be mapped to the physical-lower/physical-upper sides before the walk.
362        // For a DESC column those roles SWAP (see `physical_byte_bounds_for_slice`).
363        // An un-encodable bound makes the narrowing unsafe → decode the whole
364        // partition (honest fallback).
365        let Some((start_bytes, end_bytes)) = physical_byte_bounds_for_slice(slice, &is_reversed)?
366        else {
367            return Ok(None);
368        };
369
370        // Locate the row-body window with two O(key-length) separator walks instead
371        // of materializing every row-index block then linearly filtering (issue
372        // #1647 / L1). Both mirror Cassandra `RowIndexReader.separatorFloor`:
373        //   - floor(start): the block whose half-open interval `[s_i, s_{i+1})`
374        //     contains the physical-lower bound (largest separator `<= start`), i.e.
375        //     the first block that can hold a row of the slice. `None` when `start`
376        //     sorts below the FIRST separator — the trie-implicit first block at the
377        //     partition body start (issue #1968), which no stored walk can return.
378        //   - strict_ceiling(end): the block with the smallest separator strictly
379        //     `> end` = the EXCLUSIVE window end (the successor of `floor(end)`).
380        //     `None` when the slice reaches the last block (window runs to the
381        //     partition end).
382        // These reproduce the pre-L1 `select_row_index_blocks_for_range` window
383        // (min selected-block start .. successor of max selected-block) EXACTLY, by
384        // construction, for every boundary class (see the rows_floor unit tests).
385        let floor_block =
386            rows_floor_block(rows_db.as_slice(), root, &start_bytes).map_err(|e| {
387                Error::corruption(format!(
388                "BTI clustering seek: Rows.db floor walk at RowsOffset({rows_offset}) failed: {e}"
389            ))
390            })?;
391        let ceil_block =
392            rows_strict_ceiling_block(rows_db.as_slice(), root, &end_bytes).map_err(|e| {
393                Error::corruption(format!(
394                    "BTI clustering seek: Rows.db ceiling walk at RowsOffset({rows_offset}) \
395                     failed: {e}"
396                ))
397            })?;
398
399        // The static row precedes the clustering rows and must be merged into each
400        // emitted clustering row, so we may only fast-forward PAST it when the table
401        // has NO static columns; otherwise decode from the partition body start
402        // (`body_start_rel = 0`) so the static prefix is seen (the END bound still
403        // narrows). (`test_da.wide_table` has no static columns, so the start narrows
404        // too.)
405        let has_static = schema
406            .map(|s| s.columns.iter().any(|c| c.is_static))
407            .unwrap_or(false);
408
409        // IMPLICIT FIRST BLOCK (issue #1968): a `start` below the first separator
410        // selects the implicit block at the partition body start, which no walk can
411        // return. `rows_floor_block` returns `None` in exactly that case, so it IS
412        // the implicit-first signal; the decode must then begin at rel 0 or the
413        // earliest clustering rows are dropped.
414        let includes_implicit_first_block = floor_block.is_none();
415
416        // The start narrows to the floor block only when NEITHER a static row NOR the
417        // implicit first block forces decoding from rel 0.
418        let narrows_start = !has_static && !includes_implicit_first_block;
419
420        // CORRECTNESS GUARD (no-heuristics, never wrong results): FLAG_OPEN_MARKER on
421        // the floor block's `IndexInfo` means a range tombstone is OPEN at the START
422        // of the narrowed window — a deletion opened in an earlier block still shadows
423        // rows inside the slice. Skipping earlier blocks would drop that range-open
424        // marker and risk resurrecting a deleted row (the post-scan backstop only
425        // FILTERS rows, it cannot re-apply a missed range tombstone). Per Cassandra
426        // `RowIndexReader.IndexInfo.openDeletion`, the marker on the floor block fully
427        // captures any deletion open at the window start, so this is the exact, tight
428        // guard: when it fires (only relevant when the start actually narrows) fall
429        // back to a full-partition decode. When the decode already starts at rel 0
430        // (static / implicit-first) no earlier block is skipped, so no guard is
431        // needed. (`test_da.wide_table` has no range tombstones.)
432        if narrows_start
433            && floor_block
434                .as_ref()
435                .map(|b| b.open_marker.is_some())
436                .unwrap_or(false)
437        {
438            debug!(
439                "BTI clustering seek: floor row-index block carries an open range-tombstone \
440                 marker; decoding full partition to preserve range-deletion semantics (no \
441                 narrowing)"
442            );
443            return Ok(None);
444        }
445
446        // body_start_rel: the floor block's offset when the start narrows, else the
447        // partition body start (rel 0). The block `data_offset` is relative to the
448        // partition start (the same domain the parser sees for `window[within..]`).
449        let body_start_rel = match &floor_block {
450            Some(b) if narrows_start => b.data_offset as usize,
451            _ => 0,
452        };
453        // body_end_rel: the exclusive successor block start, or the partition end
454        // (`usize::MAX`, clamped by the caller against the authoritative partition
455        // end / data-section length) when the slice reaches the last block.
456        let body_end_rel = ceil_block
457            .map(|b| b.data_offset as usize)
458            .unwrap_or(usize::MAX);
459
460        Ok(Some(ClusteringRowWindow {
461            body_start_rel,
462            body_end_rel,
463        }))
464    }
465
466    /// BTI ("da") full scan: decompress the whole Data.db section and parse
467    /// every partition in token order (issue #660).
468    ///
469    /// BTI SSTables carry no Index.db/Summary.db, so a range/full scan cannot
470    /// use the index path. Instead we stitch the entire (chunk-compressed) data
471    /// section into one buffer and run [`parse_block_with_cell_metadata`], which
472    /// walks ALL partitions — the same per-partition decode the point-lookup
473    /// path uses, but without stopping at the first match.
474    ///
475    /// Returns entries with per-cell write metadata so the WRITETIME/TTL scan
476    /// (`scan_with_cell_metadata`) and the plain `scan` (which drops the metadata)
477    /// can share a single implementation. Results are filtered by the optional
478    /// `[start_key, end_key]` range and tombstone-suppressed, then sorted into
479    /// Murmur3 token order and truncated to `limit` — identical post-processing
480    /// to the V5CompressedLegacy stitched path.
481    ///
482    /// Uses its own per-scan
483    /// [`ScanCursor`](crate::storage::sstable::reader::source::ScanCursor), so it
484    /// runs in parallel with other scans on this reader without serialization
485    /// (issue #815).
486    ///
487    /// [`parse_block_with_cell_metadata`]: crate::storage::sstable::reader::parsing::V5CompressedLegacyParser::parse_block_with_cell_metadata
488    ///
489    /// `read_shadowing` (issue #1741): `true` for user-facing SELECT scans
490    /// (`scan`, `scan_with_cell_metadata`), `false` for the physical
491    /// `get_all_entries` (integrity verification / data-manager) which must count
492    /// every on-disk row.
493    pub(super) async fn bti_scan_with_metadata(
494        &self,
495        start_key: Option<&RowKey>,
496        end_key: Option<&RowKey>,
497        limit: Option<usize>,
498        schema: Option<&crate::schema::TableSchema>,
499        read_shadowing: bool,
500    ) -> Result<
501        Vec<(
502            RowKey,
503            ScanRow,
504            std::collections::HashMap<String, CellWriteMetadata>,
505        )>,
506    > {
507        let cursor = self.new_scan_cursor().await?;
508
509        // Decompress the entire data section. Precondition for stitch_all_chunks:
510        // cursor's file seeked to data-section start (fresh cursor is at chunk 0).
511        let header_size = self.calculate_header_size();
512        {
513            let mut file_guard = cursor.file.lock().await;
514            // A5 read-work counter (SEEK_CALLS; consumer E4): scan seek-to-data-start
515            // before stitching the section. No-op in release (design.md Decision 1/2).
516            crate::storage::sstable::read_work_counters::record_seek();
517            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
518        }
519        let whole = self.stitch_all_chunks(&cursor).await?;
520
521        // Resolve schema via the four-tier strategy (provided > header > registry).
522        // V5CompressedLegacy partition decode requires a schema (cells lack names).
523        let effective_schema = self.get_table_schema(schema);
524        let parser = self.build_v5_parser(read_shadowing);
525        let parsed =
526            parser.parse_block_with_cell_metadata(&whole, effective_schema.as_ref(), self)?;
527
528        let mut results = Vec::new();
529        for (_entry_table_id, entry_key, entry_value, cell_meta) in parsed {
530            if let Some(start) = start_key {
531                if &entry_key < start {
532                    continue;
533                }
534            }
535            if let Some(end) = end_key {
536                if &entry_key > end {
537                    continue;
538                }
539            }
540            if !self.filter_tombstone(&entry_value) {
541                continue;
542            }
543            results.push((entry_key, entry_value, cell_meta));
544        }
545
546        sort_by_token_order_with_meta(&mut results);
547        if let Some(lim) = limit {
548            results.truncate(lim);
549        }
550
551        tracing::debug!(
552            "SSTableReader::bti_scan_with_metadata - Returning {} results",
553            results.len()
554        );
555        Ok(results)
556    }
557}
