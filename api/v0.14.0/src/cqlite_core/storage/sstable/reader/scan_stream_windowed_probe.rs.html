1//! Test-only probe (issues #1143, #1333) for the windowed streaming scan.
2//!
3//! Records (a) the [`std::thread::ThreadId`] on which the windowed scan's
4//! decompress+parse half actually ran, so a guard test can deterministically
5//! prove that work executed on a `spawn_blocking` thread and NOT on a tokio async
6//! worker (#1143); and (b) how many times the per-partition scratch buffer GREW
7//! its backing store, so a guard test can prove the scratch is reused across
8//! partitions rather than reallocated per partition (#1333).
9//!
10//! Included via `#[cfg(feature = "scan-offload-probe")] #[path =
11//! "scan_stream_windowed_probe.rs"] pub mod probe;` in the parent module and
12//! compiled ONLY under the non-default `scan-offload-probe` feature. In a
13//! normal/default/release build this module, its statics, and its call-sites do
14//! not exist at all — the probe never enters the crate's public surface and adds
15//! zero cost (issue #1143 finding 1). Kept in a sibling file so the parent stays
16//! under the campsite-rule size limit (epic #1116).
17
18use std::sync::atomic::{AtomicBool, AtomicUsize, Ordering};
19use std::sync::Mutex;
20use std::thread::ThreadId;
21
22static ARMED: AtomicBool = AtomicBool::new(false);
23static LAST_PARSE_THREAD: Mutex<Option<ThreadId>> = Mutex::new(None);
24/// The [`ThreadId`] on which the scan's raw chunk READ last ran (issue #1593,
25/// F3). For a synchronously-faulting backend (mmap page fault / `O_DIRECT`
26/// `pread`) the read must run on a `spawn_blocking` thread, NOT an async worker;
27/// a guard test compares this against the enumerated async-worker set.
28static LAST_IO_READ_THREAD: Mutex<Option<ThreadId>> = Mutex::new(None);
29/// The [`ThreadId`] on which the scan's chunk DECOMPRESSION last ran (issue
30/// #1940 runtime-placement guard). The D2 substrate change moved decompression
31/// into the IO-half feed loop; that loop must run on a `spawn_blocking` thread
32/// for EVERY backend (including the genuinely-async buffered backend), so the
33/// decode CPU never lands on the async reactor. A guard test compares this
34/// against the enumerated async-worker set — recorded at the actual
35/// `decode_scan_chunk` decompress site so it pins where decompression runs, not
36/// merely where the read is dispatched.
37static LAST_DECODE_THREAD: Mutex<Option<ThreadId>> = Mutex::new(None);
38/// Number of times the per-partition scratch buffer GREW its backing store
39/// during the armed scan (issue #1333). With the scratch hoisted out of the
40/// per-partition loop and `.clear()`-reused this is a small bounded count
41/// (the buffer grows only up to its high-water mark); if the buffer were
42/// reallocated per partition it would grow from empty every partition, so this
43/// count would scale with partition count.
44static SCRATCH_ALLOCS: AtomicUsize = AtomicUsize::new(0);
45
46/// Arm the probe and clear any previously recorded thread + scratch-growth
47/// count. Call from a test before driving a scan.
48pub fn arm() {
49    if let Ok(mut g) = LAST_PARSE_THREAD.lock() {
50        *g = None;
51    }
52    if let Ok(mut g) = LAST_IO_READ_THREAD.lock() {
53        *g = None;
54    }
55    if let Ok(mut g) = LAST_DECODE_THREAD.lock() {
56        *g = None;
57    }
58    SCRATCH_ALLOCS.store(0, Ordering::SeqCst);
59    ARMED.store(true, Ordering::SeqCst);
60}
61
62/// Disarm the probe (restores the production no-op state).
63pub fn disarm() {
64    ARMED.store(false, Ordering::SeqCst);
65}
66
67/// Record the current thread as the parse thread, if armed. Called from the
68/// blocking parse half after a scan's parse work completes.
69pub(super) fn record_parse_thread() {
70    if ARMED.load(Ordering::Relaxed) {
71        if let Ok(mut g) = LAST_PARSE_THREAD.lock() {
72            *g = Some(std::thread::current().id());
73        }
74    }
75}
76
77/// The [`ThreadId`] recorded by the most recent parse, if any.
78pub fn recorded_parse_thread() -> Option<ThreadId> {
79    LAST_PARSE_THREAD.lock().ok().and_then(|g| *g)
80}
81
82/// Record the current thread as the raw-chunk READ thread, if armed. Called
83/// once per scan, at the start of the feed loop (before the first chunk read)
84/// (issue #1593, F3). For a synchronously-faulting backend this must be a
85/// `spawn_blocking` thread, not an async worker.
86pub(super) fn record_io_read_thread() {
87    if ARMED.load(Ordering::Relaxed) {
88        if let Ok(mut g) = LAST_IO_READ_THREAD.lock() {
89            *g = Some(std::thread::current().id());
90        }
91    }
92}
93
94/// The [`ThreadId`] on which the scan's raw chunk read last ran, if any (issue
95/// #1593, F3).
96pub fn recorded_io_read_thread() -> Option<ThreadId> {
97    LAST_IO_READ_THREAD.lock().ok().and_then(|g| *g)
98}
99
100/// Record the current thread as the chunk-DECODE (decompress) thread, if armed.
101/// Called from `decode_scan_chunk` at the actual decode site, once per decoded
102/// chunk (issue #1940 runtime-placement guard). Cheap enough to call per chunk:
103/// a `Relaxed` load plus, when armed, a short mutex store.
104pub(super) fn record_decode_thread() {
105    if ARMED.load(Ordering::Relaxed) {
106        if let Ok(mut g) = LAST_DECODE_THREAD.lock() {
107            *g = Some(std::thread::current().id());
108        }
109    }
110}
111
112/// The [`ThreadId`] on which the scan's chunk decompression last ran, if any
113/// (issue #1940 runtime-placement guard).
114pub fn recorded_decode_thread() -> Option<ThreadId> {
115    LAST_DECODE_THREAD.lock().ok().and_then(|g| *g)
116}

118/// Record that the per-partition scratch buffer's capacity changed from
119/// `before` to `after`. Counts one growth event whenever `after > before`
120/// (a Vec grows its backing store only by reallocating). Called once per
121/// partition from [`super::SSTableReader::drain_scan_window`]; a no-op unless
122/// armed. Issue #1333 scratch-reuse guard.
123pub(super) fn note_scratch_capacity(before: usize, after: usize) {
124    if ARMED.load(Ordering::Relaxed) && after > before {
125        SCRATCH_ALLOCS.fetch_add(1, Ordering::Relaxed);
126    }
127}
128
129/// The number of scratch-buffer growth events recorded during the armed scan.
130/// Bounded and independent of partition count when the scratch is reused
131/// (issue #1333); scales with partition count if it is reallocated per
132/// partition.
133pub fn recorded_scratch_allocs() -> usize {
134    SCRATCH_ALLOCS.load(Ordering::Relaxed)
135}
