1//! Cassandra 5.0 `Partitions.db` trie-payload decoding, traversal, and lookup.
2//!
3//! Cassandra 5.0 `Partitions.db` has NO header.  The file is a compact trie
4//! whose nodes are written bottom-up (children appear at lower offsets than
5//! parents), with the root offset stored as the **last 8 bytes** of the file as
6//! a big-endian u64.
7//!
8//! Every leaf node (ordinal 0 = PayloadOnly) carries a *payload* described by
9//! the `payloadBits` value encoded in the **low nibble** of the node's header
10//! byte.  In Cassandra 5.0 the hash byte is **always present**
11//! (`FLAG_HAS_HASH_BYTE = 8`), so `payloadBits >= 8` for every real partition
12//! leaf.  The payload layout is:
13//!
14//!   byte 0            : partition-key filter hash (lowest 8 bits of Murmur3)
15//!   bytes 1 .. N      : SizedInts-encoded signed `position`, where
16//!                        N = payloadBits − 8 + 1  (= payloadBits − 7)
17//!
18//! **Sign convention** (mirrors `PartitionIndex.java`):
19//!   - If `position < 0` → `data_offset = ~position` (bitwise NOT; integer NOT)
20//!   - If `position >= 0` → `position` points into `Rows.db` (wide partition)
21//!
22//! Reference: `PartitionIndex.java:131–135`, `BtiFormat.md:946–963`,
23//!            `SizedInts.java` (local mirror: `sized_ints.rs`).
24

33/// The `FLAG_HAS_HASH_BYTE` bit in `payloadBits` (low nibble of a BTI leaf
34/// node's header byte).  When set, the first payload byte is the partition-key
35/// filter hash; the remaining bytes encode the `position` via `SizedInts`.
36///
37/// In Cassandra 5.0 this bit is **always** set for partition index leaves.
38/// Mirrors `PartitionIndex.java:131`.
39pub const FLAG_HAS_HASH_BYTE: u8 = 8;
40
41/// Result of decoding a BTI partition leaf payload.
42///
43/// Mirrors `PartitionIndex.java` payload interpretation: either a direct
44/// `Data.db` offset (for narrow partitions) or a `Rows.db` offset (for wide
45/// partitions that need row-level indexing).
46#[derive(Debug, Clone, PartialEq, Eq)]
47pub enum BtiPartitionLocation {
48    /// Direct `Data.db` byte offset.  The partition starts here.
49    ///
50    /// Occurs when the trie leaf's encoded position is **negative**:
51    /// `data_offset = ~position` (bitwise NOT).
52    DataOffset(u64),
53
54    /// Offset into `Rows.db` for this partition's row-level trie index.
55    ///
56    /// Occurs when the trie leaf's encoded position is **non-negative**.
57    RowsOffset(u64),
58}

60/// Decode the BTI partition leaf payload at `payload_start` inside `trie_data`.
61///
62/// # Arguments
63/// * `trie_data`      – raw bytes of the trie (i.e. the file **excluding** the
64///   8-byte root-offset footer).
65/// * `payload_start`  – offset within `trie_data` where the payload begins
66///   (immediately after the 1-byte node header).
67/// * `payload_bits`   – the `payloadBits` value from the node header's low nibble.
68///
69/// # Returns
70/// `Ok(BtiPartitionLocation)` on success; a parse error if `payload_bits` is
71/// out of the expected range or the data is too short.
72///
73/// # Reference
74/// Mirrors `PartitionIndex.java` `readPayload` and Cassandra's `SizedInts.read`.
75pub fn decode_bti_partition_payload(
76    trie_data: &[u8],
77    payload_start: usize,
78    payload_bits: u8,
79) -> BtiResult<BtiPartitionLocation> {
80    // Every real Cassandra 5.0 partition leaf must have the hash byte flag set.
81    if payload_bits < FLAG_HAS_HASH_BYTE {
82        return Err(Error::Parse(format!(
83            "BTI payload_bits {payload_bits} < FLAG_HAS_HASH_BYTE (8); \
84             hash-byte-less payloads are not supported in Cassandra 5.0 BTI format"
85        )));
86    }
87    if payload_bits > 16 {
88        // SizedInts stores at most 8 bytes; plus 1 hash byte = 9 bytes max → payloadBits ≤ 16
89        return Err(Error::Parse(format!(
90            "BTI payload_bits {payload_bits} > 16; invalid BTI partition leaf"
91        )));
92    }
93
94    // N = payloadBits − FLAG_HAS_HASH_BYTE + 1
95    // For payloadBits = 8 (most common: 1-byte position): N = 1
96    // For payloadBits = 9: N = 2, etc.
97    let position_bytes = (payload_bits - FLAG_HAS_HASH_BYTE + 1) as usize;
98
99    // Minimum payload = 1 hash byte + position_bytes
100    let needed = 1 + position_bytes;
101    if payload_start + needed > trie_data.len() {
102        return Err(Error::Parse(format!(
103            "BTI partition payload at {payload_start} is too short: \
104             need {needed} bytes, have {}",
105            trie_data.len().saturating_sub(payload_start)
106        )));
107    }
108
109    // Skip hash byte (payload_start + 0), read position starting at payload_start + 1
110    let pos_data = &trie_data[payload_start + 1..payload_start + 1 + position_bytes];
111
112    // Sign-extend the big-endian bytes into a signed i64 (SizedInts semantics)
113    let position: i64 = sized_ints_read_from_slice(pos_data)?;
114
115    // Sign convention: negative → ~position is the Data.db byte offset
116    if position < 0 {
117        let data_offset = !position as u64; // bitwise NOT of the negative value
118        Ok(BtiPartitionLocation::DataOffset(data_offset))
119    } else {
120        Ok(BtiPartitionLocation::RowsOffset(position as u64))
121    }
122}

124/// Read a signed big-endian SizedInts value from a byte slice.
125///
126/// Mirrors `SizedInts.read(DataInputPlus, int)` from Cassandra — sign-extends
127/// the most-significant byte.  Accepts 1–8 bytes.
128///
129/// This is the slice-based analogue of `crate::storage::sstable::bti::sized_ints::read`
130/// (which operates on a `std::io::Read` stream).
131pub(crate) fn sized_ints_read_from_slice(data: &[u8]) -> BtiResult<i64> {
132    match data.len() {
133        0 => Ok(0),
134        1 => Ok(data[0] as i8 as i64),
135        2 => Ok(i16::from_be_bytes([data[0], data[1]]) as i64),
136        3 => {
137            let high = data[0] as i8 as i64;
138            let low = u16::from_be_bytes([data[1], data[2]]) as i64;
139            Ok((high << 16) | low)
140        }
141        4 => Ok(i32::from_be_bytes([data[0], data[1], data[2], data[3]]) as i64),
142        5 => {
143            let high = data[0] as i8 as i64;
144            let low = u32::from_be_bytes([data[1], data[2], data[3], data[4]]) as i64;
145            Ok((high << 32) | low)
146        }
147        6 => {
148            let high = i16::from_be_bytes([data[0], data[1]]) as i64;
149            let low = u32::from_be_bytes([data[2], data[3], data[4], data[5]]) as i64;
150            Ok((high << 32) | low)
151        }
152        7 => {
153            let high1 = data[0] as i8 as i64;
154            let high2 = u16::from_be_bytes([data[1], data[2]]) as i64;
155            let low = u32::from_be_bytes([data[3], data[4], data[5], data[6]]) as i64;
156            Ok((high1 << 48) | (high2 << 32) | low)
157        }
158        8 => Ok(i64::from_be_bytes([
159            data[0], data[1], data[2], data[3], data[4], data[5], data[6], data[7],
160        ])),
161        n => Err(Error::Parse(format!(
162            "SizedInts: invalid byte count {n} (expected 1–8)"
163        ))),
164    }
165}

167/// Parse a BTI trie node from `trie_data` at `node_offset` for **traversal**
168/// purposes (i.e. to follow child pointers).  Unlike [`parse_bti_node`], this
169/// function does NOT require any particular payload format: for PayloadOnly and
170/// for nodes with a non-zero `payloadBits` low nibble the payload bytes are
171/// simply skipped — the caller is responsible for reading the payload via
172/// [`read_node_payload`].
173///
174/// This is important because real Cassandra 5.0 `Partitions.db` uses a
175/// compact SizedInts-based payload (2–10 bytes) that is incompatible with the
176/// legacy 12-byte `PayloadRef` format expected by `parse_payload_ref`.
177pub(crate) fn parse_bti_node_for_traversal(
178    trie_data: &[u8],
179    node_offset: usize,
180) -> BtiResult<BtiNode> {
181    if node_offset >= trie_data.len() {
182        return Err(Error::Parse(format!(
183            "BTI traversal: node_offset {node_offset} >= trie_data.len {}",
184            trie_data.len()
185        )));
186    }
187
188    let data = &trie_data[node_offset..];
189    let header_byte = data[0];
190    let ordinal = (header_byte >> 4) & 0x0F;
191    let payload_flags = header_byte & 0x0F;
192    let node_type = classify_node_nibble(ordinal)?;
193
194    match node_type {
195        BtiNodeType::PayloadOnly => {
196            // PayloadOnly leaf: no children.  Skip payload (we decode it separately).
197            if payload_flags == 0 {
198                return Err(Error::Parse(
199                    "PayloadOnly node has no payload flags set".to_string(),
200                ));
201            }
202            // Use a stub PayloadRef — the caller uses read_node_payload for the real data
203            let stub = crate::storage::sstable::bti::node::PayloadRef::new(0, 0);
204            Ok(BtiNode {
205                node_type,
206                level: 0,
207                key_prefix: Vec::new(),
208                data: BtiNodeData::PayloadOnly { payload: stub },
209            })
210        }
211        BtiNodeType::Single => {
212            // Reuse the existing full parser — it handles pointers correctly.
213            // Payload (if any) comes after the pointer bytes and is not read here.
214            parse_bti_node(data, node_offset as u64)
215        }
216        BtiNodeType::Sparse => parse_bti_node(data, node_offset as u64),
217        BtiNodeType::Dense => parse_bti_node(data, node_offset as u64),
218    }
219}

221/// Walk one BTI trie node at `node_offset` and return the absolute offset of
222/// the child reachable via `search_byte`, or `None` if no such transition
223/// exists.
224///
225/// Returns `Ok(Some(child_offset))` when the byte is found and the child is an
226/// internal node.  Returns `Ok(None)` when there is no matching transition
227/// (key not in trie).
228///
229/// Separately, if the *current* node (at `node_offset`) has an embedded
230/// payload **and** `key_exhausted` is true, the caller should read that
231/// payload; this function does not return it — see [`read_node_payload`].
232fn find_next_child_offset(
233    trie_data: &[u8],
234    node_offset: usize,
235    search_byte: u8,
236) -> BtiResult<Option<usize>> {
237    // Issue #1574 (C3): resolve ONLY the child pointer for `search_byte` in place,
238    // instead of materializing the node's full child table via
239    // `parse_bti_node_for_traversal` just to follow one byte. The in-place decode
240    // produces bit-identical child offsets (same `saturating_sub` arithmetic and
241    // Dense delta-0 sentinel) and the same structural-error semantics.
242    super::slice_walk::find_child_offset(trie_data, node_offset, search_byte)
243}

245/// Read the `BtiPartitionLocation` from the payload attached to the BTI node at
246/// `node_offset` (`None` if it has no payload).  Pass `parsed = Some(&node)` when the
247/// caller already parsed the node (the DFS single parse, issue #1650 / L3) so a
248/// payload-bearing internal node is not re-parsed; `None` parses it lazily here.
249pub(crate) fn read_node_payload(
250    trie_data: &[u8],
251    node_offset: usize,
252    parsed: Option<&BtiNode>,
253) -> BtiResult<Option<BtiPartitionLocation>> {
254    let node = parsed;
255    if node_offset >= trie_data.len() {
256        return Err(Error::Parse(format!(
257            "BTI payload read: node_offset {node_offset} out of bounds"
258        )));
259    }
260    let header_byte = trie_data[node_offset];
261    let ordinal = (header_byte >> 4) & 0x0F;
262    let payload_flags = header_byte & 0x0F; // aka payloadBits
263
264    // Ordinals 1 (SingleNoPayload4) and 3 (SingleNoPayload12) encode their
265    // backward delta in the low nibble / low 12 bits — that nibble is NOT a
266    // payload flag, and these node types can NEVER carry a payload.  Treating
267    // their low nibble as `payloadBits` was a latent bug exposed by full DFS
268    // (issue #832): e.g. a `0x18` SingleNoPayload4 was misread as having a
269    // payload, producing a spurious entry.
270    if ordinal == 1 || ordinal == 3 {
271        return Ok(None);
272    }
273
274    if ordinal == 0 {
275        // PayloadOnly node: the entire node is just [header][payload…]
276        if payload_flags == 0 {
277            return Err(Error::Parse(
278                "PayloadOnly node has zero payload_flags".to_string(),
279            ));
280        }
281        let payload_start = node_offset + 1; // immediately after the 1-byte header
282        Ok(Some(decode_bti_partition_payload(
283            trie_data,
284            payload_start,
285            payload_flags,
286        )?))
287    } else if payload_flags != 0 {
288        // Non-leaf node with an embedded payload: skip its pointer/transition bytes
289        // to reach the payload.  Use the caller's pre-parsed node (DFS single parse,
290        // issue #1650) when supplied, else parse here.
291        let owned;
292        let node = match node {
293            Some(n) => n,
294            None => {
295                owned = parse_bti_node(&trie_data[node_offset..], node_offset as u64)?;
296                &owned
297            }
298        };
299        let payload_start = payload_start_in_node(node, trie_data, node_offset)?;
300        Ok(Some(decode_bti_partition_payload(
301            trie_data,
302            payload_start,
303            payload_flags,
304        )?))
305    } else {
306        Ok(None)
307    }
308}

310/// Compute the byte offset of the payload within `trie_data` for a non-leaf
311/// node that carries an embedded payload (`payloadBits != 0`).
312///
313/// This is the byte position immediately after all the node's transition bytes
314/// and pointer bytes, but before any subsequent node data.
315pub(crate) fn payload_start_in_node(
316    node: &BtiNode,
317    trie_data: &[u8],
318    node_offset: usize,
319) -> BtiResult<usize> {
320    use BtiNodeData::*;
321    let header_byte = trie_data[node_offset];
322    let ordinal = (header_byte >> 4) & 0x0F;
323
324    let payload_offset = match &node.data {
325        PayloadOnly { .. } => {
326            // Header(1) + payload: caller handles this case separately
327            node_offset + 1
328        }
329        Single { .. } => {
330            // Singles with payload: header(1) + transition(1) + ptr_bytes
331            let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
332            // Ordinals 1 and 3 are "NoPayload" variants; they cannot have payloads
333            // (the payload_flags nibble is always 0 for those ordinals, so this
334            // branch is unreachable for them).
335            node_offset + 1 + 1 + ptr_bytes
336        }
337        Sparse { transitions } => {
338            let count = transitions.len();
339            let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
340            // Ordinal 6 (Sparse12): packed 12-bit pointers, ceil(count*3/2) bytes
341            let ptr_area = if ordinal == 6 {
342                (count * 3).div_ceil(2)
343            } else {
344                count * ptr_bytes
345            };
346            node_offset + 1 + 1 + count + ptr_area // header + count + transitions + deltas
347        }
348        Dense { children, .. } => {
349            let range_len = children.len();
350            let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
351            let ptr_area = if ordinal == 10 {
352                // Dense12
353                (range_len * 3).div_ceil(2)
354            } else {
355                range_len * ptr_bytes
356            };
357            node_offset + 1 + 1 + 1 + ptr_area // header + start_byte + len-1 + deltas
358        }
359    };
360    Ok(payload_offset)
361}

363/// Walk the BTI partition trie loaded in `trie_data`, looking for a partition
364/// whose **byte-comparable** encoded key *prefix* routes to a leaf.
365///
366/// Cassandra BTI uses a **path-compressed (Patricia) trie**: once a path leads to
367/// a `PayloadOnly` leaf, the remaining key bytes are stored implicitly (as the
368/// compressed suffix).  Therefore a match is declared as soon as a `PayloadOnly`
369/// leaf is reached — regardless of whether the encoded key has been fully consumed.
370/// The caller is responsible for verifying the actual partition key against the
371/// Data.db bytes at the resolved offset (using the partition's hash byte for a
372/// fast pre-filter and the raw key bytes for definitive confirmation).
373///
374/// Returns:
375/// - `Ok(Some(BtiPartitionLocation))` when a leaf is reached.
376/// - `Ok(None)` when no transition exists for a key byte (key not in trie).
377/// - `Err(_)` on structural parse errors.
378///
379/// This is the inner engine for [`lookup_partition_in_bti_file`].
380pub(crate) fn walk_bti_trie(
381    trie_data: &[u8],
382    root_offset: usize,
383    encoded_key: &[u8],
384) -> BtiResult<Option<BtiPartitionLocation>> {
385    let mut current_offset = root_offset;
386    let mut key_pos = 0;
387
388    loop {
389        if current_offset >= trie_data.len() {
390            return Err(Error::Parse(format!(
391                "BTI trie walk: offset {current_offset} out of bounds (trie size {})",
392                trie_data.len()
393            )));
394        }
395
396        let header_byte = trie_data[current_offset];
397        let ordinal = (header_byte >> 4) & 0x0F;
398        let payload_flags = header_byte & 0x0F;
399        let is_leaf = ordinal == 0; // PayloadOnly
400
401        if is_leaf {
402            // Path-compressed (Patricia) trie: a PayloadOnly leaf represents the
403            // unique key whose path through the trie led here.  The remaining key
404            // bytes are stored implicitly (compressed suffix) and need not be
405            // compared here — the caller verifies the actual partition key in Data.db.
406            //
407            // Note: the original implementation required the key to be fully consumed
408            // at the leaf (return Ok(None) otherwise), which is incorrect for
409            // path-compressed tries.  Issue #755 corrects this.
410            return read_node_payload(trie_data, current_offset, None);
411        }
412
413        // Non-leaf node: if key exhausted, check for an embedded payload
414        if key_pos >= encoded_key.len() {
415            if payload_flags != 0 {
416                return read_node_payload(trie_data, current_offset, None);
417            }
418            return Ok(None);
419        }
420
421        // Advance: look for the next key byte's transition
422        let next_byte = encoded_key[key_pos];
423        match find_next_child_offset(trie_data, current_offset, next_byte)? {
424            Some(child_offset) => {
425                current_offset = child_offset;
426                key_pos += 1;
427            }
428            None => {
429                // No transition for this byte → key not in trie
430                return Ok(None);
431            }
432        }
433    }
434}

436/// Look up a partition by its **byte-comparable** encoded key in a real
437/// Cassandra 5.0 `Partitions.db` BTI trie file.
438///
439/// # Format
440/// The file is a compact trie with **no header**:
441/// - Nodes are written bottom-up (children before parents).
442/// - The root node's absolute byte offset is stored as the **last 8 bytes** of
443///   the file (big-endian u64).
444/// - Every partition leaf carries a payload consisting of a 1-byte filter hash
445///   plus a SizedInts-encoded signed `position` (see [`decode_bti_partition_payload`]).
446/// - Negative `position` → `data_offset = ~position` (direct `Data.db` pointer).
447/// - Non-negative `position` → `rows_offset` into `Rows.db` (wide partition).
448///
449/// # Arguments
450/// * `reader`      – positioned at any point; this function seeks as needed.
451/// * `encoded_key` – the byte-comparable encoding of the partition key (e.g.
452///   as produced by [`ByteComparableEncoder`](crate::storage::sstable::bti::encoder::ByteComparableEncoder)).
453///
454/// # Returns
455/// * `Ok(Some(BtiPartitionLocation::DataOffset(off)))` – the partition starts
456///   at byte `off` in `Data.db`.  **This is the headline result of issue #755.**
457/// * `Ok(Some(BtiPartitionLocation::RowsOffset(off)))` – the partition has a
458///   row index; consult `Rows.db` at `off` for the row-level trie.
459/// * `Ok(None)` – the key was not found.
460/// * `Err(_)` – structural parse error.
461///
462/// # Reference
463/// Mirrors `PartitionIndex.java` (Cassandra 5.0.8), specifically the
464/// `openBlocking` + `exactCandidate` read path; `SizedInts.read`; and the
465/// payload sign convention at `PartitionIndex.java:131–135`.
466pub fn lookup_partition_in_bti_file<R: Read + Seek>(
467    reader: &mut R,
468    encoded_key: &[u8],
469) -> BtiResult<Option<BtiPartitionLocation>> {
470    // Step 1: determine the file size
471    let file_size = reader.seek(SeekFrom::End(0))?;
472
473    if file_size < 8 {
474        return Err(Error::Parse(format!(
475            "BTI Partitions.db is too small ({file_size} bytes; need at least 8 for footer)"
476        )));
477    }
478
479    // Step 2: read the 8-byte root offset from the end of the file
480    reader.seek(SeekFrom::End(-8))?;
481    let mut footer_buf = [0u8; 8];
482    reader.read_exact(&mut footer_buf)?;
483    let root_offset = u64::from_be_bytes(footer_buf);
484
485    // The trie data is everything before the 8-byte footer
486    let trie_size = file_size - 8;
487
488    if root_offset >= trie_size {
489        return Err(Error::Parse(format!(
490            "BTI Partitions.db: root_offset {root_offset} >= trie_size {trie_size}"
491        )));
492    }
493
494    // Step 3: load the entire trie into memory
495    // For typical Partitions.db files this is small (tens of KB for millions of
496    // partitions because the trie shares prefixes aggressively).
497    reader.seek(SeekFrom::Start(0))?;
498    let mut trie_data = vec![0u8; trie_size as usize];
499    reader.read_exact(&mut trie_data)?;
500
501    // Step 4: walk the trie
502    walk_bti_trie(&trie_data, root_offset as usize, encoded_key)
503}

505// ─────────────────────────────────────────────────────────────────────────────
506// BTI partition key encoding for Murmur3Partitioner (issue #755)
507// ─────────────────────────────────────────────────────────────────────────────
508//
509// Cassandra 5.0 BTI `Partitions.db` trie keys are derived from `DecoratedKey`
510// using `ByteComparable.Version.OSS50`.  For `Murmur3Partitioner` the encoding
511// is a path-compressed representation of:
512//
513//   [type_prefix=0x40] ++ [8 bytes: murmur3_token_bc]
514//
515// where `murmur3_token_bc = (murmur3_token(raw_key_bytes) as u64) XOR 0x8000_0000_0000_0000`
516// (flips the sign bit so signed i64 tokens sort correctly as unsigned big-endian bytes).
517//
518// Because the trie uses path compression (Patricia trie), looking up the first
519// few bytes of this 9-byte key is sufficient to reach the unique leaf for a
520// given partition.  The caller must verify the actual partition key in Data.db
521// to confirm the match (the hash byte in the leaf payload provides a fast pre-filter).
522//
523// This encoding was verified against the real `da-2-bti-Partitions.db` fixture:
524//   UUID 22222222-... → token bc first byte 0x90 → DataOffset(0)   ✓
525//   UUID 11111111-... → token bc first byte 0xBC → DataOffset(63)  ✓
526//   UUID 33333333-... → token bc first byte 0xF9 → DataOffset(125) ✓
527// ─────────────────────────────────────────────────────────────────────────────
528
529/// Encode a raw partition key (any CQL type) into the BTI trie lookup key for
530/// `Murmur3Partitioner`.
531///
532/// The encoding is:
533///   `[0x40] ++ [8 bytes big-endian: (murmur3_token(raw_key) as u64) XOR 0x8000_0000_0000_0000]`
534///
535/// This 9-byte prefix uniquely identifies the partition in the trie for typical
536/// Cassandra datasets.  In a path-compressed (Patricia) trie, using only the
537/// first few bytes is correct because each leaf represents exactly one partition:
538/// once the traversal reaches a leaf, no further byte matching is required.
539///
540/// # Arguments
541/// * `raw_key_bytes` – the raw serialized partition key bytes as stored on disk
542///   (e.g. 16 bytes for a UUID, big-endian bytes for int/bigint, UTF-8 for text).
543///
544/// # Verified against
545/// Real `da-2-bti-Partitions.db` fixture — see issue #755 for derivation.
546pub fn encode_partition_key_for_bti_trie(raw_key_bytes: &[u8]) -> [u8; 9] {
547    use crate::util::cassandra_murmur3::cassandra_murmur3_token;
548
549    // C4 work-counter (KEY_HASH_CALLS, issue #1575): one per Murmur3 hash + encoding
550    // of a query key. C4 hoists this call out of the candidate-prune loop so a
551    // multi-generation point read records exactly 1 (not one per candidate). No-op in
552    // release builds (read_work_counters design.md Decision 1).
553    crate::storage::sstable::read_work_counters::record_key_hash();
554
555    let token: i64 = cassandra_murmur3_token(raw_key_bytes);
556    let bc: u64 = (token as u64) ^ 0x8000_0000_0000_0000u64;
557    let bc_bytes = bc.to_be_bytes();
558
559    let mut key = [0u8; 9];
560    key[0] = 0x40; // fixed type-discriminator prefix (observed in Partitions.db trie)
561    key[1..9].copy_from_slice(&bc_bytes);
562    key
563}

565/// Look up a raw partition key in a Cassandra 5.0 BTI `Partitions.db` trie file
566/// using the `Murmur3Partitioner` byte-comparable encoding.
567///
568/// This is a convenience wrapper over [`lookup_partition_in_bti_file`] that
569/// handles key encoding:
570///
571/// 1. Encodes `raw_key_bytes` → 9-byte BTI trie key via
572///    [`encode_partition_key_for_bti_trie`].
573/// 2. Looks up the key in the trie.
574/// 3. Returns the `BtiPartitionLocation` (Data.db or Rows.db offset).
575///
576/// Because the BTI trie uses path compression the lookup may return a candidate
577/// for a *different* key that shares the same trie prefix.  The caller must
578/// verify the actual partition key at the returned Data.db offset.
579///
580/// # Arguments
581/// * `partitions_db_reader` – seekable reader for the `Partitions.db` file.
582/// * `raw_key_bytes`        – raw on-disk partition key bytes (e.g. 16 bytes for UUID).
583///
584/// # Returns
585/// * `Ok(Some(BtiPartitionLocation::DataOffset(off)))` – candidate Data.db offset.
586/// * `Ok(None)` – definitely not in this SSTable (no trie path for this key prefix).
587/// * `Err(_)` – structural parse error.
588pub fn lookup_raw_key_in_bti_partitions_db<R: Read + Seek>(
589    partitions_db_reader: &mut R,
590    raw_key_bytes: &[u8],
591) -> BtiResult<Option<BtiPartitionLocation>> {
592    let encoded = encode_partition_key_for_bti_trie(raw_key_bytes);
593    lookup_partition_in_bti_file(partitions_db_reader, &encoded)
594}

609        // PayloadOnly header byte: 0x08 (ordinal=0, payloadBits=8)
610        // payload[0] = 0x24 (hash byte)
611        // payload[1] = 0xFF (position = i8(-1) = -1 → ~(-1) = 0)
612        let header_byte: u8 = 0x08;
613        let hash_byte: u8 = 0x24;
614        let position_byte: u8 = 0xFF; // -1 as i8
615        let trie_data = vec![header_byte, hash_byte, position_byte, 0x00, 0x00];
616        let payload_start = 1; // immediately after header
617        let payload_bits = header_byte & 0x0F; // = 8
618        let result = decode_bti_partition_payload(&trie_data, payload_start, payload_bits)
619            .expect("should decode successfully");
620        assert_eq!(
621            result,
622            BtiPartitionLocation::DataOffset(0),
623            "position=-1 (0xFF as i8) must map to data_offset=0 via ~(-1)=0"

856        if !partitions_db.exists() {
