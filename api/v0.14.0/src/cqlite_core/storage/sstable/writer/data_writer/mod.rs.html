1//! Data.db writer - writes partition and row data
2//!
3//! Generates the Data.db component with V5CompressedLegacy (NB) format.
4//! Maintains partition ordering by Murmur3 token and clustering ordering.
5//! Tracks file positions for Index.db generation.
6//!
7//! Critical requirements:
8//! - Partition ordering: By Murmur3 token, then key bytes (enforced by caller)
9//! - Clustering ordering: By clustering comparator within partition (enforced by caller)
10//! - Row size measurement: After VInt length bytes (Issue #237)
11//! - Delta encoding: Uses Statistics.db baseline for timestamps/TTL/deletion times
12//!
13//! # V5CompressedLegacy Row Format
14//!
15//! Each row is encoded as:
16//! ```text
17//! [row_flags: u8]
18//! [extended_flags: u8 if ROW_HAS_EXTENDED_FLAGS set]
19//! [clustering_prefix: variable if present]
20//! [row_size: VInt]                       ← Measured from AFTER this VInt
21//! [prev_size: VInt]
22//! [timestamp: VInt if ROW_HAS_TIMESTAMP]   ← Delta from min_timestamp
23//! [ttl: VInt if ROW_HAS_TTL]              ← Delta from min_ttl
24//! [deletion: 2 VInts if ROW_HAS_DELETION] ← local_deletion_time delta + deletion timestamp
25//! [column_bitmap: VUInt bitmask of missing columns if NOT ROW_HAS_ALL_COLUMNS]
26//! [cell_data...]
27//! ```
28//!
29//! ## Row Flags
30//! - `0x04` (HAS_TIMESTAMP): Timestamp delta present
31//! - `0x08` (HAS_TTL): TTL delta present
32//! - `0x10` (HAS_DELETION): Deletion time present (two VInts)
33//! - `0x20` (HAS_ALL_COLUMNS): All columns present (no bitmap)
34//! - `0x40` (HAS_COMPLEX_DELETION): Row contains complex column with deletion
35//! - `0x80` (HAS_EXTENDED_FLAGS): Extended flags byte follows
36//!
37//! ## Cell Format
38//! ```text
39//! [flags: u8]
40//! [timestamp: VInt if NOT USE_ROW_TIMESTAMP]  ← Delta from min_timestamp
41//! [local_deletion_time: VUInt if deleted/expiring and NOT USE_ROW_TTL]
42//! [ttl: VUInt if expiring and NOT USE_ROW_TTL]
43//! [value_length: VInt]
44//! [value_bytes]
45//! ```
46//!
47//! ## Cell Flags
48//! - `0x01` (IS_DELETED): Cell is a tombstone
49//! - `0x02` (IS_EXPIRING): TTL fields follow
50//! - `0x04` (HAS_EMPTY_VALUE): Zero-length value
51//! - `0x08` (USE_ROW_TIMESTAMP): Use row-level timestamp (no timestamp delta)
52//! - `0x10` (USE_ROW_TTL): Use row-level TTL (no TTL delta)
53//!
54//! References:
55//! - Cassandra 5.0: `org.apache.cassandra.db.rows.UnfilteredSerializer`
56//! - Parser: `cqlite-core/src/storage/sstable/reader/parsing/row_decoder.rs`
57//! - Format docs: `docs/sstables-definitive-guide/chapters/05-data-db-format.md`

103// Range tombstone bound kinds.
104//
105// These are the ordinals of Cassandra's `ClusteringPrefix.Kind` enum
106// (ClusteringPrefix.java) — the byte written on disk by
107// `ClusteringBoundOrBoundary.Serializer.serialize()`:
108//   0 = EXCL_END_BOUND, 1 = INCL_START_BOUND,
109//   2 = EXCL_END_INCL_START_BOUNDARY, 3 = STATIC_CLUSTERING,
110//   4 = CLUSTERING, 5 = INCL_END_EXCL_START_BOUNDARY,
111//   6 = INCL_END_BOUND, 7 = EXCL_START_BOUND.
112// (Issue #717: the writer previously used a private 0..5 numbering that no
113// Cassandra reader understands.)
114const EXCL_END_BOUND: u8 = 0;
115const INCL_START_BOUND: u8 = 1;
116const INCL_END_BOUND: u8 = 6;
117const EXCL_START_BOUND: u8 = 7;
118// The CLUSTERING ordinal (a regular row's clustering name). Used by
119// `partition::sort_class` as the row's Kind tiebreak so co-located bounds order
120// against rows exactly as Cassandra's comparator does (issue #1220).
121const CLUSTERING: u8 = 4;
122
123// Range tombstone BOUNDARY kinds (issue #1220). A boundary closes one range and
124// opens the next at the SAME clustering point, carrying TWO deletion-time pairs
125// (primary = end/close of the previous range, secondary = start/open of the next
126// range). Cassandra emits these whenever two adjacent ranges share a boundary
127// point with complementary inclusivity (`ClusteringBoundOrBoundary.Serializer`):
128//   2 = EXCL_END_INCL_START_BOUNDARY (close exclusive, open inclusive),
129//   5 = INCL_END_EXCL_START_BOUNDARY (close inclusive, open exclusive).
130const EXCL_END_INCL_START_BOUNDARY: u8 = 2;
131const INCL_END_EXCL_START_BOUNDARY: u8 = 5;

134const END_OF_PARTITION: u8 = 0x01;

136/// Capacity of the streaming Data.db `BufWriter` (Issue #492).
137///
138/// Large enough that each flushed partition coalesces into a handful of big
139/// `write()` syscalls instead of many small default-8 KB ones, preserving the
140/// throughput of the previous single whole-file write while keeping resident
141/// memory bounded (this buffer plus one partition's scratch).
142const DATA_SINK_BUFFER_BYTES: usize = 1024 * 1024;
143
144/// Data.db component writer
145///
146/// Writes partitions and rows in V5CompressedLegacy format with delta encoding.
147/// Caller must provide partitions in token order and rows in clustering order.
148///
149/// # Memory model (Issue #492)
150///
151/// The writer supports two modes that produce **byte-identical** Data.db output:
152///
153/// * **In-memory mode** (`DataWriter::new`): every partition is appended to the
154///   `buffer` scratch and never flushed, so `finish()` returns the full Data.db
155///   bytes. Used by unit tests that inspect the produced bytes directly.
156///
157/// * **Streaming mode** (`DataWriter::with_sink`): each partition is built in the
158///   `buffer` scratch, written to a `BufWriter<File>` over the Data.db path, and
159///   the scratch is cleared. Peak heap is therefore `O(largest partition)` rather
160///   than `O(file)`, keeping a multi-GB compaction within the 128 MB target.
161///
162/// In both modes the file offset of a partition is `position + buffer.len()`
163/// measured before any bytes are written. In streaming mode `buffer` is empty at
164/// that point (just cleared) so the offset is `position`; in memory mode
165/// `position` is always 0 and `buffer` holds all prior partitions, so the offset
166/// equals the legacy `buffer.len()`. The within-partition size math uses relative
167/// deltas into `buffer`, which are identical regardless of mode.
168#[derive(Debug)]
169pub struct DataWriter {
170    /// Per-partition scratch buffer for Data.db content.
171    ///
172    /// In streaming mode this is cleared at the start of every `write_partition`
173    /// and flushed to `sink` at the end, so only one partition is resident.
174    /// In memory mode it accumulates the entire Data.db output.
175    buffer: Vec<u8>,
176    /// Streaming sink over the Data.db path (streaming mode only).
177    ///
178    /// Lazily opened on the first `write_partition` so that the keyspace/table
179    /// directory exists before the first byte is written. `None` in in-memory
180    /// mode.
181    sink: Option<std::io::BufWriter<std::fs::File>>,
182    /// Data.db output path (streaming mode only); used for lazy sink open.
183    data_path: Option<PathBuf>,
184    /// Bytes already flushed to `sink`. Always 0 in in-memory mode.
185    position: u64,
186    /// Incremental whole-file (`Digest.crc32`) + per-chunk (`CRC.db`) checksum
187    /// accumulator, fed every flushed byte in write order (issue #1663).
188    ///
189    /// Streaming mode only: [`flush_partition`](Self::flush_partition) feeds each
190    /// flushed scratch buffer through it, so [`finish_streaming`](Self::finish_streaming)
191    /// can return both checksums without re-reading the finished `Data.db`. Left
192    /// untouched (empty) in in-memory mode, which never calls `finish_streaming`.
193    crc: StreamingCrc,
194    /// Encoding baselines used for delta encoding.
195    stats: EncodingStatsBaselines,
196    /// Issue #1741: emit the partition-level `DeletionTime` in the oa/`da`
197    /// serialization (1-byte `0x80` LIVE sentinel; `markedForDeleteAt`(i64) +
198    /// `localDeletionTime`(u32) when deleted) rather than the legacy na/`nb`
199    /// layout (`localDeletionTime`(i32) + `markedForDeleteAt`(i64), LIVE encoded as
200    /// `i32::MAX`+`i64::MIN`). `true` ONLY for `da` (BTI) SSTables, whose reader
201    /// applies `hasUIntDeletionTime` (oa) decoding; a `da` file written with the
202    /// legacy layout has its live-partition sentinel misread as a tombstone (which
203    /// the read-side shadowing then treats as a partition delete). Default `false`
204    /// preserves byte-identical `nb` output.
205    oa_partition_deletion: bool,
206}

259/// Delta-encoding baselines needed by the Data.db row/cell serializers.
260///
261/// `StatisticsMetadata` also owns Statistics.db-only accumulators such as

263/// DataWriter avoids cloning those larger accumulators on every partition.
264#[derive(Debug, Clone, Copy)]
265struct EncodingStatsBaselines {

277        }
278    }

281/// Result of finishing a streaming [`DataWriter`] (issue #1663).
282///
283/// Carries the total Data.db byte size and the checksums accumulated during the
284/// streaming write, so `SSTableWriter::finish` writes `Digest.crc32` and
285/// `CRC.db` without re-reading the finished `Data.db`.
286#[derive(Debug)]
287pub struct StreamFinish {
288    /// Total number of Data.db bytes written (the `data_size`).
289    pub data_size: u64,
290    /// Whole-file CRC32 over the raw Data.db bytes — the `Digest.crc32` value.
291    pub digest_crc32: u32,
292    /// One CRC32 per `CRC_CHUNK_SIZE` chunk of the raw Data.db bytes, in order —
293    /// the `CRC.db` chunk values (pass to `crc_writer::assemble_crc_bytes`).
294    pub chunk_crcs: Vec<u32>,
295}

298    /// Create a new in-memory Data.db writer.
299    ///
300    /// All partitions accumulate in `buffer`; `finish()` returns the full bytes.
301    /// Prefer [`DataWriter::with_sink`] for production writes to bound memory.
302    ///
303    /// # Arguments
304    /// * `stats` - Statistics metadata for delta encoding baselines
305    pub fn new(stats: StatisticsMetadata) -> Self {
306        Self {
307            buffer: Vec::new(),
308            sink: None,
309            data_path: None,
310            position: 0,
311            crc: StreamingCrc::new(),
312            stats: EncodingStatsBaselines::from(&stats),
313            oa_partition_deletion: false,
314        }
315    }

318    /// Call with `true` when producing a `da` (BTI) SSTable; the default (`false`)
319    /// keeps the legacy `nb` layout byte-identical.
320    pub fn with_oa_partition_deletion(mut self, on: bool) -> Self {
321        self.oa_partition_deletion = on;
322        self
323    }

325    /// Create a streaming Data.db writer that flushes each partition to `data_path`.
326    ///
327    /// The file is opened lazily on the first `write_partition` (creating the
328    /// parent directory if needed) so the keyspace/table layout is established
329    /// before any bytes are written. Memory is bounded to the largest single
330    /// partition.
331    ///
332    /// # Arguments
333    /// * `stats` - Statistics metadata for delta encoding baselines
334    /// * `data_path` - Destination path for the Data.db component
335    pub fn with_sink(stats: StatisticsMetadata, data_path: PathBuf) -> Self {
336        Self {
337            buffer: Vec::new(),
338            sink: None,
339            data_path: Some(data_path),
340            position: 0,
341            crc: StreamingCrc::new(),
342            stats: EncodingStatsBaselines::from(&stats),
343            oa_partition_deletion: false,
344        }
345    }
346
347    /// Lazily open the streaming sink (and create the parent directory).
348    ///
349    /// No-op in in-memory mode or once the sink is already open.
350    pub(super) fn ensure_sink(&mut self) -> Result<()> {
351        if self.sink.is_some() {
352            return Ok(());
353        }
354        if let Some(path) = self.data_path.clone() {
355            if let Some(parent) = path.parent() {
356                // Create the keyspace/table tree. The flush durability barrier
357                // fsyncs the full leaf→data-root chain unconditionally before
358                // the WAL truncate, so this creation need not track which
359                // ancestors it made (issue #1392).
360                crate::storage::write_engine::durability::create_dir_all(parent)?;
361            }
362            let file = std::fs::File::create(&path)?;
363            // Use a large BufWriter so a partition's bytes coalesce into a few
364            // big write() syscalls rather than many 8 KB-default ones, matching
365            // the throughput of the old single whole-file write.
366            self.sink = Some(std::io::BufWriter::with_capacity(
367                DATA_SINK_BUFFER_BYTES,
368                file,
369            ));
370        }
371        Ok(())
372    }
373
374    /// In streaming mode, flush the current scratch buffer to the sink, advance
375    /// `position`, and clear the scratch so only one partition is ever resident.
376    /// No-op in in-memory mode (the scratch keeps accumulating).
377    pub(super) fn flush_partition(&mut self) -> Result<()> {
378        if self.data_path.is_none() {
379            // In-memory mode: keep accumulating in `buffer`.
380            return Ok(());
381        }
382        self.ensure_sink()?;
383        if let Some(sink) = self.sink.as_mut() {
384            sink.write_all(&self.buffer)?;
385        }
386        // Feed the just-written bytes through the incremental checksum accumulator
387        // (issue #1663) so `finish_streaming` never re-reads the finished Data.db.
388        // Done here (after the sink write, before the clear) so every byte written
389        // to disk is checksummed exactly once, in write order — chunks straddle
390        // partition boundaries just as the re-read oracle chunks the raw file.
391        self.crc.update(&self.buffer);
392        self.position += self.buffer.len() as u64;
393        self.buffer.clear();
394        Ok(())
395    }

398    ///
399    /// This should be called after computing stats from all mutations
400    /// but before writing any partition data. The stats are used for
401    /// delta encoding of timestamps, TTL, and local deletion times.
402    pub fn update_stats(&mut self, stats: StatisticsMetadata) {
403        self.update_stats_from_metadata(&stats);
404    }

407    ///
408    /// The estimated histograms added for Statistics.db are intentionally not
409    /// copied into `DataWriter`; they are unrelated to Data.db delta encoding.
410    pub fn update_stats_from_metadata(&mut self, stats: &StatisticsMetadata) {
411        self.stats = EncodingStatsBaselines::from(stats);
412    }

414    /// Finish writing and return the Data.db bytes (in-memory mode).
415    ///
416    /// Only valid for writers created via [`DataWriter::new`]. In streaming mode
417    /// the bytes live on disk; use [`DataWriter::finish_streaming`] instead.
418    pub fn finish(self) -> Result<Vec<u8>> {
419        // Hard guard (not debug_assert!, which compiles out in release): on a
420        // streaming writer the bytes live on disk and `buffer` is empty after each
421        // partition flush, so returning it would silently yield a 0-byte Data.db.
422        if self.data_path.is_some() {
423            return Err(Error::InvalidInput(
424                "DataWriter::finish() called on a streaming writer; use finish_streaming()"
425                    .to_string(),
426            ));
427        }
428        Ok(self.buffer)
429    }
430
431    /// Finish a streaming writer: flush the sink to disk and return the total
432    /// byte size plus the checksums accumulated during the write (issue #1663).
433    ///
434    /// Any residual scratch (there is none in normal operation, since
435    /// `write_partition` flushes per partition) is flushed first. Returns an
436    /// error if the writer was created in in-memory mode.
437    ///
438    /// The returned [`StreamFinish`] carries the whole-file `Digest.crc32` value
439    /// and the per-chunk `CRC.db` values that were accumulated as the data
440    /// streamed to disk, so `SSTableWriter::finish` computes neither by
441    /// re-reading the finished `Data.db`.
442    pub fn finish_streaming(mut self) -> Result<StreamFinish> {
443        if self.data_path.is_none() {
444            return Err(Error::InvalidInput(
445                "finish_streaming() called on an in-memory DataWriter".to_string(),
446            ));
447        }
448        // Flush any residual scratch (normally empty), then flush the BufWriter
449        // so all bytes reach the OS file, and fsync the file so its *contents*
450        // are durable on the storage device (issue #1392). A plain `flush()`
451        // only pushes bytes into the page cache; without the fsync a crash after
452        // the WAL is truncated could lose the Data.db contents even though the
453        // directory entry was persisted. This fsync (plus the per-component
454        // fsyncs in `SSTableWriter::finish`) is the durability guarantee for
455        // both the flush and compaction write paths.
456        self.flush_partition()?;
457        if let Some(mut sink) = self.sink.take() {
458            sink.flush()?;
459            sink.get_ref()
460                .sync_all()
461                .map_err(|e| Error::Storage(format!("Failed to fsync Data.db contents: {e}")))?;
462        }
463        // Finalize the incremental checksums accumulated in `flush_partition`
464        // (issue #1663). `finalize` closes any trailing short chunk.
465        let data_size = self.position;
466        let (digest_crc32, chunk_crcs) = self.crc.finalize();
467        Ok(StreamFinish {
468            data_size,
469            digest_crc32,
470            chunk_crcs,
471        })
472    }

474    /// Get current file position (for Index.db offset tracking).
475    ///
476    /// This is the total number of Data.db bytes produced so far: bytes already
477    /// flushed to the sink (`position`) plus bytes currently buffered. Identical
478    /// in both streaming and in-memory modes.
479    pub fn position(&self) -> u64 {
480        self.position + self.buffer.len() as u64
481    }
