1//! Incremental partition-write entry point (issue #1668, stage 5c-iv, PART 1
2//! — build + prove, not yet wired to any production caller).
3//!
4//! `write_partition`/`write_partition_with_index_blocks` (`partition.rs`)
5//! need the WHOLE partition's `&[Mutation]` slice upfront: they sort it, run
6//! `collect_static_operations` over it, build `rows` via
7//! `merge_clustering_rows`, then interleave rows with range-tombstone markers
8//! via a full re-sort. [`IncrementalPartitionWriter`] proves the SAME Data.db
9//! bytes can be produced from a partition fed ONE PIECE AT A TIME:
10//!
11//! 1. [`DataWriter::begin_partition_incremental`] — writes the header
12//!    immediately (the partition tombstone, if any, is known upfront from the
13//!    caller — the eventual merge-side caller has it from the stage-1
14//!    carrier pre-scan). Builds and sorts the (typically few) range-
15//!    tombstone markers ONCE via `marker_merge::partition_item_cmp` (stage
16//!    5c-iii) — NOT the whole per-row items list.
17//! 2. [`IncrementalPartitionWriter::feed_static_row`] — called AT MOST ONCE,
18//!    right after `begin_partition_incremental`, BEFORE any clustering row.
19//!    Compaction's own merge reconciliation guarantees a static-row carrier
20//!    (if any) is the ONLY `clustering_key: None` entry in a partition's
21//!    reconciled `Vec<MergeEntry>` and ALWAYS sorts first — proven directly
22//!    in `streaming.rs`'s `static_row_carrier_always_sorts_first_regardless_
23//!    of_partition_width` test. So the caller resolves statics via
24//!    [`super::static_ops::StaticOpsTracker`] (stage 5c-ii) BEFORE streaming
25//!    any `Some(ck)` row, without buffering the clustering-row tail.
26//! 3. [`IncrementalPartitionWriter::feed_row`] — called once per clustering
27//!    row, ALREADY in clustering order (the caller's streaming source
28//!    guarantees this — stage 5c-i's schema-aware heap, modulo the stage-5d
29//!    `clustered_rows` fix). Emits any PENDING sorted markers that sort
30//!    strictly before this row (the 5c-iii merge-step logic, driven one row
31//!    at a time instead of via a bulk interleave), then the row itself.
32//! 4. [`IncrementalPartitionWriter::finish`] — emits any remaining markers,
33//!    the `END_OF_PARTITION` byte, closes the last promoted-index block, and
34//!    flushes.
35//!
36//! Promoted-index block tracking (Cassandra `BigFormatPartitionWriter`
37//! parity) is reproduced VERBATIM from `write_partition_with_index_blocks`'s
38//! inline loop body — moved, not duplicated — via
39//! [`IncrementalPartitionWriter::emit_item`], the ONE place that both the
40//! static-row/row/marker emission paths funnel through.
41//!
42//! NOT yet called by any production path: `SSTableWriter::write_partition`
43//! still calls `write_partition_with_index_blocks` (the whole-slice path).
44//! Wiring `maintenance.rs`/`KWayMerger::merge()` to this incremental entry
45//! point is stage 5c-iv, PART 2 — only after this module is proven correct
46//! in isolation (see this module's tests).
47

58/// A single range-tombstone bound marker, sorted once in
59/// [`DataWriter::begin_partition_incremental`] and drained as rows arrive
60/// (issue #1668, stage 5c-iv). A dedicated `Copy` struct — rather than
61/// storing `PartitionItem::Marker` variants directly — so converting a
62/// stored marker into the `PartitionItem` `emit_item` expects is a plain,
63/// always-valid field copy with no other-variant case to (unreachably)
64/// panic on.
65#[derive(Clone, Copy)]
66struct SortableMarker<'r> {
67    bound: &'r ClusteringBound,
68    is_open: bool,
69    deletion_time: i64,
70    local_deletion_time: i32,
71}
72
73impl<'r> SortableMarker<'r> {
74    fn as_partition_item(self) -> PartitionItem<'r> {
75        PartitionItem::Marker {
76            bound: self.bound,
77            is_open: self.is_open,
78            deletion_time: self.deletion_time,
79            local_deletion_time: self.local_deletion_time,
80        }
81    }
82}

84/// Incremental partition-write session (issue #1668, stage 5c-iv).
85///
86/// Holds `&'w mut DataWriter` (so its own methods can call the writer's
87/// existing `pub(super)` emission helpers directly) plus the running,
88/// partition-scoped state `write_partition_with_index_blocks` otherwise
89/// keeps as plain local variables inside its `for item in items` loop.
90pub(crate) struct IncrementalPartitionWriter<'w, 'r> {
91    writer: &'w mut DataWriter,
92    partition_offset: u64,
93    partition_buf_start: usize,
94    prev_unfiltered_size: u64,
95    partition_floor: Option<i64>,
96    range_tombstones: &'r [RangeTombstone],
97    /// The (typically few) range-tombstone markers, sorted ONCE via the
98    /// SAME comparator `sort_partition_items` uses (stage 5c-iii) — drained
99    /// as rows arrive via [`Self::feed_row`], never re-sorted.
100    sorted_markers: Vec<SortableMarker<'r>>,
101    marker_cursor: usize,
102    emit: PartitionEmitCounts,
103    /// `None` until the first post-header/post-static-row item is about to
104    /// be written — matches `write_partition_with_index_blocks`'s
105    /// `block_start_buf_offset`, which excludes the header AND the static
106    /// row from block 0's byte range.
107    block_start_buf_offset: Option<usize>,
108    blocks: Vec<PromotedIndexBlock>,
109    current_block_first_ck: Option<Vec<u8>>,
110    current_block_last_ck: Option<Vec<u8>>,
111    current_block_oss50: Option<Option<Vec<u8>>>,
112}
113
114impl DataWriter {
115    /// Begin an incremental partition write (issue #1668, stage 5c-iv).
116    ///
117    /// Writes the partition header immediately (the caller supplies
118    /// `partition_tombstone` upfront — known before any row arrives, from
119    /// the stage-1 carrier pre-scan on the merge side) and pre-sorts the
120    /// (small) `range_tombstones` list ONCE. Returns a session that must be
121    /// driven through `feed_static_row` (at most once, if the schema has any
122    /// static column), then `feed_row` (one call per clustering row, in
123    /// clustering order), then `finish`.
124    pub(crate) fn begin_partition_incremental<'w, 'r>(
125        &'w mut self,
126        key: &DecoratedKey,
127        partition_tombstone: Option<&PartitionTombstone>,
128        range_tombstones: &'r [RangeTombstone],
129        schema: &TableSchema,
130    ) -> Result<IncrementalPartitionWriter<'w, 'r>> {
131        let partition_offset = self.position + self.buffer.len() as u64;
132        let partition_buf_start = self.buffer.len();
133        let header_start = self.buffer.len();
134        self.write_partition_header(key, partition_tombstone)?;
135        let prev_unfiltered_size = (self.buffer.len() - header_start) as u64;
136        let partition_floor = partition_tombstone.map(|pt| pt.deletion_time);
137
138        let mut sorted_markers: Vec<SortableMarker<'r>> =
139            Vec::with_capacity(range_tombstones.len() * 2);
140        for rt in range_tombstones {
141            sorted_markers.push(SortableMarker {
142                bound: &rt.start,
143                is_open: true,
144                deletion_time: rt.deletion_time,
145                local_deletion_time: rt.local_deletion_time,
146            });
147            sorted_markers.push(SortableMarker {
148                bound: &rt.end,
149                is_open: false,
150                deletion_time: rt.deletion_time,
151                local_deletion_time: rt.local_deletion_time,
152            });
153        }
154        sorted_markers.sort_by(|a, b| {
155            marker_merge::partition_item_cmp(&a.as_partition_item(), &b.as_partition_item(), schema)
156        });
157
158        Ok(IncrementalPartitionWriter {
159            writer: self,
160            partition_offset,
161            partition_buf_start,
162            prev_unfiltered_size,
163            partition_floor,
164            range_tombstones,
165            sorted_markers,
166            marker_cursor: 0,
167            emit: PartitionEmitCounts::default(),
168            block_start_buf_offset: None,
169            blocks: Vec::new(),
170            current_block_first_ck: None,
171            current_block_last_ck: None,
172            current_block_oss50: None,
173        })
174    }
175}

177impl<'w, 'r> IncrementalPartitionWriter<'w, 'r> {
178    /// Write the static-row prelude — called AT MOST ONCE, before any
179    /// `feed_row` call, when the schema declares any static column. `merged`
180    /// is the ALREADY-RESOLVED static ops (e.g. via
181    /// `static_ops::StaticOpsTracker::finish`, stage 5c-ii); pass an empty
182    /// slice for "schema has statics but this partition writes none" (the
183    /// writer still emits the minimal empty form, matching
184    /// `write_partition_with_index_blocks` exactly). `first_mutation_ts` is
185    /// the fallback liveness timestamp used only when `merged` is non-empty
186    /// but carries no derivable liveness (mirrors
187    /// `mutations.first().map(|m| m.timestamp_micros).unwrap_or(0)` in the
188    /// whole-slice path).
189    pub(crate) fn feed_static_row(
190        &mut self,
191        merged: &[StaticMergedOp],
192        first_mutation_ts: i64,
193        schema: &TableSchema,
194    ) -> Result<()> {
195        if merged.is_empty() {
196            let static_size = self.writer.write_empty_static_row(0, schema)? as u64;
197            self.prev_unfiltered_size += static_size;
198        } else {
199            let (latest_ts, ttl) =
200                static_liveness_from_ops(merged).unwrap_or((first_mutation_ts, None));
201            let (static_size, static_cells) = self
202                .writer
203                .write_static_row_with_prev_size(merged, latest_ts, ttl, schema, 0)?;
204            self.prev_unfiltered_size += static_size as u64;
205            self.emit.rows += 1;
206            self.emit.columns += static_cells;
207        }
208        Ok(())
209    }
210
211    /// Feed one clustering-key group's mutation (issue #1668, stage 5c-iv).
212    ///
213    /// For compaction, a "cluster group" is already a single, fully
214    /// reconciled `Mutation` (the merge layer collapses to one entry per
215    /// distinct clustering key), so this treats `mutation` as its own
216    /// singleton group via `merge_row_group(&[mutation], ..)` — the SAME
217    /// function `merge_clustering_rows` calls per adjacency-group in the
218    /// whole-slice path. Emits any PENDING sorted markers that sort strictly
219    /// before this row first, then the row itself (skipping emission
220    /// entirely if the row is fully shadowed — matches
221    /// `merge_clustering_rows`'s `if let Some(row) = ...` skip).
222    pub(crate) fn feed_row(&mut self, mutation: &Mutation, schema: &TableSchema) -> Result<()> {
223        let clustering_key = mutation.clustering_key.as_ref();
224        let mut shadow_floor = self.partition_floor;
225        for rt in self.range_tombstones {
226            if range_tombstone_covers(rt, clustering_key, schema) {
227                shadow_floor =
228                    Some(shadow_floor.map_or(rt.deletion_time, |f| f.max(rt.deletion_time)));
229            }
230        }
231
232        // Emit every pending marker that sorts strictly before this row
233        // (issue #1220 tie-break: a Row and a Marker never compare Equal at
234        // the same clustering value — see marker_merge's module doc — so
235        // "strictly before" is unambiguous here).
236        let row_item = PartitionItem::Row(RowWrite {
237            clustering_key,
238            liveness_ts: None,
239            ttl_seconds: None,
240            row_deletion: None,
241            ops: Vec::new(),
242            complex_element_ops: Vec::new(),
243        });
244        while let Some(marker) = self.sorted_markers.get(self.marker_cursor).copied() {
245            if marker_merge::partition_item_cmp(&marker.as_partition_item(), &row_item, schema)
246                == std::cmp::Ordering::Greater
247            {
248                break;
249            }
250            self.emit_next_marker_or_boundary(marker, schema)?;
251        }
252
253        if let Some(row) = DataWriter::merge_row_group(&[mutation], schema, false, shadow_floor) {
254            self.emit_item(PartitionItem::Row(row), schema)?;
255        }
256        Ok(())
257    }
258
259    /// Emit `marker` (the item at `self.marker_cursor`), coalescing it with
260    /// its immediately-following pair into a single `PartitionItem::Boundary`
261    /// when they form one (issue #1220 / roborev blocker #1 on #1668):
262    /// `write_partition`/`write_partition_with_index_blocks` run every
263    /// emitted marker through `coalesce_boundaries`, but draining
264    /// `sorted_markers` one at a time (instead of coalescing the whole
265    /// pre-sorted `Vec` up front) skipped that step entirely — two adjacent
266    /// range tombstones with different deletion times would otherwise be
267    /// persisted as two separate bound markers instead of Cassandra's single
268    /// boundary marker.
269    ///
270    /// Safe to decide using ONLY `marker`'s own already-known sort position
271    /// (the caller's row-test in [`Self::feed_row`], or unconditionally in
272    /// [`Self::finish`]): a coalescible close+open pair always shares the
273    /// EXACT SAME clustering point with EQUAL sort weight (see
274    /// `marker_merge::sort_class`), so both bounds always compare identically
275    /// against any given row — advancing past both here can never skip past
276    /// a row that should have been emitted first.
277    fn emit_next_marker_or_boundary(
278        &mut self,
279        marker: SortableMarker<'r>,
280        schema: &TableSchema,
281    ) -> Result<()> {
282        if !marker.is_open {
283            if let Some(next) = self.sorted_markers.get(self.marker_cursor + 1).copied() {
284                if next.is_open {
285                    if let Some((kind, clustering)) =
286                        partition::boundary_kind_for(marker.bound, next.bound, schema)
287                    {
288                        self.marker_cursor += 2;
289                        return self.emit_item(
290                            PartitionItem::Boundary {
291                                kind,
292                                clustering,
293                                end_deletion_time: marker.deletion_time,
294                                end_local_deletion_time: marker.local_deletion_time,
295                                start_deletion_time: next.deletion_time,
296                                start_local_deletion_time: next.local_deletion_time,
297                            },
298                            schema,
299                        );
300                    }
301                }
302            }
303        }
304        self.marker_cursor += 1;
305        self.emit_item(marker.as_partition_item(), schema)
306    }

308    /// Finalize the partition: emit any remaining markers, `END_OF_PARTITION`,
309    /// close the last promoted-index block, and flush.
310    pub(crate) fn finish(
311        mut self,
312        schema: &TableSchema,
313    ) -> Result<(u64, Vec<PromotedIndexBlock>, PartitionEmitCounts)> {
314        while self.marker_cursor < self.sorted_markers.len() {
315            let marker = self.sorted_markers[self.marker_cursor];
316            self.emit_next_marker_or_boundary(marker, schema)?;
317        }
318
319        self.writer.buffer.push(END_OF_PARTITION);
320
321        if let (Some(first), Some(last)) = (
322            self.current_block_first_ck.take(),
323            self.current_block_last_ck.take(),
324        ) {
325            let current_buf_offset = self.writer.buffer.len() - self.partition_buf_start;
326            let block_start = self.block_start_buf_offset.unwrap_or(current_buf_offset);
327            let block_bytes = (current_buf_offset - block_start) as u64;
328            if block_bytes > 0 {
329                self.blocks.push(PromotedIndexBlock {
330                    first_name: first,
331                    last_name: last,
332                    offset: block_start as u64,
333                    width: block_bytes,
334                    oss50_separator: self.current_block_oss50.take().flatten(),
335                });
336            }
337        }
338
339        self.writer.flush_partition()?;
340        Ok((self.partition_offset, self.blocks, self.emit))
341    }

343    /// Write one item (row or marker) and update promoted-index-block
344    /// tracking — moved VERBATIM (not duplicated) from
345    /// `write_partition_with_index_blocks`'s `for item in items { .. }` loop
346    /// body, the ONE place both the row and marker emission paths funnel
347    /// through.
348    ///
349    /// Generic over its OWN lifetime `'a`, independent of the struct's `'r`
350    /// (which is `range_tombstones`' lifetime): a `Row` item borrows from
351    /// the CURRENT `feed_row` call's `mutation` argument, an entirely
352    /// different (and shorter-lived) object than `range_tombstones` — the
353    /// item is fully consumed (its bytes written) before this call returns,
354    /// so it never needs to outlive that.
355    fn emit_item<'a>(&mut self, item: PartitionItem<'a>, schema: &TableSchema) -> Result<()> {
356        let ck_bytes: Vec<u8> = match &item {
357            PartitionItem::Row(row) => {
358                if let Some(ck) = row.clustering_key {
359                    serialize_clustering_prefix_for_index(ck, schema)
360                        .unwrap_or_else(|_| empty_clustering_prefix_for_index())
361                } else {
362                    empty_clustering_prefix_for_index()
363                }
364            }
365            PartitionItem::Marker { bound, is_open, .. } => {
366                serialize_marker_bound_prefix_for_index(bound, *is_open, schema)
367                    .unwrap_or_else(|_| marker_bound_prefix_for_index(bound, *is_open))
368            }
369            PartitionItem::Boundary {
370                kind, clustering, ..
371            } => serialize_boundary_prefix_for_index(*kind, clustering, schema)
372                .unwrap_or_else(|_| boundary_prefix_for_index_fallback(*kind)),
373        };
374
375        let partition_buf_start = self.partition_buf_start;
376        if self.current_block_first_ck.is_none() {
377            self.current_block_first_ck = Some(ck_bytes.clone());
378            let ck_values: Option<Vec<Value>> = match &item {
379                PartitionItem::Row(row) => row
380                    .clustering_key
381                    .filter(|ck| !ck.columns.is_empty())
382                    .map(|ck| ck.columns.iter().map(|(_, v)| v.clone()).collect()),
383                PartitionItem::Marker { .. } | PartitionItem::Boundary { .. } => None,
384            };
385            let is_reversed: Vec<bool> = schema
386                .clustering_keys
387                .iter()
388                .map(|c| c.order == crate::schema::ClusteringOrder::Desc)
389                .collect();
390            self.current_block_oss50 = Some(ck_values.and_then(|vals| {
391                crate::storage::sstable::bti::encode_clustering_bound_oss50_with_order(
392                    &vals,
393                    &is_reversed,
394                )
395                .ok()
396            }));
397            // Lazily anchor the FIRST block's start now, at the first
398            // post-header/post-static-row item — matches
399            // `write_partition_with_index_blocks`'s `block_start_buf_offset`
400            // initialization point exactly.
401            self.block_start_buf_offset
402                .get_or_insert(self.writer.buffer.len() - partition_buf_start);
403        }
404        self.current_block_last_ck = Some(ck_bytes.clone());
405
406        self.prev_unfiltered_size = match item {
407            PartitionItem::Row(row) => {
408                self.emit.rows += 1;
409                let (bytes, cells) = self.writer.write_merged_row_with_prev_size(
410                    &row,
411                    schema,
412                    self.prev_unfiltered_size,
413                )?;
414                self.emit.columns += cells;
415                bytes as u64
416            }
417            PartitionItem::Marker {
418                bound,
419                is_open,
420                deletion_time,
421                local_deletion_time,
422            } => self.writer.write_range_bound(
423                bound,
424                is_open,
425                deletion_time,
426                local_deletion_time,
427                schema,
428                self.prev_unfiltered_size,
429            )? as u64,
430            PartitionItem::Boundary {
431                kind,
432                clustering,
433                end_deletion_time,
434                end_local_deletion_time,
435                start_deletion_time,
436                start_local_deletion_time,
437            } => self.writer.write_range_boundary(
438                kind,
439                clustering,
440                end_deletion_time,
441                end_local_deletion_time,
442                start_deletion_time,
443                start_local_deletion_time,
444                schema,
445                self.prev_unfiltered_size,
446            )? as u64,
447        };
448
449        let current_buf_offset = self.writer.buffer.len() - partition_buf_start;
450        let block_start = self.block_start_buf_offset.unwrap_or(current_buf_offset);
451        let block_bytes = (current_buf_offset - block_start) as u64;
452
453        if block_bytes >= COLUMN_INDEX_SIZE_BYTES {
454            self.blocks.push(PromotedIndexBlock {
455                first_name: self
456                    .current_block_first_ck
457                    .take()
458                    .unwrap_or_else(empty_clustering_prefix_for_index),
459                last_name: self
460                    .current_block_last_ck
461                    .take()
462                    .unwrap_or_else(empty_clustering_prefix_for_index),
463                offset: block_start as u64,
464                width: block_bytes,
465                oss50_separator: self.current_block_oss50.take().flatten(),
466            });
467            self.block_start_buf_offset = Some(current_buf_offset);
468            self.current_block_first_ck = None;
469            self.current_block_last_ck = None;
470            self.current_block_oss50 = None;
471        }
472        Ok(())
473    }
