1//! Statistics.db writer - writes SSTable metadata
2//!
3//! Generates the Statistics.db component with min/max timestamps, TTL, and
4//! other metadata used for delta encoding in Data.db.
5//!
6//! Critical requirements:
7//! - MUST be written BEFORE Data.db (provides delta encoding baseline)
8//! - Min timestamp, max timestamp
9//! - Min TTL, max TTL
10//! - Min local deletion time, max local deletion time
11//! - Partition count, row count
12//!
13//! # Statistics.db Format (Cassandra 5.0 Compatible)
14//!
15//! This implementation produces a full Cassandra 5.0 nb-format Statistics.db with:
16//! - TOC (Table of Contents) with checksums
17//! - Four metadata components: VALIDATION, COMPACTION, STATS, HEADER
18//! - Per-component CRC32 checksums
19//! - Global CRC32 checksum validation
20//!
21//! ## Format Structure
22//!
23//! ```text
24//! [0-3]   num_components (u32 BE) = 4
25//! [4-7]   CRC32(num_components)
26//! [8-39]  TOC entries (4 components × 8 bytes each):
27//!           [u32 BE] component_type (MetadataType ordinal)
28//!           [u32 BE] component_offset
29//! [40-43] CRC32(num_components + all TOC entries) [cumulative]
30//! [44+]   Component data:
31//!           [N bytes] component_data
32//!           [4 bytes] CRC32(component_data)
33//!           ... (repeated for each component)
34//! ```
35//!
36//! ## MetadataType Component IDs
37//!
38//! From Cassandra's `MetadataType.java` enum (ordinal values):
39//! - 0: VALIDATION (validator class name)
40//! - 1: COMPACTION (compaction metadata)
41//! - 2: STATS (statistics including EncodingStats)
42//! - 3: SERIALIZATION_HEADER (table schema)
43//!
44//! # Module layout
45//!
46//! This writer is split by responsibility (epic #1116):
47//! - [`metadata`]: `StatisticsMetadata` aggregation + `TombstoneHistogram`
48//! - [`marshal`]: CQL type name → Cassandra marshal-type conversion
49//! - [`components`]: VALIDATION / COMPACTION / STATS component builders
50//! - [`serialization_header`]: the SERIALIZATION_HEADER builder
51//! - this module: `StatisticsWriter` orchestration (TOC + `write`)
52

70/// Epoch constants for EncodingStats (from Cassandra's EncodingStats.java)
71/// These are used to compute deltas from a baseline for more compact encoding
72const TIMESTAMP_EPOCH: i64 = 1442880000000000; // Sept 22, 2015 00:00:00 UTC in microseconds
73const DELETION_TIME_EPOCH: i32 = 1442880000; // Sept 22, 2015 00:00:00 UTC in seconds
74const TTL_EPOCH: i32 = 0; // TTL epoch is 0 (no offset)
75
76/// Number of metadata components in Statistics.db
77/// Cassandra 5.0 nb-format has 4 components: VALIDATION, COMPACTION, STATS, HEADER
78const NUM_COMPONENTS: u32 = 4;
79
80/// MetadataType ordinal values (from Cassandra's MetadataType.java enum)
81const METADATA_TYPE_VALIDATION: u32 = 0;
82const METADATA_TYPE_COMPACTION: u32 = 1;
83const METADATA_TYPE_STATS: u32 = 2;
84const METADATA_TYPE_SERIALIZATION_HEADER: u32 = 3;
85
86/// Statistics.db component writer
87///
88/// Writes the Statistics.db file with metadata for SSTable delta encoding.
89#[derive(Debug)]
90pub struct StatisticsWriter {
91    /// Path to the Statistics.db file to write
92    path: PathBuf,
93    /// Emit the Cassandra-canonical `da` (BtiFormat) `StatsMetadata` layout
94    /// instead of the legacy `nb` layout.
95    ///
96    /// The `da` STATS component differs from `nb` in the fields gated by the
97    /// BtiFormat version flags (all true for `da` except `hasLegacyMinMax`):
98    /// `hasUIntDeletionTime`, `hasImprovedMinMax` (clusteringTypes + a covered
99    /// `Slice` instead of legacy min/max value lists), `hasIsTransient`,
100    /// `hasOriginatingHostId`, `hasPartitionLevelDeletionsPresenceMarker`,
101    /// `hasKeyRange` (first/last key) and `hasTokenSpaceCoverage`. Cassandra's
102    /// `sstabledump`/`sstablemetadata` deserialize a `da`-descriptor
103    /// Statistics.db with this layout and reject the `nb` layout (a
104    /// `Slice.<init>` assertion fires while reading `coveredClustering`).
105    /// Authority: cassandra-5.0.0 `StatsMetadata.StatsMetadataSerializer` +
106    /// `BtiFormat.BtiVersion` version flags.
107    bti: bool,
108}
109
110impl StatisticsWriter {
111    /// Create a new Statistics.db writer for the legacy `nb`/`oa` BIG layout.
112    ///
113    /// # Arguments
114    /// * `path` - Path where Statistics.db will be written
115    pub fn new(path: PathBuf) -> Self {
116        Self { path, bti: false }
117    }
118
119    /// Create a new Statistics.db writer that emits the Cassandra-canonical `da`
120    /// (BtiFormat) `StatsMetadata` layout.
121    pub fn new_bti(path: PathBuf) -> Self {
122        Self { path, bti: true }
123    }
124
125    /// Write Statistics.db file with the given metadata
126    ///
127    /// Generates a Cassandra 5.0 compatible Statistics.db file with full TOC structure:
128    /// 1. TOC header with component count and checksums
129    /// 2. VALIDATION component (validator class name)
130    /// 3. COMPACTION component (minimal metadata)
131    /// 4. STATS component (EncodingStats with baselines)
132    /// 5. SERIALIZATION_HEADER component (schema-derived or minimal stub)
133    ///
134    /// Each component is followed by a CRC32 checksum for validation.
135    ///
136    /// # Arguments
137    /// * `metadata` - Statistics metadata to write
138    /// * `schema` - Optional table schema for populating serialization header
139    ///
140    /// # Returns
141    /// `Ok(())` on success, or an error if writing fails
142    pub fn write(&self, metadata: &StatisticsMetadata, schema: Option<&TableSchema>) -> Result<()> {
143        let mut meta = metadata.clone();
144        meta.finalize();
145
146        // Build component data
147        let validation_data = self.build_validation_component()?;
148        let compaction_data = self.build_compaction_component()?;
149        // The STATS body is version-gated: BTI (`da`) emits the BtiFormat
150        // `StatsMetadata` layout (covered-clustering Slice, uint deletion times,
151        // key range, token-space coverage); BIG (`nb`/`oa`) emits the legacy
152        // layout. `schema` is needed for the `da` clustering-type list.
153        let stats_data = if self.bti {
154            self.build_stats_component_da(&meta, schema)?
155        } else {
156            self.build_stats_component(&meta)?
157        };
158        // Use pre-finalize metadata for the SerializationHeader EncodingStats.
159        // The baselines in the header MUST match those used by the DataWriter for
160        // delta encoding. The DataWriter uses the raw (pre-finalize) metadata values.
161        let header_data = self.build_serialization_header_component(schema, metadata)?;
162
163        // Calculate component offsets
164        // TOC structure: 4 (count) + 4 (checksum) + (4*8) TOC entries + 4 (checksum) = 44 bytes
165        let toc_size = 4 + 4 + (NUM_COMPONENTS as usize * 8) + 4;
166        let mut offset = toc_size;
167
168        let validation_offset = offset;
169        offset += validation_data.len() + 4; // +4 for component checksum
170
171        let compaction_offset = offset;
172        offset += compaction_data.len() + 4;
173
174        let stats_offset = offset;
175        offset += stats_data.len() + 4;
176
177        let header_offset = offset;
178        // header_data has its own checksum at the end
179
180        // Verify all offsets fit in u32 (Statistics.db should never exceed 4GB)
181        if offset > u32::MAX as usize {
182            return Err(Error::Storage(format!(
183                "Statistics.db too large: {} bytes exceeds u32::MAX",
184                offset
185            )));
186        }
187
188        // Build the complete file
189        let mut buffer = Vec::new();
190        let mut crc = crc32fast::Hasher::new();
191
192        // Write component count
193        buffer.write_all(&NUM_COMPONENTS.to_be_bytes())?;
194        self.update_checksum_int(&mut crc, NUM_COMPONENTS);
195
196        // Write first checksum (after count)
197        let checksum1 = crc.clone().finalize();
198        buffer.write_all(&checksum1.to_be_bytes())?;
199
200        // Reset CRC for TOC (we'll recompute cumulatively)
201        crc = crc32fast::Hasher::new();
202        self.update_checksum_int(&mut crc, NUM_COMPONENTS);
203
204        // Write TOC entries (type, offset pairs)
205        self.write_toc_entry(
206            &mut buffer,
207            &mut crc,
208            METADATA_TYPE_VALIDATION,
209            validation_offset as u32,
210        )?;
211        self.write_toc_entry(
212            &mut buffer,
213            &mut crc,
214            METADATA_TYPE_COMPACTION,
215            compaction_offset as u32,
216        )?;
217        self.write_toc_entry(
218            &mut buffer,
219            &mut crc,
220            METADATA_TYPE_STATS,
221            stats_offset as u32,
222        )?;
223        self.write_toc_entry(
224            &mut buffer,
225            &mut crc,
226            METADATA_TYPE_SERIALIZATION_HEADER,
227            header_offset as u32,
228        )?;
229
230        // Write TOC checksum (cumulative from count)
231        let toc_checksum = crc.finalize();
232        buffer.write_all(&toc_checksum.to_be_bytes())?;
233
234        // Write components with per-component checksums
235        self.write_component(&mut buffer, &validation_data)?;
236        self.write_component(&mut buffer, &compaction_data)?;
237        self.write_component(&mut buffer, &stats_data)?;
238        self.write_component(&mut buffer, &header_data)?;
239
240        // Write to file
241        std::fs::write(&self.path, buffer).map_err(|e| {
242            Error::Storage(format!(
243                "Failed to write Statistics.db to {}: {}",
244                self.path.display(),
245                e
246            ))
247        })?;
248
249        Ok(())
250    }
251
252    /// Update CRC32 checksum with a u32 value (big-endian)
253    ///
254    /// Mimics Java's FBUtilities.updateChecksumInt()
255    fn update_checksum_int(&self, crc: &mut crc32fast::Hasher, value: u32) {
256        crc.update(&value.to_be_bytes());
257    }
258
259    /// Write a TOC entry (component type and offset) with cumulative CRC update
260    fn write_toc_entry(
261        &self,
262        buffer: &mut Vec<u8>,
263        crc: &mut crc32fast::Hasher,
264        component_type: u32,
265        offset: u32,
266    ) -> Result<()> {
267        buffer.write_all(&component_type.to_be_bytes())?;
268        self.update_checksum_int(crc, component_type);
269
270        buffer.write_all(&offset.to_be_bytes())?;
271        self.update_checksum_int(crc, offset);
272
273        Ok(())
274    }
275
276    /// Write a component with its CRC32 checksum
277    fn write_component(&self, buffer: &mut Vec<u8>, data: &[u8]) -> Result<()> {
278        // Write component data
279        buffer.write_all(data)?;
280
281        // Write component checksum
282        let checksum = crc32fast::hash(data);
283        buffer.write_all(&checksum.to_be_bytes())?;
284
285        Ok(())
286    }
287}

301    /// Issue #1327: a WriteEngine-produced `Statistics.db` MUST carry a populated
302    /// `estimatedPartitionSize` `EstimatedHistogram` so the authoritative
303    /// `read_table_counts` decode (which sums that histogram's bucket counts)
304    /// returns the true partition count — not 0.
305    ///
306    /// This test fed the histogram builder from per-partition sizes exactly as
307    /// the write path does, wrote the STATS component, and read it back. Before
308    /// the fix the writer emitted a hardcoded all-zero-count histogram, so this
309    /// assertion FAILED (`partition_count == 0`).
310    #[test]
311    fn write_engine_statistics_partition_count_round_trips() {
312        let temp_dir = TempDir::new().unwrap();
313        let stats_path = temp_dir.path().join("nb-1-big-Statistics.db");
314        let writer = StatisticsWriter::new(stats_path.clone());
315
316        let mut meta = StatisticsMetadata::new();
317        // Record five partitions with distinct sizes, exactly as the write path
318        // does after each `write_partition` (authoritative per-partition size +
319        // cell count — no heuristics).
320        let partitions: [(u64, u64); 5] = [(42, 3), (256, 8), (1024, 12), (17, 1), (900, 7)];
321        for (size, cells) in partitions {
322            meta.record_partition(size, cells);
323        }
324        meta.finalize();
325        // `record_partition` populates the histograms; the authoritative
326        // partition_count is derived by the reader from Σ histogram counts, NOT
327        // from the `partition_count` field (which `increment_partition_count`
328        // maintains separately on the live write path). So we assert the DECODED
329        // count below rather than the in-memory field.
330        assert_eq!(meta.estimated_partition_size.total(), 5);
331
332        writer.write(&meta, None).expect("write should succeed");
333
334        let file_data = std::fs::read(&stats_path).expect("file should exist");
335        let counts = read_table_counts(&file_data, Some(&nb_gates())).expect("decode counts");
336        assert_eq!(
337            counts.partition_count, 5,
338            "estimatedPartitionSize histogram bucket-count sum must equal partitions written"
339        );
340        assert_ne!(
341            counts.partition_count, 0,
342            "issue #1327: write-engine Statistics.db must not report partition_count=0"
343        );
344    }

485    #[test]
486    fn test_component_checksums() {
487        let temp_dir = TempDir::new().unwrap();
488        let stats_path = temp_dir.path().join("test-Statistics.db");
489
490        let writer = StatisticsWriter::new(stats_path.clone());
491
492        let mut meta = StatisticsMetadata::new();
493        meta.min_timestamp = 1000000;
494        meta.partition_count = 100;
495
496        writer.write(&meta, None).unwrap();
497
498        // Read file and verify per-component checksums
499        let file_data = std::fs::read(&stats_path).unwrap();
500
501        // Parse TOC to get component offsets
502        let num_components =
503            u32::from_be_bytes([file_data[0], file_data[1], file_data[2], file_data[3]]);
504        assert_eq!(num_components, 4);
505
506        let mut component_offsets = Vec::new();
507        for i in 0..num_components {
508            let offset = 8 + (i as usize * 8) + 4; // +4 to skip type, get offset
509            let comp_offset = u32::from_be_bytes([
510                file_data[offset],
511                file_data[offset + 1],
512                file_data[offset + 2],
513                file_data[offset + 3],
514            ]);
515            component_offsets.push(comp_offset as usize);
516        }
517
518        // Verify each component's checksum
519        for i in 0..num_components as usize {
520            let comp_start = component_offsets[i];
521
522            // Calculate component length
523            let comp_end = if i < component_offsets.len() - 1 {
524                component_offsets[i + 1]
525            } else {
526                file_data.len()
527            };
528
529            // Component data ends 4 bytes before next component (for checksum)
530            let comp_length = comp_end - comp_start - 4;
531            let component_data = &file_data[comp_start..comp_start + comp_length];
532
533            // Read stored checksum
534            let stored_checksum = u32::from_be_bytes([
535                file_data[comp_start + comp_length],
536                file_data[comp_start + comp_length + 1],
537                file_data[comp_start + comp_length + 2],
538                file_data[comp_start + comp_length + 3],
539            ]);
540
541            // Compute expected checksum
542            let computed_checksum = crc32fast::hash(component_data);
543
544            assert_eq!(
545                stored_checksum, computed_checksum,
546                "Component {} checksum mismatch",
547                i
548            );
549        }
550    }

609        // First 4 bytes should be i32 BE length of HLL data
610        let hll_len = i32::from_be_bytes([
611            file_data[compaction_offset],
612            file_data[compaction_offset + 1],
613            file_data[compaction_offset + 2],
614            file_data[compaction_offset + 3],
615        ]);
616        assert_eq!(hll_len, 15, "HLL data length should be 15 bytes");
617
618        // Verify HLL version marker (next 4 bytes should be -2 = 0xFFFFFFFE)
619        let hll_version = i32::from_be_bytes([
620            file_data[compaction_offset + 4],
621            file_data[compaction_offset + 5],
622            file_data[compaction_offset + 6],
623            file_data[compaction_offset + 7],
624        ]);
625        assert_eq!(hll_version, -2, "HLL version should be -2");
626
627        // Verify STATS component has correct total size (+ 4 byte checksum).
628        // Issue #1327: the two leading EstimatedHistograms now use their DISTINCT
629        // Cassandra shapes — estimatedPartitionSize is EH(155) => 156 buckets
630        // (4 + 156*16 = 2500 bytes) and estimatedCellPerPartitionCount is EH(118)
631        // => 119 buckets (4 + 119*16 = 1908 bytes) — instead of the old empty
632        // 36-byte stubs. So the STATS body grew by (2500-36)+(1908-36) = 4336.
633        let partition_size_bytes = 4 + 156 * 16; // 2500
634        let cell_count_bytes = 4 + 119 * 16; // 1908 (EH(118) — distinct shape)
635        let two_histograms = partition_size_bytes + cell_count_bytes;
636        let expected_stats_size = 188 - 72 + two_histograms; // old 72 = 2*36 empty stubs
637        let stats_end = header_offset;
638        let stats_size = stats_end - stats_offset - 4; // -4 for checksum
639        assert_eq!(
640            stats_size, expected_stats_size,
641            "STATS component should be {expected_stats_size} bytes"
642        );
643
644        // Verify min_timestamp in STATS component. It follows the two histograms
645        // and the 12-byte commitLogUpperBound: two_histograms + 12 from stats_offset.
646        let ts_offset = stats_offset + two_histograms + 12;
647        let min_ts = i64::from_be_bytes([
648            file_data[ts_offset],
649            file_data[ts_offset + 1],
650            file_data[ts_offset + 2],
651            file_data[ts_offset + 3],
652            file_data[ts_offset + 4],
653            file_data[ts_offset + 5],
654            file_data[ts_offset + 6],
655            file_data[ts_offset + 7],
656        ]);
657        assert_eq!(min_ts, TIMESTAMP_EPOCH, "Min timestamp should be preserved");
658
659        // Verify SERIALIZATION_HEADER component
660        // Should start with 3 unsigned VInts for EncodingStats deltas.
661        // All metadata values == their epoch baselines, so all deltas are 0.
662        // encode_vuint(0) = [0x00].
663        assert_eq!(
664            file_data[header_offset], 0x00,
665            "EncodingStats minTimestamp delta should be 0"
666        );
667        assert_eq!(
668            file_data[header_offset + 1],
669            0x00,
670            "EncodingStats minLocalDeletionTime delta should be 0"
671        );
672        assert_eq!(
673            file_data[header_offset + 2],
674            0x00,
675            "EncodingStats minTTL delta should be 0"
676        );
677    }

679    /// Verify that Statistics.db written for a table WITH tombstones produces a
680    /// non-empty `estimatedTombstoneDropTime` histogram in the STATS component.
681    ///
682    /// This is the primary acceptance-criterion test for issue #730.
683    #[test]
684    fn test_statistics_db_tombstone_histogram_nonempty() {
685        let temp_dir = TempDir::new().unwrap();
686        let stats_path = temp_dir.path().join("tombstone-Statistics.db");
687        let writer = StatisticsWriter::new(stats_path.clone());
688
689        // Simulate two tombstone local-deletion-times
690        let ldt1 = 1_700_000_000i32;
691        let ldt2 = 1_700_100_000i32;
692
693        let mut meta = StatisticsMetadata::new();
694        meta.update_timestamp(1_600_000_000_000_000); // some live write
695        meta.update_local_deletion_time(ldt1);
696        meta.update_local_deletion_time(ldt2);
697        meta.partition_count = 1;
698        meta.row_count = 2;
699        meta.column_count = 4;
700
701        writer.write(&meta, None).expect("write should succeed");
702
703        // ---- Parse the resulting Statistics.db and find the histogram ----
704        let file_data = std::fs::read(&stats_path).expect("file should exist");
705
706        // The STATS component offset is stored in the TOC at bytes 28–31
707        // (3rd entry, 4-byte type + 4-byte offset = 8 bytes each, starting at byte 8;
708        //  entries: [8..16] VALIDATION, [16..24] COMPACTION, [24..32] STATS, [32..40] HEADER)
709        let stats_offset =
710            u32::from_be_bytes([file_data[28], file_data[29], file_data[30], file_data[31]])
711                as usize;
712
713        // Within the STATS component, the tombstone-histogram field starts after:
714        //   estimatedPartitionSize  = 4 + 156*16 = 2500 bytes (EH(155), issue #1327)
715        //   estimatedCellPerPartitionCount = 4 + 119*16 = 1908 bytes (EH(118),
716        //                             distinct Cassandra shape, issue #1327)
717        //   CommitLogPosition upper = 12 bytes
718        //   minTimestamp (i64)      =  8 bytes
719        //   maxTimestamp (i64)      =  8 bytes
720        //   minLocalDeletionTime    =  4 bytes
721        //   maxLocalDeletionTime    =  4 bytes
722        //   minTTL                  =  4 bytes
723        //   maxTTL                  =  4 bytes
724        //   compressionRatio (f64)  =  8 bytes
725        let two_histograms = (4 + 156 * 16) + (4 + 119 * 16);
726        let histogram_offset = stats_offset + two_histograms + 12 + 8 + 8 + 4 + 4 + 4 + 4 + 8;
727
728        // Read maxBinSize and size
729        let max_bin_size = i32::from_be_bytes(
730            file_data[histogram_offset..histogram_offset + 4]
731                .try_into()
732                .expect("histogram maxBinSize bytes"),
733        );
734        let histo_size = i32::from_be_bytes(
735            file_data[histogram_offset + 4..histogram_offset + 8]
736                .try_into()
737                .expect("histogram size bytes"),
738        );
739
740        assert_eq!(
741            max_bin_size,
742            TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE,
743            "estimatedTombstoneDropTime maxBinSize should be {} for a non-empty histogram (issue #730)",
744            TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE
745        );
746        assert_eq!(
747            histo_size, 2,
748            "estimatedTombstoneDropTime should have 2 bins for 2 distinct deletion times (issue #730)"
749        );
750
751        // Verify the first bin's point matches ldt1
752        let point0 = f64::from_be_bytes(
753            file_data[histogram_offset + 8..histogram_offset + 16]
754                .try_into()
755                .expect("bin0 point bytes"),
756        );
757        assert_eq!(
758            point0, ldt1 as f64,
759            "first histogram bin point should match ldt1"
760        );
761
762        // Verify the second bin's point matches ldt2
763        let point1 = f64::from_be_bytes(
764            file_data[histogram_offset + 24..histogram_offset + 32]
765                .try_into()
766                .expect("bin1 point bytes"),
767        );
768        assert_eq!(
769            point1, ldt2 as f64,
770            "second histogram bin point should match ldt2"
771        );
772
773        // Verify both bins have count = 1
774        let value0 = i64::from_be_bytes(
775            file_data[histogram_offset + 16..histogram_offset + 24]
776                .try_into()
777                .expect("bin0 value bytes"),
778        );
779        let value1 = i64::from_be_bytes(
780            file_data[histogram_offset + 32..histogram_offset + 40]
781                .try_into()
782                .expect("bin1 value bytes"),
783        );
784        assert_eq!(value0, 1, "first bin count should be 1");
785        assert_eq!(value1, 1, "second bin count should be 1");
786    }
