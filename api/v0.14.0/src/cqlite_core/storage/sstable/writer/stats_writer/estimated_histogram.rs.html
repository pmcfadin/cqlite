1//! Cassandra-canonical `EstimatedHistogram` accumulator + serialiser.
2//!
3//! Populates the STATS-component `estimatedPartitionSize` and
4//! `estimatedCellPerPartitionCount` histograms so a write-produced
5//! `Statistics.db` carries the same authoritative per-table distribution
6//! Cassandra writes (issue #1327). The read-side authoritative partition-count
7//! decode (`read_table_counts`, issue #944) sums the `estimatedPartitionSize`
8//! bucket counts; before this each histogram was empty, so it decoded `0`.
9//!
10//! # Two distinct default shapes (issue #1327 finding 1)
11//!
12//! Cassandra seeds the two STATS histograms with DIFFERENT bucket counts, not
13//! one shared shape (this was previously wrong in CQLite — the cell-count
14//! histogram reused the partition-size shape). Authority:
15//! cassandra-5.0.0 `io/sstable/metadata/MetadataCollector.java`:
16//!
17//! ```text
18//! static EstimatedHistogram defaultCellPerPartitionCountHistogram()
19//! {
20//!     // EH of 118 can track a max value of 4139110981, i.e., > 4B cells
21//!     return new EstimatedHistogram(118);
22//! }
23//! static EstimatedHistogram defaultPartitionSizeHistogram()
24//! {
25//!     // EH of 155 can track a max value of 3520571548412 i.e. 3.5TB
26//!     return new EstimatedHistogram(155);
27//! }
28//! ```
29//!
30//! `new EstimatedHistogram(size)` builds `size` bucket OFFSETS via
31//! `newOffsets(size, considerZeroes=false)` and a `bucketOffsets.length + 1`
32//! bucket-count array (the trailing overflow bucket). So:
33//!
34//! | Histogram                        | offsets | serialised buckets |
35//! |----------------------------------|---------|--------------------|
36//! | `estimatedPartitionSize`         | 155     | 156                |
37//! | `estimatedCellPerPartitionCount` | 118     | 119                |
38//!
39//! # Offset series (matches `EstimatedHistogram.newOffsets`)
40//!
41//! `newOffsets(size, false)` (cassandra-5.0.0 `utils/EstimatedHistogram.java`)
42//! produces a STRICTLY-monotonic geometric series with NO leading duplicate in
43//! the offsets array itself:
44//!
45//! ```text
46//! long last = 1;
47//! result[0] = last;                       // = 1
48//! for (i = 1; i < size; i++) {
49//!     long next = Math.round(last * 1.2);
50//!     if (next == last) next++;
51//!     result[i] = next; last = next;
52//! }
53//! ```
54//!
55//! yielding `[1, 2, 3, 4, 5, 6, 7, 8, ...]`. The two documented max offsets are
56//! reproduced exactly (offsets[154] == 3_520_571_548_412 and
57//! offsets[117] == 4_139_110_981), which pins the series + bucket counts against
58//! the Cassandra source comments.
59//!
60//! # Serialised leading duplicate is a SERIALISER artefact, not an offset
61//!
62//! The `[1, 1, 2, 3, ...]` leading duplicate observed at the head of the STATS
63//! component in the committed annotated fixture
64//! (`docs/sstables-definitive-guide/statistics-db-annotated-dump.txt`, STATS at
65//! 0x0b53) is produced by `EstimatedHistogramSerializer.serialize`, which emits
66//! `offsets[i == 0 ? 0 : i - 1]` for each of the `buckets.length` buckets — so
67//! `offsets[0]` is written TWICE (for bucket index 0 and 1). The offsets array
68//! itself is `[1, 2, 3, ...]`. CQLite therefore stores the pure series and
69//! reproduces the duplicate at serialisation, matching Cassandra byte-for-byte
70//! for the RIGHT reason.
71//!
72//! # Wire format (matches `EstimatedHistogram.EstimatedHistogramSerializer`)
73//!
74//! ```text
75//! i32 BE  bucketCount            = bucketOffsets.len() + 1
76//! for each bucket index i in 0..bucketCount:
77//!   i64 BE  offset  — offsets[i == 0 ? 0 : i - 1], or Long.MAX_VALUE for the
78//!                     overflow bucket (i == bucketCount - 1)
79//!   i64 BE  count   — number of observations that fell in this bucket
80//! ```
81
82/// Number of bucket OFFSETS in the `estimatedPartitionSize` histogram.
83/// cassandra-5.0.0 `MetadataCollector.defaultPartitionSizeHistogram` →
84/// `new EstimatedHistogram(155)`. Serialised bucket count is this + 1 (the
85/// overflow bucket) = 156, matching the committed
86/// `statistics-db-annotated-dump.txt`.
87const PARTITION_SIZE_OFFSET_COUNT: usize = 155;
88
89/// Number of bucket OFFSETS in the `estimatedCellPerPartitionCount` histogram.
90/// cassandra-5.0.0 `MetadataCollector.defaultCellPerPartitionCountHistogram` →
91/// `new EstimatedHistogram(118)`. Serialised bucket count is this + 1 = 119.
92const CELL_COUNT_OFFSET_COUNT: usize = 118;
93
94/// Sentinel offset Cassandra writes for the trailing overflow bucket
95/// (`Long.MAX_VALUE`). Any observation larger than the last real offset falls
96/// here.
97const OVERFLOW_OFFSET: i64 = i64::MAX;

99/// A Cassandra-canonical `EstimatedHistogram` used for `estimatedPartitionSize`
100/// and `estimatedCellPerPartitionCount`.
101///
102/// Bucket offsets are the fixed geometric series `[1, 2, 3, ...]` (length
103/// depends on the histogram); `counts` has one extra slot for the overflow
104/// bucket. Every observation increments exactly one bucket, so `Σ counts` equals
105/// the number of observations (one per partition), which is what the reader
106/// decodes as `partition_count`.
107#[derive(Debug, Clone)]
108pub struct EstimatedHistogram {
109    /// Bucket boundaries (inclusive upper bounds), strictly increasing:
110    /// `[1, 2, 3, 4, ...]`. This is the PURE Cassandra `newOffsets` series with
111    /// no leading duplicate; the serialised leading duplicate is applied in
112    /// [`Self::write_to`].
113    offsets: Vec<i64>,
114    /// Per-bucket observation counts; length is `offsets.len() + 1` (the last
115    /// element is the overflow bucket). `u64` cannot overflow for any real
116    /// SSTable (bounded by partition/cell counts) and is summed as `u64` by the
117    /// reader.
118    counts: Vec<u64>,
119}
120
121impl Default for EstimatedHistogram {
122    fn default() -> Self {
123        Self::partition_size()
124    }
125}

127impl EstimatedHistogram {
128    /// The `estimatedPartitionSize` histogram (155 offsets → 156 serialised
129    /// buckets). cassandra-5.0.0 `MetadataCollector.defaultPartitionSizeHistogram`.
130    pub fn partition_size() -> Self {
131        Self::with_offset_count(PARTITION_SIZE_OFFSET_COUNT)
132    }
133
134    /// The `estimatedCellPerPartitionCount` histogram (118 offsets → 119
135    /// serialised buckets). cassandra-5.0.0
136    /// `MetadataCollector.defaultCellPerPartitionCountHistogram`.
137    pub fn cell_per_partition_count() -> Self {
138        Self::with_offset_count(CELL_COUNT_OFFSET_COUNT)
139    }
140
141    /// Build an empty histogram with `offset_count` canonical bucket offsets and
142    /// `offset_count + 1` zeroed bucket counts (the trailing overflow bucket).
143    fn with_offset_count(offset_count: usize) -> Self {
144        let offsets = new_offsets(offset_count);
145        let counts = vec![0u64; offsets.len() + 1];
146        Self { offsets, counts }
147    }
148
149    /// Record one observation of `value` (a serialized partition size in bytes,
150    /// or a partition's cell count), incrementing the bucket whose offset is the
151    /// smallest `>= value`. Values above the last real offset land in the
152    /// overflow bucket.
153    ///
154    /// Mirrors `EstimatedHistogram.add(long)`:
155    /// `index = Arrays.binarySearch(offsets, value)`; on a miss (`< 0`) the
156    /// insertion point `-(index+1)` selects the first offset strictly greater
157    /// than `value`.
158    pub fn add(&mut self, value: u64) {
159        // Saturate to i64 for comparison against the (i64) offsets. Real
160        // partition sizes / cell counts never approach i64::MAX; saturation is a
161        // fail-safe, not an expected path.
162        let v = i64::try_from(value).unwrap_or(i64::MAX);
163        let idx = match self.offsets.binary_search(&v) {
164            // Exact match: that bucket's inclusive upper bound equals `value`.
165            Ok(i) => i,
166            // Miss: insertion point is the first offset strictly greater than
167            // `value`; when `value` exceeds every offset this is the overflow
168            // bucket (index == offsets.len()).
169            Err(i) => i,
170        };
171        // `idx` is always in-bounds for `counts` (length offsets.len()+1).
172        self.counts[idx] = self.counts[idx].saturating_add(1);
173    }
174
175    /// Serialise the histogram into `buffer` in Cassandra's
176    /// `EstimatedHistogramSerializer` wire format (see module docs).
177    ///
178    /// The offset written for bucket `i` is `offsets[i == 0 ? 0 : i - 1]`,
179    /// exactly as `EstimatedHistogramSerializer.serialize` does — which is why
180    /// `offsets[0]` (value `1`) appears twice in the serialised stream. The final
181    /// (overflow) bucket `i == bucketCount - 1` writes `offsets[bucketCount - 2]`,
182    /// i.e. the LAST real offset — Cassandra does NOT write `Long.MAX_VALUE` for
183    /// the overflow bucket in this format. `OVERFLOW_OFFSET` is only a defensive
184    /// fallback that never fires (`off_idx` is always in-bounds here).
185    pub fn write_to(&self, buffer: &mut Vec<u8>) {
186        let bucket_count = self.counts.len() as i32;
187        buffer.extend_from_slice(&bucket_count.to_be_bytes());
188        for (i, &count) in self.counts.iter().enumerate() {
189            // Cassandra serialiser: writeLong(offsets[i == 0 ? 0 : i - 1]).
190            let off_idx = i.saturating_sub(1);
191            let offset = self
192                .offsets
193                .get(off_idx)
194                .copied()
195                .unwrap_or(OVERFLOW_OFFSET);
196            buffer.extend_from_slice(&offset.to_be_bytes());
197            // Counts are logically non-negative; the reader decodes each as i64.
198            buffer.extend_from_slice(&(count as i64).to_be_bytes());
199        }
200    }
201
202    /// Total observation count (`Σ bucket counts`) — equals the number of
203    /// partitions/cells recorded. Exposed for tests and invariants.
204    #[cfg(test)]
205    pub fn total(&self) -> u64 {
206        self.counts.iter().copied().fold(0u64, u64::saturating_add)
207    }
208}

210/// Build the canonical `EstimatedHistogram` bucket-offset series of length
211/// `size`.
212///
213/// Matches Cassandra's `EstimatedHistogram.newOffsets(size, considerZeroes=false)`
214/// (cassandra-5.0.0 `utils/EstimatedHistogram.java`): `offsets[0] = 1`, then each
215/// subsequent offset is `round(prev * 1.2)`, bumped by one if the rounding did
216/// not advance. Result: the strictly-monotonic series `[1, 2, 3, 4, 5, 6, 7, ...]`
217/// (no leading duplicate — that duplicate is a serialiser artefact, see
218/// [`EstimatedHistogram::write_to`]).
219fn new_offsets(size: usize) -> Vec<i64> {
220    let mut offsets = Vec::with_capacity(size);
221    if size == 0 {
222        return offsets;
223    }
224    // result[0] = 1
225    let mut last: i64 = 1;
226    offsets.push(last);
227    for _ in 1..size {
228        // round(last * 1.2), matching Java Math.round (half-up); if the round did
229        // not advance, bump by one (Cassandra: `if (next == last) next++`).
230        let scaled = (last as f64) * 1.2;
231        let mut next = scaled.round() as i64;
232        if next == last {
233            next += 1;
234        }
235        offsets.push(next);
236        last = next;
237    }
238    offsets
239}

246    fn partition_size_offsets_match_cassandra_155() {
247        let off = new_offsets(PARTITION_SIZE_OFFSET_COUNT);
248        assert_eq!(off.len(), 155, "155 offsets => 156 serialised buckets");
249        // Pure Cassandra series (no leading duplicate in the offsets array).
250        assert_eq!(&off[..8], &[1, 2, 3, 4, 5, 6, 7, 8]);
251        // Documented max value from cassandra-5.0.0
252        // MetadataCollector.defaultPartitionSizeHistogram: EH(155) tracks up to
253        // 3520571548412 (3.5TB) — pins the exact series + count.
254        assert_eq!(off[154], 3_520_571_548_412);
255        // Strictly increasing throughout.
256        for w in off.windows(2) {
257            assert!(w[1] > w[0], "offsets must be strictly increasing");
258        }
