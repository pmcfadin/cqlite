4//! key range, tombstone drop-time histogram) consumed by the Statistics.db
5//! component builders.

8use std::collections::BTreeMap;

10/// Maximum number of bins in the tombstone drop-time histogram.
11///
12/// Mirrors Cassandra's `StreamingTombstoneHistogramBuilder.MAX_BIN_SIZE = 100`.
13pub(super) const TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE: i32 = 100;
14
15/// Streaming tombstone drop-time histogram.
16///
17/// Mirrors Cassandra's `StreamingTombstoneHistogramBuilder` with a fixed maximum
18/// of 100 bins. Tombstone local-deletion-times are accumulated; when the bin count
19/// exceeds the maximum the two nearest bins are merged, keeping the bin count at or
20/// below `MAX_BIN_SIZE`.
21///
22/// Serialisation uses the **legacy** (nb/mc) format:
23/// ```text
24/// i32 BE  maxBinSize  (100 when non-empty, 0 when empty)
25/// i32 BE  size        (actual number of bins)
26/// for each bin:
27///   f64 BE  point  (local-deletion-time as a double)
28///   i64 BE  value  (count of tombstones in this bin)
29/// ```
30#[derive(Debug, Clone, Default)]
31pub struct TombstoneHistogram {
32    /// Bins keyed by `local_deletion_time as i64` (for stable ordering).
33    /// Each value is the count of tombstones that map to this bin.
34    bins: BTreeMap<i64, i64>,
35}
36
37impl TombstoneHistogram {
38    /// Create an empty histogram.
39    pub fn new() -> Self {
40        Self::default()
41    }
42
43    /// Record a tombstone with the given local-deletion-time (Unix seconds).
44    ///
45    /// If adding the new point would exceed `MAX_BIN_SIZE` the two closest
46    /// existing bins are merged first, exactly as Cassandra's builder does.
47    pub fn update(&mut self, local_deletion_time: i32) {
48        let key = local_deletion_time as i64;
49        *self.bins.entry(key).or_insert(0) += 1;
50
51        // Merge bins while over capacity
52        while self.bins.len() as i32 > TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE {
53            self.merge_closest_bins();
54        }
55    }
56
57    /// Return `true` if no tombstone deletion times have been recorded.
58    pub fn is_empty(&self) -> bool {
59        self.bins.is_empty()
60    }
61
62    /// Return the number of populated bins.
63    pub fn size(&self) -> i32 {
64        self.bins.len() as i32
65    }
66
67    /// Merge the two bins whose keys are closest together.
68    ///
69    /// The merged bin's key is the weighted average of the two points;
70    /// its value is the sum of both counts.  This matches Cassandra's
71    /// `mergeNearestBins` approach.
72    fn merge_closest_bins(&mut self) {
73        if self.bins.len() < 2 {
74            return;
75        }
76
77        // Find the pair of adjacent keys with the smallest gap.
78        let keys: Vec<i64> = self.bins.keys().copied().collect();
79        let mut min_gap = i64::MAX;
80        let mut merge_idx = 0usize;
81        for i in 0..keys.len() - 1 {
82            let gap = keys[i + 1] - keys[i];
83            if gap < min_gap {
84                min_gap = gap;
85                merge_idx = i;
86            }
87        }
88
89        let k1 = keys[merge_idx];
90        let k2 = keys[merge_idx + 1];
91        let v1 = self.bins.remove(&k1).unwrap_or(0);
92        let v2 = self.bins.remove(&k2).unwrap_or(0);
93        let total = v1 + v2;
94        // Weighted average key (same formula as Cassandra)
95        let merged_key = if total == 0 {
96            (k1 + k2) / 2
97        } else {
98            (k1 * v1 + k2 * v2) / total
99        };
100        *self.bins.entry(merged_key).or_insert(0) += total;
101    }
102
103    /// Iterate over bins in ascending key order as `(point_as_f64, count)` pairs.
104    pub(super) fn entries(&self) -> impl Iterator<Item = (f64, i64)> + '_ {
105        self.bins.iter().map(|(&k, &v)| (k as f64, v))
106    }
107}

109/// Statistics metadata collected during memtable flush
110///
111/// This structure holds all the metadata needed to write Statistics.db.
112/// Values are collected as rows are written to Data.db.
113#[derive(Debug, Clone)]
114pub struct StatisticsMetadata {
115    /// Minimum timestamp in the SSTable (microseconds since epoch)
116    pub min_timestamp: i64,
117    /// Maximum timestamp in the SSTable (microseconds since epoch)
118    pub max_timestamp: i64,
119    /// Minimum local deletion time (seconds since epoch, for tombstones)
120    pub min_local_deletion_time: i32,
121    /// Maximum local deletion time (seconds since epoch)
122    pub max_local_deletion_time: i32,
123    /// Minimum TTL value (seconds, 0 if no TTL)
124    pub min_ttl: i32,
125    /// Maximum TTL value (seconds, 0 if no TTL)
126    pub max_ttl: i32,
127    /// Total number of partitions in the SSTable
128    pub partition_count: u64,
129    /// Total number of rows (live + tombstones)
130    pub row_count: u64,
131    /// Total number of columns across all rows
132    pub column_count: u64,
133    /// Total size of all rows in bytes
134    pub total_rows_size: u64,
135    /// Histogram of tombstone local-deletion-times for `estimatedTombstoneDropTime`.
136    ///
137    /// Populated by `update_local_deletion_time`; serialised into Statistics.db
138    /// so Cassandra can compute `estimatedDroppableTombstoneRatio` and schedule
139    /// tombstone compaction.  Uses the same streaming builder algorithm as
140    /// Cassandra's `StreamingTombstoneHistogramBuilder` (max 100 bins).
141    pub tombstone_histogram: TombstoneHistogram,
142    /// First (lowest-token) decorated partition-key bytes written to this
143    /// SSTable, or `None` if no partition has been written. Serialised as the
144    /// `da`-format `StatsMetadata.firstKey` (BtiFormat `hasKeyRange`); tracked in
145    /// token order by `SSTableWriter::write_partition`.
146    pub first_key: Option<Vec<u8>>,
147    /// Last (highest-token) decorated partition-key bytes written to this
148    /// SSTable. Serialised as the `da`-format `StatsMetadata.lastKey`.
149    pub last_key: Option<Vec<u8>>,
150    /// Whether any partition written to this SSTable carries a partition-level
151    /// deletion (partition tombstone).
152    ///
153    /// Serialised as the `da`-format `StatsMetadata.hasPartitionLevelDeletions`
154    /// boolean (`hasPartitionLevelDeletionsPresenceMarker`); written as a single
155    /// `writeBoolean` byte (`0x01` true, `0x00` false) — matching
156    /// cassandra-5.0.0 `StatsMetadata.StatsMetadataSerializer.serialize`
157    /// (line 495). Set by `SSTableWriter::write_partition` whenever a mutation
158    /// contributes a `partition_tombstone`. Ignored by the legacy BIG (nb/oa)
159    /// STATS body, which never serialises this field.
160    pub has_partition_level_deletions: bool,
161
162    /// `repairedAt` repair timestamp (`0` = unrepaired). Serialised verbatim
163    /// into the STATS component `repairedAt` field. Preserved through compaction
164    /// from the (compatible) input SSTables (issue #1021); a fresh memtable flush
165    /// leaves it `0`.
166    pub repaired_at: i64,
167
168    /// `pendingRepair` incremental-repair session UUID (`None` = no pending
169    /// repair). Serialised as the STATS `pendingRepair` nullable field (presence
170    /// byte then 16-byte UUID). Preserved through compaction from compatible
171    /// inputs (issue #1021); a fresh flush leaves it `None`.
172    pub pending_repair: Option<[u8; 16]>,
173
174    /// `isTransient` flag (transiently-replicated data). Serialised as the STATS
175    /// `isTransient` boolean. Preserved through compaction from compatible inputs
176    /// (issue #1021); a fresh flush leaves it `false`.
177    pub is_transient: bool,
178
179    /// `estimatedPartitionSize` EstimatedHistogram — one observation per
180    /// partition of its serialized Data.db size in bytes. Serialised as the FIRST
181    /// STATS field; the authoritative partition-count decode (`read_table_counts`,
182    /// issue #944) sums its bucket counts, so `Σ counts == partition_count`
183    /// (issue #1327). Populated by [`Self::record_partition`].
184    pub estimated_partition_size: EstimatedHistogram,
185
186    /// `estimatedCellPerPartitionCount` EstimatedHistogram — one observation per
187    /// partition of its cell count. Serialised as the SECOND STATS field.
188    /// Seeded with its OWN authoritative shape (118 offsets → 119 buckets), which
189    /// is DISTINCT from the partition-size histogram's 155/156 shape (issue #1327
190    /// finding 1; cassandra-5.0.0
191    /// `MetadataCollector.defaultCellPerPartitionCountHistogram` → `EstimatedHistogram(118)`).
192    /// Populated by [`Self::record_partition`].
193    pub estimated_cell_count: EstimatedHistogram,
194}

196impl Default for StatisticsMetadata {
197    fn default() -> Self {
198        Self {
199            min_timestamp: i64::MAX,
200            max_timestamp: i64::MIN,
201            min_local_deletion_time: i32::MAX,
202            max_local_deletion_time: i32::MIN,
203            min_ttl: i32::MAX,
204            max_ttl: 0,
205            partition_count: 0,
206            row_count: 0,
207            column_count: 0,
208            total_rows_size: 0,
209            tombstone_histogram: TombstoneHistogram::new(),
210            first_key: None,
211            last_key: None,
212            has_partition_level_deletions: false,
213            repaired_at: 0,
214            pending_repair: None,
215            is_transient: false,
216            // Two DISTINCT Cassandra-canonical shapes (issue #1327 finding 1):
217            // partition-size seeds EH(155) (156 buckets), cell-count seeds
218            // EH(118) (119 buckets). cassandra-5.0.0 MetadataCollector
219            // defaultPartitionSizeHistogram / defaultCellPerPartitionCountHistogram.
220            estimated_partition_size: EstimatedHistogram::partition_size(),
221            estimated_cell_count: EstimatedHistogram::cell_per_partition_count(),
222        }
223    }
224}

226impl StatisticsMetadata {
227    /// Create a new empty statistics metadata
228    pub fn new() -> Self {
229        Self::default()
230    }
231
232    /// Update timestamp range with a new timestamp value.
233    ///
234    /// LIVE deletion markers must never enter the min/max aggregates (issue
235    /// #851, Cassandra `d5bc7fb5`). Cassandra encodes `DeletionTime.LIVE` with
236    /// `markedForDeleteAt = Long.MIN_VALUE`, and a `NO_DELETION` / absent
237    /// liveness timestamp surfaces as `Long.MAX_VALUE`. Folding either sentinel
238    /// into `minTimestamp` / `maxTimestamp` would poison the stats, so they are
239    /// skipped here at the single aggregation chokepoint.
240    pub fn update_timestamp(&mut self, timestamp: i64) {
241        if Self::is_live_timestamp(timestamp) {
242            return;
243        }
244        self.min_timestamp = self.min_timestamp.min(timestamp);
245        self.max_timestamp = self.max_timestamp.max(timestamp);
246    }
247
248    /// Update local deletion time range (for tombstones) and the drop-time histogram.
249    ///
250    /// Call this for every tombstone local-deletion-time encountered while writing:
251    /// cell tombstones, row deletions, range tombstones, and partition tombstones.
252    ///
253    /// A LIVE marker (`DeletionTime.LIVE.localDeletionTime == Integer.MAX_VALUE`,
254    /// the same value as `Cell.NO_DELETION_TIME`) is not a tombstone: it must not
255    /// pull `minLocalDeletionTime` down to the sentinel nor inflate the tombstone
256    /// drop-time histogram (issue #851, Cassandra `d5bc7fb5`).
257    pub fn update_local_deletion_time(&mut self, deletion_time: i32) {
258        if Self::is_live_local_deletion_time(deletion_time) {
259            return;
260        }
261        self.min_local_deletion_time = self.min_local_deletion_time.min(deletion_time);
262        self.max_local_deletion_time = self.max_local_deletion_time.max(deletion_time);
263        self.tombstone_histogram.update(deletion_time);
264    }
265
266    /// Fold a LIVE (non-TTL) cell's `NO_DELETION_TIME` sentinel into the
267    /// `maxLocalDeletionTime` aggregate ONLY.
268    ///
269    /// Cassandra stamps every live, non-expiring cell with
270    /// `Cell.NO_DELETION_TIME` (`Integer.MAX_VALUE`) as its `localDeletionTime`,
271    /// and `MetadataCollector.update(Cell)` folds that value into
272    /// `maxLocalDeletionTime`. So any SSTable that contains at least one live
273    /// non-TTL cell finalizes `maxLocalDeletionTime == Integer.MAX_VALUE`,
274    /// EVEN when the same SSTable also carries older tombstones with a smaller
275    /// real LDT (a mixed SSTable — e.g. after compaction). Without this fold a
276    /// mixed SSTable would keep `maxLocalDeletionTime` at the largest TOMBSTONE
277    /// LDT and diverge from Cassandra (issue #1728).
278    ///
279    /// Unlike [`Self::update_local_deletion_time`], the live sentinel must NOT
280    /// pull `minLocalDeletionTime` down nor enter the tombstone drop-time
281    /// histogram (those describe real tombstones only; issue #851). This method
282    /// therefore raises the max alone. A pure-live SSTable ends at
283    /// `max_local_deletion_time == i32::MAX`, which serialises to the same
284    /// `NO_DELETION_TIME` sentinel as the `== 0` "no deletions" case, so the
285    /// pure-live output byte-parity is unchanged.
286    pub fn note_live_local_deletion_time(&mut self) {
287        // `i32::MAX` is the largest possible local-deletion-time, so folding it
288        // into the running max is simply an assignment (a real tombstone LDT is
289        // always strictly smaller). Written directly to avoid the no-op
290        // `.max(i32::MAX)` clippy::unnecessary_min_or_max lint.
291        self.max_local_deletion_time = i32::MAX;
292    }
293
294    /// True when `timestamp` is a LIVE / `NO_DELETION` marker rather than a real
295    /// deletion timestamp. CQLite writes `DeletionTime.LIVE` as `Long.MIN_VALUE`
296    /// (see `data_writer.rs`); Cassandra's `NO_DELETION` / `NO_TIMESTAMP` sentinel
297    /// is `Long.MAX_VALUE`. Both mean "no deletion" and must be excluded from
298    /// timestamp aggregation.
299    fn is_live_timestamp(timestamp: i64) -> bool {
300        timestamp == i64::MIN || timestamp == i64::MAX
301    }
302
303    /// True when `deletion_time` is a LIVE marker (`Integer.MAX_VALUE`) rather
304    /// than a real tombstone local-deletion-time.
305    fn is_live_local_deletion_time(deletion_time: i32) -> bool {
306        deletion_time == i32::MAX
307    }
308
309    /// Update TTL range
310    pub fn update_ttl(&mut self, ttl: i32) {
311        if ttl > 0 {
312            self.min_ttl = self.min_ttl.min(ttl);
313            self.max_ttl = self.max_ttl.max(ttl);
314        }
315    }
316
317    /// Increment partition count
318    pub fn increment_partition_count(&mut self) {
319        self.partition_count += 1;
320    }
321
322    /// Record one partition's serialized size and cell count into the
323    /// `estimatedPartitionSize` / `estimatedCellPerPartitionCount` histograms.
324    ///
325    /// Called once per partition written to Data.db with the exact serialized
326    /// partition byte length (`serialized_size`) and the number of cells emitted
327    /// (`cell_count`). Because it adds exactly one observation per partition, the
328    /// `estimatedPartitionSize` histogram's Σ bucket counts equals the SSTable's
329    /// partition count — the value the read-side authoritative decode
330    /// (`read_table_counts`, issue #944) reports. This mirrors Cassandra's
331    /// `MetadataCollector.addPartitionSizeInBytes` /
332    /// `addCellPerPartitionCount` (issue #1327).
333    pub fn record_partition(&mut self, serialized_size: u64, cell_count: u64) {
334        self.estimated_partition_size.add(serialized_size);
335        self.estimated_cell_count.add(cell_count);
336    }
337
338    /// Record a partition key in the SSTable key range.
339    ///
340    /// Partitions are written in ascending token order (enforced by
341    /// `SSTableWriter::write_partition`), so the FIRST key seen is the lowest and
342    /// the LAST is the highest. Used to populate the `da`-format
343    /// `StatsMetadata.firstKey`/`lastKey` (`hasKeyRange`). The clone is bounded by
344    /// the partition-key size (typically tens of bytes), not the partition data.
345    pub fn update_key_range(&mut self, key: &[u8]) {
346        if self.first_key.is_none() {
347            self.first_key = Some(key.to_vec());
348        }
349        self.last_key = Some(key.to_vec());
350    }
351
352    /// Record that a partition-level deletion (partition tombstone) was written.
353    ///
354    /// Once set, this stays `true` for the lifetime of the SSTable and drives the
355    /// `da`-format `StatsMetadata.hasPartitionLevelDeletions` marker. Called by
356    /// `SSTableWriter::write_partition` for any partition carrying a
357    /// `partition_tombstone`.
358    pub fn mark_partition_level_deletion(&mut self) {
359        self.has_partition_level_deletions = true;
360    }
361
362    /// Set the persisted repair state (`repairedAt`, `pendingRepair`,
363    /// `isTransient`) carried into the output STATS component.
364    ///
365    /// Used by the compaction merge path to preserve the repair metadata of
366    /// compatible inputs through to the merged output (issue #1021). A fresh
367    /// memtable flush never calls this, so the default unrepaired state
368    /// (`repaired_at = 0`, `pending_repair = None`, `is_transient = false`) is
369    /// retained.
370    pub fn set_repair_state(
371        &mut self,
372        repaired_at: i64,
373        pending_repair: Option<[u8; 16]>,
374        is_transient: bool,
375    ) {
376        self.repaired_at = repaired_at;
377        self.pending_repair = pending_repair;
378        self.is_transient = is_transient;
379    }
380
381    /// Increment row count
382    pub fn increment_row_count(&mut self) {
383        self.row_count += 1;
384    }
385
386    /// Add to column count
387    pub fn add_column_count(&mut self, count: u64) {
388        self.column_count += count;
389    }
390
391    /// Add to total rows size
392    pub fn add_rows_size(&mut self, size: u64) {
393        self.total_rows_size += size;
394    }
395
396    /// Finalize metadata before writing (normalize sentinel values)
397    pub fn finalize(&mut self) {
398        // If no timestamps were recorded, set to 0
399        if self.min_timestamp == i64::MAX {
400            self.min_timestamp = 0;
401        }
402        if self.max_timestamp == i64::MIN {
403            self.max_timestamp = 0;
404        }
405
406        // If no deletion times were recorded, set to 0
407        if self.min_local_deletion_time == i32::MAX {
408            self.min_local_deletion_time = 0;
409        }
410        if self.max_local_deletion_time == i32::MIN {
411            self.max_local_deletion_time = 0;
412        }
413
414        // If no TTLs were recorded, set min_ttl to 0
415        if self.min_ttl == i32::MAX {
416            self.min_ttl = 0;
417        }
418    }
419}

444    /// marker must not poison the timestamp aggregates. CQLite encodes

491    /// Issue #1728: a live non-TTL `Write` cell folds `NO_DELETION_TIME`
492    /// (`i32::MAX`) into `max_local_deletion_time`, matching Cassandra's
493    /// MetadataCollector. In a MIXED SSTable (an older tombstone at a smaller
494    /// real LDT plus a live cell) `max_local_deletion_time` must be the live
495    /// sentinel, NOT the tombstone LDT — while `min_local_deletion_time` and the
496    /// tombstone drop-time histogram continue to describe the real tombstone

498    #[test]
499    fn test_note_live_local_deletion_time_mixed_sstable() {
500        let mut meta = StatisticsMetadata::new();
501        // An older tombstone at a real, smaller LDT.
502        meta.update_local_deletion_time(1_500_000_000);
503        // A live non-TTL cell in the same (mixed) SSTable.
504        meta.note_live_local_deletion_time();

525    }

674    // -----------------------------------------------------------------------
675    // Issue #1668 stage 5a: prove the per-mutation stats fold is associative /
676    // order-independent, so `write_partition`'s baseline computation does NOT
677    // require the whole partition's `Vec<Mutation>` at once once baselines are
678    // pre-seeded (issue #729) — it can be fed incrementally, one row at a
679    // time, across any grouping or ordering of the SAME underlying values.
680    // -----------------------------------------------------------------------
681
682    /// `update_timestamp`/`update_local_deletion_time`/`update_ttl` are pure
683    /// min/max folds (see their doc comments): feeding the SAME set of values
684    /// in a DIFFERENT order (or split across many separate "calls" instead of
685    /// one batch) must produce an IDENTICAL final `min_timestamp` /
686    /// `max_timestamp` / `min_local_deletion_time` / `max_local_deletion_time`
687    /// / `min_ttl` / `max_ttl`. This is the concrete proof (not just a code
688    /// read) that a caller can stream rows into these folds one at a time
689    /// instead of pre-collecting a whole `Vec<Mutation>` first.
690    #[test]
691    fn stats_fold_is_order_independent_issue_1668_stage_5a() {
692        // A representative fixture: several distinct timestamps, local
693        // deletion times, and TTLs, INCLUDING the LIVE sentinels that must be
694        // filtered identically regardless of order (issue #851).
695        let timestamps = [
696            500_000i64,
697            2_000_000,
698            1_000_000,
699            i64::MIN,
700            i64::MAX,
701            750_000,
702        ];
703        let ldts = [1_700_000_000i32, 1_650_000_000, 1_690_000_000, i32::MAX];
704        let ttls = [3600i32, 60, 86400, 0, -1];
705
706        // Batch order A: as declared above (mirrors `write_partition`'s
707        // single forward `for mutation in &mutations` pass).
708        let mut batch_a = StatisticsMetadata::new();
709        for &ts in &timestamps {
710            batch_a.update_timestamp(ts);
711        }
712        for &ldt in &ldts {
713            batch_a.update_local_deletion_time(ldt);
714        }
715        for &ttl in &ttls {
716            batch_a.update_ttl(ttl);
717        }
718
719        // Batch order B: REVERSED and INTERLEAVED — simulates rows arriving
720        // one cluster group at a time, in a different relative order, across
721        // (hypothetically) multiple incremental calls.
722        let mut batch_b = StatisticsMetadata::new();
723        for &ttl in ttls.iter().rev() {
724            batch_b.update_ttl(ttl);
725        }
726        for &ts in timestamps.iter().rev() {
727            batch_b.update_timestamp(ts);
728        }
729        for &ldt in ldts.iter().rev() {
730            batch_b.update_local_deletion_time(ldt);
731        }
732
733        // Batch order C: fully interleaved, one value of EACH kind per
734        // "cluster group" — the closest analogue to a genuine per-row stream.
735        let mut batch_c = StatisticsMetadata::new();
736        let n = timestamps.len().max(ldts.len()).max(ttls.len());
737        for i in 0..n {
738            if let Some(&ts) = timestamps.get(i) {
739                batch_c.update_timestamp(ts);
740            }
741            if let Some(&ldt) = ldts.get(i) {
742                batch_c.update_local_deletion_time(ldt);
743            }
744            if let Some(&ttl) = ttls.get(i) {
745                batch_c.update_ttl(ttl);
746            }
747        }
748
749        for (name, batch) in [("B (reversed)", &batch_b), ("C (interleaved)", &batch_c)] {
750            assert_eq!(
751                batch.min_timestamp, batch_a.min_timestamp,
752                "min_timestamp diverged for order {name}"
753            );
754            assert_eq!(
755                batch.max_timestamp, batch_a.max_timestamp,
756                "max_timestamp diverged for order {name}"
757            );
758            assert_eq!(
759                batch.min_local_deletion_time, batch_a.min_local_deletion_time,
760                "min_local_deletion_time diverged for order {name}"
761            );
762            assert_eq!(
763                batch.max_local_deletion_time, batch_a.max_local_deletion_time,
764                "max_local_deletion_time diverged for order {name}"
765            );
766            assert_eq!(
767                batch.min_ttl, batch_a.min_ttl,
768                "min_ttl diverged for order {name}"
769            );
770            assert_eq!(
771                batch.max_ttl, batch_a.max_ttl,
772                "max_ttl diverged for order {name}"
773            );
774            assert_eq!(
775                batch.tombstone_histogram.size(),
776                batch_a.tombstone_histogram.size(),
777                "tombstone_histogram bin count diverged for order {name}"
778            );
779        }
780
781        // Sanity: the LIVE sentinels were actually exercised and correctly
782        // excluded (issue #851) — the real values, not the sentinels, won.
783        assert_eq!(batch_a.min_timestamp, 500_000);
784        assert_eq!(batch_a.max_timestamp, 2_000_000);
785        assert_eq!(batch_a.min_local_deletion_time, 1_650_000_000);
786        assert_eq!(batch_a.max_local_deletion_time, 1_700_000_000);
787        assert_eq!(batch_a.min_ttl, 60);
788        assert_eq!(batch_a.max_ttl, 86_400);
789    }
