1//! Startup orphan sweeps for the write engine (issue #1393).
2//!
3//! When the process crashes mid-compaction, `finalize_merge_async` can leave two
4//! kinds of orphan on disk:
5//!
6//!   (a) a `.compaction-tmp-{gen}/` directory under `data_dir` holding partial
7//!       output component files that were flushed but never renamed, and
8//!   (b) a partial set of renamed `nb-{gen}-big-*.db` components in
9//!       `data_dir/{keyspace}/{table}/` that lack a matching `TOC.txt` (the
10//!       publication barrier), left when a crash lands between the first
11//!       component rename and the `TOC.txt` rename.
12//!
13//! Both are reclaimed by the startup sweeps below, which run in
14//! [`WriteEngine::open`](super::WriteEngine) before the WAL is replayed. The
15//! sweeps are the single most dangerous delete-code in the write engine: a bug
16//! here can destroy live data. They are therefore **best-effort and observable**
17//! — each returns a [`SweepOutcome`] recording exactly what was removed and any
18//! non-fatal failure, so a failure never aborts startup yet is still surfaced
19//! (and asserted in tests) rather than swallowed into a log line.
20
21use super::{WriteEngine, WriteEngineConfig};
22use std::path::{Path, PathBuf};
23
24/// Outcome of a single startup orphan sweep.
25///
26/// The sweep is best-effort and never aborts engine startup. This struct makes
27/// its actions observable so callers (and tests) can assert what happened
28/// without scraping log output:
29///
30/// * `removed` — orphan paths that were successfully reclaimed. For the partial
31///   SSTable sweep this is the `Data.db` path whose component set was deleted.
32/// * `failures` — one human-readable message per orphan that could NOT be
33///   removed (for example a permissions error). A non-empty `failures` is the
34///   "surfaced but non-fatal" condition described in issue #1393 AC #3.
35#[derive(Debug, Default, Clone, PartialEq, Eq)]
36pub(crate) struct SweepOutcome {
37    /// Orphan paths successfully removed by this sweep.
38    pub(crate) removed: Vec<PathBuf>,
39    /// Non-fatal failures encountered (one message per un-removable orphan).
40    pub(crate) failures: Vec<String>,
41}
42
43impl SweepOutcome {
44    /// True when at least one orphan could not be removed. Startup continues
45    /// regardless; callers log the condition.
46    pub(crate) fn had_failures(&self) -> bool {
47        !self.failures.is_empty()
48    }
49}
50
51impl WriteEngine {
52    /// Run both startup orphan sweeps and log any non-fatal failures.
53    ///
54    /// Called once from [`WriteEngine::open`](super::WriteEngine) before WAL
55    /// replay. Never returns an error: an un-removable orphan is logged and left
56    /// for a later sweep, it does not abort startup (issue #1393 AC #3).
57    pub(crate) fn sweep_startup_orphans(config: &WriteEngineConfig) {
58        let data_dir = &config.data_dir;
59        let tmp = Self::sweep_orphaned_compaction_tmp(data_dir);
60        let partial = Self::sweep_orphaned_partial_sstables(
61            data_dir,
62            &config.schema.keyspace,
63            &config.schema.table,
64        );
65        for outcome in [&tmp, &partial] {
66            if outcome.had_failures() {
67                for failure in &outcome.failures {
68                    tracing::warn!(
69                        "startup orphan sweep left an un-removable orphan (non-fatal, \
70                         will retry next startup): {}",
71                        failure
72                    );
73                }
74            }
75        }
76    }
77
78    /// Startup sweep (a): remove any `.compaction-tmp-*` directories left under
79    /// `data_dir` by a previous crash mid-rename. Best-effort — individual
80    /// failures are recorded in the returned [`SweepOutcome`] and logged, but do
81    /// not abort engine startup.
82    pub(crate) fn sweep_orphaned_compaction_tmp(data_dir: &Path) -> SweepOutcome {
83        let mut outcome = SweepOutcome::default();
84        let read_dir = match std::fs::read_dir(data_dir) {
85            Ok(rd) => rd,
86            Err(e) => {
87                tracing::debug!(
88                    "sweep_orphaned_compaction_tmp: cannot read {:?}: {}",
89                    data_dir,
90                    e
91                );
92                return outcome;
93            }
94        };
95
96        for entry in read_dir.flatten() {
97            let path = entry.path();
98            let name = entry.file_name();
99            let name_str = name.to_string_lossy();
100            if name_str.starts_with(".compaction-tmp-") && path.is_dir() {
101                tracing::warn!("removing orphaned compaction tmp directory: {:?}", path);
102                match std::fs::remove_dir_all(&path) {
103                    Ok(()) => outcome.removed.push(path),
104                    Err(e) => {
105                        tracing::warn!(
106                            "failed to remove orphaned compaction tmp directory {:?}: {}",
107                            path,
108                            e
109                        );
110                        outcome.failures.push(format!("{:?}: {}", path, e));
111                    }
112                }
113            }
114        }
115        outcome
116    }
117
118    /// Startup sweep (b): remove any `nb-{gen}-big-Data.db` (and its siblings)
119    /// under `data_dir/keyspace/table/` that lack a matching `TOC.txt`.
120    ///
121    /// Such files are left when a crash occurs after some component renames but
122    /// before `TOC.txt` is renamed (the publication barrier). A published
123    /// generation — Data.db *with* a sibling TOC.txt — is never touched. The
124    /// returned [`SweepOutcome`] records each reclaimed Data.db path and any
125    /// non-fatal removal failure.
126    pub(crate) fn sweep_orphaned_partial_sstables(
127        data_dir: &Path,
128        keyspace: &str,
129        table: &str,
130    ) -> SweepOutcome {
131        let mut outcome = SweepOutcome::default();
132        let sstable_dir = data_dir.join(keyspace).join(table);
133
134        let read_dir = match std::fs::read_dir(&sstable_dir) {
135            Ok(rd) => rd,
136            Err(_) => {
137                // Directory doesn't exist yet — nothing to sweep.
138                return outcome;
139            }
140        };
141
142        for entry in read_dir.flatten() {
143            let path = entry.path();
144            let name = entry.file_name();
145            let name_str = name.to_string_lossy();
146
147            // Look for Data.db files produced by the writer (nb-{gen}-big-Data.db)
148            if !name_str.starts_with("nb-")
149                || !name_str.ends_with("-big-Data.db")
150                || !path.is_file()
151            {
152                continue;
153            }
154
155            // Extract the base prefix (e.g. "nb-5-big") to find the TOC sibling
156            let base = match name_str.strip_suffix("-Data.db") {
157                Some(b) => b.to_owned(),
158                None => continue,
159            };
160
161            // Extract the generation number for the log message
162            let gen_str = base
163                .strip_prefix("nb-")
164                .and_then(|s| s.strip_suffix("-big"))
165                .unwrap_or(&base);
166
167            let toc_path = sstable_dir.join(format!("{}-TOC.txt", base));
168            if !toc_path.exists() {
169                tracing::warn!(
170                    "removing orphaned partial SSTable components for generation {}: missing TOC.txt",
171                    gen_str
172                );
173                match Self::delete_sstable_files_static(&path) {
174                    Ok(()) => outcome.removed.push(path.clone()),
175                    Err(e) => {
176                        tracing::warn!(
177                            "failed to remove orphaned partial SSTable for generation {}: {}",
178                            gen_str,
179                            e
180                        );
181                        outcome
182                            .failures
183                            .push(format!("generation {}: {}", gen_str, e));
184                    }
185                }
186            }
187        }
188        outcome
189    }
190}

223    /// Enumerate the immediate names inside a directory as a sorted set. Used to
224    /// assert byte-for-byte set equality before/after a sweep.
225    fn dir_names(dir: &Path) -> BTreeSet<String> {
226        std::fs::read_dir(dir)
227            .map(|rd| {
228                rd.flatten()
229                    .map(|e| e.file_name().to_string_lossy().into_owned())
230                    .collect()
231            })
232            .unwrap_or_default()
233    }

235    /// Read the number of live rows from a single on-disk SSTable Data.db file.
236    fn live_row_count(data_path: &Path, schema: &TableSchema) -> usize {
237        let rt = tokio::runtime::Builder::new_current_thread()
238            .enable_all()
239            .build()
240            .unwrap();
241        rt.block_on(async {
242            let config = crate::Config::default();
243            let platform =
244                std::sync::Arc::new(crate::platform::Platform::new(&config).await.unwrap());
245            let reader = SSTableReader::open(data_path, &config, platform)
246                .await
247                .unwrap();
248            let rows = reader
249                .iterate_all_partitions_for_compaction(Some(schema))
250                .await
251                .unwrap();
252            rows.iter()
253                .filter(|r| matches!(r.row_data, CompactionRowData::Live { .. }))
254                .count()
255        })
256    }

258    /// Probe whether this process can be blocked from deleting a file by making
259    /// its parent directory read-only. Returns `true` on a normal (non-root)
260    /// account where the sweep-failure path is exercisable; `false` when running
261    /// as root (permission checks bypassed) so the permissions test can skip
262    /// rather than falsely pass. Restores permissions before returning.
263    #[cfg(unix)]
264    fn readonly_dir_blocks_delete() -> bool {
265        use std::os::unix::fs::PermissionsExt;
266        let probe = TempDir::new().unwrap();
267        let victim = probe.path().join("victim");
268        std::fs::write(&victim, b"x").unwrap();
269        std::fs::set_permissions(probe.path(), std::fs::Permissions::from_mode(0o555)).unwrap();
270        let blocked = std::fs::remove_file(&victim).is_err();
271        // Restore write permission so TempDir cleanup succeeds.
272        std::fs::set_permissions(probe.path(), std::fs::Permissions::from_mode(0o755)).unwrap();
273        blocked
274    }

325        // Live generation is byte-for-byte intact and still readable.
326        assert_eq!(

328            live_bytes_before,
329            "live Data.db must be byte-for-byte untouched"

333            live_rows_before,

399    #[cfg(unix)]
400    #[test]
401    fn undeletable_orphan_is_non_fatal_and_surfaced() {
402        use std::os::unix::fs::PermissionsExt;
403
404        if !readonly_dir_blocks_delete() {
405            // Running as root: the permission gate does not apply. Skip rather
406            // than falsely pass.
407            eprintln!("skipping: cannot exercise permission-denied path (root?)");
408            return;
409        }
410
411        let temp_dir = TempDir::new().unwrap();
412        let dir = sstable_dir(&temp_dir);
413        std::fs::create_dir_all(&dir).unwrap();
414
415        // A partial-rename orphan (no TOC.txt) that we then make un-removable by
416        // dropping write permission on its parent directory.
417        std::fs::write(dir.join("nb-42-big-Data.db"), b"orphan").unwrap();
418        std::fs::set_permissions(&dir, std::fs::Permissions::from_mode(0o555)).unwrap();
419
420        // The sweep must NOT panic/abort and MUST surface the failure.
421        let outcome = WriteEngine::sweep_orphaned_partial_sstables(
422            temp_dir.path().join("data").as_path(),
423            "test_ks",
424            "test_table",
425        );
426
427        // Restore permissions so the file survives for the assertion and so
428        // TempDir cleanup can succeed.
429        std::fs::set_permissions(&dir, std::fs::Permissions::from_mode(0o755)).unwrap();
430
431        assert!(
432            outcome.had_failures(),
433            "an un-removable orphan must be surfaced as a non-fatal failure"
434        );
435        assert!(
436            outcome.failures[0].contains("generation 42"),
437            "the surfaced failure must identify the orphan generation, got: {:?}",
438            outcome.failures
439        );
440        // Non-fatal: the file is still present (removal was blocked, not aborted).
441        assert!(dir.join("nb-42-big-Data.db").exists());
442
443        // And full engine startup over the same un-removable orphan must still
444        // succeed (best-effort sweep never aborts open()).
445        std::fs::set_permissions(&dir, std::fs::Permissions::from_mode(0o555)).unwrap();
446        let opened = WriteEngine::new(config_for(&temp_dir));
447        std::fs::set_permissions(&dir, std::fs::Permissions::from_mode(0o755)).unwrap();
448        assert!(
449            opened.is_ok(),
450            "engine startup must not abort on an un-removable orphan"
451        );
452    }

454    // ── AC #4: idempotence — running the sweep twice is a no-op the 2nd time ────
455
456    #[test]
457    fn sweeps_are_idempotent() {
458        let temp_dir = TempDir::new().unwrap();
459        let data_dir = temp_dir.path().join("data");
460        let dir = sstable_dir(&temp_dir);
461        std::fs::create_dir_all(&dir).unwrap();
462
463        // Seed both orphan kinds.
464        std::fs::create_dir_all(data_dir.join(".compaction-tmp-5")).unwrap();
465        std::fs::write(data_dir.join(".compaction-tmp-5").join("p.db"), b"x").unwrap();
466        std::fs::write(dir.join("nb-5-big-Data.db"), b"orphan").unwrap();
467        std::fs::write(dir.join("nb-5-big-Index.db"), b"orphan").unwrap();
468
469        // First sweep removes the orphans.
470        let tmp1 = WriteEngine::sweep_orphaned_compaction_tmp(&data_dir);
471        let part1 =
472            WriteEngine::sweep_orphaned_partial_sstables(&data_dir, "test_ks", "test_table");
473        assert_eq!(
474            tmp1.removed.len(),
475            1,
476            "first tmp sweep removes the orphan dir"
477        );
478        assert_eq!(
479            part1.removed.len(),
480            1,
481            "first partial sweep removes the orphan"
482        );
483        assert!(!tmp1.had_failures() && !part1.had_failures());
484
485        let state_after_first = dir_names(&data_dir);
486
487        // Second sweep is a strict no-op: nothing removed, nothing failed, and the
488        // on-disk state is unchanged.
489        let tmp2 = WriteEngine::sweep_orphaned_compaction_tmp(&data_dir);
490        let part2 =
491            WriteEngine::sweep_orphaned_partial_sstables(&data_dir, "test_ks", "test_table");
492        assert!(tmp2.removed.is_empty(), "second tmp sweep removes nothing");
493        assert!(
494            part2.removed.is_empty(),
495            "second partial sweep removes nothing"
496        );
497        assert!(!tmp2.had_failures() && !part2.had_failures());
498        assert_eq!(
499            state_after_first,
500            dir_names(&data_dir),
501            "the second sweep must not change the directory"
502        );
503    }

507    #[test]
508    fn crash_mid_compaction_then_restart_sweeps_and_recompacts() {
509        use crate::storage::write_engine::compaction::FAIL_COMPACTION_BEFORE_RENAME;
510
511        let temp_dir = TempDir::new().unwrap();
512        let schema = create_test_schema();
513        let data_dir = temp_dir.path().join("data");
514
515        // Produce 4 complete, published pre-compaction generations (5 rows each).
516        let mut engine = WriteEngine::new(config_for(&temp_dir)).unwrap();
517        let pre_paths = flush_n_sstables_sync(&mut engine, 4);
518        assert_eq!(pre_paths.len(), 4);
519
520        // Inject a crash after tmp files are written but before the publication
521        // rename, then drive a compaction. finalize fails; the tmp dir and the
522        // untouched inputs are left on disk exactly as a real crash would.
523        FAIL_COMPACTION_BEFORE_RENAME.with(|f| f.set(true));
524        let policy = crate::storage::write_engine::STCSPolicy::new(4, 32, 0.5, 1.5, 0).unwrap();
525        engine.set_merge_policy(Box::new(policy)).unwrap();
526        let crashed = engine.maintenance_step(Duration::from_secs(60));
527        FAIL_COMPACTION_BEFORE_RENAME.with(|f| f.set(false));
528        assert!(
529            crashed.is_err(),
530            "the injected pre-rename failure must surface as a compaction error"
531        );
532
533        // A tmp dir with real partial output survived the "crash".
534        let tmp_dirs: Vec<_> = std::fs::read_dir(&data_dir)
535            .unwrap()
536            .flatten()
537            .filter(|e| {
538                e.file_name()
539                    .to_string_lossy()
540                    .starts_with(".compaction-tmp-")
541            })
542            .collect();
543        assert_eq!(
544            tmp_dirs.len(),
545            1,
546            "the crash must leave exactly one .compaction-tmp-* dir"
547        );
548
549        // Pre-compaction inputs are intact (renames never happened).
550        for p in &pre_paths {
551            assert!(p.exists(), "input {:?} must survive a failed compaction", p);
552        }
553
554        // "Restart": drop and reopen the engine — startup sweeps fire.
555        drop(engine);
556        let mut engine = WriteEngine::new(config_for(&temp_dir)).unwrap();
557
558        let leftover: Vec<_> = std::fs::read_dir(&data_dir)
559            .unwrap()
560            .flatten()
561            .filter(|e| {
562                e.file_name()
563                    .to_string_lossy()
564                    .starts_with(".compaction-tmp-")
565            })
566            .collect();
567        assert!(
568            leftover.is_empty(),
569            "startup sweep must reclaim the orphaned tmp dir, found {:?}",
570            leftover.iter().map(|e| e.path()).collect::<Vec<_>>()
571        );
572
573        // Pre-compaction generations still serve reads (20 rows total across 4).
574        let total_pre: usize = pre_paths.iter().map(|p| live_row_count(p, &schema)).sum();
575        assert_eq!(
576            total_pre, 20,
577            "all 20 pre-compaction rows must still read back"
578        );
579
580        // A re-run compaction now succeeds and its output holds every row.
581        let policy = crate::storage::write_engine::STCSPolicy::new(4, 32, 0.5, 1.5, 0).unwrap();
582        engine.set_merge_policy(Box::new(policy)).unwrap();
583        let report = engine.maintenance_step(Duration::from_secs(60)).unwrap();
584        assert_eq!(
585            report.completed_merges.len(),
586            1,
587            "the re-run compaction must complete one merge"
588        );
589        let merged = &report.completed_merges[0];
590        assert_eq!(
591            live_row_count(merged, &schema),
592            20,
593            "the recompacted output must contain all 20 rows"
594        );
595    }

597    // ── issue #1959 (rust-reviewer Low 3): publication-barrier fsync faults ──────
598    //
599    // The directory durability barrier runs at two publication points inside
600    // `finalize_merge_async`: step 2b (after the non-TOC renames, before the TOC
601    // rename) and step 2c (after the TOC rename, before the inputs are deleted).
602    // A real EIO/ENOSPC on either fsync must roll back cleanly. These tests arm
603    // the `#[cfg(test)]` fault seam at each point and assert the invariant holds:
604    //   (a) the input SSTables stay intact and readable,
605    //   (b) no orphan output component and no visible-but-not-durable output TOC
606    //       survive (and the `.compaction-tmp-*` dir is dropped), and
607    //   (c) finalize returns `Error::Storage`.
608
609    /// Which publication-barrier directory fsync to fault.
610    #[derive(Clone, Copy)]
611    enum FsyncFaultPoint {
612        /// Step 2b: after the non-TOC renames, before the TOC rename.
613        BeforeToc,
614        /// Step 2c: after the TOC rename, before the inputs are deleted.
615        AfterToc,
616    }
617
618    fn drive_compaction_with_fsync_fault(point: FsyncFaultPoint) {
619        use crate::error::Error;
620        use crate::storage::write_engine::compaction::{
621            FAIL_COMPACTION_DIR_FSYNC_AFTER_TOC, FAIL_COMPACTION_DIR_FSYNC_BEFORE_TOC,
622        };
623
624        let temp_dir = TempDir::new().unwrap();
625        let schema = create_test_schema();
626        let dir = sstable_dir(&temp_dir);
627        let data_dir = temp_dir.path().join("data");
628
629        // Four complete, published input generations (5 distinct rows each).
630        let mut engine = WriteEngine::new(config_for(&temp_dir)).unwrap();
631        let pre_paths = flush_n_sstables_sync(&mut engine, 4);
632        assert_eq!(pre_paths.len(), 4);
633
634        let count_suffix = |suffix: &str| -> usize {
635            dir_names(&dir)
636                .iter()
637                .filter(|n| n.ends_with(suffix))
638                .count()
639        };
640        assert_eq!(count_suffix("-big-Data.db"), 4, "sanity: 4 input Data.db");
641        assert_eq!(count_suffix("-big-TOC.txt"), 4, "sanity: 4 input TOCs");
642
643        // Arm the fsync fault at the requested publication point.
644        match point {
645            FsyncFaultPoint::BeforeToc => {
646                FAIL_COMPACTION_DIR_FSYNC_BEFORE_TOC.with(|f| f.set(true))
647            }
648            FsyncFaultPoint::AfterToc => FAIL_COMPACTION_DIR_FSYNC_AFTER_TOC.with(|f| f.set(true)),
649        }
650        let policy = crate::storage::write_engine::STCSPolicy::new(4, 32, 0.5, 1.5, 0).unwrap();
651        engine.set_merge_policy(Box::new(policy)).unwrap();
652        let result = engine.maintenance_step(Duration::from_secs(60));
653        // Disarm BOTH flags unconditionally so a failed assertion below cannot
654        // leak the injection into another test on this thread.
655        FAIL_COMPACTION_DIR_FSYNC_BEFORE_TOC.with(|f| f.set(false));
656        FAIL_COMPACTION_DIR_FSYNC_AFTER_TOC.with(|f| f.set(false));
657
658        // (c) finalize returns Error::Storage carrying the injected marker.
659        let err = result.expect_err("the injected publication fsync fault must fail finalize");
660        assert!(
661            matches!(&err, Error::Storage(msg) if msg.contains("injected directory fsync fault")),
662            "expected Error::Storage from the injected fsync fault, got: {err:?}"
663        );
664
665        // (a) The inputs are intact and still readable (never renamed away/deleted).
666        for p in &pre_paths {
667            assert!(
668                p.exists(),
669                "input {:?} must survive a rolled-back finalize",
670                p
671            );
672        }
673        let total: usize = pre_paths.iter().map(|p| live_row_count(p, &schema)).sum();
674        assert_eq!(
675            total, 20,
676            "all 20 input rows must still read back after the rollback"
677        );
678
679        // (b) The rollback removed every renamed file — including the TOC on the
680        // 2c path — so only the four input generations remain (no orphan output
681        // component, no visible output TOC), and the tmp dir is gone.
682        assert_eq!(
683            count_suffix("-big-Data.db"),
684            4,
685            "rollback must leave exactly the 4 input Data.db (no orphan output component)"
686        );
687        assert_eq!(
688            count_suffix("-big-TOC.txt"),
689            4,
690            "rollback must leave exactly the 4 input TOCs — no visible output TOC may survive"
691        );
692        let leftover_tmp: Vec<_> = std::fs::read_dir(&data_dir)
693            .unwrap()
694            .flatten()
695            .filter(|e| {
696                e.file_name()
697                    .to_string_lossy()
698                    .starts_with(".compaction-tmp-")
699            })
700            .collect();
701        assert!(
702            leftover_tmp.is_empty(),
703            "rollback must drop the .compaction-tmp-* dir, found {:?}",
704            leftover_tmp.iter().map(|e| e.path()).collect::<Vec<_>>()
705        );
706
707        drop(engine);
708    }
709
710    #[test]
711    fn publish_fsync_fault_before_toc_rolls_back_intact() {
712        drive_compaction_with_fsync_fault(FsyncFaultPoint::BeforeToc);
713    }

715    #[test]
716    fn publish_fsync_fault_after_toc_rolls_back_intact() {
717        drive_compaction_with_fsync_fault(FsyncFaultPoint::AfterToc);
718    }
