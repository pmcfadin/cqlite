1//! Clustering-group reconciliation kernel (issue #945 decomposition).
2//!
3//! `KWayMerger::reconcile_cluster_with_overlap_counted` used to be a single
4//! ~470-line function. This module hosts the same logic decomposed into named,
5//! separately testable steps threaded through a [`ReconcileState`] accumulator.
6//! The orchestrator in `mod.rs` calls the steps **in a fixed, parity-load-bearing
7//! order** — see [`ReconcileState`] for the contract. This is a pure refactor:
8//! the per-step bodies are lifted verbatim from the original `// Step N` blocks,
9//! so compaction output (and the #1037 purge tally) is byte-for-byte identical.
10//!
11//! ## Step order (do not reorder — parity-critical, Cassandra `f66fa14f`)
12//!
13//! 1. [`fold_row_deletions`](ReconcileState::fold_row_deletions) — Step 1:
14//!    effective row deletion (max `deletion_time` + paired LDT), carried
15//!    complex/range deletions, key/run_index.
16//! 2. [`resolve_cell_winners`](ReconcileState::resolve_cell_winners) — Step 2:
17//!    per-`(column, cell_path)` last-write-wins winner resolution.
18//! 3. [`apply_complex_deletions`](ReconcileState::apply_complex_deletions) —
19//!    Step 2b: strict-supersede + shadow-before-purge. MUST run before the
20//!    row-tombstone / gc-grace filters so a surviving complex deletion cannot
21//!    resurrect a covered element on a later purge.
22//! 4. [`shadow_by_row_deletion`](ReconcileState::shadow_by_row_deletion) —
23//!    Step 3: drop cells with `ts <= row_del`.
24//! 5. [`filter_dropped_columns`](ReconcileState::filter_dropped_columns) —
25//!    Step 3b: drop cells `ts <= drop_time`, capturing the pre-purge
26//!    `had_data_before` phantom-row guard state.
27//! 5b. [`expire_ttl_cells`](ReconcileState::expire_ttl_cells) — Step 3b′
28//!    (#1382): turn each expired live TTL cell (`localDeletionTime < now_secs`)
29//!    into a cell tombstone so Step 3c purges it under the same gc/overlap gate.
30//! 6. [`purge_gc_grace`](ReconcileState::purge_gc_grace) — Step 3c: drop
31//!    purgeable tombstones (overlap-safe), tallying genuine purges (#1037).
32//! 7. [`build`](ReconcileState::build) — Step 4: phantom-row guard + emit the
33//!    merged `MergeEntry`.
34

48/// Per-cell reconcile key: `(column, cell_path)` so each element of a multi-cell
49/// column reconciles independently (epic #899). Simple cells have
50/// `cell_path == None`.
51type CellKey = (String, Option<Vec<u8>>);
52
53/// In-flight working set for reconciling one clustering-key group.
54///
55/// Built by [`Self::new`], then mutated through the step methods in the fixed
56/// order documented at the module level. The load-bearing invariants the
57/// original ~470-line function depended on are preserved:
58///
59/// - **Step 2b before Step 3/3c** — a surviving complex deletion shadows its
60///   covered elements before any marker is purged (`f66fa14f`).
61/// - **`had_data_before` captured pre-purge** — recorded in
62///   [`Self::filter_dropped_columns`] (after row-tombstone + dropped-column
63///   filtering) and consulted in [`Self::build`] AFTER the gc-grace purge, so a
64///   clustered row whose only non-key data is a purgeable cell tombstone is
65///   recognized as data-less and emits nothing instead of a phantom key-only
66///   live row (#921 finding 3).
67pub(super) struct ReconcileState {
68    /// Clustering key shared by every entry in the group (carry-through).
69    clustering_key: Option<ClusteringKey>,
70    /// Partition key, taken from the first entry seen.
71    key: Option<DecoratedKey>,
72    /// Lowest run_index seen (newest file) for stable downstream ordering.
73    run_index: usize,
74    /// Effective row deletion timestamp (`markedForDeleteAt`, micros).
75    row_del: Option<i64>,
76    /// `localDeletionTime` (GC-clock seconds) paired with the winning `row_del`.
77    row_del_ldt: i32,
78    /// First-seen cell-key order for deterministic output.
79    order: Vec<CellKey>,
80    /// Winning cell per `(column, cell_path)`.
81    winners: HashMap<CellKey, CellData>,
82    /// Carried complex (collection / UDT) deletion markers.
83    complex_deletions: Vec<ComplexDeletion>,
84    /// Carried range deletion (timestamp-max preserved).
85    range_deletion: Option<RangeTombstone>,
86    /// Cells surviving row-tombstone shadowing (Step 3 output).
87    after_row_del: Vec<CellData>,
88    /// Cells surviving dropped-column filtering (Step 3b output), then mutated
89    /// by the gc-grace purge.
90    surviving: Vec<CellData>,
91    /// Whether the row held non-key data BEFORE the gc-grace purge (phantom-row
92    /// guard, captured pre-purge).
93    had_data_before: bool,
94}

97    /// Initialize an empty accumulator for the group sharing `clustering_key`.
98    pub(super) fn new(clustering_key: Option<ClusteringKey>) -> Self {
99        Self {
100            clustering_key,
101            key: None,
102            run_index: usize::MAX,
103            row_del: None,
104            row_del_ldt: 0,
105            order: Vec::new(),
106            winners: HashMap::new(),
107            complex_deletions: Vec::new(),
108            range_deletion: None,
109            after_row_del: Vec::new(),
110            surviving: Vec::new(),
111            had_data_before: false,
112        }
113    }

115    /// True once an input entry has been folded (i.e. the group is non-empty).
116    /// Mirrors the original `let key = key?` early-out: an empty group emits
117    /// nothing.
118    pub(super) fn has_key(&self) -> bool {
119        self.key.is_some()
120    }

122    /// Step 1 — fold the per-entry deletion metadata: effective row deletion
123    /// (max `deletion_time` with its PAIRED source LDT, #873), the carried
124    /// complex-deletion union (first-seen, de-duplicated) and the
125    /// timestamp-max range deletion, plus the carry-through key / run_index.
126    ///
127    /// The per-cell winner resolution (the `RowData::Live` arm) is split out
128    /// into [`Self::resolve_cell_winners`]; both passes iterate `cluster_rows`
129    /// in heap-routing order, and the accumulators are independent, so the
130    /// result is identical to the original single combined loop.
131    pub(super) fn fold_row_deletions(&mut self, cluster_rows: &[MergeEntry]) {
132        for entry in cluster_rows {
133            if self.key.is_none() {
134                self.key = Some(entry.key.clone());
135            }
136            self.run_index = self.run_index.min(entry.run_index);
137
138            for cd in &entry.complex_deletions {
139                if !self.complex_deletions.contains(cd) {
140                    self.complex_deletions.push(cd.clone());
141                }
142            }
143            if let Some(rd) = &entry.range_deletion {
144                let replace = match &self.range_deletion {
145                    None => true,
146                    Some(current) => rd.deletion_time > current.deletion_time,
147                };
148                if replace {
149                    self.range_deletion = Some(rd.clone());
150                }
151            }
152
153            // Issue #932: a coexisting row deletion carried on a LIVE entry (a
154            // read-back row that held BOTH a row tombstone and surviving newer
155            // cells) contributes to the winning row deletion exactly as a
156            // standalone `RowData::Tombstone` does — capture its paired LDT when
157            // it sets the new max so the rebuilt deletion preserves its
158            // wall-clock `localDeletionTime` (#873).
159            if let Some((del_ts, del_ldt)) = entry.row_deletion {
160                if self.row_del.is_none_or(|d| del_ts > d) {
161                    self.row_del = Some(del_ts);
162                    self.row_del_ldt = del_ldt;
163                }
164            }
165
166            if let RowData::Tombstone {
167                deletion_time,
168                local_deletion_time,
169            } = &entry.row_data
170            {
171                // When this tombstone's deletion_time becomes (or sets) the new
172                // max, capture its paired LDT too so the winning tombstone's
173                // source `localDeletionTime` survives reconciliation (#873).
174                if self.row_del.is_none_or(|d| *deletion_time > d) {
175                    self.row_del = Some(*deletion_time);
176                    self.row_del_ldt = *local_deletion_time;
177                }
178            }
179        }
180    }
181
182    /// Step 2 — per-cell winner resolution keyed by `(column, cell_path)` so
183    /// each element of a multi-cell column reconciles independently (epic #899).
184    /// Preserves first-seen key order while resolving winners in `winners`.
185    ///
186    /// `cluster_rows` is in heap-routing order (run_index ascending within equal
187    /// keys), so when two cells tie on both timestamp and liveness the
188    /// first-seen (newer file) is kept (see
189    /// [`reconcile_rules::cell_wins`], the shared tie-break rule).
190    pub(super) fn resolve_cell_winners(&mut self, cluster_rows: &[MergeEntry]) {
191        for entry in cluster_rows {
192            if let RowData::Live { cells } = &entry.row_data {
193                for cell in cells {
194                    let cell_key: CellKey = (cell.column.clone(), cell.cell_path.clone());
195                    match self.winners.get(&cell_key) {
196                        None => {
197                            self.order.push(cell_key.clone());
198                            self.winners.insert(cell_key, cell.clone());
199                        }
200                        Some(existing) => {
201                            // Higher timestamp wins. At EQUAL timestamp a cell
202                            // DELETION (tombstone) beats a LIVE or EXPIRING
203                            // (TTL) cell, decided BEFORE any localDeletionTime
204                            // compare (parity Cassandra `a62c749`,
205                            // `Cells#reconcile`; issue #848 / #498). At equal ts
206                            // + equal deletion-status, keep the first-seen
207                            // (newer file) winner. The tie-break is the SHARED
208                            // [`reconcile_rules::cell_wins`] rule (issue #947),
209                            // also used by the flush/write path.
210                            if reconcile_rules::cell_wins(cell, existing) {
211                                self.winners.insert(cell_key, cell.clone());
212                            }
213                        }
214                    }
215                }
216            }
217        }
218    }
219
220    /// Step 2b — COMPLEX-DELETION reconcile: strict-supersede + shadow-before-purge
221    /// (issue #887). Runs AFTER per-cell winner resolution and BEFORE the
222    /// row-tombstone / dropped-column filters, so an element shadowed by a
223    /// surviving complex deletion cannot be resurrected by a later purge of the
224    /// deletion marker.
225    ///
226    /// 1. STRICT SUPERSEDE (parity Cassandra `bd244649`): per complex column the
227    ///    ACTIVE deletion is the one with the greatest `marked_for_delete_at`. A
228    ///    merged deletion supersedes the active one only when its mfda is STRICTLY
229    ///    GREATER — EQUAL timestamps do NOT supersede. Reduce the carried
230    ///    first-seen union to ONE deletion per column NAME (the strict max),
231    ///    preserving first-seen column order for deterministic output. Matched BY
232    ///    NAME (`ComplexDeletion.column`), never by header identity (consistent
233    ///    with #888's match-by-name substrate).
234    ///
235    /// 2. SHADOW BEFORE PURGE (parity Cassandra `f66fa14f`): for the surviving
236    ///    deletion on a column, drop every per-element winner of THAT column
237    ///    (matched by NAME, only complex ELEMENTS carrying a `cell_path`) whose own
238    ///    timestamp is `<= marked_for_delete_at`. The boundary mirrors the
239    ///    cell-vs-deletion rule (#498): an element with ts STRICTLY GREATER than
240    ///    mfda survives; `<=` is shadowed. A simple cell sharing the name is never
241    ///    collapsed by a complex marker.
242    pub(super) fn apply_complex_deletions(&mut self) {
243        if !self.complex_deletions.is_empty() {
244            // Strict-supersede: collapse to the max-mfda deletion per column name.
245            let mut active: HashMap<String, ComplexDeletion> = HashMap::new();
246            let mut active_order: Vec<String> = Vec::new();
247            for cd in self.complex_deletions.drain(..) {
248                match active.get_mut(&cd.column) {
249                    None => {
250                        active_order.push(cd.column.clone());
251                        active.insert(cd.column.clone(), cd);
252                    }
253                    // STRICTLY GREATER supersedes; equal/lesser does NOT
254                    // (bd244649). Shared rule (issue #947).
255                    Some(existing)
256                        if reconcile_rules::complex_deletion_supersedes(
257                            cd.marked_for_delete_at,
258                            existing.marked_for_delete_at,
259                        ) =>
260                    {
261                        *existing = cd;
262                    }
263                    Some(_) => {}
264                }
265            }
266
267            // Shadow-before-purge: drop covered per-element winners of each column
268            // whose ts <= mfda (matched by NAME), so purging the marker later cannot
269            // resurrect them (f66fa14f). Elements strictly newer than mfda survive.
270            for column in &active_order {
271                if let Some(cd) = active.get(column) {
272                    let mfda = cd.marked_for_delete_at;
273                    let winners = &mut self.winners;
274                    self.order.retain(|cell_key| {
275                        let (cell_column, cell_path) = cell_key;
276                        // Only complex elements (those with a cell_path) of THIS
277                        // column are candidates for shadowing.
278                        if cell_column != column || cell_path.is_none() {
279                            return true;
280                        }
281                        match winners.get(cell_key) {
282                            // ts > mfda survives; ts <= mfda is shadowed (purged).
283                            // Shared shadow-before-purge boundary (issue #947).
284                            Some(cell)
285                                if reconcile_rules::element_survives_complex_deletion(
286                                    cell.timestamp,
287                                    mfda,
288                                ) =>
289                            {
290                                true
291                            }
292                            Some(_) => {
293                                winners.remove(cell_key);
294                                false
295                            }
296                            None => false,
297                        }
298                    });
299                }
300            }
301
302            // Rebuild the carried union as the surviving per-column deletions in
303            // first-seen column order.
304            self.complex_deletions = active_order
305                .into_iter()
306                .filter_map(|column| active.remove(&column))
307                .collect();
308        }
309    }

311    /// Step 3 — apply row-tombstone shadowing per cell. A cell whose timestamp is
312    /// `<= row_del` is shadowed (`<=` lets the tombstone win at equal ts, #498).
313    /// Cells written strictly after `row_del` survive. This shadowing applies to
314    /// cell tombstones too: a row tombstone at ts=T supersedes a cell tombstone at
315    /// ts<=T (real Cassandra semantics).
316    ///
317    /// COMPLEX-DELETION shadowing has already been applied in Step 2b above,
318    /// BEFORE this row-tombstone filter, so a surviving complex deletion cannot
319    /// resurrect a covered element on a later purge (issue #887, parity
320    /// `f66fa14f`). The output `after_row_del` is kept so the dropped-column
321    /// stage (Step 3b) can tell whether its purge is what emptied the row of real
322    /// data (phantom-row guard).
323    pub(super) fn shadow_by_row_deletion(&mut self, purges: &mut PurgeCounts) {
324        let row_del = self.row_del;
325        // Issue #2163 (roborev r5): clustering-key pseudo-cells are intentionally
326        // RETAINED in the cell list for read-back (see `extract_clustering_key`
327        // and the `had_data_before` computation in `filter_dropped_columns`
328        // below) — they are not real data, so shadowing one must not inflate
329        // `tombstones_suppressed`. Build the SAME `is_data_cell` exclusion
330        // `filter_dropped_columns`/`build` already use, computed here (before the
331        // mutable `winners` borrow) so a clustering-key cell shadowed by the row
332        // tombstone is excluded from the count.
333        let ck_names: HashSet<&str> = self
334            .clustering_key
335            .as_ref()
336            .map(|ck| ck.columns.iter().map(|(n, _)| n.as_str()).collect())
337            .unwrap_or_default();
338        let winners = &mut self.winners;
339        self.after_row_del = std::mem::take(&mut self.order)
340            .into_iter()
341            .filter_map(|cell_key| winners.remove(&cell_key))
342            .filter(|cell| match row_del {
343                Some(d) => {
344                    let survives = cell.timestamp > d;
345                    // Issue #2163: a LIVE DATA cell dropped here was SHADOWED by
346                    // the row tombstone (suppressed) — count it, distinct from a
347                    // gc/overlap-safe purge. A cell that is itself a tombstone,
348                    // or a clustering-key pseudo-cell (retained for read-back,
349                    // never real data), is not "live data suppressed" and is not
350                    // counted.
351                    if !survives
352                        && !KWayMerger::is_cell_tombstone(cell)
353                        && !ck_names.contains(cell.column.as_str())
354                    {
355                        purges.suppressed += 1;
356                    }
357                    survives
358                }
359                None => true,
360            })
361            .collect();
362    }

364    /// Step 3b — dropped-column filtering (Cassandra `cb34ad47`,
365    /// `compaction.purge`). A column dropped at `drop_time` discards every cell
366    /// whose `timestamp <= drop_time`; a cell written strictly after the drop
367    /// (the column was re-added) survives. Scoped per column via the
368    /// `dropped_columns` map (#904 plumbing, #847 filter).
369    ///
370    /// EXACT PER-CELL GRANULARITY (#922): `cell.timestamp` is the cell's OWN
371    /// writetime, so a row mixing a pre-drop dropped-column cell with a post-drop
372    /// cell of ANOTHER column purges the dropped cell by its own writetime while
373    /// the sibling survives by its.
374    ///
375    /// Also captures the pre-purge `had_data_before` for the phantom-row guard:
376    /// clustering-key columns are intentionally left in the cell list (see
377    /// `extract_clustering_key`) for read-back, so we must remember whether the
378    /// row HAD non-key data before the gc-grace purge runs (the
379    /// `has_data_after` / `purged_to_empty` determination is DEFERRED to
380    /// [`Self::build`], after Step 3c — #921 finding 3).
381    pub(super) fn filter_dropped_columns(&mut self, dropped_columns: &HashMap<String, i64>) {
382        self.surviving = self
383            .after_row_del
384            .iter()
385            .filter(|cell| match dropped_columns.get(&cell.column) {
386                Some(drop_time) => cell.timestamp > *drop_time,
387                None => true,
388            })
389            .cloned()
390            .collect();
391
392        let ck_names: HashSet<&str> = self
393            .clustering_key
394            .as_ref()
395            .map(|ck| ck.columns.iter().map(|(n, _)| n.as_str()).collect())
396            .unwrap_or_default();
397        let is_data_cell = |cell: &CellData| !ck_names.contains(cell.column.as_str());
398        self.had_data_before = self.after_row_del.iter().any(is_data_cell);
399    }

401    /// Step 3b′ — TTL EXPIRY (issue #1382, parity Cassandra `ExpiringCell`
402    /// / `TTLExpiryTest`). Runs AFTER dropped-column filtering and BEFORE the
403    /// gc-grace purge (Step 3c) so an expired cell is first turned into a cell
404    /// tombstone and then purged by the SAME gc/overlap gate as any other cell
405    /// tombstone — matching Cassandra, which treats an expiring cell whose
406    /// `localDeletionTime` is past as a tombstone with `localDeletionTime =`
407    /// the expiry instant and the cell's own `markedForDeleteAt` unchanged.
408    ///
409    /// NO-HEURISTICS (#28): a cell is "expiring" ONLY when the reader surfaced
410    /// its authoritative expiry metadata — BOTH `ttl` (Some) AND
411    /// `local_deletion_time` (Some) — from the on-disk cell (never inferred
412    /// from the value's byte shape). `now_secs` is the compaction's pinned
413    /// evaluation instant (`None` disables expiry: a strict no-op).
414    ///
415    /// A cell that is ALREADY a tombstone (`is_cell_tombstone`, which also covers
416    /// an `is_deleted` complex-element tombstone) is left untouched — it has no
417    /// live value to expire.
418    ///
419    /// COMPLEX / COLLECTION / UDT ELEMENTS (#1537): an expiring ELEMENT of a
420    /// non-frozen complex column (`is_complex_element`) is expired here too, not
421    /// just simple cells (the #1382 scope gap — F2). Cassandra
422    /// `AbstractCell.purge(DeletionPurger, nowInSec)` treats an expired expiring
423    /// cell UNIFORMLY whether simple or complex: it converts it to
424    /// `BufferCell.tombstone(column, timestamp(), localDeletionTime() - ttl(), path())` —
425    /// PRESERVING the element's cell path and setting the tombstone's
426    /// `localDeletionTime` to `ldt - ttl` (the cell's creation time). So an
427    /// expired element becomes an element-level tombstone at the SAME path,
428    /// carrying its own IS_DELETED flag (epic #899), which the per-element writer
429    /// emits as an element tombstone.
430    ///
431    /// LDT NORMALIZATION (#921 finding 2): on-disk `localDeletionTime` is an
432    /// UNSIGNED 32-bit GC-clock second count carried as a wrapped `i32`;
433    /// reinterpret the bits as unsigned (`i64::from(ldt as u32)`) before the
434    /// compare so a far-future expiry is not treated as already-past.
435    pub(super) fn expire_ttl_cells(&mut self, now_secs: Option<i64>) {
436        let Some(now) = now_secs else {
437            return; // Expiry disabled — strict no-op.
438        };
439        for cell in &mut self.surviving {
440            // An existing tombstone (simple cell tombstone OR a complex-element
441            // tombstone via `is_deleted`) has no live value to expire.
442            if KWayMerger::is_cell_tombstone(cell) {
443                continue;
444            }
445            let (Some(ttl), Some(ldt)) = (cell.ttl, cell.local_deletion_time) else {
446                continue; // Not an authoritative expiring cell.
447            };
448            // PARITY — off-by-one at `now == ldt` (issue #1537). Cassandra
449            // `Cell.isLive(nowInSec)` (`Cell.java`) is LIVE iff
450            // `nowInSec < localDeletionTime`, so an expiring cell is EXPIRED iff
451            // `localDeletionTime <= nowInSec`. Keep the cell LIVE only when its
452            // expiry instant (unsigned GC-clock seconds) is STRICTLY in the
453            // future; `ldt == now` means already expired.
454            if i64::from(ldt as u32) > now {
455                continue; // Not yet expired (strictly-future expiry).
456            }
457            // PARITY — tombstone `localDeletionTime` == `ldt - ttl` (issue #1537).
458            // Cassandra `AbstractCell.purge` converts an expired expiring cell via
459            // `BufferCell.tombstone(column, timestamp(), localDeletionTime() - ttl(),
460            // path())`. CQLite stores an expiring cell's on-disk
461            // `local_deletion_time` as the EXPIRY instant (`now + ttl`), so the
462            // parity-correct tombstone value is `ldt - ttl` = the cell's
463            // creation-time-in-seconds. gc grace is therefore measured from the
464            // cell's creation time (see the comment in `AbstractCell.purge`).
465            //
466            // GUARD: this cannot underflow for a VALID expiring cell — `ldt`
467            // (= creation + ttl) is always `>= ttl`. A `saturating_sub` on the
468            // unsigned seconds keeps a malformed pair from ever panicking (it
469            // floors at 0, an ancient/purgeable tombstone) with no `unwrap`.
470            let creation_secs: u32 = (ldt as u32).saturating_sub(ttl);
471            let tombstone_ldt: i64 = i64::from(creation_secs);
472            // Convert the expired live cell into a (cell / complex-element)
473            // tombstone whose `localDeletionTime` is the creation-time instant
474            // (`ldt - ttl`) and whose `markedForDeleteAt` is the cell's own write
475            // timestamp (unchanged). Step 3c then purges it exactly like any other
476            // cell tombstone once its LDT is < gcBefore and the overlap gate
477            // allows. `ttl` is cleared: a tombstone carries no TTL.
478            cell.value = crate::types::Value::Tombstone(Box::new(crate::types::TombstoneInfo {
479                deletion_time: cell.timestamp,
480                tombstone_type: crate::types::TombstoneType::CellTombstone,
481                local_deletion_time: tombstone_ldt,
482                ttl: None,
483                range_start: None,
484                range_end: None,
485            }));
486            cell.ttl = None;
487            // A complex ELEMENT additionally carries its deletion via the
488            // authoritative IS_DELETED flag (epic #899). Set it so the per-element
489            // writer emits an element tombstone (preserving the element's
490            // cell_path, Cassandra `path()`) rather than a live element, and so the
491            // gc-grace purge's `cell.is_deleted` branch (Step 3c (a)) treats it as
492            // a purgeable complex-element tombstone. The cell path is untouched.
493            if cell.is_complex_element {
494                cell.is_deleted = true;
495            }
496            // PARITY: the converted tombstone's `localDeletionTime` is the
497            // creation-time instant (`ldt - ttl`), so update `cell.local_deletion_time`
498            // to match. `cell_effective_ldt` prefers this field, so both the
499            // gc-grace purge (Step 3c) and the writer surface the creation-time
500            // LDT — measuring gc grace from creation, per `AbstractCell.purge`.
501            // Stored as the wrapped `i32` GC-clock value (re-widened unsigned by
502            // `cell_effective_ldt`); `creation_secs` fits `i32` for any real
503            // creation time.
504            cell.local_deletion_time = Some(creation_secs as i32);
505        }
506    }
507
508    /// Step 3c — gc_grace / gcBefore tombstone PURGING (issue #845, parity
509    /// Cassandra `8d47ebb2`). A tombstone whose on-disk `localDeletionTime`
510    /// (GC-clock seconds) is STRICTLY LESS THAN `gcBefore` is purgeable and
511    /// dropped; one within grace (`>= gcBefore`) is retained. Runs AFTER the
512    /// complex-deletion shadow-before-purge (Step 2b) and the row-tombstone /
513    /// dropped-column filters above, so the now-redundant marker cannot
514    /// resurrect data within this set.
515    ///
516    /// OVERLAP SAFETY (#921 finding 1, #935): `gc_before_secs` is the EFFECTIVE
517    /// cutoff (caller collapses it to `None` for an unbounded partial compaction,
518    /// making this a strict no-op). With a bound, a tombstone is purged only when
519    /// BOTH its gc grace has elapsed AND its own deletion timestamp
520    /// (`markedForDeleteAt`) is STRICTLY LESS THAN `max_purgeable_timestamp` — so
521    /// it provably shadows nothing in a non-included overlapping SSTable.
522    ///
523    /// LDT NORMALIZATION (#921 finding 2): on-disk `localDeletionTime` is an
524    /// UNSIGNED 32-bit GC-clock second count carried as a wrapped `i32`;
525    /// reinterpret the bits as unsigned (`i64::from(ldt as u32)`) before every
526    /// compare so a far-future tombstone is not purged immediately. All three
527    /// sites below (cell / row / complex-deletion), plus the expiring-cell check
528    /// in `expire_ttl_cells`, MUST widen unsigned — a signed compare would treat
529    /// a value in `[2^31, 2^32)` (a negative `i32`, e.g. `i32::MIN == 2^31`) as
530    /// ANCIENT and purge a not-yet-expired far-future tombstone, resurrecting any
531    /// older cells it shadows. Pinned by `tests/issue_1386_wrapped_negative_ldt.rs`.
532    ///
533    /// DIVERGENCE FROM CASSANDRA — RECORDED DECISION (#1386): CQLite SILENTLY
534    /// reinterprets the LDT bits as unsigned (year-2106 GC-clock semantics) and
535    /// never flags the value as "suspect". Cassandra 5.0's deletion-time handling
536    /// additionally treats an out-of-range `localDeletionTime` as suspicious
537    /// (e.g. `Cell`/`DeletionTime` validation paths log / mark it). CQLite's
538    /// posture is deliberately reinterpret-unsigned WITHOUT suspect-marking: a
539    /// far-future tombstone round-trips and purges purely on the unsigned GC-clock
540    /// compare, matching Cassandra's PURGE DECISION while omitting its diagnostics.
541    ///
542    /// Each genuine gc/overlap-safe purge increments the matching `purges` field
543    /// (issue #1037); last-write-wins reconciliation collapse is NOT counted.
544    pub(super) fn purge_gc_grace(
545        &mut self,
546        gc_before_secs: Option<i64>,
547        max_purgeable_timestamp: i64,
548        purges: &mut PurgeCounts,
549    ) {
550        if let Some(gc_before) = gc_before_secs {
551            // (a) Cell tombstones: drop any purgeable simple cell tombstone (and
552            // purgeable complex-element tombstone) from the surviving set. A cell
553            // whose `local_deletion_time` is not surfaced (`None`) is conservative-
554            // ly RETAINED — we never purge on unknown LDT (no-heuristics mandate).
555            self.surviving.retain(|cell| {
556                if KWayMerger::is_cell_tombstone(cell) || cell.is_deleted {
557                    // #921 finding 1: a simple cell tombstone surfaces its LDT in
558                    // its `Value::Tombstone` payload, not `CellData.local_deletion_time`
559                    // (which the reader fills only for expiring cells). Consult both
560                    // via `cell_effective_ldt` so a purgeable cell tombstone is
561                    // actually purged here — matching the survivor pre-pass.
562                    let gc_purgeable = match KWayMerger::cell_effective_ldt(cell) {
563                        Some(ldt) => i64::from(ldt as u32) < gc_before,
564                        // Unknown LDT: never purge (no-heuristics mandate).
565                        None => false,
566                    };
567                    // #935 overlap gate: the cell tombstone's own write timestamp
568                    // (`markedForDeleteAt`) must be STRICTLY BELOW the min outside
569                    // timestamp to prove it shadows nothing in a non-included
570                    // overlapping SSTable. `i64::MAX` for a full compaction lets
571                    // every gc-purgeable tombstone through unchanged.
572                    let overlap_purgeable = cell.timestamp < max_purgeable_timestamp;
573                    // RETAIN unless BOTH gates allow the purge.
574                    let keep = !(gc_purgeable && overlap_purgeable);
575                    if !keep {
576                        // True gc/overlap-safe purge of a cell tombstone (simple
577                        // or complex-element), issue #1037.
578                        purges.cell_tombstones += 1;
579                    }
580                    keep
581                } else {
582                    true
583                }
584            });
585
586            // (b) Row tombstone: a purgeable row deletion is dropped so no
587            // tombstone entry is emitted for it. Row-tombstone shadowing already
588            // ran above (`after_row_del`), so cells it covered are gone.
589            //
590            // UNKNOWN-LDT RETENTION (#921 finding 2): `row_del_ldt == 0` is the
591            // "LDT not surfaced" placeholder; treat 0 as UNKNOWN and RETAIN the
592            // row tombstone. Only a real, non-zero, surfaced LDT strictly below
593            // `gcBefore` purges.
594            //
595            // #935 overlap gate: also require the row deletion's own timestamp
596            // (`markedForDeleteAt` = `row_del`) to be STRICTLY BELOW the min
597            // outside timestamp so it shadows nothing in a non-included
598            // overlapping SSTable. `i64::MAX` (full compaction) is a no-op.
599            if self.row_del_ldt != 0
600                && i64::from(self.row_del_ldt as u32) < gc_before
601                && self.row_del.is_some_and(|d| d < max_purgeable_timestamp)
602            {
603                self.row_del = None;
604                // True gc/overlap-safe purge of a row tombstone (issue #1037).
605                purges.row_tombstones += 1;
606            }
607
608            // (c) Complex-deletion markers: drop each purgeable marker. The
609            // strict-supersede reduction + shadow-before-purge already ran in
610            // Step 2b, so a covered element is gone before its marker is purged.
611            // #935 overlap gate: also require the marker's `marked_for_delete_at`
612            // to be STRICTLY BELOW the min outside timestamp.
613            self.complex_deletions.retain(|cd| {
614                let gc_purgeable = i64::from(cd.local_deletion_time as u32) < gc_before;
615                let overlap_purgeable = cd.marked_for_delete_at < max_purgeable_timestamp;
616                let keep = !(gc_purgeable && overlap_purgeable);
617                if !keep {
618                    // True gc/overlap-safe purge of a complex-deletion marker
619                    // (issue #1037).
620                    purges.complex_deletions += 1;
621                }
622                keep
623            });
624        }
625    }
626
627    /// Step 4 — build the merged result, applying the phantom key-only-row guard
628    /// (#921 finding 3): recompute `purged_to_empty` AFTER the gc-grace
629    /// cell-tombstone purge, so a CLUSTERED row whose only non-key data was a
630    /// purgeable cell tombstone emits NOTHING instead of a phantom key-only live
631    /// row. A row that was always key-only (a genuine row marker) keeps
632    /// `had_data_before == false` and is preserved.
633    ///
634    /// Returns `None` for an empty group (the original `let key = key?`
635    /// early-out) or a truly absent row.
636    pub(super) fn build(self, purges: &mut PurgeCounts) -> Option<MergeEntry> {
637        let ReconcileState {
638            clustering_key,
639            key,
640            run_index,
641            row_del,
642            row_del_ldt,
643            complex_deletions,
644            range_deletion,
645            surviving,
646            had_data_before,
647            ..
648        } = self;
649
650        let key = key?; // empty group => nothing to emit
651
652        let ck_names: HashSet<&str> = clustering_key
653            .as_ref()
654            .map(|ck| ck.columns.iter().map(|(n, _)| n.as_str()).collect())
655            .unwrap_or_default();
656        let is_data_cell = |cell: &CellData| !ck_names.contains(cell.column.as_str());
657        let has_data_after = surviving.iter().any(is_data_cell);
658        let purged_to_empty = had_data_before && !has_data_after;
659        drop(ck_names);
660
661        // Issue #2163 (roborev r7): a cell tombstone (simple or complex-element,
662        // the SAME predicate `purge_gc_grace`'s retain closure uses) that
663        // SURVIVES into `surviving` — whether because gc-grace/overlap kept it,
664        // or because purging was inactive for this merge entirely (no
665        // `gc_before_secs`, e.g. an unbounded partial compaction) — is a marker
666        // RETAINED into the output, matching the row/range/partition tombstone
667        // "emitted" contract established above. Counted here from the FINAL
668        // surviving set (not inside `purge_gc_grace`'s conditional retain) so a
669        // merge with purging disabled still counts its retained cell
670        // tombstones; a cell tombstone always satisfies `is_data_cell` (it is
671        // never a clustering-key pseudo-cell), so its presence here already
672        // implies `purged_to_empty == false` and the row below is genuinely
673        // emitted with it intact.
674        let retained_cell_tombstones = surviving
675            .iter()
676            .filter(|cell| KWayMerger::is_cell_tombstone(cell) || cell.is_deleted)
677            .count() as u64;
678        purges.emitted += retained_cell_tombstones;
679
680        // Attach the carried deletion metadata to whichever entry is emitted so it
681        // is not dropped by reconciliation (#886 plumbing preservation).
682        let has_carried_metadata = !complex_deletions.is_empty() || range_deletion.is_some();
683
684        // Emit a live row only when real data survives. `!purged_to_empty`
685        // suppresses a clustered row whose only data was a dropped column (phantom
686        // key-only row); a genuine row marker (always key-only) has
687        // `had_data_before == false` so it is preserved.
688        let built = if !surviving.is_empty() && !purged_to_empty {
689            // `surviving` is non-empty, so `max()` is `Some`; `unwrap_or(0)` only
690            // guards the type and never triggers.
691            let row_ts = surviving.iter().map(|c| c.timestamp).max().unwrap_or(0);
692            let live = MergeEntry::new(
693                run_index,
694                key,
695                clustering_key,
696                row_ts,
697                RowData::Live { cells: surviving },
698            );
699            // Issue #932: a row deletion that survived purging COEXISTS with the
700            // surviving (strictly-newer) cells. Carry it on the live entry so the
701            // merge→mutation step emits a `HAS_DELETION` row holding both — the row
702            // deletion keeps shadowing older cells of OTHER columns living in
703            // SSTables not part of a partial compaction, preventing resurrection.
704            // Step 3 already dropped the cells this deletion covers (ts <= row_del),
705            // so the deletion shadows nothing within `surviving`.
706            Some(match row_del {
707                Some(deletion_time) => {
708                    // Issue #2163: this row-tombstone marker is RETAINED alongside
709                    // newer live cells (the common "retained-but-coexisting" case,
710                    // NOT the sole-output case below) — count it as emitted too, so
711                    // `tombstones_emitted` covers every marker that survives
712                    // reconciliation into the output, not only the row-absent case.
713                    purges.emitted += 1;
714                    live.with_row_deletion(deletion_time, row_del_ldt)
715                }
716                None => live,
717            })
718        } else if let Some(deletion_time) = row_del {
719            // No surviving data. If a row tombstone exists, keep the row shadowed
720            // so downstream still emits the deletion (preserves #505/#498 absence).
721            // Issue #2163: this row-tombstone marker is RETAINED into the output —
722            // count it as emitted (a marker carried forward, not purged).
723            purges.emitted += 1;
724            Some(MergeEntry::new(
725                run_index,
726                key,
727                clustering_key,
728                deletion_time,
729                RowData::Tombstone {
730                    deletion_time,
731                    // Preserve the source LDT paired with the winning deletion_time
732                    // (#873) instead of resetting it to 0; the writer encodes it
733                    // verbatim and gc_grace decisions stay faithful.
734                    local_deletion_time: row_del_ldt,
735                },
736            ))
737        } else if has_carried_metadata {
738            // No surviving data AND no row tombstone, but the cluster still carries
739            // complex/range deletion metadata. Emit a metadata-only entry (an empty
740            // `Live` row) so it survives reconciliation and reaches downstream
741            // consumers (#844/#846/#899).
742            Some(MergeEntry::new(
743                run_index,
744                key,
745                clustering_key,
746                0,
747                RowData::Live { cells: vec![] },
748            ))
749        } else {
750            // Truly empty/absent row.
751            None
752        };
753
754        built.map(|entry| {
755            let entry = if complex_deletions.is_empty() {
756                entry
757            } else {
758                entry.with_complex_deletions(complex_deletions)
759            };
760            match range_deletion {
761                Some(rd) => entry.with_range_deletion(rd),
762                None => entry,
763            }
764        })
765    }
