1//! Minimal process-wide counter for Statistics.db Table-of-Contents (TOC) walks
2//! (issue #1658, epic #1606).
3//!
4//! The metadata stack re-walks the Statistics.db TOC several times during a
5//! single SSTable open — once to locate the `HEADER` (SerializationHeader)
6//! offset (`enhanced_statistics_parser::header::parse_statistics_toc_for_header_offset`)
7//! and again inside `repair_metadata::stats_component_bounds`, which
8//! `read_table_counts` and `parse_stats_extras` each invoke while building
9//! `SSTableStatistics`. This counter lets the A5 cold-open bench
10//! (`benches/open.rs`) MEASURE that redundancy so a fix can be scoped against a
11//! real number rather than a guess. It is NOT a decoding heuristic: nothing in
12//! the parse path reads the count, so it has no effect on correctness (#28).
13//!
14//! The instrumentation is gated behind `cli-helpers` (roborev #1658): the
15//! counter is only read by the cli-helpers-gated bench, so production builds
16//! (default features) carry zero overhead — `record_toc_walk()` compiles to an
17//! empty inline no-op.
18
19/// Record that a Statistics.db TOC was walked once. Called at each TOC-walk site.
20/// Gated behind cli-helpers so production builds carry zero overhead (roborev #1658).
21#[inline]
22#[cfg(feature = "cli-helpers")]
23pub(crate) fn record_toc_walk() {
24    gated::TOC_WALKS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
25}
26
27/// No-op in production builds (when cli-helpers is disabled).
28#[inline]
29#[cfg(not(feature = "cli-helpers"))]
30pub(crate) fn record_toc_walk() {
31    // Empty — zero overhead in default/production builds.
32}
33
34// The counter and its accessors are only compiled when cli-helpers is enabled
35// (the bench that reads them is also gated on cli-helpers).
36#[cfg(feature = "cli-helpers")]
37mod gated {
38    use std::sync::atomic::{AtomicU64, Ordering};
39
40    /// Process-wide count of Statistics.db TOC walks since the last reset.
41    pub(super) static TOC_WALKS: AtomicU64 = AtomicU64::new(0);
42
43    /// Read the current TOC-walk count (process-wide, since the last reset).
44    pub fn toc_walk_count() -> u64 {
45        TOC_WALKS.load(Ordering::Relaxed)
46    }
47
48    /// Reset the TOC-walk counter to zero and return the value it held. Lets a
49    /// caller (e.g. the cold-open bench) measure the walks of a single open in
50    /// isolation: reset, do one open, then read `toc_walk_count()`.
51    pub fn reset_toc_walk_count() -> u64 {
52        TOC_WALKS.swap(0, Ordering::Relaxed)
53    }
54}
55
56#[cfg(feature = "cli-helpers")]
57pub use gated::{reset_toc_walk_count, toc_walk_count};
58
59// When cli-helpers is disabled, provide stub functions so the bench module
60// (which is itself gated on cli-helpers) still compiles if accidentally reached.
61#[cfg(not(feature = "cli-helpers"))]
62pub fn toc_walk_count() -> u64 {
63    0
64}
65
66#[cfg(not(feature = "cli-helpers"))]
67pub fn reset_toc_walk_count() -> u64 {
68    0
