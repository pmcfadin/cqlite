1//! Aggregation execution for the SELECT executor (issue #1116 split; issue #1578).
2//!
3//! Two paths live here:
4//! - [`SelectExecutor::execute_aggregation`] — the buffered GROUP BY / global
5//!   aggregate accumulation over an already-materialized `Vec<QueryRow>`.
6//! - [`SelectExecutor::try_execute_global_aggregate`] — the issue #1578 (D2)
7//!   O(1) streaming fold: a GROUP-BY-free aggregate over a full table scan folds
8//!   each scanned row into a single running accumulator and drops the row, so
9//!   `COUNT(*)`/`MIN`/`MAX`/`SUM`/`AVG` never buffer the whole table.
10

28/// Resolve each aggregate's RESULT CQL type (parallel to `agg_plan.aggregates`)
29/// from the schema, using the SAME `aggregate_result_cql_type` the result
30/// metadata is built from (issue #2202). This fixes each SUM/AVG accumulator's
31/// numeric domain (integral vs floating) so the emitted value variant can never
32/// disagree with the metadata type — and it is resolved ONCE per query, then
33/// shared across every group's accumulator init.
34fn resolve_aggregate_result_types(
35    agg_plan: &AggregationPlan,
36    schema: Option<&TableSchema>,
37) -> Vec<Option<CqlType>> {
38    agg_plan
39        .aggregates
40        .iter()
41        .map(|agg| {
42            // `*` (COUNT(*)) has no argument column; SUM/AVG never take `*`.
43            let arg_type = if agg.column == "*" {
44                None
45            } else {
46                schema.and_then(|s| {
47                    s.columns
48                        .iter()
49                        .find(|c| c.name == agg.column)
50                        .and_then(|c| parse_cql_type_str(&c.data_type))
51                })
52            };
53            aggregate_result_cql_type(&agg.function, arg_type)
54        })
55        .collect()
56}

58/// Read-ahead window for the D2 aggregate fold's scan stream. The fold retains
59/// only the running accumulator, so this bounds the in-flight scan rows (not the
60/// result) — the same role `StreamingConfig::buffer_size` plays for the streaming
61/// producer.
62const AGG_FOLD_SCAN_BUFFER: usize = 1024;
63
64/// The plan shape eligible for the O(1) global-aggregate fold: one full-scan step,
65/// one GROUP-BY-free aggregate, and zero or more post-scan `Filter`s.
66struct GlobalAggregatePlan<'a> {
67    table: &'a TableId,
68    predicates: &'a [SSTablePredicate],
69    projection: &'a [String],
70    filters: Vec<&'a WhereExpression>,
71    agg_plan: &'a AggregationPlan,
72}

74impl SelectExecutor {
75    /// Execute the aggregation step over already-materialized rows. Splits
76    /// naturally into three phases: build group key, accumulate per-aggregate
77    /// state, then finalize each group into a result row.
78    pub(super) fn execute_aggregation(
79        &self,
80        rows: Vec<QueryRow>,
81        agg_plan: &AggregationPlan,
82        schema_opt: Option<&TableSchema>,
83        _context: &mut ExecutionContext,
84    ) -> Result<Vec<QueryRow>> {
85        const PER_ROW_MEMORY_ESTIMATE_BYTES: usize = 100;
86        const DEFAULT_AGGREGATION_MEMORY_LIMIT: usize = 512 * 1024 * 1024;
87
88        // Issue #1578 (D2): record how many rows this buffered aggregate input
89        // carried, so the O(1) memory guard can prove a GROUP-BY-free aggregate
90        // served by the streaming fold buffers ZERO rows here (the fold never
91        // calls this method). Zero-overhead in release (probe body is cfg-gated).
92        crate::query::agg_stream_probe::record_buffered_rows(rows.len() as u64);
93
94        let mut agg_state = AggregationState {
95            groups: Vec::new(),
96            group_index: rustc_hash::FxHashMap::default(),
97            memory_usage_bytes: 0,
98            memory_limit_bytes: DEFAULT_AGGREGATION_MEMORY_LIMIT,
99        };
100
101        // Issue #2202: resolve the per-aggregate result types ONCE, then reuse for
102        // every group's accumulator init (drives integral vs floating SUM/AVG).
103        let result_types = resolve_aggregate_result_types(agg_plan, schema_opt);
104
105        for row in rows {
106            let group_key = build_group_key(&row, &agg_plan.group_by_columns);
107            let group_index = find_or_init_group(
108                &mut agg_state,
109                group_key,
110                &agg_plan.aggregates,
111                &result_types,
112            );
113            let group_aggregates = &mut agg_state.groups[group_index].1;
114
115            for (i, agg_comp) in agg_plan.aggregates.iter().enumerate() {
116                update_aggregate(&mut group_aggregates[i], agg_comp, &row);
117            }
118
119            agg_state.memory_usage_bytes += PER_ROW_MEMORY_ESTIMATE_BYTES;
120            if agg_state.memory_usage_bytes > agg_state.memory_limit_bytes {
121                return Err(Error::query_execution(
122                    "Aggregation memory limit exceeded".to_string(),
123                ));
124            }
125        }
126
127        // Issue #2069: a GLOBAL aggregate (no GROUP BY) always emits exactly one
128        // row, even when the input is empty. With rows present a single `[Null]`
129        // group is created above (see `build_group_key`); with zero rows no group
130        // exists yet, so synthesize the empty-input row (COUNT = 0, other
131        // aggregates NULL). A GROUP BY query is UNAFFECTED — it correctly returns
132        // zero rows when there are no groups.
133        if agg_state.groups.is_empty() && agg_plan.group_by_columns.is_empty() {
134            return Ok(vec![finalize_empty_global_aggregate(agg_plan)]);
135        }
136
137        let result_rows = agg_state
138            .groups
139            .into_iter()
140            .map(|(group_key, group_aggregates)| {
141                finalize_group(group_key, group_aggregates, agg_plan)
142            })
143            .collect();
144
145        Ok(result_rows)
146    }
147
148    /// Issue #1578 (D2): if `steps` is a GROUP-BY-free aggregate over a FULL table
149    /// scan, fold the scan stream into an O(1) accumulator and return the single
150    /// finalized result row — instead of buffering the whole table into a
151    /// `Vec<QueryRow>` and calling [`Self::execute_aggregation`].
152    ///
153    /// Returns `Ok(None)` when the plan is NOT eligible — a GROUP BY (E5 territory),
154    /// a partition-targeted / multi-partition lookup (single/few small partitions
155    /// the buffered path handles fine), a WRITETIME/TTL cell-metadata projection,
156    /// or any `Sort`/`Project`/`Limit`/`PerPartitionLimit` step — and the caller
157    /// must use the buffered path. On the eligible path the per-row update and
158    /// finalize reuse the exact buffered-path helpers, so the answer is identical.
159    pub(super) async fn try_execute_global_aggregate(
160        &self,
161        steps: &[ExecutionStep],
162        schema_opt: Option<&TableSchema>,
163        context: &mut ExecutionContext,
164    ) -> Result<Option<Vec<QueryRow>>> {
165        // A WRITETIME/TTL projection needs the per-cell metadata scan, not this fold.
166        if context.projection_flags.include_cell_metadata {
167            return Ok(None);
168        }
169
170        let Some(eligible) = classify_global_aggregate(steps) else {
171            return Ok(None);
172        };
173        let GlobalAggregatePlan {
174            table,
175            predicates,
176            projection,
177            filters,
178            agg_plan,
179        } = eligible;
180
181        // A `token(...)` predicate must cover the full partition key (same rule as
182        // every scan path) before we scan/evaluate it.
183        validate_token_predicates(predicates, schema_opt)?;
184
185        // Only a full scan is folded. A targeted / multi-targeted lookup reads one
186        // or a few small partitions, so the buffered path's memory is already
187        // O(partition) — fall back and let it run unchanged.
188        let reason = match classify_partition_lookup(predicates, schema_opt) {
189            PartitionLookupOutcome::Fallback(reason) => reason,
190            PartitionLookupOutcome::Targeted(_) | PartitionLookupOutcome::MultiTargeted(_) => {
191                return Ok(None);
192            }
193        };
194
195        // Record the honest full-scan access path, exactly as the buffered scan
196        // step does, so observability/plan-family assertions are unchanged.
197        context.access_path = Some(AccessPath::FallbackFullScan { reason });
198        crate::query::access_path::record(AccessPath::FallbackFullScan { reason });
199
200        // Issue #2202: fix each SUM/AVG accumulator's numeric domain from the
201        // resolved result types (same source as the result metadata).
202        let result_types = resolve_aggregate_result_types(agg_plan, schema_opt);
203        let mut accumulators = init_aggregate_accumulators(&agg_plan.aggregates, &result_types);
204        let mut folded_any = false;
205
206        // Issue #1592: fold over the BATCHED streaming surface — one async wake per
207        // batch, not per row. Flattening each batch yields the same rows in the
208        // same order as `scan_stream`, so the aggregate result is unchanged.
209        let mut scan_stream = self
210            .storage
211            .scan_stream_batched(table, None, None, schema_opt, AGG_FOLD_SCAN_BUFFER)
212            .await?;
213        // Issue #1817: hoist the partition-key decode across a partition's rows;
214        // the cache persists across batches (a partition may straddle a batch).
215        let mut pk_cache = PartitionKeyCache::default();
216        while let Some(batch) = scan_stream.recv().await {
217            for (key, value) in batch? {
218                context.rows_processed += 1;
219                context.scan_rows += 1;
220
221                let Some(row) =
222                    build_row_from_scan_cached(key, value, projection, schema_opt, &mut pk_cache)
223                else {
224                    continue;
225                };
226                if !evaluate_predicates(&row, predicates)? {
227                    continue;
228                }
229                // Apply any residual (non-pushed) WHERE as a per-row filter.
230                let mut keep = true;
231                for filter in &filters {
232                    if !self.evaluate_where_expression(filter, &row)? {
233                        keep = false;
234                        break;
235                    }
236                }
237                if !keep {
238                    continue;
239                }
240
241                folded_any = true;
242                for (i, agg_comp) in agg_plan.aggregates.iter().enumerate() {
243                    update_aggregate(&mut accumulators[i], agg_comp, &row);
244                }
245            }
246        }
247
248        // Issue #2069: a GLOBAL aggregate (no GROUP BY) ALWAYS produces exactly one
249        // result row, even over zero input rows. Cassandra returns COUNT = 0 and
250        // NULL for every other aggregate (SUM/AVG/MIN/MAX) of empty input, so when
251        // nothing folded we synthesize that single row rather than returning zero
252        // rows. This fold only handles the GROUP-BY-free case (see
253        // `classify_global_aggregate`), so it never affects a GROUP BY query, which
254        // correctly yields zero rows when there are no groups.
255        if !folded_any {
256            let row = finalize_empty_global_aggregate(agg_plan);
257            return Ok(Some(vec![row]));
258        }
259
260        let row = finalize_group(vec![Value::Null], accumulators, agg_plan);
261        Ok(Some(vec![row]))
262    }
263}

265/// Classify `steps` as an eligible GROUP-BY-free full-scan aggregate, returning
266/// the borrowed scan + aggregate + residual-filter references, or `None` when any
267/// step disqualifies it (see [`SelectExecutor::try_execute_global_aggregate`]).
268fn classify_global_aggregate(steps: &[ExecutionStep]) -> Option<GlobalAggregatePlan<'_>> {
269    let mut scan: Option<(&TableId, &[SSTablePredicate], &[String])> = None;
270    let mut agg_plan: Option<&AggregationPlan> = None;
271    let mut filters: Vec<&WhereExpression> = Vec::new();
272
273    for step in steps {
274        match step {
275            ExecutionStep::SSTableScan {
276                table,
277                predicates,
278                projection,
279                ..
280            } => {
281                if scan.is_some() {
282                    return None;
283                }
284                scan = Some((table, predicates.as_slice(), projection.as_slice()));
285            }
286            ExecutionStep::Aggregate { plan, .. } => {
287                // Only a GLOBAL aggregate (no GROUP BY), and only one.
288                if !plan.group_by_columns.is_empty() || agg_plan.is_some() {
289                    return None;
290                }
291                agg_plan = Some(plan);
292            }
293            ExecutionStep::Filter { expression, .. } => filters.push(expression),
294            // Sort / Project / Limit / PerPartitionLimit reshape or bound the result
295            // in ways the single-row fold does not model — use the buffered path.
296            _ => return None,
297        }
298    }
299
300    let (table, predicates, projection) = scan?;
301    let agg_plan = agg_plan?;
302    Some(GlobalAggregatePlan {
303        table,
304        predicates,
305        projection,
306        filters,
307        agg_plan,
308    })
