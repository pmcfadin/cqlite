1//! Value comparison and arithmetic helpers for the SELECT executor.
2//!
3//! Pure functions over [`Value`] shared by predicate evaluation, sorting,
4//! aggregation, and constant folding. They were previously inline in
5//! `select_executor.rs`; centralising them keeps one copy of the comparison and
6//! arithmetic semantics across every execution path.
7
8use super::super::select_ast::{ArithmeticOperator, ComparisonOperator};
9use crate::{types::Value, Error, Result};

11/// Compare two `Value`s for equality, including limited cross-type numeric
12/// coercion (int↔bigint, int↔float, bigint↔float).
13///
14/// `Value` implements `PartialEq` natively but only matches identical variants;
15/// we additionally treat the small set of cross-numeric cases that show up in
16/// CQL predicates.
17pub(super) fn values_equal(a: &Value, b: &Value) -> bool {
18    if a == b {
19        return true;
20    }
21    // Integer-vs-integer must compare as the widest native integer (`i128`),
22    // NEVER via `as_f64` (issue #2231): two distinct `i64` above 2^53 collapse
23    // to the same `f64` mantissa, so an f64 fallback would report them equal and
24    // — because a fully-translated Trino conjunct is removed from the plan — leak
25    // rows Trino would drop. Every CQL integral type fits losslessly in `i128`.
26    if let (Some(x), Some(y)) = (as_integral_i128(a), as_integral_i128(b)) {
27        return x == y;
28    }
29    // Otherwise coerce only when both operands are numeric — a genuine float on
30    // at least one side. Non-numeric pairs (e.g. Text vs Integer) must not
31    // spuriously compare equal via `as_f64`.
32    if same_numeric_family(a, b) {
33        if let (Some(x), Some(y)) = (a.as_f64(), b.as_f64()) {
34            return x == y;
35        }
36    }
37    false
38}

40/// The exact integer value of an integral CQL numeric `Value`, or `None` for
41/// float/non-numeric variants. Widened to `i128` so every integral type
42/// (`tinyint`..`bigint`, `counter`) is represented losslessly, enabling exact
43/// integer equality/comparison without an `f64` round-trip (issue #2231).
44pub(super) fn as_integral_i128(v: &Value) -> Option<i128> {
45    match v {
46        Value::Integer(i) => Some(*i as i128),
47        Value::BigInt(i) => Some(*i as i128),
48        Value::Counter(i) => Some(*i as i128),
49        Value::TinyInt(i) => Some(*i as i128),
50        Value::SmallInt(i) => Some(*i as i128),
51        _ => None,
52    }
53}

55/// True when `v` is an IEEE floating-point value that is NaN (CQL `float`/
56/// `double`). Predicate (WHERE) evaluation uses this to implement SQL
57/// three-valued logic: any relational comparison (`<`, `<=`, `>`, `>=`) with a
58/// NaN operand is UNKNOWN, so the row is dropped (issue #2231). Only the `float`
59/// variants can be NaN — integral types never are, so this is principled, not a
60/// bit-pattern heuristic.
61pub(in crate::query) fn is_nan_value(v: &Value) -> bool {
62    match v {
63        Value::Float(x) => x.is_nan(),
64        Value::Float32(x) => x.is_nan(),
65        _ => false,
66    }
67}

69/// True when both `Value`s are numeric variants eligible for cross-type coercion.
70pub(super) fn same_numeric_family(a: &Value, b: &Value) -> bool {
71    a.as_f64().is_some() && b.as_f64().is_some()
72}

74/// Compare two `Value`s for ordering, returning `Ordering::Equal` for
75/// incomparable variants. Used by sorting/aggregation paths that historically
76/// swallowed comparison errors via `unwrap_or(0)`.
77pub(super) fn compare_values_ordering(a: &Value, b: &Value) -> std::cmp::Ordering {
78    try_compare_values(a, b).unwrap_or(std::cmp::Ordering::Equal)
79}

81/// Compare two `Value`s for ordering, returning an error when the operand
82/// types are not comparable. Preferred in WHERE-clause evaluation so users see
83/// a real diagnostic rather than a silent equality.
84///
85/// Cross-type numerics are coerced via `f64` first; same-variant comparisons
86/// fall back to `Value::partial_cmp`. We deliberately avoid `partial_cmp` for
87/// non-matching variants because it stringifies and would produce surprising
88/// orderings (e.g. `Text("9")` < `Text("10")` lexicographically).
89pub(super) fn try_compare_values(a: &Value, b: &Value) -> Result<std::cmp::Ordering> {
90    if same_numeric_family(a, b) {
91        if let (Some(x), Some(y)) = (a.as_f64(), b.as_f64()) {
92            // Cassandra/Java `Double.compare` total order: NaN last, -0.0 < +0.0
93            // (issues #1870, #2010). Never `partial_cmp().unwrap_or(Equal)`,
94            // which collapses NaN and signed zeros to Equal.
95            return Ok(crate::float_cmp::cassandra_double_cmp(x, y));
96        }
97    }
98    if std::mem::discriminant(a) == std::mem::discriminant(b) {
99        return a.partial_cmp(b).ok_or_else(|| {
100            Error::query_execution("Cannot compare incompatible types".to_string())
101        });
102    }
103    // Data-safety (issue #1694): log the operand TYPES, never their values.
104    tracing::debug!(
105        "Cannot compare values of incompatible types: {:?} vs {:?}",
106        a.data_type(),
107        b.data_type()
108    );
109    Err(Error::query_execution(
110        "Cannot compare incompatible types".to_string(),
111    ))
112}

114/// Relational comparison for PREDICATE (WHERE) evaluation, with SQL
115/// three-valued logic for IEEE NaN AND exact-integer ordering above the f64
116/// precision boundary.
117///
118/// Returns `Ok(None)` — SQL UNKNOWN, so the caller DROPS the row — when either
119/// operand is a NaN float. Cassandra's total order (`cassandra_double_cmp`, used
120/// by `try_compare_values`/`compare_values_ordering`) sorts NaN as the GREATEST
121/// value, which would make `d > 1.5` TRUE for `d = NaN` and leak rows Trino
122/// would drop once the conjunct is pushed down and removed from the plan (issue
123/// #2231).
124///
125/// When both operands are integral (mirroring `values_equal`'s own structure),
126/// compares them as exact `i128` BEFORE any f64 fallback: two distinct `i64`
127/// above 2^53 collapse to the same f64 mantissa, so `bigcol > 9007199254740992`
128/// against a row where `bigcol = 9007199254740993` would otherwise compare
129/// `Equal` via `cassandra_double_cmp` and wrongly evaluate `is_gt()` to `false`
130/// — the same leak-once-pushed-down mechanism as the `=` divergence.
131///
132/// This function is for filter/`WHERE` evaluation ONLY — do NOT use it for
133/// ORDER BY / MIN / MAX / clustering-key ordering, which must keep the
134/// NaN-greatest total order and existing f64-based numeric ordering
135/// (`try_compare_values`/`compare_values_ordering`, unchanged).
136pub(super) fn try_compare_values_predicate(
137    a: &Value,
138    b: &Value,
139) -> Result<Option<std::cmp::Ordering>> {
140    if is_nan_value(a) || is_nan_value(b) {
141        return Ok(None);
142    }
143    if let (Some(x), Some(y)) = (as_integral_i128(a), as_integral_i128(b)) {
144        return Ok(Some(x.cmp(&y)));
145    }
146    try_compare_values(a, b).map(Some)
147}

149/// `compare_values_ordering` counterpart for predicate evaluation: routes
150/// through [`try_compare_values_predicate`] so it shares BOTH predicate-only
151/// fixes (issue #2231) — NaN → `None` (SQL UNKNOWN → drop the row) and exact
152/// `i128` integer ordering above the f64 precision boundary — then swallows any
153/// remaining comparison error to `Ordering::Equal`, matching
154/// `compare_values_ordering`'s error-tolerant behaviour for non-NaN,
155/// non-integral pairs. Used by the SSTable leaf-predicate evaluator's
156/// inequalities so neither a NaN nor a large `bigint` pair is mishandled.
157pub(super) fn compare_values_ordering_predicate(
158    a: &Value,
159    b: &Value,
160) -> Option<std::cmp::Ordering> {
161    match try_compare_values_predicate(a, b) {
162        Ok(ordering) => ordering,
163        Err(_) => Some(std::cmp::Ordering::Equal),
164    }
165}

167/// Evaluate a scalar `ComparisonOperator` (`=`, `!=`, `<`, `<=`, `>`, `>=`)
168/// over two already-evaluated operands with SQL PREDICATE semantics (issue
169/// #2231). Shared by the expression-pushdown WHERE evaluator so equality uses
170/// exact integer comparison (`values_equal`, no `f64` collapse above 2^53) and
171/// the four inequalities treat a NaN operand as UNKNOWN → `false` (row dropped),
172/// never NaN-greatest. Non-scalar operators (`IN`, `LIKE`, `IS [NOT] NULL`, …)
173/// have their own branches and are rejected here.
174pub(super) fn eval_scalar_comparison(
175    op: &ComparisonOperator,
176    left: &Value,
177    right: &Value,
178) -> Result<bool> {
179    use ComparisonOperator::*;
180    Ok(match op {
181        Equal => values_equal(left, right),
182        NotEqual => !values_equal(left, right),
183        LessThan => try_compare_values_predicate(left, right)?.is_some_and(|o| o.is_lt()),
184        LessThanOrEqual => try_compare_values_predicate(left, right)?.is_some_and(|o| o.is_le()),
185        GreaterThan => try_compare_values_predicate(left, right)?.is_some_and(|o| o.is_gt()),
186        GreaterThanOrEqual => try_compare_values_predicate(left, right)?.is_some_and(|o| o.is_ge()),
187        other => {
188            return Err(Error::query_execution(format!(
189                "operator {:?} is not a scalar comparison",
190                other
191            )))
192        }
193    })
194}

196/// Apply an `ArithmeticOperator` to two same-typed numeric `Value`s.
197///
198/// Behaviour matches the previous inline implementations: same-type only
199/// (no implicit coercion), and division/modulo by zero are reported as
200/// query-execution errors. Float division-by-zero (matching the original
201/// runtime path) yields IEEE inf/NaN rather than an error.
202pub(super) fn eval_arithmetic(op: &ArithmeticOperator, left: Value, right: Value) -> Result<Value> {
203    use ArithmeticOperator::*;
204    macro_rules! int_op {
205        ($a:expr, $b:expr, $ctor:expr) => {
206            match op {
207                Add => Ok($ctor($a + $b)),
208                Subtract => Ok($ctor($a - $b)),
209                Multiply => Ok($ctor($a * $b)),
210                Divide => {
211                    if $b == 0 {
212                        Err(Error::query_execution("Division by zero".to_string()))
213                    } else {
214                        Ok($ctor($a / $b))
215                    }
216                }
217                Modulo => {
218                    if $b == 0 {
219                        Err(Error::query_execution("Modulo by zero".to_string()))
220                    } else {
221                        Ok($ctor($a % $b))
222                    }
223                }
224            }
225        };
226    }
227    match (left, right) {
228        (Value::Integer(a), Value::Integer(b)) => int_op!(a, b, Value::Integer),
229        (Value::BigInt(a), Value::BigInt(b)) => int_op!(a, b, Value::BigInt),
230        (Value::Float(a), Value::Float(b)) => match op {
231            Add => Ok(Value::Float(a + b)),
232            Subtract => Ok(Value::Float(a - b)),
233            Multiply => Ok(Value::Float(a * b)),
234            Divide => Ok(Value::Float(a / b)),
235            Modulo => Ok(Value::Float(a % b)),
236        },
237        _ => Err(Error::query_execution(
238            "Incompatible types for arithmetic".to_string(),
239        )),
240    }
241}

243/// Constant-folding arithmetic. Same operand-type rules as `eval_arithmetic`,
244/// plus BigInt support and per-operator error wording matching the legacy
245/// implementation (e.g. `"Cannot add incompatible types"` and
246/// `"Modulo only supported for integers"`).
247pub(super) fn const_arithmetic(
248    op: &ArithmeticOperator,
249    left: Value,
250    right: Value,
251) -> Result<Value> {
252    use ArithmeticOperator::*;
253
254    // Modulo's error wording is special: any non-integer combination must
255    // report `"Modulo only supported for integers"` regardless of which side
256    // is offending.
257    if matches!(op, Modulo) {
258        return match (left, right) {
259            (Value::Integer(a), Value::Integer(b)) => {
260                eval_arithmetic(op, Value::Integer(a), Value::Integer(b))
261            }
262            (Value::BigInt(a), Value::BigInt(b)) => {
263                eval_arithmetic(op, Value::BigInt(a), Value::BigInt(b))
264            }
265            _ => Err(Error::query_execution(
266                "Modulo only supported for integers".to_string(),
267            )),
268        };
269    }
270
271    let verb = match op {
272        Add => "add",
273        Subtract => "subtract",
274        Multiply => "multiply",
275        Divide => "divide",
276        Modulo => unreachable!("handled above"),
277    };
278
279    match (left, right) {
280        (Value::Integer(a), Value::Integer(b)) => {
281            eval_arithmetic(op, Value::Integer(a), Value::Integer(b))
282        }
283        (Value::BigInt(a), Value::BigInt(b)) => {
284            eval_arithmetic(op, Value::BigInt(a), Value::BigInt(b))
285        }
286        (Value::Float(a), Value::Float(b)) => {
287            // Constant Float Divide rejects 0.0 (legacy behaviour); runtime
288            // Float divide does not. Modulo on Float is rejected above.
289            if matches!(op, Divide) && b == 0.0 {
290                return Err(Error::query_execution("Division by zero".to_string()));
291            }
292            eval_arithmetic(op, Value::Float(a), Value::Float(b))
293        }
294        _ => Err(Error::query_execution(format!(
295            "Cannot {} incompatible types",
296            verb
297        ))),
298    }
299}

322    /// Issue #2231, divergence 2: `values_equal` distinguishes two DISTINCT
323    /// `i64` values straddling the 2^53 f64-precision boundary. An `as_f64`
324    /// fallback would round both to the same mantissa and report them equal,
325    /// leaking rows once a `bigcol = ...` conjunct is pushed down and removed
326    /// from the Trino plan.
327    #[test]
328    fn values_equal_distinguishes_large_i64_across_f64_boundary() {
329        let two53 = 1_i64 << 53; // 9_007_199_254_740_992
330        let plus1 = two53 + 1; // 9_007_199_254_740_993 (not representable as f64)
331                               // Sanity: the two integers DO collapse to the same f64.
332        assert_eq!(
333            two53 as f64, plus1 as f64,
334            "precondition: f64 collapses them"
335        );
336
337        // The exact-integer comparison must keep them distinct.
338        assert!(
339            !values_equal(&Value::BigInt(two53), &Value::BigInt(plus1)),
340            "2^53 != 2^53 + 1 as bigint"
341        );
342        assert!(
343            values_equal(&Value::BigInt(plus1), &Value::BigInt(plus1)),
344            "identical large bigints are equal"
345        );
346        // Repro operand: bigcol = 9_007_199_254_740_993 must NOT match 992.
347        assert!(!values_equal(
348            &Value::BigInt(9_007_199_254_740_992),
349            &Value::BigInt(9_007_199_254_740_993),
350        ));
351
352        // Cross-integral-type equality still holds (int == bigint numerically).
353        assert!(values_equal(&Value::Integer(7), &Value::BigInt(7)));
354        assert!(values_equal(&Value::TinyInt(7), &Value::SmallInt(7)));
355        assert!(!values_equal(&Value::Integer(7), &Value::BigInt(8)));
356        // Integer-vs-float equality still coerces (genuine float on one side).
357        assert!(values_equal(&Value::BigInt(2), &Value::Float(2.0)));
358        assert!(!values_equal(&Value::BigInt(2), &Value::Float(2.5)));
359    }

395    /// Issue #2231, divergence 1: under PREDICATE (WHERE) semantics a NaN
396    /// operand makes every relational comparison UNKNOWN, so the row is dropped.
397    /// Historically only `Lt`/`Lte`/`Eq` dropped NaN while `Gt`/`Gte` leaked it
398    /// (NaN sorts greatest under `cassandra_double_cmp`); all four inequalities
399    /// must now drop it, and `Eq` continues to.
400    #[test]
401    fn nan_predicate_comparison_is_unknown_for_all_relations() {
402        let nan = Value::Float(f64::NAN);
403        let bound = Value::Float(1.5);
404
405        // `d > 1.5` / `d >= 1.5` with d = NaN: previously TRUE (leak), now dropped.
406        let cmp = try_compare_values_predicate(&nan, &bound).unwrap();
407        assert!(cmp.is_none(), "NaN vs 1.5 is UNKNOWN (Gt/Gte)");
408        assert!(
409            !cmp.is_some_and(|o| o.is_gt()),
410            "d > 1.5 with NaN is dropped"
411        );
412        assert!(
413            !cmp.is_some_and(|o| o.is_ge()),
414            "d >= 1.5 with NaN is dropped"
415        );
416        // Lt/Lte were already consistent (dropped) — confirm they stay dropped.
417        assert!(
418            !cmp.is_some_and(|o| o.is_lt()),
419            "d < 1.5 with NaN is dropped"
420        );
421        assert!(
422            !cmp.is_some_and(|o| o.is_le()),
423            "d <= 1.5 with NaN is dropped"
424        );
425        // NaN on the right-hand side is symmetric.
426        assert!(try_compare_values_predicate(&bound, &nan)
427            .unwrap()
428            .is_none());
429        // Two NaNs are also UNKNOWN under predicate comparison.
430        assert!(try_compare_values_predicate(&nan, &nan).unwrap().is_none());
431        // Float32 (CQL `float`) NaN behaves identically.
432        assert!(
433            try_compare_values_predicate(&Value::Float32(f32::NAN), &Value::Float32(1.5))
434                .unwrap()
435                .is_none()
436        );
437
438        // Eq already dropped NaN (no NaN-aware equality) — confirm unchanged.
439        assert!(!values_equal(&nan, &bound), "NaN = 1.5 is false");
440        assert!(!values_equal(&nan, &nan), "NaN = NaN is false (SQL)");
441
442        // Non-NaN floats keep normal ordering under predicate comparison.
443        let ok = try_compare_values_predicate(&Value::Float(2.0), &bound).unwrap();
444        assert!(ok.is_some_and(|o| o.is_gt()), "2.0 > 1.5 holds");
445    }

539        assert!(f(1) == 0.0 && f(1).is_sign_negative(), "index 1 = -0.0");
540        assert!(f(2) == 0.0 && f(2).is_sign_positive(), "index 2 = +0.0");
541        assert_eq!(f(3), 1.0);
542        assert_eq!(f(4), f64::INFINITY);
543        assert!(f(5).is_nan() && f(6).is_nan(), "NaNs sort last");
544    }

566    /// MIN/MAX (aggregation uses `compare_values_ordering`): MIN over signed
567    /// zeros selects -0.0; NaN never wins MIN but is the MAX.
568    #[test]
569    fn min_max_double_matches_cassandra() {
570        let data = [
571            Value::Float(f64::NAN),
572            Value::Float(3.0),
573            Value::Float(-0.0),
574            Value::Float(0.0),
575            Value::Float(-2.0),
576        ];
577        let min = data
578            .iter()
579            .min_by(|a, b| compare_values_ordering(a, b))
580            .unwrap();
581        assert!(matches!(min, Value::Float(x) if *x == -2.0), "MIN = -2.0");
582
583        let max = data
584            .iter()
585            .max_by(|a, b| compare_values_ordering(a, b))
586            .unwrap();
587        assert!(
588            matches!(max, Value::Float(x) if x.is_nan()),
589            "MAX = NaN (sorts last)"
590        );
591
592        // MIN over just the signed zeros picks -0.0.
593        let zeros = [Value::Float(0.0), Value::Float(-0.0)];
594        let zmin = zeros
595            .iter()
596            .min_by(|a, b| compare_values_ordering(a, b))
597            .unwrap();
598        assert!(
599            matches!(zmin, Value::Float(x) if *x == 0.0 && x.is_sign_negative()),
600            "MIN of {{-0.0, +0.0}} = -0.0"
601        );
602    }
