21// Test-only counter for GROUP-key exact (`==`) comparisons performed while
22// locating a row's group (issue #1587, E5). The hash index makes this
23// `O(rows)` (amortized) rather than the old linear scan's `O(rows × groups)`.
24// Same thread-local rationale as the executor's other work counters.
25#[cfg(test)]
26thread_local! {
27    pub(crate) static GROUP_KEY_COMPARISONS: std::cell::Cell<usize> =
28        const { std::cell::Cell::new(0) };
29}

31/// Aggregation state for GROUP BY operations
32#[derive(Debug)]
33pub(super) struct AggregationState {
34    /// Groups in first-seen (insertion) order. Result-row ordering is defined by
35    /// this `Vec`, so it is preserved exactly; the hash index below is only a
36    /// lookup accelerator.
37    pub(super) groups: Vec<(Vec<Value>, Vec<AggregateValue>)>,
38    /// Issue #1587 (E5): hash index over the group key → candidate `groups`
39    /// indices sharing that hash. `Value` does not implement `Hash`/`Eq`
40    /// (floats), so we hash a normalization that is *consistent* with `Vec<Value>`
41    /// equality (equal keys hash equal; `-0.0`/`+0.0` normalized) and confirm the
42    /// match with an exact `==` against each candidate. Distinct-but-colliding
43    /// keys (and every NaN key, which is never `==` itself) simply fall through
44    /// the exact check, so grouping semantics are byte-identical to the prior
45    /// linear scan.
46    ///
47    /// Issue #1590 (E8): `FxHashMap` rather than the default `HashMap`. The key
48    /// is already a well-mixed `u64` group-key hash, so SipHashing it again is
49    /// wasted work; Fx hashes the integer key directly. Collisions remain safe —
50    /// the `Vec<usize>` bucket + exact `==` confirmation above is unchanged.
51    pub(super) group_index: FxHashMap<u64, Vec<usize>>,
52    /// Memory usage tracking
53    pub(super) memory_usage_bytes: usize,
54    /// Maximum memory limit
55    pub(super) memory_limit_bytes: usize,
56}

58/// Hash a group key consistently with `Vec<Value>` equality: if two keys are
59/// `==`, they hash identically. Only floats need care (`-0.0 == +0.0`), which
60/// are normalized; NaN keys are never `==` themselves, so their hash is
61/// irrelevant to correctness (the exact `==` check separates them). Exotic /
62/// nested variants that are not cheap to hash structurally fall back to a
63/// discriminant-only contribution — still consistent (equal values agree on
64/// discriminant), merely less selective, and only for key types that never
65/// appear in practice.
66fn hash_group_key(key: &[Value]) -> u64 {
67    let mut hasher = std::collections::hash_map::DefaultHasher::new();
68    key.len().hash(&mut hasher);
69    for v in key {
70        hash_one_value(v, &mut hasher);
71    }
72    hasher.finish()
73}
74
75fn hash_one_value<H: Hasher>(v: &Value, h: &mut H) {
76    // Tag by discriminant so different variants never alias.
77    std::mem::discriminant(v).hash(h);
78    match v {
79        Value::Null => {}
80        Value::Boolean(b) => b.hash(h),
81        Value::Integer(x) => x.hash(h),
82        Value::BigInt(x) | Value::Counter(x) | Value::Timestamp(x) | Value::Time(x) => x.hash(h),
83        Value::Date(x) => x.hash(h),
84        Value::TinyInt(x) => x.hash(h),
85        Value::SmallInt(x) => x.hash(h),
86        // `-0.0 == +0.0` must hash equal; NaN's hash is irrelevant (never `==`).
87        Value::Float(f) => (if *f == 0.0 { 0.0f64 } else { *f }).to_bits().hash(h),
88        Value::Float32(f) => (if *f == 0.0 { 0.0f32 } else { *f }).to_bits().hash(h),
89        Value::Text(s) => s.hash(h),
90        Value::Blob(b) | Value::Varint(b) | Value::Inet(b) => b.hash(h),
91        Value::Uuid(u) => u.hash(h),
92        Value::Decimal { scale, unscaled } => {
93            scale.hash(h);
94            unscaled.hash(h);
95        }
96        Value::Duration {
97            months,
98            days,
99            nanos,
100        } => {
101            months.hash(h);
102            days.hash(h);
103            nanos.hash(h);
104        }
105        Value::List(items) | Value::Set(items) | Value::Tuple(items) => {
106            for item in items {
107                hash_one_value(item, h);
108            }
109        }
110        Value::Map(pairs) => {
111            for (k, val) in pairs {
112                hash_one_value(k, h);
113                hash_one_value(val, h);
114            }
115        }
116        Value::Frozen(inner) => hash_one_value(inner, h),
117        // Exotic / rarely-grouped variants: discriminant-only (still consistent).
118        Value::Json(_) | Value::Udt(_) | Value::Tombstone(_) => {}
119    }
120}

122/// Aggregate value accumulator
123#[derive(Debug, Clone)]
124pub(super) enum AggregateValue {
125    Count(u64),
126    Sum(NumericAcc),
127    Avg { acc: NumericAcc, count: u64 },
128    Min(Value),
129    Max(Value),
130}

132/// Build the initial per-aggregate accumulators for one group. Shared by
133/// [`find_or_init_group`] (the buffered GROUP BY path) and the streaming
134/// GROUP-BY-free fold (issue #1578, D2) so the initial state — and thus the
135/// per-row update + finalize semantics — are defined in exactly one place.
136///
137/// Issue #2202: `result_types[i]` is the aggregate's resolved RESULT CQL type
138/// (from `select_naming::aggregate_result_cql_type`, the SAME source the result
139/// metadata uses), parallel to `aggregates`. It fixes each SUM/AVG accumulator's
140/// numeric domain (integral vs floating) up front, so the emitted value variant
141/// matches the metadata type even for an all-null group that folds no input.
142pub(super) fn init_aggregate_accumulators(
143    aggregates: &[AggregateComputation],
144    result_types: &[Option<CqlType>],
145) -> Vec<AggregateValue> {
146    aggregates
147        .iter()
148        .enumerate()
149        .map(|(i, c)| {
150            let result_type = result_types.get(i).and_then(|t| t.as_ref());
151            match c.function {
152                AggregateType::Count => AggregateValue::Count(0),
153                AggregateType::Sum => AggregateValue::Sum(NumericAcc::init(result_type)),
154                AggregateType::Avg => AggregateValue::Avg {
155                    acc: NumericAcc::init(result_type),
156                    count: 0,
157                },
158                AggregateType::Min => AggregateValue::Min(Value::Null),
159                AggregateType::Max => AggregateValue::Max(Value::Null),
160            }
161        })
162        .collect()
163}

165/// Build the GROUP BY key for `row`. With no GROUP BY, all rows hash into a
166/// single `[Null]` bucket (global aggregation).
167pub(super) fn build_group_key(row: &QueryRow, group_by_columns: &[String]) -> Vec<Value> {
168    if group_by_columns.is_empty() {
169        return vec![Value::Null];
170    }
171    group_by_columns
172        .iter()
173        .map(|col| row.values.get(col.as_str()).cloned().unwrap_or(Value::Null))
174        .collect()
175}

177/// Locate the group matching `key`, or push a fresh entry with initial
178/// aggregator state. Returns the index into `state.groups`.
179///
180/// Issue #1587 (E5): a hash index (`state.group_index`) makes this `O(1)`
181/// amortized instead of the old `O(groups)` linear scan (which made a full
182/// aggregation `O(rows × groups)`). Result-row ordering is still defined solely
183/// by insertion order into `state.groups`; the index only accelerates lookup,
184/// and the exact `==` confirmation preserves grouping semantics byte-for-byte
185/// (colliding-but-unequal keys, and NaN keys, create distinct groups exactly as
186/// the linear scan did).
187pub(super) fn find_or_init_group(
188    state: &mut AggregationState,
189    key: Vec<Value>,
190    aggregates: &[AggregateComputation],
191    result_types: &[Option<CqlType>],
192) -> usize {
193    let hash = hash_group_key(&key);
194    if let Some(candidates) = state.group_index.get(&hash) {
195        for &idx in candidates {
196            #[cfg(test)]
197            GROUP_KEY_COMPARISONS.with(|c| c.set(c.get() + 1));
198            if state.groups[idx].0 == key {
199                return idx;
200            }
201        }
202    }
203    let initial = init_aggregate_accumulators(aggregates, result_types);
204    let new_index = state.groups.len();
205    state.groups.push((key, initial));
206    state.group_index.entry(hash).or_default().push(new_index);
207    new_index
208}

210/// Apply one row's contribution to a single aggregate accumulator.
211///
212/// COUNT(*) always increments; COUNT(col) only increments on non-null. SUM and
213/// AVG ignore non-numeric values. MIN/MAX clone the value only when it
214/// becomes the new extremum, sparing per-row clones in the common case.
215pub(super) fn update_aggregate(
216    state: &mut AggregateValue,
217    agg_comp: &AggregateComputation,
218    row: &QueryRow,
219) {
220    let is_star = agg_comp.column == "*";
221    // Look up the column once; for COUNT(*) we don't need it.
222    let value: Option<&Value> = if is_star {
223        None
224    } else {
225        row.values.get(agg_comp.column.as_str())
226    };
227    let is_null = !is_star && value.is_none_or(Value::is_null);
228
229    match state {
230        AggregateValue::Count(count) => {
231            if is_star || !is_null {
232                *count += 1;
233            }
234        }
235        AggregateValue::Sum(acc) => {
236            if let Some(v) = value {
237                if !v.is_null() {
238                    acc.add(v);
239                }
240            }
241        }
242        AggregateValue::Avg { acc, count } => {
243            if let Some(v) = value {
244                // Count only inputs the accumulator's numeric domain accepts, so
245                // AVG's divisor matches SUM's contributor set — a `0` still
246                // counts (issue #2202).
247                if !v.is_null() && acc.add(v) {
248                    *count += 1;
249                }
250            }
251        }
252        AggregateValue::Min(min_val) => {
253            if let Some(v) = value {
254                if !v.is_null()
255                    && (min_val.is_null() || compare_values_ordering(v, min_val).is_lt())
256                {
257                    *min_val = v.clone();
258                }
259            }
260        }
261        AggregateValue::Max(max_val) => {
262            if let Some(v) = value {
263                if !v.is_null()
264                    && (max_val.is_null() || compare_values_ordering(v, max_val).is_gt())
265                {
266                    *max_val = v.clone();
267                }
268            }
269        }
270    }
271}

273/// Materialize a single aggregation group into a `QueryRow`.
274pub(super) fn finalize_group(
275    group_key: Vec<Value>,
276    group_aggregates: Vec<AggregateValue>,
277    agg_plan: &AggregationPlan,
278) -> QueryRow {
279    // Issue #1584: one sized allocation per group — the row holds exactly the
280    // GROUP BY columns plus the aggregate results, so pre-size to that count and
281    // avoid rehash growth.
282    let mut row_values: HashMap<std::sync::Arc<str>, Value> =
283        HashMap::with_capacity(agg_plan.group_by_columns.len() + agg_plan.aggregates.len());
284
285    // Issue #1763: emit each grouped dimension under its SELECT OUTPUT name
286    // (alias when present, else the column name) — the SAME `select_naming` source
287    // that names the result metadata column — so the grouped dimension's row value
288    // key can never diverge from its metadata name. The group KEY was read by the
289    // raw stored column (`build_group_key`); only the emitted key differs.
290    for (i, out_name) in agg_plan.group_by_output_names.iter().enumerate() {
291        if let Some(v) = group_key.get(i) {
292            row_values.insert(out_name.as_str().into(), v.clone());
293        }
294    }
295
296    for (i, agg_comp) in agg_plan.aggregates.iter().enumerate() {
297        let result_value = match &group_aggregates[i] {
298            AggregateValue::Count(count) => Value::BigInt(*count as i64),
299            // Issue #2202: emit the value variant matching the accumulator's
300            // numeric domain (int/bigint/double), which was fixed from the
301            // resolved result type, so it agrees with the result metadata type.
302            AggregateValue::Sum(acc) => acc.finalize_sum(),
303            AggregateValue::Avg { acc, count } => {
304                if *count > 0 {
305                    acc.finalize_avg(*count)
306                } else {
307                    Value::Null
308                }
309            }
310            AggregateValue::Min(val) | AggregateValue::Max(val) => val.clone(),
311        };
312        row_values.insert(agg_comp.alias.as_str().into(), result_value);
313    }
314
315    QueryRow {
316        values: row_values,
317        key: RowKey::new(vec![]),
318        metadata: Default::default(),
319        cell_metadata: None,
320    }
321}

323/// Issue #2069: materialize the single result row of a GLOBAL aggregate (no
324/// GROUP BY) over ZERO input rows.
325///
326/// Cassandra's CQL semantics say a bare aggregate SELECT with no GROUP BY is a
327/// "global aggregate" and ALWAYS produces exactly one row, even over an empty
328/// table: `COUNT(*)`/`COUNT(col)` = `0`, and `MIN`/`MAX`/`SUM`/`AVG` = `NULL`
329/// (notably SUM of empty input is NULL, not `0`). This is DISTINCT from a
330/// GROUP BY query, which yields zero rows when there are no groups — the callers
331/// only reach this helper on the GROUP-BY-free path.
332///
333/// We cannot reuse [`init_aggregate_accumulators`] + [`finalize_group`] for this
334/// because a fresh SUM accumulator finalizes to a typed zero (`0`/`0.0`), whereas
335/// the empty-input answer must be `NULL`. This builds the row directly instead.
336pub(super) fn finalize_empty_global_aggregate(agg_plan: &AggregationPlan) -> QueryRow {
337    let mut row_values: HashMap<std::sync::Arc<str>, Value> =
338        HashMap::with_capacity(agg_plan.aggregates.len());
339
340    for agg_comp in &agg_plan.aggregates {
341        let result_value = match agg_comp.function {
342            // COUNT of empty input is 0 (BigInt, matching the non-empty finalize).
343            AggregateType::Count => Value::BigInt(0),
344            // SUM/AVG/MIN/MAX of empty input are NULL in Cassandra.
345            AggregateType::Sum | AggregateType::Avg | AggregateType::Min | AggregateType::Max => {
346                Value::Null
347            }
348        };
349        row_values.insert(agg_comp.alias.as_str().into(), result_value);
350    }
351
352    QueryRow {
353        values: row_values,
354        key: RowKey::new(vec![]),
355        metadata: Default::default(),
356        cell_metadata: None,
357    }
358}

387    /// Issue #1587 (E5): locating a row's group is `O(1)` amortized via the hash
388    /// index, so total group-key comparisons across a high-cardinality GROUP BY
389    /// stay `O(rows)` — NOT the old linear scan's `O(rows × groups)`. Assert a
390    /// tight linear bound the quadratic scan could never meet.
391    #[test]
392    fn find_or_init_group_is_subquadratic() {
393        let plan = count_star_plan();
394        let groups_count = 200usize;
395        let repeats = 3usize; // each group is hit multiple times (exercises "found")
396        let rows = groups_count * repeats;
397
398        let mut state = new_state();
399        GROUP_KEY_COMPARISONS.with(|c| c.set(0));
400
401        for r in 0..repeats {
402            for g in 0..groups_count {
403                let key = vec![Value::Integer(g as i32)];
404                let idx = find_or_init_group(&mut state, key, &plan.aggregates, &[]);
405                // Groups are created on first pass in ascending order.
406                assert_eq!(idx, g, "row (repeat {r}, group {g}) maps to a stable index");
407            }
408        }
409
410        let comparisons = GROUP_KEY_COMPARISONS.with(|c| c.get());
411        assert_eq!(
412            state.groups.len(),
413            groups_count,
414            "exactly one group per key"
415        );
416        // Hash index: ~1 comparison per row on the "found" path, 0 on "new".
417        // Linear scan would be ~rows × groups / 2 = {} here.
418        assert!(
419            comparisons <= rows * 2,
420            "issue #1587: group lookup must be sub-quadratic — {comparisons} comparisons \
421             for {rows} rows / {groups_count} groups (linear scan would be ~{})",
422            rows * groups_count / 2
423        );
424    }

515    /// Drive `init → update per value → finalize_group` for one scalar aggregate
516    /// with a resolved result type, returning the emitted result value.
517    fn run_scalar_agg(
518        function: AggregateType,
519        result_type: Option<CqlType>,
520        values: &[Value],
521    ) -> Value {
522        let plan = scalar_agg_plan(function, "v");
523        let result_types = vec![result_type];
524        let mut accs = init_aggregate_accumulators(&plan.aggregates, &result_types);
525        for v in values {
526            let row = QueryRow {
527                values: [(std::sync::Arc::<str>::from("v"), v.clone())]
528                    .into_iter()
529                    .collect(),
530                key: RowKey::new(vec![]),
531                metadata: Default::default(),
532                cell_metadata: None,
533            };
534            update_aggregate(&mut accs[0], &plan.aggregates[0], &row);
535        }
536        let row = finalize_group(vec![Value::Null], accs, &plan);
537        row.values.get("r").cloned().expect("result present")
538    }
