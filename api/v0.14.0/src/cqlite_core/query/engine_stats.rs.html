1//! Lock-free per-query statistics for the query engine (issue #1595, Epic F/F5).
2//!
3//! The engine previously held `QueryStats` behind an `Arc<RwLock<_>>` and took a
4//! **write lock 2–3× per query** just to bump counters — a process-wide
5//! serialization point on the hot path with no functional need. This module
6//! replaces that with a struct of `AtomicU64` counters bumped with
7//! [`Ordering::Relaxed`]: the counters are independent totals, no reader depends
8//! on a happens-before relationship between two different counters, and the
9//! public snapshot is inherently a racy point-in-time read (as it was under the
10//! old lock). Relaxed `fetch_add` is exact under concurrency (no lost updates),
11//! which is the only invariant the audit requires ("counts must remain correct").
12//!
13//! The public [`QueryStats`] snapshot shape is unchanged; derived values
14//! (`avg_execution_time_us`, `cache_hit_ratio`) are computed at read time in
15//! [`AtomicQueryStats::snapshot`].
16
17use std::sync::atomic::{AtomicU64, Ordering};
18
19use super::QueryStats;
20
21/// Lock-free query statistics accumulator held by the query engine.
22///
23/// Every field is an independent monotonic counter incremented with relaxed
24/// ordering. [`snapshot`](Self::snapshot) reads them into the public
25/// [`QueryStats`] view, computing the derived averages/ratios at read time.
26#[derive(Debug, Default)]
27pub(crate) struct AtomicQueryStats {
28    /// Total queries executed.
29    total_queries: AtomicU64,
30    /// Queries that resulted in errors.
31    error_queries: AtomicU64,
32    /// Number of plan-cache hits (numerator of `cache_hit_ratio`).
33    cache_hits: AtomicU64,
34    /// Total rows affected by write operations.
35    rows_affected: AtomicU64,
36    /// Sum of recorded execution times, in microseconds. Divided by
37    /// `total_queries` at read time to derive `avg_execution_time_us`.
38    exec_time_us_sum: AtomicU64,
39}
40
41impl AtomicQueryStats {
42    /// Count one executed query.
43    pub(crate) fn record_query(&self) {
44        self.total_queries.fetch_add(1, Ordering::Relaxed);
45    }
46
47    /// Count one query that resulted in an error.
48    pub(crate) fn record_error(&self) {
49        self.error_queries.fetch_add(1, Ordering::Relaxed);
50    }
51
52    /// Count one plan-cache hit.
53    pub(crate) fn record_cache_hit(&self) {
54        self.cache_hits.fetch_add(1, Ordering::Relaxed);
55    }
56
57    /// Record the outcome of a materialized query: add its execution time (µs)
58    /// and the rows it affected.
59    pub(crate) fn record_execution(&self, exec_time_us: u64, rows_affected: u64) {
60        self.exec_time_us_sum
61            .fetch_add(exec_time_us, Ordering::Relaxed);
62        self.rows_affected
63            .fetch_add(rows_affected, Ordering::Relaxed);
64    }
65
66    /// Read a consistent-shaped point-in-time [`QueryStats`] snapshot.
67    ///
68    /// Derived values are computed here: `avg_execution_time_us` is the mean
69    /// recorded execution time over all queries, and `cache_hit_ratio` is the
70    /// fraction of queries served from the plan cache. Both are `0` when no
71    /// queries have been recorded.
72    pub(crate) fn snapshot(&self) -> QueryStats {
73        let total_queries = self.total_queries.load(Ordering::Relaxed);
74        let cache_hits = self.cache_hits.load(Ordering::Relaxed);
75        let exec_time_us_sum = self.exec_time_us_sum.load(Ordering::Relaxed);
76
77        let (avg_execution_time_us, cache_hit_ratio) = if total_queries == 0 {
78            (0, 0.0)
79        } else {
80            (
81                exec_time_us_sum / total_queries,
82                // Clamp the numerator: `total_queries` and `cache_hits` are
83                // read as independent `Relaxed` atomics, so a concurrent
84                // snapshot could observe `cache_hits > total_queries` and yield
85                // a ratio > 1.0, violating the documented 0.0..=1.0 contract.
86                cache_hits.min(total_queries) as f64 / total_queries as f64,
87            )
88        };
89
90        QueryStats {
91            total_queries,
92            error_queries: self.error_queries.load(Ordering::Relaxed),
93            avg_execution_time_us,
94            cache_hit_ratio,
95            rows_affected: self.rows_affected.load(Ordering::Relaxed),
96        }
97    }
98}

100#[cfg(test)]
101mod tests {
102    use super::*;
103    use std::sync::Arc;
104    use std::thread;
105
106    #[test]
107    fn empty_snapshot_is_all_zero() {
108        let snap = AtomicQueryStats::default().snapshot();
109        assert_eq!(snap.total_queries, 0);
110        assert_eq!(snap.error_queries, 0);
111        assert_eq!(snap.rows_affected, 0);
112        assert_eq!(snap.avg_execution_time_us, 0);
113        assert_eq!(snap.cache_hit_ratio, 0.0);
114    }
115
116    #[test]
117    fn derived_values_have_expected_shape() {
118        let stats = AtomicQueryStats::default();
119        // 4 queries, 1 cache hit, executions summing to 400us over 4 queries.
120        for _ in 0..4 {
121            stats.record_query();
122        }
123        stats.record_cache_hit();
124        stats.record_execution(100, 2);
125        stats.record_execution(300, 3);
126
127        let snap = stats.snapshot();
128        assert_eq!(snap.total_queries, 4);
129        assert_eq!(snap.rows_affected, 5);
130        // avg = exec_time_us_sum / total_queries = 400 / 4
131        assert_eq!(snap.avg_execution_time_us, 100);
132        // ratio = cache_hits / total_queries = 1 / 4, and is within [0, 1] and > 0.
133        assert!((0.0..=1.0).contains(&snap.cache_hit_ratio));
134        assert!(snap.cache_hit_ratio > 0.0);
135        assert!((snap.cache_hit_ratio - 0.25).abs() < f64::EPSILON);
136    }
137
138    /// Concurrent increments must not lose updates: the snapshot totals must
139    /// equal the exact number of issued events. This asserts only counter *sums*
140    /// against known issued counts — no wall-clock window — so it cannot flake on
141    /// a timing boundary.
142    #[test]
143    fn concurrent_increments_are_exact() {
144        const THREADS: u64 = 16;
145        const PER_THREAD: u64 = 5_000;
146
147        let stats = Arc::new(AtomicQueryStats::default());
148        let handles: Vec<_> = (0..THREADS)
149            .map(|_| {
150                let stats = Arc::clone(&stats);
151                thread::spawn(move || {
152                    for _ in 0..PER_THREAD {
153                        stats.record_query();
154                        stats.record_error();
155                        stats.record_cache_hit();
156                        stats.record_execution(2, 3);
157                    }
158                })
159            })
160            .collect();
161        for h in handles {
162            h.join().expect("worker thread panicked");
163        }
164
165        let snap = stats.snapshot();
166        let issued = THREADS * PER_THREAD;
167        assert_eq!(snap.total_queries, issued);
168        assert_eq!(snap.error_queries, issued);
169        assert_eq!(snap.rows_affected, issued * 3);
170        // avg = (issued * 2) / issued == 2; ratio = issued hits / issued == 1.0
171        assert_eq!(snap.avg_execution_time_us, 2);
172        assert!((snap.cache_hit_ratio - 1.0).abs() < f64::EPSILON);
173    }
174}
