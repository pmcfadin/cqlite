1//! Minimal dataset helpers for Issue #78
2//!
3//! Provides simple functions to resolve canonical dataset paths from metadata.yml
4//!
5//! ## AppleDouble File Handling
6//! This module includes robust filtering for AppleDouble files (macOS metadata files
7//! with ._ prefix) to prevent CI test failures. All file iteration functions use
8//! should_ignore_file() to consistently filter these metadata files.

14/// Error type for dataset operations
15#[derive(Debug, thiserror::Error)]
16pub enum DatasetError {
17    #[error("Dataset not found: {keyspace}.{table}. Available datasets: {available}")]
18    DatasetNotFound {
19        keyspace: String,
20        table: String,
21        available: String,
22    },
23    #[error("Metadata file not found at {path}")]
24    MetadataNotFound { path: String },
25    #[error("IO error: {0}")]
26    Io(#[from] std::io::Error),
27    #[error("YAML parsing error: {0}")]
28    Yaml(#[from] serde_yaml::Error),
29    #[error("Directory not found: {path}")]
30    DirectoryNotFound { path: String },
31}

61/// Get the datasets root directory, checking CQLITE_DATASETS_ROOT env var first
62fn get_datasets_root() -> PathBuf {
63    if let Ok(root) = std::env::var("CQLITE_DATASETS_ROOT") {
64        PathBuf::from(root)
65    } else {
66        // Use compile-time manifest dir to calculate workspace root
67        // CARGO_MANIFEST_DIR points to cqlite-core/, parent is workspace root
68        let manifest_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
69        manifest_dir
70            .parent()
71            .map(|workspace| workspace.join("test-data/datasets"))
72            .unwrap_or_else(|| PathBuf::from("test-data/datasets"))
73    }
74}

76/// `true` when strict fixture mode is requested (issue #1230). Either
77/// `CQLITE_REQUIRE_FIXTURES` (the repo-wide convention used by the newer
78/// `issue_10xx`/`issue_99x` parity tests) or `CQLITE_PARITY_REQUIRE_DATASETS`
79/// (issue #1205) set to a truthy value ("1"/"true") flips dataset-dependent
80/// tests FAIL-CLOSED: an absent/empty fixture PANICS (test failure) instead of
81/// skipping, so a required CI lane cannot false-green when a table is dropped or
82/// a #773-class path regression hides the data. When neither is set, the default
83/// skip-on-absence behavior is preserved so local dev without the binaries still
84/// works. This is the single shared definition — do NOT fork a parallel copy.
85pub fn require_fixtures_strict() -> bool {
86    matches!(
87        std::env::var("CQLITE_REQUIRE_FIXTURES").as_deref(),
88        Ok("1") | Ok("true")
89    ) || matches!(
90        std::env::var("CQLITE_PARITY_REQUIRE_DATASETS").as_deref(),
91        Ok("1") | Ok("true")
92    )
93}

100/// Load metadata.yml from specified root path
101pub fn load_metadata_at(root: &Path) -> Result<Metadata, DatasetError> {
102    let metadata_path = root.join("metadata.yml");
103
104    if !metadata_path.exists() {
105        return Err(DatasetError::MetadataNotFound {
106            path: metadata_path.to_string_lossy().to_string(),
107        });
108    }
109
110    let content = fs::read_to_string(metadata_path)?;
111    let metadata: Metadata = serde_yaml::from_str(&content)?;
112
113    Ok(metadata)
114}

122/// Resolve table to SSTable path under specified root/sstables/
123pub fn resolve_table_to_sstable_path_at(
124    root: &Path,
125    keyspace: &str,
126    table: &str,
127) -> Result<PathBuf, DatasetError> {
128    let metadata = load_metadata_at(root)?;
129
130    // Verify table exists in metadata
131    let mut found = false;
132    for ks in &metadata.keyspaces {
133        if ks.name == keyspace {
134            for tbl in &ks.tables {
135                if tbl.name == table {
136                    found = true;
137                    break;
138                }
139            }
140        }
141    }
142
143    if !found {
144        let available = list_tables_at(root, None)?
145            .into_iter()
146            .map(|t| format!("{}.{}", t.keyspace, t.table))
147            .collect::<Vec<_>>()
148            .join(", ");
149
150        return Err(DatasetError::DatasetNotFound {
151            keyspace: keyspace.to_string(),
152            table: table.to_string(),
153            available,
154        });
155    }
156
157    // Look for directory under root/sstables/keyspace/
158    let sstables_dir = root.join("sstables").join(keyspace);
159
160    if !sstables_dir.exists() {
161        return Err(DatasetError::DirectoryNotFound {
162            path: sstables_dir.to_string_lossy().to_string(),
163        });
164    }
165
166    // Find table directories (format: table-{hash}) and prefer one with Data.db
167    let mut data_db_candidate: Option<PathBuf> = None;
168    let mut index_db_candidate: Option<PathBuf> = None;
169    let mut any_candidate: Option<PathBuf> = None;
170
171    for entry in fs::read_dir(&sstables_dir)? {
172        let entry = entry?;
173        let path = entry.path();
174        if !path.is_dir() {
175            continue;
176        }
177        if let Some(name) = path.file_name().and_then(|n| n.to_str()) {
178            if !name.starts_with(&format!("{}-", table)) {
179                continue;
180            }
181        }
182
183        // Scan the directory to classify candidates
184        let mut has_data = false;
185        let mut has_index = false;
186        let mut has_any = false;
187        if let Ok(files) = fs::read_dir(&path) {
188            for f in files.flatten() {
189                if let Some(fname) = f.file_name().to_str() {
190                    if should_ignore_file(fname) {
191                        continue;
192                    }
193                    if fname.ends_with("-Data.db") {
194                        has_data = true;
195                        has_any = true;
196                        break;
197                    } else if fname.ends_with("-Index.db") {
198                        has_index = true;
199                        has_any = true;
200                    } else if fname.ends_with("-Data.db.jsonl")
201                        || fname.ends_with("-Statistics.db.txt")
202                        || fname.ends_with("-Summary.db.txt")
203                    {
204                        has_any = true;
205                    }
206                }
207            }
208        }
209
210        if has_data {
211            data_db_candidate = Some(path);
212            // Highest priority met; we can stop searching further
213            break;
214        }
215        if has_index && index_db_candidate.is_none() {
216            index_db_candidate = Some(path.clone());
217        }
218        if has_any && any_candidate.is_none() {
219            any_candidate = Some(path.clone());
220        }
221    }
222
223    if let Some(p) = data_db_candidate.or(index_db_candidate).or(any_candidate) {
224        return Ok(p);
225    }
226
227    Err(DatasetError::DirectoryNotFound {
228        path: format!("{}/{}-*", sstables_dir.to_string_lossy(), table),
229    })
230}

237/// List tables from specified root, optionally filtered by keyspace  
238pub fn list_tables_at(
239    root: &Path,
240    keyspace_filter: Option<&str>,
241) -> Result<Vec<TableInfo>, DatasetError> {
242    let metadata = load_metadata_at(root)?;
243    let mut tables = Vec::new();
244
245    for keyspace in &metadata.keyspaces {
246        if let Some(filter) = keyspace_filter {
247            if keyspace.name != filter {
248                continue;
249            }
250        }
251
252        for table in &keyspace.tables {
253            tables.push(TableInfo {
254                keyspace: keyspace.name.clone(),
255                table: table.name.clone(),
256                row_count: table.row_count,
257            });
258        }
259    }
260
261    Ok(tables)
262}

264/// Check if a filename should be ignored (AppleDouble or other metadata files)
265///
266/// AppleDouble files are macOS metadata files with ._ prefix that should be
267/// ignored when scanning for SSTable files to prevent CI test failures.
268pub fn should_ignore_file(filename: &str) -> bool {
269    // Filter AppleDouble files (macOS metadata with ._ prefix)
270    filename.starts_with("._")
271}

277/// Given a path to a Data.db file, derive sibling reference paths produced by export.sh
278///
279/// This function first tries to find files with exact naming patterns, then falls back to searching
280/// for any valid reference files in the directory. This handles cases where CI datasets have
281/// different UUIDs or naming patterns than expected.
282pub fn derive_reference_paths_from_data_db(data_db: &Path) -> Option<(PathBuf, PathBuf, PathBuf)> {
283    let file_name = data_db.file_name()?.to_str()?;
284    if !file_name.ends_with("-Data.db") {
285        return None;
286    }
287    let prefix = &file_name[..file_name.len() - "-Data.db".len()];
288    let dir = data_db.parent()?;
289
290    // First, try exact naming patterns
291    let expected_jsonl = dir.join(format!("{}-Data.db.jsonl", prefix));
292    let expected_stats = dir.join(format!("{}-Statistics.db.txt", prefix));
293    let expected_summary = dir.join(format!("{}-Summary.db.txt", prefix));
294
295    // If all expected files exist, return them
296    if expected_jsonl.exists() && expected_stats.exists() && expected_summary.exists() {
297        return Some((expected_jsonl, expected_stats, expected_summary));
298    }
299
300    // Otherwise, search for any valid reference files in the directory
301    let mut found_jsonl = None;
302    let mut found_stats = None;
303    let mut found_summary = None;
304
305    if let Ok(entries) = std::fs::read_dir(dir) {
306        for entry in entries.flatten() {
307            if let Some(name) = entry.file_name().to_str() {
308                // Skip AppleDouble files
309                if should_ignore_file(name) {
310                    continue;
311                }
312
313                // Look for any JSONL reference file
314                if found_jsonl.is_none() && name.ends_with("-Data.db.jsonl") {
315                    found_jsonl = Some(entry.path());
316                }
317
318                // Look for any Statistics reference file
319                if found_stats.is_none() && name.ends_with("-Statistics.db.txt") {
320                    found_stats = Some(entry.path());
321                }
322
323                // Look for any Summary reference file
324                if found_summary.is_none() && name.ends_with("-Summary.db.txt") {
325                    found_summary = Some(entry.path());
326                }
327            }
328        }
329    }
330
331    // Return found files, or expected paths as fallback
332    let data_jsonl = found_jsonl.unwrap_or(expected_jsonl);
333    let stats_txt = found_stats.unwrap_or(expected_stats);
334    let summary_txt = found_summary.unwrap_or(expected_summary);
335
336    Some((data_jsonl, stats_txt, summary_txt))
337}

339/// Derive a companion SSTable file path (e.g., Index.db, Summary.db) from a Data.db path
340///
341/// This function handles cases where files might have different naming patterns or UUIDs
342/// by searching for any valid companion file if the expected one doesn't exist.
343pub fn derive_companion_file(data_file: &Path, companion_type: &str) -> Option<PathBuf> {
344    let data_name = data_file.file_name()?.to_str()?;
345    if !data_name.ends_with("-Data.db") {
346        return None;
347    }
348
349    let prefix = &data_name[..data_name.len() - "-Data.db".len()];
350    let dir = data_file.parent()?;
351
352    // First, try exact naming pattern
353    let expected_companion = dir.join(format!("{}-{}", prefix, companion_type));
354    if expected_companion.exists() {
355        return Some(expected_companion);
356    }
357
358    // Otherwise, search for any valid companion file in the directory
359    let companion_suffix = format!("-{}", companion_type);
360    if let Ok(entries) = std::fs::read_dir(dir) {
361        for entry in entries.flatten() {
362            if let Some(name) = entry.file_name().to_str() {
363                // Skip AppleDouble files
364                if should_ignore_file(name) {
365                    continue;
366                }
367
368                // Look for any file with the companion suffix
369                if name.ends_with(&companion_suffix) {
370                    return Some(entry.path());
371                }
372            }
373        }
374    }
375
376    // Return expected path as fallback (for error messages)
377    Some(expected_companion)
378}

380/// Stream JSONL rows from sstabledump output file, yielding serde_json::Value for each line
381pub fn read_jsonl_rows(
382    path: &Path,
383) -> Result<impl Iterator<Item = serde_json::Value>, DatasetError> {
384    use std::io::{BufRead, BufReader};
385    let file = std::fs::File::open(path)?;
386    let reader = BufReader::new(file);
387    let iter = reader
388        .lines()
389        .map_while(Result::ok)
390        .filter(|l| !l.trim().is_empty())
391        .filter_map(|l| serde_json::from_str::<serde_json::Value>(&l).ok());
392    Ok(iter)
393}

395/// Parse sstablemetadata text to a simple key->value map for assertions
396pub fn parse_sstablemetadata_text(
397    path: &Path,
398) -> Result<std::collections::HashMap<String, String>, DatasetError> {
399    let content = std::fs::read_to_string(path)?;
400    let mut map = std::collections::HashMap::new();
401    for line in content.lines() {
402        let line = line.trim();
403        if line.is_empty() {
404            continue;
405        }
406        if let Some((k, v)) = line.split_once(':') {
407            map.insert(k.trim().to_string(), v.trim().to_string());
408        }
409    }
410    Ok(map)
411}

416struct RefManifest {

427    table: String,
428    #[allow(dead_code)]

430    #[allow(dead_code)]

432}

434/// Load references.yml if present
435pub fn load_references_manifest_at(root: &Path) -> Option<RefManifest> {
436    let path = root.join("references.yml");
437    let content = std::fs::read_to_string(path).ok()?;
438    serde_yaml::from_str::<RefManifest>(&content).ok()
439}

441/// Resolve a stable table directory using references.yml when available.
442/// We normalize absolute paths inside the manifest by replacing their root
443/// with the provided datasets root.
444pub fn resolve_table_dir_via_manifest(root: &Path, keyspace: &str, table: &str) -> Option<PathBuf> {
445    let manifest = load_references_manifest_at(root)?;
446    let entry = manifest
447        .tables
448        .into_iter()
449        .find(|e| e.keyspace == keyspace && e.table == table)?;
450
451    // Extract the hashed directory basename from the recorded path
452    let hashed_dir_name = std::path::Path::new(&entry.sstable_dir)
453        .file_name()
454        .and_then(|n| n.to_str())?
455        .to_string();
456
457    let normalized = root.join("sstables").join(keyspace).join(hashed_dir_name);
458    Some(normalized)
459}
