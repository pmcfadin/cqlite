100/// Ordered interned cells of a single decoded row (issue #1334).
101///
102/// Each entry is `(column_name, value)` where the name is a shared `Arc<str>`
103/// handle interned once by the row decoder. Carrying a cell's name into
104/// `QueryRow.values` is therefore a reference-count bump — NOT a per-cell heap
105/// `String` allocation. Entries are ordered exactly as the producer emits them,
106/// which is deterministic per producer: the PRIMARY V5 decoder emits positionally
107/// in serialization-header (schema) column order by CONSTRUCTION, with no per-row
108/// sort (issue #1642 / K3); the cold schema-less fallback paths collect into an
109/// unordered map and emit SORTED order. This order is NOT user-visible: the public
110/// query result (`QueryRow.values`) is a name-keyed map, so all consumers key by
111/// name regardless of producer.
112pub type RowCells = Vec<(Arc<str>, Value)>;
113
114/// Payload of a single scanned row as it crosses the storage → query boundary
115/// (issue #1334).
116///
117/// This is a **dedicated internal carrier — deliberately NOT a variant of the
118/// public [`Value`] enum**. Keeping it off `Value` keeps the public value type
119/// closed (no breaking variant, no defensive match arms) and lets the interned
120/// [`RowCells`] names reach `QueryRow.values` without ever round-tripping
121/// through a `String`.
122///
123/// There is exactly ONE row-carrier path: every scan/compaction producer builds
124/// a `ScanRow` and every consumer disassembles a `ScanRow`, so a live row's
125/// column values can never silently fall through to a non-row fallback.
126#[derive(Debug, Clone, PartialEq)]
127pub enum ScanRow {
128    /// A live row: its interned cells, ordered exactly as emitted — the PRIMARY V5
129    /// decoder emits positionally in serialization-header column order (issue #1642
130    /// / K3), while cold schema-less fallback paths emit sorted order (both
131    /// deterministic). These cells are ALREADY DECODED — a consumer surfaces them
132    /// as-is (never re-decoding a cell value). The order is not user-visible (the
133    /// public result is name-keyed).
134    Row(RowCells),
135    /// A RAW, UNDECODED whole-row value carried with explicit provenance
136    /// (issue #1334). Emitted only by the fallback producers — the `data_access`
137    /// offset-read placeholder and the legacy `parse_block_entries*` fallback —
138    /// which pre-#1334 returned a bare `Value::Blob` of a row's raw value bytes.
139    ///
140    /// This is DELIBERATELY distinct from [`ScanRow::Row`]: it tells a consumer,
141    /// by provenance (not by inspecting cell name/shape), that the bytes still
142    /// need schema-decoding. A schema-aware consumer schema-decodes these bytes
143    /// into the table's columns; a no-schema consumer surfaces them as the exact
144    /// pre-#1334 shape (a single `"data"` [`Value::Blob`] column). An already
145    /// decoded row — even one whose only non-key column is a blob column named
146    /// `"data"` — is a [`ScanRow::Row`] and is NEVER treated as raw.
147    RawRow(Vec<u8>),
148    /// A non-row scan marker carried on the same channel — a row tombstone
149    /// (`Value::Tombstone`), an absent/null row (`Value::Null`), or a synthetic
150    /// compaction carrier value. Suppressed from user-visible output but
151    /// preserved so the compaction merge can reconcile deletions.
152    Marker(Value),
153}
154
155impl ScanRow {
156    /// Cells surfaced to a no-schema consumer: a live row's interned cells, the
157    /// single synthetic `"data"` blob for a raw fallback row, or `None` for a
158    /// marker (tombstone/null).
159    ///
160    /// A [`ScanRow::RawRow`] surfaces as one `("data", Value::Blob(bytes))` cell —
161    /// the exact pre-#1334 bare-`Value::Blob` shape a no-schema consumer produced.
162    pub fn into_cells(self) -> Option<RowCells> {
163        match self {
164            ScanRow::Row(cells) => Some(cells),
165            ScanRow::RawRow(bytes) => {
166                Some(vec![(std::sync::Arc::from("data"), Value::Blob(bytes))])
167            }
168            ScanRow::Marker(_) => None,
169        }
170    }
171
172    /// Cells surfaced to a **schema-discovery sampler** (issue #1334).
173    ///
174    /// Like [`into_cells`](Self::into_cells), but a [`ScanRow::RawRow`] — undecoded
175    /// whole-row bytes — is mapped onto `fallback_column` (the SSTable header's
176    /// first column name) rather than a synthetic `"data"` blob. This matches the
177    /// pre-#1334 sampler, which mapped a raw single `Value` onto the first header
178    /// column for type inference. When no `fallback_column` is available the raw
179    /// row is skipped (`None`), exactly as the pre-#1334 sampler skipped an entry
180    /// with an empty header column list. Markers are suppressed (`None`).
181    pub fn into_sample_cells(self, fallback_column: Option<&str>) -> Option<RowCells> {
182        match self {
183            ScanRow::Row(cells) => Some(cells),
184            ScanRow::RawRow(bytes) => {
185                let name = fallback_column?;
186                Some(vec![(std::sync::Arc::from(name), Value::Blob(bytes))])
187            }
188            ScanRow::Marker(_) => None,
189        }
190    }
191
192    /// True when this is a suppressed marker (row tombstone or absent/null row).
193    pub fn is_marker(&self) -> bool {
194        matches!(self, ScanRow::Marker(_))
195    }
196
197    /// Number of cells in a live row, or the byte length of a raw/marker value.
198    ///
199    /// Inspection convenience: a live [`ScanRow::Row`] reports its cell count; a
200    /// [`ScanRow::RawRow`] reports its undecoded byte length; a
201    /// [`ScanRow::Marker`] delegates to the wrapped value's [`Value::len`].
202    pub fn len(&self) -> usize {
203        match self {
204            ScanRow::Row(cells) => cells.len(),
205            ScanRow::RawRow(bytes) => bytes.len(),
206            ScanRow::Marker(v) => v.len(),
207        }
208    }
209
210    /// True when a live row has no cells, a raw row has no bytes, or the marker
211    /// value is empty/null.
212    pub fn is_empty(&self) -> bool {
213        match self {
214            ScanRow::Row(cells) => cells.is_empty(),
215            ScanRow::RawRow(bytes) => bytes.is_empty(),
216            ScanRow::Marker(v) => v.is_empty(),
217        }
218    }
219
220    /// Byte view of a raw fallback row's undecoded bytes, or a marker's wrapped
221    /// value bytes; `None` for a decoded live row.
222    pub fn as_bytes(&self) -> Option<&[u8]> {
223        match self {
224            ScanRow::RawRow(bytes) => Some(bytes),
225            ScanRow::Marker(v) => v.as_bytes(),
226            ScanRow::Row(_) => None,
227        }
228    }
229
230    /// Classifier that maps a single ALREADY-DECODED named cell onto the
231    /// scan → query row carrier (issue #1334). Used by the block-emit producers
232    /// for genuine static/clustering cells so a live value can never be
233    /// mis-wrapped as a suppressed marker.
234    ///
235    /// A genuinely absent cell ([`Value::Null`]) or a tombstone
236    /// ([`Value::Tombstone`]) stays a suppressible [`ScanRow::Marker`] —
237    /// `build_row_from_scan`/`into_cells()` drop those from user-visible output.
238    /// ANY other (decoded, live) value becomes a live single-cell
239    /// [`ScanRow::Row`] under `column_name` (interned `Arc<str>`) so it surfaces
240    /// in SELECT/export/schema-discovery. This is for DECODED cells only; a raw
241    /// undecoded whole-row value uses [`ScanRow::RawRow`] instead.
242    pub fn classify_cell(column_name: &str, value: Value) -> ScanRow {
243        match value {
244            Value::Null | Value::Tombstone(_) => ScanRow::Marker(value),
245            _ => ScanRow::Row(vec![(std::sync::Arc::from(column_name), value)]),
246        }
247    }
248}

363/// Types of tombstones in Cassandra
364#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
365pub enum TombstoneType {
366    /// Row-level deletion (entire row is deleted)
367    RowTombstone,
368    /// Cell-level deletion (specific column is deleted)
369    CellTombstone,
370    /// Range tombstone (a range of clustering rows is deleted)
371    RangeTombstone,
372    /// TTL expiration (data expired due to TTL)
373    TtlExpiration,
374    /// Partition-level deletion (`DELETE FROM t WHERE pk = ?`).
375    ///
376    /// Distinct from `RowTombstone` (which targets a single clustered row).
377    /// A partition tombstone is stored in the partition header's
378    /// `deletionInfo` field (`markedForDeleteAt` / `localDeletionTime`) and
379    /// supersedes every row and cell within the partition whose writetime is
380    /// older than the tombstone timestamp.
381    ///
382    /// In the delta-scan layer (`feature = "delta-scan"`) this maps directly
383    /// to `DeltaRecord::PartitionDelete`.  In the normal read path the reader
384    /// already parses the deletion-time bytes from the partition header (see
385    /// `version_gate::has_partition_level_deletion_presence_marker`); this
386    /// variant makes that information expressible in the shared type system.
387    PartitionTombstone,
388}

459    /// Validate UDT value against this type definition
460    pub fn validate_value(&self, value: &UdtValue) -> crate::Result<()> {
461        // Check that type names match
462        if value.type_name != self.name {
463            return Err(crate::Error::schema(format!(
464                "UDT type name mismatch: expected '{}', found '{}'",
465                self.name, value.type_name
466            )));
467        }
468
469        // Check that keyspace matches
470        if value.keyspace != self.keyspace {
471            return Err(crate::Error::schema(format!(
472                "UDT keyspace mismatch: expected '{}', found '{}'",
473                self.keyspace, value.keyspace
474            )));
475        }
476
477        // Validate each field
478        for field_def in &self.fields {
479            if let Some(field_value) = value.get_field(&field_def.name) {
480                // Field is present, check type compatibility
481                if !Self::is_compatible_type(&field_value.data_type(), &field_def.field_type) {
482                    return Err(crate::Error::schema(format!(
483                        "Field '{}' type mismatch: expected {:?}, found {:?}",
484                        field_def.name,
485                        field_def.field_type,
486                        field_value.data_type()
487                    )));
488                }
489            } else if !field_def.nullable {
490                // Field is missing but not nullable
491                return Err(crate::Error::schema(format!(
492                    "Non-nullable field '{}' is missing",
493                    field_def.name
494                )));
495            }
496        }
497
498        Ok(())
499    }
500
501    fn is_compatible_type(value_type: &CqlType, expected_type: &CqlType) -> bool {
502        // For now, require exact match - could be extended for type coercion
503        value_type == expected_type
504    }
505}

531impl Value {
532    /// Get the data type of this value
533    pub fn data_type(&self) -> CqlType {
534        match self {
535            Value::Null => CqlType::Text, // Default type for null
536            Value::Boolean(_) => CqlType::Boolean,
537            Value::Integer(_) => CqlType::Int,
538            Value::BigInt(_) => CqlType::BigInt,
539            Value::Counter(_) => CqlType::Counter, // Counter type (will be added to CqlType)
540            Value::Float(_) => CqlType::Double,
541            Value::Text(_) => CqlType::Text,
542            Value::Blob(_) => CqlType::Blob,
543            Value::Timestamp(_) => CqlType::Timestamp,
544            Value::Time(_) => CqlType::Time,
545            Value::Date(_) => CqlType::Date,
546            Value::Uuid(_) => CqlType::Uuid,
547            Value::Json(_) => CqlType::Text, // JSON stored as text
548            Value::TinyInt(_) => CqlType::TinyInt,
549            Value::SmallInt(_) => CqlType::SmallInt,
550            Value::Float32(_) => CqlType::Float,
551            Value::List(elements) => {
552                let element_type = if elements.is_empty() {
553                    CqlType::Text
554                } else {
555                    elements[0].data_type()
556                };
557                CqlType::List(Box::new(element_type))
558            }
559            Value::Set(elements) => {
560                let element_type = if elements.is_empty() {
561                    CqlType::Text
562                } else {
563                    elements[0].data_type()
564                };
565                CqlType::Set(Box::new(element_type))
566            }
567            Value::Map(pairs) => {
568                let (key_type, value_type) = if pairs.is_empty() {
569                    (CqlType::Text, CqlType::Text)
570                } else {
571                    (pairs[0].0.data_type(), pairs[0].1.data_type())
572                };
573                CqlType::Map(Box::new(key_type), Box::new(value_type))
574            }
575            Value::Tuple(fields) => {
576                let field_types = fields.iter().map(|f| f.data_type()).collect();
577                CqlType::Tuple(field_types)
578            }
579            Value::Udt(udt) => {
580                let fields = udt
581                    .fields
582                    .iter()
583                    .map(|f| {
584                        let field_type = if let Some(ref value) = f.value {
585                            value.data_type()
586                        } else {
587                            CqlType::Text // Default for null fields
588                        };
589                        (f.name.clone(), field_type)
590                    })
591                    .collect();
592                CqlType::Udt(udt.type_name.clone(), fields)
593            }
594            Value::Frozen(inner) => CqlType::Frozen(Box::new(inner.data_type())),
595            Value::Varint(_) => CqlType::Varint,
596            Value::Decimal { .. } => CqlType::Decimal,
597            Value::Duration { .. } => CqlType::Duration,
598            Value::Tombstone(_) => CqlType::Text, // Tombstones don't have a specific type
599            Value::Inet(_) => CqlType::Inet,
600        }
601    }
602
603    /// Check if this value is null
604    pub fn is_null(&self) -> bool {
605        matches!(self, Value::Null)
606    }
607
608    /// Check if this value is a tombstone (deleted)
609    pub fn is_tombstone(&self) -> bool {
610        matches!(self, Value::Tombstone(_))
611    }
612

869    /// Get the size in bytes for this value when serialized
870    pub fn size_estimate(&self) -> usize {
871        match self {
872            Value::Null => BOOL_SIZE,
873            Value::Boolean(_) => BOOL_SIZE,
874            Value::TinyInt(_) => TINYINT_SIZE,
875            Value::SmallInt(_) => SMALLINT_SIZE,
876            Value::Integer(_) => INT_SIZE,
877            Value::BigInt(_) | Value::Counter(_) => BIGINT_SIZE,
878            Value::Float32(_) => FLOAT32_SIZE,
879            Value::Float(_) => FLOAT64_SIZE,
880            Value::Timestamp(_) | Value::Time(_) => BIGINT_SIZE,
881            Value::Date(_) => INT_SIZE,
882            Value::Uuid(_) => UUID_SIZE,
883            Value::Duration { .. } => DURATION_SIZE,
884            Value::Tombstone(_) => TOMBSTONE_SIZE,
885
886            // Variable-length types with prefix
887            Value::Text(s) => VINT_LENGTH_PREFIX + s.len(),
888            Value::Blob(b) => VINT_LENGTH_PREFIX + b.len(),
889            Value::Inet(addr) => VINT_LENGTH_PREFIX + addr.len(),
890            Value::Varint(data) => VINT_LENGTH_PREFIX + data.len(),
891            Value::Decimal { unscaled, .. } => INT_SIZE + VINT_LENGTH_PREFIX + unscaled.len(),
892            Value::Json(j) => VINT_LENGTH_PREFIX + j.to_string().len(),
893
894            // Collections
895            Value::List(items) | Value::Set(items) => {
896                Self::collection_size(items.iter(), VINT_LENGTH_PREFIX + BOOL_SIZE)
897            }
898            Value::Map(pairs) => {
899                let overhead = VINT_LENGTH_PREFIX + 2 * BOOL_SIZE; // count + key_type + value_type
900                pairs.iter().fold(overhead, |acc, (k, v)| {
901                    acc + k.size_estimate() + v.size_estimate()
902                })
903            }
904            Value::Tuple(items) => Self::collection_size(items.iter(), VINT_LENGTH_PREFIX),
905            Value::Udt(udt_value) => {
906                let mut size = VINT_LENGTH_PREFIX
907                    + udt_value.type_name.len()
908                    + VINT_LENGTH_PREFIX
909                    + udt_value.keyspace.len()
910                    + VINT_LENGTH_PREFIX; // field count
911                for field in &udt_value.fields {
912                    size += VINT_LENGTH_PREFIX + field.name.len();
913                    size += match &field.value {
914                        Some(val) => BOOL_SIZE + val.size_estimate(),
915                        None => BOOL_SIZE,
916                    };
917                }
918                size
919            }
920            Value::Frozen(inner) => inner.size_estimate(),
921        }
922    }

962    /// Validate collection type consistency
963    pub fn validate_collection_types(&self) -> crate::Result<()> {
964        match self {
965            Value::List(list) => {
966                Self::validate_homogeneous_collection(list.iter(), "List")?;
967                Ok(())
968            }
969            Value::Set(set) => {
970                Self::validate_homogeneous_collection(set.iter(), "Set")?;
971                Self::check_unique_items(set.iter(), "Set")?;
972                Ok(())
973            }
974            Value::Map(map) => {
975                if !map.is_empty() {
976                    let (first_key, first_value) = &map[0];
977                    let key_type = first_key.data_type();
978                    let value_type = first_value.data_type();
979
980                    for (key, value) in map.iter().skip(1) {
981                        if key.data_type() != key_type {
982                            return Err(crate::Error::schema(format!(
983                                "Map contains mixed key types: {:?} and {:?}",
984                                key_type,
985                                key.data_type()
986                            )));
987                        }
988                        if value.data_type() != value_type {
989                            return Err(crate::Error::schema(format!(
990                                "Map contains mixed value types: {:?} and {:?}",
991                                value_type,
992                                value.data_type()
993                            )));
994                        }
995                    }
996
997                    Self::check_unique_items(map.iter().map(|(k, _)| k), "Map keys")?;
998                }
999                Ok(())
1000            }
1001            _ => Ok(()), // Non-collections are always valid
1002        }
1003    }

1048/// Ordering for `Value`.
1049///
1050/// NOTE (contract split, #1870/#2010): this `PartialOrd` INTENTIONALLY diverges
1051/// from the derived `PartialEq`. For `float`/`double` it uses the Cassandra/Java
1052/// total order (`-0.0 < +0.0`, and every `NaN` sorts last and compares Equal to
1053/// every other `NaN`), whereas `PartialEq` keeps IEEE semantics (`-0.0 == +0.0`,
1054/// `NaN != NaN`). This is deliberate: ordering (ORDER BY / MIN / MAX / clustering
1055/// order) must be a TOTAL order, while equality (GROUP BY) stays IEEE. As a
1056/// result `partial_cmp` may report `Equal` where `eq` reports `false` (two NaNs)
1057/// and vice-versa (`-0.0`/`+0.0`). If an `impl Ord for Value` is ever added it
1058/// MUST reuse this comparator (never derive from `PartialEq`) and callers must
1059/// not assume the `PartialOrd`/`PartialEq` consistency the std traits normally
1060/// imply.
1061impl PartialOrd for Value {
1062    fn partial_cmp(&self, other: &Self) -> Option<std::cmp::Ordering> {
1063        use crate::float_cmp::{cassandra_double_cmp as dcmp, cassandra_float_cmp as fcmp};
1064        use std::cmp::Ordering;
1065        match (self, other) {
1066            (Value::Null, Value::Null) => Some(Ordering::Equal),
1067            (Value::Null, _) => Some(Ordering::Less),
1068            (_, Value::Null) => Some(Ordering::Greater),
1069
1070            (Value::Boolean(a), Value::Boolean(b)) => a.partial_cmp(b),
1071            (Value::Integer(a), Value::Integer(b)) => a.partial_cmp(b),
1072            (Value::BigInt(a), Value::BigInt(b)) => a.partial_cmp(b),
1073            (Value::Counter(a), Value::Counter(b)) => a.partial_cmp(b),
1074            (Value::Float(a), Value::Float(b)) => Some(dcmp(*a, *b)),
1075            (Value::Text(a), Value::Text(b)) => a.partial_cmp(b),
1076            (Value::Blob(a), Value::Blob(b)) => a.partial_cmp(b),
1077            (Value::Timestamp(a), Value::Timestamp(b)) => a.partial_cmp(b),
1078            (Value::Time(a), Value::Time(b)) => a.partial_cmp(b),
1079            (Value::Date(a), Value::Date(b)) => a.partial_cmp(b),
1080            (Value::Uuid(a), Value::Uuid(b)) => a.partial_cmp(b),
1081            (Value::TinyInt(a), Value::TinyInt(b)) => a.partial_cmp(b),
1082            (Value::SmallInt(a), Value::SmallInt(b)) => a.partial_cmp(b),
1083            (Value::Float32(a), Value::Float32(b)) => Some(fcmp(*a, *b)),
1084            (Value::Inet(a), Value::Inet(b)) => a.partial_cmp(b),
1085
1086            // For complex types, compare by string representation
1087            (a, b) => a.to_string().partial_cmp(&b.to_string()),
1088        }
1089    }
1090}

1376/// Row key type - used for indexing and sorting.
1377///
1378/// Issue #1643 (K4): the raw key bytes are stored behind an `Arc<[u8]>` so that
1379/// the partition key, materialized ONCE when a partition header is parsed, can be
1380/// shared across every row of that partition by a pointer bump instead of a
1381/// per-row heap allocation. A 10k-row partition now allocates its key once, not
1382/// 10k times. This is a pure ownership/allocation change: the bytes and the
1383/// derived comparison order (`Ord`/`Eq`/`Hash` all delegate to the pointed-to
1384/// `[u8]`) are byte-identical to the former `Vec<u8>` representation, and serde's
1385/// `rc` feature keeps the wire format unchanged (a byte sequence).
1386#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
1387pub struct RowKey(pub std::sync::Arc<[u8]>);
1388
1389impl RowKey {
1390    /// Create a new row key from bytes
1391    pub fn new(bytes: Vec<u8>) -> Self {
1392        Self(std::sync::Arc::from(bytes))
1393    }
1394
1395    /// Create a row key from a value
1396    pub fn from_value(value: &Value) -> crate::Result<Self> {
1397        let bytes =
1398            bincode::serialize(value).map_err(|e| crate::Error::serialization(e.to_string()))?;
1399        Ok(Self(std::sync::Arc::from(bytes)))
1400    }
1401
1402    /// Get the byte representation
1403    pub fn as_bytes(&self) -> &[u8] {
1404        &self.0
1405    }
1406
1407    /// Get the length in bytes
1408    pub fn len(&self) -> usize {
1409        self.0.len()
1410    }

1413    pub fn is_empty(&self) -> bool {
1414        self.0.is_empty()
1415    }

1418    /// the same partition-key `Arc` return the SAME pointer, so a test can prove

1424    }

1431    }
