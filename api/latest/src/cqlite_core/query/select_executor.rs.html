132/// Split a `TableId` of the form `"keyspace.table"` into its parts.
133///
134/// If no dot is present, the whole name becomes the table component and the
135/// keyspace is `None`.
136fn parse_table_id(table_id: &TableId) -> (Option<String>, String) {
137    let table_str = table_id.name();
138    match table_str.rfind('.') {
139        Some(dot) => (
140            Some(table_str[..dot].to_string()),
141            table_str[dot + 1..].to_string(),
142        ),
143        None => (None, table_str.to_string()),
144    }
145}

147/// Compare two `Value`s for equality, including limited cross-type numeric
148/// coercion (int↔bigint, int↔float, bigint↔float).
149///
150/// `Value` implements `PartialEq` natively but only matches identical variants;
151/// we additionally treat the small set of cross-numeric cases that show up in
152/// CQL predicates.
153fn values_equal(a: &Value, b: &Value) -> bool {
154    if a == b {
155        return true;
156    }
157    // Only coerce when both operands are numeric — otherwise non-numeric
158    // pairs (e.g. Text vs Integer) would spuriously compare equal via `as_f64`.
159    if same_numeric_family(a, b) {
160        if let (Some(x), Some(y)) = (a.as_f64(), b.as_f64()) {
161            return x == y;
162        }
163    }
164    false
165}
166
167/// True when both `Value`s are numeric variants eligible for cross-type coercion.
168fn same_numeric_family(a: &Value, b: &Value) -> bool {
169    a.as_f64().is_some() && b.as_f64().is_some()
170}

172/// Compare two `Value`s for ordering, returning `Ordering::Equal` for
173/// incomparable variants. Used by sorting/aggregation paths that historically
174/// swallowed comparison errors via `unwrap_or(0)`.
175fn compare_values_ordering(a: &Value, b: &Value) -> std::cmp::Ordering {
176    try_compare_values(a, b).unwrap_or(std::cmp::Ordering::Equal)
177}
178
179/// Compare two `Value`s for ordering, returning an error when the operand
180/// types are not comparable. Preferred in WHERE-clause evaluation so users see
181/// a real diagnostic rather than a silent equality.
182///
183/// Cross-type numerics are coerced via `f64` first; same-variant comparisons
184/// fall back to `Value::partial_cmp`. We deliberately avoid `partial_cmp` for
185/// non-matching variants because it stringifies and would produce surprising
186/// orderings (e.g. `Text("9")` < `Text("10")` lexicographically).
187fn try_compare_values(a: &Value, b: &Value) -> Result<std::cmp::Ordering> {
188    use std::cmp::Ordering;
189    if same_numeric_family(a, b) {
190        if let (Some(x), Some(y)) = (a.as_f64(), b.as_f64()) {
191            return Ok(x.partial_cmp(&y).unwrap_or(Ordering::Equal));
192        }
193    }
194    if std::mem::discriminant(a) == std::mem::discriminant(b) {
195        return a.partial_cmp(b).ok_or_else(|| {
196            Error::query_execution("Cannot compare incompatible types".to_string())
197        });
198    }
199    log::debug!("Cannot compare {:?} with {:?}", a, b);
200    Err(Error::query_execution(
201        "Cannot compare incompatible types".to_string(),
202    ))
203}

205/// Three-valued (SQL Kleene) outcome of evaluating a single leaf predicate
206/// against one row.
207///
208/// `Unknown` is produced when the predicate references a column that is absent
209/// from the row or whose value is `Null` — i.e. a SQL `NULL` operand. Callers
210/// that only need pure-`AND` rejection (the historical [`evaluate_predicates`])
211/// treat `Unknown` and `False` identically; callers that evaluate `OR`/`NOT`
212/// (the Flight nested-predicate evaluator, issue #834) must distinguish them to
213/// match SQL `WHERE` semantics.
214#[derive(Debug, Clone, Copy, PartialEq, Eq)]
215pub enum LeafOutcome {
216    /// The predicate is definitely satisfied.
217    True,
218    /// The predicate is definitely not satisfied.
219    False,
220    /// The predicate's column is missing or `NULL` (SQL `UNKNOWN`).
221    Unknown,
222}
223
224/// Evaluate a single SSTable leaf predicate against one `QueryRow` with SQL
225/// three-valued (Kleene) semantics.
226///
227/// Returns [`LeafOutcome::Unknown`] when the predicate's column is absent from
228/// the row or its value is `Null`; otherwise a definite `True`/`False` from the
229/// typed comparison. This is the finer-grained primitive underlying both
230/// [`evaluate_predicates`] (pure AND, where `Unknown` rejects like `False`) and
231/// the Flight nested-predicate evaluator (issue #834), so all three paths share
232/// one copy of the comparison logic (`values_equal` / `compare_values_ordering`).
233///
234/// `IN` and `Prefix` follow `evaluate_predicates`: `IN` is membership over the
235/// value list, `Prefix` is a text `starts_with`. `Range` (two-bound) is included
236/// for completeness even though Flight lowers single bounds to `Gt`/`Lt`/etc.
237/// `BloomFilter` is always `True` (checked upstream).
238pub fn evaluate_leaf(row: &QueryRow, predicate: &SSTablePredicate) -> LeafOutcome {
239    use super::select_optimizer::SSTableFilterOp;
240    let column_value = match row.values.get(&predicate.column) {
241        // A SQL `NULL` operand (absent column or explicit `Null`) is `UNKNOWN`.
242        None | Some(Value::Null) => return LeafOutcome::Unknown,
243        Some(v) => v,
244    };
245    let matches = match &predicate.operation {
246        SSTableFilterOp::Equal => predicate
247            .values
248            .first()
249            .is_some_and(|v| values_equal(column_value, v)),
250        // Membership uses the same coercing equality as `Equal` so a pushed-down
251        // `IN` operand that lowers to a wider numeric type (e.g. `Integer`) still
252        // matches a narrow column value (`TinyInt`/`SmallInt`/`Float32`).
253        SSTableFilterOp::In => predicate
254            .values
255            .iter()
256            .any(|v| values_equal(column_value, v)),
257        SSTableFilterOp::Range => {
258            if predicate.values.len() < 2 {
259                false
260            } else {
261                let lo = &predicate.values[0];
262                let hi = &predicate.values[1];
263                compare_values_ordering(column_value, lo).is_ge()
264                    && compare_values_ordering(column_value, hi).is_le()
265            }
266        }
267        // Single-bound clustering inequalities (Issue #788). A missing bound
268        // rejects the row, mirroring the `Range` len-guard above.
269        SSTableFilterOp::Gt => predicate
270            .values
271            .first()
272            .is_some_and(|b| compare_values_ordering(column_value, b).is_gt()),
273        SSTableFilterOp::Gte => predicate
274            .values
275            .first()
276            .is_some_and(|b| compare_values_ordering(column_value, b).is_ge()),
277        SSTableFilterOp::Lt => predicate
278            .values
279            .first()
280            .is_some_and(|b| compare_values_ordering(column_value, b).is_lt()),
281        SSTableFilterOp::Lte => predicate
282            .values
283            .first()
284            .is_some_and(|b| compare_values_ordering(column_value, b).is_le()),
285        SSTableFilterOp::Prefix => matches!(
286            (column_value, predicate.values.first()),
287            (Value::Text(s), Some(Value::Text(p))) if s.starts_with(p)
288        ),
289        SSTableFilterOp::BloomFilter => true, // already checked upstream
290    };
291    if matches {
292        LeafOutcome::True
293    } else {
294        LeafOutcome::False
295    }
296}

298/// Evaluate the SSTable predicate set against a single `QueryRow`.
299///
300/// Returns `Ok(true)` only if every predicate is satisfied. A missing column
301/// causes the row to be rejected.
302///
303/// Exposed publicly so the Arrow Flight server can apply identical predicate
304/// pushdown semantics to its merged rows (output parity with SELECT).
305///
306/// Implemented in terms of [`evaluate_leaf`]: under pure `AND`, both
307/// [`LeafOutcome::False`] and [`LeafOutcome::Unknown`] reject the row, so this
308/// preserves the historical "missing column → false" behaviour exactly.
309pub fn evaluate_predicates(row: &QueryRow, predicates: &[SSTablePredicate]) -> Result<bool> {
310    for predicate in predicates {
311        if evaluate_leaf(row, predicate) != LeafOutcome::True {
312            return Ok(false);
313        }
314    }
315    Ok(true)
316}

318/// Build a `QueryRow` from a single `(RowKey, Value)` produced by storage scan,
319/// applying optional projection and synthesising partition-key columns from the
320/// raw key bytes when a schema is available.
321///
322/// Partition-key columns are never stored in the cell payload, so they are
323/// reconstructed from the raw row key via the canonical
324/// [`crate::storage::partition_key_codec::decode_partition_key_columns`] (the
325/// same codec the write engine uses). This is the fix for Issue #586: the
326/// previous decoder assumed a `u16` length prefix for every TEXT key, which is
327/// only correct for composite components — a single-component TEXT partition key
328/// is raw bytes, so its column was silently dropped from scan-built rows.
329///
330/// Returns `None` for tombstoned rows (so the caller can `continue`).
331///
332/// Exposed publicly so other readers (e.g. the Arrow Flight server's compaction
333/// merge producer) can assemble rows identically to the SELECT path, guaranteeing
334/// output parity. The `value` is expected to be a `Value::Map` of decoded
335/// non-partition-key cells; partition-key columns are reconstructed from `key`.
336pub fn build_row_from_scan(
337    key: RowKey,
338    value: Value,
339    projection: &[String],
340    schema: Option<&crate::schema::TableSchema>,
341) -> Option<QueryRow> {
342    // Suppress tombstoned rows from user-visible output. A row tombstone reaches
343    // here as `Value::Tombstone` (Issue #505); before that change it was `Value::Null`.
344    // Both must be suppressed identically so deleted rows never appear in query results.
345    if matches!(value, Value::Null | Value::Tombstone(_)) {
346        return None;
347    }
348
349    let mut row_values = HashMap::new();
350    let project = |name: &str| projection.is_empty() || projection.iter().any(|p| p == name);
351
352    if let Value::Map(map) = value {
353        for (col_name, col_value) in map {
354            if let Value::Text(name) = col_name {
355                if project(&name) {
356                    row_values.insert(name, col_value);
357                }
358            }
359        }
360        // Cassandra never serialises partition-key columns in the cell payload;
361        // reconstruct them from the raw row key when the schema is known. We
362        // decode through the canonical codec shared with the write engine so
363        // single-component (raw bytes) and composite (`[u16 len][bytes][0x00]`)
364        // keys are handled identically on both paths (Issue #586).
365        if let Some(schema) = schema {
366            match crate::storage::partition_key_codec::decode_partition_key_columns(&key.0, schema)
367            {
368                Ok(pk_columns) => {
369                    for (name, value) in pk_columns {
370                        if project(&name) {
371                            row_values.insert(name, value);
372                        }
373                    }
374                }
375                // Surface — never silently swallow — a decode failure, so a
376                // missing partition-key column can't ship invisibly (Issue #586).
377                Err(e) => {
378                    log::warn!(
379                        "Failed to reconstruct partition-key columns from row key \
380                         (len={} bytes) for {}.{}: {}",
381                        key.0.len(),
382                        schema.keyspace,
383                        schema.table,
384                        e
385                    );
386                }
387            }
388        }
389    } else {
390        // Non-map fallback: expose the raw value plus a debug-formatted id.
391        row_values.insert("data".to_string(), value);
392        if project("id") {
393            row_values.insert("id".to_string(), Value::Text(format!("{:?}", key)));
394        }
395    }
396
397    Some(QueryRow {
398        values: row_values,
399        key,
400        metadata: Default::default(),
401        cell_metadata: None,
402    })
403}

405/// If the pushed-down predicates fully constrain the partition key with
406/// equality, return the raw on-disk partition-key bytes so the scan can be
407/// turned into a partition-targeted lookup (Issue #949).
408///
409/// Returns `None` — and the caller falls back to a full table scan — when:
410/// - no schema is available (we cannot identify the partition-key columns),
411/// - any partition-key column is missing an `=` predicate (partial key, or an
412///   `IN`/range restriction — those still require the scan path today), or
413/// - the constrained values cannot be encoded to the on-disk key form (e.g. a
414///   type mismatch), in which case a full scan is the safe, correct fallback.
415///
416/// Only an all-equality restriction over the complete partition key qualifies,
417/// mirroring Cassandra's requirement for a single-partition read.
418fn full_partition_key_lookup(
419    predicates: &[SSTablePredicate],
420    schema: Option<&crate::schema::TableSchema>,
421) -> Option<Vec<u8>> {
422    use super::select_optimizer::SSTableFilterOp;
423
424    let schema = schema?;
425    if schema.partition_keys.is_empty() {
426        return None;
427    }
428
429    let mut values = Vec::with_capacity(schema.partition_keys.len());
430    for pk in &schema.partition_keys {
431        let predicate = predicates
432            .iter()
433            .find(|p| p.column == pk.name && matches!(p.operation, SSTableFilterOp::Equal))?;
434        // An `Equal` predicate always carries exactly one value.
435        values.push(predicate.values.first()?.clone());
436    }
437
438    crate::storage::partition_key_codec::encode_partition_key_columns(&values, schema).ok()
439}

441/// Apply an `ArithmeticOperator` to two same-typed numeric `Value`s.
442///
443/// Behaviour matches the previous inline implementations: same-type only
444/// (no implicit coercion), and division/modulo by zero are reported as
445/// query-execution errors. Float division-by-zero (matching the original
446/// runtime path) yields IEEE inf/NaN rather than an error.
447fn eval_arithmetic(op: &ArithmeticOperator, left: Value, right: Value) -> Result<Value> {
448    use ArithmeticOperator::*;
449    macro_rules! int_op {
450        ($a:expr, $b:expr, $ctor:expr) => {
451            match op {
452                Add => Ok($ctor($a + $b)),
453                Subtract => Ok($ctor($a - $b)),
454                Multiply => Ok($ctor($a * $b)),
455                Divide => {
456                    if $b == 0 {
457                        Err(Error::query_execution("Division by zero".to_string()))
458                    } else {
459                        Ok($ctor($a / $b))
460                    }
461                }
462                Modulo => {
463                    if $b == 0 {
464                        Err(Error::query_execution("Modulo by zero".to_string()))
465                    } else {
466                        Ok($ctor($a % $b))
467                    }
468                }
469            }
470        };
471    }
472    match (left, right) {
473        (Value::Integer(a), Value::Integer(b)) => int_op!(a, b, Value::Integer),
474        (Value::BigInt(a), Value::BigInt(b)) => int_op!(a, b, Value::BigInt),
475        (Value::Float(a), Value::Float(b)) => match op {
476            Add => Ok(Value::Float(a + b)),
477            Subtract => Ok(Value::Float(a - b)),
478            Multiply => Ok(Value::Float(a * b)),
479            Divide => Ok(Value::Float(a / b)),
480            Modulo => Ok(Value::Float(a % b)),
481        },
482        _ => Err(Error::query_execution(
483            "Incompatible types for arithmetic".to_string(),
484        )),
485    }
486}

488/// Build the GROUP BY key for `row`. With no GROUP BY, all rows hash into a
489/// single `[Null]` bucket (global aggregation).
490fn build_group_key(row: &QueryRow, group_by_columns: &[String]) -> Vec<Value> {
491    if group_by_columns.is_empty() {
492        return vec![Value::Null];
493    }
494    group_by_columns
495        .iter()
496        .map(|col| row.values.get(col).cloned().unwrap_or(Value::Null))
497        .collect()
498}

500/// Locate the group matching `key` in `groups`, or push a fresh entry with
501/// initial aggregator state. Returns the index into `groups`.
502///
503/// `Value` doesn't implement `Hash`, so groups live in a `Vec` and lookup is
504/// linear. This is unchanged from the legacy implementation; switching to a
505/// hash map would change result-row ordering for callers that rely on
506/// insertion order.
507fn find_or_init_group(
508    groups: &mut Vec<(Vec<Value>, Vec<AggregateValue>)>,
509    key: Vec<Value>,
510    aggregates: &[super::select_optimizer::AggregateComputation],
511) -> usize {
512    if let Some(idx) = groups.iter().position(|(k, _)| k == &key) {
513        return idx;
514    }
515    let initial: Vec<_> = aggregates
516        .iter()
517        .map(|c| match c.function {
518            AggregateType::Count => AggregateValue::Count(0),
519            AggregateType::Sum => AggregateValue::Sum(0.0),
520            AggregateType::Avg => AggregateValue::Avg { sum: 0.0, count: 0 },
521            AggregateType::Min => AggregateValue::Min(Value::Null),
522            AggregateType::Max => AggregateValue::Max(Value::Null),
523        })
524        .collect();
525    groups.push((key, initial));
526    groups.len() - 1
527}

529/// Apply one row's contribution to a single aggregate accumulator.
530///
531/// COUNT(*) always increments; COUNT(col) only increments on non-null. SUM and
532/// AVG ignore non-numeric values. MIN/MAX clone the value only when it
533/// becomes the new extremum, sparing per-row clones in the common case.
534fn update_aggregate(
535    state: &mut AggregateValue,
536    agg_comp: &super::select_optimizer::AggregateComputation,
537    row: &QueryRow,
538) {
539    let is_star = agg_comp.column == "*";
540    // Look up the column once; for COUNT(*) we don't need it.
541    let value: Option<&Value> = if is_star {
542        None
543    } else {
544        row.values.get(&agg_comp.column)
545    };
546    let is_null = !is_star && value.is_none_or(Value::is_null);
547
548    match state {
549        AggregateValue::Count(count) => {
550            if is_star || !is_null {
551                *count += 1;
552            }
553        }
554        AggregateValue::Sum(sum) => {
555            if let Some(v) = value.and_then(Value::as_f64) {
556                *sum += v;
557            }
558        }
559        AggregateValue::Avg { sum, count } => {
560            if let Some(v) = value.and_then(Value::as_f64) {
561                *sum += v;
562                *count += 1;
563            }
564        }
565        AggregateValue::Min(min_val) => {
566            if let Some(v) = value {
567                if !v.is_null()
568                    && (min_val.is_null() || compare_values_ordering(v, min_val).is_lt())
569                {
570                    *min_val = v.clone();
571                }
572            }
573        }
574        AggregateValue::Max(max_val) => {
575            if let Some(v) = value {
576                if !v.is_null()
577                    && (max_val.is_null() || compare_values_ordering(v, max_val).is_gt())
578                {
579                    *max_val = v.clone();
580                }
581            }
582        }
583    }
584}

586/// Materialize a single aggregation group into a `QueryRow`.
587fn finalize_group(
588    group_key: Vec<Value>,
589    group_aggregates: Vec<AggregateValue>,
590    agg_plan: &AggregationPlan,
591) -> QueryRow {
592    let mut row_values = HashMap::new();
593
594    for (i, col) in agg_plan.group_by_columns.iter().enumerate() {
595        if let Some(v) = group_key.get(i) {
596            row_values.insert(col.clone(), v.clone());
597        }
598    }
599
600    for (i, agg_comp) in agg_plan.aggregates.iter().enumerate() {
601        let result_value = match &group_aggregates[i] {
602            AggregateValue::Count(count) => Value::BigInt(*count as i64),
603            AggregateValue::Sum(sum) => Value::Float(*sum),
604            AggregateValue::Avg { sum, count } => {
605                if *count > 0 {
606                    Value::Float(sum / (*count as f64))
607                } else {
608                    Value::Null
609                }
610            }
611            AggregateValue::Min(val) | AggregateValue::Max(val) => val.clone(),
612        };
613        row_values.insert(agg_comp.alias.clone(), result_value);
614    }
615
616    QueryRow {
617        values: row_values,
618        key: RowKey::new(vec![]),
619        metadata: Default::default(),
620        cell_metadata: None,
621    }
622}

624/// Constant-folding arithmetic. Same operand-type rules as `eval_arithmetic`,
625/// plus BigInt support and per-operator error wording matching the legacy
626/// implementation (e.g. `"Cannot add incompatible types"` and
627/// `"Modulo only supported for integers"`).
628fn const_arithmetic(op: &ArithmeticOperator, left: Value, right: Value) -> Result<Value> {
629    use ArithmeticOperator::*;
630
631    // Modulo's error wording is special: any non-integer combination must
632    // report `"Modulo only supported for integers"` regardless of which side
633    // is offending.
634    if matches!(op, Modulo) {
635        return match (left, right) {
636            (Value::Integer(a), Value::Integer(b)) => {
637                eval_arithmetic(op, Value::Integer(a), Value::Integer(b))
638            }
639            (Value::BigInt(a), Value::BigInt(b)) => {
640                eval_arithmetic(op, Value::BigInt(a), Value::BigInt(b))
641            }
642            _ => Err(Error::query_execution(
643                "Modulo only supported for integers".to_string(),
644            )),
645        };
646    }
647
648    let verb = match op {
649        Add => "add",
650        Subtract => "subtract",
651        Multiply => "multiply",
652        Divide => "divide",
653        Modulo => unreachable!("handled above"),
654    };
655
656    match (left, right) {
657        (Value::Integer(a), Value::Integer(b)) => {
658            eval_arithmetic(op, Value::Integer(a), Value::Integer(b))
659        }
660        (Value::BigInt(a), Value::BigInt(b)) => {
661            eval_arithmetic(op, Value::BigInt(a), Value::BigInt(b))
662        }
663        (Value::Float(a), Value::Float(b)) => {
664            // Constant Float Divide rejects 0.0 (legacy behaviour); runtime
665            // Float divide does not. Modulo on Float is rejected above.
666            if matches!(op, Divide) && b == 0.0 {
667                return Err(Error::query_execution("Division by zero".to_string()));
668            }
669            eval_arithmetic(op, Value::Float(a), Value::Float(b))
670        }
671        _ => Err(Error::query_execution(format!(
672            "Cannot {} incompatible types",
673            verb
674        ))),
675    }
676}

678/// Return `true` when the select clause contains at least one `WRITETIME` or
679/// `TTL` call — used during planning to set `ProjectionFlags::include_cell_metadata`.
680fn select_has_writetime_ttl(statement: &SelectStatement) -> bool {
681    let exprs = match &statement.select_clause {
682        SelectClause::All => return false,
683        SelectClause::Columns(e) | SelectClause::Distinct(e) => e,
684    };
685    exprs
686        .iter()
687        .any(|e| matches!(e, SelectExpression::WriteTimeTtl(_)))
688}

690/// Compute the Cassandra-convention output column name for a `WriteTimeTtlCall`.
691///
692/// - No alias: `writetime(col)` or `ttl(col)` (lowercase, matching Cassandra).
693/// - Explicit alias: the alias string, exactly as parsed.
694fn writetime_ttl_column_name(call: &WriteTimeTtlCall) -> String {
695    if let Some(alias) = &call.alias {
696        return alias.clone();
697    }
698    match call.function {
699        WriteTimeTtlFunction::WriteTime => format!("writetime({})", call.column),
700        WriteTimeTtlFunction::Ttl => format!("ttl({})", call.column),
701    }
702}

704/// Evaluate a `WRITETIME(col)` or `TTL(col)` call against a single `QueryRow`.
705///
706/// `now_secs` is the current epoch-second used only for TTL subtraction. It
707/// **must** be injected by the caller rather than read here so that unit tests
708/// can produce deterministic results.
709///
710/// Return values:
711/// - `WRITETIME(col)` → `Value::BigInt(micros)` when metadata exists; `Value::Null` otherwise.
712/// - `TTL(col)` → `Value::Integer(remaining_secs)` when the cell has an unexpired TTL;
713///   `Value::Null` when no expiration exists **or** the cell has already expired.
714fn evaluate_writetime_ttl(call: &WriteTimeTtlCall, row: &QueryRow, now_secs: i64) -> Value {
715    let meta = match row.get_cell_metadata(&call.column) {
716        Some(m) => m,
717        None => return Value::Null,
718    };
719
720    match call.function {
721        WriteTimeTtlFunction::WriteTime => Value::BigInt(meta.write_timestamp_micros),
722        WriteTimeTtlFunction::Ttl => match &meta.expiration {
723            None => Value::Null,
724            Some(exp) => {
725                let remaining = exp.expires_at_seconds - now_secs;
726                if remaining <= 0 {
727                    // Cell has already expired — Cassandra returns NULL.
728                    Value::Null
729                } else {
730                    // Safe cast: remaining is in (0, i32::MAX] range for any
731                    // realistic TTL (Cassandra caps TTL at 630_720_000 seconds).
732                    Value::Integer(remaining.min(i32::MAX as i64) as i32)
733                }
734            }
735        },
736    }
737}

739/// Translate a CQL LIKE pattern (`%`, `_`) into an anchored regex.
740fn like_pattern_to_regex(pattern: &str) -> String {
741    let mut out = String::with_capacity(pattern.len() + 4);
742    out.push('^');
743    for ch in pattern.chars() {
744        match ch {
745            '%' => out.push_str(".*"),
746            '_' => out.push('.'),
747            _ => out.push(ch),
748        }
749    }
750    out.push('$');
751    out
752}

791    /// Execute an optimized query plan
792    pub async fn execute(&self, plan: OptimizedQueryPlan) -> Result<QueryResult> {
793        let table_id = if let Some(ref from_clause) = plan.statement.from_clause {
794            self.extract_table_id(from_clause)?
795        } else {
796            // For queries without FROM clause (like SELECT 1), use a dummy table ID
797            TableId::new("_dummy_")
798        };
799
800        // Issue #692: detect whether any WRITETIME/TTL select items are present
801        // during planning and set the opt-in flag so the reader threads per-cell
802        // metadata. This is the "planning" half of the executor wiring; the
803        // "evaluation" half lives in `evaluate_select_expression`.
804        let projection_flags = ProjectionFlags {
805            include_cell_metadata: select_has_writetime_ttl(&plan.statement),
806        };
807        log::debug!(
808            "Query plan: include_cell_metadata={}",
809            projection_flags.include_cell_metadata
810        );
811
812        let mut context = ExecutionContext {
813            table_id,
814            columns: self.get_result_columns(&plan.statement).await?,
815            rows_processed: 0,
816            projection_flags,
817        };
818
819        // Handle queries without FROM clause (like SELECT 1)
820        if plan.statement.from_clause.is_none() {
821            return self.execute_constant_query(&plan.statement, &context).await;
822        }
823
824        // Execute the plan step by step
825        let mut intermediate_results = Vec::new();
826
827        // If no execution steps are provided, add a default table scan
828        let execution_steps = if plan.execution_steps.is_empty() {
829            vec![ExecutionStep::SSTableScan {
830                table: context.table_id.clone(),
831                predicates: vec![],
832                projection: context.columns.iter().map(|c| c.name.clone()).collect(),
833            }]
834        } else {
835            plan.execution_steps.clone()
836        };
837
838        for step in &execution_steps {
839            match step {
840                ExecutionStep::SSTableScan {
841                    table,
842                    predicates,
843                    projection,
844                    ..
845                } => {
846                    let rows = self
847                        .execute_sstable_scan(table, predicates, projection, &mut context)
848                        .await?;
849                    intermediate_results = rows;
850                }
851                ExecutionStep::Filter { expression, .. } => {
852                    intermediate_results = self
853                        .execute_filter(intermediate_results, expression, &mut context)
854                        .await?;
855                }
856                ExecutionStep::Sort { order_by, .. } => {
857                    intermediate_results = self
858                        .execute_sort(intermediate_results, order_by, &mut context)
859                        .await?;
860                }
861                ExecutionStep::Aggregate { plan: agg_plan, .. } => {
862                    intermediate_results = self
863                        .execute_aggregation(intermediate_results, agg_plan, &mut context)
864                        .await?;
865                }
866                ExecutionStep::PerPartitionLimit { count } => {
867                    intermediate_results =
868                        Self::execute_per_partition_limit(intermediate_results, *count);
869                }
870                ExecutionStep::Limit { count, offset } => {
871                    intermediate_results = self
872                        .execute_limit(intermediate_results, *count, *offset, &mut context)
873                        .await?;
874                }
875                ExecutionStep::Project { columns } => {
876                    intermediate_results = self
877                        .execute_projection(intermediate_results, columns, &mut context)
878                        .await?;
879                }
880            }
881        }
882
883        let total_rows = intermediate_results.len() as u64;
884
885        // CRITICAL FIX (Issue #129/#140): Populate metadata.columns for SELECT *
886        // When SELECT * is used and no schema was found, context.columns is empty.
887        // Fall back to inferring column names from the first row's HashMap keys.
888        // IMPORTANT: Must be sorted alphabetically for deterministic JSON output (Issue #129)!
889        let mut columns = context.columns;
890        if columns.is_empty() && !intermediate_results.is_empty() {
891            // Try to resolve schema to get proper CQL types (Issue #674).
892            let schema_opt = if let Some(ref from_clause) = plan.statement.from_clause {
893                if let Ok(table_id) = self.extract_table_id(from_clause) {
894                    let (keyspace, table_name) = parse_table_id(&table_id);
895                    self._schema
896                        .find_schema_by_table(&keyspace, &table_name)
897                        .await
898                } else {
899                    None
900                }
901            } else {
902                None
903            };
904
905            let first_row = &intermediate_results[0];
906            let mut col_names: Vec<_> = first_row.values.keys().collect();
907            col_names.sort(); // Sort alphabetically for deterministic ordering (Issue #129)
908
909            let table_name_for_meta = schema_opt
910                .as_ref()
911                .map(|s| format!("{}.{}", s.keyspace, s.table));
912
913            for (idx, col_name) in col_names.iter().enumerate() {
914                // Look up CQL type from schema; derive flat DataType from it (Issue #674).
915                let cql_type_opt = schema_opt.as_ref().and_then(|schema| {
916                    schema
917                        .columns
918                        .iter()
919                        .find(|c| c.name.as_str() == col_name.as_str())
920                        .and_then(|c| parse_cql_type_str(&c.data_type))
921                });
922
923                let data_type = cql_type_opt
924                    .as_ref()
925                    .map(cql_type_to_data_type)
926                    .unwrap_or(crate::types::DataType::Text);
927
928                let mut col_info = ColumnInfo {
929                    name: (*col_name).clone(),
930                    data_type,
931                    nullable: true,
932                    position: idx,
933                    table_name: table_name_for_meta.clone(),
934                    cql_type: None,
935                };
936                if let Some(cql_type) = cql_type_opt {
937                    col_info = col_info.with_cql_type(cql_type);
938                }
939                columns.push(col_info);
940            }
941        }
942
943        Ok(QueryResult {
944            rows: intermediate_results,
945            rows_affected: total_rows, // Use actual number of rows returned
946            execution_time_ms: 0,      // Will be set by the engine
947            metadata: crate::query::result::QueryMetadata {
948                columns,
949                total_rows: Some(total_rows),
950                plan_info: None,
951                performance: Default::default(),
952                warnings: vec![],
953            },
954        })
955    }

957    /// Execute an optimized query plan with streaming results (Issue #280)
958    ///
959    /// Instead of materializing all rows in memory, this method returns a
960    /// `QueryResultIterator` that yields rows incrementally via a bounded channel.
961    /// This enables memory-efficient processing of large result sets.
962    ///
963    /// # Memory Budget
964    ///
965    /// With default `StreamingConfig::buffer_size` of 1024 rows and ~1KB avg row size:
966    /// - Channel buffer: ~1MB in flight
967    /// - Background task: minimal overhead
968    /// - Total streaming overhead: ~1-2MB (well within 128MB target)
969    ///
970    /// # Limitations
971    ///
972    /// Currently supports:
973    /// - SSTableScan with predicates (streaming)
974    /// - Filter/Limit/Project (applied during scan)
975    ///
976    /// `LIMIT` (and `OFFSET`, when present in the plan) is enforced by the
977    /// streaming producer (`execute_streaming_background`): it skips `OFFSET`
978    /// matches and stops scanning once `count` rows have been sent, so a
979    /// `LIMIT N` query yields exactly `N` rows without materializing the rest
980    /// (Issue #581).
981    ///
982    /// For ORDER BY/GROUP BY/DISTINCT, falls back to full execution then streams results.
983    pub async fn execute_streaming(
984        &self,
985        plan: OptimizedQueryPlan,
986        config: StreamingConfig,
987    ) -> Result<QueryResultIterator> {
988        // Check if query requires full materialization (ORDER BY, GROUP BY, aggregates)
989        if self.requires_materialization(&plan) {
990            log::info!("Query requires materialization (ORDER BY/GROUP BY/aggregates), using execute-then-stream");
991            return self.execute_and_stream(plan, config).await;
992        }
993
994        let table_id = if let Some(ref from_clause) = plan.statement.from_clause {
995            self.extract_table_id(from_clause)?
996        } else {
997            // For queries without FROM clause (like SELECT 1), fall back to execute
998            return self.execute_and_stream(plan, config).await;
999        };
1000
1001        let columns = self.get_result_columns(&plan.statement).await?;
1002
1003        // Create bounded channel for backpressure
1004        let (tx, rx) = mpsc::channel(config.buffer_size);
1005
1006        // Determine execution steps
1007        let execution_steps = if plan.execution_steps.is_empty() {
1008            vec![ExecutionStep::SSTableScan {
1009                table: table_id.clone(),
1010                predicates: vec![],
1011                projection: columns.iter().map(|c| c.name.clone()).collect(),
1012            }]
1013        } else {
1014            plan.execution_steps.clone()
1015        };
1016
1017        // Clone what we need for the background task
1018        let storage = Arc::clone(&self.storage);
1019        let schema_manager = Arc::clone(&self._schema);
1020        let buffer_size = config.buffer_size;
1021
1022        // Spawn background task to stream rows
1023        tokio::spawn(async move {
1024            if let Err(e) = Self::execute_streaming_background(
1025                storage,
1026                schema_manager,
1027                table_id,
1028                execution_steps,
1029                tx,
1030                buffer_size,
1031            )
1032            .await
1033            {
1034                log::error!("Streaming execution error: {}", e);
1035                // Error is logged; channel will close and consumer will see None
1036            }
1037        });
1038
1039        // Create metadata for the iterator
1040        let metadata = QueryMetadata {
1041            columns,
1042            total_rows: None, // Unknown for streaming
1043            plan_info: None,
1044            performance: Default::default(),
1045            warnings: vec![],
1046        };
1047
1048        Ok(QueryResultIterator::new(rx, metadata))
1049    }

1051    /// Check if query plan requires full materialization before streaming
1052    fn requires_materialization(&self, plan: &OptimizedQueryPlan) -> bool {
1053        for step in &plan.execution_steps {
1054            match step {
1055                ExecutionStep::Sort { .. } => return true,
1056                ExecutionStep::Aggregate { .. } => return true,
1057                _ => {}
1058            }
1059        }
1060
1061        // Check for DISTINCT
1062        if matches!(plan.statement.select_clause, SelectClause::Distinct(_)) {
1063            return true;
1064        }
1065
1066        // Issue #693: WRITETIME()/TTL() expressions require full materialisation
1067        // because the streaming background task only emits raw scan rows without
1068        // applying the WRITETIME/TTL projection (cell metadata extraction and
1069        // value computation).  Falling back to execute_and_stream ensures the
1070        // complete execute() path runs, which correctly populates writetime(col)/
1071        // ttl(col) keys in each row's values map.
1072        select_has_writetime_ttl(&plan.statement)
1073    }

1075    /// Fallback: Execute query fully, then stream the results
1076    async fn execute_and_stream(
1077        &self,
1078        plan: OptimizedQueryPlan,
1079        config: StreamingConfig,
1080    ) -> Result<QueryResultIterator> {
1081        // Execute full query
1082        let result = self.execute(plan).await?;
1083
1084        // Create channel to stream results
1085        let (tx, rx) = mpsc::channel(config.buffer_size);
1086
1087        // Spawn task to send rows through channel
1088        tokio::spawn(async move {
1089            for row in result.rows {
1090                if tx.send(Ok(row)).await.is_err() {
1091                    break; // Consumer dropped
1092                }
1093            }
1094            // Channel closes automatically when tx drops
1095        });
1096
1097        Ok(QueryResultIterator::new(rx, result.metadata))
1098    }

1100    /// Background task: Execute streaming scan and send rows through channel
1101    async fn execute_streaming_background(
1102        storage: Arc<StorageEngine>,
1103        schema_manager: Arc<SchemaManager>,
1104        _table_id: TableId,
1105        execution_steps: Vec<ExecutionStep>,
1106        tx: mpsc::Sender<Result<QueryRow>>,
1107        buffer_size: usize,
1108    ) -> Result<()> {
1109        // Issue #581: LIMIT/OFFSET must be enforced by the producer in the
1110        // streaming path. The `ExecutionStep::Limit` arm previously only logged a
1111        // message and relied on a consumer that never applied it, so
1112        // `execute_streaming` yielded the full result set regardless of LIMIT.
1113        // Extract the bound up front (steps are ordered with Limit after the scan)
1114        // and stop sending once it is satisfied — mirroring `execute_limit`
1115        // (drain OFFSET, then truncate to `count`) row-by-row so the producer
1116        // stops scanning early.
1117        let limit = execution_steps.iter().find_map(|step| match step {
1118            ExecutionStep::Limit { count, offset } => Some((*count, offset.unwrap_or(0))),
1119            _ => None,
1120        });
1121        let (limit_count, mut offset_remaining) = match limit {
1122            Some((count, offset)) => (Some(count), offset),
1123            None => (None, 0),
1124        };
1125
1126        // A `LIMIT 0` means no rows can ever be sent; return before scanning.
1127        if limit_count == Some(0) {
1128            return Ok(());
1129        }
1130
1131        // Issue #757: PER PARTITION LIMIT caps rows per partition before the
1132        // query-wide LIMIT/OFFSET. The scan yields rows grouped by partition
1133        // key, so we track the current partition (by its raw key bytes) and
1134        // reset the counter at each boundary.
1135        let per_partition_limit = execution_steps.iter().find_map(|step| match step {
1136            ExecutionStep::PerPartitionLimit { count } => Some(*count),
1137            _ => None,
1138        });
1139        let mut current_partition: Option<Vec<u8>> = None;
1140        let mut partition_count: u64 = 0;
1141
1142        let mut sent: u64 = 0;
1143
1144        for step in &execution_steps {
1145            match step {
1146                ExecutionStep::SSTableScan {
1147                    table,
1148                    predicates,
1149                    projection,
1150                    ..
1151                } => {
1152                    let (keyspace, table_name) = parse_table_id(table);
1153                    let schema_opt = schema_manager
1154                        .find_schema_by_table(&keyspace, &table_name)
1155                        .await;
1156
1157                    // Issue #949: a fully-constrained `WHERE pk = ?` is served by a
1158                    // partition-targeted lookup that prunes SSTables via bloom/BTI,
1159                    // instead of streaming a scan over every SSTable. The resulting
1160                    // rows are sent through the same per-row pipeline below
1161                    // (predicates, PER PARTITION LIMIT, OFFSET, LIMIT). Note
1162                    // `scan_partition` reconciles across SSTable generations like the
1163                    // materializing `scan()` (last-write-wins + tombstone shadowing),
1164                    // which is the authoritative read semantics; it does not merely
1165                    // mirror `scan_stream`'s per-key merge.
1166                    if let Some(pk_bytes) =
1167                        full_partition_key_lookup(predicates, schema_opt.as_ref())
1168                    {
1169                        let rows = storage
1170                            .scan_partition(table, &pk_bytes, schema_opt.as_ref())
1171                            .await?;
1172                        for (key, value) in rows {
1173                            let part_sig = per_partition_limit.map(|_| key.0.clone());
1174                            let Some(row) =
1175                                build_row_from_scan(key, value, projection, schema_opt.as_ref())
1176                            else {
1177                                continue;
1178                            };
1179                            if !evaluate_predicates(&row, predicates)? {
1180                                continue;
1181                            }
1182                            if let (Some(cap), Some(sig)) = (per_partition_limit, part_sig) {
1183                                if current_partition.as_deref() != Some(sig.as_slice()) {
1184                                    current_partition = Some(sig);
1185                                    partition_count = 0;
1186                                }
1187                                if partition_count >= cap {
1188                                    continue;
1189                                }
1190                                partition_count += 1;
1191                            }
1192                            if offset_remaining > 0 {
1193                                offset_remaining -= 1;
1194                                continue;
1195                            }
1196                            if tx.send(Ok(row)).await.is_err() {
1197                                return Ok(());
1198                            }
1199                            sent += 1;
1200                            if let Some(count) = limit_count {
1201                                if sent >= count {
1202                                    return Ok(());
1203                                }
1204                            }
1205                        }
1206                        // This SSTableScan step is fully served by the lookup.
1207                        continue;
1208                    }
1209
1210                    // Issue #790: pull rows lazily from a bounded streaming scan
1211                    // instead of materializing the full result `Vec`. The reader
1212                    // parses one entry at a time into this channel, so live heap
1213                    // stays bounded by `buffer_size` rather than O(result rows).
1214                    let mut scan_stream = storage
1215                        .scan_stream(table, None, None, schema_opt.as_ref(), buffer_size)
1216                        .await?;
1217
1218                    while let Some(item) = scan_stream.recv().await {
1219                        let (key, value) = item?;
1220                        // Capture the partition key bytes before `key` is moved
1221                        // into row construction (only when needed).
1222                        let part_sig = per_partition_limit.map(|_| key.0.clone());
1223                        let Some(row) =
1224                            build_row_from_scan(key, value, projection, schema_opt.as_ref())
1225                        else {
1226                            continue;
1227                        };
1228
1229                        if !evaluate_predicates(&row, predicates)? {
1230                            continue;
1231                        }
1232
1233                        // Apply PER PARTITION LIMIT: cap matching rows per
1234                        // partition, before OFFSET/LIMIT (Cassandra semantics).
1235                        if let (Some(cap), Some(sig)) = (per_partition_limit, part_sig) {
1236                            if current_partition.as_deref() != Some(sig.as_slice()) {
1237                                current_partition = Some(sig);
1238                                partition_count = 0;
1239                            }
1240                            if partition_count >= cap {
1241                                continue;
1242                            }
1243                            partition_count += 1;
1244                        }
1245
1246                        // Apply OFFSET: skip the first `offset_remaining` matches.
1247                        if offset_remaining > 0 {
1248                            offset_remaining -= 1;
1249                            continue;
1250                        }
1251
1252                        // Send row through channel (with backpressure). Consumer drop ends the scan.
1253                        if tx.send(Ok(row)).await.is_err() {
1254                            return Ok(());
1255                        }
1256                        sent += 1;
1257
1258                        // Apply LIMIT: stop scanning once `count` rows have been
1259                        // sent. Dropping `scan_stream` here signals the producer
1260                        // (via a closed channel) to stop parsing early.
1261                        if let Some(count) = limit_count {
1262                            if sent >= count {
1263                                return Ok(());
1264                            }
1265                        }
1266                    }
1267                }
1268                ExecutionStep::Limit { .. } | ExecutionStep::PerPartitionLimit { .. } => {
1269                    // Enforced inline during the scan above (see the bounds
1270                    // extracted before the loop).
1271                }
1272                // Projection and predicate filtering are pushed into SSTableScan above.
1273                ExecutionStep::Project { .. } | ExecutionStep::Filter { .. } => {}
1274                _ => {
1275                    log::warn!("Streaming execution: skipping unsupported step {:?}", step);
1276                }
1277            }
1278        }
1279
1280        Ok(())
1281    }

1283    /// Execute SSTable scan with predicate pushdown.
1284    ///
1285    /// Per-row work (build row, decode partition key, evaluate predicates) is
1286    /// handled by the free helpers `build_row_from_scan` and
1287    /// `evaluate_predicates`, which are shared with the streaming background
1288    /// task to keep the two execution paths in lockstep.
1289    async fn execute_sstable_scan(
1290        &self,
1291        table: &TableId,
1292        predicates: &[SSTablePredicate],
1293        projection: &[String],
1294        context: &mut ExecutionContext,
1295    ) -> Result<Vec<QueryRow>> {
1296        const MAX_RESULTS: usize = 1_000_000;
1297
1298        log::info!(
1299            "Executing SSTableScan: table=\"{}\", predicates={:?}, include_cell_metadata={}",
1300            table,
1301            predicates,
1302            context.projection_flags.include_cell_metadata,
1303        );
1304
1305        let (keyspace, table_name) = parse_table_id(table);
1306        let schema_opt = self
1307            ._schema
1308            .find_schema_by_table(&keyspace, &table_name)
1309            .await;
1310
1311        match schema_opt.as_ref() {
1312            Some(schema) => log::info!(
1313                "Found schema for {}.{} with {} columns",
1314                schema.keyspace,
1315                schema.table,
1316                schema.columns.len()
1317            ),
1318            None => log::info!(
1319                "No schema found for {}.{}, proceeding without schema-aware parsing",
1320                keyspace.as_deref().unwrap_or("unknown"),
1321                table_name
1322            ),
1323        }
1324
1325        // Issue #693: When WRITETIME(col) or TTL(col) is in the SELECT, use the
1326        // metadata-carrying scan so per-cell timestamps reach the QueryRow.
1327        let mut results = Vec::new();
1328        if context.projection_flags.include_cell_metadata {
1329            let scan_results = self
1330                .storage
1331                .scan_with_cell_metadata(table, None, None, None, schema_opt.as_ref())
1332                .await?;
1333
1334            log::info!("Scan (with metadata) returned {} rows", scan_results.len());
1335
1336            for (key, value, cell_meta) in scan_results {
1337                context.rows_processed += 1;
1338
1339                let Some(mut row) =
1340                    build_row_from_scan(key, value, projection, schema_opt.as_ref())
1341                else {
1342                    continue;
1343                };
1344
1345                // Attach per-cell metadata so evaluate_writetime_ttl can read it.
1346                if !cell_meta.is_empty() {
1347                    row.set_cell_metadata(cell_meta);
1348                }
1349
1350                if evaluate_predicates(&row, predicates)? {
1351                    results.push(row);
1352                }
1353
1354                if results.len() > MAX_RESULTS {
1355                    return Err(Error::query_execution(
1356                        "Result set too large, consider adding LIMIT".to_string(),
1357                    ));
1358                }
1359            }
1360        } else {
1361            // Issue #949: a fully-constrained `WHERE pk = ?` is served by a
1362            // partition-targeted lookup that prunes SSTables via bloom/BTI and only
1363            // parses the candidates, instead of scanning every SSTable for the
1364            // table. Falls back to a full scan when the partition key isn't fully
1365            // pinned or can't be encoded. The per-row predicate evaluation below is
1366            // unchanged, so clustering predicates and the pk equality itself are
1367            // still applied (and any over-inclusion is filtered out).
1368            let scan_results = if let Some(pk_bytes) =
1369                full_partition_key_lookup(predicates, schema_opt.as_ref())
1370            {
1371                log::info!(
1372                    "SSTableScan: partition-key point lookup (key len={}) for \"{}\"",
1373                    pk_bytes.len(),
1374                    table
1375                );
1376                self.storage
1377                    .scan_partition(table, &pk_bytes, schema_opt.as_ref())
1378                    .await?
1379            } else {
1380                self.storage
1381                    .scan(table, None, None, None, schema_opt.as_ref())
1382                    .await?
1383            };
1384
1385            log::info!("Scan returned {} rows", scan_results.len());
1386
1387            for (key, value) in scan_results {
1388                context.rows_processed += 1;
1389
1390                // build_row_from_scan returns None for tombstoned/null rows (Issue #191).
1391                let Some(row) = build_row_from_scan(key, value, projection, schema_opt.as_ref())
1392                else {
1393                    continue;
1394                };
1395
1396                if evaluate_predicates(&row, predicates)? {
1397                    results.push(row);
1398                }
1399
1400                if results.len() > MAX_RESULTS {
1401                    return Err(Error::query_execution(
1402                        "Result set too large, consider adding LIMIT".to_string(),
1403                    ));
1404                }
1405            }
1406        }
1407
1408        Ok(results)
1409    }

1430    /// Evaluate WHERE expression against a row
1431    fn evaluate_where_expression(&self, expr: &WhereExpression, row: &QueryRow) -> Result<bool> {
1432        match expr {
1433            WhereExpression::Comparison(comp) => self.evaluate_comparison(comp, row),
1434            WhereExpression::And(exprs) => {
1435                for expr in exprs {
1436                    if !self.evaluate_where_expression(expr, row)? {
1437                        return Ok(false);
1438                    }
1439                }
1440                Ok(true)
1441            }
1442            WhereExpression::Or(exprs) => {
1443                for expr in exprs {
1444                    if self.evaluate_where_expression(expr, row)? {
1445                        return Ok(true);
1446                    }
1447                }
1448                Ok(false)
1449            }
1450            WhereExpression::Not(expr) => Ok(!self.evaluate_where_expression(expr, row)?),
1451            WhereExpression::Parentheses(expr) => self.evaluate_where_expression(expr, row),
1452        }
1453    }

1455    /// Evaluate comparison expression. Operators that need a single right
1456    /// operand share one `evaluate` call; IN/LIKE/IS NULL fall through to
1457    /// their custom branches.
1458    fn evaluate_comparison(&self, comp: &ComparisonExpression, row: &QueryRow) -> Result<bool> {
1459        use ComparisonOperator::*;
1460
1461        let left_value = self.evaluate_select_expression(&comp.left, row)?;
1462
1463        // Fast path for null tests, which ignore the right side.
1464        match comp.operator {
1465            IsNull => return Ok(left_value.is_null()),
1466            IsNotNull => return Ok(!left_value.is_null()),
1467            _ => {}
1468        }
1469
1470        match (&comp.operator, &comp.right) {
1471            (
1472                op @ (Equal | NotEqual | LessThan | LessThanOrEqual | GreaterThan
1473                | GreaterThanOrEqual),
1474                ComparisonRightSide::Value(right_expr),
1475            ) => {
1476                let right_value = self.evaluate_select_expression(right_expr, row)?;
1477                let result = match op {
1478                    Equal => values_equal(&left_value, &right_value),
1479                    NotEqual => !values_equal(&left_value, &right_value),
1480                    LessThan => try_compare_values(&left_value, &right_value)?.is_lt(),
1481                    LessThanOrEqual => try_compare_values(&left_value, &right_value)?.is_le(),
1482                    GreaterThan => try_compare_values(&left_value, &right_value)?.is_gt(),
1483                    GreaterThanOrEqual => try_compare_values(&left_value, &right_value)?.is_ge(),
1484                    _ => unreachable!("guarded by outer match"),
1485                };
1486                Ok(result)
1487            }
1488            (In, ComparisonRightSide::ValueList(value_exprs)) => {
1489                for value_expr in value_exprs {
1490                    let value = self.evaluate_select_expression(value_expr, row)?;
1491                    if left_value == value {
1492                        return Ok(true);
1493                    }
1494                }
1495                Ok(false)
1496            }
1497            (Like, ComparisonRightSide::Value(pattern_expr)) => {
1498                let pattern = self.evaluate_select_expression(pattern_expr, row)?;
1499                if let (Value::Text(text), Value::Text(pattern_str)) = (&left_value, &pattern) {
1500                    Ok(self.match_like_pattern(text, pattern_str))
1501                } else {
1502                    Ok(false)
1503                }
1504            }
1505            _ => Err(Error::query_execution(
1506                "Unsupported comparison operator".to_string(),
1507            )),
1508        }
1509    }

1511    /// Evaluate SELECT expression against a row
1512    fn evaluate_select_expression(&self, expr: &SelectExpression, row: &QueryRow) -> Result<Value> {
1513        match expr {
1514            SelectExpression::Column(col_ref) => {
1515                row.values.get(&col_ref.column).cloned().ok_or_else(|| {
1516                    Error::query_execution(format!("Column not found: {}", col_ref.column))
1517                })
1518            }
1519            SelectExpression::Literal(value) => Ok(value.clone()),
1520            SelectExpression::CollectionAccess(access) => {
1521                self.evaluate_collection_access(access, row)
1522            }
1523            SelectExpression::Arithmetic(arith) => {
1524                let left = self.evaluate_select_expression(&arith.left, row)?;
1525                let right = self.evaluate_select_expression(&arith.right, row)?;
1526                self.evaluate_arithmetic(&arith.operator, left, right)
1527            }
1528            SelectExpression::Aliased(expr, _) => self.evaluate_select_expression(expr, row),
1529            SelectExpression::Aggregate(_) => {
1530                // Aggregate expressions should not be evaluated at row level
1531                // They should only be processed during the aggregation step
1532                Err(Error::query_execution(
1533                    "Aggregate expressions should be processed during aggregation step, not row evaluation".to_string(),
1534                ))
1535            }
1536            SelectExpression::Function(_) => {
1537                // Function expressions not yet implemented
1538                Err(Error::query_execution(
1539                    "Function expressions not yet implemented".to_string(),
1540                ))
1541            }
1542            // Issue #692: evaluate WRITETIME(col) / TTL(col) against the per-cell
1543            // metadata carrier threaded by the reader when `ProjectionFlags::include_cell_metadata`
1544            // is set. Returns `Value::Null` when metadata is absent (e.g. no schema-aware
1545            // read path or the column was a partition-key column with no cell header).
1546            SelectExpression::WriteTimeTtl(call) => {
1547                let now_secs = self.clock.now_seconds();
1548                Ok(evaluate_writetime_ttl(call, row, now_secs))
1549            }
1550        }
1551    }

1553    /// Evaluate collection access operations (`list[idx]`, `map['key']`,
1554    /// `value IN set_column`).
1555    fn evaluate_collection_access(
1556        &self,
1557        access: &CollectionAccessExpression,
1558        row: &QueryRow,
1559    ) -> Result<Value> {
1560        let lookup_column = |col: &ColumnRef| -> Result<&Value> {
1561            row.values
1562                .get(&col.column)
1563                .ok_or_else(|| Error::query_execution(format!("Column not found: {}", col.column)))
1564        };
1565
1566        match access {
1567            CollectionAccessExpression::ListIndex(col_ref, index_expr) => {
1568                let list_value = lookup_column(col_ref)?;
1569                let index_value = self.evaluate_select_expression(index_expr, row)?;
1570
1571                let (Value::List(list), Value::Integer(index)) = (list_value, &index_value) else {
1572                    return Err(Error::query_execution("Invalid list access".to_string()));
1573                };
1574                if *index >= 0 && (*index as usize) < list.len() {
1575                    Ok(list[*index as usize].clone())
1576                } else {
1577                    Ok(Value::Null)
1578                }
1579            }
1580            CollectionAccessExpression::MapKey(col_ref, key_expr) => {
1581                let map_value = lookup_column(col_ref)?;
1582                let key_value = self.evaluate_select_expression(key_expr, row)?;
1583
1584                let Value::Map(map) = map_value else {
1585                    return Err(Error::query_execution("Invalid map access".to_string()));
1586                };
1587                Ok(map
1588                    .iter()
1589                    .find(|(k, _)| *k == key_value)
1590                    .map(|(_, v)| v.clone())
1591                    .unwrap_or(Value::Null))
1592            }
1593            CollectionAccessExpression::SetContains(col_ref, value_expr) => {
1594                let set_value = lookup_column(col_ref)?;
1595                let test_value = self.evaluate_select_expression(value_expr, row)?;
1596
1597                let Value::Set(set) = set_value else {
1598                    return Err(Error::query_execution(
1599                        "Invalid set contains operation".to_string(),
1600                    ));
1601                };
1602                Ok(Value::Boolean(set.contains(&test_value)))
1603            }
1604        }
1605    }

1667    /// Execute the aggregation step. Splits naturally into three phases:
1668    /// build group key, accumulate per-aggregate state, then finalize each
1669    /// group into a result row.
1670    async fn execute_aggregation(
1671        &self,
1672        rows: Vec<QueryRow>,
1673        agg_plan: &AggregationPlan,
1674        _context: &mut ExecutionContext,
1675    ) -> Result<Vec<QueryRow>> {
1676        const PER_ROW_MEMORY_ESTIMATE_BYTES: usize = 100;
1677        const DEFAULT_AGGREGATION_MEMORY_LIMIT: usize = 512 * 1024 * 1024;
1678
1679        let mut agg_state = AggregationState {
1680            groups: Vec::new(),
1681            memory_usage_bytes: 0,
1682            memory_limit_bytes: DEFAULT_AGGREGATION_MEMORY_LIMIT,
1683        };
1684
1685        for row in rows {
1686            let group_key = build_group_key(&row, &agg_plan.group_by_columns);
1687            let group_index =
1688                find_or_init_group(&mut agg_state.groups, group_key, &agg_plan.aggregates);
1689            let group_aggregates = &mut agg_state.groups[group_index].1;
1690
1691            for (i, agg_comp) in agg_plan.aggregates.iter().enumerate() {
1692                update_aggregate(&mut group_aggregates[i], agg_comp, &row);
1693            }
1694
1695            agg_state.memory_usage_bytes += PER_ROW_MEMORY_ESTIMATE_BYTES;
1696            if agg_state.memory_usage_bytes > agg_state.memory_limit_bytes {
1697                return Err(Error::query_execution(
1698                    "Aggregation memory limit exceeded".to_string(),
1699                ));
1700            }
1701        }
1702
1703        let result_rows = agg_state
1704            .groups
1705            .into_iter()
1706            .map(|(group_key, group_aggregates)| {
1707                finalize_group(group_key, group_aggregates, agg_plan)
1708            })
1709            .collect();
1710
1711        Ok(result_rows)
1712    }

1714    /// Execute PER PARTITION LIMIT: keep at most `count` rows per partition,
1715    /// preserving order (Issue #757). Counts are keyed on the partition (raw key
1716    /// bytes) rather than tracking only the most recent partition, so the cap
1717    /// holds even when a partition's rows are not contiguous — e.g. when an
1718    /// upstream `ORDER BY` interleaves rows from different partitions (roborev
1719    /// job 38).
1720    fn execute_per_partition_limit(rows: Vec<QueryRow>, count: u64) -> Vec<QueryRow> {
1721        let mut out = Vec::with_capacity(rows.len());
1722        let mut counts: HashMap<Vec<u8>, u64> = HashMap::new();
1723        for row in rows {
1724            let seen = counts.entry(row.key.0.clone()).or_insert(0);
1725            if *seen < count {
1726                *seen += 1;
1727                out.push(row);
1728            }
1729        }
1730        out
1731    }

1733    /// Execute limit step (apply OFFSET then truncate to LIMIT).
1734    async fn execute_limit(
1735        &self,
1736        mut rows: Vec<QueryRow>,
1737        count: u64,
1738        offset: Option<u64>,
1739        _context: &mut ExecutionContext,

1862    /// Extract a `TableId` from a FROM clause. Cassandra CQL has no JOINs, so
1863    /// either form (bare table or aliased table) yields the same result.

1872    async fn get_result_columns(&self, statement: &SelectStatement) -> Result<Vec<ColumnInfo>> {
1873        let mut columns = Vec::new();
1874
1875        match &statement.select_clause {
1876            SelectClause::All => {
1877                // For SELECT *, look up the schema to get column names and CQL types.
1878                // This is needed for streaming mode where we can't wait for the first row.
1879                if let Some(ref from_clause) = statement.from_clause {
1880                    let table_id = self.extract_table_id(from_clause)?;
1881                    let (keyspace_opt, table_name) = parse_table_id(&table_id);
1882
1883                    // Look up schema from SchemaManager
1884                    if let Some(schema) = self
1885                        ._schema
1886                        .find_schema_by_table(&keyspace_opt, &table_name)
1887                        .await
1888                    {
1889                        // Collect all schema columns (sorted alphabetically for determinism)
1890                        let mut schema_cols: Vec<&crate::schema::Column> =
1891                            schema.columns.iter().collect();
1892                        schema_cols.sort_by_key(|c| c.name.as_str());
1893
1894                        let keyspace_str = keyspace_opt.as_deref().unwrap_or("");
1895                        let table_name_str = format!("{}.{}", keyspace_str, table_name);
1896
1897                        for (idx, schema_col) in schema_cols.iter().enumerate() {
1898                            // Parse the CQL type string into a structured CqlType (Issue #674).
1899                            let cql_type_opt = parse_cql_type_str(&schema_col.data_type);
1900                            // Derive the flat DataType from the CqlType; avoids hardcoded Text.
1901                            let data_type = cql_type_opt
1902                                .as_ref()
1903                                .map(cql_type_to_data_type)
1904                                .unwrap_or(crate::types::DataType::Text);
1905
1906                            let mut col_info = ColumnInfo {
1907                                name: schema_col.name.clone(),
1908                                data_type,
1909                                nullable: true,
1910                                position: idx,
1911                                table_name: Some(table_name_str.clone()),
1912                                cql_type: None,
1913                            };
1914                            if let Some(cql_type) = cql_type_opt {
1915                                col_info = col_info.with_cql_type(cql_type);
1916                            }
1917                            columns.push(col_info);
1918                        }
1919
1920                        log::debug!(
1921                            "SELECT * resolved {} columns from schema for {:?}.{}",
1922                            columns.len(),
1923                            keyspace_opt,
1924                            table_name
1925                        );
1926                    }
1927                    // If schema not found, columns stay empty - will be populated from first row at runtime
1928                }
1929            }
1930            SelectClause::Columns(exprs) | SelectClause::Distinct(exprs) => {
1931                // Try to resolve a schema for the FROM table (if present) so we can
1932                // attach authoritative CQL types to explicitly projected columns (Issue #674).
1933                let schema_opt = if let Some(ref from_clause) = statement.from_clause {
1934                    if let Ok(table_id) = self.extract_table_id(from_clause) {
1935                        let (keyspace_opt, table_name) = parse_table_id(&table_id);
1936                        self._schema
1937                            .find_schema_by_table(&keyspace_opt, &table_name)
1938                            .await
1939                    } else {
1940                        None
1941                    }
1942                } else {
1943                    None
1944                };
1945
1946                for (i, expr) in exprs.iter().enumerate() {
1947                    // Issue #692: WriteTimeTtl expressions produce fixed-schema output
1948                    // columns with Cassandra-convention names, independent of the table schema.
1949                    if let SelectExpression::WriteTimeTtl(call) = expr {
1950                        let col_name = writetime_ttl_column_name(call);
1951                        let (data_type, cql_type) = match call.function {
1952                            // WRITETIME returns bigint (µs since epoch)
1953                            WriteTimeTtlFunction::WriteTime => {
1954                                (crate::types::DataType::BigInt, Some(CqlType::BigInt))
1955                            }
1956                            // TTL returns int (remaining seconds)
1957                            WriteTimeTtlFunction::Ttl => {
1958                                (crate::types::DataType::Integer, Some(CqlType::Int))
1959                            }
1960                        };
1961                        let mut col_info = ColumnInfo {
1962                            name: col_name,
1963                            data_type,
1964                            nullable: true, // always nullable — absent cell → NULL
1965                            position: i,
1966                            table_name: None,
1967                            cql_type: None,
1968                        };
1969                        if let Some(ct) = cql_type {
1970                            col_info = col_info.with_cql_type(ct);
1971                        }
1972                        columns.push(col_info);
1973                        continue;
1974                    }
1975
1976                    let column_name = match expr {
1977                        SelectExpression::Column(col_ref) => col_ref.column.clone(),
1978                        SelectExpression::Aliased(_, alias) => alias.clone(),
1979                        _ => format!("col_{i}"),
1980                    };
1981
1982                    // Look up CQL type for this column in the schema (Issue #674).
1983                    let cql_type_opt = schema_opt.as_ref().and_then(|schema| {
1984                        schema
1985                            .columns
1986                            .iter()
1987                            .find(|c| c.name == column_name)
1988                            .and_then(|c| parse_cql_type_str(&c.data_type))
1989                    });
1990                    let data_type = cql_type_opt
1991                        .as_ref()
1992                        .map(cql_type_to_data_type)
1993                        .unwrap_or(crate::types::DataType::Text);
1994
1995                    let mut col_info = ColumnInfo {
1996                        name: column_name,
1997                        data_type,
1998                        nullable: true,
1999                        position: i,
2000                        table_name: None,
2001                        cql_type: None,
2002                    };
2003                    if let Some(cql_type) = cql_type_opt {
2004                        col_info = col_info.with_cql_type(cql_type);
2005                    }
2006                    columns.push(col_info);
2007                }
2008            }
2009        }
2010
2011        Ok(columns)
2012    }
