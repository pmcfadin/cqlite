1//! Row assembly and table-id/type helpers for the SELECT executor.
2//!
3//! [`build_row_from_scan`] turns a raw `(RowKey, ScanRow)` scan entry into a
4//! `QueryRow`, reconstructing partition-key columns from the raw key. It is part
5//! of the public surface (re-exported via `query::mod`) so other readers (e.g.
6//! the Arrow Flight compaction-merge producer) assemble rows identically.

18/// Split a `TableId` of the form `"keyspace.table"` into its parts.
19///
20/// If no dot is present, the whole name becomes the table component and the
21/// keyspace is `None`.
22pub(super) fn parse_table_id(table_id: &TableId) -> (Option<String>, String) {
23    let table_str = table_id.name();
24    match table_str.rfind('.') {
25        Some(dot) => (
26            Some(table_str[..dot].to_string()),
27            table_str[dot + 1..].to_string(),
28        ),
29        None => (None, table_str.to_string()),
30    }
31}

33/// Parse a CQL type string (e.g. `"list<int>"`, `"text"`) into a [`CqlType`].
34///
35/// Returns `None` when the type string cannot be parsed (unknown or malformed
36/// types). Used to populate `ColumnInfo::cql_type` from the schema's string
37/// representation, satisfying the no-heuristics mandate (Issue #28).
38pub(super) fn parse_cql_type_str(type_str: &str) -> Option<CqlType> {
39    let parser = ComplexTypeParser::new();
40    parser
41        .parse_type(type_str)
42        .ok()
43        .map(|parsed| parsed.cql_type)
44}

46/// Build a `ColumnInfo` from a schema column name + CQL type string, deriving
47/// the flat `DataType` from the parsed `CqlType` (Issue #674). Centralises the
48/// "name + type string → ColumnInfo" pattern shared by the SELECT-* and
49/// explicit-projection column builders.
50pub(super) fn column_info_from_type_str(
51    name: String,
52    type_str: &str,
53    position: usize,
54    table_name: Option<String>,
55) -> ColumnInfo {
56    let cql_type_opt = parse_cql_type_str(type_str);
57    let data_type = cql_type_opt
58        .as_ref()
59        .map(cql_type_to_data_type)
60        .unwrap_or(crate::types::DataType::Text);
61    let mut col_info = ColumnInfo {
62        name,
63        data_type,
64        nullable: true,
65        position,
66        table_name,
67        cql_type: None,
68    };
69    if let Some(cql_type) = cql_type_opt {
70        col_info = col_info.with_cql_type(cql_type);
71    }
72    col_info
73}

75/// Build a `QueryRow` from a single `(RowKey, ScanRow)` produced by storage scan,
76/// applying optional projection and synthesising partition-key columns from the
77/// raw key bytes when a schema is available.
78///
79/// Partition-key columns are never stored in the cell payload, so they are
80/// reconstructed from the raw row key via the canonical
81/// [`crate::storage::partition_key_codec::decode_partition_key_columns`] (the
82/// same codec the write engine uses). This is the fix for Issue #586: the
83/// previous decoder assumed a `u16` length prefix for every TEXT key, which is
84/// only correct for composite components — a single-component TEXT partition key
85/// is raw bytes, so its column was silently dropped from scan-built rows.
86///
87/// Returns `None` for tombstoned rows (so the caller can `continue`).
88///
89/// Exposed publicly so other readers (e.g. the Arrow Flight server's compaction
90/// merge producer) can assemble rows identically to the SELECT path, guaranteeing
91/// output parity. The `row` carries the decoded non-partition-key cells as the
92/// single [`ScanRow`] row carrier (issue #1334); partition-key columns are
93/// reconstructed from `key`.
94pub fn build_row_from_scan(
95    key: RowKey,
96    row: ScanRow,
97    projection: &[String],
98    schema: Option<&crate::schema::TableSchema>,
99) -> Option<QueryRow> {
100    // Suppress tombstoned / absent rows from user-visible output. A row tombstone
101    // or null row reaches here as `ScanRow::Marker` (Issue #505); it must never
102    // appear in query results. A live row is always `ScanRow::Row`, so there is
103    // exactly ONE row-carrier path — a live row's column values can never silently
104    // fall through to a non-row fallback (issue #1334 / roborev H2).
105    let cells = row.into_cells()?;
106
107    // Issue #1584: pre-size the value map to the upper bound of inserts — the
108    // decoded cell count plus the reconstructed partition-key columns. This is a
109    // single sized allocation with no rehash growth (for `SELECT *` it is exact;
110    // for a narrower projection it slightly over-reserves, still one alloc).
111    let pk_hint = schema.map(|s| s.partition_keys.len()).unwrap_or(0);
112    let mut row_values: HashMap<Arc<str>, Value> = HashMap::with_capacity(cells.len() + pk_hint);
113    let project = |name: &str| projection.is_empty() || projection.iter().any(|p| p == name);
114
115    // Issue #1334: the decoder carries interned `Arc<str>` column-name handles in
116    // the row carrier; move them straight into `QueryRow.values` (an `Arc` move —
117    // NO `String` re-allocation of the name).
118    for (name, col_value) in cells {
119        if project(&name) {
120            row_values.insert(name, col_value);
121        }
122    }
123    // Cassandra never serialises partition-key columns in the cell payload;
124    // reconstruct them from the raw row key when the schema is known. We
125    // decode through the canonical codec shared with the write engine so
126    // single-component (raw bytes) and composite (`[u16 len][bytes][0x00]`)
127    // keys are handled identically on both paths (Issue #586).
128    if let Some(schema) = schema {
129        match crate::storage::partition_key_codec::decode_partition_key_columns(&key.0, schema) {
130            Ok(pk_columns) => {
131                for (name, value) in pk_columns {
132                    if project(&name) {
133                        row_values.insert(name.into(), value);
134                    }
135                }
136            }
137            // Surface — never silently swallow — a decode failure, so a
138            // missing partition-key column can't ship invisibly (Issue #586).
139            Err(e) => {
140                log::warn!(
141                    "Failed to reconstruct partition-key columns from row key \
142                     (len={} bytes) for {}.{}: {}",
143                    key.0.len(),
144                    schema.keyspace,
145                    schema.table,
146                    e
147                );
148            }
149        }
150    }
151
152    Some(QueryRow {
153        values: row_values,
154        key,
155        metadata: Default::default(),
156        cell_metadata: None,
157    })
158}

166    /// Issue #586: a single-component TEXT partition key is stored as raw bytes
167    /// with NO length prefix. `build_row_from_scan` must materialise it from the
168    /// `RowKey`. Before the fix the column was silently dropped (the decoder
169    /// read a phantom `u16` prefix, errored, and the error was swallowed).
170    #[test]
171    fn build_row_from_scan_materialises_single_text_pk() {
172        let key = RowKey::new(b"k0000000000000000".to_vec());
173        let value = ScanRow::Row(vec![(Arc::from("name"), Value::Text("name-0".to_string()))]);
174        let schema = single_pk_schema("id", "text");
175
176        let row = build_row_from_scan(key, value, &[], Some(&schema))
177            .expect("row must be built (not tombstoned)");
178
179        assert_eq!(
180            row.values.get("id"),
181            Some(&Value::Text("k0000000000000000".to_string())),
182            "Issue #586: single TEXT PK column must be reconstructed from the raw row key"
183        );
184        // Regular columns must still be present.
185        assert_eq!(
186            row.values.get("name"),
187            Some(&Value::Text("name-0".to_string()))
188        );
189    }

214    /// Issue #1334 / roborev H2: a multi-column live `ScanRow::Row` must
215    /// disassemble into EVERY named column value — never collapse into a
216    /// synthetic `"data"` fallback (the bug where a non-`ScanRow::Row` carrier
217    /// dropped all column values). This pins the single-carrier contract:
218    /// `build_row_from_scan` yields the real columns and NO fallback key.
219    #[test]
220    fn build_row_from_scan_multi_column_row_has_no_data_fallback() {
221        let key = RowKey::new(b"k0000000000000000".to_vec());
222        let value = ScanRow::Row(vec![
223            (Arc::from("name"), Value::Text("alice".to_string())),
224            (Arc::from("score"), Value::Integer(42)),
225        ]);
226
227        // No schema → no partition-key reconstruction; only the row's own cells.
228        let row = build_row_from_scan(key, value, &[], None)
229            .expect("a live row must build (not tombstoned)");
230
231        assert_eq!(
232            row.values.get("name"),
233            Some(&Value::Text("alice".to_string())),
234            "real text column value must survive the row-carrier disassembly"
235        );
236        assert_eq!(
237            row.values.get("score"),
238            Some(&Value::Integer(42)),
239            "real int column value must survive the row-carrier disassembly"
240        );
241        assert!(
242            !row.values.contains_key("data"),
243            "roborev H2: column values must NOT collapse into a synthetic 'data' fallback"
244        );
245        assert_eq!(
246            row.values.len(),
247            2,
248            "exactly the two real columns, no extras"
249        );
250    }

252    /// Issue #1584: the row value map is pre-sized to the decoded cell count so a
253    /// single sized allocation covers the row (no rehash growth). Pinned via a
254    /// narrow projection: 8 cells are decoded but only ONE survives projection —
255    /// the map's capacity must still reflect the 8-cell hint (`>= 8`). With the
256    /// pre-fix `HashMap::new()` the map is grown from empty to the single inserted
257    /// entry (capacity 3), which fails this lower bound.
258    #[test]
259    fn build_row_from_scan_presizes_value_map() {
260        let cells: Vec<(Arc<str>, Value)> = (0..8)
261            .map(|i| (Arc::from(format!("c{i}").as_str()), Value::Integer(i)))
262            .collect();
263        let key = RowKey::new(b"k".to_vec());
264
265        let row = build_row_from_scan(key, ScanRow::Row(cells), &["c0".to_string()], None)
266            .expect("a live row must build");
267
268        assert_eq!(row.values.len(), 1, "projection keeps exactly one column");
269        assert!(
270            row.values.capacity() >= 8,
271            "issue #1584: value map must be pre-sized to the decoded cell count \
272             (>= 8), not grown from empty to the projected size; got capacity {}",
273            row.values.capacity()
274        );
275    }

287    /// Issue #1334 (roborev round 9, finding 2): the canonical query consumer
288    /// SUPPRESSES a `ScanRow::Marker` but SURFACES a live `ScanRow::Row`. This is
289    /// exactly why the CLI `read`/`inspect`/`benchmark` bulletproof-reader fallback
290    /// producers must NOT wrap a LIVE synthetic value in `Marker` — doing so drops
291    /// it from user-visible output. The SAME live value must survive as a `Row`.
292    #[test]
293    fn live_value_dropped_as_marker_surfaces_as_row() {
294        let key = RowKey::new(b"k".to_vec());
295        let live = Value::Text("synthetic-fallback".to_string());
296
297        // Wrapped as a Marker (the pre-fix producer): dropped entirely.
298        assert!(
299            build_row_from_scan(key.clone(), ScanRow::Marker(live.clone()), &[], None).is_none(),
300            "a LIVE value mis-wrapped as Marker is dropped — the bug the producers must avoid"
301        );
302
303        // Wrapped as a live Row (the fixed producer): surfaces under its cell name.
304        let row = build_row_from_scan(
305            key,
306            ScanRow::Row(vec![(Arc::from("data"), live.clone())]),
307            &[],
308            None,
309        )
310        .expect("a live Row must surface");
311        assert_eq!(row.values.get("data"), Some(&live));
312    }
