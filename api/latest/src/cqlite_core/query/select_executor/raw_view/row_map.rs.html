1//! Map a per-generation [`CompactionRow`] into the raw view's [`QueryRow`]
2//! shape (design.md D7's column contract, issue #4222). No reconciliation:
3//! one `CompactionRow` becomes one `QueryRow` (two for a `RangeMarker` — its
4//! start and end bound), each carrying THIS generation's source identity
5//! untouched — the deliberate absence of `KWayMerger` (design.md D5).
6

18/// Source-SSTable identity attached to every row a generation contributes
19/// (design.md D7's "source columns, present on every row").
20#[derive(Clone)]
21pub(in crate::query::select_executor) struct RawViewSource {
22    pub sstable: String,
23    pub generation: u64,
24    pub format: &'static str,
25    /// Byte offset of the partition in `Data.db`, when cheaply resolvable.
26    /// The point-key producer resolves this from the SAME index lookup it
27    /// uses to seek the partition (issue #4222); the full-scan producer
28    /// currently leaves this `None` — exposing it there needs the streaming
29    /// compaction walk to surface a per-partition offset, which nothing in
30    /// this codebase does today (a documented scope limitation, not a
31    /// heuristic substitute).
32    pub position: Option<i64>,
33}
34
35impl RawViewSource {
36    /// Build a source identity from an open reader (issue #4222). Shared by
37    /// the point-key and full-scan producers so `sstable`/`generation`/
38    /// `format` are derived identically on both paths.
39    pub(in crate::query::select_executor) fn from_reader(
40        reader: &SSTableReader,
41        position: Option<i64>,
42    ) -> Self {
43        let sstable = reader
44            .file_path()
45            .file_name()
46            .and_then(|n| n.to_str())
47            .unwrap_or_default()
48            .to_string();
49        Self {
50            sstable,
51            generation: reader.generation,
52            format: reader.sstable_format_label(),
53            position,
54        }
55    }
56}

58/// Narrow an `i64` to `i32` by SATURATION, never silent two's-complement
59/// wraparound (roborev overflow-class self-check) — every field this touches
60/// (TTL seconds, generation number, local-deletion-time) is authoritative
61/// on-disk/reader data that is always small in practice, so saturation is a
62/// safety net, not a correctness path any committed fixture exercises.
63fn saturating_i32(v: i64) -> i32 {
64    i32::try_from(v).unwrap_or(if v > 0 { i32::MAX } else { i32::MIN })
65}
66
67fn insert_source_values(values: &mut HashMap<String, Value>, source: &RawViewSource) {
68    values.insert("sstable".to_string(), Value::text(source.sstable.clone()));
69    // `bigint`, never a saturating `i32` narrow (roborev finding, issue
70    // #4222): `SSTableReader::generation` is a `u64`, and this view's whole
71    // purpose is authoritative per-generation source identity — collapsing
72    // two distinct generations to the same saturated value would be exactly
73    // the fabricated-fact class the rest of this module fails closed on.
74    // `i64::MAX` remains a saturating fallback ONLY for a generation number
75    // past `i64::MAX`, a value no real corpus can produce.
76    values.insert(
77        "generation".to_string(),
78        Value::BigInt(i64::try_from(source.generation).unwrap_or(i64::MAX)),
79    );
80    values.insert("format".to_string(), Value::text(source.format));
81    values.insert(
82        "position".to_string(),
83        source.position.map(Value::BigInt).unwrap_or(Value::Null),
84    );
85}
86
87fn insert_pk_values(values: &mut HashMap<String, Value>, pk_values: &[(String, Value)]) {
88    for (name, value) in pk_values {
89        values.insert(name.clone(), value.clone());
90    }
91}

93/// Map one [`CompactionRow`] into its raw-view [`QueryRow`]s — one for
94/// `Live`/`Tombstone`/`PartitionDelete`, two (start + end bound) for
95/// `RangeMarker`.
96///
97/// KNOWN GAP, declared rather than left unexamined (roborev finding, issue
98/// #4222 — round 8; spec.md's "A static row is distinguishable from a
99/// clustering row" scenario, DEFERRED): `CompactionRow`/`CompactionRowData`
100/// do not carry whether the decoder classified this row as a Cassandra
101/// STATIC row (`compaction_row_build.rs`'s `is_static`, issue #3809), so a
102/// static row reaches here as an ordinary `Live`/`Tombstone` with EMPTY
103/// clustering and renders as the same `row_kind = 'row'` a genuine
104/// clustering row with missing clustering components would — a silent
105/// fidelity gap. Threading `is_static` onto `CompactionRow` (a shared
106/// compaction-read data model, not just this view's) is future work.
107pub(in crate::query::select_executor) fn map_compaction_row(
108    row: CompactionRow,
109    schema: &TableSchema,
110    source: &RawViewSource,
111) -> Result<Vec<QueryRow>> {
112    let clustering_names: HashSet<&str> = schema
113        .clustering_keys
114        .iter()
115        .map(|k| k.name.as_str())
116        .collect();
117    let pk_values = decode_partition_key_columns(row.key.as_bytes(), schema)?;
118
119    let rows = match row.row_data {
120        CompactionRowData::Live {
121            simple,
122            complex,
123            row_deletion,
124            row_liveness,
125        } => {
126            let mut values: HashMap<String, Value> = HashMap::new();
127            insert_pk_values(&mut values, &pk_values);
128            for cell in &simple {
129                insert_simple_cell(&mut values, cell, &clustering_names);
130            }
131            for col in &complex {
132                insert_complex_column(&mut values, col);
133            }
134            values.insert("row_kind".to_string(), Value::text("row"));
135
136            if row_liveness.has_marker {
137                if let Some(ts) = row_liveness.marker_timestamp {
138                    values.insert("row_timestamp".to_string(), Value::BigInt(ts));
139                }
140                // The AUTHORITATIVE on-disk row TTL, reported VERBATIM
141                // (roborev finding, issue #4222 — round 11). This used to be
142                // DERIVED as `expires_at_seconds - marker_timestamp/1_000_000`,
143                // which is not a fact: Cassandra computes `localExpirationTime`
144                // from the COORDINATOR's `nowInSec` at write time, never from
145                // the mutation's write timestamp, so the subtraction diverges
146                // by exactly the gap between "now" and an explicit
147                // `USING TIMESTAMP`. `INSERT ... USING TIMESTAMP
148                // 4102444800000000 AND TTL 60` (year 2100) left the expiry at
149                // `now + 60`, so the subtraction underflowed to ≈ `-2.4e9` and
150                // `saturating_i32` published `row_ttl = i32::MIN` — a
151                // fabricated fact in the very branch whose comment claimed
152                // "never a guess (issue #28)". `RowHeader::ttl` was decoded all
153                // along and merely dropped when `RowLiveness` was built; it is
154                // now threaded through as `ttl_seconds`, matching how the
155                // per-cell `<col>_ttl` path has always read `cell.ttl`. A
156                // marker with no on-disk TTL reports NO `row_ttl` at all.
157                if let Some(ttl) = row_liveness.ttl_seconds {
158                    values.insert("row_ttl".to_string(), Value::Integer(ttl));
159                }
160                if let Some(expires_at) = row_liveness.expires_at_seconds {
161                    // ITS OWN COLUMN, never `row_local_deletion_time`
162                    // (roborev finding, issue #4222 — round 11). A row
163                    // deletion and a TTL'd liveness marker COEXIST on one
164                    // physical row (`CompactionRowData::Live::row_deletion`,
165                    // issue #932 — e.g. a `DELETE … USING TIMESTAMP 100`
166                    // flushed together with a later
167                    // `INSERT … USING TIMESTAMP 200 AND TTL 60`), and this
168                    // value used to be written into `row_local_deletion_time`
169                    // only to be UNCONDITIONALLY OVERWRITTEN by the
170                    // tombstone's LDT below — silently discarding a physical
171                    // fact in a view whose contract is to lose none. The two
172                    // clocks are different facts and now have different
173                    // columns: `row_local_deletion_time` is the row
174                    // TOMBSTONE's GC clock and nothing else.
175                    //
176                    // `expires_at` is an HONEST `i64` (write-time + TTL
177                    // seconds), never a wrapped on-disk LDT bit pattern — no
178                    // cast needed for this `bigint` column (roborev finding,
179                    // issue #4222 — round 8).
180                    values.insert(
181                        "row_liveness_expires_at".to_string(),
182                        Value::BigInt(expires_at),
183                    );
184                }
185            }
186            if let Some((deletion_time, local_deletion_time)) = row_deletion {
187                values.insert("row_tombstone".to_string(), Value::text("row"));
188                values.insert(
189                    "row_local_deletion_time".to_string(),
190                    // `i64::from(x as u32)`, never a bare narrow-to-i32
191                    // (roborev finding, issue #4222 — round 8): this
192                    // `local_deletion_time` IS the wrapped on-disk bit
193                    // pattern (module doc, `compaction_row.rs`'s header),
194                    // matching `partition_deletion_time`/`range_deletion_time`'s
195                    // round-7 fix.
196                    Value::BigInt(i64::from(local_deletion_time as u32)),
197                );
198                values.insert(
199                    "row_deletion_timestamp".to_string(),
200                    Value::BigInt(deletion_time),
201                );
202            }
203
204            insert_source_values(&mut values, source);
205            vec![QueryRow::with_values(row.key.clone(), values)]
206        }
207        CompactionRowData::Tombstone {
208            deletion_time,
209            local_deletion_time,
210            clustering,
211        } => {
212            let mut values: HashMap<String, Value> = HashMap::new();
213            insert_pk_values(&mut values, &pk_values);
214            for (name, value) in &clustering {
215                values.insert(name.clone(), value.clone());
216            }
217            values.insert("row_kind".to_string(), Value::text("row"));
218            values.insert("row_tombstone".to_string(), Value::text("row"));
219            values.insert(
220                "row_deletion_timestamp".to_string(),
221                Value::BigInt(deletion_time),
222            );
223            values.insert(
224                "row_local_deletion_time".to_string(),
225                // See the identical fix + rationale in the `Live` arm above
226                // (roborev finding, issue #4222 — round 8).
227                Value::BigInt(i64::from(local_deletion_time as u32)),
228            );
229            insert_source_values(&mut values, source);
230            vec![QueryRow::with_values(row.key.clone(), values)]
231        }
232        CompactionRowData::PartitionDelete {
233            deletion_time,
234            local_deletion_time,
235        } => {
236            let mut values: HashMap<String, Value> = HashMap::new();
237            insert_pk_values(&mut values, &pk_values);
238            values.insert("row_kind".to_string(), Value::text("partition_tombstone"));
239            values.insert(
240                "partition_deletion_timestamp".to_string(),
241                Value::BigInt(deletion_time),
242            );
243            values.insert(
244                "partition_deletion_time".to_string(),
245                // `i64::from(x as u32)`, NEVER a bare `as i64` (roborev
246                // finding, issue #4222 — round 7): `local_deletion_time` is
247                // an `i32` that deliberately carries a far-future LDT via
248                // wrapping `as u32 as i32` (`compaction_row.rs`), matching
249                // the SAME widening convention `write_engine/merge/{mod,
250                // reconcile,streaming}.rs` already use. A bare `as i64`
251                // SIGN-EXTENDS that wrapped bit pattern instead, rendering a
252                // real far-future LDT as a fabricated large-negative epoch
253                // second in this `bigint` column — precisely the
254                // authoritative-facts violation this view exists to avoid.
255                Value::BigInt(i64::from(local_deletion_time as u32)),
256            );
257            insert_source_values(&mut values, source);
258            vec![QueryRow::with_values(row.key.clone(), values)]
259        }
260        CompactionRowData::RangeMarker {
261            start,
262            end,
263            deletion_time,
264            local_deletion_time,
265        } => {
266            vec![
267                range_bound_row(
268                    &row.key,
269                    &pk_values,
270                    &start,
271                    "range_tombstone_start",
272                    deletion_time,
273                    local_deletion_time,
274                    source,
275                ),
276                range_bound_row(
277                    &row.key,
278                    &pk_values,
279                    &end,
280                    "range_tombstone_end",
281                    deletion_time,
282                    local_deletion_time,
283                    source,
284                ),
285            ]
286        }
287    };
288    Ok(rows)
289}

291#[allow(clippy::too_many_arguments)]
292fn range_bound_row(
293    key: &RowKey,
294    pk_values: &[(String, Value)],
295    bound: &CompactionBound,
296    row_kind: &'static str,
297    deletion_time: i64,
298    local_deletion_time: i32,
299    source: &RawViewSource,
300) -> QueryRow {
301    let mut values: HashMap<String, Value> = HashMap::new();
302    insert_pk_values(&mut values, pk_values);
303    let bound_inclusive = match bound {
304        CompactionBound::Inclusive(components) => {
305            for (name, value) in components {
306                values.insert(name.clone(), value.clone());
307            }
308            Some(true)
309        }
310        CompactionBound::Exclusive(components) => {
311            for (name, value) in components {
312                values.insert(name.clone(), value.clone());
313            }
314            Some(false)
315        }
316        // Open bounds (before/after every clustering key) carry no clustering
317        // component and no literal inclusivity to report.
318        CompactionBound::Bottom | CompactionBound::Top => None,
319    };
320    values.insert("row_kind".to_string(), Value::text(row_kind));
321    if let Some(inclusive) = bound_inclusive {
322        values.insert("bound_inclusive".to_string(), Value::Boolean(inclusive));
323    }
324    values.insert(
325        "range_deletion_timestamp".to_string(),
326        Value::BigInt(deletion_time),
327    );
328    values.insert(
329        "range_deletion_time".to_string(),
330        // See the identical fix + rationale on `partition_deletion_time`
331        // above (roborev finding, issue #4222 — round 7): `i64::from(x as
332        // u32)`, never a bare `as i64` sign-extension of the wrapped bit
333        // pattern.
334        Value::BigInt(i64::from(local_deletion_time as u32)),
335    );
336    insert_source_values(&mut values, source);
337    QueryRow::with_values(key.clone(), values)
338}

340fn insert_simple_cell(
341    values: &mut HashMap<String, Value>,
342    cell: &SimpleCell,
343    clustering_names: &HashSet<&str>,
344) {
345    let live_value = match &cell.value {
346        Value::Tombstone(_) => Value::Null,
347        other => other.clone(),
348    };
349
350    if clustering_names.contains(cell.column.as_str()) {
351        // Clustering columns are surfaced as plain data columns, not per-cell
352        // metadata quads (design.md D7: the quad is for non-key columns only).
353        values.insert(cell.column.clone(), live_value);
354        return;
355    }
356
357    // A tombstoned cell's authoritative local-deletion-time lives on
358    // `TombstoneInfo` (`Value::Tombstone(info).local_deletion_time`), NOT on
359    // `SimpleCell::local_deletion_time` (which the decoder leaves `None` for
360    // a tombstone — that field is populated for a LIVE expiring cell). Prefer
361    // the tombstone's own field when the cell is a tombstone; fall back to
362    // the cell's own field otherwise (the live-TTL case).
363    //
364    // `TombstoneInfo::local_deletion_time` is an HONEST `i64` (never a
365    // wrapped bit pattern — `types.rs`'s own doc), so no cast is needed
366    // there. `SimpleCell::local_deletion_time: Option<i32>` DOES follow the
367    // wrapped `as u32 as i32` convention for a far-future LDT
368    // (`compaction_row.rs`'s module-header invariant) — widened via
369    // `i64::from(x as u32)`, never a bare narrow-to-i32 (roborev finding,
370    // issue #4222 — round 8, matching round 7's `partition_deletion_time`
371    // fix for the same defect class).
372    let (tombstone_kind, tombstone_ldt): (Option<&str>, Option<i64>) = match &cell.value {
373        Value::Tombstone(info) => (
374            Some(match info.tombstone_type {
375                TombstoneType::TtlExpiration => "expired",
376                _ => "cell",
377            }),
378            Some(info.local_deletion_time),
379        ),
380        _ => (None, None),
381    };
382    values.insert(cell.column.clone(), live_value);
383    values.insert(
384        format!("{}_timestamp", cell.column),
385        Value::BigInt(cell.timestamp),
386    );
387    if let Some(ttl) = cell.ttl {
388        values.insert(
389            format!("{}_ttl", cell.column),
390            Value::Integer(saturating_i32(ttl as i64)),
391        );
392    }
393    let ldt = tombstone_ldt.or_else(|| cell.local_deletion_time.map(|x| i64::from(x as u32)));
394    if let Some(ldt) = ldt {
395        values.insert(
396            format!("{}_local_deletion_time", cell.column),
397            Value::BigInt(ldt),
398        );
399    }
400    if let Some(kind) = tombstone_kind {
401        values.insert(format!("{}_tombstone", cell.column), Value::text(kind));
402    }
403}

405fn insert_complex_column(values: &mut HashMap<String, Value>, col: &ComplexColumn) {
406    values.insert(col.column.clone(), col.collapsed_value.clone());
407    values.insert(
408        format!("{}_complex_deletion", col.column),
409        Value::Boolean(col.complex_deletion.is_some()),
410    );
411    if let Some((deletion_time, local_deletion_time)) = col.complex_deletion {
412        values.insert(
413            format!("{}_complex_deletion_timestamp", col.column),
414            Value::BigInt(deletion_time),
415        );
416        values.insert(
417            format!("{}_complex_deletion_time", col.column),
418            // `ComplexColumn::complex_deletion`'s `i32` element follows the
419            // SAME wrapped `as u32 as i32` convention as every other LDT
420            // field in this module (`compaction_row.rs`'s header) — widen
421            // via `i64::from(x as u32)`, never a bare narrow-to-i32
422            // (roborev finding, issue #4222 — round 8).
423            Value::BigInt(i64::from(local_deletion_time as u32)),
424        );
425    }
426}
