1//! The raw SSTable view — `<keyspace>.<table>_raw_sstable_data` (issue #4222,
2//! `openspec/changes/raw-sstable-view/design.md`).
3//!
4//! Exposes every physical row CQLite has already decoded for a table,
5//! UNRECONCILED, one row per physical row per SSTable generation, with
6//! per-cell/row/partition metadata and source SSTable identity — the raw
7//! facts a reconciled `SELECT` on the base table discards.
8//!
9//! ## Interception point (design.md D6)
10//!
11//! Since the query engine has no catalog/`TableProvider` registry at all
12//! (table resolution runs a `TableId` string straight to on-disk SSTables),
13//! recognizing this new virtual table means intercepting resolution BEFORE it
14//! reaches disk. [`SelectExecutor::execute`] (`execute.rs`) and
15//! [`SelectExecutor::execute_streaming`] (`mod.rs`) both call
16//! [`SelectExecutor::raw_view_base_name`] immediately after extracting the
17//! FROM-clause table id, BEFORE the normal `resolve_table_schema` /
18//! execution-step pipeline runs, and route a recognized suffix entirely to
19//! [`SelectExecutor::execute_raw_sstable_view`] — never through
20//! `StorageEngine::scan`/`scan_partition`/`SSTableManager::scan_with_meter`,
21//! which RECONCILE across generations (design.md D5's "why not `KWayMerger`").
22//!
23//! ## Fail-closed (design.md D8)
24//!
25//! Unlike the base SELECT path (which silently returns zero rows for an
26//! unknown table and silently falls back to row-derived columns for a missing
27//! schema), an unresolvable BASE table/schema here raises a typed
28//! `Error::Table`/`Error::Schema` — this is new surface, built fail-closed
29//! from the start, and the base table's own permissive behavior is untouched.

54impl super::SelectExecutor {
55    /// Decide whether `table_id` should be intercepted as a raw-view
56    /// reference, returning the (owned) base table name when so.
57    ///
58    /// A LITERAL table actually named with the `_raw_sstable_data` suffix
59    /// takes precedence (roborev finding, issue #4222): unconditional
60    /// stripping made such a real table permanently unreachable through
61    /// this executor. Only when the FULL, unstripped name resolves to NO
62    /// registered schema does the suffix count as the D1 naming convention.
63    /// Checked on the bare table name only (no keyspace segment), matching
64    /// how the suffix is a convention over the flat `keyspace.table`
65    /// namespace.
66    pub(super) async fn raw_view_base_name(&self, table_id: &TableId) -> Option<String> {
67        let (_, bare_table_name) = parse_table_id(table_id);
68        let base = strip_raw_view_suffix(&bare_table_name)?;
69        if self.resolve_table_schema(table_id).await.is_some() {
70            // A real table is literally registered under the suffixed name —
71            // let it resolve normally, never shadow it.
72            return None;
73        }
74        Some(base.to_string())
75    }

77    /// Entry point for a `_raw_sstable_data` table reference (design.md D6).
78    /// `table_id` carries the RAW-VIEW name (suffix included, as the FROM
79    /// clause named it — needed to resolve the reader snapshot and to keep
80    /// the keyspace); `base_name` is the already-stripped base table name.
81    pub(super) async fn execute_raw_sstable_view(
82        &self,
83        plan: &OptimizedQueryPlan,
84        table_id: &TableId,
85        base_name: &str,
86    ) -> Result<QueryResult> {
87        // Fail closed on every reshaping clause this slice does not implement
88        // (roborev finding, issue #4222): ORDER BY needs a real sort over the
89        // producers' unordered output, DISTINCT/aggregates need dedup/fold
90        // machinery — none of which this view builds (design.md D4's
91        // JOIN-gap precedent: a general query-engine capability, not a
92        // raw-view concern). Checked BEFORE any producer work runs, so a
93        // rejected query never pays for a scan it will discard.
94        if plan.aggregation_plan.is_some() {
95            return Err(Error::unsupported_query(
96                "aggregate functions are not supported over a _raw_sstable_data view",
97            ));
98        }
99        if matches!(plan.statement.select_clause, SelectClause::Distinct(_)) {
100            return Err(Error::unsupported_query(
101                "SELECT DISTINCT is not supported over a _raw_sstable_data view",
102            ));
103        }
104        if plan.statement.order_by.is_some() {
105            return Err(Error::unsupported_query(
106                "ORDER BY is not supported over a _raw_sstable_data view",
107            ));
108        }
109        // PER PARTITION LIMIT (roborev finding, issue #4222): silently
110        // ignoring it (rather than failing closed, like the three guards
111        // above) would return every row of every partition instead of
112        // capping each one — the same silent-wrong-answer class.
113        if plan.statement.per_partition_limit.is_some() {
114            return Err(Error::unsupported_query(
115                "PER PARTITION LIMIT is not supported over a _raw_sstable_data view",
116            ));
117        }
118        // A WHERE clause with ANY comparison that did not fully lower to a
119        // pushed-down predicate (roborev finding, issue #4222 — High,
120        // round 2 of this same class): OR/NOT are ONE way that happens
121        // (`collect_sstable_predicates` skips those branches entirely), but
122        // `!=`/`NotIn`/`LIKE`/`IS NULL`/`IS NOT NULL`/a non-literal RHS
123        // etc. ALSO silently drop even inside a perfectly pushable `AND`
124        // position — `column_comparison_to_predicate` returns `None` for
125        // every one of those shapes. The base pipeline compensates with a
126        // residual `Filter` execution step over the ORIGINAL where-
127        // expression tree — a step this view's early return never reaches.
128        // Comparing the PUSHABLE leaf count against the predicate count
129        // (rather than just checking for OR/NOT, or checking
130        // `sstable_predicates.is_empty()`) catches BOTH classes: `pk = 1 AND
131        // val != 'x'` still yields the non-empty list `[pk = 1]`, which an
132        // emptiness check would miss. Failing closed here (rather than
133        // re-implementing the general WHERE-expression evaluator with this
134        // view's row-kind exemption semantics) is the same documented scope
135        // boundary as ORDER BY/DISTINCT/aggregates above.
136        if let Some(where_clause) = &plan.statement.where_clause {
137            let fully_pushed = count_pushable_comparison_leaves(where_clause)
138                .map(|n| n == plan.sstable_predicates.len())
139                .unwrap_or(false);
140            if !fully_pushed {
141                return Err(Error::unsupported_query(
142                    "a WHERE clause containing OR/NOT, a comparison shape (!=, LIKE, IS \
143                     [NOT] NULL, NOT IN, a non-literal comparison, etc.) that cannot be pushed \
144                     down to an SSTable-level predicate, or an IN (...) list with a \
145                     non-literal element (which would lower to a SILENTLY NARROWED subset of \
146                     the list — roborev finding, issue #4222), is not supported over a \
147                     _raw_sstable_data view — every restriction must lower to a pushable \
148                     column/token comparison IN FULL, or it would be silently dropped",
149                ));
150            }
151        }
152        // A predicate on 'position' is rejected OUTRIGHT (roborev finding,
153        // issue #4222 — round 6), never allowed to reach either producer:
154        // unlike every other source-identity column, it is populated
155        // differently by the two internal access paths — real on the
156        // point-key path (an index lookup), always `Null` on the full-scan
157        // path (never consulted per row) — so filtering on it would make
158        // the SAME query text silently return different rows depending on
159        // an access-path choice (`classify_partition_lookup`) the caller
160        // does not control. See `always_applicable_column_names`'s doc for
161        // the fuller rationale.
162        if plan
163            .sstable_predicates
164            .iter()
165            .any(|p| p.column == "position")
166        {
167            return Err(Error::unsupported_query(
168                "a predicate on 'position' is not supported over a _raw_sstable_data view — \
169                 this column is populated only on the point-key access path (never on a full \
170                 scan), so filtering on it would silently return different rows depending on \
171                 an internal access-path choice the query has no control over",
172            ));
173        }
174
175        let (keyspace, _) = parse_table_id(table_id);
176        let base_schema = self
177            .resolve_base_schema_for_raw_view(keyspace.as_deref(), base_name)
178            .await?;
179
180        // The reader map (`SSTableManager::table_readers`) is keyed by the
181        // BASE table's own name — the raw-view suffix is a query-engine-side
182        // naming convention only (design.md D1) and is never itself a
183        // registered table identity. Every reader-snapshot lookup below MUST
184        // use this, never `table_id` (which still carries the suffix).
185        let base_table_id = TableId::new(format!(
186            "{}{}",
187            keyspace
188                .as_deref()
189                .map(|k| format!("{k}."))
190                .unwrap_or_default(),
191            base_name
192        ));
193
194        let (columns, metadata_names) = raw_view_columns(&base_schema)?;
195        let readers = self
196            .resolve_raw_view_readers(&base_table_id, keyspace.is_some())
197            .await?;
198
199        // Validate the SELECT-list shape/names BEFORE either producer runs
200        // (roborev finding, issue #4222 — round 9, correcting round 5's own
201        // gap): this check previously lived in the POST-producer
202        // projection-trimming match below, so a projection naming an
203        // unknown column paid a full corpus walk before erroring — and on
204        // a corpus exceeding `max_result_bytes`/`max_result_rows` the
205        // producer's OWN budget guard tripped first, masking the intended
206        // `Error::Schema` behind `Error::ResultTooLarge`. Mirrors the
207        // WHERE-predicate validation immediately below for the same
208        // reason. `SelectClause::All` needs no validation; a non-bare-
209        // column `Columns(exprs)` shape and `Distinct` are refused here,
210        // matching the (now trimming-only) match below.
211        match &plan.statement.select_clause {
212            SelectClause::All => {}
213            SelectClause::Columns(exprs)
214                if exprs
215                    .iter()
216                    .all(|e| matches!(e, SelectExpression::Column(_))) =>
217            {
218                for expr in exprs {
219                    let SelectExpression::Column(c) = expr else {
220                        unreachable!("filtered to bare Column above")
221                    };
222                    if !columns.iter().any(|col| col.name == c.column) {
223                        return Err(Error::Schema(format!(
224                            "raw SSTable view: SELECT names unknown column '{}' — not part \
225                             of the raw view's column contract",
226                            c.column
227                        )));
228                    }
229                }
230            }
231            SelectClause::Columns(_) | SelectClause::Distinct(_) => {
232                return Err(Error::unsupported_query(
233                    "a SELECT expression other than a plain column reference (WRITETIME/TTL, \
234                     an alias, an arithmetic expression, or a collection-element access) is \
235                     not supported over a _raw_sstable_data view — use 'SELECT *' or a plain \
236                     column list",
237                ));
238            }
239        }
240
241        // A predicate naming a column that is NOT part of the raw view's
242        // column contract at all must fail closed (roborev finding, issue
243        // #4222 — round 7), never be silently misapplied: it would land in
244        // `data_predicates` below, and the per-column-presence exemption
245        // that predicate group applies to a synthetic row (see
246        // `row_passes_predicates`) treats an unknown column exactly like a
247        // genuinely-absent one — exempting EVERY `partition_tombstone`/
248        // open-bound `range_tombstone_*` row from it while `evaluate_leaf`'s
249        // `Unknown` still rejects every plain `row_kind = 'row'` row. So
250        // `WHERE typo_col = 1` would silently return only the corpus's
251        // tombstone rows instead of erroring — asymmetric with the SELECT
252        // list, which already fails closed on an unknown column below, and
253        // with this view's whole D8 posture. Checked BEFORE the split, so
254        // neither producer runs on an unvalidated predicate set. Token
255        // predicates are exempt from this check (`p.column` is a
256        // human-readable `"token(...)"` label, never a real column name).
257        if let Some(bad) = plan
258            .sstable_predicates
259            .iter()
260            .find(|p| !p.is_token() && !columns.iter().any(|c| c.name == p.column))
261        {
262            return Err(Error::Schema(format!(
263                "raw SSTable view: WHERE names unknown column '{}' — not part of the raw \
264                 view's column contract",
265                bad.column
266            )));
267        }
268
269        // Split the predicate set ONCE (roborev finding, issue #4222): a
270        // partition-key predicate (or `token(...)`), source-identity/
271        // `row_kind`/partition-and-range-deletion predicate applies to
272        // EVERY row, synthetic or not; a clustering/regular-column
273        // predicate exempts a SYNTHETIC row ONLY WHEN that row genuinely
274        // lacks the column (round 6's fix — see `row_passes_predicates`),
275        // never by blanket row-kind. Threaded into BOTH producers so the
276        // filtering happens where LIMIT can stop the walk early, not after
277        // the whole corpus is already materialized.
278        let (always_predicates, data_predicates) =
279            split_predicates_by_key_role(&plan.sstable_predicates, &base_schema);
280
281        // LIMIT/OFFSET (roborev finding, issue #4222): the raw view returns
282        // straight from `execute_raw_sstable_view`, never reaching the
283        // execution-step pipeline's `Limit` step. `stop_after` tells the
284        // FULL-SCAN producer to stop once this many ACCEPTED (post-predicate)
285        // rows are collected — mirroring issue #1577's LIMIT-pushdown intent —
286        // so `SELECT * FROM big_raw_sstable_data LIMIT 1` does not decode the
287        // whole corpus, and does not spuriously trip `max_result_rows` on a
288        // table that exceeds it (issue #1578's explicit-LIMIT exemption,
289        // applied at the SOURCE this time, not just the final check below).
290        let offset = plan.statement.offset.unwrap_or(0) as usize;
291        let limit = plan
292            .statement
293            .limit
294            .as_ref()
295            .map(|l| l.count as usize)
296            .unwrap_or(usize::MAX);
297        let stop_after = plan
298            .statement
299            .limit
300            .as_ref()
301            .map(|_| offset.saturating_add(limit));
302
303        // Reuse the SAME sstable-predicate extraction the optimizer already
304        // ran for this statement's WHERE clause (predicates are plain
305        // column=value facts, independent of which table string the SELECT
306        // named) to decide whether this is a partition-targeted point read
307        // or an unbounded full scan (spec: "pushed to the point-read path,
308        // never a scan" for a full `WHERE pk = ?`).
309        let outcome = classify_partition_lookup(&plan.sstable_predicates, Some(&base_schema));
310
311        let rows = match outcome {
312            PartitionLookupOutcome::Targeted(pk_bytes) => {
313                raw_view_point_rows(
314                    &readers,
315                    &base_schema,
316                    std::slice::from_ref(&pk_bytes),
317                    &always_predicates,
318                    &data_predicates,
319                    &metadata_names,
320                    self.max_result_bytes,
321                    self.max_result_rows,
322                    stop_after,
323                )
324                .await?
325            }
326            PartitionLookupOutcome::MultiTargeted(pk_keys) => {
327                raw_view_point_rows(
328                    &readers,
329                    &base_schema,
330                    &pk_keys,
331                    &always_predicates,
332                    &data_predicates,
333                    &metadata_names,
334                    self.max_result_bytes,
335                    self.max_result_rows,
336                    stop_after,
337                )
338                .await?
339            }
340            PartitionLookupOutcome::Fallback(_) => {
341                raw_view_full_scan_rows(
342                    &readers,
343                    &base_schema,
344                    &always_predicates,
345                    &data_predicates,
346                    &metadata_names,
347                    self.max_result_bytes,
348                    self.max_result_rows,
349                    stop_after,
350                )
351                .await?
352            }
353        };
354
355        // Plain-column projection TRIMMING ONLY (`SELECT a, b, ...`) —
356        // reusing the SAME `trim_projection` the base pipeline's `Project`
357        // step uses so the two never drift. The SHAPE/NAME validation for
358        // every branch here already ran ABOVE, before either producer, so
359        // the `ok_or_else` below is defense-in-depth (never actually
360        // reachable in practice — roborev finding, issue #4222 — round 9,
361        // moving that validation earlier so a bad projection cannot pay for
362        // a full corpus walk first). `SELECT *` is the only other shape
363        // handled here; DISTINCT/aggregates/a non-bare-column shape were
364        // already rejected above.
365        let (rows, columns) = match &plan.statement.select_clause {
366            SelectClause::All => (rows, columns),
367            SelectClause::Columns(exprs)
368                if exprs
369                    .iter()
370                    .all(|e| matches!(e, SelectExpression::Column(_))) =>
371            {
372                let selected: Vec<&str> = exprs
373                    .iter()
374                    .filter_map(|e| match e {
375                        SelectExpression::Column(c) => Some(c.column.as_str()),
376                        _ => None,
377                    })
378                    .collect();
379                let trimmed_rows = self.trim_projection(rows, exprs);
380                // Look up every selected name BEFORE enumerating (roborev
381                // finding, issue #4222): assigning `position` from the
382                // pre-filter index left a `filter_map`'d-out unknown column
383                // a GAP in the surviving positions (`SELECT pk, bogus, ck`
384                // produced positions `0, 2`), violating the dense/ordered
385                // invariant `metadata.columns` must hold. An unknown column
386                // now fails closed (D8) instead of being silently dropped
387                // (defense-in-depth only — the SAME check already ran
388                // above, before either producer).
389                let mut trimmed_columns = Vec::with_capacity(selected.len());
390                for name in &selected {
391                    let col = columns
392                        .iter()
393                        .find(|c| c.name == *name)
394                        .cloned()
395                        .ok_or_else(|| {
396                            Error::Schema(format!(
397                                "raw SSTable view: SELECT names unknown column '{name}' — \
398                                 not part of the raw view's column contract"
399                            ))
400                        })?;
401                    trimmed_columns.push(col);
402                }
403                for (idx, c) in trimmed_columns.iter_mut().enumerate() {
404                    c.position = idx;
405                }
406                (trimmed_rows, trimmed_columns)
407            }
408            SelectClause::Columns(_) | SelectClause::Distinct(_) => {
409                return Err(Error::unsupported_query(
410                    "a SELECT expression other than a plain column reference (WRITETIME/TTL, \
411                     an alias, an arithmetic expression, or a collection-element access) is \
412                     not supported over a _raw_sstable_data view — use 'SELECT *' or a plain \
413                     column list",
414                ));
415            }
416        };
417
418        // Apply the SAME LIMIT/OFFSET the producer was told to `stop_after`
419        // (mirroring the constant-query branch, `execute.rs`'s `SELECT 1`
420        // handling): the producer stopped once `offset + limit` ACCEPTED
421        // rows were collected (or ran out of corpus first), so this trims
422        // exactly `offset` off the front and caps at `limit` — never fewer
423        // than available, never more than requested.
424        let rows: Vec<_> = rows.into_iter().skip(offset).take(limit).collect();
425
426        // Same final budget check every other query path applies (issue
427        // #1582/D6) — belt-and-braces alongside BOTH producers' own
428        // incremental checks (issue #4222 roborev finding: the point-key
429        // path now bounds accumulation the same way the full-scan path
430        // does, via `PointCollector` in `point.rs`), never materializing
431        // past the budget before this runs. Applied to the POST-limit rows,
432        // so a `LIMIT 10` raw-view query is never penalized for the corpus
433        // it did not keep.
434        let effective_max_rows = if plan.statement.limit.is_some() {
435            usize::MAX
436        } else {
437            self.max_result_rows
438        };
439        enforce_materialized_rows(&rows, self.max_result_bytes, effective_max_rows)?;
440
441        let total_rows = rows.len() as u64;
442        Ok(QueryResult {
443            rows,
444            rows_affected: total_rows,
445            execution_time_ms: 0,
446            metadata: QueryMetadata {
447                columns,
448                total_rows: Some(total_rows),
449                plan_info: None,
450                performance: Default::default(),
451                warnings: vec![],
452                access_path: None,
453            },
454        })
455    }

457    /// Resolve the BASE table's schema for a raw-view request, FAILING CLOSED
458    /// (design.md D8) — never falling back to row-derived columns the way
459    /// [`SelectExecutor::resolve_table_schema`]'s best-effort `Option` does.
460    async fn resolve_base_schema_for_raw_view(
461        &self,
462        keyspace: Option<&str>,
463        base_name: &str,
464    ) -> Result<TableSchema> {
465        let keyspace_opt = keyspace.map(|s| s.to_string());
466        let qualified = format!(
467            "{}{}",
468            keyspace.map(|k| format!("{k}.")).unwrap_or_default(),
469            base_name
470        );
471        self._schema
472            .find_schema_by_table(&keyspace_opt, base_name)
473            .await
474            .map_err(|e| {
475                Error::Schema(format!(
476                    "raw SSTable view: could not resolve schema for base table \
477                     '{qualified}': {e}"
478                ))
479            })?
480            .ok_or_else(|| {
481                Error::Table(format!(
482                    "raw SSTable view: base table '{qualified}' does not exist or has \
483                     no resolvable schema (queried as '{qualified}{}')",
484                    columns::RAW_VIEW_SUFFIX,
485                ))
486            })
487    }

489    /// Resolve the per-generation reader snapshot for `base_table_id`,
490    /// refusing a QUALIFIED (keyspace-carrying) raw-view request that only
491    /// resolved via the reader map's bare-table-name FALLBACK (roborev
492    /// finding, issue #4222).
493    ///
494    /// `resolve_reader_list` falls back to a bare-name match when no exact
495    /// `keyspace.table` key exists; `manager_point_read.rs` threads that same
496    /// `fully_qualified_match` signal into `get_with_resolution_unmetered`
497    /// specifically so a qualified query never silently reads a DIFFERENT
498    /// keyspace's same-named table (#1321) — this view had no equivalent
499    /// guard, so `ks_b.t_raw_sstable_data` could read `ks_a.t`'s rows tagged
500    /// as `ks_b.t`'s. `was_qualified` is `false` for an unqualified request
501    /// (no keyspace to mismatch, matching `fully_qualified_match`'s own
502    /// contract), so this never refuses a legitimately bare table name.
503    async fn resolve_raw_view_readers(
504        &self,
505        base_table_id: &TableId,
506        was_qualified: bool,
507    ) -> Result<Vec<Arc<crate::storage::sstable::reader::SSTableReader>>> {
508        let (readers, fully_qualified_match) =
509            self.storage.raw_view_reader_snapshot(base_table_id).await;
510        cross_keyspace_guard(was_qualified, fully_qualified_match, readers.is_empty()).map_err(
511            |_| {
512                Error::Table(format!(
513                    "raw SSTable view: '{base_table_id}' resolved only via a bare-table-name \
514                     fallback, not an exact keyspace match — refusing to read a possibly \
515                     DIFFERENT keyspace's same-named table (issue #1321's guard, applied here \
516                     for issue #4222)"
517                ))
518            },
519        )?;
520        Ok(readers)
521    }

554    /// Roborev finding (issue #4222): a REAL table literally named with the
555    /// `_raw_sstable_data` suffix must resolve as an ordinary table, never be
556    /// shadowed by the naming convention. Registering a schema for the exact
557    /// literal name and asserting `raw_view_base_name` returns `None` for it
558    /// is the direct regression test for that precedence.
559    #[tokio::test]
560    async fn literal_raw_sstable_data_table_is_never_shadowed() {
561        let (executor, schema) = test_executor().await;
562        schema
563            .parse_and_register_cql_schema(
564                "CREATE TABLE ks.foo_raw_sstable_data (pk int PRIMARY KEY, v text);",
565            )
566            .await
567            .expect("registering the literal table's schema must succeed");
568
569        let literal_id = TableId::new("ks.foo_raw_sstable_data");
570        assert_eq!(
571            executor.raw_view_base_name(&literal_id).await,
572            None,
573            "a REAL table literally named '..._raw_sstable_data' must resolve normally, \
574             never be intercepted as the raw view's naming convention"
575        );
576
577        // Sanity: a table with NO literal registration under the suffixed
578        // name IS recognized as the raw-view convention.
579        schema
580            .parse_and_register_cql_schema("CREATE TABLE ks.bar (pk int PRIMARY KEY, v text);")
581            .await
582            .expect("registering bar's schema must succeed");
583        let convention_id = TableId::new("ks.bar_raw_sstable_data");
584        assert_eq!(
585            executor.raw_view_base_name(&convention_id).await,
586            Some("bar".to_string()),
587            "with no literal 'bar_raw_sstable_data' table registered, the suffix must be \
588             recognized as the D1 naming convention over 'bar'"
589        );
590    }
