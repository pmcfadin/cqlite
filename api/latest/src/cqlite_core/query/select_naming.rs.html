1//! Output/column name derivation for SELECT expressions.
2//!
3//! Extracted from the optimizer so result-metadata naming, the aggregation
4//! plan's row-value keys, and scan-projection column extraction all share ONE
5//! set of pure functions and cannot disagree (issue #1763). The invariant this
6//! module upholds: an aggregate's `metadata.columns[i].name` (via
7//! [`result_column_name`]) equals the row value key that `finalize_group` emits
8//! (via the aggregation plan's alias, derived from [`aggregate_output_name`]) —
9//! an explicit alias when present, else the derived expression text, never a
10//! synthetic `col_N`.
11
12use super::select_ast::*;
13use crate::schema::CqlType;

15/// The scan-projection column name for a SELECT expression, or `None` when the
16/// expression does not name a stored column (aggregates, literals, arithmetic).
17///
18/// Issue #1763: an aliased aggregate (`COUNT(*) AS total`) is NOT a stored
19/// column — it is excluded exactly as a bare aggregate is, so the alias never
20/// enters the scan as a phantom column; its output name comes from the
21/// aggregation plan, not a projected cell.
22pub(crate) fn extract_column_name(expr: &SelectExpression) -> Option<String> {
23    match expr {
24        SelectExpression::Column(col_ref) => Some(col_ref.column.clone()),
25        SelectExpression::Aliased(inner, _) if inner.is_aggregate() => None,
26        SelectExpression::Aliased(_, alias) => Some(alias.clone()),
27        _ => None,
28    }
29}

31/// For a plain-column SELECT projection (possibly aliased), return
32/// `(source_column, output_name)` where `source_column` is the stored column the
33/// scan reads and `output_name` is what appears in `metadata.columns`.
34///
35/// Issue #1763 (grouped dimensions): `finalize_group` keys a grouped dimension's
36/// row VALUE by its SELECT OUTPUT name so it matches the metadata name, while the
37/// group KEY is still read from the row by the raw stored column. This pairs the
38/// two: `Column(x)` → `(x, x)`, `Aliased(Column(x), "a")` → `(x, "a")`. The
39/// output side is exactly what [`extract_column_name`] returns, so a grouped
40/// dimension's row key can never diverge from its metadata name. Returns `None`
41/// for aggregates, literals, arithmetic, etc.
42pub(crate) fn projection_source_and_output(expr: &SelectExpression) -> Option<(String, String)> {
43    match expr {
44        SelectExpression::Column(col_ref) => Some((col_ref.column.clone(), col_ref.column.clone())),
45        SelectExpression::Aliased(inner, alias) => match inner.as_ref() {
46            SelectExpression::Column(col_ref) => Some((col_ref.column.clone(), alias.clone())),
47            _ => None,
48        },
49        _ => None,
50    }
51}

53/// Borrow the `AggregateFunction` from an aggregate expression, unwrapping a
54/// surrounding `Aliased` (`COUNT(*) AS total`). Returns `None` for non-aggregate
55/// expressions.
56pub(crate) fn unwrap_aggregate(expr: &SelectExpression) -> Option<&AggregateFunction> {
57    match expr {
58        SelectExpression::Aggregate(agg) => Some(agg),
59        SelectExpression::Aliased(inner, _) => unwrap_aggregate(inner),
60        _ => None,
61    }
62}

64/// Source (stored) column names referenced by an aggregate's ARGUMENT
65/// expressions, excluding the `*` wildcard.
66///
67/// Issue #1952: the SSTable scan projection MUST include these so grouped,
68/// non-star aggregates read their input cells. `build_row_from_scan` filters
69/// decoded cells by the scan projection; without the argument column a grouped
70/// query that also projects a group dimension (`SELECT category, SUM(value) ...
71/// GROUP BY category`) scans ONLY the dimension and silently aggregates from
72/// missing inputs (SUM/AVG → 0/null, COUNT(col) → 0, MIN/MAX → null). Returns an
73/// empty vec for non-aggregate expressions and for `COUNT(*)` (no argument
74/// column). The SOURCE column is read directly (never an alias), matching the
75/// column the aggregation plan accumulates via [`aggregate_column_and_alias`].
76pub(crate) fn aggregate_arg_source_columns(expr: &SelectExpression) -> Vec<String> {
77    let Some(agg) = unwrap_aggregate(expr) else {
78        return Vec::new();
79    };
80    agg.args
81        .iter()
82        .filter_map(|arg| match arg {
83            SelectExpression::Column(col_ref) if col_ref.column != "*" => {
84                Some(col_ref.column.clone())
85            }
86            _ => None,
87        })
88        .collect()
89}

91/// The output column name for a (possibly aliased) aggregate SELECT expression,
92/// or `None` if `expr` is not an aggregate.
93///
94/// Issue #1763: the SINGLE source of truth for aggregate output names, shared by
95/// the aggregation plan (which keys row values via `finalize_group`) and result-
96/// metadata construction ([`result_column_name`]). Deriving both from this one
97/// function guarantees `metadata.columns[i].name` equals the emitted row value
98/// key for aggregates — an explicit alias when present, else the derived
99/// expression text (never a synthetic `col_N`).
100pub(crate) fn aggregate_output_name(expr: &SelectExpression) -> Option<String> {
101    match expr {
102        SelectExpression::Aggregate(agg) => Some(aggregate_column_and_alias(agg).1),
103        SelectExpression::Aliased(inner, alias) if inner.is_aggregate() => Some(alias.clone()),
104        _ => None,
105    }
106}

108/// The result (output) column name for a projected SELECT expression at position
109/// `index`, as it appears in `metadata.columns`.
110///
111/// Issue #1763: an aggregate is named by [`aggregate_output_name`] — the SAME
112/// source that keys the emitted row values (`finalize_group` via the aggregation
113/// plan alias) — so result metadata can never disagree with aggregate row value
114/// keys. Non-aggregate expressions use the column name, an explicit alias, or the
115/// synthetic `col_{index}` fallback. `WriteTimeTtl` is intentionally NOT handled
116/// here — the caller names those before reaching this function.
117pub(crate) fn result_column_name(expr: &SelectExpression, index: usize) -> String {
118    if let Some(name) = aggregate_output_name(expr) {
119        return name;
120    }
121    match expr {
122        SelectExpression::Column(col_ref) => col_ref.column.clone(),
123        SelectExpression::Aliased(_, alias) => alias.clone(),
124        _ => format!("col_{index}"),
125    }
126}

128/// Derive the CQL result type of an aggregate from the aggregate FUNCTION and
129/// its argument's schema type — never from a schema lookup of the aggregate's
130/// output name/alias (issue #1941). The name-lookup approach mis-typed
131/// `COUNT(*) AS value` as the `value` column's type and fell back to `Text` for
132/// any aggregate whose derived name matched no column.
133///
134/// Oracle — Cassandra's aggregate type rules, reconciled with the value type
135/// CQLite's executor actually emits in `finalize_group`
136/// (`select_executor::aggregation`):
137///   * `COUNT(*)` / `COUNT(col)` → `bigint` (emitted `Value::BigInt`)
138///   * `SUM(_)` / `AVG(_)`       → the RESULT type Cassandra returns for the
139///     argument (issue #2202, `AggregateFcts.java`): every INTEGRAL argument
140///     (`tinyint`/`smallint`/`int`/`bigint`/`counter`) returns THE SAME type
141///     back — Cassandra does NOT promote `tinyint`/`smallint` to `int`; every
142///     other numeric argument — `float`, `double`, `decimal`, `varint`, or an
143///     unknown/unresolved argument — → `double`. The executor accumulates
144///     integrally (`i64`, Cassandra wrapping, narrowed to the argument's own
145///     width at finalize) for the integral cases and in `f64` otherwise, and
146///     `finalize_group` emits the matching `Value` variant, so this metadata
147///     type never lies about the produced value (the #1941 invariant).
148///   * `MIN(col)` / `MAX(col)`   → the argument column's type (the value is
149///     cloned through unchanged)
150///
151/// `arg_type` is the pre-resolved CQL type of the aggregate's argument column
152/// (from the schema); it drives the SUM/AVG promotion above and the MIN/MAX
153/// passthrough. Returns `None` solely for a MIN/MAX whose argument type is
154/// unknown (no schema, or the argument is not a resolvable column), leaving the
155/// caller's existing untyped fallback — SUM/AVG always resolve to a concrete
156/// type (`double` when the argument is unknown).
157pub(crate) fn aggregate_result_cql_type(
158    function: &AggregateType,
159    arg_type: Option<CqlType>,
160) -> Option<CqlType> {
161    match function {
162        AggregateType::Count => Some(CqlType::BigInt),
163        // Issue #2202: preserve Cassandra's integral SUM/AVG result types instead
164        // of collapsing every numeric input to `double`.
165        AggregateType::Sum | AggregateType::Avg => Some(sum_avg_result_cql_type(arg_type)),
166        AggregateType::Min | AggregateType::Max => arg_type,
167    }
168}
169
170/// Cassandra's SUM/AVG result-type rule (issue #2202, verified against
171/// `AggregateFcts.java`): EVERY integral argument type returns THE SAME type —
172/// `tinyint`→`tinyint`, `smallint`→`smallint`, `int`→`int`, `bigint`/`counter`→
173/// `bigint`. Cassandra does NOT promote the narrow integral types to `int`
174/// (its own docs warn of overflow risk precisely because the result stays the
175/// input's narrow width). Every other numeric argument (`float`, `double`,
176/// `decimal`, `varint`) — plus an unknown argument — yields `double`,
177/// preserving CQLite's prior float behaviour with no regression. This is the
178/// SINGLE source of truth the executor's accumulator
179/// (`init_aggregate_accumulators`) also consults, so the emitted value variant
180/// and the result metadata type can never disagree.
181pub(crate) fn sum_avg_result_cql_type(arg_type: Option<CqlType>) -> CqlType {
182    match arg_type {
183        Some(t @ (CqlType::TinyInt | CqlType::SmallInt | CqlType::Int | CqlType::BigInt)) => t,
184        Some(CqlType::Counter) => CqlType::BigInt,
185        _ => CqlType::Double,
186    }
187}

189/// Resolve `(column, alias)` for an aggregate. `COUNT(*)` and any aggregate
190/// referencing `*` yields `("*", "Func(*)")`; a single named column yields
191/// `(name, "Func_name")`; anything else falls back to `("*", "Func")`.
192pub(crate) fn aggregate_column_and_alias(agg: &AggregateFunction) -> (String, String) {
193    let references_star = agg.args.is_empty()
194        || agg
195            .args
196            .iter()
197            .any(|arg| matches!(arg, SelectExpression::Column(c) if c.column == "*"));
198
199    if references_star {
200        return ("*".to_string(), format!("{:?}(*)", agg.function));
201    }
202
203    match agg.args.first().and_then(extract_column_name) {
204        Some(col_name) => {
205            let alias = format!("{:?}_{}", agg.function, col_name);
206            (col_name, alias)
207        }
208        None => ("*".to_string(), format!("{:?}", agg.function)),
209    }
210}

216    fn col(name: &str) -> SelectExpression {
217        SelectExpression::Column(ColumnRef {
218            table: None,
219            column: name.to_string(),

223    fn count_star() -> SelectExpression {
224        SelectExpression::Aggregate(AggregateFunction {
225            function: AggregateType::Count,
226            args: vec![],
227            distinct: false,
228        })
229    }

231    fn sum_of(c: &str) -> SelectExpression {
232        SelectExpression::Aggregate(AggregateFunction {
233            function: AggregateType::Sum,
234            args: vec![col(c)],
235            distinct: false,
236        })

239    /// The output name a bare aggregate exposes in metadata MUST equal the alias
240    /// the aggregation plan keys its row value by — the derived expression text,
241    /// never `col_N` (issue #1763).
242    #[test]
243    fn unaliased_aggregate_name_is_expression_text() {
244        assert_eq!(result_column_name(&count_star(), 0), "Count(*)");
245        assert_eq!(
246            aggregate_output_name(&count_star()).as_deref(),
247            Some("Count(*)")
248        );
249        // Metadata name == aggregation-plan alias (the single-source invariant).
250        assert_eq!(
251            result_column_name(&count_star(), 0),
252            aggregate_column_and_alias(unwrap_aggregate(&count_star()).unwrap()).1
253        );
254        assert_eq!(result_column_name(&sum_of("value"), 3), "Sum_value");
255    }
256
257    /// An explicit alias wins for BOTH metadata naming and the plan's row-value
258    /// key, so the two remain identical.
259    #[test]
260    fn aliased_aggregate_name_is_the_alias() {
261        let aliased = SelectExpression::Aliased(Box::new(count_star()), "total".to_string());
262        assert_eq!(result_column_name(&aliased, 0), "total");
263        assert_eq!(aggregate_output_name(&aliased).as_deref(), Some("total"));
264        // Unwraps to the inner aggregate for accumulation.
265        assert!(unwrap_aggregate(&aliased).is_some());
266    }

268    /// An aliased aggregate is never emitted as a scan-projection column (it is
269    /// not a stored cell), matching the bare-aggregate exclusion.
270    #[test]
271    fn aliased_aggregate_is_not_a_projection_column() {
272        let aliased = SelectExpression::Aliased(Box::new(count_star()), "total".to_string());
273        assert_eq!(extract_column_name(&aliased), None);
274        assert_eq!(extract_column_name(&count_star()), None);
275        // A plain column and a plain-aliased column still project normally.
276        assert_eq!(extract_column_name(&col("name")).as_deref(), Some("name"));
277        let aliased_col = SelectExpression::Aliased(Box::new(col("name")), "n".to_string());
278        assert_eq!(extract_column_name(&aliased_col).as_deref(), Some("n"));
279    }

281    /// Issue #1952: aggregate argument source columns are extracted for the scan
282    /// projection — the SOURCE column for a named aggregate, nothing for
283    /// `COUNT(*)` or a non-aggregate, so the scan never drops an aggregate input.
284    #[test]
285    fn aggregate_arg_source_columns_extracts_named_argument_only() {
286        assert_eq!(
287            aggregate_arg_source_columns(&sum_of("value")),
288            vec!["value"]
289        );
290        // COUNT(*) references no stored column.
291        assert!(aggregate_arg_source_columns(&count_star()).is_empty());
292        // Aliased aggregate still yields its inner argument's source column.
293        let aliased = SelectExpression::Aliased(Box::new(sum_of("value")), "s".to_string());
294        assert_eq!(aggregate_arg_source_columns(&aliased), vec!["value"]);
295        // Plain columns / non-aggregates contribute no aggregate-argument columns.
296        assert!(aggregate_arg_source_columns(&col("value")).is_empty());
297    }

299    /// Issue #1941/#2202: aggregate result type comes from the function (+
300    /// argument type), never a name lookup. COUNT → bigint; SUM/AVG return the
301    /// SAME integral type back for every integral argument (Cassandra does NOT
302    /// promote tinyint/smallint to int — verified against `AggregateFcts.java`)
303    /// and fall back to double for float/double/unknown; MIN/MAX preserve the
304    /// argument type and are the ONLY variants that return `None` when the
305    /// argument type is unknown.
306    #[test]
307    fn aggregate_result_cql_type_derives_from_function_and_argument() {
308        assert_eq!(
309            aggregate_result_cql_type(&AggregateType::Count, Some(CqlType::Int)),
310            Some(CqlType::BigInt),
311            "COUNT is bigint regardless of any argument type"
312        );
313        // Issue #2202: SUM/AVG preserve Cassandra's narrow integral result types —
314        // NO promotion to int for tinyint/smallint.
315        assert_eq!(
316            aggregate_result_cql_type(&AggregateType::Sum, Some(CqlType::Int)),
317            Some(CqlType::Int),
318            "SUM(int) is int"
319        );
320        assert_eq!(
321            aggregate_result_cql_type(&AggregateType::Sum, Some(CqlType::BigInt)),
322            Some(CqlType::BigInt),
323            "SUM(bigint) is bigint"
324        );
325        assert_eq!(
326            aggregate_result_cql_type(&AggregateType::Sum, Some(CqlType::SmallInt)),
327            Some(CqlType::SmallInt),
328            "SUM(smallint) stays smallint (Cassandra does not promote to int)"
329        );
330        assert_eq!(
331            aggregate_result_cql_type(&AggregateType::Sum, Some(CqlType::TinyInt)),
332            Some(CqlType::TinyInt),
333            "SUM(tinyint) stays tinyint (Cassandra does not promote to int)"
334        );
335        assert_eq!(
336            aggregate_result_cql_type(&AggregateType::Avg, Some(CqlType::Int)),
337            Some(CqlType::Int),
338            "AVG(int) is int"
339        );
340        assert_eq!(
341            aggregate_result_cql_type(&AggregateType::Avg, Some(CqlType::BigInt)),
342            Some(CqlType::BigInt),
343            "AVG(bigint) is bigint"
344        );
345        assert_eq!(
346            aggregate_result_cql_type(&AggregateType::Avg, Some(CqlType::SmallInt)),
347            Some(CqlType::SmallInt),
348            "AVG(smallint) stays smallint"
349        );
350        assert_eq!(
351            aggregate_result_cql_type(&AggregateType::Avg, Some(CqlType::TinyInt)),
352            Some(CqlType::TinyInt),
353            "AVG(tinyint) stays tinyint"
354        );
355        assert_eq!(
356            aggregate_result_cql_type(&AggregateType::Sum, Some(CqlType::Counter)),
357            Some(CqlType::BigInt),
358            "SUM(counter) is bigint"
359        );
360        // Float/double inputs still return double (no regression).
361        assert_eq!(
362            aggregate_result_cql_type(&AggregateType::Sum, Some(CqlType::Double)),
363            Some(CqlType::Double),
364            "SUM(double) stays double"
365        );
366        assert_eq!(
367            aggregate_result_cql_type(&AggregateType::Avg, Some(CqlType::Float)),
368            Some(CqlType::Double),
369            "AVG(float) is double (no regression)"
370        );
371        // Unknown SUM/AVG argument → double (never None, unlike MIN/MAX).
372        assert_eq!(
373            aggregate_result_cql_type(&AggregateType::Sum, None),
374            Some(CqlType::Double),
375            "SUM with unknown argument falls back to double"
376        );
377        assert_eq!(
378            aggregate_result_cql_type(&AggregateType::Min, Some(CqlType::Int)),
379            Some(CqlType::Int),
380            "MIN preserves the argument column's type"
381        );
382        assert_eq!(
383            aggregate_result_cql_type(&AggregateType::Max, Some(CqlType::Double)),
384            Some(CqlType::Double)
385        );
386        // Only MIN/MAX with an unknown argument type stay untyped.
387        assert_eq!(aggregate_result_cql_type(&AggregateType::Min, None), None);
388        assert_eq!(
389            aggregate_result_cql_type(&AggregateType::Count, None),
390            Some(CqlType::BigInt)
391        );
392    }
