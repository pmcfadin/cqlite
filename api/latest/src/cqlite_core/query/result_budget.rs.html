1//! Byte-bounded result-budget enforcement (issue #1582 / D6, narrow subset).
2//!
3//! The modern materializing SELECT path ([`crate::query::select_executor`])
4//! guards its FINAL result with a BYTE ceiling
5//! ([`crate::config::QueryConfig::max_result_bytes`]) plus a secondary row-count
6//! safety valve ([`crate::config::QueryConfig::max_result_rows`]). The check is
7//! applied ONCE, on the fully-materialized result (after the whole step
8//! pipeline), by [`crate::query::select_executor::SelectExecutor::execute`].
9//!
10//! SCOPE: the LEGACY [`QueryExecutor`](crate::query::executor::QueryExecutor)
11//! path — non-SELECT statements and any cached legacy SELECT plan reused via the
12//! plan-cache HIT branch — is NOT covered by this module's estimator directly;
13//! `QueryEngine::enforce_legacy_result_budget` applies the same ceiling at those
14//! sites. Since issue #1750 ad-hoc SELECTs (including simple `WHERE id = <value>`
15//! point lookups) route through the modern `SelectExecutor` and are budgeted
16//! there. (D6 redesign tracked on #1582.)
17//!
18//! This module is compiled unconditionally (unlike `select_executor`, which is
19//! `state_machine`-gated).
20
21use super::result::QueryRow;
22use crate::{Error, Result};

24/// Estimate the logical size, in bytes, of a materialized [`QueryRow`], reusing
25/// the shared row-cache estimator ([`crate::memory::estimate_value_size`]). Sums
26/// the per-value estimate over the row's column values, plus the retained
27/// [`RowKey`](crate::types::RowKey) bytes (issue #1582): every materialized row
28/// keeps the original partition/clustering key, so a query projecting small
29/// non-key columns from rows with very large keys must still count the key bytes
30/// against `max_result_bytes`. Uses saturating addition so a pathological row can
31/// never overflow `usize`.
32pub(crate) fn estimate_query_row_bytes(row: &QueryRow) -> usize {
33    let values_bytes: usize = row
34        .values
35        .values()
36        .map(crate::memory::estimate_value_size)
37        .sum();
38    values_bytes.saturating_add(row.key.as_bytes().len())
39}
40
41/// Sum [`estimate_query_row_bytes`] over a result set, saturating so a pathological
42/// estimate can never overflow `usize` (it stays at `usize::MAX`, which still
43/// trips the byte budget).
44pub(crate) fn estimate_result_bytes(rows: &[QueryRow]) -> usize {
45    rows.iter().fold(0usize, |acc, row| {
46        acc.saturating_add(estimate_query_row_bytes(row))
47    })
48}

50/// Enforce the byte-bounded result budget (primary) and the row-count safety
51/// valve (secondary) on a materialized result set (issue #1582 / D6).
52///
53/// `result_bytes` is the logical-byte estimate of `results`. Returns
54/// [`Error::ResultTooLarge`] (with a remedy message: add `LIMIT` or stream) when
55/// the byte budget is exceeded, or the legacy query-execution error when the
56/// row-count valve trips.
57pub(crate) fn enforce_result_budget(
58    results: &[QueryRow],
59    result_bytes: usize,
60    byte_budget: usize,
61    max_rows: usize,
62) -> Result<()> {
63    if result_bytes > byte_budget {
64        return Err(Error::ResultTooLarge {
65            budget_bytes: byte_budget,
66            estimated_bytes: result_bytes,
67            rows: results.len(),
68        });
69    }
70    if results.len() > max_rows {
71        return Err(Error::query_execution(
72            "Result set too large, consider adding LIMIT".to_string(),
73        ));
74    }
75    Ok(())
76}
77
78/// Enforce the budget on a fully-materialized set of rows in one call: estimate
79/// the total bytes, then apply [`enforce_result_budget`]. Used by the legacy
80/// engine point-lookup path, which materializes the whole result before checking.
81pub(crate) fn enforce_materialized_rows(
82    rows: &[QueryRow],
83    byte_budget: usize,
84    max_rows: usize,
85) -> Result<()> {
86    let result_bytes = estimate_result_bytes(rows);
87    enforce_result_budget(rows, result_bytes, byte_budget, max_rows)
88}

90#[cfg(test)]
91mod tests {
92    use super::*;
93    use crate::memory::estimate_value_size;
94    use crate::query::result::{QueryRow, RowMetadata};
95    use crate::types::{RowKey, Value};
96    use std::collections::HashMap;
97    use std::sync::Arc;
98
99    fn row_with(key_bytes: usize, value: Value) -> QueryRow {
100        let mut values: HashMap<Arc<str>, Value> = HashMap::new();
101        values.insert(Arc::from("c"), value);
102        QueryRow {
103            values,
104            key: RowKey::new(vec![b'k'; key_bytes]),
105            metadata: RowMetadata::default(),
106            cell_metadata: None,
107        }
108    }
109
110    /// Issue #1582: a wide partition key with a tiny projected value must count
111    /// the retained key bytes against the budget. Value bytes alone stay under,
112    /// so the budget only trips because the key is included.
113    #[test]
114    fn wide_key_small_value_trips_budget() {
115        // Value::Integer estimates 4 bytes; key is 1000 bytes.
116        let rows = vec![row_with(1000, Value::Integer(1))];
117        let value_only = rows[0]
118            .values
119            .values()
120            .map(estimate_value_size)
121            .sum::<usize>();
122        assert_eq!(value_only, 4, "projected value estimate should be small");
123
124        // Budget sits above the value-only estimate but below value+key.
125        let budget = 500;
126        assert!(value_only <= budget, "keys-ignored estimate would NOT trip");
127
128        let err = enforce_materialized_rows(&rows, budget, usize::MAX)
129            .expect_err("wide key must push the row over the byte budget");
130        match err {
131            Error::ResultTooLarge {
132                budget_bytes,
133                estimated_bytes,
134                rows: n,
135            } => {
136                assert_eq!(budget_bytes, budget);
137                assert!(estimated_bytes > budget, "estimate must include key bytes");
138                assert!(
139                    estimated_bytes >= 1000,
140                    "estimate must include the 1000-byte key"
141                );
142                assert_eq!(n, 1);
143            }
144            other => panic!("expected ResultTooLarge, got {other:?}"),
145        }
146    }
147
148    /// Sanity: within-budget rows (key + value) still pass.
149    #[test]
150    fn small_key_small_value_passes() {
151        let rows = vec![row_with(8, Value::Integer(1))];
152        enforce_materialized_rows(&rows, 500, usize::MAX)
153            .expect("small key + value must stay under budget");
154    }
155}
