16const NOT_IMPLEMENTED: &str = "ANTLR parser not yet implemented";
17
18fn not_implemented<T>() -> Result<T> {
19    Err(ParserError::backend("antlr", NOT_IMPLEMENTED).into())
20}
21
22/// ANTLR-based parser implementation
23#[derive(Debug, Default)]
24pub struct AntlrParser;
25
26impl AntlrParser {
27    /// Create a new ANTLR parser with the given configuration
28    pub fn new(config: ParserConfig) -> Result<Self> {
29        Self::validate_config(&config)?;
30        Ok(Self)
31    }
32
33    fn validate_config(config: &ParserConfig) -> Result<()> {
34        use super::config::ParserFeature;
35
36        if config.has_feature(&ParserFeature::Streaming) {
37            return Err(ParserError::unsupported_feature("antlr", "streaming").into());
38        }
39
40        Ok(())
41    }
42
43    /// Get backend information
44    pub fn backend_info() -> ParserBackendInfo {
45        ParserBackendInfo {
46            name: "antlr".to_string(),
47            version: "4.0".to_string(),
48            features: vec![
49                ParserFeature::ErrorRecovery,
50                ParserFeature::SyntaxHighlighting,
51                ParserFeature::CodeCompletion,
52                ParserFeature::AstTransformation,
53            ],
54            performance: PerformanceCharacteristics {
55                statements_per_second: 5_000,
56                memory_per_statement: 2048,
57                startup_time_ms: 50,
58                async_support: true,
59            },
60        }
61    }
62}

103#[cfg(test)]
104mod tests {
105    use super::super::config::ParserConfig;

107
108    #[test]
109    fn test_antlr_parser_creation() {
110        let config =
111            ParserConfig::default().with_backend(super::super::config::ParserBackend::Antlr);
112        let parser = AntlrParser::new(config).unwrap();
113
114        let info = parser.backend_info();
115        assert_eq!(info.name, "antlr");
116    }
117
118    #[tokio::test]
119    async fn test_not_implemented_error() {
120        let config = ParserConfig::default();
121        let parser = AntlrParser::new(config).unwrap();
122
123        let err = parser.parse("SELECT * FROM users").await.unwrap_err();
124        assert!(err.to_string().contains("not yet implemented"));
125    }
