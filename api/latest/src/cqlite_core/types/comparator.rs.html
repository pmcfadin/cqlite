1//! ComparatorType system for type comparison logic in CQLite
2//!
3//! This module provides a comprehensive comparison system for all CQL data types,
4//! including primitive types, collections, UDTs, and frozen types. The comparator
5//! system ensures proper ordering and equality comparison that matches Cassandra's
6//! comparison semantics.
7
8// `Custom("inet")` / `Custom("time")` value ordering (issue #3790).
9mod custom;

17/// ComparatorType enum that handles comparison logic for different CQL types
18#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
19pub enum ComparatorType {
20    /// Boolean type comparator
21    Boolean,
22    /// 8-bit signed integer comparator
23    TinyInt,
24    /// 16-bit signed integer comparator
25    SmallInt,
26    /// 32-bit signed integer comparator
27    Int,
28    /// 64-bit signed integer comparator
29    BigInt,
30    /// Counter type comparator (stored as 64-bit signed integer)
31    Counter,
32    /// 32-bit floating point comparator
33    Float32,
34    /// 64-bit floating point comparator
35    Float,
36    /// UTF-8 text comparator (lexicographic)
37    Text,
38    /// Binary data comparator (byte-wise)
39    Blob,
40    /// Timestamp comparator (chronological)
41    Timestamp,
42    /// UUID comparator (byte-wise)
43    Uuid,
44    /// Variable-length integer comparator
45    Varint,
46    /// Decimal type comparator
47    Decimal,
48    /// Duration type comparator
49    Duration,
50    /// Date type comparator (days since epoch)
51    Date,
52    /// JSON document comparator
53    Json,
54    /// List comparator with element comparator
55    List(Box<ComparatorType>),
56    /// Set comparator with element comparator
57    Set(Box<ComparatorType>),
58    /// Map comparator with key and value comparators
59    Map(Box<ComparatorType>, Box<ComparatorType>),
60    /// Tuple comparator with field comparators
61    Tuple(Vec<ComparatorType>),
62    /// UDT comparator with field definitions
63    Udt {
64        type_name: String,
65        keyspace: Option<String>,
66        field_comparators: Vec<(String, ComparatorType)>,
67    },
68    /// Frozen type comparator (byte-wise comparison of serialized form)
69    Frozen(Box<ComparatorType>),
70    /// Custom type comparator (fallback)
71    /// `vector<element, n>` — issue #4114 / roborev job 112.
72    ///
73    /// This variant exists so that CONSTRUCTING parsing metadata for a
74    /// vector-bearing schema SUCCEEDS. Comparator construction is metadata, not
75    /// ordering: `SchemaRegistry`'s `DerivedParsingState::compute` builds a
76    /// comparator for EVERY column (`schema/registry.rs:136-141`, not just key
77    /// columns), and `get_parsing_context` RE-COMPUTES and PROPAGATES that error on
78    /// the `derived: None` branch (`:852`). So refusing at construction did not
79    /// merely decline to order a vector — it made `get_parsing_context` fail for any
80    /// table carrying one, i.e. it broke schema-driven reads of exactly the tables
81    /// #4114 exists to make readable.
82    ///
83    /// The refusal is preserved where it belongs: `compare` and `supports_ordering`.
84    /// Cassandra's `VectorType` is `ComparisonType.CUSTOM` comparing element-wise via
85    /// its serializer (`VectorType.java:88`, `:122-125`), and #4114 implements
86    /// READING, not ORDERING — so an actual comparison still fails closed by name
87    /// rather than guessing a rule (e.g. reusing the `List` comparator) that nothing
88    /// here has verified against Cassandra.
89    Vector {
90        /// Comparator for the element type, carried so a future ordering
91        /// implementation has what it needs without re-deriving it.
92        element: Box<ComparatorType>,
93        /// Declared dimension.
94        dimension: usize,
95    },
96    Custom(String),
97}

99// Struct-size regression guard (issue #1616, Epic H/H3; see
100// docs/reports/parser-performance-audit-2026-07-01.md §Epic H (finding H3)).
101// `ComparatorType` is constructed per column during schema-aware decode and
102// clustering-key comparison on the read hot path; its widest variant (`Udt`)
103// sets the layout. Measured 72 bytes today on 64-bit targets. Update this pin
104// DELIBERATELY, never silently: any change — growth or shrink — must be a
105// reviewed edit here.
106#[cfg(target_pointer_width = "64")]
107const _: () = assert!(std::mem::size_of::<ComparatorType>() == 72);

109/// The ONE refusal both `from_cql_type` constructors return for a vector column,
110/// so the two cannot drift into two different messages (issue #4114).
111fn vector_ordering_unsupported() -> Error {
112    Error::unsupported_format(
113        "ordering/comparing a vector column is not implemented (issue #4114 covers \
114         READING vector<float, n> values); Cassandra's VectorType is a CUSTOM \
115         comparator that compares element-wise via its serializer \
116         (VectorType.java:88,122-125)"
117            .to_string(),
118    )
119}

122    /// Create a ComparatorType from a CqlType
123    pub fn from_cql_type(cql_type: &CqlType) -> Result<Self> {
124        let comparator = match cql_type {
125            CqlType::Boolean => ComparatorType::Boolean,
126            CqlType::TinyInt => ComparatorType::TinyInt,
127            CqlType::SmallInt => ComparatorType::SmallInt,
128            CqlType::Int => ComparatorType::Int,
129            CqlType::BigInt => ComparatorType::BigInt,
130            CqlType::Counter => ComparatorType::Counter,
131            CqlType::Float => ComparatorType::Float32,
132            CqlType::Double => ComparatorType::Float,
133            CqlType::Text | CqlType::Varchar => ComparatorType::Text,
134            CqlType::Ascii => ComparatorType::Text, // Map ASCII to Text for simplicity
135            CqlType::Blob => ComparatorType::Blob,
136            CqlType::Timestamp => ComparatorType::Timestamp,
137            CqlType::Uuid => ComparatorType::Uuid,
138            CqlType::TimeUuid => ComparatorType::Uuid, // Map TimeUuid to Uuid for simplicity
139            CqlType::Varint => ComparatorType::Varint,
140            CqlType::List(element_type) => {
141                let element_comparator = Self::from_cql_type(element_type)?;
142                ComparatorType::List(Box::new(element_comparator))
143            }
144            CqlType::Set(element_type) => {
145                let element_comparator = Self::from_cql_type(element_type)?;
146                ComparatorType::Set(Box::new(element_comparator))
147            }
148            CqlType::Map(key_type, value_type) => {
149                let key_comparator = Self::from_cql_type(key_type)?;
150                let value_comparator = Self::from_cql_type(value_type)?;
151                ComparatorType::Map(Box::new(key_comparator), Box::new(value_comparator))
152            }
153            CqlType::Tuple(field_types) => {
154                let mut field_comparators = Vec::new();
155                for field_type in field_types {
156                    field_comparators.push(Self::from_cql_type(field_type)?);
157                }
158                ComparatorType::Tuple(field_comparators)
159            }
160            CqlType::Udt(type_name, fields) => {
161                let mut field_comparators = Vec::new();
162                for (field_name, field_type) in fields {
163                    let field_comparator = Self::from_cql_type(field_type)?;
164                    field_comparators.push((field_name.clone(), field_comparator));
165                }
166                ComparatorType::Udt {
167                    type_name: type_name.clone(),
168                    keyspace: None, // Default keyspace, should be set by caller
169                    field_comparators,
170                }
171            }
172            CqlType::Frozen(inner_type) => {
173                let inner_comparator = Self::from_cql_type(inner_type)?;
174                ComparatorType::Frozen(Box::new(inner_comparator))
175            }
176            CqlType::Custom(type_name) => ComparatorType::Custom(type_name.clone()),
177            // CONSTRUCT, do not refuse (roborev job 112). Refusing here broke
178            // `get_parsing_context` for every vector-bearing table — see the
179            // `ComparatorType::Vector` doc comment. Ordering is still refused, at
180            // `compare`/`supports_ordering`, which is where the unsupported
181            // OPERATION actually happens.
182            CqlType::Vector(element_type, dimension) => ComparatorType::Vector {
183                element: Box::new(Self::from_cql_type(element_type)?),
184                dimension: *dimension,
185            },
186            // Map newly supported types
187            CqlType::Decimal => ComparatorType::Decimal,
188            CqlType::Duration => ComparatorType::Duration,
189            CqlType::Date => ComparatorType::Date,
190            // Map remaining unsupported types to custom for now
191            CqlType::Time => ComparatorType::Custom("time".to_string()),
192            CqlType::Inet => ComparatorType::Custom("inet".to_string()),
193        };
194
195        Ok(comparator)
196    }

211    /// This method resolves UDT type references (e.g., "address_type") to their full

213    /// properly parsing UDTs inside collections like `list<frozen<address_type>>`.
214    ///
215    /// # Arguments
216    /// * `data_type` - The data type string (e.g., "list<frozen<address_type>>")
217    /// * `registry` - The UDT registry containing type definitions
218    /// * `keyspace` - The keyspace to use for UDT lookup
219    pub fn from_data_type_with_registry(
220        data_type: &str,
221        registry: &UdtRegistry,
222        keyspace: &str,
223    ) -> Result<Self> {
224        let cql_type = CqlType::parse(data_type)?;
225        Self::from_cql_type_with_registry(&cql_type, registry, keyspace)
226    }

228    /// Create a ComparatorType from a CqlType with UDT registry resolution (`pub` since issue #2339: a caller may already hold a parsed `CqlType`, e.g. the merged-read assembler resolving a collection's element/key type)
229    pub fn from_cql_type_with_registry(
230        cql_type: &CqlType,
231        registry: &UdtRegistry,
232        keyspace: &str,
233    ) -> Result<Self> {
234        let comparator = match cql_type {
235            CqlType::Boolean => ComparatorType::Boolean,
236            CqlType::TinyInt => ComparatorType::TinyInt,
237            CqlType::SmallInt => ComparatorType::SmallInt,
238            CqlType::Int => ComparatorType::Int,
239            CqlType::BigInt => ComparatorType::BigInt,
240            CqlType::Counter => ComparatorType::Counter,
241            CqlType::Float => ComparatorType::Float32,
242            CqlType::Double => ComparatorType::Float,
243            CqlType::Text | CqlType::Varchar => ComparatorType::Text,
244            CqlType::Ascii => ComparatorType::Text,
245            CqlType::Blob => ComparatorType::Blob,
246            CqlType::Timestamp => ComparatorType::Timestamp,
247            CqlType::Uuid => ComparatorType::Uuid,
248            CqlType::TimeUuid => ComparatorType::Uuid,
249            CqlType::Varint => ComparatorType::Varint,
250            CqlType::Decimal => ComparatorType::Decimal,
251            CqlType::Duration => ComparatorType::Duration,
252            CqlType::Date => ComparatorType::Date,
253            CqlType::Time => ComparatorType::Custom("time".to_string()),
254            CqlType::Inet => ComparatorType::Custom("inet".to_string()),
255            CqlType::List(element_type) => {
256                let element_comparator =
257                    Self::from_cql_type_with_registry(element_type, registry, keyspace)?;
258                ComparatorType::List(Box::new(element_comparator))
259            }
260            CqlType::Set(element_type) => {
261                let element_comparator =
262                    Self::from_cql_type_with_registry(element_type, registry, keyspace)?;
263                ComparatorType::Set(Box::new(element_comparator))
264            }
265            CqlType::Map(key_type, value_type) => {
266                let key_comparator =
267                    Self::from_cql_type_with_registry(key_type, registry, keyspace)?;
268                let value_comparator =
269                    Self::from_cql_type_with_registry(value_type, registry, keyspace)?;
270                ComparatorType::Map(Box::new(key_comparator), Box::new(value_comparator))
271            }
272            CqlType::Tuple(field_types) => {
273                let mut field_comparators = Vec::new();
274                for field_type in field_types {
275                    field_comparators.push(Self::from_cql_type_with_registry(
276                        field_type, registry, keyspace,
277                    )?);
278                }
279                ComparatorType::Tuple(field_comparators)
280            }
281            CqlType::Udt(type_name, fields) => {
282                let mut field_comparators = Vec::new();
283                for (field_name, field_type) in fields {
284                    let field_comparator =
285                        Self::from_cql_type_with_registry(field_type, registry, keyspace)?;
286                    field_comparators.push((field_name.clone(), field_comparator));
287                }
288                ComparatorType::Udt {
289                    type_name: type_name.clone(),
290                    keyspace: Some(keyspace.to_string()),
291                    field_comparators,
292                }
293            }
294            CqlType::Frozen(inner_type) => {
295                let inner_comparator =
296                    Self::from_cql_type_with_registry(inner_type, registry, keyspace)?;
297                ComparatorType::Frozen(Box::new(inner_comparator))
298            }
299            CqlType::Custom(type_name) => {
300                // Check if this is a UDT reference (format: "udt:type_name" or just "type_name")
301                let udt_name = type_name.strip_prefix("udt:").unwrap_or(type_name);
302
303                // Look up UDT in registry - try with and without keyspace
304                let udt_def = registry
305                    .get_udt(keyspace, udt_name)
306                    .or_else(|| registry.get_udt(keyspace, type_name));
307
308                if let Some(udt_def) = udt_def {
309                    let mut field_comparators = Vec::new();
310                    for field in &udt_def.fields {
311                        // Recursively resolve field types (field_type is already CqlType)
312                        let field_comparator = Self::from_cql_type_with_registry(
313                            &field.field_type,
314                            registry,
315                            keyspace,
316                        )?;
317                        field_comparators.push((field.name.clone(), field_comparator));
318                    }
319                    ComparatorType::Udt {
320                        type_name: udt_def.name.clone(),
321                        keyspace: Some(keyspace.to_string()),
322                        field_comparators,
323                    }
324                } else {
325                    // UDT not found in registry - return as custom
326                    ComparatorType::Custom(type_name.clone())
327                }
328            }
329            // CONSTRUCT here too (roborev job 113). Fixing only `from_cql_type` left
330            // this REGISTRY-AWARE TWIN still refusing, so
331            // `from_data_type_with_registry("vector<float, 3>", ..)` kept failing and a
332            // registry-resolved composite containing a vector field still could not
333            // build its comparator — the same defect, one function over. The element
334            // recurses through the REGISTRY-aware call so a UDT element still resolves.
335            CqlType::Vector(element_type, dimension) => ComparatorType::Vector {
336                element: Box::new(Self::from_cql_type_with_registry(
337                    element_type,
338                    registry,
339                    keyspace,
340                )?),
341                dimension: *dimension,
342            },
343        };
344
345        Ok(comparator)
346    }

348    /// Compare two values using this comparator
349    pub fn compare(&self, left: &Value, right: &Value) -> Result<Ordering> {
350        // Handle null values first (nulls are always considered less than non-nulls)
351        match (left.is_null(), right.is_null()) {
352            (true, true) => return Ok(Ordering::Equal),
353            (true, false) => return Ok(Ordering::Less),
354            (false, true) => return Ok(Ordering::Greater),
355            (false, false) => {} // Both non-null, continue with type-specific comparison
356        }
357
358        match self {
359            ComparatorType::Boolean => self.compare_boolean(left, right),
360            ComparatorType::TinyInt => self.compare_tinyint(left, right),
361            ComparatorType::SmallInt => self.compare_smallint(left, right),
362            ComparatorType::Int => self.compare_int(left, right),
363            ComparatorType::BigInt => self.compare_bigint(left, right),
364            ComparatorType::Counter => self.compare_bigint(left, right), // Counter uses same comparison as bigint
365            ComparatorType::Float32 => self.compare_float32(left, right),
366            ComparatorType::Float => self.compare_float(left, right),
367            ComparatorType::Text => self.compare_text(left, right),
368            ComparatorType::Blob => self.compare_blob(left, right),
369            ComparatorType::Timestamp => self.compare_timestamp(left, right),
370            ComparatorType::Uuid => self.compare_uuid(left, right),
371            ComparatorType::Varint => self.compare_varint(left, right),
372            ComparatorType::Decimal => self.compare_decimal(left, right),
373            ComparatorType::Duration => self.compare_duration(left, right),
374            ComparatorType::Date => self.compare_date(left, right),
375            ComparatorType::Json => self.compare_json(left, right),
376            ComparatorType::List(element_comparator) => {
377                self.compare_list(left, right, element_comparator)
378            }
379            ComparatorType::Set(element_comparator) => {
380                self.compare_set(left, right, element_comparator)
381            }
382            ComparatorType::Map(key_comparator, value_comparator) => {
383                self.compare_map(left, right, key_comparator, value_comparator)
384            }
385            ComparatorType::Tuple(field_comparators) => {
386                self.compare_tuple(left, right, field_comparators)
387            }
388            ComparatorType::Udt {
389                field_comparators, ..
390            } => self.compare_udt(left, right, field_comparators),
391            ComparatorType::Frozen(inner_comparator) => {
392                self.compare_frozen(left, right, inner_comparator)
393            }
394            // ORDERING refused HERE — at the unsupported OPERATION — not at
395            // construction. Refusing at construction broke get_parsing_context for
396            // every vector-bearing table (roborev job 112).
397            ComparatorType::Vector { .. } => Err(vector_ordering_unsupported()),
398            ComparatorType::Custom(name) => custom::compare(name, left, right),
399        }
400    }

402    /// Check if two values are equal using this comparator
403    pub fn equals(&self, left: &Value, right: &Value) -> Result<bool> {
404        Ok(self.compare(left, right)? == Ordering::Equal)
405    }

407    /// Check if left value is less than right value
408    pub fn less_than(&self, left: &Value, right: &Value) -> Result<bool> {
409        Ok(self.compare(left, right)? == Ordering::Less)
410    }
411
412    /// Check if left value is greater than right value
413    pub fn greater_than(&self, left: &Value, right: &Value) -> Result<bool> {
414        Ok(self.compare(left, right)? == Ordering::Greater)
415    }

417    /// Check if left value is less than or equal to right value
418    pub fn less_than_or_equal(&self, left: &Value, right: &Value) -> Result<bool> {
419        let ordering = self.compare(left, right)?;
420        Ok(ordering == Ordering::Less || ordering == Ordering::Equal)
421    }
422
423    /// Check if left value is greater than or equal to right value
424    pub fn greater_than_or_equal(&self, left: &Value, right: &Value) -> Result<bool> {
425        let ordering = self.compare(left, right)?;
426        Ok(ordering == Ordering::Greater || ordering == Ordering::Equal)
427    }

461    /// Check if this comparator supports ordering (some types only support equality)
462    pub fn supports_ordering(&self) -> bool {
463        match self {
464            ComparatorType::Boolean
465            | ComparatorType::TinyInt
466            | ComparatorType::SmallInt
467            | ComparatorType::Int
468            | ComparatorType::BigInt
469            | ComparatorType::Counter
470            | ComparatorType::Float32
471            | ComparatorType::Float
472            | ComparatorType::Text
473            | ComparatorType::Blob
474            | ComparatorType::Timestamp
475            | ComparatorType::Uuid
476            | ComparatorType::Varint
477            | ComparatorType::Decimal
478            | ComparatorType::Duration
479            | ComparatorType::Date
480            | ComparatorType::Json => true,
481            ComparatorType::List(element_comparator) => element_comparator.supports_ordering(),
482            ComparatorType::Set(_) => false, // Sets only support equality
483            ComparatorType::Map(_, _) => false, // Maps only support equality
484            ComparatorType::Tuple(field_comparators) => {
485                field_comparators.iter().all(|c| c.supports_ordering())
486            }
487            ComparatorType::Udt {
488                field_comparators, ..
489            } => field_comparators.iter().all(|(_, c)| c.supports_ordering()),
490            ComparatorType::Frozen(inner_comparator) => inner_comparator.supports_ordering(),
491            // `inet` / `time` order by value (issue #3790); the residual
492            // unresolved-UDT / unknown name does not.
493            // A vector has no verified ordering rule here, so it does NOT support
494            // ordering. Callers that ask first get an honest `false` instead of an
495            // error from `compare`.
496            ComparatorType::Vector { .. } => false,
497            ComparatorType::Custom(name) => custom::supports_ordering(name),
498        }
499    }

503    fn compare_boolean(&self, left: &Value, right: &Value) -> Result<Ordering> {
504        match (left.as_bool(), right.as_bool()) {
505            (Some(l), Some(r)) => Ok(l.cmp(&r)),
506            _ => Err(Error::Schema(
507                "Type mismatch: expected boolean values".to_string(),
508            )),
509        }

548    fn compare_float32(&self, left: &Value, right: &Value) -> Result<Ordering> {
549        match (left, right) {
550            // Cassandra/Java total order (NaN last, -0.0 < +0.0), NOT IEEE
551            // partial_cmp — this is the schema-aware clustering-key read hot
552            // path and must agree with the writer's ClusteringKey ordering.
553            // See float_cmp.rs and issues #1870/#2010.
554            (Value::Float32(l), Value::Float32(r)) => {
555                Ok(crate::float_cmp::cassandra_float_cmp(*l, *r))
556            }

720    fn compare_list_values(
721        &self,
722        left: &[Value],
723        right: &[Value],
724        element_comparator: &ComparatorType,
725    ) -> Result<Ordering> {
726        let min_len = left.len().min(right.len());
727
728        // Compare elements pairwise
729        for i in 0..min_len {
730            let cmp = element_comparator.compare(&left[i], &right[i])?;
731            if cmp != Ordering::Equal {
732                return Ok(cmp);
733            }
734        }
735
736        // If all compared elements are equal, compare lengths
737        Ok(left.len().cmp(&right.len()))
738    }

740    fn compare_set(
741        &self,
742        left: &Value,
743        right: &Value,
744        _element_comparator: &ComparatorType,
745    ) -> Result<Ordering> {
746        match (left, right) {
747            (Value::Set(l), Value::Set(r)) => {
748                // Sets only support equality comparison
749                if l.len() != r.len() {
750                    return Ok(Ordering::Greater); // Arbitrary but consistent
751                }
752
753                // Check if all elements in left set are in right set
754                // This is a simplified implementation; a full implementation would need proper set logic
755                for l_val in l {
756                    let mut found = false;
757                    for r_val in r {
758                        if l_val == r_val {
759                            found = true;
760                            break;
761                        }
762                    }
763                    if !found {
764                        return Ok(Ordering::Greater); // Arbitrary but consistent
765                    }
766                }
767
768                Ok(Ordering::Equal)
769            }
770            _ => Err(Error::Schema(
771                "Type mismatch: expected set values".to_string(),
772            )),
773        }
774    }

805        left: &Value,
806        right: &Value,
807        field_comparators: &[ComparatorType],
808    ) -> Result<Ordering> {
809        match (left, right) {
810            (Value::Tuple(l), Value::Tuple(r)) => {
811                let min_len = l.len().min(r.len()).min(field_comparators.len());
812
813                // Compare fields pairwise using their respective comparators

815                    let cmp = field_comparators[i].compare(&l[i], &r[i])?;
816                    if cmp != Ordering::Equal {
817                        return Ok(cmp);
818                    }
819                }
820
821                // If all compared fields are equal, compare tuple lengths
822                Ok(l.len().cmp(&r.len()))
823            }
824            _ => Err(Error::Schema(
825                "Type mismatch: expected tuple values".to_string(),
826            )),
827        }

830    fn compare_udt(
831        &self,
832        left: &Value,
833        right: &Value,
834        field_comparators: &[(String, ComparatorType)],
835    ) -> Result<Ordering> {
836        match (left, right) {
837            (Value::Udt(l), Value::Udt(r)) => {
838                // Compare UDT type names first
839                if l.type_name != r.type_name {
840                    return Ok(l.type_name.cmp(&r.type_name));
841                }
842
843                // Compare fields in definition order
844                for (field_name, field_comparator) in field_comparators {
845                    // Find field in UDT fields
846                    let l_val = l
847                        .fields
848                        .iter()
849                        .find(|f| f.name == *field_name)
850                        .and_then(|f| f.value.as_ref())
851                        .unwrap_or(&Value::Null);
852
853                    let r_val = r
854                        .fields
855                        .iter()
856                        .find(|f| f.name == *field_name)
857                        .and_then(|f| f.value.as_ref())
858                        .unwrap_or(&Value::Null);
859
860                    let cmp = field_comparator.compare(l_val, r_val)?;
861                    if cmp != Ordering::Equal {
862                        return Ok(cmp);
863                    }
864                }
865
866                Ok(Ordering::Equal)
867            }
868            _ => Err(Error::Schema(
869                "Type mismatch: expected UDT values".to_string(),
870            )),
871        }
872    }

892impl std::fmt::Display for ComparatorType {
893    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
894        match self {
895            ComparatorType::List(element_type) => write!(f, "list<{}>", element_type),
896            ComparatorType::Set(element_type) => write!(f, "set<{}>", element_type),
897            ComparatorType::Map(key_type, value_type) => {
898                write!(f, "map<{}, {}>", key_type, value_type)
899            }
900            ComparatorType::Tuple(field_types) => {
901                write!(f, "tuple<")?;
902                for (i, field_type) in field_types.iter().enumerate() {
903                    if i > 0 {
904                        write!(f, ", ")?;
905                    }
906                    write!(f, "{}", field_type)?;
907                }
908                write!(f, ">")
909            }
910            ComparatorType::Udt {
911                type_name,
912                keyspace,
913                ..
914            } => {
915                if let Some(ks) = keyspace {
916                    write!(f, "{}.{}", ks, type_name)
917                } else {
918                    write!(f, "{}", type_name)
919                }
920            }
921            ComparatorType::Frozen(inner_type) => write!(f, "frozen<{}>", inner_type),
922            _ => write!(f, "{}", self.type_name()),
923        }
924    }
925}
