1//! Tombstone (deletion) value decoding.
2//!
3//! Decodes the legacy CQL binary tombstone encoding into `Value::Tombstone`.
4
5use super::super::vint::parse_vint_length_signed;
6use crate::types::{RowKey, TombstoneInfo, TombstoneType, Value};
7use nom::{
8    bytes::complete::take,
9    number::complete::{be_i64, be_u8},
10    IResult,
11};
12
13/// Parse tombstone information with enhanced Cassandra 5.0 compatibility
14pub fn parse_tombstone(input: &[u8]) -> IResult<&[u8], Value> {
15    // Parse deletion timestamp (microseconds since epoch)
16    let (input, deletion_time) = be_i64(input)?;
17
18    // Parse tombstone type byte
19    let (input, tombstone_type_byte) = be_u8(input)?;
20
21    let tombstone_type = match tombstone_type_byte {
22        0 => TombstoneType::RowTombstone,
23        1 => TombstoneType::CellTombstone,
24        2 => TombstoneType::RangeTombstone,
25        3 => TombstoneType::TtlExpiration,
26        _ => {
27            return Err(nom::Err::Error(nom::error::Error::new(
28                input,
29                nom::error::ErrorKind::Verify,
30            )));
31        }
32    };
33
34    // Parse optional TTL for TTL-based tombstones
35    let (input, ttl) = if tombstone_type == TombstoneType::TtlExpiration {
36        let (input, ttl_value) = be_i64(input)?;
37        (input, Some(ttl_value))
38    } else {
39        (input, None)
40    };
41
42    // Parse optional clustering key range for range tombstones
43    let (input, range_start, range_end) = if tombstone_type == TombstoneType::RangeTombstone {
44        let (input, has_range) = be_u8(input)?;
45        if has_range != 0 {
46            // CQLite-internal: encoder is ZigZag encode_vint (see types/mod.rs
47            // serialize_cql_value); intentionally signed.
48            let (input, start_len) = parse_vint_length_signed(input)?;
49            let (input, start_data) = take(start_len)(input)?;
50            // CQLite-internal: encoder is ZigZag encode_vint (see types/mod.rs
51            // serialize_cql_value); intentionally signed.
52            let (input, end_len) = parse_vint_length_signed(input)?;
53            let (input, end_data) = take(end_len)(input)?;
54            (input, Some(start_data.to_vec()), Some(end_data.to_vec()))
55        } else {
56            (input, None, None)
57        }
58    } else {
59        (input, None, None)
60    };
61
62    let tombstone_info = TombstoneInfo {
63        deletion_time,
64        tombstone_type,
65        // The legacy CQL binary tombstone encoding carries no localDeletionTime
66        // (only deletion_time); default to 0 (#873).
67        local_deletion_time: 0,
68        ttl,
69        range_start: range_start.map(RowKey::new),
70        range_end: range_end.map(RowKey::new),
71    };
72
73    Ok((input, Value::Tombstone(Box::new(tombstone_info))))
74}

81    #[test]
82    fn test_tombstone_parsing() {
83        // Test row tombstone parsing
84        let row_tombstone = Value::row_tombstone(1000);
85        let serialized = serialize_cql_value(&row_tombstone).unwrap();
86
87        // Parse it back
88        let (remaining, parsed_value) = parse_tombstone(&serialized[1..]).unwrap(); // Skip type ID
89        assert!(remaining.is_empty());
90        assert_eq!(parsed_value, row_tombstone);
91
92        // Test TTL tombstone parsing
93        let ttl_tombstone = Value::ttl_tombstone(2000, 1000);
94        let serialized_ttl = serialize_cql_value(&ttl_tombstone).unwrap();
95
96        let (remaining, parsed_ttl) = parse_tombstone(&serialized_ttl[1..]).unwrap(); // Skip type ID
97        assert!(remaining.is_empty());
98        assert_eq!(parsed_ttl, ttl_tombstone);
99    }

126    #[test]
127    fn test_range_tombstone_range_roundtrip_zigzag() {
128        // Issue #1623 regression: the range_start/range_end length prefixes are
129        // written by serialize_cql_value with the CQLite ZigZag encoder
130        // (encode_vint), so they MUST be read back with the signed decoder.
131        // A length of 1 encodes as ZigZag 0x02; decoding it as UNSIGNED would
132        // yield 2 and corrupt the round-trip (take too many bytes / mismatch).
133        use crate::types::{TombstoneInfo, TombstoneType};
134        let range_tombstone = Value::Tombstone(Box::new(TombstoneInfo {
135            deletion_time: 4242,
136            tombstone_type: TombstoneType::RangeTombstone,
137            local_deletion_time: 0,
138            ttl: None,
139            // Byte length exactly 1 so ZigZag (0x02) vs unsigned (0x01) disagree.
140            range_start: Some(RowKey::new(vec![0xAB])),
141            range_end: Some(RowKey::new(vec![0xCD])),
142        }));
143        let serialized = serialize_cql_value(&range_tombstone).unwrap();
144
145        let (remaining, parsed) = parse_tombstone(&serialized[1..]).unwrap(); // Skip type ID
146        assert!(
147            remaining.is_empty(),
148            "range tombstone should consume all bytes"
149        );
150        assert_eq!(
151            parsed, range_tombstone,
152            "range tombstone round-trip mismatch"
153        );
154
155        if let Value::Tombstone(info) = parsed {
156            assert_eq!(
157                info.range_start.map(|k| k.as_bytes().to_vec()),
158                Some(vec![0xAB])
159            );
160            assert_eq!(
161                info.range_end.map(|k| k.as_bytes().to_vec()),
162                Some(vec![0xCD])
163            );
164        } else {
165            panic!("expected Value::Tombstone");
166        }
167    }

169    #[test]
170    fn test_partition_tombstone_serialize_errors() {
171        // PartitionTombstone has no encoding in the legacy CQL binary format.
172        // Serializing one must return an explicit error, not silently alias to
173        // RowTombstone (byte 0) — that would be a silent lossy round-trip.
174        use crate::types::{TombstoneInfo, TombstoneType};
175        let partition_tombstone = Value::Tombstone(Box::new(TombstoneInfo {
176            deletion_time: 9999,
177            tombstone_type: TombstoneType::PartitionTombstone,
178            local_deletion_time: 0,
179            ttl: None,
180            range_start: None,
181            range_end: None,
182        }));
183        let result = serialize_cql_value(&partition_tombstone);
184        assert!(
185            result.is_err(),
186            "serializing PartitionTombstone must return Err, not Ok"
187        );
188        let err_msg = result.unwrap_err().to_string();
189        assert!(
190            err_msg.contains("PartitionTombstone"),
191            "error message should mention PartitionTombstone, got: {err_msg}"
192        );
193    }
