1//! CQL type system parsing and serialization
2//!
3//! This module handles parsing and serialization of CQL data types as they
4//! appear in Cassandra SSTable format. It maps between the binary representation
5//! and the CQLite Value types.
6//!
7//! The implementation is split by responsibility:
8//! - [`primitives`] — scalar types (int/text/uuid/timestamp/decimal/…).
9//! - [`collections`] — list/set/map (legacy, v5, registry-aware, schema-driven).
10//! - [`udt`] — UDT, tuple, and frozen decoding.
11//! - [`tombstones`] — tombstone/deletion decoding.
12//!
13//! This `mod.rs` keeps the shared model ([`CqlTypeId`], type-id ⇄ value mapping,
14//! empty-value construction), the value-dispatch entry points
15//! ([`parse_cql_value`], [`parse_cql_value_raw`]), the schema-driven dispatcher,
16//! buffer-consumption validation helpers, and serialization — and re-exports the
17//! full public surface so callers using `parser::types::*` are unaffected.
18

20mod primitives;
21mod tombstones;
22mod udt;

49/// CQL type identifiers as they appear in the binary format
50#[derive(Debug, Clone, Copy, PartialEq, Eq)]
51#[repr(u8)]
52pub enum CqlTypeId {
53    Custom = 0x00,
54    Ascii = 0x01,
55    BigInt = 0x02,
56    Blob = 0x03,
57    Boolean = 0x04,
58    Counter = 0x05,
59    Decimal = 0x06,
60    Double = 0x07,
61    Float = 0x08,
62    Int = 0x09,
63    Timestamp = 0x0B,
64    Uuid = 0x0C,
65    Varchar = 0x0D,
66    Varint = 0x0E,
67    Timeuuid = 0x0F,
68    Inet = 0x10,
69    Date = 0x11,
70    Time = 0x12,
71    Smallint = 0x13,
72    Tinyint = 0x14,
73    Duration = 0x15,
74    List = 0x20,
75    Map = 0x21,
76    Set = 0x22,
77    Udt = 0x30,
78    Tuple = 0x31,
79    // Custom CQLite extension for tombstones
80    Tombstone = 0xFF,
81}

83impl TryFrom<u8> for CqlTypeId {
84    type Error = Error;
85
86    fn try_from(value: u8) -> Result<Self> {
87        match value {
88            0x00 => Ok(CqlTypeId::Custom),
89            0x01 => Ok(CqlTypeId::Ascii),
90            0x02 => Ok(CqlTypeId::BigInt),
91            0x03 => Ok(CqlTypeId::Blob),
92            0x04 => Ok(CqlTypeId::Boolean),
93            0x05 => Ok(CqlTypeId::Counter),
94            0x06 => Ok(CqlTypeId::Decimal),
95            0x07 => Ok(CqlTypeId::Double),
96            0x08 => Ok(CqlTypeId::Float),
97            0x09 => Ok(CqlTypeId::Int),
98            0x0B => Ok(CqlTypeId::Timestamp),
99            0x0C => Ok(CqlTypeId::Uuid),
100            0x0D => Ok(CqlTypeId::Varchar),
101            0x0E => Ok(CqlTypeId::Varint),
102            0x0F => Ok(CqlTypeId::Timeuuid),
103            0x10 => Ok(CqlTypeId::Inet),
104            0x11 => Ok(CqlTypeId::Date),
105            0x12 => Ok(CqlTypeId::Time),
106            0x13 => Ok(CqlTypeId::Smallint),
107            0x14 => Ok(CqlTypeId::Tinyint),
108            0x15 => Ok(CqlTypeId::Duration),
109            0x20 => Ok(CqlTypeId::List),
110            0x21 => Ok(CqlTypeId::Map),
111            0x22 => Ok(CqlTypeId::Set),
112            0x30 => Ok(CqlTypeId::Udt),
113            0x31 => Ok(CqlTypeId::Tuple),
114            0xFF => Ok(CqlTypeId::Tombstone),
115            _ => Err(Error::corruption(format!(
116                "Unknown CQL type ID: 0x{:02X}",
117                value
118            ))),
119        }
120    }
121}

128/// Parse a CQL value based on its type with enhanced collection support
129pub fn parse_cql_value(input: &[u8], type_id: CqlTypeId) -> IResult<&[u8], Value> {
130    match type_id {
131        CqlTypeId::Boolean => parse_boolean(input),
132        CqlTypeId::Tinyint => parse_tinyint(input),
133        CqlTypeId::Smallint => parse_smallint(input),
134        CqlTypeId::Int => parse_int(input),
135        CqlTypeId::BigInt => parse_bigint(input),
136        CqlTypeId::Counter => parse_counter(input),
137        CqlTypeId::Float => parse_float(input),
138        CqlTypeId::Double => parse_double(input),
139        CqlTypeId::Ascii | CqlTypeId::Varchar => {
140            // In the SSTable cell format the caller has already extracted exactly
141            // the bytes belonging to this cell (length-framing is done at the cell
142            // level before parse_cql_value is invoked).  The entire `input` slice
143            // IS the text value — no additional length prefix to strip.
144            //
145            // For Ascii, Cassandra's AsciiSerializer requires every byte to be in
146            // the 0x00-0x7F range.  For Varchar/Text any valid UTF-8 is accepted.
147            // We accept both under the same validation path here; a stricter
148            // Ascii-only byte check can be added when needed.
149            #[cfg(feature = "legacy-heuristics")]
150            {
151                // Legacy path: try 4-byte big-endian length prefix, then null-terminated,
152                // then raw UTF-8 (kept for pre-5.0 compatibility only).
153                if input.len() >= 4 {
154                    let length =
155                        u32::from_be_bytes([input[0], input[1], input[2], input[3]]) as usize;
156                    if input.len() >= 4 + length {
157                        let text_bytes = &input[4..4 + length];
158                        if let Ok(text) = String::from_utf8(text_bytes.to_vec()) {
159                            return Ok((&input[4 + length..], Value::Text(text)));
160                        }
161                    }
162                }
163                if let Some(null_pos) = input.iter().position(|&b| b == 0) {
164                    if let Ok(text) = String::from_utf8(input[..null_pos].to_vec()) {
165                        return Ok((&input[null_pos + 1..], Value::Text(text)));
166                    }
167                }
168                if let Ok(text) = String::from_utf8(input.to_vec()) {
169                    return Ok((&[], Value::Text(text)));
170                }
171                parse_text(input)
172            }
173            #[cfg(not(feature = "legacy-heuristics"))]
174            {
175                // Deterministic path: treat the entire input as the UTF-8 text value.
176                let text = String::from_utf8(input.to_vec()).map_err(|_| {
177                    nom::Err::Error(nom::error::Error::new(input, nom::error::ErrorKind::Verify))
178                })?;
179                Ok((&[], Value::Text(text)))
180            }
181        }
182        CqlTypeId::Blob => {
183            // The caller has already extracted exactly the bytes belonging to
184            // this cell (length-framing is done at the cell level before
185            // parse_cql_value is invoked), so the entire `input` slice IS the
186            // blob value verbatim — mirroring the sibling Ascii/Varchar arm,
187            // parse_cql_value_raw and parse_blob_value. Framing is the caller's
188            // responsibility via this function's contract; the decoder never
189            // sniffs byte patterns to infer a length (no-heuristics mandate,
190            // issues #28 / #1630). The genuinely VInt-framed decode lives in
191            // parse_blob (used by parse_cql_value_with_schema and the write
192            // side's tagged serialization), not here.
193            Ok((&[], Value::Blob(input.to_vec())))
194        }
195        CqlTypeId::Uuid | CqlTypeId::Timeuuid => parse_uuid(input),
196        CqlTypeId::Timestamp => parse_timestamp(input),
197        CqlTypeId::Date => parse_date(input),
198        CqlTypeId::Time => parse_time(input),
199        CqlTypeId::Varint => parse_varint(input),
200        CqlTypeId::Decimal => parse_decimal(input),
201        CqlTypeId::Duration => parse_duration(input),
202        CqlTypeId::Inet => parse_inet(input),
203        CqlTypeId::List => parse_list_enhanced(input),
204        CqlTypeId::Set => parse_set_enhanced(input),
205        CqlTypeId::Map => parse_map_enhanced(input),
206        CqlTypeId::Udt => parse_udt_enhanced(input),
207        CqlTypeId::Tuple => parse_tuple(input),
208        CqlTypeId::Tombstone => parse_tombstone(input),
209        CqlTypeId::Custom => {
210            // Custom types require additional metadata, return as blob for now
211            parse_blob(input)
212        }
213    }
214}

216/// Parse a CQL value from raw data (without length prefix)
217/// This is used for collection elements where length is already parsed
218pub fn parse_cql_value_raw(input: &[u8], type_id: CqlTypeId) -> IResult<&[u8], Value> {
219    match type_id {
220        CqlTypeId::Boolean => parse_boolean(input),
221        CqlTypeId::Tinyint => parse_tinyint(input),
222        CqlTypeId::Smallint => parse_smallint(input),
223        CqlTypeId::Int => parse_int(input),
224        CqlTypeId::BigInt => parse_bigint(input),
225        CqlTypeId::Counter => parse_counter(input),
226        CqlTypeId::Float => parse_float(input),
227        CqlTypeId::Double => parse_double(input),
228        CqlTypeId::Ascii | CqlTypeId::Varchar => {
229            // For map/collection contexts, the input is already length-prefixed at the collection level
230            // So we can treat all input as the text content directly
231            let text = String::from_utf8(input.to_vec()).map_err(|_| {
232                nom::Err::Error(nom::error::Error::new(input, nom::error::ErrorKind::Verify))
233            })?;
234            Ok((&[], Value::Text(text)))
235        }
236        CqlTypeId::Blob => {
237            // For blob, use all input as blob data
238            Ok((&[], Value::Blob(input.to_vec())))
239        }
240        CqlTypeId::Uuid | CqlTypeId::Timeuuid => parse_uuid(input),
241        CqlTypeId::Timestamp => parse_timestamp(input),
242        CqlTypeId::Date => parse_date(input),
243        CqlTypeId::Time => parse_time(input),
244        CqlTypeId::Varint => parse_varint(input),
245        CqlTypeId::Decimal => parse_decimal(input),
246        CqlTypeId::Duration => parse_duration(input),
247        CqlTypeId::Inet => parse_inet(input),
248        // Collections and complex types should not be called with raw parsing
249        CqlTypeId::List | CqlTypeId::Set | CqlTypeId::Map | CqlTypeId::Udt | CqlTypeId::Tuple => {
250            parse_cql_value(input, type_id) // Fallback to normal parsing
251        }
252        CqlTypeId::Tombstone => parse_tombstone(input),
253        CqlTypeId::Custom => {
254            // Custom types require additional metadata, return as blob for now
255            Ok((&[], Value::Blob(input.to_vec())))
256        }
257    }
258}

260/// Create empty value for a CQL type
261pub(super) fn create_empty_value_for_cql_type(cql_type: &CqlType) -> Result<Value> {
262    match cql_type {
263        CqlType::Boolean => Ok(Value::Boolean(false)),
264        CqlType::TinyInt => Ok(Value::TinyInt(0)),
265        CqlType::SmallInt => Ok(Value::SmallInt(0)),
266        CqlType::Int => Ok(Value::Integer(0)),
267        CqlType::BigInt => Ok(Value::BigInt(0)),
268        CqlType::Float => Ok(Value::Float32(0.0)),
269        CqlType::Double => Ok(Value::Float(0.0)),
270        CqlType::Text | CqlType::Ascii | CqlType::Varchar => Ok(Value::Text(String::new())),
271        CqlType::Blob => Ok(Value::Blob(Vec::new())),
272        CqlType::Uuid | CqlType::TimeUuid => Ok(Value::Uuid([0; 16])),
273        CqlType::Timestamp => Ok(Value::Timestamp(0)),
274        CqlType::Date => Ok(Value::Timestamp(0)),
275        CqlType::Time => Ok(Value::Timestamp(0)),
276        CqlType::List(_) => Ok(Value::List(Vec::new())),
277        CqlType::Set(_) => Ok(Value::Set(Vec::new())),
278        CqlType::Map(_, _) => Ok(Value::Map(Vec::new())),
279        CqlType::Tuple(_) => Ok(Value::Tuple(Vec::new())),
280        CqlType::Udt(name, _) => Ok(Value::Udt(UdtValue::new(
281            name.clone(),
282            "unknown".to_string(),
283        ))),
284        CqlType::Frozen(inner) => create_empty_value_for_cql_type(inner),
285        _ => Ok(Value::Null),
286    }
287}

344/// Serialize a CQL value to bytes
345pub fn serialize_cql_value(value: &Value) -> Result<Vec<u8>> {
346    let mut result = Vec::new();
347
348    match value {
349        Value::Null => {
350            // Null values are represented with a special length marker
351            result.extend_from_slice(&encode_vint(-1));
352        }
353        Value::Boolean(b) => {
354            result.push(CqlTypeId::Boolean as u8);
355            result.push(if *b { 1 } else { 0 });
356        }
357        Value::Integer(i) => {
358            result.push(CqlTypeId::Int as u8);
359            result.extend_from_slice(&i.to_be_bytes());
360        }
361        Value::BigInt(i) => {
362            result.push(CqlTypeId::BigInt as u8);
363            result.extend_from_slice(&i.to_be_bytes());
364        }
365        Value::Counter(c) => {
366            result.push(CqlTypeId::Counter as u8);
367            result.extend_from_slice(&c.to_be_bytes());
368        }
369        Value::Float(f) => {
370            result.push(CqlTypeId::Double as u8);
371            result.extend_from_slice(&f.to_be_bytes());
372        }
373        Value::Text(s) => {
374            result.push(CqlTypeId::Varchar as u8);
375            result.extend_from_slice(&encode_vint(s.len() as i64));
376            result.extend_from_slice(s.as_bytes());
377        }
378        Value::Blob(b) => {
379            result.push(CqlTypeId::Blob as u8);
380            result.extend_from_slice(&encode_vint(b.len() as i64));
381            result.extend_from_slice(b);
382        }
383        Value::Timestamp(ts) => {
384            result.push(CqlTypeId::Timestamp as u8);
385            let millis = ts / 1000; // Convert microseconds to milliseconds
386            result.extend_from_slice(&millis.to_be_bytes());
387        }
388        Value::Date(days) => {
389            result.push(CqlTypeId::Date as u8);
390            result.extend_from_slice(&days.to_be_bytes());
391        }
392        Value::Time(nanos) => {
393            result.push(CqlTypeId::Time as u8);
394            result.extend_from_slice(&nanos.to_be_bytes());
395        }
396        Value::Uuid(uuid) => {
397            result.push(CqlTypeId::Uuid as u8);
398            result.extend_from_slice(uuid);
399        }
400        Value::Json(json) => {
401            // Store JSON as text
402            let json_str = json.to_string();
403            result.push(CqlTypeId::Varchar as u8);
404            result.extend_from_slice(&encode_vint(json_str.len() as i64));
405            result.extend_from_slice(json_str.as_bytes());
406        }
407        Value::List(list) => {
408            result.push(CqlTypeId::List as u8);
409            result.extend_from_slice(&encode_vint(list.len() as i64));
410
411            // For simplicity, assume all elements are the same type
412            if let Some(first) = list.first() {
413                let element_type = map_value_to_cql_type(first);
414                result.push(element_type as u8);
415
416                for element in list {
417                    if let Value::Null = element {
418                        // Null element: length = -1
419                        result.extend_from_slice(&encode_vint(-1));
420                    } else {
421                        let element_data = serialize_value_without_type_prefix(element)?;
422                        result.extend_from_slice(&encode_vint(element_data.len() as i64));
423                        result.extend_from_slice(&element_data);
424                    }
425                }
426            }
427        }
428        Value::Map(map) => {
429            result.push(CqlTypeId::Map as u8);
430            result.extend_from_slice(&encode_vint(map.len() as i64));
431
432            // For simplicity, assume all keys and values are the same type
433            if let Some((first_key, first_value)) = map.first() {
434                let key_type = map_value_to_cql_type(first_key);
435                let value_type = map_value_to_cql_type(first_value);
436                result.push(key_type as u8);
437                result.push(value_type as u8);
438
439                for (key, value) in map {
440                    // Serialize key with length prefix
441                    if let Value::Null = key {
442                        result.extend_from_slice(&encode_vint(-1));
443                    } else {
444                        let key_data = serialize_value_without_type_prefix(key)?;
445                        result.extend_from_slice(&encode_vint(key_data.len() as i64));
446                        result.extend_from_slice(&key_data);
447                    }
448
449                    // Serialize value with length prefix
450                    if let Value::Null = value {
451                        result.extend_from_slice(&encode_vint(-1));
452                    } else {
453                        let value_data = serialize_value_without_type_prefix(value)?;
454                        result.extend_from_slice(&encode_vint(value_data.len() as i64));
455                        result.extend_from_slice(&value_data);
456                    }
457                }
458            }
459        }
460        Value::TinyInt(i) => {
461            result.push(CqlTypeId::Tinyint as u8);
462            result.push(*i as u8);
463        }
464        Value::Inet(bytes) => {
465            result.push(CqlTypeId::Inet as u8);
466            result.extend_from_slice(&encode_vint(bytes.len() as i64));
467            result.extend_from_slice(bytes);
468        }
469        Value::SmallInt(i) => {
470            result.push(CqlTypeId::Smallint as u8);
471            result.extend_from_slice(&i.to_be_bytes());
472        }
473        Value::Float32(f) => {
474            result.push(CqlTypeId::Float as u8);
475            result.extend_from_slice(&f.to_be_bytes());
476        }
477        Value::Set(set) => {
478            result.push(CqlTypeId::Set as u8);
479            result.extend_from_slice(&encode_vint(set.len() as i64));
480
481            if let Some(first) = set.first() {
482                let element_type = map_value_to_cql_type(first);
483                result.push(element_type as u8);
484
485                for element in set {
486                    if let Value::Null = element {
487                        // Null element: length = -1
488                        result.extend_from_slice(&encode_vint(-1));
489                    } else {
490                        let element_data = serialize_value_without_type_prefix(element)?;
491                        result.extend_from_slice(&encode_vint(element_data.len() as i64));
492                        result.extend_from_slice(&element_data);
493                    }
494                }
495            }
496        }
497        Value::Tuple(tuple) => {
498            result.push(CqlTypeId::Tuple as u8);
499            result.extend_from_slice(&encode_vint(tuple.len() as i64));
500
501            // Serialize type information for each field
502            for element in tuple {
503                let element_type = map_value_to_cql_type(element);
504                result.push(element_type as u8);
505            }
506
507            // Serialize field values with proper length prefixes
508            for element in tuple {
509                if let Value::Null = element {
510                    // Null field: length = -1
511                    result.extend_from_slice(&(-1i32).to_be_bytes());
512                } else {
513                    let element_data = serialize_value_without_type_prefix(element)?;
514                    result.extend_from_slice(&(element_data.len() as i32).to_be_bytes());
515                    result.extend_from_slice(&element_data);
516                }
517            }
518        }
519        Value::Udt(udt) => {
520            result.push(CqlTypeId::Udt as u8);
521
522            // Serialize type name
523            result.extend_from_slice(&encode_vint(udt.type_name.len() as i64));
524            result.extend_from_slice(udt.type_name.as_bytes());
525
526            // Serialize field count
527            result.extend_from_slice(&encode_vint(udt.fields.len() as i64));
528
529            // Serialize field definitions
530            for field in &udt.fields {
531                result.extend_from_slice(&encode_vint(field.name.len() as i64));
532                result.extend_from_slice(field.name.as_bytes());
533
534                // Serialize field type (inferred from value or use blob as fallback)
535                let field_type = match &field.value {
536                    Some(value) => map_value_to_cql_type(value),
537                    None => CqlTypeId::Blob, // Null field, use generic type
538                };
539                result.push(field_type as u8);
540            }
541
542            // Serialize field values
543            for field in &udt.fields {
544                match &field.value {
545                    None => {
546                        // Null field: length = -1
547                        result.extend_from_slice(&(-1i32).to_be_bytes());
548                    }
549                    Some(value) => {
550                        let field_data = serialize_value_without_type_prefix(value)?;
551                        result.extend_from_slice(&(field_data.len() as i32).to_be_bytes());
552                        result.extend_from_slice(&field_data);
553                    }
554                }
555            }
556        }
557        Value::Frozen(boxed_value) => {
558            // For frozen values, just serialize the inner value
559            let inner_bytes = serialize_cql_value(boxed_value)?;
560            result.extend_from_slice(&inner_bytes);
561        }
562        Value::Tombstone(info) => {
563            result.push(CqlTypeId::Tombstone as u8);
564            result.extend_from_slice(&info.deletion_time.to_be_bytes());
565
566            let tombstone_type_byte = match info.tombstone_type {
567                TombstoneType::RowTombstone => 0u8,
568                TombstoneType::CellTombstone => 1u8,
569                TombstoneType::RangeTombstone => 2u8,
570                TombstoneType::TtlExpiration => 3u8,
571                // PartitionTombstone is a delta-scan concept that has no encoding
572                // in the legacy binary format.  Silently mapping it to byte 0
573                // (RowTombstone) would cause a silent lossy round-trip, violating
574                // the no-silent-corruption mandate.  Fail loudly instead.
575                TombstoneType::PartitionTombstone => {
576                    return Err(Error::invalid_operation(
577                        "PartitionTombstone cannot be serialized via the legacy CQL value \
578                         binary format; it is a delta-scan concept only",
579                    ));
580                }
581            };
582            result.push(tombstone_type_byte);
583
584            // Add TTL if present
585            if let Some(ttl) = info.ttl {
586                result.extend_from_slice(&ttl.to_be_bytes());
587            }
588
589            // Add range information for range tombstones
590            if info.tombstone_type == TombstoneType::RangeTombstone {
591                if let (Some(start), Some(end)) = (&info.range_start, &info.range_end) {
592                    result.push(1u8); // Has range marker
593                    result.extend_from_slice(&encode_vint(start.len() as i64));
594                    result.extend_from_slice(start.as_bytes());
595                    result.extend_from_slice(&encode_vint(end.len() as i64));
596                    result.extend_from_slice(end.as_bytes());
597                } else {
598                    result.push(0u8); // No range marker
599                }
600            }
601        }
602        Value::Varint(data) => {
603            result.push(CqlTypeId::Varint as u8);
604            result.extend_from_slice(&encode_vint(data.len() as i64));
605            result.extend_from_slice(data);
606        }
607        Value::Decimal { scale, unscaled } => {
608            result.push(CqlTypeId::Decimal as u8);
609            result.extend_from_slice(&scale.to_be_bytes());
610            result.extend_from_slice(&encode_vint(unscaled.len() as i64));
611            result.extend_from_slice(unscaled);
612        }
613        Value::Duration {
614            months,
615            days,
616            nanos,
617        } => {
618            result.push(CqlTypeId::Duration as u8);
619            result.extend_from_slice(&months.to_be_bytes());
620            result.extend_from_slice(&days.to_be_bytes());
621            result.extend_from_slice(&nanos.to_be_bytes());
622        }
623    }
624
625    Ok(result)
626}

628/// Serialize a CQL value without the type prefix byte
629fn serialize_value_without_type_prefix(value: &Value) -> Result<Vec<u8>> {
630    match value {
631        Value::Null => Ok(vec![]), // Null should be handled at higher level
632        Value::Boolean(b) => Ok(vec![if *b { 1 } else { 0 }]),
633        Value::Integer(i) => Ok(i.to_be_bytes().to_vec()),
634        Value::BigInt(i) => Ok(i.to_be_bytes().to_vec()),
635        Value::Float(f) => Ok(f.to_be_bytes().to_vec()),
636        Value::Text(s) => {
637            // For raw serialization without type prefix, just include the string bytes
638            // The length will be handled by the map format itself
639            Ok(s.as_bytes().to_vec())
640        }
641        Value::Blob(b) => {
642            // For raw serialization without type prefix, just include the blob bytes
643            // The length will be handled by the map format itself
644            Ok(b.to_vec())
645        }
646        Value::Timestamp(ts) => {
647            let millis = ts / 1000; // Convert microseconds to milliseconds
648            Ok(millis.to_be_bytes().to_vec())
649        }
650        Value::Date(days) => Ok(days.to_be_bytes().to_vec()),
651        Value::Time(nanos) => Ok(nanos.to_be_bytes().to_vec()),
652        Value::Uuid(uuid) => Ok(uuid.to_vec()),
653        Value::TinyInt(i) => Ok(vec![*i as u8]),
654        Value::Inet(bytes) => Ok(bytes.clone()),
655        Value::SmallInt(i) => Ok(i.to_be_bytes().to_vec()),
656        Value::Float32(f) => Ok(f.to_be_bytes().to_vec()),
657        // For complex types, fall back to full serialization and strip type byte
658        _ => {
659            let full_bytes = serialize_cql_value(value)?;
660            Ok(full_bytes[1..].to_vec()) // Skip the type byte
661        }
662    }
663}

665fn map_value_to_cql_type(value: &Value) -> CqlTypeId {
666    match value {
667        Value::Null => CqlTypeId::Blob, // Generic fallback
668        Value::Boolean(_) => CqlTypeId::Boolean,
669        Value::Integer(_) => CqlTypeId::Int,
670        Value::BigInt(_) => CqlTypeId::BigInt,
671        Value::Counter(_) => CqlTypeId::Counter,
672        Value::Float(_) => CqlTypeId::Double,
673        Value::Text(_) => CqlTypeId::Varchar,
674        Value::Blob(_) => CqlTypeId::Blob,
675        Value::Timestamp(_) => CqlTypeId::Timestamp,
676        Value::Date(_) => CqlTypeId::Date,
677        Value::Time(_) => CqlTypeId::Time,
678        Value::Uuid(_) => CqlTypeId::Uuid,
679        Value::Json(_) => CqlTypeId::Varchar,
680        Value::TinyInt(_) => CqlTypeId::Tinyint,
681        Value::Inet(_) => CqlTypeId::Inet,
682        Value::SmallInt(_) => CqlTypeId::Smallint,
683        Value::Float32(_) => CqlTypeId::Float,
684        Value::List(_) => CqlTypeId::List,
685        Value::Set(_) => CqlTypeId::Set,
686        Value::Map(_) => CqlTypeId::Map,
687        Value::Tuple(_) => CqlTypeId::Tuple,
688        Value::Udt(_) => CqlTypeId::Udt,
689        Value::Frozen(_) => CqlTypeId::Blob, // Frozen is a wrapper, use blob as fallback
690        Value::Tombstone(_) => CqlTypeId::Tombstone,
691        Value::Varint(_) => CqlTypeId::Varint,
692        Value::Decimal { .. } => CqlTypeId::Decimal,
693        Value::Duration { .. } => CqlTypeId::Duration,
694    }

701/// Assert that all input bytes have been consumed during parsing
702///
703/// This helper enforces complete buffer consumption to prevent silent data
704/// truncation in collection and UDT parsing. Per Issue #61 acceptance criteria,
705/// all parsers must validate full consumption.
706///
707/// # Arguments
708/// * `remaining` - The remaining bytes after parsing
709/// * `context` - Description of parsing context for error messages
710///
711/// # Returns
712/// * `Ok(())` if buffer is fully consumed
713/// * `Err` if bytes remain unconsumed
714#[inline]
715#[allow(dead_code)] // Will be used in collection parsing functions
716pub(crate) fn assert_full_buffer_consumption(remaining: &[u8], context: &str) -> Result<()> {
717    if !remaining.is_empty() {
718        return Err(Error::corruption(format!(
719            "Buffer not fully consumed in {}: {} bytes remaining",
720            context,
721            remaining.len()
722        )));
723    }
724    Ok(())
725}
726
727/// Validate that a parsed element consumed its entire allocated buffer
728///
729/// Used within collection parsing to ensure each element's buffer is fully
730/// consumed before moving to the next element.
731#[inline]
732#[allow(dead_code)] // Will be used in collection parsing functions
733pub(crate) fn validate_element_consumption<'a>(
734    _input: &'a [u8],
735    remaining: &'a [u8],
736    element_index: usize,
737    collection_type: &str,
738) -> Result<()> {
739    if !remaining.is_empty() {
740        return Err(Error::corruption(format!(
741            "{} element {} did not consume full buffer: {} bytes remaining",
742            collection_type,
743            element_index,
744            remaining.len()
745        )));
746    }
747    Ok(())
748}

751// Schema-Aware Collection Parsing (Issue #61)
752// ============================================================================
753
754/// Parse CQL value using schema (no heuristics)
755///
756/// Schema-driven parser that never falls back to heuristics. Used by
757/// schema-aware collection parsers to ensure deterministic decoding.
758pub(super) fn parse_cql_value_with_schema<'a>(
759    input: &'a [u8],
760    schema: &CqlType,
761) -> IResult<&'a [u8], Value> {
762    match schema {
763        CqlType::Boolean => parse_boolean(input),
764        CqlType::TinyInt => parse_tinyint(input),
765        CqlType::SmallInt => parse_smallint(input),
766        CqlType::Int => parse_int(input),
767        CqlType::BigInt => parse_bigint(input),
768        CqlType::Counter => parse_counter(input),
769        CqlType::Float => parse_float(input),
770        CqlType::Double => parse_double(input),
771        CqlType::Text | CqlType::Ascii | CqlType::Varchar => parse_text(input),
772        CqlType::Blob => parse_blob(input),
773        CqlType::Uuid | CqlType::TimeUuid => parse_uuid(input),
774        CqlType::Timestamp => parse_timestamp(input),
775        CqlType::Date => parse_date(input),
776        CqlType::Time => parse_time(input),
777        CqlType::Duration => parse_duration(input),
778        CqlType::Inet => parse_inet(input),
779        CqlType::Decimal => parse_decimal(input),
780        CqlType::Varint => parse_varint(input),
781        CqlType::List(element_type) => parse_list_with_schema(input, element_type),
782        CqlType::Set(element_type) => {
783            let (remaining, Value::List(elements)) = parse_list_with_schema(input, element_type)?
784            else {
785                return Err(nom::Err::Error(nom::error::Error::new(
786                    input,
787                    nom::error::ErrorKind::Verify,
788                )));
789            };
790            Ok((remaining, Value::Set(elements)))
791        }
792        CqlType::Map(key_type, value_type) => parse_map_with_schema(input, key_type, value_type),
793        CqlType::Tuple(_) => parse_tuple(input),
794        CqlType::Udt(_, _) => parse_udt(input),
795        CqlType::Frozen(inner) => parse_cql_value_with_schema(input, inner),
796        CqlType::Custom(_) => {
797            // Custom types require additional metadata, parse as blob
798            parse_blob(input)
799        }
800    }
801}
