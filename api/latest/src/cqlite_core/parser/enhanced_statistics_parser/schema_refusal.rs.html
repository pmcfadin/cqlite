1//! The typed failure channel of the post-EncodingStats SerializationHeader
2//! schema decoder.
3//!
4//! `parse_serialization_header_schema` can fail for two reasons that a caller
5//! must NOT treat alike, and a `nom::Err` cannot tell them apart:
6//!
7//! * The bytes at the TOC-anchored offset do not hold a readable header at all —
8//!   truncated, mispositioned, an implausible declared length, a non-UTF-8 type
9//!   name. Retrying with the marker-search decoder is legitimate: it is a second
10//!   attempt to find WHERE the header is.
11//! * The bytes decoded fine and DECLARE a type Cassandra cannot have written (a
12//!   `frozen<scalar>`; `CQL3Type.Raw::freeze()` throws for every
13//!   non-collection/tuple/UDT/vector — cassandra-5.0.8
14//!   `src/java/org/apache/cassandra/cql3/CQL3Type.java:647-651`). Nothing about
15//!   WHERE we are looking is in doubt, so a second decoder cannot improve the
16//!   answer — it can only replace a correct refusal with a marker-search guess,
17//!   which is both fail-open and a no-heuristics violation (#28).
18//!
19//! Issue #4104 (roborev job 116): the second case used to reach the caller as a
20//! bare `nom::Err` and was retried through the heuristic fallback, so a header
21//! the frozen-scalar gate had correctly refused was accepted anyway — while the
22//! sibling KEY-type gate in the same function failed closed. This enum is what
23//! makes the two gates agree.
24//!
25//! # Its reach, after the #4158 review round
26//!
27//! It is no longer only the anchored decoder's channel. Job 119 found the same
28//! fail-open in the MARKER-SEARCH decoders — a refusal there became an ordinary
29//! failed candidate, so the search continued to its empty-schema success — so
30//! `serialization_header::{mod, sequential}` and `parse_regular_columns` and the
31//! ASCII fallback all carry this type now, and `Refused` terminates the search
32//! wherever it arises. Blocker A extended it in the other direction: the carried
33//! `Error` is propagated out through `parse_nb_format_statistics_data_with_toc`
34//! and the `_detailed` entry points to `StatisticsReader::open`, so the refusal
35//! the user sees is the message the gate wrote.
36
37use crate::error::Error;

39/// Why [`super::serialization_header::parse_serialization_header_schema`] did not
40/// return a schema.
41///
42/// Callers MUST match on the variant: only [`Self::Structural`] may be retried
43/// with the marker-search fallback.
44#[derive(Debug)]
45pub(super) enum HeaderSchemaError<'a> {
46    /// The bytes at this offset are not a readable header (truncated,
47    /// mispositioned, implausible length, non-UTF-8). Position is in doubt, so
48    /// the marker-search fallback may legitimately be tried.
49    Structural(nom::Err<nom::error::Error<&'a [u8]>>),
50    /// The header decoded, and what it declares is not writable by Cassandra.
51    /// FAIL-CLOSED: never retried with a heuristic. The carried [`Error`] is
52    /// PROPAGATED to the user (#4158 review, blocker A) — it is the only text that
53    /// names the refused column, the refused type and the Cassandra citation, and
54    /// it used to be logged and then dropped, so the refusal reached the caller as
55    /// a bare `ErrorKind::Verify` that read like data corruption. The DECISION is
56    /// still taken on the VARIANT, never on the message text.
57    Refused(Error),
58}
59
60impl<'a> From<nom::Err<nom::error::Error<&'a [u8]>>> for HeaderSchemaError<'a> {
61    /// Every nom failure the decoder propagates with `?` is structural by
62    /// construction: it comes from a length/`take`/VInt primitive, none of which
63    /// can express a semantic judgement about a declared type.
64    fn from(err: nom::Err<nom::error::Error<&'a [u8]>>) -> Self {
65        Self::Structural(err)
66    }
67}
68
69impl<'a> HeaderSchemaError<'a> {
70    /// A structural refusal raised by the decoder's own sanity checks at
71    /// `input` (implausible declared count/length, non-UTF-8 name or type).
72    pub(super) fn structural_at(input: &'a [u8]) -> Self {
73        Self::Structural(nom::Err::Error(nom::error::Error::new(
74            input,
75            nom::error::ErrorKind::Verify,
76        )))
77    }
78
79    /// A semantic refusal of `detail`, which already carries the Cassandra
80    /// citation from the frozen-scalar gate.
81    pub(super) fn refused(detail: impl Into<String>) -> Self {
82        Self::Refused(Error::schema(detail.into()))
83    }
84}
