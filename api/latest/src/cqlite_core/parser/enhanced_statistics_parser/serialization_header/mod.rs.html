30/// Parse SerializationHeader from Statistics.db (Issue #163)
31///
32/// This function locates and parses the complete SerializationHeader section including:
33/// 1. Partition key types
34/// 2. Clustering key types
35/// 3. Regular column definitions
36///
37/// Returns: (partition_key_types, clustering_key_types, regular_columns)
38///
39/// # A SEMANTIC REFUSAL STOPS THE SEARCH — ONCE THE CANDIDATE IS CONFIRMED (#4104)
40///
41/// The search tries many candidate offsets, so a failed candidate is normally just
42/// "not here — try the next one", and that is right for every STRUCTURAL failure:
43/// where the header lives is exactly what is in doubt.
44///
45/// A semantic refusal (`FrozenType(<scalar>)`, which no Cassandra writer can have
46/// recorded) is a different answer, and it needed THREE attempts to place
47/// correctly:
48///
49/// 1. Originally it was demoted to an ordinary failed candidate, so the search
50///    continued and ended at this function's own
51///    `Ok((input, (empty, empty, empty)))` no-header success, which the caller
52///    accepts as a schema-less header. FAIL-OPEN — every individual gate refused
53///    and the net outcome was still `Ok` (roborev job 119).
54/// 2. Then it was propagated immediately from wherever it arose. OVER-REFUSAL —
55///    this search walks ARBITRARY bytes and the decoders accept any UTF-8 string
56///    as the key type, so an earlier FALSE candidate carrying a plausible
57///    `FrozenType(<scalar>)` run aborted an otherwise VALID SSTable before the
58///    real header was reached (roborev job 120).
59/// 3. Now: a refusal is authoritative only once the candidate is CONFIRMED to be
60///    a SerializationHeader — a complete decode of its own declared column lists
61///    plus marshal class SPELLINGS for its key and clustering types, judged by
62///    [`candidate_confirmed_as_header`] from Cassandra's writer. Confirmed ⇒
63///    propagate (never fall through to another candidate or to the no-header
64///    success). Unconfirmed ⇒ it was a guess that did not pan out, so the search
65///    continues, exactly as for a structural failure.
66///
67/// On the TOC-ANCHORED path the location is known, so the refusal is authoritative
68/// on arrival and is propagated there without any of this — see
69/// `parse_minimal_encoding_stats`.
70pub(super) fn parse_serialization_header(
71    input: &[u8],
72) -> Result<(&[u8], SerializationHeaderResult), HeaderSchemaError<'_>> {
73    tracing::debug!(
74        "Searching for SerializationHeader in {} bytes (max search: 8KB)",
75        input.len()
76    );
77
78    // Log input buffer state at function entry
79    let preview_len = std::cmp::min(64, input.len());
80    let preview_hex: String = input[..preview_len]
81        .iter()
82        .map(|b| format!("{:02x}", b))
83        .collect::<Vec<_>>()
84        .join(" ");
85    tracing::debug!(
86        "Input buffer size: {} bytes, first 64 bytes: {}",
87        input.len(),
88        preview_hex
89    );
90
91    // Search for SerializationHeader start marker: VInt followed by 0x00 0x00 and '(' character
92    // This marks the beginning of the partition key type descriptor
93    let mut search_offset = 0;
94
95    // Search for SerializationHeader by finding "org.apache.cassandra.db.marshal" string
96    // and working backwards to find the 0x00 0x00 marker
97    // Format: [VInt unknown] [0x00 0x00] [VInt partition_type_len] [partition_type_string]
98    let marshal_pattern = b"org.apache.cassandra.db.marshal";
99
100    while search_offset + marshal_pattern.len() < input.len() && search_offset < 8192 {
101        if &input[search_offset..search_offset + marshal_pattern.len()] == marshal_pattern {
102            let context_start = search_offset.saturating_sub(10);
103            let context_end = (search_offset + 50).min(input.len());
104            tracing::debug!(
105                "Found 'org.apache.cassandra.db.marshal' at offset {}, context (offset-10 to offset+50): {:02x?}",
106                search_offset,
107                &input[context_start..context_end]
108            );
109
110            // Issue #216 fix: Look for the pattern [prev_zero] [pk_type_len] "org.apache..."
111            // where pk_type_len is a valid VInt length (0x01-0x7F for single byte, or multi-byte VInt)
112            // The prev_zero is typically the last byte of EncodingStats (minTTL=0) or another zero field.
113            //
114            // We need to find the START of the partition key type length, which is:
115            // - 1 byte before "org.apache..." for single-byte lengths (0x28 = 40 bytes for UUIDType)
116            // - 2 bytes before for two-byte VInt lengths (0x80 0xXX)
117
118            for lookback in 1..=15 {
119                if search_offset < lookback {
120                    break;
121                }
122                let type_len_offset = search_offset - lookback;
123
124                // Check if this could be a valid pk_type_len
125                // For single-byte VInt: values 0x01-0x7F
126                // For two-byte VInt: first byte has high bit set (0x80-0xFF)
127                let first_byte = input[type_len_offset];
128
129                // Common partition key type lengths:
130                // - UUIDType: 40 bytes (0x28)
131                // - UTF8Type: 40 bytes (0x28)
132                // - Int32Type: 41 bytes (0x29)
133                // - TimestampType: 45 bytes (0x2D)
134                // - CompositeType: ~80-150 bytes (0x50-0x96 or multi-byte VInt)
135
136                // Single-byte VInt: 0x20-0x7F are reasonable pk_type lengths (32-127 bytes)
137                let is_valid_single_byte_len = (0x20..=0x7F).contains(&first_byte);
138
139                // Two-byte VInt: 0x80-0xBF with continuation
140                let is_multi_byte_vint = first_byte >= 0x80;
141
142                if is_valid_single_byte_len || is_multi_byte_vint {
143                    // Try parsing from this offset using sequential parser
144                    let candidate = &input[type_len_offset..];
145                    match parse_serialization_header_sequential(candidate, SemanticGate::Enforce) {
146                        // A semantic refusal fails CLOSED — but only once these
147                        // bytes are CONFIRMED to be a header rather than a
148                        // candidate that briefly looked like one. The confirmation
149                        // re-decodes the SAME bytes with the refusal deferred, so
150                        // the completeness the refusal cut short can be judged
151                        // (#4104, roborev job 120). Unconfirmed falls through to
152                        // the next candidate offset, like a structural failure.
153                        Err(refused @ HeaderSchemaError::Refused(_)) => {
154                            if candidate_confirmed_as_header(parse_serialization_header_sequential(
155                                candidate,
156                                SemanticGate::Survey,
157                            )) {
158                                return Err(refused);
159                            }
160                        }
161                        Ok((remaining, (pk_types, ck_types, cols))) => {
162                            // Validate: partition key type should contain expected substring
163                            if !pk_types.is_empty()
164                                && pk_types[0].contains("org.apache.cassandra.db.marshal")
165                            {
166                                tracing::debug!(
167                                    "Successfully parsed SerializationHeader at offset {} (lookback: {}): pk_type={}",
168                                    type_len_offset,
169                                    lookback,
170                                    pk_types[0]
171                                );
172                                return Ok((remaining, (pk_types, ck_types, cols)));
173                            }
174                        }
175                        Err(HeaderSchemaError::Structural(_)) => {}
176                    }
177                }
178
179                // Also try the legacy 0x00 0x00 marker for backward compatibility
180                if type_len_offset > 0 {
181                    let prev_offset = type_len_offset - 1;
182                    if input[prev_offset] == 0x00 && input[type_len_offset] == 0x00 {
183                        let candidate = &input[prev_offset..];
184                        match parse_serialization_header_at_offset(candidate, SemanticGate::Enforce)
185                        {
186                            // Same confirm-then-fail-closed rule as the sequential
187                            // candidate above: this offset is a guess too.
188                            Err(refused @ HeaderSchemaError::Refused(_)) => {
189                                if candidate_confirmed_as_header(
190                                    parse_serialization_header_at_offset(
191                                        candidate,
192                                        SemanticGate::Survey,
193                                    ),
194                                ) {
195                                    return Err(refused);
196                                }
197                            }
198                            Ok(parsed) => {
199                                tracing::debug!(
200                                    "Successfully parsed SerializationHeader at legacy marker offset {}",
201                                    prev_offset
202                                );
203                                return Ok(parsed);
204                            }
205                            Err(HeaderSchemaError::Structural(_)) => {}
206                        }
207                    }
208                }
209            }
210        }
211        search_offset += 1;
212    }
213
214    tracing::debug!(
215        "Search completed: searched {} bytes, no partition key type found",
216        search_offset
217    );
218
219    // Partition key type not found - try to find regular columns directly
220    // This handles files where SerializationHeader contains only regular columns
221    tracing::debug!("Attempting to parse regular columns without partition key metadata");
222    let (remaining, (partition_keys, columns)) = parse_regular_columns(input)?;
223
224    if !columns.is_empty() {
225        tracing::debug!(
226            "Successfully parsed {} regular columns, {} partition keys via backtracking",
227            columns.len(),
228            partition_keys.len()
229        );
230        return Ok((remaining, (partition_keys, Vec::new(), columns)));
231    }
232
233    // Nothing found - return empty results
234    tracing::warn!(
235        "Failed to locate SerializationHeader or regular columns: searched {} bytes",
236        search_offset
237    );
238
239    if let Some((pk_types, ck_types, cols)) = fallback_parse_serialization_header_ascii(input)? {
240        tracing::debug!(
241            "ASCII fallback extracted SerializationHeader: {} partition keys, {} clustering keys, {} regular columns",
242            pk_types.len(),
243            ck_types.len(),
244            cols.len()
245        );
246        return Ok((input, (pk_types, ck_types, cols)));
247    }
248
249    Ok((input, (Vec::new(), Vec::new(), Vec::new())))
250}

252/// ASCII fallback parser for SerializationHeader when structured parsing fails.
253///
254/// `Ok(None)` = nothing recognisable here (the caller falls through to its
255/// no-header success). `Err(Refused)` = a type Cassandra cannot have written was
256/// found, which fails CLOSED — it must not degrade to `Ok(None)` and thence to an
257/// accepted empty schema (#4104, roborev job 119).
258fn fallback_parse_serialization_header_ascii(
259    input: &[u8],
260) -> Result<Option<SerializationHeaderResult>, HeaderSchemaError<'_>> {
261    // Helper to find subsequence
262    fn find_subsequence(haystack: &[u8], needle: &[u8]) -> Option<usize> {
263        haystack
264            .windows(needle.len())
265            .position(|window| window == needle)
266    }
267
268    let mut partition_types = Vec::new();
269    let mut clustering_types = Vec::new();
270    let mut columns = Vec::new();
271
272    // Extract partition key types from CompositeType(...)
273    if let Some(comp_idx) = find_subsequence(input, b"CompositeType(") {
274        let start = comp_idx + "CompositeType(".len();
275        let mut end = start;
276        while end < input.len() && input[end] != b')' {
277            end += 1;
278        }
279        if end <= input.len() {
280            if let Ok(inner) = std::str::from_utf8(&input[start..end]) {
281                partition_types = inner
282                    .split(',')
283                    .map(|s| s.trim().to_string())
284                    .filter(|s| !s.is_empty())
285                    .collect();
286            }
287
288            // Attempt to extract clustering types immediately after the composite type string
289            let mut cursor = end + 1;
290            while cursor < input.len() && input[cursor] < 0x20 {
291                cursor += 1;
292            }
293            if cursor < input.len() && input[cursor] == b'(' {
294                cursor += 1;
295                let mut cluster_end = cursor;
296                while cluster_end < input.len() && input[cluster_end] >= 0x20 {
297                    cluster_end += 1;
298                }
299                if cluster_end > cursor {
300                    if let Ok(cluster_str) = std::str::from_utf8(&input[cursor..cluster_end]) {
301                        if cluster_str.contains("org.apache.cassandra.db.marshal") {
302                            clustering_types = cluster_str
303                                .split(',')
304                                .map(|s| s.trim().to_string())
305                                .filter(|s| !s.is_empty())
306                                .collect();
307                        }
308                    }
309                }
310                // Set scan position for column parsing after clustering types/control bytes
311                let mut scan_start = cluster_end;
312                while scan_start < input.len() && input[scan_start] < 0x20 {
313                    scan_start += 1;
314                }
315
316                // Parse regular columns using [len][name][type] pattern with control-byte delimiters
317                let mut idx = scan_start;
318                while idx < input.len() {
319                    let name_len = input[idx] as usize;
320                    if name_len == 0 || name_len > 64 {
321                        idx += 1;
322                        continue;
323                    }
324
325                    let name_start = idx + 1;
326                    let name_end = name_start + name_len;
327                    if name_end > input.len() {
328                        break;
329                    }
330
331                    let name_bytes = &input[name_start..name_end];
332                    if !name_bytes
333                        .iter()
334                        .all(|b| b.is_ascii_alphanumeric() || *b == b'_')
335                    {
336                        idx += 1;
337                        continue;
338                    }
339
340                    if name_end >= input.len() || input[name_end] != b'(' {
341                        idx += 1;
342                        continue;
343                    }
344
345                    let type_start = name_end + 1;
346                    let mut type_end = type_start;
347                    while type_end < input.len() && input[type_end] >= 0x20 {
348                        type_end += 1;
349                    }
350
351                    if type_end == type_start {
352                        idx += 1;
353                        continue;
354                    }
355
356                    let type_bytes = &input[type_start..type_end];
357                    if !type_bytes.windows(10).any(|w| w == b"org.apach") {
358                        idx += 1;
359                        continue;
360                    }
361
362                    let column_name = match std::str::from_utf8(name_bytes) {
363                        Ok(s) => s.to_string(),
364                        Err(_) => {
365                            idx += 1;
366                            continue;
367                        }
368                    };
369
370                    let internal_type = match std::str::from_utf8(type_bytes) {
371                        Ok(s) => s.trim().to_string(),
372                        Err(_) => {
373                            idx += 1;
374                            continue;
375                        }
376                    };
377
378                    // Gate 2 of #4104: a refused type fails the WHOLE header rather
379                    // than silently dropping the column, which would hide the refusal
380                    // behind a schema that merely looks short — and it propagates as
381                    // a REFUSAL, not as `None`, which the caller would accept.
382                    let cql_type = convert_marshal_type_to_cql_checked(&internal_type)
383                        .map_err(HeaderSchemaError::Refused)?;
384                    columns.push(ColumnInfo {
385                        name: column_name,
386                        column_type: cql_type,
387                        is_primary_key: false,
388                        key_position: None,
389                        is_static: false,
390                        is_clustering: false,
391                        clustering_reversed: false,
392                    });
393
394                    // Advance past control bytes to next potential column entry
395                    idx = type_end;
396                    while idx < input.len() && input[idx] < 0x20 {
397                        idx += 1;
398                    }
399                }
400            }
401        }
402    }
403
404    if partition_types.is_empty() && columns.is_empty() {
405        return Ok(None);
406    }
407
408    Ok(Some((partition_types, clustering_types, columns)))
409}
