1//! Cassandra marshal-type string handling and `ColumnInfo` construction.
2//!
3//! These helpers convert Cassandra internal `org.apache.cassandra.db.marshal.*`
4//! comparator/type strings (as they appear in the SerializationHeader of
5//! Statistics.db) into CQL type names, and build the partition/clustering
6//! `ColumnInfo` vectors used by schema discovery.

10/// Extract inner type from parameterized type string with proper parenthesis matching
11///
12/// Given a string that starts AFTER the opening parenthesis of a wrapper type,
13/// returns the content up to (but not including) the matching closing parenthesis.
14///
15/// Example: For input "ListType(Int32Type))" (after stripping "FrozenType("),
16/// returns Some("ListType(Int32Type)") - the content before the MATCHING close paren.
17fn extract_inner_type(type_with_close_paren: &str) -> Option<&str> {
18    let mut depth = 1; // We're already inside one opening paren (the wrapper type)
19    for (idx, ch) in type_with_close_paren.char_indices() {
20        match ch {
21            '(' => depth += 1,
22            ')' => {
23                depth -= 1;
24                if depth == 0 {
25                    // Return None if extracted string is empty (malformed input like ")")
26                    if idx == 0 {
27                        return None;
28                    }
29                    return Some(&type_with_close_paren[..idx]);
30                }
31            }
32            _ => {}
33        }
34    }
35    None // Unmatched parentheses
36}

38/// Split a type argument list on top-level commas, ignoring nested parentheses
39fn split_type_arguments(input: &str) -> Vec<&str> {
40    let mut args = Vec::new();
41    let mut depth = 0;
42    let mut start = 0;
43    for (idx, ch) in input.char_indices() {
44        match ch {
45            '(' => depth += 1,
46            ')' => {
47                if depth > 0 {
48                    depth -= 1;
49                } else {
50                    tracing::warn!(
51                        "Unmatched closing parenthesis at position {} in type arguments: '{}'",
52                        idx,
53                        input
54                    );
55                }
56            }
57            ',' if depth == 0 => {
58                let part = input[start..idx].trim();
59                if !part.is_empty() {
60                    args.push(part);
61                }
62                start = idx + ch.len_utf8();
63            }
64            _ => {}
65        }
66    }
67
68    let tail = input[start..].trim();
69    if !tail.is_empty() {
70        args.push(tail);
71    }
72
73    args
74}

76/// Strict variant of [`split_type_arguments`] for `TupleType(..)` (issue #4627,
77/// roborev round 2): every slot must be non-empty. The lenient splitter drops empty
78/// components, so `TupleType(Int32Type,,UTF8Type)` would silently change arity.
79/// Cassandra's WRITER (`TypeParser.stringifyTypeParameters`, pinned 5.0.8) joins
80/// exactly the printed element types with one `,` and never emits an empty slot, so
81/// refusing a leading, trailing, repeated or whitespace-only slot cannot reject a
82/// form Cassandra wrote. (Its reader, `getTypeParameters`/`skipBlankAndComma`, is
83/// more tolerant, which is why strictness is chosen here rather than inherited.)
84/// `Map`/`UDT` callers keep the lenient splitter: they are untouched by this issue.
85fn split_tuple_arguments(input: &str) -> crate::error::Result<Vec<&str>> {
86    let mut slots: Vec<&str> = Vec::new();
87    let mut depth = 0usize;
88    let mut start = 0;
89    let mut bounds: Vec<usize> = Vec::new();
90    for (idx, ch) in input.char_indices() {
91        match ch {
92            '(' => depth += 1,
93            ')' => depth = depth.saturating_sub(1),
94            ',' if depth == 0 => bounds.push(idx),
95            _ => {}
96        }
97    }
98    bounds.push(input.len());
99    for end in bounds {
100        let slot = input[start..end].trim();
101        if slot.is_empty() {
102            return Err(crate::error::Error::schema(format!(
103                "malformed TupleType arguments '{input}': empty element slot \
104                 (leading, trailing or repeated comma)"
105            )));
106        }
107        slots.push(slot);
108        start = end + 1;
109    }
110    Ok(slots)
111}

113/// Detect whether a clustering comparator class denotes descending order.
114///
115/// Issue #759: Cassandra encodes a DESC clustering column by wrapping its
116/// comparator class in `org.apache.cassandra.db.marshal.ReversedType(...)`.
117/// This is the documented, authoritative signal for descending order
118/// (definitive guide Ch.7 / Appendix B) — no heuristics. We mirror the
119/// leading-paren stripping that `convert_marshal_type_to_cql` performs (the
120/// header sometimes prefixes types with `(`), then check for the `ReversedType(`
121/// prefix with or without the marshal package namespace.
122pub(super) fn is_reversed_comparator(marshal_type: &str) -> bool {
123    let mut value = marshal_type.trim();
124    // Strip the optional leading wrapping paren(s) the header adds to the
125    // top-level type (matches `convert_marshal_type_to_cql`'s normalization).
126    while value.starts_with('(') && value.ends_with(')') && value.len() > 2 {
127        value = value[1..value.len() - 1].trim();
128    }
129    // Some accepted header encodings prefix the comparator list with a `[`
130    // (e.g. `[org.apache.cassandra.db.marshal.ReversedType(...)`); strip a
131    // leading bracket so the ReversedType detection matches that form too
132    // (roborev job 43).
133    let value = value.trim_start_matches('[').trim_start();
134    value.starts_with("org.apache.cassandra.db.marshal.ReversedType(")
135        || value.starts_with("ReversedType(")
136}

138/// ═══ GATE 2 OF 2: the `Statistics.db` SerializationHeader type parser ═══
139///
140/// [`convert_marshal_type_to_cql`] with the `frozen<scalar>` refusal applied, and
141/// THE ONLY WAY OUT OF THIS MODULE for a header type string. The converter itself
142/// is private and infallible so that it can recurse; the gate sits here, on the
143/// whole string, exactly once — which also means it descends into a
144/// `UserType(…)`-bearing string that the converter returns verbatim.
145///
146/// A `FrozenType(<scalar>)` header is not something a Cassandra writer can emit:
147/// the header records `column.type`, and `CQL3Type.Raw::freeze()` throws for every
148/// non-collection/tuple/UDT, so no such column can have been declared. Refusing it
149/// fail-closed is the no-heuristics answer (#28) — see [`validate_marshal_frozen`]
150/// for the citation, the override set, and the corpus census. Issue #4104.
151pub(super) fn convert_marshal_type_to_cql_checked(
152    marshal_type: &str,
153) -> crate::error::Result<String> {
154    validate_marshal_frozen(marshal_type)?;
155    validate_marshal_tuples(marshal_type)?;
156    Ok(convert_marshal_type_to_cql(marshal_type))
157}

159/// Fail-closed structural check of every `TupleType(..)` claim in a header type
160/// (issue #4627, roborev round 1). A string that claims to be a tuple but whose
161/// parentheses do not balance, or that has text after the matching `)`, is refused
162/// with a schema error: the converter is infallible and would otherwise fall into
163/// the scalar fallback and pick a framing (#28). Recurses through the wrappers and
164/// collections the converter recurses through; `UserType(..)` is preserved verbatim
165/// by the converter and is not descended into here.
166fn validate_marshal_tuples(marshal_type: &str) -> crate::error::Result<()> {
167    const TUPLE: [&str; 2] = ["org.apache.cassandra.db.marshal.TupleType(", "TupleType("];
168    const WRAP: [&str; 10] = [
169        "org.apache.cassandra.db.marshal.ReversedType(",
170        "ReversedType(",
171        "org.apache.cassandra.db.marshal.FrozenType(",
172        "FrozenType(",
173        "org.apache.cassandra.db.marshal.ListType(",
174        "ListType(",
175        "org.apache.cassandra.db.marshal.SetType(",
176        "SetType(",
177        "org.apache.cassandra.db.marshal.MapType(",
178        "MapType(",
179    ];
180    let mut value = marshal_type.trim().trim_start_matches('[').trim_start();
181    while value.starts_with('(') && value.ends_with(')') && value.len() > 2 {
182        value = value[1..value.len() - 1].trim();
183    }
184    let malformed = |why: &str| {
185        crate::error::Error::schema(format!(
186            "malformed TupleType in SerializationHeader type '{marshal_type}': {why}"
187        ))
188    };
189    for prefix in TUPLE {
190        if let Some(rest) = value.strip_prefix(prefix) {
191            let inner =
192                extract_inner_type(rest).ok_or_else(|| malformed("unbalanced parentheses"))?;
193            if rest.len() != inner.len() + 1 {
194                return Err(malformed("trailing text after the closing parenthesis"));
195            }
196            return split_tuple_arguments(inner)?
197                .into_iter()
198                .try_for_each(validate_marshal_tuples);
199        }
200    }
201    // Same container set the converter recurses through: Vector (element) ...
202    match crate::schema::vector_type::marshal_vector_kind(value).into_args(value) {
203        Ok(Some(args)) => return validate_marshal_tuples(args.element),
204        Ok(None) => {}
205        // A malformed vector is spelled fail-closed by the converter itself; only a
206        // tuple claim inside it is refused here (it can no longer be located).
207        Err(_) if value.contains("TupleType(") => {
208            return Err(malformed(
209                "unterminated or malformed VectorType around a tuple",
210            ));
211        }
212        Err(_) => {}
213    }
214    // ... and the single-parameter / map wrappers. A recognised wrapper whose
215    // parameter list does not close, or that has text after the matching `)`,
216    // fails closed: it can hide a tuple claim.
217    for prefix in WRAP {
218        if let Some(rest) = value.strip_prefix(prefix) {
219            let inner =
220                extract_inner_type(rest).ok_or_else(|| malformed("unbalanced parentheses"))?;
221            if rest.len() != inner.len() + 1 {
222                return Err(malformed("trailing text after the closing parenthesis"));
223            }
224            return split_type_arguments(inner)
225                .into_iter()
226                .try_for_each(validate_marshal_tuples);
227        }
228    }
229    Ok(())
230}

240/// Convert Cassandra internal marshal type to CQL type name.
241///
242/// PRIVATE and infallible by design: it recurses, so gating here would re-scan the
243/// string at every level. Every caller outside this module goes through
244/// [`convert_marshal_type_to_cql_checked`], which is what makes gate 2 unmissable.
245fn convert_marshal_type_to_cql(marshal_type: &str) -> String {
246    fn strip_wrapping_parens(mut value: &str) -> &str {
247        loop {
248            // Also strip a leading structural `[` the header may prefix the
249            // comparator list with (e.g. `[org...ReversedType(...)`), so the
250            // wrapper-prefix checks below match the bracketed form too and the
251            // inner CQL type is derived correctly (roborev job 48). This mirrors
252            // the same normalization in `is_reversed_comparator`.
253            let trimmed = value.trim().trim_start_matches('[').trim_start();
254            if trimmed.starts_with('(') && trimmed.ends_with(')') && trimmed.len() > 2 {
255                value = &trimmed[1..trimmed.len() - 1];
256            } else {
257                return trimmed;
258            }
259        }
260    }
261
262    fn strip_namespace(type_name: &str) -> &str {
263        type_name.rsplit('.').next().unwrap_or(type_name)
264    }
265
266    fn strip_type_suffix(name: &str) -> &str {
267        name.trim_end_matches("Type")
268    }
269
270    let mut cleaned = strip_wrapping_parens(marshal_type);
271
272    // `TupleType(T1,T2,...)` -> CQL `tuple<t1, t2, ...>` (issue #4627). Cassandra
273    // prints a tuple as `getClass().getName() + stringifyTypeParameters(types,
274    // true)` (`TupleType.java:557-560`), never wrapped in `FrozenType` because a
275    // tuple is implicitly frozen. Handled BEFORE the `UserType` preservation below
276    // so a tuple with a UDT element converts element by element (the UDT element
277    // keeps its definition verbatim via the recursion). A malformed tuple never
278    // reaches here: `validate_marshal_tuples` (gate 2) refuses it first.
279    for prefix in ["org.apache.cassandra.db.marshal.TupleType(", "TupleType("] {
280        if let Some(params_with_close) = cleaned.strip_prefix(prefix) {
281            if let Some(inner) = extract_inner_type(params_with_close) {
282                let elements: Vec<String> = split_tuple_arguments(inner)
283                    .unwrap_or_default()
284                    .into_iter()
285                    .map(convert_marshal_type_to_cql)
286                    .collect();
287                return format!("tuple<{}>", elements.join(", "));
288            }
289        }
290    }
291
292    // Special case: Preserve UserType definitions unchanged
293    // UserType contains critical metadata (keyspace, type name, field definitions) that must
294    // reach the parser intact. Converting it to a simplified CQL type would lose this information.
295    if cleaned.contains("org.apache.cassandra.db.marshal.UserType(") {
296        return marshal_type.to_string();
297    }
298
299    // Normalize known wrappers by recursively converting inner types
300    // Use extract_inner_type() for proper parenthesis matching (fixes nested types)
301    for prefix in [
302        "org.apache.cassandra.db.marshal.ReversedType(",
303        "ReversedType(",
304    ] {
305        if let Some(params_with_close) = cleaned.strip_prefix(prefix) {
306            if let Some(inner) = extract_inner_type(params_with_close) {
307                return convert_marshal_type_to_cql(inner);
308            }
309        }
310    }
311
312    for prefix in ["org.apache.cassandra.db.marshal.FrozenType(", "FrozenType("] {
313        if let Some(params_with_close) = cleaned.strip_prefix(prefix) {
314            if let Some(inner) = extract_inner_type(params_with_close) {
315                return format!("frozen<{}>", convert_marshal_type_to_cql(inner));
316            }
317        }
318    }
319
320    for prefix in ["org.apache.cassandra.db.marshal.ListType(", "ListType("] {
321        if let Some(params_with_close) = cleaned.strip_prefix(prefix) {
322            if let Some(inner) = extract_inner_type(params_with_close) {
323                return format!("list<{}>", convert_marshal_type_to_cql(inner));
324            }
325        }
326    }
327
328    for prefix in ["org.apache.cassandra.db.marshal.SetType(", "SetType("] {
329        if let Some(params_with_close) = cleaned.strip_prefix(prefix) {
330            if let Some(inner) = extract_inner_type(params_with_close) {
331                return format!("set<{}>", convert_marshal_type_to_cql(inner));
332            }
333        }
334    }
335
336    // Cassandra 5.0 `VectorType(element , n)` -> CQL `vector<element, n>`
337    // (issue #4114). THIS is the conversion the measured `read-sstable` path uses,
338    // and before it the vector fell through to the `other => other.to_lowercase()`
339    // fallback at the bottom of this function, which SILENTLY produced the junk
340    // type string `floattype , 3`. The value was then decoded as a
341    // vint-length-prefixed blob and, on Cassandra-written bytes, could be returned
342    // WRONG with exit 0 (see `.drive-issue-4114/silent-misdecode-measurement.md`).
343    //
344    // The dimension must survive this conversion: a fixed-width vector value
345    // carries no length prefix and no element count, so `n` is the ONLY thing that
346    // makes it parseable (#28 — nothing may be inferred from the bytes). A
347    // MALFORMED vector type is deliberately NOT silently degraded either: it is
348    // returned in a spelling no decoder claims (`vector<unparseable:…>` never
349    // parses as a vector and matches no other type arm), because this function's
350    // signature cannot carry an error and inventing a dimension would put a
351    // made-up width on the decode path.
352    //
353    // The probe is THREE-valued (roborev job 109): a malformed `VectorType(`, whose
354    // parameter list cannot even be EXTRACTED, used to be indistinguishable from
355    // "not a vector" and so reached the lowercase fallback — the very defect above,
356    // by a second route. `into_args` folds both malformed states into `Err`.
357    match crate::schema::vector_type::marshal_vector_kind(cleaned).into_args(cleaned) {
358        Ok(Some(args)) => {
359            return format!(
360                "vector<{}, {}>",
361                convert_marshal_type_to_cql(args.element),
362                args.dimension
363            )
364        }
365        Err(e) => {
366            tracing::warn!(
367                "malformed vector marshal type '{}' in SerializationHeader: {}",
368                cleaned,
369                e
370            );
371            // Fail closed, with the type NAMED, in a spelling that (a) still
372            // begins `vector<` so the decoder's VECTOR arm intercepts it — anything
373            // else would reach the blob fall-through, which is the defect — and (b)
374            // can never parse as a vector.
375            //
376            // (b) is why the `, unparseable-dimension` tail is appended rather than
377            // just wrapping `cleaned`: `vector<unparseable:{cleaned}>` alone CAN
378            // parse when `cleaned` happens to end in `, <digits>` (measured: a
379            // trailing-text VectorType parsed as `vector<…, 4>`). With the tail,
380            // every case fails — either the `<`/`>` matcher rejects the brackets, or
381            // `cleaned`'s own unclosed bracket leaves NO top-level comma, or the last
382            // top-level comma is this one and `unparseable-dimension` is not a
383            // decimal dimension.
384            return format!("vector<unparseable:{}, unparseable-dimension>", cleaned);
385        }
386        // Not a vector at all: fall through to the remaining type arms.
387        Ok(None) => {}
388    }
389
390    for prefix in ["org.apache.cassandra.db.marshal.MapType(", "MapType("] {
391        if let Some(params_with_close) = cleaned.strip_prefix(prefix) {
392            if let Some(inner) = extract_inner_type(params_with_close) {
393                let args = split_type_arguments(inner);
394                if args.len() == 2 {
395                    let key = convert_marshal_type_to_cql(args[0]);
396                    let value = convert_marshal_type_to_cql(args[1]);
397                    return format!("map<{}, {}>", key, value);
398                } else if args.len() == 1 {
399                    let value = convert_marshal_type_to_cql(args[0]);
400                    return format!("map<text, {}>", value);
401                }
402            }
403        }
404    }
405
406    cleaned = strip_wrapping_parens(cleaned);
407    let base = strip_type_suffix(strip_namespace(cleaned)).trim_end_matches(')');
408
409    // Map common types to CQL equivalents
410    match base {
411        "UTF8" => "text".to_string(),
412        "Int32" => "int".to_string(),
413        "Integer" => "int".to_string(),
414        "Long" => "bigint".to_string(),
415        "Short" => "smallint".to_string(),
416        "Byte" => "tinyint".to_string(),
417        "SimpleDate" => "date".to_string(),
418        "Timestamp" => "timestamp".to_string(),
419        "Boolean" => "boolean".to_string(),
420        "Decimal" => "decimal".to_string(),
421        "Float" => "float".to_string(),
422        "Double" => "double".to_string(),
423        "Bytes" => "blob".to_string(),
424        "Ascii" => "ascii".to_string(),
425        "InetAddress" => "inet".to_string(),
426        "UUID" => "uuid".to_string(),
427        "TimeUUID" => "timeuuid".to_string(),
428        "Duration" => "duration".to_string(),
429        "Time" => "time".to_string(),
430        "Counter" | "CounterColumn" => "counter".to_string(),
431        other => other.to_lowercase(),
432    }
433}

435/// Construct ColumnInfo entries for partition key definitions found in SerializationHeader
436pub(super) fn build_partition_key_columns(
437    partition_types: &[String],
438) -> crate::error::Result<Vec<super::super::header::ColumnInfo>> {
439    if partition_types.is_empty() {
440        return Ok(Vec::new());
441    }
442
443    let total = partition_types.len();
444    partition_types
445        .iter()
446        .enumerate()
447        .map(|(idx, marshal_type)| {
448            let cql_type = convert_marshal_type_to_cql_checked(marshal_type)?;
449            let name = if total == 1 {
450                match cql_type.as_str() {
451                    "uuid" | "timeuuid" => "id".to_string(),
452                    _ => "partition_key".to_string(),
453                }
454            } else {
455                format!("partition_key_{}", idx)
456            };
457
458            Ok(super::super::header::ColumnInfo {
459                name,
460                column_type: cql_type,
461                is_primary_key: true,
462                key_position: Some(idx as u16),
463                is_static: false,
464                is_clustering: false,
465                clustering_reversed: false,
466            })
467        })
468        .collect()
469}

471/// Construct ColumnInfo entries for clustering key definitions found in SerializationHeader
472pub(super) fn build_clustering_key_columns(
473    clustering_types: &[String],
474) -> crate::error::Result<Vec<super::super::header::ColumnInfo>> {
475    if clustering_types.is_empty() {
476        return Ok(Vec::new());
477    }
478
479    let total = clustering_types.len();
480    clustering_types
481        .iter()
482        .enumerate()
483        .map(|(idx, marshal_type)| {
484            // Issue #759: a DESC clustering column is encoded by wrapping its
485            // comparator class in `ReversedType(...)`. Detect that authoritative
486            // signal here, BEFORE `convert_marshal_type_to_cql` unwraps it, so
487            // schema discovery can report `ClusteringOrder::Desc`. The CQL type
488            // continues to come from `convert_marshal_type_to_cql`, which strips
489            // `ReversedType` so the inner type's deserialization is undisturbed.
490            let clustering_reversed = is_reversed_comparator(marshal_type);
491            let cql_type = convert_marshal_type_to_cql_checked(marshal_type)?;
492            let name = if total == 1 {
493                "clustering_key".to_string()
494            } else {
495                format!("clustering_key_{}", idx)
496            };
497
498            Ok(super::super::header::ColumnInfo {
499                name,
500                column_type: cql_type,
501                is_primary_key: true,
502                key_position: Some(idx as u16),
503                is_static: false,
504                is_clustering: true,
505                clustering_reversed,
506            })
507        })
508        .collect()
509}

511/// Build ColumnInfo vectors from parsed type strings.
512pub(super) fn build_column_infos(
513    partition_types: &[String],
514    clustering_types: &[String],
515) -> crate::error::Result<(
516    Vec<super::super::header::ColumnInfo>,
517    Vec<super::super::header::ColumnInfo>,
518)> {
519    let partition_key_columns = build_partition_key_columns(partition_types)?;
520    let clustering_key_columns = build_clustering_key_columns(clustering_types)?;
521
522    tracing::debug!(
523        "Constructed ColumnInfo entries from SerializationHeader: {} partition keys, {} clustering keys",
524        partition_key_columns.len(),
525        clustering_key_columns.len()
526    );
527
528    Ok((partition_key_columns, clustering_key_columns))
529}

535    // Issue #759: ReversedType wrapping is the authoritative DESC signal.
536    #[test]
537    fn test_is_reversed_comparator() {
538        // Fully-qualified and short forms both denote DESC.
539        assert!(is_reversed_comparator(
540            "org.apache.cassandra.db.marshal.ReversedType(org.apache.cassandra.db.marshal.TimestampType)"
541        ));
542        assert!(is_reversed_comparator("ReversedType(Int32Type)"));
543        // The header sometimes prefixes the top-level type with '(' — still DESC.
544        assert!(is_reversed_comparator(
545            "(org.apache.cassandra.db.marshal.ReversedType(org.apache.cassandra.db.marshal.SimpleDateType))"
546        ));
547        // ...and sometimes with a structural '[' (roborev job 43) — still DESC.
548        assert!(is_reversed_comparator(
549            "[org.apache.cassandra.db.marshal.ReversedType(org.apache.cassandra.db.marshal.TimestampType)"
550        ));
551        assert!(is_reversed_comparator("[ReversedType(Int32Type)"));
552        // Non-reversed comparators are ASC.
553        assert!(!is_reversed_comparator(
554            "org.apache.cassandra.db.marshal.UTF8Type"
555        ));
556        assert!(!is_reversed_comparator("Int32Type"));

665    /// Regression, roborev job 109 (issue #4114): a MALFORMED `VectorType` must FAIL
666    /// CLOSED here, never reach the `other => other.to_lowercase()` fallback at the
667    /// bottom of `convert_marshal_type_to_cql`.
668    ///
669    /// The probe used to answer `None` for BOTH "not a vector" and "a vector whose
670    /// parameter list cannot be extracted", so an unmatched paren fell through and
671    /// `org…VectorType(org…FloatType , 3` converted to the junk scalar spelling
672    /// `floattype , 3` — which no arm recognises, so the column was decoded as a
673    /// vint-length-prefixed BLOB: exactly the framing #4114 removed, restored by a
674    /// second route.
675    ///
676    /// The type strings are literals, not fixture-derived: a malformed type STRING is
677    /// a text-parsing question, not a byte-framing one, and Cassandra's writer cannot
678    /// emit one (`TypeParser.stringifyVectorParameters` concatenates an `int`), so no
679    /// Cassandra-written bytes can express this case.
680    #[test]
681    fn a_malformed_vector_marshal_type_never_reaches_the_lowercase_fallback() {
682        const PKG: &str = "org.apache.cassandra.db.marshal.";
683        // The well-formed spelling Cassandra writes still converts, dimension intact.
684        assert_eq!(
685            convert_marshal_type_to_cql(&format!("{PKG}VectorType({PKG}FloatType , 3)")),
686            "vector<float, 3>"
687        );
688
689        for malformed in [
690            // Unmatched open paren — the case that reached the lowercase fallback.
691            format!("{PKG}VectorType({PKG}FloatType , 3"),
692            // A `VectorType` claim with no parameter list at all.
693            format!("{PKG}VectorType"),
694            // Trailing text after the matched close paren.
695            format!("{PKG}VectorType({PKG}FloatType , 3) , 4"),
696            // Extractable parameters that do not split into (element, dimension).
697            format!("{PKG}VectorType({PKG}FloatType)"),
698            // A dimension Cassandra rejects at construction (VectorType.java:89-90).
699            format!("{PKG}VectorType({PKG}FloatType , 0)"),
700        ] {
701            let converted = convert_marshal_type_to_cql(&malformed);
702            assert!(
703                converted.starts_with("vector<unparseable:"),
704                "'{malformed}' converted to '{converted}': a malformed vector must \
705                 fail closed, not degrade to a spelling the blob arm accepts"
706            );
707            // And the fail-closed spelling must really be unparseable — no decoder
708            // may claim it, or "fail closed" would be a label rather than a property.
709            assert!(
710                crate::schema::CqlType::parse(&converted).is_err(),
711                "the fail-closed spelling '{converted}' must not parse as any type"
712            );
713        }
714    }
