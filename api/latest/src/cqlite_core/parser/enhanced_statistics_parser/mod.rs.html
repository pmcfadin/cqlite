1//! Enhanced Statistics.db parser for Cassandra 5.0 'nb' format
2//!
3//! # Implementation Status (Issue #162)
4//!
5//! This module provides **MINIMAL PARSING** of nb-format Statistics.db files to support
6//! delta-coded timestamp decoding in V5CompressedLegacy parser.
7//!
8//! ## Current Implementation
9//!
10//! Parses ONLY the EncodingStats fields required for delta decoding:
11//! - Header (32 bytes): version, data_length, checksum, metadata
12//! - EncodingStats section: partitioner, minTimestamp, minLocalDeletionTime, minTTL
13//!
14//! All other statistics (row counts, histograms, column stats, etc.) are populated with
15//! placeholder values. This is sufficient for V5CompressedLegacy parser baseline values.
16//!
17//! ## Previous Implementation (REMOVED)
18//!
19//! The previous implementation violated the no-heuristics mandate (Issue #28) by fabricating
20//! statistics from header metadata. It was removed and replaced with this minimal real-data
21//! parser that extracts only what's needed from the actual binary format.
22//!
23//! ## Deferred to Future Milestones
24//!
25//! Complete Statistics.db parsing including:
26//! - Row count statistics and distribution histograms
27//! - Column-level statistics and cardinality estimates
28//! - Partition size histograms and percentiles
29//! - Compression ratio and performance metrics
30//! - Checksum validation (header.checksum field not yet validated)
31//!
32//! ## References
33//!
34//! - Issue #162: Fix Statistics reader for Cassandra 5 nb format
35//! - Issue #28: No-heuristics mandate for modern Cassandra 5.0 paths
36//! - Issue #105: Remove heuristic estimation from enhanced_statistics_parser.rs
37//! - `docs/development/rust_developer_guide.md`: Architecture decisions
38//!
39//! # Module layout (Issue #1125 split)
40//!
41//! - [`header`] — 32-byte file header + Table-of-Contents (component offsets).
42//! - [`encoding_stats`] — EncodingStats (min timestamp/deletion-time/TTL) decode.
43//! - [`serialization_header`] — table-schema (partition/clustering/static/regular columns).
44//! - [`marshal_type`] — Cassandra marshal-type → CQL conversion + `ColumnInfo` builders.
45//! - this `mod.rs` — the public entry points that orchestrate the above.
46

73/// Parse minimal nb-format statistics data for delta-coding baseline (Issue #162)
74///
75/// This implementation parses ONLY the EncodingStats fields required for delta decoding:
76/// - partitioner (string)
77/// - minTimestamp (VInt)
78/// - minLocalDeletionTime (VInt)
79/// - minTTL (VInt)
80///
81/// All other fields (histograms, column stats, etc.) are skipped to minimize complexity.
82/// This is sufficient for V5CompressedLegacy parser which needs baseline values for
83/// delta-coded timestamps and TTLs.
84///
85/// # Format (observed from real nb-format Statistics.db files)
86///
87/// After 32-byte header:
88/// - metadata_type (u32 BE) = 0x00000003 (indicates EncodingStats section)
89/// - data_length (VInt) - length of remaining data
90/// - partitioner_length (VInt) - length of partitioner class name string
91/// - partitioner (UTF-8 string) - e.g., "org.apache.cassandra.dht.Murmur3Partitioner"
92/// - additional_metadata (various VInts) - skipped
93/// - minTimestamp (VInt, microseconds)
94/// - minLocalDeletionTime (VInt, seconds)
95/// - minTTL (VInt, seconds)
96///
97/// # Returns
98///
99/// Partial statistics with only TimestampStatistics populated from real data.
100#[allow(clippy::type_complexity)]
101pub fn parse_nb_format_statistics_data(
102    input: &[u8],
103    header: &StatisticsHeader,
104    full_input: &[u8],
105    // VG3 plumbing: gates are threaded here so version-sensitive decisions in
106    // parse_encoding_stats_vuints (e.g. has_uint_deletion_time) can be flipped
107    // without re-deriving gates from the filename.
108    // Pass `None` from callers that do not have gates (standalone tools, tests).
109    gates: Option<&VersionGates>,
110) -> Result<(
111    RowStatistics,
112    TimestampStatistics,
113    TableStatistics,
114    PartitionStatistics,
115    CompressionStatistics,
116    Vec<super::header::ColumnInfo>,
117    Vec<super::header::ColumnInfo>,
118    Vec<super::header::ColumnInfo>,
119)> {
120    // Get HEADER offset from TOC (Issue #216)
121    let header_offset = parse_statistics_toc_for_header_offset(full_input);
122
123    // Parse the EncodingStats section from the data following the header
124    let result = parse_minimal_encoding_stats(input, full_input, header_offset, gates);
125
126    match result {
127        Ok((
128            _,
129            (
130                min_timestamp,
131                min_deletion_time,
132                min_ttl,
133                partition_columns,
134                clustering_columns,
135                regular_columns,
136            ),
137        )) => {
138            // Populate the authoritative row/partition counts from the STATS
139            // component (issue #1325). We reuse the single source of truth added
140            // in #944 — `repair_metadata::read_table_counts` — rather than
141            // re-deriving the walk here:
142            //   * `partition_count` = Σ `estimatedPartitionSize` histogram bucket
143            //     counts (self-describing; always decodable).
144            //   * `total_rows`      = STATS `totalRows` via the version-gated walk,
145            //     `None` when not authoritatively traversable (e.g. a clustered
146            //     covered-Slice whose bound values are not modeled).
147            // No-heuristics mandate (#28): when a value is NOT authoritatively
148            // reachable we leave the documented 0 rather than fabricating a count.
149            //   * `total_rows`: `None` from `read_table_counts` → 0 (meaning
150            //     "not authoritatively reachable", never a guessed count).
151            //   * `live_rows`: STATS carries no per-SSTable live-row count that is
152            //     authoritatively distinguishable from `total_rows` here, so it is
153            //     left as 0 (documented "not authoritatively derivable") rather
154            //     than guessed. (Consumers that need an upper bound use
155            //     `total_rows`; see cqlite-flight `read_table_counts` doctrine.)
156            //   * `avg_rows_per_partition`: computed only when BOTH counts are
157            //     authoritative and `partition_count > 0`; otherwise left 0.0.
158            // `full_input` is the raw Statistics.db buffer (with TOC), exactly the
159            // input `read_table_counts` expects. This is strictly additive — an
160            // Err is logged and the fields stay at 0 (a Statistics.db that parses
161            // today must keep parsing).
162            //
163            // No-heuristics gate discipline (#28, #1325 roborev finding): the
164            // version-gated `read_table_counts` walk MUST run with gates that match
165            // the file's actual on-disk layout — Statistics.db does NOT self-describe
166            // its SSTable version (the version comes from the filename → the gates).
167            // The `nb` and `oa`/`da` STATS min/max blocks have DIFFERENT layouts, so
168            // synthesizing `nb` gates for a file whose format we have not
169            // authoritatively established would mis-walk an `oa`/`da` buffer and could
170            // read a BOGUS nonzero `totalRows`. This function is the SINGLE entry point
171            // for `parse_statistics_with_fallback`, which is a public API reachable with
172            // `None` gates for ANY format (standalone tools, tests) — there is no
173            // format-detection dispatch that guarantees only `nb` bytes arrive here.
174            //
175            // Therefore we pass the caller's gates through UNCHANGED and NEVER
176            // synthesize a version:
177            //   * `Some(gates)` — the caller (e.g. `StatisticsReader::open`) derived
178            //     these from the filename, so they are authoritative for the file's
179            //     real version (nb / oa / da). The gated walk reaches `totalRows`.
180            //   * `None` — the format is UNKNOWN. We cannot authoritatively establish
181            //     it, so `read_table_counts` stops before the gated min/max block and
182            //     reports `total_rows = None`. We surface that as `0` meaning "not
183            //     authoritatively available", NEVER a guessed count. This is strictly
184            //     safer than the old `nb` synthesis, which could fabricate a nonzero
185            //     count for an oa/da file.
186            let (total_rows, partition_count) =
187                match crate::parser::repair_metadata::read_table_counts(full_input, gates) {
188                    // `total_rows.unwrap_or(0)`: a `None` here means STATS does NOT
189                    // authoritatively expose `totalRows` (e.g. an unmodeled
190                    // improvedMinMax covered-Slice bound blocks the walk). The 0 is
191                    // therefore "not authoritatively available from STATS", NOT a
192                    // measured zero — never a guessed count (#1325, no-heuristics #28).
193                    Ok(counts) => (counts.total_rows.unwrap_or(0), counts.partition_count),
194                    Err(e) => {
195                        log::debug!(
196                            "Best-effort authoritative row/partition-count decode failed; \
197                             leaving RowStatistics counts at 0: {:?}",
198                            e
199                        );
200                        (0, 0)
201                    }
202                };
203            let avg_rows_per_partition = if partition_count > 0 && total_rows > 0 {
204                total_rows as f64 / partition_count as f64
205            } else {
206                0.0
207            };
208
209            let row_stats = RowStatistics {
210                total_rows,
211                // `live_rows = 0` here means "NOT authoritatively available from
212                // STATS for nb", NOT a measured zero. STATS carries no per-SSTable
213                // live-row count that is authoritatively distinguishable from
214                // `total_rows`, so per #1325's acceptance criteria this unreachable
215                // field stays "0-with-documented-meaning" rather than being guessed
216                // or aliased to `total_rows` (no-heuristics mandate #28). Changing
217                // this field to `Option` is a broader API change tracked separately.
218                live_rows: 0,
219                tombstone_count: 0,
220                partition_count,
221                avg_rows_per_partition,
222                row_size_histogram: vec![],
223            };
224
225            let timestamp_stats = TimestampStatistics {
226                min_timestamp,
227                // Fail-closed placeholder. The authoritative `maxTimestamp` is
228                // recovered from STATS field 4 by the best-effort post-pass in
229                // `parse_enhanced_statistics_file` (issue #1729). The inner
230                // EncodingStats parser does not reach field 4, so until the
231                // post-pass runs we use `i64::MIN` — Cassandra's "no max write
232                // timestamp recorded" sentinel — NOT the old `min_timestamp`
233                // placeholder (which silently reported the MIN as the max).
234                // A consumer that sees `i64::MIN` must treat the max as
235                // unavailable (do not proceed as if a real max were known).
236                max_timestamp: i64::MIN,
237                min_deletion_time,
238                max_deletion_time: min_deletion_time,
239                min_ttl,
240                max_ttl: min_ttl,
241                rows_with_ttl: 0,
242            };
243
244            let table_stats = TableStatistics {
245                disk_size: 0,
246                uncompressed_size: 0,
247                compressed_size: 0,
248                compression_ratio: 1.0,
249                block_count: 0,
250                avg_block_size: 0.0,
251                index_size: 0,
252                bloom_filter_size: 0,
253                level_count: 0,
254            };
255
256            let partition_stats = PartitionStatistics {
257                avg_partition_size: 0.0,
258                min_partition_size: 0,
259                max_partition_size: 0,
260                large_partition_percentage: 0.0,
261                size_histogram: vec![],
262            };
263
264            let compression_stats = CompressionStatistics {
265                algorithm: "unknown".to_string(),
266                original_size: 0,
267                compressed_size: 0,
268                ratio: 1.0,
269                compression_speed: 0.0,
270                decompression_speed: 0.0,
271                compressed_blocks: 0,
272            };
273
274            Ok((
275                row_stats,
276                timestamp_stats,
277                table_stats,
278                partition_stats,
279                compression_stats,
280                partition_columns,
281                clustering_columns,
282                regular_columns,
283            ))
284        }
285        Err(e) => {
286            log::debug!(
287                "Failed to parse minimal EncodingStats from Statistics.db: {:?}",
288                e
289            );
290            Err(Error::UnsupportedFormat(format!(
291                "Failed to parse minimal nb-format Statistics.db EncodingStats: {:?}. \
292                         This is required for delta-coded timestamp decoding. \
293                         Header checksum: 0x{:08x}, data_length: {}",
294                e, header.checksum, header.data_length
295            )))
296        }
297    }
298}

300/// Main enhanced parser for real Statistics.db files (minimal implementation for Issue #162)
301///
302/// This function parses the header and minimal EncodingStats fields from nb-format
303/// Statistics.db files. Only timestamp-related fields are extracted; all other
304/// statistics (histograms, column stats, etc.) are populated with placeholder values.
305///
306/// This is sufficient for V5CompressedLegacy parser which requires min_timestamp,
307/// min_local_deletion_time, and min_ttl for delta decoding baseline.
308///
309/// # Arguments
310///
311/// * `gates` - Optional [`VersionGates`] for version-sensitive decoding decisions
312///   (VG1 plumbing).  Pass `None` from standalone tools/tests to use nb-compatible
313///   defaults; pass `Some(&gates)` from `SSTableReader` to enable VG3 gating.
314///
315/// # Returns
316///
317/// SSTableStatistics with only header and timestamp_stats populated from real data.
318pub fn parse_enhanced_statistics_file<'a>(
319    input: &'a [u8],
320    gates: Option<&VersionGates>,
321) -> nom::IResult<&'a [u8], SSTableStatistics> {
322    // Parse the 32-byte header
323    let (remaining, header) = parse_nb_format_header(input)?;
324
325    // Parse minimal statistics data (EncodingStats + SerializationHeader columns)
326    // Pass full input for TOC-based HEADER offset lookup (Issue #216)
327    let result = parse_nb_format_statistics_data(remaining, &header, input, gates);
328
329    match result {
330        Ok((
331            row_stats,
332            mut timestamp_stats,
333            table_stats,
334            partition_stats,
335            compression_stats,
336            partition_columns,
337            clustering_columns,
338            columns,
339        )) => {
340            log::debug!(
341                "Successfully parsed Statistics.db serialization header: {} partition keys, {} clustering keys, {} regular columns",
342                partition_columns.len(),
343                clustering_columns.len(),
344                columns.len()
345            );
346
347            // Best-effort STATS-extras decode over the FULL buffer (with TOC):
348            // recover the authoritative maxLocalDeletionTime (the inner parser
349            // leaves it as a placeholder equal to min_deletion_time), the
350            // authoritative maxTimestamp (issue #1729 — the inner parser leaves
351            // it as the `i64::MIN` "unavailable" sentinel), and the estimated
352            // tombstone-drop-times histogram (Issue #1073). This is strictly
353            // additive — a Statistics.db that parses today must keep parsing, so
354            // an Err here is logged and the placeholders are kept. When
355            // maxTimestamp is unavailable (extras.max_timestamp == None, i.e.
356            // Cassandra's Long.MIN_VALUE sentinel) we deliberately leave the
357            // `i64::MIN` fail-closed sentinel in place rather than substituting
358            // the min — consumers requiring a real max must not proceed.
359            let tombstone_drop_times =
360                match crate::parser::repair_metadata::parse_stats_extras(input, gates) {
361                    Ok(extras) => {
362                        // Only set max_deletion_time. Do NOT touch min_deletion_time
363                        // (the EncodingStats min baseline consumed elsewhere).
364                        timestamp_stats.max_deletion_time = extras.max_local_deletion_time;
365                        if let Some(max_ts) = extras.max_timestamp {
366                            timestamp_stats.max_timestamp = max_ts;
367                        }
368                        extras.tombstone_drop_times
369                    }
370                    Err(e) => {
371                        log::debug!(
372                            "Best-effort STATS-extras decode failed; keeping placeholders: {:?}",
373                            e
374                        );
375                        vec![]
376                    }
377                };
378
379            let statistics = SSTableStatistics {
380                header,
381                row_stats,
382                timestamp_stats,
383                column_stats: vec![],
384                table_stats,
385                partition_stats,
386                compression_stats,
387                metadata: std::collections::HashMap::new(),
388                serialization_header_columns: columns,
389                serialization_header_partition_keys: partition_columns,
390                serialization_header_clustering_keys: clustering_columns,
391                tombstone_drop_times,
392            };
393
394            Ok((remaining, statistics))
395        }
396        Err(e) => {
397            // Convert Error to nom::Err
398            log::warn!("Failed to parse nb-format Statistics.db: {}", e);
399            Err(nom::Err::Error(nom::error::Error::new(
400                input,
401                nom::error::ErrorKind::Verify,
402            )))
403        }
404    }
405}

407/// Enhanced statistics reader with fallback (minimal implementation for Issue #162)
408///
409/// Attempts to parse nb-format Statistics.db with minimal EncodingStats extraction.
410/// This provides the minimum fields needed for delta-coded timestamp decoding.
411///
412/// # Arguments
413///
414/// * `gates` - Optional [`VersionGates`] threaded from `SSTableReader` for VG3
415///   version-sensitive decoding.  Pass `None` from standalone tools/tests; the
416///   nb-compatible behaviour is used when `gates` is `None`.
417///
418/// # Returns
419///
420/// SSTableStatistics with minimal fields populated, or error if parsing fails.
421pub fn parse_statistics_with_fallback<'a>(
422    input: &'a [u8],
423    gates: Option<&VersionGates>,
424) -> nom::IResult<&'a [u8], SSTableStatistics> {
425    // Try the minimal enhanced parser
426    parse_enhanced_statistics_file(input, gates)
427}

435        // Test with insufficient/invalid data - should fail to parse VInts
436        let header = StatisticsHeader {
437            version: 4,

439            data_length: 44,
440            metadata1: 1,

442            metadata3: 2,
443            checksum: 0x14d4,
444            table_id: None,
445        };
446
447        let dummy_data = vec![0xFF; 10]; // Too short to parse properly
448        let result = parse_nb_format_statistics_data(&dummy_data, &header, &dummy_data, None);

451        assert!(result.is_err());
