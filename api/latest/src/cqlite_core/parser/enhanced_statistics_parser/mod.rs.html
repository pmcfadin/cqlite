1//! Enhanced Statistics.db parser for Cassandra 5.0 'nb' format
2//!
3//! # Implementation Status (Issue #162)
4//!
5//! This module provides **MINIMAL PARSING** of nb-format Statistics.db files to support
6//! delta-coded timestamp decoding in V5CompressedLegacy parser.
7//!
8//! ## Current Implementation
9//!
10//! Parses ONLY the EncodingStats fields required for delta decoding:
11//! - Header (32 bytes): version, data_length, checksum, metadata
12//! - EncodingStats section: partitioner, minTimestamp, minLocalDeletionTime, minTTL
13//!
14//! All other statistics (row counts, histograms, column stats, etc.) are populated with
15//! placeholder values. This is sufficient for V5CompressedLegacy parser baseline values.
16//!
17//! ## Previous Implementation (REMOVED)
18//!
19//! The previous implementation violated the no-heuristics mandate (Issue #28) by fabricating
20//! statistics from header metadata. It was removed and replaced with this minimal real-data
21//! parser that extracts only what's needed from the actual binary format.
22//!
23//! ## Deferred to Future Milestones
24//!
25//! Complete Statistics.db parsing including:
26//! - Row count statistics and distribution histograms
27//! - Column-level statistics and cardinality estimates
28//! - Partition size histograms and percentiles
29//! - Compression ratio and performance metrics
30//! - Checksum validation (header.checksum field not yet validated)
31//!
32//! ## References
33//!
34//! - Issue #162: Fix Statistics reader for Cassandra 5 nb format
35//! - Issue #28: No-heuristics mandate for modern Cassandra 5.0 paths
36//! - Issue #105: Remove heuristic estimation from enhanced_statistics_parser.rs
37//! - `docs/development/rust_developer_guide.md`: Architecture decisions
38//!
39//! # Module layout (Issue #1125 split)
40//!
41//! - [`header`] — 32-byte file header + Table-of-Contents (component offsets).
42//! - [`encoding_stats`] — EncodingStats (min timestamp/deletion-time/TTL) decode.
43//! - [`serialization_header`] — table-schema (partition/clustering/static/regular columns).
44//! - [`marshal_type`] — Cassandra marshal-type → CQL conversion + `ColumnInfo` builders.
45//! - this `mod.rs` — the public entry points that orchestrate the above.

73/// Parse minimal nb-format statistics data for delta-coding baseline (Issue #162)
74///
75/// This implementation parses ONLY the EncodingStats fields required for delta decoding:
76/// - partitioner (string)
77/// - minTimestamp (VInt)
78/// - minLocalDeletionTime (VInt)
79/// - minTTL (VInt)
80///
81/// All other fields (histograms, column stats, etc.) are skipped to minimize complexity.
82/// This is sufficient for V5CompressedLegacy parser which needs baseline values for
83/// delta-coded timestamps and TTLs.
84///
85/// # Format (observed from real nb-format Statistics.db files)
86///
87/// After 32-byte header:
88/// - metadata_type (u32 BE) = 0x00000003 (indicates EncodingStats section)
89/// - data_length (VInt) - length of remaining data
90/// - partitioner_length (VInt) - length of partitioner class name string
91/// - partitioner (UTF-8 string) - e.g., "org.apache.cassandra.dht.Murmur3Partitioner"
92/// - additional_metadata (various VInts) - skipped
93/// - minTimestamp (VInt, microseconds)
94/// - minLocalDeletionTime (VInt, seconds)
95/// - minTTL (VInt, seconds)
96///
97/// # Returns
98///
99/// Partial statistics with only TimestampStatistics populated from real data.
100#[allow(clippy::type_complexity)]
101pub fn parse_nb_format_statistics_data(
102    input: &[u8],
103    header: &StatisticsHeader,
104    full_input: &[u8],
105    // VG3 plumbing: gates are threaded here so version-sensitive decisions in
106    // parse_encoding_stats_vuints (e.g. has_uint_deletion_time) can be flipped
107    // without re-deriving gates from the filename.
108    // Pass `None` from callers that do not have gates (standalone tools, tests).
109    gates: Option<&VersionGates>,
110) -> Result<(
111    RowStatistics,
112    TimestampStatistics,
113    Vec<super::header::ColumnInfo>,
114    Vec<super::header::ColumnInfo>,
115    Vec<super::header::ColumnInfo>,
116)> {
117    // Standalone/test entry: parse the TOC once here, then delegate. The main
118    // open path (`parse_enhanced_statistics_file`) parses the TOC ONCE and threads
119    // it to both this decode and the STATS post-pass (issue #2148).
120    let toc = parse_statistics_toc(full_input);
121    parse_nb_format_statistics_data_with_toc(input, header, full_input, &toc, gates)
122}

124/// [`parse_nb_format_statistics_data`] over a `Statistics.db` TOC that was already
125/// parsed once (issue #2148): reuses the resolved HEADER offset and STATS bounds
126/// from `toc` instead of re-walking the TOC, so a single metadata parse walks the
127/// TOC exactly once.
128#[allow(clippy::type_complexity)]
129pub(crate) fn parse_nb_format_statistics_data_with_toc(
130    input: &[u8],
131    header: &StatisticsHeader,
132    full_input: &[u8],
133    toc: &StatisticsToc,
134    gates: Option<&VersionGates>,
135) -> Result<(
136    RowStatistics,
137    TimestampStatistics,
138    Vec<super::header::ColumnInfo>,
139    Vec<super::header::ColumnInfo>,
140    Vec<super::header::ColumnInfo>,
141)> {
142    // HEADER offset from the shared single TOC parse (Issue #216, #2148).
143    let header_offset = toc.header_offset();
144
145    // Parse the EncodingStats section from the data following the header
146    let result = parse_minimal_encoding_stats(input, full_input, header_offset, gates);
147
148    match result {
149        Ok((
150            _,
151            (
152                min_timestamp,
153                min_deletion_time,
154                min_ttl,
155                partition_columns,
156                clustering_columns,
157                regular_columns,
158            ),
159        )) => {
160            // Populate the authoritative row/partition counts from the STATS
161            // component (issue #1325). We reuse the single source of truth added
162            // in #944 — `repair_metadata::read_table_counts` — rather than
163            // re-deriving the walk here:
164            //   * `partition_count` = Σ `estimatedPartitionSize` histogram bucket
165            //     counts (self-describing; always decodable).
166            //   * `total_rows`      = STATS `totalRows` via the version-gated walk,
167            //     `None` when not authoritatively traversable (e.g. a clustered
168            //     covered-Slice whose bound values are not modeled).
169            // No-heuristics mandate (#28): when a value is NOT authoritatively
170            // reachable we leave the documented 0 rather than fabricating a count.
171            //   * `total_rows`: `None` from `read_table_counts` → 0 (meaning
172            //     "not authoritatively reachable", never a guessed count).
173            //   * `live_rows`: STATS carries no per-SSTable live-row count that is
174            //     authoritatively distinguishable from `total_rows` here, so it is
175            //     left as 0 (documented "not authoritatively derivable") rather
176            //     than guessed. (Consumers that need an upper bound use
177            //     `total_rows`; see cqlite-flight `read_table_counts` doctrine.)
178            //   * `avg_rows_per_partition`: computed only when BOTH counts are
179            //     authoritative and `partition_count > 0`; otherwise left 0.0.
180            // `full_input` is the raw Statistics.db buffer (with TOC), exactly the
181            // input `read_table_counts` expects. This is strictly additive — an
182            // Err is logged and the fields stay at 0 (a Statistics.db that parses
183            // today must keep parsing).
184            //
185            // No-heuristics gate discipline (#28, #1325 roborev finding): the
186            // version-gated `read_table_counts` walk MUST run with gates that match
187            // the file's actual on-disk layout — Statistics.db does NOT self-describe
188            // its SSTable version (the version comes from the filename → the gates).
189            // The `nb` and `oa`/`da` STATS min/max blocks have DIFFERENT layouts, so
190            // synthesizing `nb` gates for a file whose format we have not
191            // authoritatively established would mis-walk an `oa`/`da` buffer and could
192            // read a BOGUS nonzero `totalRows`. This function is the SINGLE entry point
193            // for `parse_statistics_with_fallback`, which is a public API reachable with
194            // `None` gates for ANY format (standalone tools, tests) — there is no
195            // format-detection dispatch that guarantees only `nb` bytes arrive here.
196            //
197            // Therefore we pass the caller's gates through UNCHANGED and NEVER
198            // synthesize a version:
199            //   * `Some(gates)` — the caller (e.g. `StatisticsReader::open`) derived
200            //     these from the filename, so they are authoritative for the file's
201            //     real version (nb / oa / da). The gated walk reaches `totalRows`.
202            //   * `None` — the format is UNKNOWN. We cannot authoritatively establish
203            //     it, so `read_table_counts` stops before the gated min/max block and
204            //     reports `total_rows = None`. We surface that as `0` meaning "not
205            //     authoritatively available", NEVER a guessed count. This is strictly
206            //     safer than the old `nb` synthesis, which could fabricate a nonzero
207            //     count for an oa/da file.
208            let (total_rows, partition_count) =
209                match crate::parser::repair_metadata::read_table_counts_with_toc(
210                    full_input, toc, gates,
211                ) {
212                    // `total_rows.unwrap_or(0)`: a `None` here means STATS does NOT
213                    // authoritatively expose `totalRows` (e.g. an unmodeled
214                    // improvedMinMax covered-Slice bound blocks the walk). The 0 is
215                    // therefore "not authoritatively available from STATS", NOT a
216                    // measured zero — never a guessed count (#1325, no-heuristics #28).
217                    Ok(counts) => (counts.total_rows.unwrap_or(0), counts.partition_count),
218                    Err(e) => {
219                        tracing::debug!(
220                            "Best-effort authoritative row/partition-count decode failed; \
221                             leaving RowStatistics counts at 0: {:?}",
222                            e
223                        );
224                        (0, 0)
225                    }
226                };
227            let avg_rows_per_partition = if partition_count > 0 && total_rows > 0 {
228                total_rows as f64 / partition_count as f64
229            } else {
230                0.0
231            };
232
233            let row_stats = RowStatistics {
234                total_rows,
235                // `live_rows = 0` here means "NOT authoritatively available from
236                // STATS for nb", NOT a measured zero. STATS carries no per-SSTable
237                // live-row count that is authoritatively distinguishable from
238                // `total_rows`, so per #1325's acceptance criteria this unreachable
239                // field stays "0-with-documented-meaning" rather than being guessed
240                // or aliased to `total_rows` (no-heuristics mandate #28). Changing
241                // this field to `Option` is a broader API change tracked separately.
242                live_rows: 0,
243                tombstone_count: 0,
244                partition_count,
245                avg_rows_per_partition,
246                row_size_histogram: vec![],
247            };
248
249            let timestamp_stats = TimestampStatistics {
250                min_timestamp,
251                // `None` = maxTimestamp NOT yet authoritatively available (issue
252                // #1653). The real `maxTimestamp` is recovered from STATS field 4
253                // by the best-effort post-pass in `parse_enhanced_statistics_file`
254                // (issue #1729), which sets `Some(..)`. The inner EncodingStats
255                // parser does not reach field 4, so it stays `None` here — an
256                // honest "unavailable", NOT the old `i64::MIN` sentinel or the even
257                // older `min_timestamp` alias (both lied about the max). A consumer
258                // that needs a real max must treat `None` as unavailable.
259                max_timestamp: None,
260                min_deletion_time,
261                // Genuinely parsed by the STATS post-pass (#1073/#1011), which
262                // overwrites this below. Until then use the authoritative
263                // `NO_DELETION_TIME` (`i64::MAX`) "no deletions recorded" value so a
264                // post-pass failure fails CLOSED (never classified fully-expired),
265                // NOT the old `= min_deletion_time` placeholder which reported the
266                // MIN as the max (issue #1653).
267                max_deletion_time: i64::MAX,
268                min_ttl,
269                // `None` = maxTTL not authoritatively available (issue #1653). The
270                // enhanced parser does not decode a per-SSTable max TTL, so it is
271                // honestly `None`, NOT the old `= min_ttl` alias (which lied).
272                max_ttl: None,
273                // `None` = rows-with-TTL count not authoritatively available (issue
274                // #1653), NOT a fabricated `0` (which claimed "no rows have a TTL").
275                rows_with_ttl: None,
276            };
277
278            // Table/partition/compression statistics are NOT decoded by the
279            // enhanced (nb) parser. They are left `None` on `SSTableStatistics`
280            // (built by the caller) rather than fabricated with all-zero /
281            // "unknown" placeholders that were type-indistinguishable from real
282            // values (issue #1653; no-heuristics/no-fabrication mandate #28).
283            Ok((
284                row_stats,
285                timestamp_stats,
286                partition_columns,
287                clustering_columns,
288                regular_columns,
289            ))
290        }
291        Err(e) => {
292            tracing::debug!(
293                "Failed to parse minimal EncodingStats from Statistics.db: {:?}",
294                e
295            );
296            Err(Error::UnsupportedFormat(format!(
297                "Failed to parse minimal nb-format Statistics.db EncodingStats: {:?}. \
298                         This is required for delta-coded timestamp decoding. \
299                         Header checksum: 0x{:08x}, data_length: {}",
300                e, header.checksum, header.data_length
301            )))
302        }
303    }
304}

306/// Main enhanced parser for real Statistics.db files (minimal implementation for Issue #162)
307///
308/// This function parses the header and minimal EncodingStats fields from nb-format
309/// Statistics.db files. Only timestamp-related fields are extracted; all other
310/// statistics (histograms, column stats, etc.) are populated with placeholder values.
311///
312/// This is sufficient for V5CompressedLegacy parser which requires min_timestamp,
313/// min_local_deletion_time, and min_ttl for delta decoding baseline.
314///
315/// # Arguments
316///
317/// * `gates` - Optional [`VersionGates`] for version-sensitive decoding decisions
318///   (VG1 plumbing).  Pass `None` from standalone tools/tests to use nb-compatible
319///   defaults; pass `Some(&gates)` from `SSTableReader` to enable VG3 gating.
320///
321/// # Returns
322///
323/// SSTableStatistics with only header and timestamp_stats populated from real data.
324pub fn parse_enhanced_statistics_file<'a>(
325    input: &'a [u8],
326    gates: Option<&VersionGates>,
327) -> nom::IResult<&'a [u8], SSTableStatistics> {
328    // Parse the 32-byte header
329    let (remaining, header) = parse_nb_format_header(input)?;
330
331    // Parse the Statistics.db TOC ONCE (issue #2148) and thread it to every
332    // downstream consumer — the EncodingStats/row-count decode below and the
333    // STATS-extras post-pass — so the TOC is walked exactly once per open,
334    // never re-walked to relocate an already-resolved component offset.
335    let toc = parse_statistics_toc(input);
336
337    // Parse minimal statistics data (EncodingStats + SerializationHeader columns)
338    // Pass full input for TOC-based HEADER offset lookup (Issue #216)
339    let result = parse_nb_format_statistics_data_with_toc(remaining, &header, input, &toc, gates);
340
341    match result {
342        Ok((row_stats, mut timestamp_stats, partition_columns, clustering_columns, columns)) => {
343            tracing::debug!(
344                "Successfully parsed Statistics.db serialization header: {} partition keys, {} clustering keys, {} regular columns",
345                partition_columns.len(),
346                clustering_columns.len(),
347                columns.len()
348            );
349
350            // Best-effort STATS-extras decode over the FULL buffer (with TOC):
351            // recover the authoritative maxLocalDeletionTime (the inner parser
352            // leaves it as a placeholder equal to min_deletion_time), the
353            // authoritative maxTimestamp (issue #1729 — the inner parser leaves
354            // it as the `i64::MIN` "unavailable" sentinel), and the estimated
355            // tombstone-drop-times histogram (Issue #1073). This is strictly
356            // additive — a Statistics.db that parses today must keep parsing, so
357            // an Err here is logged and the placeholders are kept. When
358            // maxTimestamp is unavailable (extras.max_timestamp == None, i.e.
359            // Cassandra's Long.MIN_VALUE sentinel) we deliberately leave
360            // `max_timestamp == None` in place (issue #1653) rather than
361            // substituting the min — consumers requiring a real max must not
362            // proceed.
363            let tombstone_drop_times =
364                match crate::parser::repair_metadata::parse_stats_extras_with_toc(
365                    input, &toc, gates,
366                ) {
367                    Ok(extras) => {
368                        // Only set max_deletion_time. Do NOT touch min_deletion_time
369                        // (the EncodingStats min baseline consumed elsewhere).
370                        timestamp_stats.max_deletion_time = extras.max_local_deletion_time;
371                        // `extras.max_timestamp` is already `Option`: `Some` only
372                        // when the authoritative max was decoded (#1729). Propagate
373                        // it directly — `None` stays `None` (issue #1653).
374                        if let Some(max_ts) = extras.max_timestamp {
375                            timestamp_stats.max_timestamp = Some(max_ts);
376                        }
377                        // Authoritative maxTTL (issue #1537): the inner nb walk only
378                        // decodes the EncodingStats minTTL baseline and honestly
379                        // leaves `max_ttl = None`. Recover the true maximum from the
380                        // STATS-extras walk so the compaction TTL-expiry LDT floor
381                        // (`compute_expiry_ttl_ldt_floor`) can lower the output's
382                        // min-LDT delta baseline below a creation-time (`ldt - ttl`)
383                        // tombstone. `None` (no expiring cells) stays `None`.
384                        if extras.max_ttl.is_some() {
385                            timestamp_stats.max_ttl = extras.max_ttl;
386                        }
387                        extras.tombstone_drop_times
388                    }
389                    Err(e) => {
390                        tracing::debug!(
391                            "Best-effort STATS-extras decode failed; keeping placeholders: {:?}",
392                            e
393                        );
394                        vec![]
395                    }
396                };
397
398            let statistics = SSTableStatistics {
399                header,
400                row_stats,
401                timestamp_stats,
402                column_stats: vec![],
403                // NOT decoded by the enhanced (nb) parser → honestly `None`
404                // rather than fabricated all-zero / "unknown" placeholders
405                // (issue #1653; no-heuristics mandate #28).
406                table_stats: None,
407                partition_stats: None,
408                compression_stats: None,
409                metadata: std::collections::HashMap::new(),
410                serialization_header_columns: columns,
411                serialization_header_partition_keys: partition_columns,
412                serialization_header_clustering_keys: clustering_columns,
413                tombstone_drop_times,
414            };
415
416            Ok((remaining, statistics))
417        }
418        Err(e) => {
419            // Convert Error to nom::Err
420            tracing::warn!("Failed to parse nb-format Statistics.db: {}", e);
421            Err(nom::Err::Error(nom::error::Error::new(
422                input,
423                nom::error::ErrorKind::Verify,
424            )))
425        }
426    }
427}

429/// Enhanced statistics reader with fallback (minimal implementation for Issue #162)
430///
431/// Attempts to parse nb-format Statistics.db with minimal EncodingStats extraction.
432/// This provides the minimum fields needed for delta-coded timestamp decoding.
433///
434/// # Arguments
435///
436/// * `gates` - Optional [`VersionGates`] threaded from `SSTableReader` for VG3
437///   version-sensitive decoding.  Pass `None` from standalone tools/tests; the
438///   nb-compatible behaviour is used when `gates` is `None`.
439///
440/// # Returns
441///
442/// SSTableStatistics with minimal fields populated, or error if parsing fails.
443pub fn parse_statistics_with_fallback<'a>(
444    input: &'a [u8],
445    gates: Option<&VersionGates>,
446) -> nom::IResult<&'a [u8], SSTableStatistics> {
447    // Try the minimal enhanced parser
448    parse_enhanced_statistics_file(input, gates)
449}
