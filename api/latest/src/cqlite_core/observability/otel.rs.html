1//! OpenTelemetry runtime wiring (feature = "observability").
2//!
3//! This module is only compiled when the `observability` feature is on. It owns
4//! the OTLP trace + metric exporter setup, the parent-based trace-ID-ratio
5//! sampler, the global providers, the lazily-built metric instruments keyed by
6//! the [`catalog`](crate::observability::catalog) names, and the
7//! [`ObservabilityGuard`] that flushes/shuts down on drop.
8//!
9//! The public re-exports live in [`crate::observability`]; everything here is
10//! reached through those.

32/// Concrete SDK tracer provider stashed by [`init`] so [`tracing_layer`] can
33/// build its layer from an `SdkTracer` (which implements `PreSampledTracer`).
34/// When `init` is not called (or is inert), a default no-export provider is
35/// lazily created so the layer type stays monomorphic and simply drops spans.
36static TRACER_PROVIDER: OnceLock<SdkTracerProvider> = OnceLock::new();
37static METRICS_ACTIVE: AtomicBool = AtomicBool::new(false);

39/// Build the OTel `Resource` describing this process.
40fn build_resource(cfg: &ObservabilityConfig) -> Resource {
41    Resource::builder()
42        .with_service_name(cfg.service_name.clone())
43        .with_attribute(KeyValue::new(
44            semconv::attribute::SERVICE_VERSION,
45            cfg.service_version.clone(),
46        ))
47        .build()
48}

50/// Parent-based trace-ID-ratio sampler from the configured ratio.
51fn build_sampler(ratio: f64) -> Sampler {
52    Sampler::ParentBased(Box::new(Sampler::TraceIdRatioBased(ratio.clamp(0.0, 1.0))))
53}

55fn otlp_protocol(p: OtelProtocol) -> Protocol {
56    match p {
57        OtelProtocol::Grpc => Protocol::Grpc,
58        OtelProtocol::Http => Protocol::HttpBinary,
59    }
60}

62fn build_span_exporter(cfg: &ObservabilityConfig) -> Result<SpanExporter> {
63    let builder = match cfg.protocol {
64        OtelProtocol::Grpc => SpanExporter::builder()
65            .with_tonic()
66            .with_endpoint(cfg.endpoint.clone())
67            .with_protocol(otlp_protocol(cfg.protocol))
68            .with_timeout(cfg.timeout)
69            .build(),
70        OtelProtocol::Http => SpanExporter::builder()
71            .with_http()
72            .with_endpoint(cfg.endpoint.clone())
73            .with_protocol(otlp_protocol(cfg.protocol))
74            .with_timeout(cfg.timeout)
75            .build(),
76    };
77    builder.map_err(|e| Error::configuration(format!("OTLP span exporter init failed: {e}")))
78}

80fn build_metric_exporter(cfg: &ObservabilityConfig) -> Result<MetricExporter> {
81    let builder = match cfg.protocol {
82        OtelProtocol::Grpc => MetricExporter::builder()
83            .with_tonic()

85            .with_protocol(otlp_protocol(cfg.protocol))
86            .with_timeout(cfg.timeout)

92            .with_timeout(cfg.timeout)

98/// RAII guard returned by [`init`](crate::observability::init).
99///
100/// Dropping it force-flushes and shuts down the trace and metric providers so
101/// buffered telemetry is exported before the process exits. An *inert* guard
102/// (built when observability is disabled) does nothing on drop.
103pub struct ObservabilityGuard {
104    tracer_provider: Option<SdkTracerProvider>,
105    meter_provider: Option<SdkMeterProvider>,
106}
107
108impl std::fmt::Debug for ObservabilityGuard {
109    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
110        f.debug_struct("ObservabilityGuard")
111            .field("active", &self.tracer_provider.is_some())
112            .finish()
113    }
114}
115
116impl ObservabilityGuard {
117    /// An inert guard that installs and flushes nothing.
118    pub(crate) fn inert() -> Self {
119        Self {
120            tracer_provider: None,
121            meter_provider: None,
122        }
123    }
124
125    /// Whether this guard owns live exporters.
126    pub fn is_active(&self) -> bool {
127        self.tracer_provider.is_some() || self.meter_provider.is_some()
128    }
129
130    /// Explicitly flush all pending telemetry without shutting down. Useful for
131    /// tests; a normal program can just rely on the drop behaviour.
132    pub fn force_flush(&self) {
133        if let Some(tp) = &self.tracer_provider {
134            let _ = tp.force_flush();
135        }
136        if let Some(mp) = &self.meter_provider {
137            let _ = mp.force_flush();
138        }
139    }
140}
141
142impl Drop for ObservabilityGuard {
143    fn drop(&mut self) {
144        if let Some(tp) = &self.tracer_provider {
145            let _ = tp.force_flush();
146            let _ = tp.shutdown();
147        }
148        if let Some(mp) = &self.meter_provider {
149            let _ = mp.force_flush();
150            let _ = mp.shutdown();
151            METRICS_ACTIVE.store(false, Ordering::Relaxed);
152        }
153    }
154}

156/// Initialise OTLP trace + metric export and install the global providers.
157///
158/// Returns an inert [`ObservabilityGuard`] when `cfg.enabled == false` (no
159/// exporters, no global providers touched). Otherwise installs SDK providers as
160/// the OpenTelemetry globals and returns a guard that flushes on drop.
161///
162/// Note: this does NOT install a `tracing` subscriber. Callers compose their own
163/// subscriber and add [`tracing_layer`] to it — see the module docs on
164/// [`crate::observability`].
165pub fn init(cfg: ObservabilityConfig) -> Result<ObservabilityGuard> {
166    if !cfg.enabled {
167        return Ok(ObservabilityGuard::inert());
168    }
169
170    let resource = build_resource(&cfg);
171
172    // Tracing pipeline.
173    let span_exporter = build_span_exporter(&cfg)?;
174    let tracer_provider = SdkTracerProvider::builder()
175        .with_batch_exporter(span_exporter)
176        .with_sampler(build_sampler(cfg.sampling_ratio))
177        .with_resource(resource.clone())
178        .build();
179    // Set the global provider (for cross-crate context propagation) and stash a
180    // concrete clone so `tracing_layer` can build its layer from the SDK tracer
181    // (the boxed global tracer does not implement `PreSampledTracer`).
182    let _ = TRACER_PROVIDER.set(tracer_provider.clone());
183    global::set_tracer_provider(tracer_provider.clone());
184
185    // Metrics pipeline.
186    let metric_exporter = build_metric_exporter(&cfg)?;
187    let reader = PeriodicReader::builder(metric_exporter).build();
188    let meter_provider = SdkMeterProvider::builder()
189        .with_reader(reader)
190        .with_resource(resource)
191        .build();
192    global::set_meter_provider(meter_provider.clone());
193    METRICS_ACTIVE.store(true, Ordering::Relaxed);
194
195    Ok(ObservabilityGuard {
196        tracer_provider: Some(tracer_provider),
197        meter_provider: Some(meter_provider),
198    })
199}

211/// Return the `tracing` layer that bridges spans/events into OpenTelemetry, or
212/// `None` if observability has not been initialised.
213///
214/// **Call [`init`] before composing this layer.** The layer is bound at
215/// construction time to the exporting tracer provider that `init` installs, so
216/// it can only be built once that provider exists. When `init` has not run (or
217/// ran with a disabled config), this returns `None` — an honestly inert result —
218/// rather than a layer permanently bound to a non-exporting provider. `Option<L>`
219/// itself implements `Layer`, so the `None` case is a no-op and callers compose
220/// it directly:
221///
222/// ```ignore
223/// use tracing_subscriber::prelude::*;
224/// let _guard = cqlite_core::observability::init(cfg)?;   // installs the provider
225/// tracing_subscriber::registry()
226///     .with(tracing_subscriber::fmt::layer())
227///     .with(tracing_subscriber::EnvFilter::from_default_env())
228///     .with(cqlite_core::observability::tracing_layer())  // Some(..) after init, else None
229///     .init();
230/// ```
231pub fn tracing_layer<S>() -> Option<impl tracing_subscriber::Layer<S>>
232where
233    S: tracing::Subscriber + for<'span> tracing_subscriber::registry::LookupSpan<'span>,
234{
235    // Only build a layer once `init` has installed the exporting provider. No
236    // ephemeral fallback: a layer bound to a throwaway no-export provider can
237    // never start exporting, so returning `None` (inert) is the correct
238    // behaviour when uninitialised.
239    let provider = TRACER_PROVIDER.get()?;
240    let tracer = provider.tracer(SCOPE);
241    Some(tracing_opentelemetry::layer().with_tracer(tracer))
242}

513/// Add to a u64 counter identified by a catalog name.
514///
515/// Catalog names use the cached instrument; unknown names build an ad-hoc
516/// counter so call sites never silently drop data (catalog names should always
517/// be used).
518pub(crate) fn add_counter(name: &'static str, value: u64, attributes: &[KeyValue]) {
519    let i = instruments();
520    let counter = match name {
521        catalog::READ_ROWS => &i.read_rows,
522        catalog::READ_BYTES => &i.read_bytes,
523        catalog::READ_PARTITIONS => &i.read_partitions,
524        catalog::READ_PARTITION_LOOKUP => &i.read_partition_lookup,
525        catalog::READ_BLOOM_CHECKS => &i.read_bloom_checks,
526        catalog::READ_SCAN_WINDOW_REFILL => &i.read_scan_window_refill,
527        catalog::STORAGE_OPEN_SSTABLES => &i.storage_open_sstables,
528        catalog::STORAGE_OPEN_BYTES => &i.storage_open_bytes,
529        catalog::STORAGE_OPEN_TABLES => &i.storage_open_tables,
530        catalog::QUERY_ROWS => &i.query_rows,
531        catalog::QUERY_ROWS_SCANNED => &i.query_rows_scanned,
532        catalog::ERRORS_TOTAL => &i.errors_total,
533        catalog::WRITE_MUTATIONS => &i.write_mutations,
534        catalog::FLUSH_ROWS => &i.flush_rows,
535        catalog::FLUSH_BYTES => &i.flush_bytes,
536        catalog::FLUSH_SSTABLES => &i.flush_sstables,
537        catalog::WRITE_PARTITIONS => &i.write_partitions,
538        catalog::WRITE_BYTES => &i.write_bytes,
539        catalog::COMPACTION_ROWS_MERGED => &i.compaction_rows_merged,
540        catalog::COMPACTION_BYTES_WRITTEN => &i.compaction_bytes_written,
541        catalog::COMPACTION_SSTABLES_IN => &i.compaction_sstables_in,
542        catalog::COMPACTION_SSTABLES_OUT => &i.compaction_sstables_out,
543        catalog::COMPACTION_TOMBSTONES_PURGED => &i.compaction_tombstones_purged,
544        catalog::RPC_REQUESTS => &i.rpc_requests,
545        catalog::RPC_ROWS => &i.rpc_rows,
546        catalog::RPC_BYTES => &i.rpc_bytes,
547        _ => {
548            meter().u64_counter(name).build().add(value, attributes);
549            return;
550        }
551    };
552    counter.add(value, attributes);
553}

555/// Record into an f64 histogram identified by a catalog name.
556pub(crate) fn record_histogram(name: &'static str, value: f64, attributes: &[KeyValue]) {
557    let i = instruments();
558    let hist = match name {
559        catalog::READ_DURATION => &i.read_duration,
560        catalog::QUERY_DURATION => &i.query_duration,
561        catalog::COMPACTION_DURATION => &i.compaction_duration,
562        catalog::WAL_SYNC_DURATION => &i.wal_sync_duration,
563        catalog::FLUSH_DURATION => &i.flush_duration,
564        catalog::COMPRESSION_RATIO => &i.compression_ratio,
565        catalog::COMPACTION_FINALIZE_DURATION => &i.compaction_finalize_duration,
566        catalog::COMPACTION_BUDGET_REQUESTED => &i.compaction_budget_requested,
567        catalog::COMPACTION_BUDGET_CONSUMED => &i.compaction_budget_consumed,
568        catalog::RPC_DURATION => &i.rpc_duration,
569        _ => {
570            meter()
571                .f64_histogram(name)
572                .build()
573                .record(value, attributes);
574            return;
575        }
576    };
577    hist.record(value, attributes);
578}

607/// Extract a W3C `traceparent` header and set the resulting remote context as
608/// the parent of `span`. No-op when `traceparent` is absent/empty/unparseable.
609///
610/// Uses the standard [`TraceContextPropagator`] to parse the header into an
611/// OpenTelemetry [`Context`](opentelemetry::Context), then
612/// `tracing-opentelemetry`'s `set_parent` to link the `tracing` span to that
613/// remote span. Only the `traceparent` header is consulted (no baggage /
614/// tracestate), keeping the surface minimal and the behaviour identical to
615/// other CQLite hosts.
616pub(crate) fn set_span_parent_from_traceparent(span: &tracing::Span, traceparent: Option<&str>) {
617    use opentelemetry::propagation::{Extractor, TextMapPropagator};
618    use opentelemetry::trace::TraceContextExt;
619    use opentelemetry_sdk::propagation::TraceContextPropagator;
620    use tracing_opentelemetry::OpenTelemetrySpanExt;
621
622    let header = match traceparent {
623        Some(h) if !h.trim().is_empty() => h,
624        _ => return,
625    };
626
627    /// Single-header carrier exposing only `traceparent` to the propagator.
628    struct TraceParentCarrier<'a>(&'a str);
629    impl Extractor for TraceParentCarrier<'_> {
630        fn get(&self, key: &str) -> Option<&str> {
631            if key.eq_ignore_ascii_case("traceparent") {
632                Some(self.0)
633            } else {
634                None
635            }
636        }
637        fn keys(&self) -> Vec<&str> {
638            vec!["traceparent"]
639        }
640    }
641
642    let propagator = TraceContextPropagator::new();
643    let cx = propagator.extract(&TraceParentCarrier(header));
644    // Only re-parent when extraction produced a valid remote span context;
645    // otherwise leave the span attached to its in-process parent.
646    if cx.span().span_context().is_valid() {
647        span.set_parent(cx);
648    }
649}
