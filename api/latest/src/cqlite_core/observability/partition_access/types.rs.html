1//! Value types for the bounded partition access-distribution probe (issue #2827):
2//! the closed bucket/size-source label sets, the per-access byte weight, and the
3//! summary of one closed measurement window.
4//!
5//! Split out of `mod.rs` to keep each file inside the campsite-rule source target
6//! (#1116): this file owns the vocabulary the probe EMITS, while `mod.rs` owns the
7//! recorder that produces it and `table.rs` owns the counting structure.

9/// The six repeat-access buckets, verbatim as specified. A closed set — the whole
10/// cardinality budget of the emitted series.
11#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
12pub enum RepeatBucket {
13    /// Accessed exactly once in the window.
14    One,
15    /// Accessed exactly twice.
16    Two,
17    /// Accessed 3 or 4 times.
18    ThreeToFour,
19    /// Accessed 5–8 times.
20    FiveToEight,
21    /// Accessed 9–16 times.
22    NineToSixteen,
23    /// Accessed 17 or more times.
24    SeventeenPlus,
25}
26
27impl RepeatBucket {
28    /// Every bucket, in ascending order.
29    pub const ALL: [RepeatBucket; 6] = [
30        RepeatBucket::One,
31        RepeatBucket::Two,
32        RepeatBucket::ThreeToFour,
33        RepeatBucket::FiveToEight,
34        RepeatBucket::NineToSixteen,
35        RepeatBucket::SeventeenPlus,
36    ];
37
38    /// The bounded attribute value for `cqlite.read.repeat_bucket`.
39    pub fn label(self) -> &'static str {
40        match self {
41            RepeatBucket::One => "1",
42            RepeatBucket::Two => "2",
43            RepeatBucket::ThreeToFour => "3-4",
44            RepeatBucket::FiveToEight => "5-8",
45            RepeatBucket::NineToSixteen => "9-16",
46            RepeatBucket::SeventeenPlus => "17+",
47        }
48    }
49
50    /// Classify a repeat count. `0` is not a valid input (an entry exists only
51    /// because it was accessed at least once) and is classified as [`Self::One`].
52    pub fn from_count(count: u32) -> Self {
53        match count {
54            0 | 1 => RepeatBucket::One,
55            2 => RepeatBucket::Two,
56            3..=4 => RepeatBucket::ThreeToFour,
57            5..=8 => RepeatBucket::FiveToEight,
58            9..=16 => RepeatBucket::NineToSixteen,
59            _ => RepeatBucket::SeventeenPlus,
60        }
61    }
62
63    pub(super) fn index(self) -> usize {
64        match self {
65            RepeatBucket::One => 0,
66            RepeatBucket::Two => 1,
67            RepeatBucket::ThreeToFour => 2,
68            RepeatBucket::FiveToEight => 3,
69            RepeatBucket::NineToSixteen => 4,
70            RepeatBucket::SeventeenPlus => 5,
71        }
72    }
73}

75/// Provenance of an access's on-disk byte weight — the closed
76/// `cqlite.read.size_source` value set of THREE values.
77///
78/// A reader must always be able to tell a MEASURED weight from a weight the index
79/// handed over, and both from a genuinely unknown one — so the provenance is a
80/// distinct label rather than being folded into `index`.
81#[derive(Clone, Copy, Debug, PartialEq, Eq)]
82pub enum SizeSource {
83    /// Every SSTable resolved for the access reported a size directly in its index
84    /// metadata (`PartitionLoc.data_size`). No Cassandra 5.0 index format records
85    /// one, so this is unreachable for Cassandra-written SSTables today; it is kept
86    /// because a producer that genuinely knows a size must not be forced to report
87    /// a measured one.
88    Index,
89    /// The weight was MEASURED as the partition's successor gap —
90    /// `[data_offset, successor_offset)`, bounding to the authoritative uncompressed
91    /// data-section length for the last partition. Authoritative index-layout
92    /// metadata, the same bound the single-partition seek uses to size its decode
93    /// window; never an estimate.
94    SuccessorGap,
95    /// At least one resolved SSTable yielded no authoritative extent at all. The
96    /// access is still counted; it contributes ZERO bytes.
97    Unavailable,
98}
99
100impl SizeSource {
101    /// Every value, in order.
102    pub const ALL: [SizeSource; 3] = [
103        SizeSource::Index,
104        SizeSource::SuccessorGap,
105        SizeSource::Unavailable,
106    ];
107
108    /// The bounded attribute value for `cqlite.read.size_source`.
109    pub fn label(self) -> &'static str {
110        match self {
111            SizeSource::Index => "index",
112            SizeSource::SuccessorGap => "successor_gap",
113            SizeSource::Unavailable => "unavailable",
114        }
115    }
116}

118/// The on-disk byte weight of one logical partition access.
119///
120/// **`Unavailable` fails closed and is never filled in.** A resolution that yields
121/// no size — BTI trie resolution records only an offset
122/// (`PartitionLoc::offset_only`, `data_size = 0`), and see the reachability note
123/// below — has no authoritative weight. Such an access is still COUNTED
124/// as a partition access — dropping it would make the histogram itself wrong — but
125/// it contributes ZERO bytes and is reported under
126/// `distinct_partitions{cqlite.read.size_source="unavailable"}`, so an incomplete
127/// byte total always has a visible `unavailable` series beside it and the decision
128/// procedure can refuse the window. A size is never estimated, interpolated from a
129/// successor offset, or defaulted to a nominal value (no-heuristics, #28).
130///
131/// # Which variant a Cassandra-written SSTable produces
132///
133/// [`AccessWeight::SuccessorGap`], essentially always. Neither Cassandra 5.0 index
134/// format records a per-partition size — a BIG index entry is
135/// `[key][data_offset vint][promoted_index_len vint][promoted_index]`
136/// (`docs/sstables-definitive-guide/chapters/06-index-and-summary.md`, "Index.db
137/// Entry Format"), and the BTI `Partitions.db` trie resolves an offset only — which
138/// is why the reader's own seek path bounds a partition with the SUCCESSOR offset
139/// and treats that as authoritative. The probe measures the same extent.
140/// [`AccessWeight::Index`] therefore stays unreachable for such SSTables and exists
141/// so a producer that genuinely knows a size is never forced to report a measured
142/// one.
143#[derive(Clone, Copy, Debug, PartialEq, Eq)]
144pub enum AccessWeight {
145    /// On-disk bytes taken directly from index-recorded sizes, summed across the
146    /// SSTables resolved for this one logical access.
147    Index(u64),
148    /// On-disk bytes MEASURED as the successor gap, summed across the SSTables
149    /// resolved for this one logical access.
150    SuccessorGap(u64),
151    /// No authoritative extent was available for at least one resolved SSTable.
152    Unavailable,
153}
154
155impl AccessWeight {
156    pub(super) fn bytes(self) -> Option<u64> {
157        match self {
158            AccessWeight::Index(b) | AccessWeight::SuccessorGap(b) => Some(b),
159            AccessWeight::Unavailable => None,
160        }
161    }
162
163    pub(super) fn source(self) -> SizeSource {
164        match self {
165            AccessWeight::Index(_) => SizeSource::Index,
166            AccessWeight::SuccessorGap(_) => SizeSource::SuccessorGap,
167            AccessWeight::Unavailable => SizeSource::Unavailable,
168        }
169    }
170}

172/// Accumulates the byte weight of ONE logical partition access across the SSTables
173/// that access resolved.
174///
175/// Fails closed in both directions: an accumulator that saw no authoritative size
176/// at all finishes as [`AccessWeight::Unavailable`] (an access with nothing to
177/// price is not an access priced at zero), and a single unsized SSTable poisons the
178/// whole access.
179#[derive(Clone, Copy, Debug, Default)]
180pub struct AccessWeightBuilder {
181    bytes: u64,
182    sized: u32,
183    measured: bool,
184    unavailable: bool,
185}
186
187impl AccessWeightBuilder {
188    /// A fresh accumulator for one logical access.
189    pub fn new() -> Self {
190        Self::default()
191    }
192
193    /// Record that one resolved SSTable reported `data_size` on-disk bytes for this
194    /// partition. A `data_size` of `0` is NOT a size — it is how BTI resolution
195    /// records "the trie knows the offset and nothing else" — so it is folded in as
196    /// [`Self::note_unsized`].
197    pub fn note_sized(&mut self, data_size: u32) {
198        if data_size == 0 {
199            self.note_unsized();
200            return;
201        }
202        self.bytes = self.bytes.saturating_add(u64::from(data_size));
203        self.sized = self.sized.saturating_add(1);
204    }
205
206    /// Record that one resolved SSTable's extent was MEASURED as its successor gap
207    /// (`[data_offset, successor_offset)`, or the data-section-length bound for the
208    /// last partition). A zero-length gap is not an extent and folds to
209    /// [`Self::note_unsized`].
210    pub fn note_measured(&mut self, gap_bytes: u64) {
211        if gap_bytes == 0 {
212            self.note_unsized();
213            return;
214        }
215        self.bytes = self.bytes.saturating_add(gap_bytes);
216        self.sized = self.sized.saturating_add(1);
217        self.measured = true;
218    }
219
220    /// Record that one resolved SSTable yielded no authoritative extent.
221    pub fn note_unsized(&mut self) {
222        self.unavailable = true;
223    }
224
225    /// Finish the accumulation.
226    ///
227    /// Provenance is the WEAKEST of the contributions: a total that mixes an
228    /// index-recorded size with a measured gap is reported as measured, because the
229    /// total is only as well-founded as its weakest component. Any unpriceable
230    /// contribution — or no contribution at all — makes the whole access
231    /// unavailable.
232    pub fn finish(self) -> AccessWeight {
233        if self.unavailable || self.sized == 0 {
234            AccessWeight::Unavailable
235        } else if self.measured {
236            AccessWeight::SuccessorGap(self.bytes)
237        } else {
238            AccessWeight::Index(self.bytes)
239        }
240    }
241}

243/// Per-bucket totals for one closed window.
244#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
245pub struct BucketStats {
246    /// Distinct partitions in this bucket whose bytes came from an index-recorded
247    /// size.
248    pub distinct_index: u64,
249    /// Distinct partitions in this bucket whose bytes were MEASURED as the
250    /// successor gap.
251    pub distinct_successor_gap: u64,
252    /// Distinct partitions in this bucket whose bytes could not be priced at all.
253    pub distinct_unavailable: u64,
254    /// Sum of the repeat counts of every partition in this bucket.
255    pub accesses: u64,
256    /// Sum of DISTINCT-partition on-disk bytes in this bucket (unavailable
257    /// partitions contribute zero, by construction).
258    pub bytes: u64,
259}
260
261impl BucketStats {
262    /// Distinct partitions in this bucket, priced or not.
263    pub fn distinct(&self) -> u64 {
264        self.distinct_index + self.distinct_successor_gap + self.distinct_unavailable
265    }
266
267    /// Distinct partitions in this bucket that carry an authoritative extent.
268    pub fn distinct_priced(&self) -> u64 {
269        self.distinct_index + self.distinct_successor_gap
270    }
271}

273/// The summary of one CLOSED measurement window — the complete input to the
274/// decision procedure.
275#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
276pub struct WindowSummary {
277    pub(super) buckets: [BucketStats; 6],
278    /// `2^k` for the sampling prefix width `k` in force at close. `1` = census.
279    pub sample_denominator: u64,
280    /// The recorder hit its sampling floor: the surviving sample is too small to
281    /// be worth anything and the decision procedure refuses the window.
282    pub at_sampling_floor: bool,
283    /// Every access the recorder was asked to record, including accesses to keys
284    /// the sampling predicate did not admit. Always `>= total_accesses()`.
285    pub recorded_accesses: u64,
286    /// Monotonic close sequence, assigned under the recorder's write lock at the
287    /// moment this window closed.
288    ///
289    /// Windows close atomically but their series are EMITTED after the lock is
290    /// released, so with several recording threads window `N`'s emit can be overtaken
291    /// by `N+1`'s. That is harmless for the additive counter families, but the three
292    /// unlabelled scalars — `sample_denominator`, `sampling_floor` and
293    /// `window_dropped_accesses` — are gauges, and a late emit of an older window
294    /// would leave them describing it. Since `sampling_floor` and
295    /// `window_dropped_accesses` are the "was the LAST CLOSED window clean" signals
296    /// (normative in the catalog docs and the spec), that would invert exactly the
297    /// property they exist to provide.
298    ///
299    /// Each recorder therefore keeps its OWN high-water mark and skips the gauge
300    /// writes for a stale sequence, holding that mark's lock across the comparison
301    /// AND all three writes. Both properties are load-bearing: a process-wide mark
302    /// would let independent recorders suppress each other's gauges (silently — a
303    /// suppressed gauge is an ABSENT series), and comparing without holding the lock
304    /// through the writes would let an admitted older emitter be preempted, let a
305    /// newer one write, then resume and overwrite with stale values. See
306    /// `PartitionAccessRecorder::emit_window`.
307    ///
308    /// This sequence is per-recorder and monotonic within one recorder; two
309    /// independent recorders number independently, which is exactly why the mark
310    /// cannot be shared.
311    pub close_sequence: u64,
312    /// Accesses that could NOT be landed in the counting table at all: it was at its
313    /// load factor with the sampling prefix already at its cap, so no slot could be
314    /// claimed.
315    ///
316    /// Distinct from an access whose key the sampling predicate declined to admit —
317    /// that is the sample working as designed and is unbiased. This is input LOST,
318    /// and a measurement instrument must never lose input silently, so it is
319    /// reported. Non-zero only in a window that also reports
320    /// [`Self::at_sampling_floor`], which the decision procedure already refuses.
321    pub dropped_accesses: u64,
322}
323
324impl WindowSummary {
325    /// Stats for one bucket.
326    pub fn bucket(&self, b: RepeatBucket) -> BucketStats {
327        self.buckets[b.index()]
328    }
329
330    /// `A = Σ a_b` — accesses attributable to the admitted sample.
331    pub fn total_accesses(&self) -> u64 {
332        self.buckets.iter().map(|b| b.accesses).sum()
333    }
334
335    /// Distinct partitions in the admitted sample.
336    pub fn distinct_partitions(&self) -> u64 {
337        self.buckets.iter().map(|b| b.distinct()).sum()
338    }
339
340    /// Distinct partitions whose on-disk bytes could not be priced.
341    pub fn unavailable_partitions(&self) -> u64 {
342        self.buckets.iter().map(|b| b.distinct_unavailable).sum()
343    }
344
345    /// Total distinct-partition on-disk bytes across every bucket.
346    pub fn total_bytes(&self) -> u64 {
347        self.buckets.iter().map(|b| b.bytes).sum()
348    }
349
350    /// A census window counted every distinct partition it saw.
351    pub fn is_census(&self) -> bool {
352        self.sample_denominator == 1
353    }
354
355    /// Fraction of distinct partitions whose bytes could not be priced. `0.0` for
356    /// an empty window.
357    pub fn unavailable_fraction(&self) -> f64 {
358        let total = self.distinct_partitions();
359        if total == 0 {
360            return 0.0;
361        }
362        self.unavailable_partitions() as f64 / total as f64
363    }
