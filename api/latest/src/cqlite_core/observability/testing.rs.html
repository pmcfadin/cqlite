1//! Shared in-memory OpenTelemetry capture harness for observability tests
2//! (epic #1031, issue #1043).
3//!
4//! This module is the **single shared fixture** every observability-correctness
5//! test (and future child issue) reuses. It installs in-memory OTLP exporters,
6//! runs a flow, force-flushes, and hands back the captured spans and metrics so a
7//! test can assert the expected span tree, metric names/units, and bounded
8//! attribute sets — all deterministically, with no collector, network, or timing
9//! dependence.
10//!
11//! It is gated behind the `observability` feature (it needs the OTel SDK) and is
12//! intended for tests, so it lives under `#[cfg(feature = "observability")]` and
13//! depends on `opentelemetry_sdk`'s in-memory exporters (available because the
14//! crate enables the SDK's `testing` feature as a dev-dependency; the
15//! library/default build links no OTel at all).
16//!
17//! # Why a single process-global meter provider
18//!
19//! The metric helpers in [`crate::observability`] record through
20//! `opentelemetry::global::meter("cqlite")`, and the concrete `Meter` is cached
21//! in a `OnceLock` the first time any metric is recorded in the process. So the
22//! global meter provider must be installed **once, before the first metric
23//! record**, and cannot be swapped per-test. [`metrics_capture`] therefore
24//! returns a handle to a single process-wide [`InMemoryMetricExporter`]; tests
25//! call [`MetricsCapture::reset`] + [`MetricsCapture::flush_and_collect`] around
26//! their own flow rather than installing a fresh provider each time.
27//!
28//! ## Metric isolation guarantee
29//!
30//! The shared exporter is configured with **DELTA** temporality
31//! ([`Temporality::Delta`]). Each [`MetricsCapture::flush_and_collect`] therefore
32//! reports only the values aggregated *since the previous collect*, not a
33//! running cumulative total. [`MetricsCapture::reset`] force-flushes (draining and
34//! discarding any prior deltas) and clears the exporter buffer, so a subsequent
35//! `flush_and_collect` reflects **only the work done between `reset` and
36//! `flush_and_collect`** — never metrics accumulated by an earlier flow or test.
37//! This holds even under parallel test execution: although the meter provider is
38//! process-global, all metric assertions run inside a single serial test (see
39//! `tests/observability_correctness.rs`), and DELTA temporality means that test
40//! never re-observes another flow's already-collected counters.
41//!
42//! Spans, by contrast, flow through `tracing`, so a test can install its own
43//! tracer + subscriber for the duration of a flow via [`capture_spans`] and get
44//! perfectly isolated span trees with no global state.
45//!
46//! # Typical use
47//!
48//! ```ignore
49//! use cqlite_core::observability::{self as obs, catalog};
50//! use cqlite_core::observability::testing;
51//!
52//! // Spans: scoped, isolated per call.
53//! let spans = testing::capture_spans(|| {
54//!     obs::record_histogram(catalog::QUERY_DURATION, 0.001, &[]);
55//!     // ... run an instrumented flow inside a `tracing` span ...
56//! });
57//! assert!(spans.iter().any(|s| s.name == "query.execute"));
58//!
59//! // Metrics: process-global, reset around the flow.
60//! let mc = testing::metrics_capture();
61//! mc.reset();
62//! obs::add_counter(catalog::READ_ROWS, 7, &[]);
63//! let metrics = mc.flush_and_collect();
64//! assert_eq!(metrics.counter_sum(catalog::READ_ROWS), 7);
65//! ```

87/// A captured, simplified view of an exported span. Wraps the SDK [`SpanData`]
88/// with ergonomic accessors so tests read span trees without depending on the
89/// SDK's exact field layout.
90#[derive(Debug, Clone)]
91pub struct CapturedSpan {
92    /// Span name (`#[tracing::instrument]` name or `info_span!` name).
93    pub name: String,
94    /// The raw SDK span data, for advanced assertions (timing, events, links).
95    pub data: SpanData,
96}
97
98impl CapturedSpan {
99    /// This span's own 16-byte span id.
100    pub fn span_id(&self) -> opentelemetry::trace::SpanId {
101        self.data.span_context.span_id()
102    }
103
104    /// This span's parent span id (`SpanId::INVALID` for a root span).
105    pub fn parent_span_id(&self) -> opentelemetry::trace::SpanId {
106        self.data.parent_span_id
107    }
108
109    /// The trace id this span belongs to.
110    pub fn trace_id(&self) -> opentelemetry::trace::TraceId {
111        self.data.span_context.trace_id()
112    }
113
114    /// Look up a bounded span attribute value (as a display string) by key.
115    pub fn attribute(&self, key: &str) -> Option<String> {
116        self.data
117            .attributes
118            .iter()
119            .find(|kv| kv.key.as_str() == key)
120            .map(|kv| kv.value.as_str().into_owned())
121    }
122
123    /// Whether the span's status was set to error (via `record_error` /
124    /// `mark_span_error`).
125    pub fn is_error(&self) -> bool {
126        matches!(self.data.status, opentelemetry::trace::Status::Error { .. })
127            || self.attribute("otel.status_code").as_deref() == Some("ERROR")
128    }
129}

131/// A captured set of spans from one [`capture_spans`] call, with tree helpers.
132#[derive(Debug, Clone, Default)]
133pub struct CapturedSpans {
134    spans: Vec<CapturedSpan>,
135}
136
137impl CapturedSpans {
138    /// Build a [`CapturedSpans`] from already-exported `(name, SpanData)` pairs.
139    ///
140    /// Lets a test that drives its own tracer (e.g. with a custom sampler) reuse
141    /// the tree/name helpers here instead of duplicating them.
142    pub fn from_raw(raw: Vec<(String, SpanData)>) -> Self {
143        Self {
144            spans: raw
145                .into_iter()
146                .map(|(name, data)| CapturedSpan { name, data })
147                .collect(),
148        }
149    }
150
151    /// All captured spans, in export order.
152    pub fn all(&self) -> &[CapturedSpan] {
153        &self.spans
154    }
155
156    /// Iterate the captured spans.
157    pub fn iter(&self) -> std::slice::Iter<'_, CapturedSpan> {
158        self.spans.iter()
159    }
160
161    /// Find the first span with the given name.
162    pub fn find(&self, name: &str) -> Option<&CapturedSpan> {
163        self.spans.iter().find(|s| s.name == name)
164    }
165
166    /// Whether any captured span has the given name.
167    pub fn contains(&self, name: &str) -> bool {
168        self.find(name).is_some()
169    }
170
171    /// Whether `child` is a direct child of `parent` (matched by name): there
172    /// exists a span named `parent` whose span id equals a span named `child`'s
173    /// parent span id, sharing the same trace.
174    pub fn is_parent_of(&self, parent: &str, child: &str) -> bool {
175        let parents: Vec<_> = self.spans.iter().filter(|s| s.name == parent).collect();
176        self.spans.iter().filter(|s| s.name == child).any(|c| {
177            parents
178                .iter()
179                .any(|p| p.span_id() == c.parent_span_id() && p.trace_id() == c.trace_id())
180        })
181    }
182}

184/// Run `flow` with an isolated in-memory tracer installed as the default
185/// `tracing` subscriber, then return every span it produced.
186///
187/// The tracer + subscriber are scoped to this call (via
188/// [`tracing::subscriber::with_default`]), so concurrent or subsequent calls do
189/// not see each other's spans. A [`SimpleSpanProcessor`] exports each span
190/// synchronously as it closes, and the provider is force-flushed before reading,
191/// so the result is deterministic with no timing dependence.
192///
193/// Spans only close (and thus export) when their guard is dropped — make sure
194/// `flow` fully completes any instrumented work (e.g. `block_on` an async flow)
195/// before returning.
196pub fn capture_spans<F, R>(flow: F) -> CapturedSpans
197where
198    F: FnOnce() -> R,
199{
200    let exporter = InMemorySpanExporter::default();
201    let provider = SdkTracerProvider::builder()
202        .with_simple_exporter(exporter.clone())
203        .build();
204    let tracer = provider.tracer(SCOPE);
205    let layer = tracing_opentelemetry::layer().with_tracer(tracer);
206    let subscriber = tracing_subscriber::registry().with(layer);
207
208    tracing::subscriber::with_default(subscriber, flow);
209
210    // Ensure everything is flushed before we read.
211    let _ = provider.force_flush();
212
213    let spans = exporter
214        .get_finished_spans()
215        .expect("in-memory span exporter must yield finished spans")
216        .into_iter()
217        .map(|data| CapturedSpan {
218            name: data.name.to_string(),
219            data,
220        })
221        .collect();
222    let _ = provider.shutdown();
223    CapturedSpans { spans }
224}

230/// Process-wide in-memory metric exporter handle.
231///
232/// There is exactly one of these per process because the production metric
233/// helpers bind a single global `Meter` on first use (see the module docs).
234/// Obtain it via [`metrics_capture`]; [`reset`](Self::reset) before your flow and
235/// [`flush_and_collect`](Self::flush_and_collect) after it.
236#[derive(Clone)]

242/// A snapshot of metrics collected by [`MetricsCapture::flush_and_collect`],
243/// with name/unit/value accessors for assertions.
244#[derive(Debug, Default)]
245pub struct CapturedMetrics {
246    entries: Vec<MetricEntry>,
247}
248
249/// A single collected metric: its name, unit, kind, and aggregated value(s).
250#[derive(Debug, Clone)]
251pub struct MetricEntry {
252    /// Instrument name (a `catalog::*` constant).
253    pub name: String,
254    /// Instrument unit (a `catalog::unit::*` string).
255    pub unit: String,
256    /// The aggregated data points.
257    pub points: Vec<MetricPoint>,
258}
259
260/// One aggregated data point: its numeric value (as f64) and bounded attributes.
261#[derive(Debug, Clone)]
262pub struct MetricPoint {
263    /// The aggregated value (counter sum, gauge value, or histogram sum).
264    pub value: f64,
265    /// The bounded attributes as `(key, value-as-string)` pairs.
266    pub attributes: Vec<(String, String)>,
267}
268
269impl CapturedMetrics {
270    /// Every collected metric entry.
271    pub fn entries(&self) -> &[MetricEntry] {
272        &self.entries
273    }
274
275    /// Find a collected metric by name.
276    pub fn find(&self, name: &str) -> Option<&MetricEntry> {
277        self.entries.iter().find(|m| m.name == name)
278    }
279
280    /// Whether a metric with the given name was collected.
281    pub fn contains(&self, name: &str) -> bool {
282        self.find(name).is_some()
283    }
284
285    /// The unit string reported for a metric, if collected.
286    pub fn unit(&self, name: &str) -> Option<&str> {
287        self.find(name).map(|m| m.unit.as_str())
288    }
289
290    /// Sum of all data-point values for a metric (counter total, histogram sum,
291    /// or gauge value), or `0.0` if the metric was not collected.
292    pub fn counter_sum(&self, name: &str) -> f64 {
293        self.find(name)
294            .map(|m| m.points.iter().map(|p| p.value).sum())
295            .unwrap_or(0.0)
296    }
297
298    /// Sum of values for the data points whose attribute set contains ALL of the
299    /// given `(key, value)` pairs. Lets a test assert, e.g., that
300    /// `cqlite.errors.total{category=…,subsystem=…}` incremented.
301    pub fn sum_where(&self, name: &str, required: &[(&str, &str)]) -> f64 {
302        let Some(m) = self.find(name) else {
303            return 0.0;
304        };
305        m.points
306            .iter()
307            .filter(|p| {
308                required
309                    .iter()
310                    .all(|(k, v)| p.attributes.iter().any(|(pk, pv)| pk == k && pv == v))
311            })
312            .map(|p| p.value)
313            .sum()
314    }
315}

317impl MetricsCapture {
318    /// Drop everything collected so far so the next
319    /// [`flush_and_collect`](Self::flush_and_collect) reflects only the flow run
320    /// after this call. Call this immediately before your flow.
321    ///
322    /// This force-flushes first (draining the SDK's pending DELTA aggregation so
323    /// any straggler deltas land in the exporter), then clears the exporter
324    /// buffer. Because the exporter uses DELTA temporality, the *next* collect
325    /// starts a fresh aggregation window — no cumulative state from before this
326    /// `reset` can leak into a later `flush_and_collect`.
327    pub fn reset(&self) {
328        // Flush first so any straggler deltas land, then clear the buffer. With
329        // DELTA temporality this also resets the aggregation window.
330        let _ = self.provider.force_flush();
331        self.exporter.reset();
332    }
333
334    /// Force the meter provider to collect + export, then return a snapshot.
335    pub fn flush_and_collect(&self) -> CapturedMetrics {
336        let _ = self.provider.force_flush();
337        let resource_metrics = self
338            .exporter
339            .get_finished_metrics()
340            .expect("in-memory metric exporter must yield collected metrics");
341
342        let mut entries = Vec::new();
343        for rm in &resource_metrics {
344            for sm in rm.scope_metrics() {
345                for metric in sm.metrics() {
346                    entries.push(metric_to_entry(metric));
347                }
348            }
349        }
350        CapturedMetrics { entries }
351    }
352}

354/// The process-wide [`MetricsCapture`], installing the global in-memory meter
355/// provider exactly once on first call.
356///
357/// MUST be called before any code path records a catalog metric in the process,
358/// otherwise the global `Meter` binds to a no-op provider and nothing is
359/// captured. Observability tests call this at the start of the relevant test.
360pub fn metrics_capture() -> MetricsCapture {
361    static CAPTURE: OnceLock<MetricsCapture> = OnceLock::new();
362    CAPTURE
363        .get_or_init(|| {
364            // DELTA temporality so each `flush_and_collect` returns only the
365            // values recorded since the previous collect (see the module-level
366            // "Metric isolation guarantee"). This prevents an earlier flow/test's
367            // cumulative counters from re-appearing in a later collect.
368            let exporter = InMemoryMetricExporterBuilder::new()
369                .with_temporality(Temporality::Delta)
370                .build();
371            let reader = PeriodicReader::builder(exporter.clone()).build();
372            let provider = SdkMeterProvider::builder().with_reader(reader).build();
373            opentelemetry::global::set_meter_provider(provider.clone());
374            super::otel::set_metrics_active_for_testing();
375            MetricsCapture { exporter, provider }
376        })
377        .clone()
378}

380fn value_to_string(v: &Value) -> String {
381    match v {
382        Value::String(s) => s.to_string(),
383        Value::Bool(b) => b.to_string(),
384        Value::I64(i) => i.to_string(),
385        Value::F64(f) => f.to_string(),
386        other => format!("{other:?}"),
387    }
388}

390/// Convert one SDK `Metric` into the simplified [`MetricEntry`], flattening the
391/// f64/u64/i64 aggregation variants into f64 values for uniform assertions.
392fn metric_to_entry(metric: &opentelemetry_sdk::metrics::data::Metric) -> MetricEntry {
393    let mut points = Vec::new();
394    match metric.data() {
395        AggregatedMetrics::F64(data) => collect_points_f64(data, &mut points),
396        AggregatedMetrics::U64(data) => collect_points_u64(data, &mut points),
397        AggregatedMetrics::I64(data) => collect_points_i64(data, &mut points),
398    }
399    MetricEntry {
400        name: metric.name().to_string(),
401        unit: metric.unit().to_string(),
402        points,
403    }
404}
405
406macro_rules! collect_points_impl {
407    ($fn_name:ident, $ty:ty, $to_f64:expr) => {
408        fn $fn_name(data: &MetricData<$ty>, out: &mut Vec<MetricPoint>) {
409            let to_f64: fn($ty) -> f64 = $to_f64;
410            match data {
411                MetricData::Sum(sum) => {
412                    for dp in sum.data_points() {
413                        out.push(MetricPoint {
414                            value: to_f64(dp.value()),
415                            attributes: attrs(dp.attributes()),
416                        });
417                    }
418                }
419                MetricData::Gauge(gauge) => {
420                    for dp in gauge.data_points() {
421                        out.push(MetricPoint {
422                            value: to_f64(dp.value()),
423                            attributes: attrs(dp.attributes()),
424                        });
425                    }
426                }
427                MetricData::Histogram(hist) => {
428                    for dp in hist.data_points() {
429                        out.push(MetricPoint {
430                            // For histograms, expose the running sum so a test can
431                            // assert "something was recorded" and read total magnitude.
432                            value: to_f64(dp.sum()),
433                            attributes: attrs(dp.attributes()),
434                        });
435                    }
436                }
437                MetricData::ExponentialHistogram(hist) => {
438                    for dp in hist.data_points() {
439                        out.push(MetricPoint {
440                            value: to_f64(dp.sum()),
441                            attributes: attrs(dp.attributes()),
442                        });
443                    }
444                }
445            }
446        }
447    };
448}

450fn attrs<'a>(it: impl Iterator<Item = &'a opentelemetry::KeyValue>) -> Vec<(String, String)> {
451    it.map(|kv| (kv.key.as_str().to_string(), value_to_string(&kv.value)))
452        .collect()
453}

455collect_points_impl!(collect_points_f64, f64, |v| v);
456collect_points_impl!(collect_points_u64, u64, |v| v as f64);
457collect_points_impl!(collect_points_i64, i64, |v| v as f64);

463    // Span capture uses a per-call isolated tracer, so this self-test needs no
464    // global state and is safe under parallel test execution.
465    #[test]
466    fn capture_spans_records_tree_and_attributes() {
467        let spans = capture_spans(|| {
468            let parent = tracing::info_span!("parent.span");
469            let _g = parent.enter();
470            let child = tracing::info_span!("child.span", cqlite.result = "hit");
471            child.in_scope(|| {});
472        });
473
474        assert!(spans.contains("parent.span"));
475        assert!(spans.contains("child.span"));
476        assert!(
477            spans.is_parent_of("parent.span", "child.span"),
478            "child.span must nest under parent.span; saw {:?}",
479            spans.iter().map(|s| s.name.clone()).collect::<Vec<_>>()
480        );
481        let child = spans.find("child.span").expect("child present");
482        assert_eq!(child.attribute("cqlite.result").as_deref(), Some("hit"));
483    }

485    #[test]
486    fn captured_spans_from_raw_roundtrips_empty() {
487        let empty = CapturedSpans::from_raw(Vec::new());
488        assert!(empty.all().is_empty());
489        assert!(!empty.contains("anything"));
