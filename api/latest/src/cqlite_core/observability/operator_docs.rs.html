1//! Operator-facing metric reference — GENERATED source of truth (issue #2426).
2//!
3//! The developer-facing catalog ([`super::catalog`]) owns the metric *names*,
4//! *units*, and cardinality contracts. This module adds the **operator**
5//! annotation for every catalogued instrument (one-sentence meaning, bounded
6//! attribute set, healthy-vs-alarming interpretation, and the #2399 round
7//! scoreboard item it feeds) and renders a deterministic markdown reference so
8//! the AWS field team has a single consumable page that CANNOT drift from the
9//! code.
10//!
11//! # Anti-drift contract (mirrors the #1338 parity-report pattern)
12//!
13//! - The renderer's source of truth is the REAL catalog: it walks
14//!   [`super::catalog::ALL_METRICS`] and references the `catalog::attr::*`
15//!   constants, never a hand-copied name list.
16//! - Generation is **fail-closed**: a metric present in `ALL_METRICS` with no
17//!   operator annotation makes [`operator_metric_docs`] return
18//!   [`DocGenError::MissingAnnotation`] — no undocumented instrument can ship.
19//!   An annotation for a name absent from `ALL_METRICS` is likewise rejected.
20//! - The rendered markdown is committed at [`COMMITTED_DOC_REL`]; a lib test
21//!   (`operator_metrics_doc_is_fresh`) and the `operator-metrics-doc` agent-gate
22//!   component fail if the committed file drifts from a fresh render.
23//! - Regenerate with:
24//!   `cargo run -p cqlite-core --example gen_operator_metrics_doc`.
25//!
26//! Everything here is always compiled (it pulls in no OpenTelemetry types), like
27//! the catalog itself, so the generator and its freshness test build in any
28//! feature configuration.
29

56/// OpenTelemetry instrument kind, as the field team reads it on a dashboard.
57#[derive(Debug, Clone, Copy, PartialEq, Eq)]
58pub enum MetricKind {
59    /// Monotonically increasing total.
60    Counter,
61    /// Current up/down value.
62    Gauge,
63    /// Distribution of observed values (durations, sizes, ratios).
64    Histogram,
65}
66
67impl MetricKind {
68    /// Lowercase label used in the rendered reference.
69    pub fn label(self) -> &'static str {
70        match self {
71            MetricKind::Counter => "counter",
72            MetricKind::Gauge => "gauge",
73            MetricKind::Histogram => "histogram",
74        }
75    }
76}

78/// One operator annotation for a catalogued instrument.
79#[derive(Debug, Clone)]
80pub struct MetricDoc {
81    /// Metric name (a `catalog::*` constant — never a copied literal).
82    pub name: &'static str,
83    /// Instrument kind.
84    pub kind: MetricKind,
85    /// UCUM unit string.
86    pub unit: &'static str,
87    /// One operator sentence: what this instrument means in the field.
88    pub summary: &'static str,
89    /// Bounded attribute keys this metric may carry (`catalog::attr::*`), empty
90    /// when the metric carries none.
91    pub attributes: &'static [&'static str],
92    /// Healthy value vs the shape that should raise an alarm.
93    pub interpretation: &'static str,
94    /// #2399 round-scoreboard item this metric feeds, or `"—"`.
95    pub round_item: &'static str,
96}
97
98/// Error raised when the operator annotations and the catalog disagree — the
99/// fail-closed guard against an undocumented (or phantom) instrument shipping.
100#[derive(Debug, Clone, PartialEq, Eq)]
101pub enum DocGenError {
102    /// A metric in [`ALL_METRICS`] has no operator annotation.
103    MissingAnnotation(&'static str),
104    /// An annotation names a metric that is not in [`ALL_METRICS`].
105    UnknownMetric(&'static str),
106    /// An annotation names a metric more than once.
107    DuplicateAnnotation(&'static str),
108}
109
110impl std::fmt::Display for DocGenError {
111    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
112        match self {
113            DocGenError::MissingAnnotation(m) => write!(
114                f,
115                "metric `{m}` is catalogued but has no operator annotation in \
116                 operator_docs::ANNOTATIONS — no undocumented instrument may ship (#2426)"
117            ),
118            DocGenError::UnknownMetric(m) => write!(
119                f,
120                "operator annotation names `{m}`, which is not in catalog::ALL_METRICS \
121                 (stale annotation)"
122            ),
123            DocGenError::DuplicateAnnotation(m) => {
124                write!(f, "operator annotation for `{m}` appears more than once")
125            }
126        }
127    }
128}

132/// Return every catalogued instrument paired with its operator annotation, in
133/// catalog (`ALL_METRICS`) order.
134///
135/// Fail-closed: returns [`DocGenError`] if any metric lacks an annotation, if an
136/// annotation names a metric absent from `ALL_METRICS`, or if a metric is
137/// annotated twice — so no undocumented (or phantom) instrument can ship.
138pub fn operator_metric_docs() -> Result<Vec<MetricDoc>, DocGenError> {
139    // Reject duplicate annotations first (a duplicate would mask a missing one).
140    let mut seen = std::collections::HashSet::new();
141    for a in all_annotations() {
142        if !seen.insert(a.name) {
143            return Err(DocGenError::DuplicateAnnotation(a.name));
144        }
145        if !ALL_METRICS.contains(&a.name) {
146            return Err(DocGenError::UnknownMetric(a.name));
147        }
148    }
149    // Every catalogued metric must have an annotation, walked in catalog order.
150    let mut out = Vec::with_capacity(ALL_METRICS.len());
151    for name in ALL_METRICS {
152        match all_annotations().find(|a| a.name == *name) {
153            Some(a) => out.push(a.clone()),
154            None => return Err(DocGenError::MissingAnnotation(name)),
155        }
156    }
157    Ok(out)
158}

160/// The bounded attribute keys of `d`, as a markdown table cell.
161fn render_attributes(d: &MetricDoc) -> String {
162    if d.attributes.is_empty() {
163        "_(none)_".to_string()
164    } else {
165        d.attributes
166            .iter()
167            .map(|a| format!("`{a}`"))
168            .collect::<Vec<_>>()
169            .join("<br>")
170    }
171}

173/// Render the deterministic operator-facing markdown reference from the catalog.
174///
175/// Output is stable across runs (metrics sorted by name, no timestamps) so a
176/// `--check` regeneration can fail on a stale committed artifact — the #1338
177/// derived-artifact pattern.
178pub fn render_markdown() -> Result<String, DocGenError> {
179    let mut docs = operator_metric_docs()?;
180    docs.sort_by(|a, b| a.name.cmp(b.name));
181
182    // Split the catalog into the two populations an operator experiences
183    // DIFFERENTLY (issue #1705): a live OTel instrument becomes scrapeable once it
184    // records a value, a `catalog::STATS_ONLY_METRICS` entry never does — a
185    // provable difference (that list is exactly
186    // the set of catalogued names with no instrument, enforced by
187    // `stats_only_metrics_are_catalogued_and_never_otel_registered`). Rendering
188    // them under one "all instruments" heading with one total told operators to
189    // go looking on a Prometheus scrape for names that cannot be there.
190    let stats_only: std::collections::HashSet<&str> =
191        STATS_ONLY_METRICS.iter().map(|m| m.name).collect();
192    let (stats_only_docs, live): (Vec<MetricDoc>, Vec<MetricDoc>) = docs
193        .iter()
194        .cloned()
195        .partition(|d| stats_only.contains(d.name));
196
197    let mut s = String::new();
198    let mut line = |t: &str| {
199        s.push_str(t);
200        s.push('\n');
201    };
202
203    line("# CQLite Flight metrics — operator reference");
204    line("");
205    line(
206        "> GENERATED from the observability catalog \
207         (`cqlite-core/src/observability/catalog.rs` + `operator_docs.rs`) by \
208         `cargo run -p cqlite-core --example gen_operator_metrics_doc`. \
209         Do NOT edit by hand — edit the catalog/annotations and regenerate. \
210         The `operator-metrics-doc` agent-gate component fails if this file drifts \
211         from the catalog (issue #2426).",
212    );
213    line("");
214    line(
215        "Operator-facing reference for every `cqlite.*` metric name in CQLite's \
216         observability catalog, covering Arrow Flight and the storage/write/compaction \
217         paths. Names, units, and bounded attribute sets are generated from the code.",
218    );
219    line("");
220    line(
221        "**Two populations.** Most catalogued names are LIVE OTel instruments: \
222         registered with the meter, and scrapeable once they have recorded a value. \
223         The rest are **stats-only** — no instrument is ever registered, so they are \
224         never on a scrape at all; read them from the in-process \
225         `Database::stats().memory_stats` snapshot. The two are listed in separate \
226         sections below.",
227    );
228    line("");
229    line("Related: the Flight/Trino operator docs (`docs/flight-trino/`) and the round scoreboard template (issue #2399) link back to the entries here.");
230    line("");
231    line(&format!(
232        "Catalogued metrics: **{}** — **{}** live OTel instruments and **{}** \
233         stats-only (not scrapeable).",
234        docs.len(),
235        live.len(),
236        stats_only_docs.len(),
237    ));
238    line("");
239
240    line("## Live instruments");
241    line("");
242    line(
243        "Registered with the OTel meter; each appears on a Prometheus scrape / OTel \
244         collector export once it has recorded a value.",
245    );
246    line("");
247    line("| Metric | Type | Unit | Attributes | Operator meaning | Healthy vs alarming |");
248    line("|---|---|---|---|---|---|");
249    for d in &live {
250        line(&format!(
251            "| `{}` | {} | `{}` | {} | {} | {} |",
252            d.name,
253            d.kind.label(),
254            d.unit,
255            render_attributes(d),
256            d.summary,
257            d.interpretation,
258        ));
259    }
260    if live.is_empty() {
261        line("| _(none)_ | | | | | |");
262    }
263    line("");
264
265    line("## Stats-only metrics — NOT OTel instruments");
266    line("");
267    line(&format!(
268        "These **{}** catalogued names have NO OTel instrument: nothing registers them \
269         with a meter, so a Prometheus scrape or OTel collector will never show them. \
270         Read them from the in-process `Database::stats()` snapshot at the field named \
271         in the last column. This is enforced, not documentation: \
272         `catalog::STATS_ONLY_METRICS` is what this section is generated from, and the \
273         `stats_only_metrics_are_catalogued_and_never_otel_registered` guard fails the \
274         build if one of them ever does get an instrument (issue #1705).",
275        stats_only_docs.len(),
276    ));
277    line("");
278    line("| Metric | Type | Unit | Attributes | Operator meaning | Healthy vs alarming | Read it from |");
279    line("|---|---|---|---|---|---|---|");
280    for d in &stats_only_docs {
281        let field = STATS_ONLY_METRICS
282            .iter()
283            .find(|m| m.name == d.name)
284            .map(|m| format!("`Database::stats().{}`", m.stats_field))
285            .unwrap_or_else(|| "—".to_string());
286        line(&format!(
287            "| `{}` | {} | `{}` | {} | {} | {} | {} |",
288            d.name,
289            d.kind.label(),
290            d.unit,
291            render_attributes(d),
292            d.summary,
293            d.interpretation,
294            field,
295        ));
296    }
297    if stats_only_docs.is_empty() {
298        line("| _(none)_ | | | | | | |");
299    }
300    line("");
301
302    // Round-scoreboard mapping (#2399): only metrics that feed a template item.
303    line("## Round-scoreboard mapping (issue #2399)");
304    line("");
305    line("Metrics a #2399 round-template scoreboard item consumes. Round handoffs (#2367-style) link the item to its metric here instead of re-explaining it.");
306    line("");
307    line("| Metric | Scoreboard item |");
308    line("|---|---|");
309    let mut any = false;
310    for d in &docs {
311        if d.round_item != "—" {
312            any = true;
313            line(&format!("| `{}` | {} |", d.name, d.round_item));
314        }
315    }
316    if !any {
317        line("| _(none)_ | _(none)_ |");
318    }
319    line("");
320
321    Ok(s)
322}

324/// Render the published website (Astro/Starlight) variant: the same catalog

327pub fn render_website_markdown() -> Result<String, DocGenError> {
328    Ok(format!("{WEBSITE_FRONT_MATTER}{}", render_markdown()?))
329}

386    /// The rendered counts and section split must be DERIVED from the catalog
387    /// (issue #1705).
388    ///
389    /// The `operator-metrics-doc` gate component only compares the committed file
390    /// against a fresh render, so it is structurally blind to a generator whose
391    /// PROSE is false: before this, the page announced "every `cqlite.*` instrument
392    /// CQLite emits" and "Total instruments: 84" under one "All instruments"
393    /// heading, while 6 of those 84 are `catalog::STATS_ONLY_METRICS` and are
394    /// GUARANTEED (by `stats_only_metrics_are_catalogued_and_never_otel_registered`)
395    /// to have no instrument at all — the page sent operators looking on a
396    /// Prometheus scrape for six names that provably cannot appear there. That is
397    /// issue #1705's own bug class: a doc promising behaviour the code does not have.
398    ///
399    /// The counts here are computed from the catalog independently of the renderer,
400    /// so a hard-coded number in the generator reds this the moment the catalog
401    /// moves — the claim has to be derived, not asserted.
402    #[test]
403    fn rendered_counts_and_sections_are_derived_from_the_catalog() {
404        let doc = render_markdown().expect("render must succeed");
405        let stats_only: std::collections::HashSet<&str> =
406            STATS_ONLY_METRICS.iter().map(|m| m.name).collect();
407        let total = ALL_METRICS.len();
408        let n_stats_only = ALL_METRICS
409            .iter()
410            .filter(|n| stats_only.contains(*n))
411            .count();
412        let n_live = total - n_stats_only;
413
414        let expected_counts = format!(
415            "Catalogued metrics: **{total}** — **{n_live}** live OTel instruments and \
416             **{n_stats_only}** stats-only (not scrapeable)."
417        );
418        assert!(
419            doc.contains(&expected_counts),
420            "the rendered counts must be derived from the catalog; expected \
421             {expected_counts:?}"
422        );
423
424        // The retired claim must not come back: there is no single "all instruments"
425        // population, because 6 catalogued names are not instruments.
426        assert!(
427            !doc.contains("## All instruments"),
428            "a single 'All instruments' section misrepresents the stats-only names"
429        );
430        assert!(
431            !doc.contains("Total instruments:"),
432            "a single instrument total counts stats-only names as instruments"
433        );
434
435        // A synchronous OTel instrument is exported only after it records a value, so
436        // the page may promise a scrape appearance only conditionally. `COMPRESSION_RATIO`
437        // is the standing counter-example: registered, and silent forever on the
438        // production write surface, which emits no compressed SSTables (#1406).
439        assert!(
440            doc.contains("once it has recorded a value"),
441            "the live-instrument section must condition a scrape appearance on the \
442             instrument having recorded"
443        );
444        assert!(
445            !doc.contains("so these appear on a Prometheus scrape"),
446            "an unconditional 'these appear on a scrape' claim is false for a \
447             registered instrument that has never recorded"
448        );
449
450        // Each population is rendered in ITS OWN section, and no name is in both.
451        let live_head = "## Live instruments";
452        let stats_head = "## Stats-only metrics — NOT OTel instruments";
453        let live_at = doc.find(live_head).expect("live-instrument section");
454        let stats_at = doc.find(stats_head).expect("stats-only section");
455        assert!(live_at < stats_at, "sections must render in a stable order");
456        let live_section = &doc[live_at..stats_at];
457        // Bounded at the next heading: the round-scoreboard table that follows lists
458        // rows in the same `| `name` |` shape, and swallowing it would make every
459        // live metric look like it were also declared stats-only.
460        let stats_end = doc[stats_at..]
461            .find("\n## ")
462            .map(|o| stats_at + o)
463            .unwrap_or(doc.len());
464        let stats_section = &doc[stats_at..stats_end];
465
466        for name in ALL_METRICS {
467            let row = format!("| `{name}` |");
468            if stats_only.contains(name) {
469                assert!(
470                    stats_section.contains(&row),
471                    "stats-only `{name}` must be listed in the stats-only section"
472                );
473                assert!(
474                    !live_section.contains(&row),
475                    "`{name}` has no OTel instrument, so it must NOT be listed as a \
476                     live instrument"
477                );
478            } else {
479                assert!(
480                    live_section.contains(&row),
481                    "`{name}` is a live instrument and must be listed as one"
482                );
483                assert!(
484                    !stats_section.contains(&row),
485                    "`{name}` has a live instrument, so it must NOT be listed as \
486                     stats-only"
487                );
488            }
489        }
490
491        // And the stats-only rows tell the operator where to actually read them.
492        for m in STATS_ONLY_METRICS {
493            assert!(
494                stats_section.contains(&format!("`Database::stats().{}`", m.stats_field)),
495                "the stats-only row for `{}` must name its stats field",
496                m.name
497            );
498        }
499    }

501    #[test]
502    fn committed_operator_metrics_doc_is_fresh() {
503        // Anti-drift guard (#2426 AC2/AC5): the committed reference MUST match a
504        // fresh render of the catalog. If a metric is added/renamed or an
505        // annotation changes without regenerating, this fails — mirroring the
506        // #1338 parity-report freshness check, at the lib-test layer so it runs in
507        // both the full gate's core-tests and the lite gate's cqlite-core --lib.
508        let repo_root = std::path::Path::new(env!("CARGO_MANIFEST_DIR"))
509            .parent()
510            .expect("cqlite-core has a repo-root parent");
511        let path = repo_root.join(COMMITTED_DOC_REL);
512        let committed = std::fs::read_to_string(&path).unwrap_or_else(|e| {
513            panic!(
514                "committed operator metrics doc {} is missing ({e}); regenerate with \
515                 `cargo run -p cqlite-core --example gen_operator_metrics_doc`",
516                path.display()
517            )
518        });
519        let fresh = render_markdown().expect("render must succeed");
520        assert_eq!(
521            committed,
522            fresh,
523            "{} is STALE vs the catalog; regenerate with \
524             `cargo run -p cqlite-core --example gen_operator_metrics_doc`",
525            path.display()
526        );
527
528        // The published website page is the same render with front matter.
529        let web_path = repo_root.join(WEBSITE_DOC_REL);
530        let web_committed = std::fs::read_to_string(&web_path).unwrap_or_else(|e| {
531            panic!(
532                "published website metrics page {} is missing ({e}); regenerate with \
533                 `cargo run -p cqlite-core --example gen_operator_metrics_doc`",
534                web_path.display()
535            )
536        });
537        let web_fresh = render_website_markdown().expect("website render must succeed");
538        assert_eq!(
539            web_committed,
540            web_fresh,
541            "{} is STALE vs the catalog; regenerate with \
542             `cargo run -p cqlite-core --example gen_operator_metrics_doc`",
543            web_path.display()
544        );
545    }
