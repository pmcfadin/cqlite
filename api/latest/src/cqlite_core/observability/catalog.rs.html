1//! Metric catalog — the single source of truth for CQLite metric names + units.
2//!
3//! Downstream observability issues (epic #1031) MUST import the constants from
4//! this module instead of hard-coding metric-name strings. Centralising the
5//! names here keeps the emitted telemetry consistent across subsystems and lets
6//! us evolve names in exactly one place.
7//!
8//! # Naming conventions
9//!
10//! Names follow the OpenTelemetry metric semantic conventions:
11//! - dot-separated, lower-snake namespaces, rooted under `cqlite.`
12//! - units use the UCUM annotations OTel recommends (`{row}`, `s`, `By`, `1`,
13//!   `{error}`, …); see [`unit`].
14//! - counters describe a monotonically increasing total; their name reflects the
15//!   thing being counted (rows, bytes, errors).
16//! - histograms describe a distribution (durations, sizes).
17//! - gauges describe a current value (in-flight, cache occupancy).
18//!
19//! # Attribute cardinality
20//!
21//! Every metric is documented with the *bounded* attribute set it may carry.
22//! NEVER attach unbounded values (raw error messages, partition keys, full
23//! queries) as attributes — those explode cardinality and cost. Bounded
24//! attribute keys live in [`attr`].
25
26/// Recommended UCUM units for the catalog metrics.
27///
28/// Use these constants when constructing instruments so the unit string stays
29/// consistent with the metric definitions below.
30pub mod unit {
31    /// Dimensionless count / ratio.
32    pub const DIMENSIONLESS: &str = "1";
33    /// A count of rows (UCUM annotation).
34    pub const ROWS: &str = "{row}";
35    /// A count of partitions.
36    pub const PARTITIONS: &str = "{partition}";
37    /// A count of SSTables.
38    pub const SSTABLES: &str = "{sstable}";
39    /// A count of errors.
40    pub const ERRORS: &str = "{error}";
41    /// A count of tombstones.
42    pub const TOMBSTONES: &str = "{tombstone}";
43    /// Bytes.
44    pub const BYTES: &str = "By";
45    /// Seconds (OTel prefers base-unit seconds for durations).
46    pub const SECONDS: &str = "s";
47    /// A count of OS threads (UCUM annotation).
48    pub const THREADS: &str = "{thread}";
49}

51/// Bounded attribute keys for catalog metrics.
52///
53/// These are the ONLY attribute keys downstream code should attach to the
54/// catalog metrics. Each is documented with its allowed value space so the
55/// cardinality stays bounded.
56pub mod attr {
57    /// Low-cardinality error category. Values come from
58    /// [`crate::observability::ErrorCategory::as_str`] (≈10 distinct values).
59    pub const ERROR_CATEGORY: &str = "cqlite.error.category";
60    /// Subsystem that produced an event, e.g. `"reader"`, `"query"`,
61    /// `"compaction"`. Callers pass a `&'static str`, so the value space is
62    /// bounded by the code itself.
63    pub const SUBSYSTEM: &str = "cqlite.subsystem";
64    /// SSTable on-disk format family, e.g. `"big"` or `"bti"`. Bounded.
65    pub const SSTABLE_FORMAT: &str = "cqlite.sstable.format";
66    /// Compression algorithm, e.g. `"lz4"`, `"snappy"`, `"none"`. Bounded.
67    pub const COMPRESSION: &str = "cqlite.compression";
68    /// Outcome of a lookup/check, exactly `"hit"` or `"miss"`. Bounded to two
69    /// values; used by partition-lookup and bloom-check counters so a single
70    /// metric carries both arms for ratio dashboards.
71    pub const RESULT: &str = "cqlite.result";
72    /// Read-path access route for a partition lookup, e.g. `"index"` (BIG
73    /// Index.db) or `"bti_trie"` (BTI Partitions.db). Bounded by the code.
74    /// Distinct from [`ACCESS_PATH`], which is the query-engine SELECT access
75    /// path (#1035); this is the storage-layer lookup route (#1034).
76    pub const LOOKUP_ROUTE: &str = "cqlite.read.lookup_route";
77    /// Access path a `SELECT` chose for its SSTable-scan step (issue #1035).
78    ///
79    /// Values come from [`crate::query::access_path::AccessPath::label`] — a
80    /// closed set such as `"full_scan"`, `"partition_lookup"`,
81    /// `"multi_partition_lookup"`, `"clustering_slice"`,
82    /// `"fallback_full_scan"`. Bounded by the `AccessPath` enum itself, so it is
83    /// safe as a metric dimension and span attribute. NEVER carries key values.
84    pub const ACCESS_PATH: &str = "cqlite.query.access_path";
85    /// Query plan family chosen by the planner / executor, e.g. `"table_scan"`,
86    /// `"point_lookup"`, `"index_scan"`, `"range_scan"`, `"aggregation"`
87    /// (issue #1035). Bounded by the executor's plan-type taxonomy.
88    pub const PLAN_TYPE: &str = "cqlite.query.plan_type";
89    /// Arrow Flight RPC method name (issue #1041), e.g. `"do_get"`,
90    /// `"get_flight_info"`, `"get_schema"`, `"handshake"`. Bounded by the
91    /// `FlightService` trait's fixed set of methods — never a request payload.
92    pub const RPC_METHOD: &str = "cqlite.rpc.method";
93    /// Arrow Flight RPC outcome (issue #1041), exactly `"ok"` or `"error"`.
94    /// Bounded to two values so a single metric series carries both arms for
95    /// success/error-rate dashboards.
96    pub const RPC_STATUS: &str = "cqlite.rpc.status";
97    /// `do_get` execution phase (issue #2162). Bounded to the closed set
98    /// `"resolve"`, `"merge_setup"`, `"stream"` — a `&'static str` from a fixed
99    /// slot table, never a per-query, per-ticket, key, or query-text value. Used
100    /// as the bounded dimension on [`super::RPC_PHASE_DURATION`] so a stalled
101    /// `do_get` localizes to a phase (time piling up in `merge_setup`) from
102    /// metrics alone.
103    pub const RPC_PHASE: &str = "cqlite.rpc.phase";
104    /// Reason a `SELECT` fell back to a degraded (full-scan) read path
105    /// (issue #2163). Values come from
106    /// [`crate::query::access_path::FallbackReason::label`] — a documented closed
107    /// set (`no_schema`, `partition_key_not_fully_constrained`,
108    /// `partition_key_encoding_failed`, `metadata_scan_path`, `legacy_executor_path`,
109    /// `tombstones_build_no_prune`). Bounded by the enum itself; NEVER carries a
110    /// partition key, predicate value, or query string.
111    pub const FALLBACK_REASON: &str = "cqlite.query.fallback_reason";
112    /// Flight warm-handle refresh outcome (issue #2310). Bounded to the closed set
113    /// `"unchanged"`, `"rebuilt_delta"`, `"fail_closed_retained"` — a `&'static str`
114    /// from a fixed slot table, never a ticket, key, or path value.
115    pub const WARM_REFRESH_OUTCOME: &str = "cqlite.warm.refresh_outcome";
116}

118/// `cqlite.read.rows` — counter `{row}`.
119///
120/// Total rows materialised by the read path. On the Flight k-way merge scan
121/// (issue #2162) the delta is emitted incrementally during a long-running scan,
122/// at a bounded row threshold, so the counter climbs before the scan returns;
123/// the total is unchanged. That merge-scan emission is FORMAT-AGNOSTIC (carries
124/// no attributes): the k-way merge reconciles rows across potentially several
125/// input SSTables — of possibly mixed BIG/BTI format — into one row set before
126/// this counter's grain, so no single format label is honest at the point of
127/// emission without per-input-file tallies threaded through reconciliation (no
128/// consumer needs that split today; a future extension could add it). A direct
129/// single-SSTable read-path caller may still attach [`attr::SSTABLE_FORMAT`]
130/// where the format is known at its own emission site — this metric's attribute
131/// set is therefore [`attr::SSTABLE_FORMAT`] OR no attributes, never a fabricated
132/// format label.
133pub const READ_ROWS: &str = "cqlite.read.rows";
134

159/// `cqlite.sstable.index_parses_total` — counter `1` (issue #2383).
160///
161/// Incremented ONCE each time the full BIG/NB `Index.db` partition index is
162/// parsed end-to-end into partition entries (`parse_all_partition_keys_with_summary`,
163/// the O(entries) `memcmp`/vint loop that dominates opening a reader for a
164/// many-partition SSTable). It is the scale-free, path-independent probe for the
165/// #2383 resolve-phase CPU spin: the field failure re-parsed the SAME 1.58M-entry
166/// Index.db 8× for one logical query (redundant per-generation opens across
167/// per-query snapshot teardown + un-coalesced concurrent splits), so this counter
168/// climbs far past the number of distinct generations. A correct read path parses
169/// each generation's Index.db at most ONCE per query (warm reuse / rebind
170/// thereafter), so `index_parses_total ≈ Σ generations`, not `Σ generations ×
171/// opens`. Emitted from every full-parse site (warm rebuild, aggregate merge,
172/// point-read routing) so no single call path can hide a redundant parse. No
173/// high-cardinality attributes.
174pub const INDEX_PARSES_TOTAL: &str = "cqlite.sstable.index_parses_total";
