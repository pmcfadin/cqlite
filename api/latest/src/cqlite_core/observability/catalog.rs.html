1//! Metric catalog — the single source of truth for CQLite metric names + units.
2//!
3//! Downstream observability issues (epic #1031) MUST import the constants from
4//! this module instead of hard-coding metric-name strings. Centralising the
5//! names here keeps the emitted telemetry consistent across subsystems and lets
6//! us evolve names in exactly one place.
7//!
8//! # Naming conventions
9//!
10//! Names follow the OpenTelemetry metric semantic conventions:
11//! - dot-separated, lower-snake namespaces, rooted under `cqlite.`
12//! - units use the UCUM annotations OTel recommends (`{row}`, `s`, `By`, `1`,
13//!   `{error}`, …); see [`unit`].
14//! - counters describe a monotonically increasing total; their name reflects the
15//!   thing being counted (rows, bytes, errors).
16//! - histograms describe a distribution (durations, sizes).
17//! - gauges describe a current value (in-flight, cache occupancy).
18//!
19//! # Attribute cardinality
20//!
21//! Every metric is documented with the *bounded* attribute set it may carry.
22//! NEVER attach unbounded values (raw error messages, partition keys, full
23//! queries) as attributes — those explode cardinality and cost. Bounded
24//! attribute keys live in [`attr`].
25
26/// Recommended UCUM units for the catalog metrics.
27///
28/// Use these constants when constructing instruments so the unit string stays
29/// consistent with the metric definitions below.
30pub mod unit {
31    /// Dimensionless count / ratio.
32    pub const DIMENSIONLESS: &str = "1";
33    /// A count of rows (UCUM annotation).
34    pub const ROWS: &str = "{row}";
35    /// A count of partitions.
36    pub const PARTITIONS: &str = "{partition}";
37    /// A count of SSTables.
38    pub const SSTABLES: &str = "{sstable}";
39    /// A count of errors.
40    pub const ERRORS: &str = "{error}";
41    /// A count of tombstones.
42    pub const TOMBSTONES: &str = "{tombstone}";
43    /// Bytes.
44    pub const BYTES: &str = "By";
45    /// Seconds (OTel prefers base-unit seconds for durations).
46    pub const SECONDS: &str = "s";
47    /// A count of OS threads (UCUM annotation).
48    pub const THREADS: &str = "{thread}";
49    /// A count of open file descriptors (UCUM annotation, issue #2419).
50    pub const FDS: &str = "{fd}";
51    /// A count of channel entries / queued items (UCUM annotation, issue #2419).
52    pub const ENTRIES: &str = "{entry}";
53}

55/// Bounded attribute keys for catalog metrics.
56///
57/// These are the ONLY attribute keys downstream code should attach to the
58/// catalog metrics. Each is documented with its allowed value space so the
59/// cardinality stays bounded.
60pub mod attr {
61    /// Low-cardinality error category. Values come from
62    /// [`crate::observability::ErrorCategory::as_str`] (≈10 distinct values).
63    pub const ERROR_CATEGORY: &str = "cqlite.error.category";
64    /// Subsystem that produced an event, e.g. `"reader"`, `"query"`,
65    /// `"compaction"`. Callers pass a `&'static str`, so the value space is
66    /// bounded by the code itself.
67    pub const SUBSYSTEM: &str = "cqlite.subsystem";
68    /// SSTable on-disk format family, e.g. `"big"` or `"bti"`. Bounded.
69    pub const SSTABLE_FORMAT: &str = "cqlite.sstable.format";
70    /// Compression algorithm, e.g. `"lz4"`, `"snappy"`, `"none"`. Bounded.
71    pub const COMPRESSION: &str = "cqlite.compression";
72    /// Outcome of a lookup/check, exactly `"hit"` or `"miss"`. Bounded to two
73    /// values; used by partition-lookup and bloom-check counters so a single
74    /// metric carries both arms for ratio dashboards.
75    pub const RESULT: &str = "cqlite.result";
76    /// Read-path access route for a partition lookup, e.g. `"index"` (BIG
77    /// Index.db) or `"bti_trie"` (BTI Partitions.db). Bounded by the code.
78    /// Distinct from [`ACCESS_PATH`], which is the query-engine SELECT access
79    /// path (#1035); this is the storage-layer lookup route (#1034).
80    pub const LOOKUP_ROUTE: &str = "cqlite.read.lookup_route";
81    /// Access path a `SELECT` chose for its SSTable-scan step (issue #1035).
82    ///
83    /// Values come from [`crate::query::access_path::AccessPath::label`] — a
84    /// closed set such as `"full_scan"`, `"partition_lookup"`,
85    /// `"multi_partition_lookup"`, `"clustering_slice"`,
86    /// `"fallback_full_scan"`. Bounded by the `AccessPath` enum itself, so it is
87    /// safe as a metric dimension and span attribute. NEVER carries key values.
88    pub const ACCESS_PATH: &str = "cqlite.query.access_path";
89    /// Query plan family chosen by the planner / executor, e.g. `"table_scan"`,
90    /// `"point_lookup"`, `"index_scan"`, `"range_scan"`, `"aggregation"`
91    /// (issue #1035). Bounded by the executor's plan-type taxonomy.
92    pub const PLAN_TYPE: &str = "cqlite.query.plan_type";
93    /// Arrow Flight RPC method name (issue #1041), e.g. `"do_get"`,
94    /// `"get_flight_info"`, `"get_schema"`, `"handshake"`. Bounded by the
95    /// `FlightService` trait's fixed set of methods — never a request payload.
96    pub const RPC_METHOD: &str = "cqlite.rpc.method";
97    /// Arrow Flight RPC outcome (issue #1041), exactly `"ok"` or `"error"`.
98    /// Bounded to two values so a single metric series carries both arms for
99    /// success/error-rate dashboards.
100    pub const RPC_STATUS: &str = "cqlite.rpc.status";
101    /// `do_get` execution phase (issue #2162; `"admission"` added #2420
102    /// roborev-1700; `"validate"` added #2420 roborev-1702). Bounded to the
103    /// closed set `"validate"`, `"admission"`, `"resolve"`, `"merge_setup"`,
104    /// `"stream"` — a `&'static str` from a fixed slot table, never a
105    /// per-query, per-ticket, key, or query-text value. Used as the bounded
106    /// dimension on [`super::RPC_PHASE_DURATION`] so a stalled `do_get`
107    /// localizes to a phase (time piling up in `merge_setup`, queued behind the
108    /// admission semaphore in `admission`, or stuck parsing/validating a
109    /// malformed ticket in `validate`) from metrics alone.
110    pub const RPC_PHASE: &str = "cqlite.rpc.phase";
111    /// Reason a `SELECT` fell back to a degraded (full-scan) read path
112    /// (issue #2163). Values come from
113    /// [`crate::query::access_path::FallbackReason::label`] — a documented closed
114    /// set (`no_schema`, `partition_key_not_fully_constrained`,
115    /// `partition_key_encoding_failed`, `metadata_scan_path`, `legacy_executor_path`,
116    /// `tombstones_build_no_prune`). Bounded by the enum itself; NEVER carries a
117    /// partition key, predicate value, or query string.
118    pub const FALLBACK_REASON: &str = "cqlite.query.fallback_reason";
119    /// Flight warm-handle refresh outcome (issue #2310). Bounded to the closed set
120    /// `"unchanged"`, `"rebuilt_delta"`, `"fail_closed_retained"` — a `&'static str`
121    /// from a fixed slot table, never a ticket, key, or path value.
122    pub const WARM_REFRESH_OUTCOME: &str = "cqlite.warm.refresh_outcome";
123}

125/// `cqlite.read.rows` — counter `{row}`.
126///
127/// Total rows materialised by the read path. On the Flight k-way merge scan
128/// (issue #2162) the delta is emitted incrementally during a long-running scan,
129/// at a bounded row threshold, so the counter climbs before the scan returns;
130/// the total is unchanged. That merge-scan emission is FORMAT-AGNOSTIC (carries
131/// no attributes): the k-way merge reconciles rows across potentially several
132/// input SSTables — of possibly mixed BIG/BTI format — into one row set before
133/// this counter's grain, so no single format label is honest at the point of
134/// emission without per-input-file tallies threaded through reconciliation (no
135/// consumer needs that split today; a future extension could add it). A direct
136/// single-SSTable read-path caller may still attach [`attr::SSTABLE_FORMAT`]
137/// where the format is known at its own emission site — this metric's attribute
138/// set is therefore [`attr::SSTABLE_FORMAT`] OR no attributes, never a fabricated
139/// format label.
140pub const READ_ROWS: &str = "cqlite.read.rows";

166/// `cqlite.sstable.index_parses_total` — counter `1` (issue #2383).
167///
168/// Incremented ONCE each time the full BIG/NB `Index.db` partition index is
169/// parsed end-to-end into partition entries (`parse_all_partition_keys_with_summary`,
170/// the O(entries) `memcmp`/vint loop that dominates opening a reader for a
171/// many-partition SSTable). It is the scale-free, path-independent probe for the
172/// #2383 resolve-phase CPU spin: the field failure re-parsed the SAME 1.58M-entry
173/// Index.db 8× for one logical query (redundant per-generation opens across
174/// per-query snapshot teardown + un-coalesced concurrent splits), so this counter
175/// climbs far past the number of distinct generations. A correct read path parses
176/// each generation's Index.db at most ONCE per query (warm reuse / rebind
177/// thereafter), so `index_parses_total ≈ Σ generations`, not `Σ generations ×
178/// opens`. Emitted from every full-parse site (warm rebuild, aggregate merge,
179/// point-read routing) so no single call path can hide a redundant parse. No
180/// high-cardinality attributes.
181pub const INDEX_PARSES_TOTAL: &str = "cqlite.sstable.index_parses_total";
182
183/// `cqlite.sstable.index_interval_parses_total` — counter `1` (issue #2412).
184///
185/// Incremented once per **bounded** `Index.db` interval parse performed by the
186/// lazy Summary-guided BIG partition index: a point lookup binary-searches
187/// `Summary.db`, seeks to the covering sample's position, and parses at most one
188/// `min_index_interval` of entries (§B of the #2412 design). This is a DISTINCT
189/// counter from [`INDEX_PARSES_TOTAL`], which continues to count only WHOLE-file
190/// `Index.db` parses (so a lazy-open regression that accidentally full-parses is
191/// still visible there, exactly as the #2367 field rounds check). A cold lazy
192/// open of K generations yields `index_parses_total += 0` and
193/// `index_interval_parses_total += 0`, then `+= 1` per point lookup — the
194/// scale-free work-probe for #2412. No high-cardinality attributes.
195pub const INDEX_INTERVAL_PARSES_TOTAL: &str = "cqlite.sstable.index_interval_parses_total";

560/// `cqlite.proc.threads` — gauge `{thread}` (issue #2419, WS2).
561///
562/// Process-wide OS thread count, sampled from `/proc/self/task` on Linux by the
563/// background saturation sampler (~2s cadence). Aggregates the thread footprint
564/// across ALL concurrent queries (unlike [`MERGE_PRODUCER_THREADS`], which is
565/// per-merge), so N wide `do_get` merges over-subscribing the box is legible on
566/// the server's own metric surface, not only through out-of-band `kubectl top`.
567///
568/// **Healthy vs alarming**: rises with concurrent scans (each opens producer
569/// threads) and settles back toward baseline as they complete; a level that
570/// keeps climbing toward the container thread ceiling is the thread/scheduler
571/// collapse the ramp watches for. **Absence rule**: on a non-`/proc` platform
572/// the reader returns None and this gauge is ABSENT from the exposition (never
573/// `0`). No high-cardinality attributes.
574pub const PROC_THREADS: &str = "cqlite.proc.threads";

576/// `cqlite.proc.fds` — gauge `{fd}` (issue #2419, WS2).
577///
578/// Process-wide open file-descriptor count, sampled from `/proc/self/fd` on
579/// Linux. The read path opens a fresh `File` per SSTable per scan (no reader
580/// pool, by #815 design), so N×M fds accumulate against a container ulimit
581/// (~1024) → `EMFILE`. This gauge makes fd pressure visible before exhaustion.

583/// **Healthy vs alarming**: rises as concurrent scans open SSTables and falls as

1042    #[test]
1043    fn every_instrument_registered_in_otel_is_catalogued() {
1044        // Issue #2426 (roborev MEDIUM, F1): guard the "emitted instrument absent
1045        // from ALL_METRICS" bug class. `otel.rs` is the canonical instrument
1046        // construction + record-routing site (every cross-crate emission — incl.
1047        // cqlite-flight's warm-cache/admission metrics — routes through its
1048        // `add_counter`/`record_histogram`/`record_gauge` dedicated arms). Any
1049        // `catalog::SCREAMING_CONST` referenced there is a metric name bound to a
1050        // real instrument, so it MUST appear in `ALL_METRICS`. This is a
1051        // fully-automatic source-level check (no `observability` feature needed):
1052        // add an instrument in `otel.rs` and forget to catalogue it → this fails.
1053        //
1054        // Automation note (#2426): this scans the core `otel.rs` registration site.
1055        // Because every catalogued instrument that cqlite-flight emits now has a
1056        // dedicated arm here (never the ad-hoc `_ =>` fallback), the check
1057        // transitively covers the flight emission sites too. A future metric emitted
1058        // ONLY via the ad-hoc fallback (no dedicated arm, no catalog entry) would not
1059        // be caught here — that path is reserved for genuinely non-catalog names.
1060        let otel_src = include_str!("otel.rs");
1061        let catalogued: std::collections::HashSet<&str> = ALL_METRICS.iter().copied().collect();
1062
1063        // Collect the const IDENTIFIERS present in the ALL_METRICS array so we can
1064        // map an `otel.rs` `catalog::IDENT` reference to a catalogued name. The
1065        // constants are `pub const IDENT: &str = "cqlite. …";`, so build the map
1066        // from this source file.
1067        let this_src = include_str!("catalog.rs");
1068        let mut ident_to_value = std::collections::HashMap::new();
1069        for line in this_src.lines() {
1070            let line = line.trim_start();
1071            if let Some(rest) = line.strip_prefix("pub const ") {
1072                if let Some((ident, tail)) = rest.split_once(':') {
1073                    if let Some(start) = tail.find('"') {
1074                        let after = &tail[start + 1..];
1075                        if let Some(end) = after.find('"') {
1076                            ident_to_value.insert(ident.trim(), &after[..end]);
1077                        }
1078                    }
1079                }
1080            }
1081        }
1082
1083        // Extract every `catalog::SCREAMING_CONST` reference in otel.rs. `unit`/
1084        // `attr` submodule refs (`catalog::unit::…`, `catalog::attr::…`) start with
1085        // a lowercase char after `catalog::`, so they are excluded by construction.
1086        let mut missing = Vec::new();
1087        for (i, _) in otel_src.match_indices("catalog::") {
1088            let rest = &otel_src[i + "catalog::".len()..];
1089            let ident: String = rest
1090                .chars()
1091                .take_while(|c| c.is_ascii_uppercase() || c.is_ascii_digit() || *c == '_')
1092                .collect();
1093            // Skip lowercase submodule paths (unit/attr) — `ident` is empty then.
1094            if ident.is_empty() {
1095                continue;
1096            }
1097            let value = ident_to_value.get(ident.as_str()).copied().unwrap_or_else(|| {
1098                panic!("otel.rs references catalog::{ident}, which is not a metric-name constant in catalog.rs")
1099            });
1100            if !catalogued.contains(value) {
1101                missing.push(format!("catalog::{ident} (\"{value}\")"));
1102            }
1103        }
1104        assert!(
1105            missing.is_empty(),
1106            "otel.rs registers instruments for metrics ABSENT from ALL_METRICS \
1107             (add them to catalog::ALL_METRICS): {missing:?}"
1108        );
1109    }
