1//! Operator annotation data table for [`super::operator_docs`] (issue #2426).
2//!
3//! Split out of `operator_docs.rs` to keep each file under the campsite source
4//! target (#1116); this is pure data. The annotations are keyed by the
5//! `catalog::*` name constants so they stay wired to the real catalog — the
6//! fail-closed completeness/uniqueness checks live in `operator_docs.rs`.
7
8use super::catalog::{self, attr};
9use super::operator_docs::{MetricDoc, MetricKind};
10
11// Read-PHASE timings + reader/WAL resource gauges (issue #1707): a SECOND
12// annotation table in its own file, because this one was already at the campsite
13// source target (#1116) and the table is append-only data.
14#[path = "operator_docs_annotations_read_phase.rs"]
15mod read_phase;
16
17/// Every operator annotation table, in declaration order. The renderer walks the
18/// UNION (and sorts by name), so a family lives in whichever file it was declared
19/// in — `operator_docs::operator_metric_docs` never reads one table directly.
20pub(super) const ANNOTATION_TABLES: &[&[MetricDoc]] = &[ANNOTATIONS, read_phase::ANNOTATIONS];
21
22/// Every operator annotation across [`ANNOTATION_TABLES`].
23pub(super) fn all_annotations() -> impl Iterator<Item = &'static MetricDoc> {
24    ANNOTATION_TABLES.iter().copied().flatten()
25}
