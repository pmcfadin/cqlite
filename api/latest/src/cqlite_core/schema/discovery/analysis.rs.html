21    pub(super) async fn extract_header_metadata(
22        &self,
23        context: &mut DiscoveryContext,
24    ) -> Result<()> {
25        // Parse headers from each source file
26        for file_path in &context.source_files.clone() {
27            match self.parse_sstable_header(file_path).await {
28                Ok(header) => {
29                    context.headers.push(header.clone());
30
31                    // Update Cassandra version if not set
32                    if context.cassandra_version.is_none() {
33                        context.cassandra_version = Some(header.cassandra_version);
34                    }
35                }
36                Err(e) => {
37                    // Log error but continue with other files
38                    tracing::warn!("Failed to parse header from {:?}: {}", file_path, e);
39                }
40            }
41        }
42
43        Ok(())
44    }

46    /// Parse SSTable header from file
47    async fn parse_sstable_header(&self, file_path: &std::path::Path) -> Result<SSTableHeader> {
48        use crate::storage::sstable::reader::SSTableReader;
49
50        let reader =
51            SSTableReader::open(file_path, &self.core_config, self.platform.clone()).await?;
52        Ok(reader.header().clone())

55    pub(super) async fn sample_data_for_inference(
56        &self,
57        context: &mut DiscoveryContext,
58    ) -> Result<()> {
59        let mut total_sampled = 0;
60
61        for file_path in &context.source_files.clone() {
62            if total_sampled >= self.config.max_sample_rows {
63                break;
64            }
65
66            match self
67                .sample_sstable_data(file_path, self.config.max_sample_rows - total_sampled)
68                .await
69            {
70                Ok(samples) => {
71                    total_sampled += samples.len();
72
73                    // Organize samples by column
74                    for row in samples {
75                        for (column_name, value) in row {
76                            context
77                                .column_samples
78                                .entry(column_name)
79                                .or_default()
80                                .push(value);
81                        }
82                    }
83                }
84                Err(e) => {
85                    tracing::warn!("Failed to sample data from {:?}: {}", file_path, e);
86                }
87            }
88        }
89
90        context.total_rows_sampled = total_sampled;
91        Ok(())
92    }

94    /// Sample data from an SSTable file
95    async fn sample_sstable_data(
96        &self,
97        file_path: &std::path::Path,
98        max_rows: usize,
99    ) -> Result<Vec<HashMap<String, Value>>> {
100        use crate::storage::sstable::reader::SSTableReader;
101
102        let reader =
103            SSTableReader::open(file_path, &self.core_config, self.platform.clone()).await?;
104
105        // Header column names for the raw-fallback case (issue #1334): a
106        // `ScanRow::RawRow` is undecoded whole-row bytes, so — matching the
107        // pre-#1334 sampler — map it onto the first header column rather than a
108        // synthetic `"data"` blob that would infer a bogus column.
109        let header_first_column: Option<String> =
110            reader.header().columns.first().map(|c| c.name.clone());
111
112        // Get all entries and sample up to max_rows
113        let all_entries = reader.get_all_entries().await?;
114
115        // Issue #1334: each entry carries a `ScanRow` row. Disassemble a live row's
116        // interned cells into a name→value map for type inference; map a raw
117        // fallback row onto the header column name; suppress markers (row
118        // tombstone / null row) which carry no columns.
119        let samples: Vec<HashMap<String, Value>> = all_entries
120            .into_iter()
121            .take(max_rows)
122            .filter_map(|(_table_id, _row_key, row)| {
123                let cells = row.into_sample_cells(header_first_column.as_deref())?;
124                if cells.is_empty() {
125                    return None;
126                }
127                let row_data: HashMap<String, Value> = cells
128                    .into_iter()
129                    .map(|(name, v)| (name.to_string(), v))
130                    .collect();
131                Some(row_data)
132            })
133            .collect();
134
135        Ok(samples)
136    }

138    pub(super) async fn discover_udts(&self, context: &mut DiscoveryContext) -> Result<()> {
139        // UDT discovery from headers and data samples
140        for header in &context.headers {
141            for column_def in &header.columns {
142                // Check if column type indicates a UDT
143                if self.is_udt_type(&column_def.column_type) {
144                    let udt_name = self.extract_udt_name(&column_def.column_type);
145
146                    if !context.discovered_udts.contains_key(&udt_name) {
147                        // Create basic UDT definition from type info
148                        let udt_def = UDTDefinition {
149                            name: udt_name.clone(),
150                            keyspace: context.keyspace.clone(),
151                            fields: self.parse_udt_fields(&column_def.column_type),
152                            version: Some(1),
153                        };
154
155                        context.discovered_udts.insert(udt_name, udt_def);
156                    }
157                }
158            }
159        }
160
161        // Analyze UDT usage in sample data
162        for (column_name, samples) in &context.column_samples {
163            for sample in samples {
164                if let Value::Udt(udt_map) = sample {
165                    // Infer UDT structure from data
166                    let udt_name = format!("{}_udt", column_name); // Generate name if not available
167
168                    if !context.discovered_udts.contains_key(&udt_name) {
169                        let mut fields = Vec::new();
170
171                        for (position, field) in udt_map.fields.iter().enumerate() {
172                            let field_def = UdtFieldDefinition {
173                                name: field.name.clone(),
174                                field_type: "text".to_string(), // Would need type inference here
175                                position,
176                                nullable: true,
177                            };
178                            fields.push(field_def);
179                        }
180
181                        let udt_def = UDTDefinition {
182                            name: udt_name.clone(),
183                            keyspace: context.keyspace.clone(),
184                            fields,
185                            version: Some(1),
186                        };
187
188                        context.discovered_udts.insert(udt_name, udt_def);
189                    }
190                }
191            }
192        }
193
194        Ok(())
195    }

197    /// Check if a type string indicates a UDT
198    fn is_udt_type(&self, type_str: &str) -> bool {
199        // UDTs are typically not standard CQL types
200        !matches!(
201            type_str.to_lowercase().as_str(),
202            "text"
203                | "varchar"
204                | "ascii"
205                | "int"
206                | "bigint"
207                | "smallint"
208                | "tinyint"
209                | "float"
210                | "double"
211                | "boolean"
212                | "timestamp"
213                | "date"
214                | "time"
215                | "uuid"
216                | "timeuuid"
217                | "blob"
218                | "varint"
219                | "decimal"
220                | "duration"
221                | "inet"
222                | "counter"
223        ) && !type_str.starts_with("list<")
224            && !type_str.starts_with("set<")
225            && !type_str.starts_with("map<")
226            && !type_str.starts_with("tuple<")
227            && !type_str.starts_with("frozen<")
228    }

248    pub(super) async fn analyze_collection_types(
249        &self,
250        context: &mut DiscoveryContext,
251    ) -> Result<()> {
252        // Analyze collection types from headers
253        for header in &context.headers {
254            for column_def in &header.columns {
255                if let Some(collection_type) = self.parse_collection_type(&column_def.column_type) {
256                    context
257                        .collection_types
258                        .insert(column_def.name.clone(), collection_type);
259                }
260            }
261        }
262
263        // Analyze collection types from sample data
264        for (column_name, samples) in &context.column_samples {
265            let mut detected_collections = Vec::new();
266
267            for sample in samples {
268                match sample {
269                    Value::List(elements) => {
270                        let element_type = if !elements.is_empty() {
271                            self.infer_element_type(elements)
272                        } else {
273                            "text".to_string()
274                        };
275
276                        detected_collections.push(CollectionType {
277                            kind: CollectionKind::List,
278                            element_type: Some(element_type),
279                            key_type: None,
280                            value_type: None,
281                            is_frozen: false,
282                        });
283                    }
284                    Value::Set(elements) => {
285                        let element_type = if !elements.is_empty() {
286                            self.infer_element_type(elements)
287                        } else {
288                            "text".to_string()
289                        };
290
291                        detected_collections.push(CollectionType {
292                            kind: CollectionKind::Set,
293                            element_type: Some(element_type),
294                            key_type: None,
295                            value_type: None,
296                            is_frozen: false,
297                        });
298                    }
299                    Value::Map(map) => {
300                        let (key_type, value_type) = if !map.is_empty() {
301                            let keys: Vec<Value> = map.iter().map(|(k, _)| k.clone()).collect();
302                            let values: Vec<Value> = map.iter().map(|(_, v)| v.clone()).collect();
303                            (
304                                self.infer_element_type(&keys),
305                                self.infer_element_type(&values),
306                            )
307                        } else {
308                            ("text".to_string(), "text".to_string())
309                        };
310
311                        detected_collections.push(CollectionType {
312                            kind: CollectionKind::Map,
313                            element_type: None,
314                            key_type: Some(key_type),
315                            value_type: Some(value_type),
316                            is_frozen: false,
317                        });
318                    }
319                    Value::Tuple(_) => {
320                        detected_collections.push(CollectionType {
321                            kind: CollectionKind::Tuple,
322                            element_type: None,
323                            key_type: None,
324                            value_type: None,
325                            is_frozen: false,
326                        });
327                    }
328                    _ => {}
329                }
330            }
331
332            // Use the most common collection type for this column
333            if let Some(collection_type) =
334                self.select_most_common_collection_type(detected_collections)
335            {
336                context
337                    .collection_types
338                    .insert(column_name.clone(), collection_type);
339            }
340        }
341
342        Ok(())
343    }

345    /// Parse collection type from type string
346    fn parse_collection_type(&self, type_str: &str) -> Option<CollectionType> {
347        let lower_type = type_str.to_lowercase();
348
349        if lower_type.starts_with("list<") {
350            let element_type = self.extract_inner_type(type_str, "list<");
351            Some(CollectionType {
352                kind: CollectionKind::List,
353                element_type: Some(element_type),
354                key_type: None,
355                value_type: None,
356                is_frozen: false,
357            })
358        } else if lower_type.starts_with("set<") {
359            let element_type = self.extract_inner_type(type_str, "set<");
360            Some(CollectionType {
361                kind: CollectionKind::Set,
362                element_type: Some(element_type),
363                key_type: None,
364                value_type: None,
365                is_frozen: false,
366            })
367        } else if lower_type.starts_with("map<") {
368            let (key_type, value_type) = self.extract_map_types(type_str);
369            Some(CollectionType {
370                kind: CollectionKind::Map,
371                element_type: None,
372                key_type: Some(key_type),
373                value_type: Some(value_type),
374                is_frozen: false,
375            })
376        } else if lower_type.starts_with("tuple<") {
377            Some(CollectionType {
378                kind: CollectionKind::Tuple,
379                element_type: None,
380                key_type: None,
381                value_type: None,
382                is_frozen: false,
383            })
384        } else if lower_type.starts_with("frozen<") {
385            // Parse the inner type and mark as frozen
386            if let Some(mut inner_collection) =
387                self.parse_collection_type(&self.extract_inner_type(type_str, "frozen<"))
388            {
389                inner_collection.is_frozen = true;
390                Some(inner_collection)
391            } else {
392                None
393            }
394        } else {
395            None
396        }
397    }

399    /// Extract inner type from generic type string
400    fn extract_inner_type(&self, type_str: &str, _prefix: &str) -> String {
401        if let Some(start) = type_str.find('<') {
402            if let Some(end) = type_str.rfind('>') {
403                return type_str[start + 1..end].to_string();
404            }
405        }
406        "text".to_string()
407    }

409    /// Extract key and value types from map type string
410    fn extract_map_types(&self, type_str: &str) -> (String, String) {
411        if let Some(start) = type_str.find('<') {
412            if let Some(end) = type_str.rfind('>') {
413                let inner = &type_str[start + 1..end];
414                if let Some(comma_pos) = inner.find(',') {
415                    let key_type = inner[..comma_pos].trim().to_string();
416                    let value_type = inner[comma_pos + 1..].trim().to_string();
417                    return (key_type, value_type);
418                }
419            }
420        }
421        ("text".to_string(), "text".to_string())
422    }

424    /// Infer element type from a collection of values
425    fn infer_element_type(&self, elements: &[Value]) -> String {
426        if elements.is_empty() {
427            return "text".to_string();
428        }
429
430        // Count type occurrences
431        let mut type_counts = HashMap::new();
432        for element in elements {
433            let type_name = match element {
434                Value::Text(_) => "text",
435                Value::Integer(_) => "int",
436                Value::BigInt(_) => "bigint",
437                Value::Float(_) => "double",
438                Value::Boolean(_) => "boolean",
439                Value::Uuid(_) => "uuid",
440                Value::Timestamp(_) => "timestamp",
441                Value::Blob(_) => "blob",
442                _ => "text",
443            };
444            *type_counts.entry(type_name).or_insert(0) += 1;
445        }
446
447        // Return most common type
448        type_counts
449            .into_iter()
450            .max_by_key(|(_, count)| *count)
451            .map(|(type_name, _)| type_name.to_string())
452            .unwrap_or_else(|| "text".to_string())
453    }

455    /// Select the most common collection type from detected types
456    fn select_most_common_collection_type(
457        &self,
458        mut types: Vec<CollectionType>,
459    ) -> Option<CollectionType> {
460        if types.is_empty() {
461            return None;
462        }
463
464        // For simplicity, just return the first type
465        // In a real implementation, you'd want to find the most common type
466        types.sort_by(|a, b| format!("{:?}", a.kind).cmp(&format!("{:?}", b.kind)));
467        types.into_iter().next()
468    }

470    pub(super) async fn discover_indexes(&self, context: &mut DiscoveryContext) -> Result<()> {
471        // Index discovery from SSTable metadata and headers
472        for header in &context.headers {
473            // Check for index-related metadata in the header
474            if let Some(index_info) = self.extract_index_info_from_header(header) {
475                context.indexes.extend(index_info);
476            }
477        }
478
479        // Infer potential indexes from column usage patterns
480        for (column_name, samples) in &context.column_samples {
481            // Check if column has characteristics that suggest indexing
482            if self.should_suggest_index(column_name, samples) {
483                let index_def = IndexDefinition {
484                    name: format!("{}_idx", column_name),
485                    target_column: column_name.clone(),
486                    index_type: IndexType::Secondary,
487                    options: HashMap::new(),
488                };
489
490                // Only add if not already present
491                if !context
492                    .indexes
493                    .iter()
494                    .any(|idx| idx.target_column == *column_name)
495                {
496                    context.indexes.push(index_def);
497                }
498            }
499        }
500
501        Ok(())
502    }

515    /// Determine if a column should have an index based on usage patterns
516    fn should_suggest_index(&self, column_name: &str, samples: &[Value]) -> bool {
517        // Simple heuristics for index suggestion
518
519        // 1. Columns with "id" in the name (common lookup pattern)
520        if column_name.to_lowercase().contains("id") {
521            return true;
522        }
523
524        // 2. Columns with high cardinality (many unique values)
525        let unique_values: std::collections::HashSet<_> = samples.iter().collect();
526        let cardinality_ratio = unique_values.len() as f64 / samples.len() as f64;
527        if cardinality_ratio > 0.7 && samples.len() > 10 {
528            return true;
529        }
530
531        // 3. Columns that look like foreign keys or references
532        if column_name.to_lowercase().ends_with("_id")
533            || column_name.to_lowercase().ends_with("_ref")
534            || column_name.to_lowercase().contains("email")
535            || column_name.to_lowercase().contains("username")
536        {
537            return true;
538        }
539
540        // 4. UUID columns (often used for lookups)
541        if samples.iter().any(|v| matches!(v, Value::Uuid(_))) {
542            return true;
543        }
544
545        false
546    }

548    pub(super) async fn build_schema_info(
549        &self,
550        context: &mut DiscoveryContext,
551    ) -> Result<SchemaInfo> {
552        let mut partition_key = Vec::new();
553        let mut clustering_keys = Vec::new();
554        let mut regular_columns = Vec::new();
555        let mut static_columns = Vec::new();
556
557        // Extract column information from headers
558        let mut position = 0;
559        for header in &context.headers {
560            for column_def in &header.columns {
561                // Infer type from header and samples
562                let samples = context
563                    .column_samples
564                    .get(&column_def.name)
565                    .cloned()
566                    .unwrap_or_default();
567                let type_info = self
568                    .type_inference
569                    .infer_column_type(&samples)
570                    .await
571                    .unwrap_or_else(|_| TypeInfo {
572                        type_id: column_def.column_type.clone(),
573                        type_params: vec![],
574                        is_frozen: false,
575                        element_type: None,
576                        key_type: None,
577                        value_type: None,
578                        udt_fields: None,
579                        tuple_elements: None,
580                    });
581
582                let confidence = self.calculate_type_confidence(&samples, &type_info);
583
584                let column = ColumnDefinition {
585                    name: column_def.name.clone(),
586                    data_type: column_def.column_type.clone(),
587                    type_info,
588                    nullable: true,   // Assume nullable unless proven otherwise
589                    is_static: false, // Would need additional logic to detect static columns
590                    default_value: None,
591                    position,
592                    confidence,
593                };
594
595                // Classify column based on position and naming patterns
596                if self.is_partition_key_column(&column_def.name, position) {
597                    partition_key.push(column);
598                } else if self.is_clustering_column(&column_def.name, position) {
599                    clustering_keys.push(ClusteringColumn {
600                        name: column_def.name.clone(),
601                        data_type: column_def.column_type.clone(),
602                        position: clustering_keys.len(), // Use current size as position
603                        order: crate::schema::ClusteringOrder::Asc, // Default to ascending
604                    });
605                } else if column.is_static {
606                    static_columns.push(column);
607                } else {
608                    regular_columns.push(column);
609                }
610
611                position += 1;
612            }
613        }
614
615        // If no partition key was found, assume first column is partition key
616        if partition_key.is_empty() && !regular_columns.is_empty() {
617            let first_column = regular_columns.remove(0);
618            partition_key.push(first_column);
619        }
620
621        // Build validation results
622        let validation_results = ValidationResults {
623            status: self.determine_validation_status(&partition_key, &regular_columns),
624            errors: Vec::new(), // Would be populated by actual validation
625            warnings: self.generate_validation_warnings(&partition_key, &regular_columns),
626            consistency_results: ConsistencyResults {
627                files_analyzed: context.source_files.len(),
628                schema_mismatches: 0,
629                type_inconsistencies: Vec::new(),
630                udt_conflicts: Vec::new(),
631            },
632        };
633
634        Ok(SchemaInfo {
635            keyspace: context.keyspace.clone(),
636            table: context.table.clone(),
637            partition_key,
638            clustering_keys,
639            regular_columns,
640            static_columns,
641            collection_types: context.collection_types.clone(),
642            user_defined_types: context.discovered_udts.values().cloned().collect(),
643            indexes: context.indexes.clone(),
644            table_options: context.table_options.clone(),
645            metadata: SchemaMetadata {
646                discovered_at: std::time::SystemTime::now(),
647                source_files: context.source_files.clone(),
648                total_rows_sampled: context.total_rows_sampled,
649                cassandra_version: context.cassandra_version,
650                discovery_method: DiscoveryMethod::Hybrid,
651                version: 1,
652                validation_results,
653                performance_metrics: DiscoveryMetrics {
654                    total_time_ms: 0, // Will be filled in later
655                    header_parsing_time_ms: 0,
656                    data_sampling_time_ms: 0,
657                    type_inference_time_ms: 0,
658                    validation_time_ms: 0,
659                    peak_memory_usage_bytes: 0,
660                },
661            },
662        })
663    }

665    /// Heuristic to determine if a column is a partition key
666    fn is_partition_key_column(&self, column_name: &str, position: usize) -> bool {
667        // Simple heuristics - in practice this would be more sophisticated
668        position == 0
669            || column_name.to_lowercase().contains("key")
670            || column_name.to_lowercase() == "id"
671            || column_name.to_lowercase().ends_with("_id")
672    }
673
674    /// Heuristic to determine if a column is a clustering column
675    fn is_clustering_column(&self, column_name: &str, position: usize) -> bool {
676        // Simple heuristics
677        (position == 1 && !self.is_partition_key_column(column_name, position))
678            || column_name.to_lowercase().contains("time")
679            || column_name.to_lowercase().contains("date")
680            || column_name.to_lowercase().contains("order")
681    }
