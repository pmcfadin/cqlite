1//! The ONE parser for a `vector` type's two declared parameters — element type
2//! and dimension (issue #4114).
3//!
4//! # Why this is shared rather than repeated
5//!
6//! Four independent type-string parsers plus the STATS clustering-layout resolver
7//! all have to learn `vector`, and every one of them needs the SAME two answers:
8//! which element type, and what dimension. Repeating the dimension rules (reject
9//! `0`, reject non-numeric, reject an overflowing product) five times is five
10//! chances to disagree, and the disagreement would be silent — the defect #4114
11//! exists to remove.
12//!
13//! # Authority (pinned `cassandra-5.0.8`, never CQLite's own code — #3041)
14//!
15//! * The marshal spelling is written by `TypeParser.stringifyVectorParameters`
16//!   (`TypeParser.java:239-242`):
17//!   `"(" + type.toString(ignoreFreezing) + " , " + dimension + ")"`, reached from
18//!   `VectorType.toString` (`VectorType.java:339-342`) as
19//!   `getClass().getName() + stringifyVectorParameters(…)`. So Cassandra WRITES
20//!   `org.apache.cassandra.db.marshal.VectorType(org.apache.cassandra.db.marshal.FloatType , 3)`
21//!   — note the spaces around the comma.
22//! * Cassandra's own READER tolerates whitespace variation:
23//!   `TypeParser.getVectorParameters` (`TypeParser.java:244-263`) calls
24//!   `skipBlankAndComma()` between the two parameters. So this parser TRIMS rather
25//!   than matching Cassandra's exact output spacing — matching the literal `" , "`
26//!   would refuse a legal spelling another writer may emit.
27//! * Parameter ORDER is (element type, dimension) — same function.
28//! * `dimension <= 0` is rejected by Cassandra at construction
29//!   (`VectorType.java:89-90`, `InvalidRequestException`: "vectors may only have
30//!   positive dimensions"), so a `0` dimension is a MALFORMED TYPE here, never an
31//!   empty vector. There is NO upper bound in the type itself
32//!   (`Guardrails.vectorDimensions` is a soft, disabled-by-default cluster
33//!   setting — `Config.java:924-925`), so none is imposed here either; the only
34//!   rejected large value is one whose byte width overflows `usize`.
35//! * The CQL3 surface spelling is `vector<float, 3>` (`CQL3Type.java:589,:938`).
36//!
37//! # No-heuristics (#28)
38//!
39//! The dimension is read from the DECLARED type and from nowhere else. Nothing in
40//! this module looks at a value byte, and no width is ever inferred from how many
41//! bytes happen to be available.
42

54///
55/// `FloatType.valueLengthIfFixed()` returns 4 (`FloatType.java:148-152`), and

60/// The two declared parameters of a `vector` type, as borrowed substrings of the
61/// type string they came from.
62#[derive(Debug, Clone, Copy, PartialEq, Eq)]
63pub(crate) struct VectorTypeArgs<'a> {
64    /// The element type, verbatim and trimmed (a CQL short form such as `float`
65    /// for the CQL spelling, or a marshal class name such as
66    /// `org.apache.cassandra.db.marshal.FloatType` for the marshal spelling).
67    pub element: &'a str,
68    /// The dimension `n`, guaranteed `>= 1`.
69    pub dimension: usize,
70}

72/// The three answers a vector-type probe can give (issue #4114, roborev job 109).
73///
74/// Two-valued `Option` was a DEFECT, not a simplification: `None` meant both "this
75/// is not a vector" (so the caller's other type arms apply) and "this IS a vector
76/// but its parameter list is unparseable" (so the caller must FAIL CLOSED). Every
77/// caller took the first reading, so a malformed `VectorType(` reached a generic
78/// fallback — in `enhanced_statistics_parser::marshal_type` the
79/// `other => other.to_lowercase()` one, which restored exactly the blob/phantom-vint
80/// framing #4114 exists to remove.
81#[derive(Debug, Clone, Copy, PartialEq, Eq)]
82pub(crate) enum VectorInner<'a> {
83    /// `type_str` does not claim to be a vector type.
84    NotAVector,
85    /// `type_str` is a vector type; the payload is its raw parameter list, still to
86    /// be split by [`split_vector_args`].
87    Args(&'a str),
88    /// `type_str` CLAIMS to be a vector type but its parameter list could not be
89    /// extracted. The payload says why, for the refusal message.
90    Malformed(&'static str),
91}
92
93impl<'a> VectorInner<'a> {
94    /// Collapse the probe into the answer a caller actually needs:
95    /// `Ok(None)` = not a vector, `Ok(Some(args))` = a well-formed vector,
96    /// `Err` = a vector that must fail closed, with `type_str` named.
97    ///
98    /// Both malformed states — an unextractable parameter list AND a parameter list
99    /// that does not split into (element, dimension) — come out as `Err`, so no
100    /// caller can accidentally handle only one of them.
101    pub(crate) fn into_args(self, type_str: &'a str) -> Result<Option<VectorTypeArgs<'a>>> {
102        match self {
103            VectorInner::NotAVector => Ok(None),
104            VectorInner::Args(inner) => split_vector_args(inner, type_str).map(Some),
105            VectorInner::Malformed(why) => Err(malformed(type_str, why)),
106        }
107    }
108}

110/// Probe `type_str` for the CQL spelling `vector<element, n>`.
111///
112/// The keyword is matched case-insensitively because CQL type keywords are
113/// (`SET<TEXT>` == `set<text>`), which the sibling parsers in
114/// [`super::cql_type_parser`] already rely on.
115///
116/// A bare `vector` with NO parameter list is [`VectorInner::NotAVector`], NOT
117/// malformed: `vector` is not a reserved word in CQL (`CQL3Type.java:589,:938`
118/// recognises it only WITH parameters), so an unparameterised `vector` can be a
119/// legitimate UDT name and must stay available to the caller's UDT arm. Once a `<`
120/// follows, no other type can be meant, so an unterminated parameter list is
121/// [`VectorInner::Malformed`].
122pub(crate) fn cql_vector_kind(type_str: &str) -> VectorInner<'_> {
123    let trimmed = type_str.trim();
124    let (keyword, rest) = split_leading_identifier(trimmed);
125    if !keyword.eq_ignore_ascii_case("vector") {
126        return VectorInner::NotAVector;
127    }
128    let rest = rest.trim_start();
129    let Some(after_open) = rest.strip_prefix('<') else {
130        // No parameter list: a UDT may legitimately be named `vector`.
131        return VectorInner::NotAVector;
132    };
133    match matched_bracket_body(after_open, '<', '>') {
134        Some((inner, tail)) if tail.trim().is_empty() => VectorInner::Args(inner),
135        Some(_) => {
136            VectorInner::Malformed("trailing text after the closing '>' of the type parameters")
137        }
138        None => VectorInner::Malformed("the type parameters are not terminated by a matching '>'"),
139    }
140}

142/// Probe `type_str` for the marshal spelling `VectorType(element , n)`.
143///
144/// The constructor name is matched EXACTLY on its package-stripped simple name
145/// (`VectorType`), the same way the sibling marshal resolvers in
146/// `parser::repair_clustering` and `parser::enhanced_statistics_parser` identify a
147/// constructor: Java class names are case-sensitive, so a lowercased spelling is
148/// not a class reference and must not be accepted here.
149///
150/// Unlike the CQL probe, a `VectorType` with no usable parameter list is
151/// [`VectorInner::Malformed`] rather than `NotAVector`: `VectorType` is a Java class
152/// name, nothing else can be spelled that way, and `VectorType.getInstance` has no
153/// parameterless form (`TypeParser.getVectorParameters`, `TypeParser.java:244-263`,
154/// requires both parameters). So there is no honest reading of it other than "a
155/// broken vector type".
156pub(crate) fn marshal_vector_kind(type_str: &str) -> VectorInner<'_> {
157    let trimmed = type_str.trim();
158    let (ctor, rest) = split_leading_identifier(trimmed);
159    let simple = ctor.rsplit('.').next().unwrap_or(ctor);
160    if simple != "VectorType" {
161        return VectorInner::NotAVector;
162    }
163    let rest = rest.trim_start();
164    let Some(after_open) = rest.strip_prefix('(') else {
165        return VectorInner::Malformed(
166            "a VectorType constructor with no '(' parameter list (Cassandra has no \
167             parameterless VectorType)",
168        );
169    };
170    match matched_bracket_body(after_open, '(', ')') {
171        Some((inner, tail)) if tail.trim().is_empty() => VectorInner::Args(inner),
172        Some(_) => {
173            VectorInner::Malformed("trailing text after the closing ')' of the parameter list")
174        }
175        None => VectorInner::Malformed("the parameter list is not terminated by a matching ')'"),
176    }
177}

179/// Split off the leading Java/CQL type identifier — the run of characters legal in
180/// a (possibly package-qualified) type name — and return `(identifier, rest)`.
181///
182/// `.` is part of the identifier so a package-qualified class name comes back
183/// whole; the caller package-strips. `_` and `$` are legal Java identifier
184/// characters. Nothing else is consumed, so the very next character decides
185/// whether a parameter list follows.
186fn split_leading_identifier(value: &str) -> (&str, &str) {
187    let end = value
188        .find(|c: char| !(c.is_ascii_alphanumeric() || c == '.' || c == '_' || c == '$'))
189        .unwrap_or(value.len());
190    (&value[..end], &value[end..])
191}
192
193/// The body between an ALREADY-CONSUMED `open` bracket and its MATCHING `close`,
194/// plus whatever follows that `close`. `None` on unbalanced brackets.
195fn matched_bracket_body(after_open: &str, open: char, close: char) -> Option<(&str, &str)> {
196    let mut depth = 1usize;
197    for (idx, ch) in after_open.char_indices() {
198        if ch == open {
199            depth += 1;
200        } else if ch == close {
201            depth -= 1;
202            if depth == 0 {
203                return Some((&after_open[..idx], &after_open[idx + close.len_utf8()..]));
204            }
205        }
206    }
207    None
208}

210/// Split a vector type's argument list into its (element type, dimension) pair.
211///
212/// `inner` is the already-extracted argument list — the [`VectorInner::Args`]
213/// payload from [`cql_vector_kind`] or [`marshal_vector_kind`], normally reached
214/// via [`VectorInner::into_args`] rather than called directly. `type_str` is the
215/// full type string, used only so a refusal NAMES the type that was refused.
216///
217/// EXACTLY ONE top-level comma is required, and that is not a restatement of
218/// "split on the last one" — the two differ, measurably. This function used to take
219/// the LAST top-level comma and fold everything before it into the element, so
220/// `vector<int, text, 3>` parsed as element `int, text` (which then degraded to a
221/// `CqlType::Custom` carrier) with dimension `3`, ADMITTING a spelling Cassandra
222/// refuses. Caught by the frozen-scalar gate's own accept/refuse pins (#4104): it
223/// was refused before `CqlType::Vector` existed and silently admitted after.
224///
225/// Cassandra permits no third parameter, in either spelling:
226/// * CQL — `Parser.g:1916-1919`,
227///   `vector_type : K_VECTOR '<' t1=comparatorType ',' d=INTEGER '>'`;
228/// * marshal — `TypeParser.getVectorParameters` (`TypeParser.java:244-263`) parses
229///   ONE type, then `skipBlankAndComma()`, then ONE identifier, and then
230///   `if (str.charAt(idx) != ')') throw new IllegalStateException()`.
231///
232/// Commas INSIDE the element's own brackets are at depth > 0 and are not counted,
233/// so `vector<map<int, text>, 3>` and `VectorType(MapType(Int32Type,UTF8Type) , 3)`
234/// are unaffected. Order is fixed by the same functions: element first, dimension
235/// last.
236pub(crate) fn split_vector_args<'a>(inner: &'a str, type_str: &str) -> Result<VectorTypeArgs<'a>> {
237    let mut depth = 0usize;
238    let mut split_at: Option<usize> = None;
239    let mut top_level_commas = 0usize;
240    for (idx, ch) in inner.char_indices() {
241        match ch {
242            '(' | '<' => depth += 1,
243            ')' | '>' => {
244                depth = depth.saturating_sub(1);
245            }
246            ',' if depth == 0 => {
247                top_level_commas += 1;
248                split_at = Some(idx);
249            }
250            _ => {}
251        }
252    }
253    if top_level_commas > 1 {
254        return Err(malformed(
255            type_str,
256            &format!(
257                "expected exactly two parameters (element type, dimension) but found \
258                 {} top-level comma(s): Cassandra's grammar admits no third parameter \
259                 (Parser.g:1916-1919; TypeParser.getVectorParameters requires ')' after \
260                 the dimension, TypeParser.java:244-263)",
261                top_level_commas
262            ),
263        ));
264    }
265    let Some(comma) = split_at else {
266        return Err(malformed(
267            type_str,
268            "expected two parameters (element type, dimension) separated by a comma",
269        ));
270    };
271    let element = inner[..comma].trim();
272    let dimension_raw = inner[comma + 1..].trim();
273    if element.is_empty() {
274        return Err(malformed(type_str, "the element type parameter is empty"));
275    }
276    let dimension = parse_vector_dimension(dimension_raw, type_str)?;
277    Ok(VectorTypeArgs { element, dimension })
278}

280/// Parse a vector's dimension parameter.
281///
282/// Rejects — each with a named error, never a fallback value:
283/// * an empty parameter, or one that is not PLAIN DECIMAL DIGITS — including the
284///   signed spellings Cassandra's own reader would tolerate but its writer can
285///   never emit (see the body for the pinned citation and why the divergence is
286///   one-directional);
287/// * `0` and any negative spelling — `VectorType.java:89-90` rejects
288///   `dimension <= 0`, so a zero-dimension vector does not exist and a
289///   zero-length value is an error rather than an empty vector;
290/// * a dimension whose byte width would overflow `usize` (see
291///   [`vector_byte_width`]).
292pub(crate) fn parse_vector_dimension(raw: &str, type_str: &str) -> Result<usize> {
293    if raw.is_empty() {
294        return Err(malformed(type_str, "the dimension parameter is empty"));
295    }
296    // DIGITS ONLY, checked explicitly rather than left to `str::parse`, which
297    // ACCEPTS a leading `+` (`"+3".parse::<usize>() == Ok(3)`) — measured, and it
298    // is what the sibling test caught.
299    //
300    // This is deliberately STRICTER THAN CASSANDRA'S OWN READER, which would
301    // accept `+3`: `TypeParser.readNextIdentifier` treats `+`, `-`, `.`, `_` and
302    // `&` as identifier characters (`TypeParser.java:578-583`) and
303    // `getVectorParameters` hands the result to `Integer.parseInt`
304    // (`:255-258`), which tolerates a leading sign. The divergence is safe and
305    // one-directional: Cassandra's WRITER concatenates the `int` dimension
306    // directly (`stringifyVectorParameters`, `TypeParser.java:239-242`), so no
307    // Cassandra-written type string can carry a sign, a `.` or a `_` in this
308    // parameter. Refusing them therefore rejects nothing Cassandra wrote, while
309    // accepting them would mean two spellings of one dimension reaching the
310    // decode path — and a dimension IS the width, so an unnormalised spelling is
311    // a width nobody declared (#28).
312    if !raw.bytes().all(|b| b.is_ascii_digit()) {
313        return Err(malformed(
314            type_str,
315            &format!(
316                "dimension '{raw}' is not a plain decimal integer (digits only: a \
317                 sign, a decimal point, whitespace or a non-ASCII digit is not a \
318                 dimension Cassandra's writer can emit)"
319            ),
320        ));
321    }
322    let dimension: usize = raw.parse().map_err(|_| {
323        malformed(
324            type_str,
325            &format!("dimension '{raw}' is not a non-negative decimal integer"),
326        )
327    })?;
328    if dimension == 0 {
329        return Err(malformed(
330            type_str,
331            "dimension 0 is not a legal vector dimension (Cassandra rejects \
332             dimension <= 0: VectorType.java:89-90)",
333        ));
334    }
335    // The doc comment above promises this function rejects a dimension whose BYTE
336    // WIDTH would overflow, and before this it only rejected zero (roborev job 113).
337    // On a 64-bit target any `dimension > usize::MAX / 4` was accepted into
338    // `CqlType::Vector` and failed much later, at decode — a promise the code did not
339    // keep, and a late failure where an early one was available. Validated through the
340    // SAME helper the decoders use, so the parser and the decode path cannot disagree
341    // about what is representable.
342    // Cassandra's dimension is a Java `int`: `getVectorParameters` hands the token to
343    // `Integer.parseInt` (`TypeParser.java:255-258`), so `Integer.MAX_VALUE` is the
344    // hard ceiling and NO Cassandra-written type string can exceed it (roborev job
345    // 114). This is the same one-directional argument the digits-only check above
346    // rests on — Cassandra's WRITER concatenates the `int` directly
347    // (`stringifyVectorParameters`, `:239-242`) — so refusing a larger value rejects
348    // nothing Cassandra wrote, while accepting one would admit an IMPOSSIBLE schema
349    // and hand a multi-gigabyte declared width to the decode path.
350    if dimension > i32::MAX as usize {
351        return Err(malformed(
352            type_str,
353            "the dimension exceeds Cassandra's Integer.MAX_VALUE ceiling",
354        ));
355    }
356    // NOTE the ORDER: the Integer.MAX_VALUE ceiling above runs FIRST, and on a 64-bit
357    // target it SUBSUMES this width guard (usize::MAX/4 is 4.6e18, far above
358    // i32::MAX = 2.1e9, so nothing can reach here overflowing). This guard is NOT dead
359    // code though: on a 32-bit target usize::MAX/4 is 1_073_741_823, BELOW i32::MAX, so
360    // a dimension in (1.07e9, 2.1e9] is legal per Cassandra and still overflows `4 * n`
361    // — and there this is the only thing standing between that and a wrapped byte
362    // count. Both guards are load-bearing; which one reports is target-dependent, so
363    // no test may pin the message.
364    //
365    // Checked at the FLOAT element width, which is element-SPECIFIC and deliberately
366    // so: this parser sees only the dimension token, never the element, so there is no
367    // generally-correct width to check against — a check at width 1 would be
368    // vacuously true. `float` is the only element #4114 decodes, and every other
369    // element is refused by name at `require_float_element` before any width is
370    // computed, so no unchecked path exists today. A future element type wider than 4
371    // bytes must add its own check; it cannot inherit this one.
372    if vector_byte_width(FLOAT_ELEMENT_WIDTH, dimension).is_none() {
373        return Err(malformed(
374            type_str,
375            "the dimension's byte width overflows the addressable range",
376        ));
377    }
378    Ok(dimension)
379}

381/// The total on-disk byte width of a FIXED-width vector: `element_width *
382/// dimension`, or `None` when that product overflows `usize`.
383///
384/// `VectorType.java:94-96`:
385/// `valueLengthIfFixed = elementType.valueLengthIfFixed() * dimension`.
386pub(crate) fn vector_byte_width(element_width: usize, dimension: usize) -> Option<usize> {
387    // `dimension == 0` is REFUSED, not multiplied (issue #4114, roborev job 111).
388    // Cassandra cannot even CONSTRUCT a zero-dimension vector — `VectorType.java:89-90`
389    // throws `InvalidRequestException("vectors may only have positive dimensions")` —
390    // and an empty vector VALUE is `MarshalException("Invalid empty vector value")`
391    // (`:365-368` via `:515-517`). Without this guard `0` multiplied cleanly to a width
392    // of 0 and an EMPTY buffer then decoded as a successful `Value::List([])`: a value
393    // Cassandra says cannot exist, produced without error. `CqlType::Vector` is
394    // publicly constructible, so the string parser's own `n > 0` check is NOT
395    // sufficient — a programmatically built `Vector(Float, 0)` bypasses it entirely,
396    // which is why the guard belongs HERE, at the shared width helper every decode
397    // entry point funnels through, rather than at one caller.
398    if dimension == 0 {
399        return None;
400    }
401    element_width.checked_mul(dimension)
402}

404/// The width Cassandra's `valueLengthIfFixed()` reports for a vector ELEMENT, or
405/// `None` when that element is variable-length in Cassandra's FRAMING.
406///
407/// Deliberately NOT `CqlType::fixed_size()` (issue #4114, roborev job 111).
408/// `fixed_size()` answers "how many bytes does one logical value occupy", which is a
409/// DIFFERENT question from "does Cassandra frame this without a length prefix", and
410/// it disagrees with the framing authority on five types: it reports `TinyInt(1)`,
411/// `SmallInt(2)`, `Date(4)`, `Time(8)` and `Inet(16)` as fixed, while
412/// `parser/repair_clustering.rs` — whose classification is derived from
413/// `valueLengthIfFixed()` and pinned by tests — treats `InetAddressType`, `TimeType`
414/// and `SimpleDateType` as VARIABLE, and its own tests assert tinyint and date
415/// clusterings are vint-prefixed. (`fixed_size()`'s `Inet` arm even concedes the point
416/// in a comment: IPv4 is 4 bytes, IPv6 is 16.)
417///
418/// That pre-existing disagreement is NOT corrected here — `fixed_size()` has no
419/// callers other than its own recursion, so re-plumbing it is a separate change with
420/// its own blast radius. What this function fixes is the vector arm's DEPENDENCE on
421/// it: a vector must not inherit a framing claim from a method that does not answer
422/// the framing question. Only element types whose Cassandra framing is KNOWN fixed
423/// are listed, so an unlisted element yields `None` (variable) rather than a guess
424/// (#28).
425pub(crate) fn cassandra_fixed_element_width(element: &crate::schema::CqlType) -> Option<usize> {
426    use crate::schema::CqlType as T;
427    match element {
428        T::Boolean => Some(1),
429        T::Int => Some(4),
430        T::Float => Some(4),
431        T::BigInt | T::Counter | T::Double | T::Timestamp => Some(8),
432        T::Uuid | T::TimeUuid => Some(16),
433        // Everything else — including TinyInt, SmallInt, Date, Time and Inet — is
434        // either variable in Cassandra's framing or not a scalar this function
435        // speaks for. `None` means "not known fixed", never "zero width".
436        _ => None,
437    }
438}

440fn malformed(type_str: &str, why: &str) -> Error {
441    Error::schema(format!("malformed vector type '{type_str}': {why}"))
442}

484    #[test]

490            format!("{PKG}VectorType"),

654    // ── roborev job 111 ────────────────────────────────────────────────────────
655
656    /// A zero dimension is REFUSED at the shared width helper, so no decode entry
657    /// point can be reached with it.
658    ///
659    /// `CqlType::Vector` is publicly constructible, so the string parser's `n > 0`
660    /// check does not cover a programmatically built `Vector(Float, 0)`. Before this
661    /// guard, `0` multiplied cleanly to width 0 and an EMPTY buffer decoded as a
662    /// successful `Value::List([])` — a value Cassandra says cannot exist
663    /// (`VectorType.java:89-90` refuses n <= 0 at construction; `:365-368` throws
664    /// `MarshalException("Invalid empty vector value")`).
665    /// roborev job 113: an overflowing dimension is refused AT PARSE TIME, not only
666    /// later at width computation.
667    ///
668    /// TWO guards now stand between a raw token and `CqlType::Vector`, and WHICH ONE
669    /// fires is TARGET-DEPENDENT — which is why this asserts REFUSAL rather than a
670    /// message. Measured:
671    ///
672    /// | target | `usize::MAX / 4`      | `i32::MAX`    | width guard reachable? |
673    /// |--------|-----------------------|---------------|------------------------|
674    /// | 64-bit | 4_611_686_018_427_387_903 | 2_147_483_647 | NO — the ceiling always fires first |
675    /// | 32-bit | 1_073_741_823         | 2_147_483_647 | YES — (1.07e9, 2.1e9] passes the ceiling and overflows `4 * n` |
676    ///
677    /// So on 64-bit the Integer.MAX_VALUE ceiling subsumes the width guard, while on
678    /// 32-bit the width guard is the only thing standing between a legal-per-Cassandra
679    /// dimension and a wrapped byte count. BOTH are load-bearing; neither is dead.
680    ///
681    /// An earlier version of this test asserted the message contained "overflow" and
682    /// broke the moment the ceiling landed (it fired first and said
683    /// "Integer.MAX_VALUE" instead) — one unit test failing FOUR gate components,
684    /// since core-tests / write-tests / legacy-heuristics / feature-iso-delta-scan all
685    /// run this lib. Pinning a message that guard ORDERING can change was the mistake.
686    #[test]
687    fn an_overflowing_dimension_is_refused_at_parse_time_not_only_at_width() {
688        let ty = "org.apache.cassandra.db.marshal.VectorType(org.apache.cassandra.db.marshal.FloatType , n)";
689
690        // A dimension whose 4*n width cannot be represented must be REFUSED at parse
691        // time by SOME guard. Which one is target-dependent, so do not pin the text.
692        let width_overflow = (usize::MAX / FLOAT_ELEMENT_WIDTH) + 1;
693        assert!(
694            parse_vector_dimension(&width_overflow.to_string(), ty).is_err(),
695            "a dimension whose byte width cannot be represented must be refused at \
696             PARSE time, by whichever guard applies on this target"
697        );
698        assert!(
699            parse_vector_dimension(&usize::MAX.to_string(), ty).is_err(),
700            "usize::MAX must be refused"
701        );
702
703        // The representable ones still parse.
704        for ok in ["1", "3", "384", "4096"] {
705            assert!(
706                parse_vector_dimension(ok, ty).is_ok(),
707                "{ok} is representable and must parse"
708            );
709        }
710    }

732    /// The element width used for a vector's framing is Cassandra's
733    /// `valueLengthIfFixed()`, NOT `CqlType::fixed_size()`.
734    ///
735    /// `fixed_size()` reports TinyInt/SmallInt/Date/Time/Inet as fixed, while the
736    /// framing authority (`parser/repair_clustering.rs`, derived from
737    /// `valueLengthIfFixed()` and pinned by its own tests) treats
738    /// `InetAddressType`/`TimeType`/`SimpleDateType` as VARIABLE and asserts tinyint
739    /// and date clusterings are vint-prefixed. A vector must not inherit a framing
740    /// claim from a method that answers a different question.
741    #[test]
742    fn framing_width_is_not_fixed_size() {
743        use crate::schema::CqlType as T;
744
745        // KNOWN-FIXED elements, with Cassandra's widths.
746        for (ty, want) in [
747            (T::Boolean, 1usize),
748            (T::Int, 4),
749            (T::Float, 4),
750            (T::BigInt, 8),
751            (T::Counter, 8),
752            (T::Double, 8),
753            (T::Timestamp, 8),
754            (T::Uuid, 16),
755            (T::TimeUuid, 16),
756        ] {
757            assert_eq!(
758                cassandra_fixed_element_width(&ty),
759                Some(want),
760                "{ty:?} must be fixed at {want} bytes"
761            );
762        }
763
764        // THE REGRESSION: the five types `fixed_size()` gets wrong for framing.
765        // Each of these returns Some(..) from fixed_size(), so a vector arm built on
766        // that method claimed a fixed width Cassandra does not use.
767        for ty in [T::TinyInt, T::SmallInt, T::Date, T::Time, T::Inet] {
768            assert_eq!(
769                cassandra_fixed_element_width(&ty),
770                None,
771                "{ty:?} is VARIABLE in Cassandra framing and must NOT report a fixed width"
772            );
773            assert!(
774                ty.fixed_size().is_some(),
775                "{ty:?} is expected to still report a logical fixed_size() — this test \
776                 exists precisely because the two answers differ"
777            );
778        }
779
780        // Genuinely variable elements stay None.
781        for ty in [T::Text, T::Blob, T::Decimal, T::Duration, T::Varint] {
782            assert_eq!(cassandra_fixed_element_width(&ty), None, "{ty:?}");
783        }
