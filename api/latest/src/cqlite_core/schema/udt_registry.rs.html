1//! UDT (User Defined Type) schema registry.
2//!
3//! Houses [`UdtRegistry`], which stores and resolves User Defined Type
4//! definitions per keyspace, including dependency validation and CREATE TYPE
5//! export. Extracted from `schema/mod.rs` (issue #1134, source-split doctrine)
6//! with no behavior change.

13/// UDT Schema Registry for managing User Defined Type definitions
14#[derive(Debug, Clone, Default, Serialize, Deserialize)]
15pub struct UdtRegistry {
16    /// Registered UDT type definitions by keyspace and type name
17    udts: HashMap<String, HashMap<String, UdtTypeDef>>,
18}
19
20impl UdtRegistry {
21    /// Create a new UDT registry
22    pub fn new() -> Self {
23        Self {
24            udts: HashMap::new(),
25        }
26    }
27
28    /// Create a new UDT registry with enhanced Cassandra 5.0 defaults
29    pub fn with_cassandra5_defaults() -> Self {
30        let mut registry = Self::new();
31        registry.load_cassandra5_system_udts();
32        registry
33    }

35    /// Register a UDT type definition
36    pub fn register_udt(&mut self, udt_def: UdtTypeDef) {
37        let keyspace_udts = self.udts.entry(udt_def.keyspace.clone()).or_default();
38        keyspace_udts.insert(udt_def.name.clone(), udt_def);
39    }
40
41    /// Get a UDT definition by keyspace and name
42    pub fn get_udt(&self, keyspace: &str, name: &str) -> Option<&UdtTypeDef> {
43        self.udts.get(keyspace)?.get(name)
44    }
45
46    /// Get all UDTs in a keyspace
47    pub fn get_keyspace_udts(&self, keyspace: &str) -> Option<&HashMap<String, UdtTypeDef>> {
48        self.udts.get(keyspace)
49    }

51    /// List all registered UDT names in a keyspace
52    pub fn list_udt_names(&self, keyspace: &str) -> Vec<&str> {
53        self.udts
54            .get(keyspace)
55            .map(|udts| udts.keys().map(|s| s.as_str()).collect())
56            .unwrap_or_default()

59    /// Check if a UDT is registered
60    pub fn contains_udt(&self, keyspace: &str, name: &str) -> bool {
61        self.udts
62            .get(keyspace)
63            .map(|udts| udts.contains_key(name))
64            .unwrap_or(false)

148    /// Resolve UDT with full dependency chain
149    pub fn resolve_udt_with_dependencies(
150        &self,
151        keyspace: &str,
152        name: &str,
153    ) -> crate::Result<&UdtTypeDef> {
154        let udt = self.get_udt(keyspace, name).ok_or_else(|| {
155            crate::Error::schema(format!(
156                "UDT '{}' not found in keyspace '{}'",
157                name, keyspace
158            ))
159        })?;
160
161        // Validate all field dependencies exist
162        for field in &udt.fields {
163            self.validate_field_type_dependencies(&field.field_type, keyspace)?;
164        }
165
166        Ok(udt)
167    }
168
169    /// Validate that all UDT field type dependencies exist in the registry
170    fn validate_field_type_dependencies(
171        &self,
172        field_type: &CqlType,
173        keyspace: &str,
174    ) -> crate::Result<()> {
175        match field_type {
176            CqlType::Udt(udt_name, _) => {
177                if !self.contains_udt(keyspace, udt_name) {
178                    return Err(crate::Error::schema(format!(
179                        "UDT dependency '{}' not found in keyspace '{}'",
180                        udt_name, keyspace
181                    )));
182                }
183            }
184            CqlType::List(inner) | CqlType::Set(inner) | CqlType::Frozen(inner) => {
185                self.validate_field_type_dependencies(inner, keyspace)?;
186            }
187            CqlType::Map(key_type, value_type) => {
188                self.validate_field_type_dependencies(key_type, keyspace)?;
189                self.validate_field_type_dependencies(value_type, keyspace)?;
190            }
191            CqlType::Tuple(field_types) => {
192                for tuple_field_type in field_types {
193                    self.validate_field_type_dependencies(tuple_field_type, keyspace)?;
194                }
195            }
196            _ => {} // Primitive types don't need validation
197        }
198        Ok(())
199    }

201    /// Get all UDTs that depend on a given UDT (for cascade operations)
202    pub fn get_dependent_udts(&self, keyspace: &str, udt_name: &str) -> Vec<&UdtTypeDef> {
203        let mut dependents = Vec::new();
204
205        if let Some(keyspace_udts) = self.udts.get(keyspace) {
206            for udt in keyspace_udts.values() {
207                if udt.name == udt_name {
208                    continue; // Skip self
209                }
210
211                // Check if this UDT depends on the target UDT
212                if self.udt_depends_on(udt, udt_name) {
213                    dependents.push(udt);
214                }
215            }
216        }
217
218        dependents
219    }

221    /// Check if a UDT depends on another UDT (recursively)
222    fn udt_depends_on(&self, udt: &UdtTypeDef, target_udt: &str) -> bool {
223        for field in &udt.fields {
224            if self.field_type_depends_on(&field.field_type, target_udt) {
225                return true;
226            }
227        }
228        false
229    }
230
231    /// Check if a field type depends on a UDT
232    #[allow(clippy::only_used_in_recursion)]
233    fn field_type_depends_on(&self, field_type: &CqlType, target_udt: &str) -> bool {
234        match field_type {
235            CqlType::Udt(udt_name, _) => udt_name == target_udt,
236            CqlType::List(inner) | CqlType::Set(inner) | CqlType::Frozen(inner) => {
237                self.field_type_depends_on(inner, target_udt)
238            }
239            CqlType::Map(key_type, value_type) => {
240                self.field_type_depends_on(key_type, target_udt)
241                    || self.field_type_depends_on(value_type, target_udt)
242            }
243            CqlType::Tuple(field_types) => field_types
244                .iter()
245                .any(|ft| self.field_type_depends_on(ft, target_udt)),
246            _ => false,
247        }
248    }

250    /// Register UDT with dependency validation
251    pub fn register_udt_with_validation(&mut self, udt_def: UdtTypeDef) -> crate::Result<()> {
252        // Validate dependencies exist
253        for field in &udt_def.fields {
254            self.validate_field_type_dependencies(&field.field_type, &udt_def.keyspace)?;
255        }
256
257        // Check for circular dependencies
258        if self.would_create_circular_dependency(&udt_def) {
259            return Err(crate::Error::schema(format!(
260                "Registering UDT '{}' would create circular dependency",
261                udt_def.name
262            )));
263        }
264
265        self.register_udt(udt_def);
266        Ok(())
267    }

269    /// Check if registering a UDT would create circular dependencies
270    fn would_create_circular_dependency(&self, udt_def: &UdtTypeDef) -> bool {
271        // This is complex - for now, just check direct self-reference
272        for field in &udt_def.fields {
273            if self.field_type_depends_on(&field.field_type, &udt_def.name) {
274                return true;
275            }
276        }
277        false
278    }

280    /// Export UDT definitions for debugging
281    pub fn export_definitions(&self, keyspace: &str) -> Vec<String> {
282        let mut definitions = Vec::new();
283
284        if let Some(keyspace_udts) = self.udts.get(keyspace) {
285            for udt in keyspace_udts.values() {
286                let mut def = format!("CREATE TYPE {}.{} (\n", keyspace, udt.name);
287
288                for (i, field) in udt.fields.iter().enumerate() {
289                    if i > 0 {
290                        def.push_str(",\n");
291                    }
292                    def.push_str(&format!(
293                        "  {} {}",
294                        field.name,
295                        self.format_cql_type(&field.field_type)
296                    ));
297                }
298
299                def.push_str("\n);");
300                definitions.push(def);
301            }
302        }
303
304        definitions
305    }

307    /// Format CQL type for CREATE TYPE statements
308    #[allow(clippy::only_used_in_recursion)]
309    fn format_cql_type(&self, cql_type: &CqlType) -> String {
310        match cql_type {
311            CqlType::Boolean => "boolean".to_string(),
312            CqlType::TinyInt => "tinyint".to_string(),
313            CqlType::SmallInt => "smallint".to_string(),
314            CqlType::Int => "int".to_string(),
315            CqlType::BigInt => "bigint".to_string(),
316            CqlType::Counter => "counter".to_string(),
317            CqlType::Float => "float".to_string(),
318            CqlType::Double => "double".to_string(),
319            CqlType::Text | CqlType::Varchar => "text".to_string(),
320            CqlType::Ascii => "ascii".to_string(),
321            CqlType::Blob => "blob".to_string(),
322            CqlType::Timestamp => "timestamp".to_string(),
323            CqlType::Date => "date".to_string(),
324            CqlType::Time => "time".to_string(),
325            CqlType::Uuid => "uuid".to_string(),
326            CqlType::TimeUuid => "timeuuid".to_string(),
327            CqlType::Inet => "inet".to_string(),
328            CqlType::Duration => "duration".to_string(),
329            CqlType::Varint => "varint".to_string(),
330            CqlType::Decimal => "decimal".to_string(),
331            CqlType::List(inner) => format!("list<{}>", self.format_cql_type(inner)),
332            CqlType::Set(inner) => format!("set<{}>", self.format_cql_type(inner)),
333            CqlType::Map(key, value) => format!(
334                "map<{}, {}>",
335                self.format_cql_type(key),
336                self.format_cql_type(value)
337            ),
338            CqlType::Udt(name, _) => name.clone(),
339            CqlType::Tuple(types) => {
340                let type_strs: Vec<String> =
341                    types.iter().map(|t| self.format_cql_type(t)).collect();
342                format!("tuple<{}>", type_strs.join(", "))
343            }
344            CqlType::Frozen(inner) => format!("frozen<{}>", self.format_cql_type(inner)),
345            CqlType::Custom(name) => name.clone(),
346        }
347    }
