1//! Ingestion Module for M2-CLI One-Shot Execution
2//!
3//! This module orchestrates schema loading and SSTable discovery to build
4//! a fully-configured Database instance for one-shot query execution.
5
6use std::path::PathBuf;
7use std::sync::Arc;

13    aggregator::{AggregatorConfig, LoadResult, SchemaAggregator},

15};
16use crate::Config;
17use crate::Database;

20/// Configuration for one-shot ingestion
21#[derive(Debug, Clone)]
22pub struct IngestionConfig {
23    /// Schema file paths (.cql or .json) to load
24    pub schema_paths: Vec<PathBuf>,
25
26    /// Root data directory containing SSTables (e.g., /var/lib/cassandra/data)
27    pub data_dir: PathBuf,
28
29    /// Optional Cassandra version hint (e.g., "5.0")
30    pub version_hint: Option<String>,
31
32    /// Core database configuration
33    pub core_config: Config,
34
35    /// Optional filter for table directories (e.g., contains "/test_basic/")
36    /// Only table directories matching this pattern will be loaded
37    pub table_directory_filter: Option<String>,
38}
39
40/// Result of ingestion operation
41#[derive(Debug)]
42pub struct IngestionResult {
43    /// The initialized Database instance
44    pub database: Database,
45
46    /// Schema loading summary
47    pub schema_load_result: LoadResult,
48
49    /// SSTable discovery summary
50    pub discovery_summary: DiscoverySummary,
51
52    /// Schema registry for coverage reporting
53    pub schema_registry: Arc<RwLock<SchemaRegistry>>,
54}
55
56/// SSTable discovery summary
57#[derive(Debug, Clone)]
58pub struct DiscoverySummary {
59    /// Total number of SSTables discovered
60    pub sstables_found: usize,
61    /// Keyspaces discovered
62    pub keyspaces: Vec<String>,
63    /// Tables discovered per keyspace
64    pub tables: Vec<String>,
65    /// Full directory paths for each discovered table
66    pub table_directories: Vec<PathBuf>,
67    /// Resolved Cassandra version (from precedence)
68    pub resolved_version: Option<String>,
69}
70
71/// Main ingestion function for one-shot execution
72///
73/// This function orchestrates:
74/// 1. Schema loading from provided paths (CQL/JSON files)
75/// 2. SSTable discovery from data directory
76/// 3. Database construction with QueryEngine
77///
78/// # Errors
79///
80/// Returns Error::Schema for schema loading failures (exit code 3)
81/// Returns Error::Io for discovery/data-dir failures (exit code 4)
82/// Returns Error::QueryExecution for query engine setup failures (exit code 5)
83pub async fn ingest(config: IngestionConfig) -> Result<IngestionResult> {
84    // Step 1: Validate data directory exists
85    if !config.data_dir.exists() {
86        return Err(Error::Io(std::io::Error::new(
87            std::io::ErrorKind::NotFound,
88            format!(
89                "Data directory does not exist: {}",
90                config.data_dir.display()
91            ),
92        )));
93    }
94
95    if !config.data_dir.is_dir() {
96        return Err(Error::Io(std::io::Error::new(
97            std::io::ErrorKind::InvalidInput,
98            format!(
99                "Data directory path is not a directory: {}",
100                config.data_dir.display()
101            ),
102        )));
103    }
104
105    // Step 2: Initialize Platform
106    let platform = Arc::new(Platform::new(&config.core_config).await?);
107
108    // Step 3: Load schemas using SchemaAggregator
109    let registry_config = SchemaRegistryConfig::default();
110    let schema_registry = Arc::new(RwLock::new(
111        SchemaRegistry::new(
112            registry_config,
113            platform.clone(),
114            config.core_config.clone(),
115        )
116        .await
117        .map_err(|e| Error::Schema(format!("Failed to create schema registry: {}", e)))?,
118    ));
119
120    // Use the SchemaRegistry's internal UDT registry so UDTs are available for query execution
121    // This ensures UDTs loaded from CQL files are available when parsing SSTable data (Issue #238)
122    let udt_registry = schema_registry.read().await.get_udt_registry();
123
124    let aggregator_config = AggregatorConfig {
125        graceful_degradation: false, // Fail fast for one-shot execution
126        validate_udt_dependencies: true,
127    };
128
129    let mut aggregator = SchemaAggregator::new(
130        schema_registry.clone(),
131        udt_registry.clone(),
132        aggregator_config,
133    );
134
135    let schema_load_result = if !config.schema_paths.is_empty() {
136        aggregator
137            .load_from_paths(&config.schema_paths)
138            .await
139            .map_err(|e| Error::Schema(format!("Schema loading failed: {}", e)))?
140    } else {
141        // No schema paths provided - return empty result
142        LoadResult {
143            schemas_loaded: 0,
144            udts_loaded: 0,
145            errors: Vec::new(),
146            warnings: Vec::new(),
147        }
148    };
149
150    // Check for schema loading errors (fail fast)
151    if !schema_load_result.errors.is_empty() {
152        let error_messages: Vec<String> = schema_load_result
153            .errors
154            .iter()
155            .map(|e| format!("{:?}: {}", e.error_type, e.message))
156            .collect();
157        return Err(Error::Schema(format!(
158            "Schema loading failed with {} error(s): {}",
159            schema_load_result.errors.len(),
160            error_messages.join("; ")
161        )));
162    }
163
164    // Step 4: Discover SSTables using DiscoveryService
165    let discovery_service = DiscoveryService::with_schema_registry(
166        config.data_dir.clone(),
167        config.version_hint.clone(),
168        schema_registry.clone(),
169    );
170
171    let service_summary = discovery_service.scan().await.map_err(|e| {
172        // Map discovery errors to appropriate error types
173        match e {
174            Error::Io(_) => e,
175            _ => Error::Io(std::io::Error::other(format!(
176                "SSTable discovery failed: {}",
177                e
178            ))),
179        }
180    })?;
181
182    // Step 5: Filter table directories if a filter is specified
183    let filtered_table_dirs = if let Some(ref filter_pattern) = config.table_directory_filter {
184        service_summary
185            .table_directories
186            .iter()
187            .filter(|path| path.to_string_lossy().contains(filter_pattern))
188            .cloned()
189            .collect()
190    } else {
191        service_summary.table_directories.clone()
192    };
193
194    // Step 6: Build Database with discovered (and optionally filtered) SSTables
195    // Pass the loaded schema_registry to the Database so schemas are available to the query engine
196    // Storage path is data_dir (for runtime storage), discovered directories from DiscoveryService
197    let database = Database::open_with_discovered_sstables_and_registry(
198        &config.data_dir,
199        filtered_table_dirs.clone(),
200        config.core_config.clone(),
201        Some(schema_registry.clone()),
202    )
203    .await
204    .map_err(|e| {
205        // Map database creation errors appropriately
206        match e {
207            Error::Schema(_) => e,
208            Error::Io(_) => e,
209            #[cfg(feature = "state_machine")]
210            Error::QueryExecution(_) => e,
211            _ => Error::QueryExecution(format!("Database initialization failed: {}", e)),
212        }
213    })?;
214
215    // Convert from discovery module's DiscoverySummary to ingestion's DiscoverySummary
216    // Use the filtered table directories in the summary
217    let discovery_summary = DiscoverySummary {
218        sstables_found: service_summary.sstables_found,
219        keyspaces: service_summary.keyspaces,
220        tables: service_summary.tables,
221        table_directories: filtered_table_dirs,
222        resolved_version: service_summary.resolved_version,
223    };
224
225    Ok(IngestionResult {
226        database,
227        schema_load_result,
228        discovery_summary,
229        schema_registry: schema_registry.clone(),
230    })
231}

233#[cfg(test)]
234mod tests {
235    use super::*;
236    use tempfile::TempDir;
237
238    // Note: Tests for discover_sstables() and resolve_cassandra_version()
239    // have been removed as these functions are now in the discovery module
240    // and are tested there.
241
242    #[tokio::test]
243    async fn test_ingest_invalid_data_dir() {
244        let config = IngestionConfig {
245            schema_paths: vec![],
246            data_dir: PathBuf::from("/nonexistent/path"),
247            version_hint: None,
248            core_config: Config::default(),
249            table_directory_filter: None,
250        };
251
252        let result = ingest(config).await;
253        assert!(result.is_err());
254
255        if let Err(Error::Io(io_err)) = result {
256            assert_eq!(io_err.kind(), std::io::ErrorKind::NotFound);
257        } else {
258            panic!("Expected Io error for nonexistent directory");
259        }
260    }
261
262    #[tokio::test]
263    async fn test_ingest_with_empty_schema_paths() {
264        let temp_dir = TempDir::new().unwrap();
265
266        let config = IngestionConfig {
267            schema_paths: vec![],
268            data_dir: temp_dir.path().to_path_buf(),
269            version_hint: Some("5.0".to_string()),
270            core_config: Config::default(),
271            table_directory_filter: None,
272        };
273
274        let result = ingest(config).await;
275        assert!(result.is_ok());
276
277        let ingestion_result = result.unwrap();
278        assert_eq!(ingestion_result.schema_load_result.schemas_loaded, 0);
279        assert_eq!(ingestion_result.schema_load_result.udts_loaded, 0);
280    }
