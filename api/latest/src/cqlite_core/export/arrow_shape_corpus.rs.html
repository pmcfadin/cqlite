1//! The SHARED Arrow row-shape corpus (issues #2825 / #2821 / #2932).
2//!
3//! One corpus, two consumers, so neither can drift from the other:
4//!
5//! * `export::arrow_size_tests::estimate_is_conservative_across_shape_corpus`
6//!   asserts the estimator contract
7//!   `Sum estimate_arrow_row_bytes >= arrow_payload_bytes(batch)`.
8//! * `cqlite_flight::batch_bytes_tests::the_capacity_bound_holds_over_the_shared_shape_corpus`
9//!   asserts the PUBLISHED capacity conversion
10//!   `get_array_memory_size() <= worst_case_batch_capacity_bytes(Sum estimate, nodes, 0)`.
11//!
12//! The second consumer is why this module exists at all rather than staying a
13//! private test helper. Issue #2821 turned that capacity bound from a loose doc
14//! claim into a **fail-closed runtime check**: a shape whose fixed per-array-node
15//! cost exceeds `BATCH_BYTES_PER_COLUMN_SLACK` does not under-report a metric, it
16//! terminates a live `do_get` with an internal error. A hand-written list of six
17//! shapes in the flight crate is exactly how that gap appeared (#2932), so the
18//! guard now runs over EVERY shape the estimator is validated against —
19//! `FixedSizeBinary(16)` (uuid/timeuuid), boolean/decimal/varint/timestamp,
20//! tuple/UDT (Struct), `set`, deeply nested collections, and the
21//! `cql_type = None` flat dispatch arms that route through different builders
22//! (`build_binary_array` / `build_list_array` / `build_map_array`).
23//!
24//! Adding a shape here therefore strengthens BOTH contracts at once; adding a CQL
25//! type the converter handles without adding it here is the omission both tests
26//! exist to catch.
27//!
28//! Compiled only under `cfg(test)` (for the in-crate estimator tests) or the
29//! opt-in `arrow-shape-corpus` feature (for the `cqlite-flight` dev-dependency),
30//! following the `fuzz` / `bench-internals` / `work-counters` precedent: a
31//! default `cargo build -p cqlite-core` links none of it.

41pub fn col(name: &str, data_type: DataType, cql_type: Option<CqlType>) -> ColumnInfo {
42    ColumnInfo {
43        name: name.to_string(),
44        data_type,
45        nullable: true,
46        position: 0,
47        table_name: None,
48        cql_type,
49    }
50}

52pub fn row(pairs: Vec<(&str, Value)>) -> QueryRow {
53    let mut values: HashMap<Arc<str>, Value> = HashMap::new();
54    for (name, value) in pairs {
55        values.insert(Arc::from(name), value);
56    }
57    QueryRow::with_interned_values(RowKey::new(Vec::new()), values)
58}

60/// One named corpus entry: a projected column set plus the rows to convert.
61pub struct Shape {
62    /// Human-readable shape name, quoted in every failure message.
63    pub name: &'static str,
64    /// The projected column set (authoritative `ColumnInfo`, no heuristics).
65    pub columns: Vec<ColumnInfo>,
66    /// The rows to convert.
67    pub rows: Vec<QueryRow>,
68}
69
70pub fn text(s: &str) -> Value {
71    Value::Text(s.as_bytes().to_vec().into())
72}
73
74pub fn blob(n: usize) -> Value {
75    Value::Blob(vec![0xABu8; n].into())
76}

78/// `list<list<…<text>>>` nested `depth` levels deep.
79pub fn nested_list_type(depth: usize) -> CqlType {
80    let mut t = CqlType::Text;
81    for _ in 0..depth {
82        t = CqlType::List(Box::new(t));
83    }
84    t
85}
