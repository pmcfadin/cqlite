1//! CQL → Arrow **schema** mapping (feature = "arrow").
2//!
3//! Split out of `arrow_convert` (epic #1116 file-size split, issue #3096 Phase 0a)
4//! with no behaviour change. This module owns the type side of the conversion —
5//! turning `ColumnInfo` / `CqlType` / `DataType` into Arrow `Field`s and
6//! `DataType`s — while the value side (building the arrays themselves) lives in
7//! the `arrow_builders_*` / `arrow_typed_value` siblings and the entry points in
8//! [`super::arrow_convert`].
9//!
10//! The full CQL → Arrow mapping table (and the rationale for each choice) is
11//! documented on [`super::arrow_convert`].
12

24/// Fixed scale used for all `decimal` columns mapped to `Decimal128`.
25///
26/// We choose 9 (nanosecond-like resolution) as a reasonable default that
27/// accommodates most CQL decimal use-cases without requiring per-column
28/// inspection of the data.
29pub(crate) const DECIMAL_FIXED_SCALE: i32 = 9;
30
31/// Maximum precision for `Decimal128` (Arrow/Parquet limit).
32pub(crate) const DECIMAL_MAX_PRECISION: u8 = 38;
33
34/// Arrow UUID extension type name, as specified by the Arrow spec.
35/// The field metadata key `ARROW:extension:name` = `arrow.uuid` triggers
36/// the Parquet UUID logical type annotation.
37pub(crate) const ARROW_EXTENSION_NAME_KEY: &str = "ARROW:extension:name";
38pub(crate) const ARROW_UUID_EXTENSION_NAME: &str = "arrow.uuid";

44/// Convert a CQL column to an Arrow [`Field`].
45///
46/// When `col.cql_type` is `Some`, the high-fidelity schema mapping
47/// (`cql_type_to_arrow_field`) is used.  For scalar types this produces the
48/// correct Arrow logical type (e.g. `Date32`, `Time64`, `Decimal128`).
49pub(crate) fn column_to_field(col: &ColumnInfo) -> Field {
50    if let Some(cql_type) = &col.cql_type {
51        if let Some(field) = cql_type_to_arrow_field(&col.name, cql_type, col.nullable) {
52            return field;
53        }
54    }
55    let arrow_type = data_type_to_arrow(&col.data_type);
56    Field::new(&col.name, arrow_type, col.nullable)
57}
58
59/// Map a scalar `CqlType` to an Arrow `Field`, returning `None` for complex
60/// or unknown types so the caller can fall back to `data_type_to_arrow`.
61///
62/// UUID and TimeUUID columns receive the canonical Arrow UUID extension
63/// metadata (`ARROW:extension:name` = `arrow.uuid`) so that Parquet readers
64/// emit the Parquet UUID logical type.
65///
66/// `CqlType::Frozen(inner)` transparently unwraps to `inner`.
67pub(crate) fn cql_type_to_arrow_field(
68    name: &str,
69    cql_type: &CqlType,
70    nullable: bool,
71) -> Option<Field> {
72    match cql_type {
73        CqlType::Date => Some(Field::new(name, ArrowDataType::Date32, nullable)),
74        CqlType::Time => Some(Field::new(
75            name,
76            ArrowDataType::Time64(TimeUnit::Nanosecond),
77            nullable,
78        )),
79        CqlType::Decimal => Some(Field::new(
80            name,
81            ArrowDataType::Decimal128(DECIMAL_MAX_PRECISION, DECIMAL_FIXED_SCALE as i8),
82            nullable,
83        )),
84        CqlType::Varint => {
85            // varint → Decimal128(38, 0) — the integer domain.
86            // Values that exceed 38 digits will be detected at write time and
87            // produce an error (never silently truncated).
88            Some(Field::new(
89                name,
90                ArrowDataType::Decimal128(DECIMAL_MAX_PRECISION, 0),
91                nullable,
92            ))
93        }
94        CqlType::Duration => {
95            // NOTE: The Parquet format's INTERVAL logical type does not support
96            // nanosecond precision (only months + days + milliseconds).  The
97            // `parquet` crate v59 therefore refuses to write
98            // `Interval(MonthDayNano)` and returns an NYI error at write time.
99            //
100            // Arrow `Interval(MonthDayNano)` is the correct *Arrow* type for
101            // CQL duration (months + days + nanos), but it cannot be persisted
102            // to Parquet in this crate version.  We fall back to `Utf8` using
103            // the canonical CQL textual representation (e.g. "1mo2d3ns") so
104            // that the data is always readable.  Once the parquet crate gains
105            // MonthDayNano write support this can be upgraded.
106            Some(Field::new(name, ArrowDataType::Utf8, nullable))
107        }
108        CqlType::Uuid | CqlType::TimeUuid => {
109            // FixedSizeBinary(16) with the Arrow UUID extension metadata so that
110            // Parquet readers interpret the column as UUID logical type.
111            Some(uuid_field(name, nullable))
112        }
113        CqlType::Inet => Some(Field::new(name, ArrowDataType::Utf8, nullable)),
114        CqlType::Counter => Some(Field::new(name, ArrowDataType::Int64, nullable)),
115        // List and Set: map to Arrow List with recursively mapped element type.
116        // Arrow has no dedicated Set type; Set is represented as List.
117        // #4114: a `vector<element, n>` decodes to `Value::List` of its elements,
118        // so it takes the SAME Arrow List mapping as a list/set. Arrow's
119        // `FixedSizeList(element, n)` would additionally carry the dimension, but it
120        // needs its own array builder and #4114's scope is the READ path; the value
121        // shape here is correct either way.
122        CqlType::List(inner) | CqlType::Set(inner) | CqlType::Vector(inner, _) => {
123            let item_field = Arc::new(cql_type_to_arrow_child_field("item", inner, true));
124            Some(Field::new(name, ArrowDataType::List(item_field), nullable))
125        }
126        // Frozen<T> is transparent: same Arrow type as T.
127        CqlType::Frozen(inner) => cql_type_to_arrow_field(name, inner, nullable),
128        // Map: emit typed Arrow Map with non-nullable keys and nullable values.
129        // The entries struct is conventionally named "entries" with children
130        // "key" (non-nullable) and "value" (nullable).
131        CqlType::Map(key_type, val_type) => {
132            let entries_field = Arc::new(Field::new(
133                "entries",
134                ArrowDataType::Struct(Fields::from(vec![
135                    cql_type_to_arrow_child_field("key", key_type, false),
136                    cql_type_to_arrow_child_field("value", val_type, true),
137                ])),
138                false,
139            ));
140            Some(Field::new(
141                name,
142                ArrowDataType::Map(entries_field, false),
143                nullable,
144            ))
145        }
146        // Tuple<A,B,…> → Arrow Struct with positional field names.
147        // Zero-field tuples fall back to Utf8 (Arrow Struct requires ≥1 field).
148        CqlType::Tuple(element_types) => {
149            if element_types.is_empty() {
150                return Some(Field::new(name, ArrowDataType::Utf8, nullable));
151            }
152            let struct_type = cql_type_to_arrow_data_type(cql_type);
153            Some(Field::new(name, struct_type, nullable))
154        }
155        // UDT → Arrow Struct with the UDT's field names.
156        // Zero-field UDTs fall back to Utf8 (Arrow Struct requires ≥1 field).
157        CqlType::Udt(_udt_name, udt_fields) => {
158            if udt_fields.is_empty() {
159                return Some(Field::new(name, ArrowDataType::Utf8, nullable));
160            }
161            let struct_type = cql_type_to_arrow_data_type(cql_type);
162            Some(Field::new(name, struct_type, nullable))
163        }
164        // Remaining scalar types are already handled correctly by the flat
165        // DataType mapping; return None to allow that path to run.
166        _ => None,
167    }
168}

170/// Build a `FixedSizeBinary(16)` Field carrying the Arrow UUID extension metadata
171/// (`ARROW:extension:name` = `arrow.uuid`).
172fn uuid_field(name: &str, nullable: bool) -> Field {
173    let mut meta = HashMap::new();
174    meta.insert(
175        ARROW_EXTENSION_NAME_KEY.to_string(),
176        ARROW_UUID_EXTENSION_NAME.to_string(),
177    );
178    Field::new(name, ArrowDataType::FixedSizeBinary(16), nullable).with_metadata(meta)
179}

181/// Build a CHILD `Field` for `cql_type` — the element of a `List`/`Set`/`Vector`, a `Map`
182/// key/value, or a `Tuple`/`Udt` position — carrying the Arrow UUID extension metadata when
183/// `cql_type` (after unwrapping `Frozen`) is `Uuid`/`TimeUuid`.
184///
185/// Issue #4237: [`cql_type_to_arrow_data_type`] returns a bare `ArrowDataType`, which cannot
186/// carry field-level metadata — so before this function existed, every nested UUID/TimeUUID
187/// (inside a collection, map, tuple, or UDT, at ANY depth) lost the `arrow.uuid` extension
188/// metadata that [`cql_type_to_arrow_field`] attaches at the TOP level. Parquet never
189/// noticed (a Parquet reader that wants the UUID logical type reads the metadata when
190/// present and falls back to raw `FixedSizeBinary(16)` otherwise), but Vortex's Arrow
191/// importer REFUSES a bare `FixedSizeBinary(16)` outright — discovered via the
192/// `test_collections`/`test_wide_rows` fixture tables that carry `list<uuid>` /
193/// `map<text, uuid>` / UDT-uuid fields (`cqlite-cli/tests/issue_4237_vortex_parquet_differential.rs`).
194///
195/// This is the single recursion point [`cql_type_to_arrow_data_type`]'s own List/Set/Vector/
196/// Map/Tuple/Udt branches (below) and the identically-shaped value-building constructors in
197/// `arrow_builders_nested.rs` both call, so the schema Field and the array's own embedded
198/// Field can never drift apart (`RecordBatch::try_new` requires them to match exactly,
199/// metadata included, for nested List/Struct/Map arrays).
200pub(crate) fn cql_type_to_arrow_child_field(
201    name: &str,
202    cql_type: &CqlType,
203    nullable: bool,
204) -> Field {
205    match cql_type {
206        CqlType::Uuid | CqlType::TimeUuid => uuid_field(name, nullable),
207        CqlType::Frozen(inner) => cql_type_to_arrow_child_field(name, inner, nullable),
208        _ => Field::new(name, cql_type_to_arrow_data_type(cql_type), nullable),
209    }
210}

216/// Recursively map a `CqlType` to an Arrow `DataType`.
217///
218/// This function is the single source of truth for element-type mapping used
219/// by both the schema-building path (`cql_type_to_arrow_field`) and the
220/// value-building path (`build_typed_value_array`).  It handles all scalar
221/// types and recursively handles `List`, `Set`, `Frozen`, `Map`, `Tuple`,
222/// and `Udt`.
223///
224/// `CqlType::Frozen(inner)` is transparent: the same Arrow type as `inner`.
225///
226/// Zero-field `Tuple` and `Udt` fall back to `Utf8` because Arrow `Struct`
227/// with zero fields cannot be represented in Parquet.
228pub(crate) fn cql_type_to_arrow_data_type(cql_type: &CqlType) -> ArrowDataType {
229    match cql_type {
230        // Scalar types
231        CqlType::Boolean => ArrowDataType::Boolean,
232        CqlType::TinyInt => ArrowDataType::Int8,
233        CqlType::SmallInt => ArrowDataType::Int16,
234        CqlType::Int => ArrowDataType::Int32,
235        CqlType::BigInt => ArrowDataType::Int64,
236        CqlType::Counter => ArrowDataType::Int64,
237        CqlType::Float => ArrowDataType::Float32,
238        CqlType::Double => ArrowDataType::Float64,
239        CqlType::Text | CqlType::Ascii | CqlType::Varchar => ArrowDataType::Utf8,
240        CqlType::Blob => ArrowDataType::Binary,
241        CqlType::Timestamp => ArrowDataType::Timestamp(TimeUnit::Millisecond, Some("UTC".into())),
242        CqlType::Date => ArrowDataType::Date32,
243        CqlType::Time => ArrowDataType::Time64(TimeUnit::Nanosecond),
244        CqlType::Decimal => {
245            ArrowDataType::Decimal128(DECIMAL_MAX_PRECISION, DECIMAL_FIXED_SCALE as i8)
246        }
247        CqlType::Varint => ArrowDataType::Decimal128(DECIMAL_MAX_PRECISION, 0),
248        // Duration: Utf8 fallback (parquet crate v59 MonthDayNano NYI)
249        CqlType::Duration => ArrowDataType::Utf8,
250        CqlType::Uuid | CqlType::TimeUuid => ArrowDataType::FixedSizeBinary(16),
251        // Inet: canonical text form
252        CqlType::Inet => ArrowDataType::Utf8,
253        // List/Set → Arrow List with recursively mapped element type.
254        // Arrow has no dedicated Set type; both map to List.
255        // Same rule as `cql_type_to_arrow_field` above (#4114).
256        CqlType::List(inner) | CqlType::Set(inner) | CqlType::Vector(inner, _) => {
257            ArrowDataType::List(Arc::new(cql_type_to_arrow_child_field("item", inner, true)))
258        }
259        // Frozen<T> is transparent in type mapping.
260        CqlType::Frozen(inner) => cql_type_to_arrow_data_type(inner),
261        // Map: Arrow Map type with typed key (non-nullable) and value (nullable).
262        // The entries struct field is named "entries" with children "key" and
263        // "value".
264        CqlType::Map(key_type, val_type) => ArrowDataType::Map(
265            Arc::new(Field::new(
266                "entries",
267                ArrowDataType::Struct(Fields::from(vec![
268                    cql_type_to_arrow_child_field("key", key_type, false),
269                    cql_type_to_arrow_child_field("value", val_type, true),
270                ])),
271                false,
272            )),
273            false,
274        ),
275        // Tuple<A, B, …> → Struct(field_0: A, field_1: B, …).
276        // Zero-field tuples fall back to Utf8 (Arrow Struct requires ≥1 field).
277        CqlType::Tuple(element_types) => {
278            if element_types.is_empty() {
279                return ArrowDataType::Utf8;
280            }
281            let struct_fields: Vec<Field> = element_types
282                .iter()
283                .enumerate()
284                .map(|(i, t)| {
285                    cql_type_to_arrow_child_field(
286                        &format!("field_{i}"),
287                        t,
288                        true, // tuple positions are always nullable
289                    )
290                })
291                .collect();
292            ArrowDataType::Struct(Fields::from(struct_fields))
293        }
294        // UDT → Struct with the UDT's schema field names and recursively mapped types.
295        // Zero-field UDTs fall back to Utf8 (Arrow Struct requires ≥1 field).
296        CqlType::Udt(_udt_name, udt_fields) => {
297            if udt_fields.is_empty() {
298                return ArrowDataType::Utf8;
299            }
300            let struct_fields: Vec<Field> = udt_fields
301                .iter()
302                .map(|(field_name, field_type)| {
303                    cql_type_to_arrow_child_field(
304                        field_name.as_str(),
305                        field_type,
306                        true, // UDT fields are always nullable (can be unset)
307                    )
308                })
309                .collect();
310            ArrowDataType::Struct(Fields::from(struct_fields))
311        }
312        // Custom/unknown types: Utf8
313        CqlType::Custom(_) => ArrowDataType::Utf8,
314    }
315}

321/// Map CQL `DataType` to Arrow `DataType` (flat fallback path).
322///
323/// This is used when `ColumnInfo.cql_type` is `None`.
324pub(crate) fn data_type_to_arrow(data_type: &DataType) -> ArrowDataType {
325    match data_type {
326        DataType::Null => ArrowDataType::Null,
327        DataType::Boolean => ArrowDataType::Boolean,
328        DataType::TinyInt => ArrowDataType::Int8,
329        DataType::SmallInt => ArrowDataType::Int16,
330        DataType::Integer => ArrowDataType::Int32,
331        DataType::BigInt => ArrowDataType::Int64,
332        DataType::Float32 => ArrowDataType::Float32,
333        DataType::Float => ArrowDataType::Float64,
334        DataType::Text => ArrowDataType::Utf8,
335        DataType::Blob => ArrowDataType::Binary,
336        DataType::Timestamp => ArrowDataType::Timestamp(TimeUnit::Millisecond, Some("UTC".into())),
337        DataType::Uuid => ArrowDataType::FixedSizeBinary(16),
338        DataType::Json => ArrowDataType::Utf8,
339        DataType::List => {
340            ArrowDataType::List(Arc::new(Field::new("item", ArrowDataType::Utf8, true)))
341        }
342        DataType::Set => {
343            ArrowDataType::List(Arc::new(Field::new("item", ArrowDataType::Utf8, true)))
344        }
345        DataType::Map => ArrowDataType::Map(
346            Arc::new(Field::new(
347                "entries",
348                ArrowDataType::Struct(Fields::from(vec![
349                    Field::new("key", ArrowDataType::Utf8, false),
350                    Field::new("value", ArrowDataType::Utf8, true),
351                ])),
352                false,
353            )),
354            false,
355        ),
356        DataType::Tuple => ArrowDataType::Utf8, // Serialize as JSON string
357        DataType::Udt => ArrowDataType::Utf8,   // Serialize as JSON string
358        DataType::Frozen => ArrowDataType::Utf8,
359        DataType::Tombstone => ArrowDataType::Utf8,
360    }
361}

368    fn has_uuid_extension(field: &Field) -> bool {
369        field
370            .metadata()
371            .get(ARROW_EXTENSION_NAME_KEY)
372            .map(String::as_str)
373            == Some(ARROW_UUID_EXTENSION_NAME)
374    }
375
376    /// Issue #4237 (roborev job 4372, Medium): the nested-UUID extension-metadata fix
377    /// (`cql_type_to_arrow_child_field`) is a behavior change to every collection/tuple/UDT
378    /// schema, previously pinned only by two Parquet golden-schema assertions in a
379    /// gate-quarantined test file. This is the gate-EXECUTED pin (runs in `core-tests` and
380    /// the new `vortex-parquet-differential` lane's `export::vortex::`-filtered pass, since
381    /// it lives in this module): a UUID nested at every position this module builds a child
382    /// Field for carries the `arrow.uuid` extension.
383    #[test]
384    fn nested_uuid_carries_extension_metadata_at_every_position() {
385        // list<uuid> -> List(item: FixedSizeBinary(16) + arrow.uuid)
386        let list_field =
387            cql_type_to_arrow_field("ids", &CqlType::List(Box::new(CqlType::Uuid)), true)
388                .expect("List maps to an explicit field");
389        let ArrowDataType::List(item_field) = list_field.data_type() else {
390            panic!("expected List, got {:?}", list_field.data_type());
391        };
392        assert!(
393            has_uuid_extension(item_field),
394            "list<uuid> element field must carry the arrow.uuid extension"
395        );
396
397        // map<text, uuid> -> Map(entries: Struct(key: Utf8, value: FixedSizeBinary(16) + arrow.uuid))
398        let map_field = cql_type_to_arrow_field(
399            "m",
400            &CqlType::Map(Box::new(CqlType::Text), Box::new(CqlType::Uuid)),
401            true,
402        )
403        .expect("Map maps to an explicit field");
404        let ArrowDataType::Map(entries_field, _) = map_field.data_type() else {
405            panic!("expected Map, got {:?}", map_field.data_type());
406        };
407        let ArrowDataType::Struct(entry_fields) = entries_field.data_type() else {
408            panic!(
409                "expected Map entries Struct, got {:?}",
410                entries_field.data_type()
411            );
412        };
413        let value_field = entry_fields
414            .iter()
415            .find(|f| f.name() == "value")
416            .expect("entries struct has a value field");
417        assert!(
418            has_uuid_extension(value_field),
419            "map<text, uuid> value field must carry the arrow.uuid extension"
420        );
421
422        // map<uuid, text> -> the KEY field must also carry it.
423        let map_key_field = cql_type_to_arrow_field(
424            "m2",
425            &CqlType::Map(Box::new(CqlType::Uuid), Box::new(CqlType::Text)),
426            true,
427        )
428        .expect("Map maps to an explicit field");
429        let ArrowDataType::Map(entries_field2, _) = map_key_field.data_type() else {
430            panic!("expected Map, got {:?}", map_key_field.data_type());
431        };
432        let ArrowDataType::Struct(entry_fields2) = entries_field2.data_type() else {
433            panic!("expected Map entries Struct");
434        };
435        let key_field = entry_fields2
436            .iter()
437            .find(|f| f.name() == "key")
438            .expect("entries struct has a key field");
439        assert!(
440            has_uuid_extension(key_field),
441            "map<uuid, text> key field must carry the arrow.uuid extension"
442        );
443
444        // tuple<int, uuid> -> Struct(field_0: Int32, field_1: FixedSizeBinary(16) + arrow.uuid)
445        let tuple_field = cql_type_to_arrow_field(
446            "t",
447            &CqlType::Tuple(vec![CqlType::Int, CqlType::Uuid]),
448            true,
449        )
450        .expect("Tuple maps to an explicit field");
451        let ArrowDataType::Struct(tuple_fields) = tuple_field.data_type() else {
452            panic!("expected Struct, got {:?}", tuple_field.data_type());
453        };
454        assert!(
455            has_uuid_extension(&tuple_fields[1]),
456            "tuple position 1 (uuid) must carry the arrow.uuid extension"
457        );
458
459        // UDT { id: uuid, name: text } -> Struct(id: FixedSizeBinary(16) + arrow.uuid, name: Utf8)
460        let udt_field = cql_type_to_arrow_field(
461            "u",
462            &CqlType::Udt(
463                "my_udt".to_string(),
464                vec![
465                    ("id".to_string(), CqlType::Uuid),
466                    ("name".to_string(), CqlType::Text),
467                ],
468            ),
469            true,
470        )
471        .expect("Udt maps to an explicit field");
472        let ArrowDataType::Struct(udt_fields) = udt_field.data_type() else {
473            panic!("expected Struct, got {:?}", udt_field.data_type());
474        };
475        let udt_id_field = udt_fields
476            .iter()
477            .find(|f| f.name() == "id")
478            .expect("udt struct has an id field");
479        assert!(
480            has_uuid_extension(udt_id_field),
481            "UDT field 'id' (uuid) must carry the arrow.uuid extension"
482        );
483
484        // list<list<uuid>> -> nested List(List(item: FixedSizeBinary(16) + arrow.uuid))
485        let nested_list_field = cql_type_to_arrow_field(
486            "ll",
487            &CqlType::List(Box::new(CqlType::List(Box::new(CqlType::Uuid)))),
488            true,
489        )
490        .expect("nested List maps to an explicit field");
491        let ArrowDataType::List(outer_item) = nested_list_field.data_type() else {
492            panic!("expected List, got {:?}", nested_list_field.data_type());
493        };
494        let ArrowDataType::List(inner_item) = outer_item.data_type() else {
495            panic!("expected nested List, got {:?}", outer_item.data_type());
496        };
497        assert!(
498            has_uuid_extension(inner_item),
499            "list<list<uuid>> innermost element field must carry the arrow.uuid extension"
500        );
501    }

503    /// `cql_type_to_arrow_data_type` (the bare-`ArrowDataType` path used by the value-building
504    /// side) must resolve to the SAME `DataType` shape `cql_type_to_arrow_field` does for a
505    /// container type — this test pins that the two paths did not diverge when
506    /// `cql_type_to_arrow_child_field` was introduced as their shared recursion point.
507    #[test]
508    fn data_type_path_agrees_with_field_path_for_nested_uuid() {
509        let cql = CqlType::List(Box::new(CqlType::Uuid));
510        let via_field = cql_type_to_arrow_field("x", &cql, true)
511            .expect("List maps to an explicit field")
512            .data_type()

515        assert_eq!(via_field, via_data_type);
516    }
