1//! CQL → Arrow RecordBatch conversion (feature = "arrow")
2//!
3//! This module contains the pure CQL-to-Arrow type mapping and array building
4//! logic that was previously embedded in the `parquet` module.  Exposing it
5//! behind a standalone `arrow` cargo feature allows consumers (e.g. a separate
6//! Arrow IPC writer crate) to reuse the conversion without depending on the
7//! `parquet` crate.
8//!
9//! The `parquet` feature depends on the `arrow` feature, so all of the
10//! conversion logic is available to the Parquet writer without code duplication.
11//!
12//! # CQL → Arrow type mapping
13//!
14//! When `ColumnInfo.cql_type` is `Some`, the following high-fidelity mappings
15//! are used instead of the flat `data_type` fallback:
16//!
17//! | CQL type          | Arrow type                            | Notes                             |
18//! |-------------------|---------------------------------------|-----------------------------------|
19//! | date              | `Date32`                              | Signed days since 1970-01-01      |
20//! | time              | `Time64(Nanosecond)`                  | Nanos since midnight              |
21//! | decimal           | `Decimal128(38, DECIMAL_FIXED_SCALE)` | Rescaled; see strategy below      |
22//! | varint            | `Decimal128(38, 0)`                   | Err on >38-digit overflow (fail-closed, never Utf8) |
23//! | duration          | `Utf8` (CQL text form)                | Parquet crate v53 NYI MonthDayNano|
24//! | uuid/timeuuid     | `FixedSizeBinary(16)` + UUID ext      | Arrow UUID extension metadata     |
25//! | inet              | `Utf8`                                | Canonical textual form (deliberate)|
26//! | counter           | `Int64`                               | Unchanged                         |
27//! | list\<X\>         | `List<mapped(X)>`                     | Recursive element mapping         |
28//! | set\<X\>          | `List<mapped(X)>`                     | Arrow has no Set type; uses List  |
29//! | map\<K,V\>        | `Map<Struct(key:K,value:V)>`          | Typed keys/values; nested OK      |
30//! | tuple\<A,B,…\>    | `Struct(field_0:A, field_1:B, …)`     | Positional names; per-position types|
31//! | udt\<name\>       | `Struct(f1:T1, f2:T2, …)`             | Field names from schema; null OK  |
32
33//!
34//! # Module layout (epic #1116 file-size split, issue #3096 Phase 0a)
35//!
36//! This file keeps the public entry points and the top-level column dispatch;
37//! the rest of the converter lives in siblings, with no behaviour change:
38//!
39//! | Module                     | Responsibility                                   |
40//! |----------------------------|--------------------------------------------------|
41//! | `arrow_schema`             | `CqlType`/`DataType` → Arrow `Field`/`DataType`  |
42//! | `arrow_typed_value`        | recursive value → `ArrayRef` dispatcher (scalars)|
43//! | `arrow_builders_scalar`    | per-column scalar/primitive array builders       |
44//! | `arrow_builders_nested`    | collection / tuple / UDT array builders          |
45//! | `arrow_convert_util`       | fail-closed offset guards, `Frozen` unwrapping   |
46//! | `arrow_decimal`            | bounded decimal rescaling (issue #1755)          |
47//! | `arrow_columnar`           | row → column transpose (issue #1495)             |

78/// Errors produced by the CQL → Arrow conversion.
79///
80/// A dedicated `thiserror` enum so that callers (e.g. `ParquetExportError`)
81/// can wrap or delegate to this error without pulling in Parquet-specific
82/// error types.
83#[derive(Debug, Error)]
84pub enum ArrowConvertError {
85    /// Arrow array or schema construction failure.
86    #[error("Arrow error: {0}")]
87    Arrow(#[from] arrow::error::ArrowError),
88    /// A value could not be represented in the target Arrow type.
89    #[error("{0}")]
90    InvalidValue(String),
91    /// A caller-supplied Arrow schema does not describe the column set it was
92    /// passed with (issue #3096 review).
93    ///
94    /// Raised by [`rows_to_record_batch_with_schema`], which is the only entry
95    /// point that accepts a schema it did not build. `RecordBatch::try_new`
96    /// checks field DATA TYPES and array lengths (plus a non-nullable field
97    /// holding actual nulls), so a schema whose fields are REORDERED, RENAMED,
98    /// re-flagged NULLABLE, or stripped of their METADATA is accepted by Arrow and
99    /// silently mislabels every affected column — this variant is what makes
100    /// that a rejection instead.
101    ///
102    /// The two in-crate error-conversion boundaries (`ParquetExportError`,
103    /// `DeltaParquetError`) carry it as their `InvalidValue` **through `Display`**,
104    /// so the prefix above survives the hop. Neither writer supplies its own
105    /// schema today, so neither can raise it; they map it rather than drop it so a
106    /// future caller that does gets the reason instead of a bare Arrow error.
107    #[error("Arrow schema does not match the column set: {0}")]
108    SchemaMismatch(String),
109}

115/// Build an Arrow [`Schema`] from CQL column metadata.
116///
117/// Each column is mapped through the high-fidelity CQL type path when
118/// `col.cql_type` is `Some`, falling back to the flat `DataType` mapping
119/// otherwise.
120///
121/// This is the same schema building logic used by [`rows_to_record_batch`].
122pub fn build_arrow_schema(columns: &[ColumnInfo]) -> Result<Schema, ArrowConvertError> {
123    let fields: Vec<Field> = columns.iter().map(column_to_field).collect();
124    Ok(Schema::new(fields))
125}

127/// Convert a slice of [`QueryRow`] values to an Arrow [`RecordBatch`].
128///
129/// The schema is derived from `columns` via [`build_arrow_schema`].  Each
130/// column is converted to an Arrow array using the same type mapping.
131///
132/// # Cost
133///
134/// The schema is built HERE from `columns` and handed straight to
135/// [`rows_to_record_batch_trusted_schema`], so it is NOT revalidated: this entry
136/// point pays ONE `column_to_field` per column, not two. Routing it through
137/// [`rows_to_record_batch_with_schema`] instead would reconstruct every expected
138/// `Field` a second time to compare it against the fields
139/// [`build_arrow_schema`] had just produced from the same `columns` — pure
140/// duplicate work, since a mismatch is unconstructible on this path (issue
141/// #3096, third review).
142///
143/// # Errors
144///
145/// Returns [`ArrowConvertError`] if any value cannot be represented in the
146/// target Arrow type, or if the Arrow schema/array construction fails.
147pub fn rows_to_record_batch(
148    columns: &[ColumnInfo],
149    rows: &[QueryRow],
150) -> Result<RecordBatch, ArrowConvertError> {
151    let schema = Arc::new(build_arrow_schema(columns)?);
152    // Trusted: `schema` IS `build_arrow_schema(columns)`, one line up.
153    rows_to_record_batch_trusted_schema(schema, columns, rows)
154}
155
156/// Build the arrays for `columns`/`rows` and assemble the batch under a schema
157/// the CALLER HAS ALREADY ESTABLISHED is [`build_arrow_schema`]'s output for
158/// `columns` (issue #3096, third review).
159///
160/// The shared tail of both public entry points, and the only place
161/// `RecordBatch::try_new` is called. What differs is the precondition:
162///
163/// * [`rows_to_record_batch`] built `schema` from these same `columns`
164///   immediately before calling, so it holds by CONSTRUCTION — checking it would
165///   re-derive, per batch, the very fields `build_arrow_schema` had just built.
166/// * [`rows_to_record_batch_with_schema`] accepts a schema it did not build, so
167///   it must PROVE the precondition first via [`check_schema_matches_columns`];
168///   that is a documented public contract and is not weakened by this helper's
169///   existence.
170///
171/// This is deliberately NOT public — it is module-private, reaching the two callers
172/// above and stopping there, so it is not a "skip the checks" door for external
173/// callers, and the schema-mismatch guarantee of the public API is unchanged.
174/// `RecordBatch::try_new` still owns array lengths and field data types, so an
175/// internal caller that broke its precondition would surface as
176/// [`ArrowConvertError::Arrow`] rather than undefined behaviour.
177fn rows_to_record_batch_trusted_schema(
178    schema: Arc<Schema>,
179    columns: &[ColumnInfo],
180    rows: &[QueryRow],
181) -> Result<RecordBatch, ArrowConvertError> {
182    let arrays = convert_to_arrays(columns, rows)?;
183    let batch = RecordBatch::try_new(schema, arrays)?;
184    Ok(batch)
185}

187/// [`rows_to_record_batch`] over a schema the caller already holds.
188///
189/// Identical output — `schema` MUST be the [`build_arrow_schema`] of the same
190/// `columns`, which is what [`rows_to_record_batch`] derives inline. This entry
191/// point exists for a caller whose schema arrives from somewhere it did not derive
192/// it, and its job is to REJECT one that does not describe `columns`; the
193/// `Arc<Schema>` it takes is incidental to that.
194///
195/// It is deliberately NOT a performance door. Reusing one `Arc<Schema>` across a
196/// scan's batches through this entry point is not cheaper than deriving it per
197/// batch — see `# Cost` below — and hoisting the derivation out of the Flight
198/// `do_get` egress loop was measured twice on the WS0 corpus and delivered nothing
199/// (issue #3096, lever 6, reverted: +0.30%, 95% CI covering zero, 4.5x below the
200/// measuring box's between-binary code-layout noise floor).
201///
202/// # What is validated, exactly (issue #3096, second review)
203///
204/// This function accepts a schema it did not build, so the mismatch question is
205/// real — and what is enforced is that `schema` **IS** [`build_arrow_schema`]'s
206/// output for `columns`, not merely that it is compatible with the arrays:
207///
208/// * **Full `Field` identity, position by position** — checked HERE, by
209///   [`check_schema_matches_columns`], against `column_to_field(col)`: the exact
210///   function [`build_arrow_schema`] itself uses, so there is ONE source of truth
211///   and no re-derived mapping that can drift. Arrow's `Field: PartialEq`
212///   compares **name, data type, nullability and metadata** (deliberately
213///   excluding the IPC-only `dict_id`/`dict_is_ordered`, which arrow-schema
214///   documents as irrelevant to schema equality), so every one of those four axes
215///   is covered — including the `ARROW:extension:name` = `arrow.uuid` metadata a
216///   uuid/timeuuid column carries. Field ORDER and ARITY follow from comparing
217///   position by position.
218/// * **Schema-level metadata** — must be empty, because `build_arrow_schema`
219///   builds with `Schema::new`, which sets none.
220/// * **Array lengths** — left to `RecordBatch::try_new`, which owns them.
221///
222/// All of the above is checked here because **`RecordBatch::try_new` does not
223/// check it**: it compares field DATA TYPES and array lengths, and rejects a
224/// non-nullable field that actually holds nulls. It never compares field NAMES,
225/// field METADATA, or schema metadata, and a nullability difference the data does
226/// not contradict passes it. So a schema whose fields are REORDERED or RENAMED
227/// among columns of the same Arrow type, or which drops the uuid extension
228/// metadata, or which flips `nullable`, is accepted by Arrow and silently
229/// mislabels the batch. The first version of this guard checked arity and names
230/// only — leaving nullability and metadata unchecked while documenting the
231/// contract as enforced; the full-identity comparison is what makes the
232/// documented guarantee TRUE.
233///
234/// A near-miss worth naming: `cqlite-flight`'s `MergeProducer::arrow_schema()`
235/// (the WIRE schema for `GetFlightInfo`/`GetSchema`) augments every field with
236/// `cqlite:pushdown` metadata, so it is NOT this function's schema and is
237/// correctly rejected here. The egress path passes
238/// `build_arrow_schema(&self.columns)` unmodified and is unaffected.
239///
240/// # Cost — borne by THIS entry point only
241///
242/// One `column_to_field` construction and one `Field` comparison per COLUMN per
243/// BATCH — for a 12-column table, 12 short `String` allocations plus one small
244/// metadata `HashMap` per uuid/timeuuid column, against per-batch work
245/// proportional to rows x columns.
246///
247/// [`rows_to_record_batch`] does NOT pay it: it goes through the private trusted
248/// tail both entry points share, because its schema is
249/// [`build_arrow_schema`]'s output for the same `columns` by construction (issue
250/// #3096, third review — before that fix it delegated here and every caller paid
251/// the mapping twice).
252///
253/// So state the schema-reuse tradeoff exactly, without overclaiming: reusing one
254/// `Arc<Schema>` across a scan's batches THROUGH THIS ENTRY POINT saves the
255/// `Schema`/`Fields` allocation and a fresh `SchemaRef` per batch, and costs a
256/// `column_to_field` construction plus a `Field` comparison per column per batch.
257/// It is therefore CHEAPER IN ALLOCATIONS but NOT strictly less work per batch than
258/// rebuilding the schema through [`rows_to_record_batch`] — the per-field
259/// construction is paid either way, once as `build_arrow_schema` or once as this
260/// validation.
261///
262/// So a caller emitting many batches over one column set should use
263/// [`rows_to_record_batch`], not this function with a hoisted schema. Issue #3096
264/// pursued the hoist on the Flight `do_get` egress with the validation removed
265/// as well (a schema newtype bound to its columns), measured it against the WS0
266/// corpus, and found nothing to keep: the per-batch work at stake is 1.53
267/// cycles/row of 23,940, so the whole lever was reverted. This entry point remains
268/// for the case it is actually for — a schema the caller did not derive, which must
269/// be REJECTED if it does not describe `columns`.
270///
271/// # Errors
272///
273/// Returns [`ArrowConvertError::SchemaMismatch`] if `schema` is not
274/// [`build_arrow_schema`]'s output for `columns` (field count, order, name, data
275/// type, nullability, field metadata, or schema-level metadata);
276/// [`ArrowConvertError::InvalidValue`] if a value cannot be represented in the
277/// target Arrow type; and [`ArrowConvertError::Arrow`] if array construction
278/// fails or the built arrays do not match `schema`'s field types or lengths.
279pub fn rows_to_record_batch_with_schema(
280    schema: Arc<Schema>,
281    columns: &[ColumnInfo],
282    rows: &[QueryRow],
283) -> Result<RecordBatch, ArrowConvertError> {
284    check_schema_matches_columns(&schema, columns)?;
285    // The precondition is now PROVEN, not assumed, so the shared tail may trust it.
286    rows_to_record_batch_trusted_schema(schema, columns, rows)
287}

289/// Reject a caller-supplied `schema` that is not [`build_arrow_schema`]'s output
290/// for `columns` (issue #3096, second review).
291///
292/// FULL `Field` identity — name, data type, nullability, metadata — position by
293/// position, plus empty schema-level metadata. See
294/// [`rows_to_record_batch_with_schema`] for which of those axes Arrow itself
295/// checks (data types, lengths) and which it silently accepts (everything else),
296/// and for the cost.
297///
298/// The expected field is CONSTRUCTED by `column_to_field`, the same function
299/// [`build_arrow_schema`] uses — nothing is inferred from a name's or a type's
300/// shape, and no part of the mapping is restated here where it could drift
301/// (no-heuristics, #28).
302fn check_schema_matches_columns(
303    schema: &Schema,
304    columns: &[ColumnInfo],
305) -> Result<(), ArrowConvertError> {
306    // Probe instrumentation (issue #3096, third review): the ONLY way a test can
307    // distinguish "the trusted path skipped validation" from "validation ran and
308    // happened to pass", since a schema `build_arrow_schema` just produced can
309    // never FAIL this check. Thread-local, so a `cargo test` thread's count is
310    // unaffected by the other tests running concurrently in the same process, and
311    // compiled out entirely of any non-`cfg(test)` build.
312    #[cfg(test)]
313    SCHEMA_VALIDATIONS.with(|n| n.set(n.get() + 1));
314    if !schema.metadata().is_empty() {
315        return Err(ArrowConvertError::SchemaMismatch(format!(
316            "schema carries top-level metadata {:?} but build_arrow_schema sets none \
317             (it builds with Schema::new), so this is not the schema of the supplied \
318             columns; RecordBatch::try_new never compares schema metadata",
319            sorted_pairs(schema.metadata())
320        )));
321    }
322    let fields = schema.fields();
323    if fields.len() != columns.len() {
324        return Err(ArrowConvertError::SchemaMismatch(format!(
325            "schema has {} field(s) but {} column(s) were supplied",
326            fields.len(),
327            columns.len()
328        )));
329    }
330    for (i, (field, col)) in fields.iter().zip(columns.iter()).enumerate() {
331        let expected = column_to_field(col);
332        if field.as_ref() != &expected {
333            return Err(ArrowConvertError::SchemaMismatch(field_mismatch_reason(
334                i, field, &expected,
335            )));
336        }
337    }
338    Ok(())
339}

341/// The FIRST axis on which `field` differs from the `expected` field of column
342/// `i`, as an operator-facing reason.
343///
344/// One named axis rather than two `Debug` dumps of a whole `Field`, and each arm
345/// says what Arrow would have done with that difference — which is the question a
346/// reader has when a batch is refused. The axes are exactly the four Arrow's
347/// `Field: PartialEq` compares, so the last arm is reached only for a metadata
348/// difference (all four matching means `==` held and this function was not
349/// called).
350fn field_mismatch_reason(i: usize, field: &Field, expected: &Field) -> String {
351    if field.name() != expected.name() {
352        return format!(
353            "field {i} is '{}' but column {i} is '{}' — a reordered or renamed \
354             schema is accepted by RecordBatch::try_new whenever the Arrow types \
355             line up, and would mislabel the batch",
356            field.name(),
357            expected.name()
358        );
359    }
360    if field.data_type() != expected.data_type() {
361        return format!(
362            "field {i} '{}' has Arrow type {:?} but column {i} maps to {:?}",
363            field.name(),
364            field.data_type(),
365            expected.data_type()
366        );
367    }
368    if field.is_nullable() != expected.is_nullable() {
369        return format!(
370            "field {i} '{}' is nullable={} but column {i} maps to nullable={} — \
371             RecordBatch::try_new only rejects a non-nullable field that actually \
372             holds nulls, so this difference is otherwise accepted silently",
373            field.name(),
374            field.is_nullable(),
375            expected.is_nullable()
376        );
377    }
378    format!(
379        "field {i} '{}' carries metadata {:?} but column {i} maps to {:?} — \
380         RecordBatch::try_new never compares field metadata, so dropping or altering \
381         the Arrow UUID extension key is otherwise accepted silently",
382        field.name(),
383        sorted_pairs(field.metadata()),
384        sorted_pairs(expected.metadata())
385    )
386}

389thread_local! {
390    /// How many times [`check_schema_matches_columns`] has run on THIS thread.
391    ///
392    /// Test-only. Exists so a test can assert the negative —
393    /// [`rows_to_record_batch`] performs ZERO schema validations while
394    /// [`rows_to_record_batch_with_schema`] performs exactly one — which is not

396    /// identical batch).
397    static SCHEMA_VALIDATIONS: std::cell::Cell<usize> = const { std::cell::Cell::new(0) };
398}

400/// The current thread's [`SCHEMA_VALIDATIONS`] count. Test-only.
401#[cfg(test)]
402fn schema_validations_on_this_thread() -> usize {
403    SCHEMA_VALIDATIONS.with(|n| n.get())
404}

406/// Metadata as key-sorted pairs, so a rejection message is deterministic.
407///
408/// `Field::metadata` is a `HashMap`, whose `Debug` order is not stable across
409/// runs; a message a test can assert on (and an operator can diff) must be.
410fn sorted_pairs(metadata: &HashMap<String, String>) -> Vec<(&str, &str)> {
411    let mut pairs: Vec<(&str, &str)> = metadata
412        .iter()
413        .map(|(k, v)| (k.as_str(), v.as_str()))
414        .collect();
415    pairs.sort_unstable();
416    pairs
417}

423/// Convert all rows to Arrow arrays (one per column).
424///
425/// Per-column accessors are resolved ONCE via [`transpose_columns`] (issue #1495
426/// / parser epic J1): each builder gets its column's pre-resolved, row-aligned
427/// slice instead of re-hashing `col.name` per cell. Output is byte-identical.
428///
429/// [`transpose_columns`]: super::arrow_columnar::transpose_columns
430pub(crate) fn convert_to_arrays(
431    columns: &[ColumnInfo],
432    rows: &[QueryRow],
433) -> Result<Vec<ArrayRef>, ArrowConvertError> {
434    let columnar = super::arrow_columnar::transpose_columns(columns, rows);
435    columns
436        .iter()
437        .zip(columnar.iter())
438        .map(|(col, cells)| convert_column_to_array(col, cells))
439        .collect()
440}

442/// Convert a single column's pre-resolved, row-aligned value slice to an Arrow
443/// array. `cells[i]` is row `i`'s value (`None` when the column is absent),
444/// resolved once by the transpose — no per-cell `col.name` hashing (issue #1495).
445///
446/// When `col.cql_type` is `Some` and the type is a high-fidelity scalar
447/// (date, time, decimal, varint, duration, uuid/timeuuid, inet, counter),
448/// the corresponding typed builder is used.
449///
450/// For `List`, `Set`, and `Frozen(List|Set)` with a `cql_type`, the
451/// recursive `build_typed_value_array` path is used, which maps element
452/// types through the same scalar mapping above.
453///
454/// All other cases fall through to the existing flat `data_type`-based
455/// dispatch.
456pub(crate) fn convert_column_to_array(
457    col: &ColumnInfo,
458    cells: Cells,
459) -> Result<ArrayRef, ArrowConvertError> {
460    // High-fidelity CQL-type dispatch
461    if let Some(cql_type) = &col.cql_type {
462        // Check if the (possibly Frozen-wrapped) type is a List or Set.
463        let effective = unwrap_frozen_type(cql_type);
464        match effective {
465            CqlType::Date => return build_date32_array(col, cells),
466            CqlType::Time => return build_time64_ns_array(col, cells),
467            CqlType::Decimal => return build_decimal128_array(col, cells),
468            CqlType::Varint => return build_varint_as_decimal128_array(col, cells),
469            CqlType::Duration => return build_duration_utf8_array(col, cells),
470            CqlType::Uuid | CqlType::TimeUuid => return build_uuid_fixed_binary_array(col, cells),
471            CqlType::Inet => return build_inet_utf8_array(col, cells),
472            CqlType::Counter => return build_int64_array(col, cells),
473            // List, Set, Map, Tuple, Udt and Vector: use the recursive typed builder.
474            //
475            // `Vector` MUST be here (issue #4114, roborev job 110). Both
476            // `cql_type_to_arrow_field` and `cql_type_to_arrow_data_type`
477            // (`arrow_schema.rs:127`, `:222`) DECLARE a vector as
478            // `List<element>` — for `vector<float, n>` that is `List<Float32>`.
479            // Without this arm a vector fell through to the flat `DataType`
480            // dispatch below and was built by a path that does NOT produce
481            // `List<Float32>`, so the array did not match the schema the very
482            // same module had declared for it. A RecordBatch whose column array
483            // disagrees with its declared field is invalid, so this was not a
484            // cosmetic mismatch: the export either errors or emits a batch no
485            // reader can trust. The typed builder is the one that honours the
486            // declared element type, which is exactly what a vector needs since
487            // its element type is load-bearing.
488            CqlType::List(_)
489            | CqlType::Set(_)
490            | CqlType::Map(_, _)
491            | CqlType::Tuple(_)
492            | CqlType::Udt(_, _)
493            | CqlType::Vector(_, _) => {
494                return build_typed_value_array(cql_type, cells);
495            }
496            // All other complex/collection types fall through to the flat dispatch.
497            _ => {}
498        }
499    }
500
501    // Flat data_type dispatch (legacy path)
502    match &col.data_type {
503        DataType::Boolean => build_boolean_array(col, cells),
504        DataType::TinyInt => build_int8_array(col, cells),
505        DataType::SmallInt => build_int16_array(col, cells),
506        DataType::Integer => build_int32_array(col, cells),
507        DataType::BigInt => build_int64_array(col, cells),
508        DataType::Float32 => build_float32_array(col, cells),
509        DataType::Float => build_float64_array(col, cells),
510        DataType::Text | DataType::Json => build_string_array(col, cells),
511        DataType::Blob => build_binary_array(col, cells),
512        DataType::Timestamp => build_timestamp_array(col, cells),
513        DataType::Uuid => build_uuid_array(col, cells),
514        DataType::List | DataType::Set => build_list_array(col, cells),
515        DataType::Map => build_map_array(col, cells),
516        DataType::Tuple
517        | DataType::Udt
518        | DataType::Frozen
519        | DataType::Tombstone
520        | DataType::Null => {
521            build_string_array(col, cells) // Fallback to string representation
522        }
523    }
524}
