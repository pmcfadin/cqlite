1//! Panic-hook silencing for the injected faults (issues #3106/#3124), split out of
2//! [`super`] so that file stays under the ~800-line campsite target (epic #1116).
3//!
4//! The whole module is gated by its `mod` declaration in [`super`] — it exists only
5//! where a fault can be armed at all — so no item below repeats that `cfg`.
6
7use super::INJECTED_PANIC_MESSAGE;
8
9/// A boxed panic hook, as [`std::panic::set_hook`] takes it.
10type PanicHook = Box<dyn Fn(&std::panic::PanicHookInfo<'_>) + Sync + Send + 'static>;
11
12/// Suppress the console noise of the INJECTED panic — and only that one — for the
13/// returned guard's lifetime.
14///
15/// Deliberately NOT a blanket `set_hook(|_| {})`: the panic hook is
16/// process-global, so silencing everything would swallow a genuine assertion
17/// failure message from this test or any test running in parallel, exactly the
18/// "masked assertion" failure mode. Panics whose payload does not carry
19/// [`INJECTED_PANIC_MESSAGE`] are delegated to the hook that was installed
20/// before (libtest's capture hook, normally), and that hook is reinstated when
21/// this guard drops — EXCEPT on an unwinding drop, where restoring is skipped
22/// (see [`SilencedInjectedPanics::drop`]).
23#[must_use = "the injected panic is only silenced while the guard is alive"]
24pub fn silence_injected_panics() -> SilencedInjectedPanics {
25    let previous: std::sync::Arc<PanicHook> = std::sync::Arc::new(std::panic::take_hook());
26    let installed = previous.clone();
27    std::panic::set_hook(Box::new(move |info| {
28        if is_injected(info) {
29            return;
30        }
31        installed(info);
32    }));
33    SilencedInjectedPanics { previous }
34}
35
36/// Whether `info` describes an injected fault panic. Matched on the payload
37/// STRING, so no real panic is ever swallowed.
38fn is_injected(info: &std::panic::PanicHookInfo<'_>) -> bool {
39    let payload = info.payload();
40    let message = payload
41        .downcast_ref::<String>()
42        .map(String::as_str)
43        .or_else(|| payload.downcast_ref::<&'static str>().copied());
44    message.is_some_and(|m| m.contains(INJECTED_PANIC_MESSAGE))
45}
46
47/// Guard returned by [`silence_injected_panics`]; restores the previous hook on a
48/// normal drop (see [`Self::drop`] for the unwinding case).
49pub struct SilencedInjectedPanics {
50    previous: std::sync::Arc<PanicHook>,
51}
52
53impl Drop for SilencedInjectedPanics {
54    fn drop(&mut self) {
55        // NEVER touch the hook while unwinding (roborev, issue #3106):
56        // `std::panic::set_hook` PANICS if called from a panicking thread, so a
57        // guard still alive when an assertion (or any fallible call inside the
58        // silenced block) fails would double-panic and ABORT the process — under
59        // libtest's capture that loses the original message entirely and, in an
60        // integration-test binary, takes every sibling test's result with it.
61        // Skipping the restore leaves the filtering hook installed for the rest of
62        // the process, which is harmless: it only ever suppresses
63        // `INJECTED_PANIC_MESSAGE` and delegates everything else.
64        if std::thread::panicking() {
65            return;
66        }
67        // `set_hook` needs an owned `Box`, and the previous hook is behind an
68        // `Arc` (it is borrowed by the filtering hook installed above), so it is
69        // reinstated through one thin delegating wrapper. Behaviourally identical;
70        // the only cost is a pointer hop per nested guard.
71        let previous = self.previous.clone();
72        std::panic::set_hook(Box::new(move |info| previous(info)));
73    }
74}
