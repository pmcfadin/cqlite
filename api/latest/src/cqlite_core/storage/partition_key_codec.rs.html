1//! Canonical partition-key (de)serialization codec.
2//!
3//! This module is the single source of truth for turning the raw partition-key
4//! bytes stored in SSTables (and produced by the write engine) back into typed
5//! [`Value`]s. Both the read path (scan-built rows in the query engine) and the
6//! write path ([`crate::storage::write_engine::mutation::PartitionKey`]) decode
7//! through here so the two never drift apart.
8//!
9//! ## Wire layout (matches Cassandra and `PartitionKey::to_bytes`)
10//!
11//! - **Single-component keys** — raw value bytes, *no* length prefix
12//!   (UUID = 16 bytes, int = 4 BE bytes, text = raw UTF-8, …).
13//! - **Multi-component (composite) keys** — `[len: u16 BE][value bytes][0x00]`
14//!   per component, including a trailing `0x00` after the final component.
15//!
16//! Historically the query engine carried its own ad-hoc decoder that assumed a
17//! `u16` length prefix for *every* TEXT key. That was correct for composite
18//! components but wrong for single-component text partition keys (whose bytes
19//! are raw), so reconstructing a TEXT single-PK column silently failed and the
20//! column was dropped from scan-built rows (Issue #586). Routing both paths
21//! through this module fixes that and prevents a recurrence.
22
23use crate::schema::TableSchema;
24use crate::types::ComparatorType;
25use crate::{Error, Result, Value};

27/// Decode the partition-key columns from their raw on-disk bytes.
28///
29/// Returns the `(column_name, value)` pairs in schema-declared partition-key
30/// order. `data` must be the raw partition-key bytes (single-component: raw
31/// value; multi-component: `[len:u16 BE][value][0x00]` framing).
32pub fn decode_partition_key_columns(
33    data: &[u8],
34    schema: &TableSchema,
35) -> Result<Vec<(String, Value)>> {
36    if schema.partition_keys.is_empty() {
37        return Err(Error::InvalidInput(
38            "Schema has no partition keys".to_string(),
39        ));
40    }
41
42    if data.is_empty() {
43        return Err(Error::InvalidInput("Empty partition key bytes".to_string()));
44    }
45
46    let mut columns = Vec::with_capacity(schema.partition_keys.len());
47
48    if schema.partition_keys.len() == 1 {
49        // Single-component: the whole buffer is the raw value (no length prefix).
50        let key_col = &schema.partition_keys[0];
51        let comparator = ComparatorType::from_data_type(&key_col.data_type)?;
52        let value = deserialize_value_bytes(data, &comparator)?;
53        columns.push((key_col.name.clone(), value));
54    } else {
55        // Multi-component: [len:u16 BE][value bytes][0x00] per component.
56        let mut offset = 0;
57        for key_col in &schema.partition_keys {
58            if offset + 2 > data.len() {
59                return Err(Error::InvalidInput(format!(
60                    "Truncated multi-component partition key at offset {}",
61                    offset
62                )));
63            }
64            let len = u16::from_be_bytes([data[offset], data[offset + 1]]) as usize;
65            offset += 2;
66
67            if offset + len > data.len() {
68                return Err(Error::InvalidInput(format!(
69                    "Partition key component extends beyond data: offset={}, len={}, data_len={}",
70                    offset,
71                    len,
72                    data.len()
73                )));
74            }
75
76            let comparator = ComparatorType::from_data_type(&key_col.data_type)?;
77            let value = deserialize_value_bytes(&data[offset..offset + len], &comparator)?;
78            columns.push((key_col.name.clone(), value));
79            offset += len;
80
81            // Skip the 0x00 end-of-component marker.
82            if offset < data.len() {
83                offset += 1;
84            }
85        }
86    }
87
88    Ok(columns)
89}

91/// Encode partition-key column values into the raw on-disk partition-key bytes.
92///
93/// This is the inverse of [`decode_partition_key_columns`] and produces byte-for-byte
94/// the same layout the write path emits (`PartitionKey::to_bytes`):
95///
96/// - **Single-component keys** — raw value bytes, no length prefix.
97/// - **Multi-component (composite) keys** — `[len:u16 BE][value bytes][0x00]` per
98///   component, including a trailing `0x00` after the final component.
99///
100/// `values` must be in schema-declared partition-key order and have exactly one
101/// entry per partition-key column. Because the encoding reproduces the on-disk raw
102/// key bytes exactly, the result can be matched directly against (a) the partition
103/// RowKeys returned by a scan and (b) the bloom filter / BTI trie, which are keyed
104/// on the same raw bytes. Used by the query engine to turn a fully-constrained
105/// `WHERE pk = ?` into a partition-targeted lookup.
106///
107/// Returns an error (so callers can fall back to a full scan) when the column count
108/// disagrees with the schema or a value cannot be serialized for its declared type.
109pub fn encode_partition_key_columns(values: &[Value], schema: &TableSchema) -> Result<Vec<u8>> {
110    if schema.partition_keys.is_empty() {
111        return Err(Error::InvalidInput(
112            "Schema has no partition keys".to_string(),
113        ));
114    }
115    if values.len() != schema.partition_keys.len() {
116        return Err(Error::InvalidInput(format!(
117            "Partition key column count mismatch: expected {}, got {}",
118            schema.partition_keys.len(),
119            values.len()
120        )));
121    }
122
123    // Serialize one component to its raw value bytes (the framing differs
124    // between single- and multi-component keys, but the value encoding does not).
125    let encode_one = |data_type: &str, value: &Value| -> Result<Vec<u8>> {
126        let comparator = ComparatorType::from_data_type(data_type)?;
127        let coerced = coerce_value_for_comparator(value, &comparator);
128        serialize_value_bytes(&coerced, &comparator)
129    };
130
131    // Single-component key: raw value bytes, no length prefix.
132    if schema.partition_keys.len() == 1 {
133        return encode_one(&schema.partition_keys[0].data_type, &values[0]);
134    }
135
136    // Multi-component: [len:u16 BE][value bytes][0x00] per component.
137    let mut result = Vec::new();
138    for (key_col, value) in schema.partition_keys.iter().zip(values.iter()) {
139        let value_bytes = encode_one(&key_col.data_type, value)?;
140        if value_bytes.len() > u16::MAX as usize {
141            return Err(Error::InvalidInput(format!(
142                "Partition key component too large: {} bytes",
143                value_bytes.len()
144            )));
145        }
146        result.extend_from_slice(&(value_bytes.len() as u16).to_be_bytes());
147        result.extend_from_slice(&value_bytes);
148        result.push(0x00);
149    }
150    Ok(result)
151}

153/// Best-effort coercion of a parsed CQL literal `Value` to the variant the
154/// partition-key serializer expects for `comparator`.
155///
156/// The SELECT parser emits every integer literal as [`Value::BigInt`], so a
157/// `WHERE id = 5` against an `int`/`smallint`/`tinyint`/`timestamp` partition key
158/// would otherwise fail to serialize. This narrows those numeric literals to the
159/// declared type (only when the value fits); every other case is returned
160/// unchanged so an unrepresentable value falls through to a serialization error
161/// and the caller reverts to a full scan.
162///
163/// The UUID and Blob arms are explicit (rather than relying on the catch-all)
164/// to document that the SELECT parser's unquoted-UUID and `0x...` blob literals
165/// (Issue #956) are first-class fast-path inputs: a UUID literal serializes for
166/// both `uuid` and `timeuuid` partition keys (both map to
167/// [`ComparatorType::Uuid`]), and a blob literal for a `blob` key. Keeping them
168/// here as named, no-op coercions guards against a future change to the
169/// serializer silently dropping the #949 fast path for the most common
170/// Cassandra partition-key type.
171fn coerce_value_for_comparator(value: &Value, comparator: &ComparatorType) -> Value {
172    match (value, comparator) {
173        (Value::BigInt(n), ComparatorType::Int)
174            if *n >= i32::MIN as i64 && *n <= i32::MAX as i64 =>
175        {
176            Value::Integer(*n as i32)
177        }
178        (Value::BigInt(n), ComparatorType::SmallInt)
179            if *n >= i16::MIN as i64 && *n <= i16::MAX as i64 =>
180        {
181            Value::SmallInt(*n as i16)
182        }
183        (Value::BigInt(n), ComparatorType::TinyInt)
184            if *n >= i8::MIN as i64 && *n <= i8::MAX as i64 =>
185        {
186            Value::TinyInt(*n as i8)
187        }
188        (Value::BigInt(n), ComparatorType::Timestamp) => Value::Timestamp(*n),
189        (Value::Integer(n), ComparatorType::BigInt) => Value::BigInt(*n as i64),
190        // UUID / TIMEUUID literal -> already in the serializer's expected shape.
191        (Value::Uuid(bytes), ComparatorType::Uuid) => Value::Uuid(*bytes),
192        // Blob literal (0x...) -> matches the `blob` serializer directly.
193        (Value::Blob(bytes), ComparatorType::Blob) => Value::Blob(bytes.clone()),
194        _ => value.clone(),
195    }
196}

198/// Serialize a single partition-key value to its raw on-disk bytes.
199///
200/// Mirrors the write engine's `serialize_value_bytes`
201/// ([`crate::storage::write_engine::mutation`]); kept here as the inverse of
202/// [`deserialize_value_bytes`] so the read-side encoder has no dependency on the
203/// (feature-gated) write engine.
204fn serialize_value_bytes(value: &Value, comparator: &ComparatorType) -> Result<Vec<u8>> {
205    match (value, comparator) {
206        (Value::Boolean(b), ComparatorType::Boolean) => Ok(vec![u8::from(*b)]),
207        (Value::TinyInt(n), ComparatorType::TinyInt) => Ok(vec![*n as u8]),
208        (Value::SmallInt(n), ComparatorType::SmallInt) => Ok(n.to_be_bytes().to_vec()),
209        (Value::Integer(n), ComparatorType::Int) => Ok(n.to_be_bytes().to_vec()),
210        (Value::BigInt(n), ComparatorType::BigInt) => Ok(n.to_be_bytes().to_vec()),
211        (Value::Counter(n), ComparatorType::Counter) => Ok(n.to_be_bytes().to_vec()),
212        (Value::Float32(f), ComparatorType::Float32) => Ok(f.to_bits().to_be_bytes().to_vec()),
213        (Value::Float(f), ComparatorType::Float) => Ok(f.to_bits().to_be_bytes().to_vec()),
214        (Value::Text(s), ComparatorType::Text) => Ok(s.as_bytes().to_vec()),
215        (Value::Blob(bytes), ComparatorType::Blob) => Ok(bytes.clone()),
216        (Value::Timestamp(millis), ComparatorType::Timestamp) => Ok(millis.to_be_bytes().to_vec()),
217        (Value::Date(days), ComparatorType::Date) => {
218            let stored = days.wrapping_sub(i32::MIN) as u32;
219            Ok(stored.to_be_bytes().to_vec())
220        }
221        (Value::Uuid(bytes), ComparatorType::Uuid) => Ok(bytes.to_vec()),
222        (Value::Time(nanos), ComparatorType::Custom(name)) if name == "time" => {
223            Ok(nanos.to_be_bytes().to_vec())
224        }
225        (Value::Inet(bytes), ComparatorType::Custom(name)) if name == "inet" => Ok(bytes.clone()),
226        (Value::Varint(bytes), ComparatorType::Varint) => Ok(bytes.clone()),
227        (Value::Decimal { scale, unscaled }, ComparatorType::Decimal) => {
228            let mut result = Vec::with_capacity(4 + unscaled.len());
229            result.extend_from_slice(&scale.to_be_bytes());
230            result.extend_from_slice(unscaled);
231            Ok(result)
232        }
233        (
234            Value::Duration {
235                months,
236                days,
237                nanos,
238            },
239            ComparatorType::Duration,
240        ) => {
241            let mut result = Vec::with_capacity(16);
242            result.extend_from_slice(&months.to_be_bytes());
243            result.extend_from_slice(&days.to_be_bytes());
244            result.extend_from_slice(&nanos.to_be_bytes());
245            Ok(result)
246        }
247        _ => Err(Error::InvalidInput(format!(
248            "Type mismatch encoding partition key: value {:?} does not match comparator {:?}",
249            value, comparator
250        ))),
251    }
252}

254/// Deserialize a single raw value from its byte-comparable key encoding.
255///
256/// `data` is the raw value bytes for one component (no length prefix). This is
257/// the inverse of the write engine's `serialize_value_bytes`.
258pub fn deserialize_value_bytes(data: &[u8], comparator: &ComparatorType) -> Result<Value> {
259    match comparator {
260        ComparatorType::Boolean => {
261            if data.is_empty() {
262                return Err(Error::InvalidInput("Empty boolean value".to_string()));
263            }
264            Ok(Value::Boolean(data[0] != 0))
265        }
266        ComparatorType::TinyInt => {
267            if data.is_empty() {
268                return Err(Error::InvalidInput("Empty tinyint value".to_string()));
269            }
270            Ok(Value::TinyInt(data[0] as i8))
271        }
272        ComparatorType::SmallInt => {
273            let bytes: [u8; 2] = data.try_into().map_err(|_| {
274                Error::InvalidInput(format!("SmallInt requires 2 bytes, got {}", data.len()))
275            })?;
276            Ok(Value::SmallInt(i16::from_be_bytes(bytes)))
277        }
278        ComparatorType::Int => {
279            let bytes: [u8; 4] = data.try_into().map_err(|_| {
280                Error::InvalidInput(format!("Int requires 4 bytes, got {}", data.len()))
281            })?;
282            Ok(Value::Integer(i32::from_be_bytes(bytes)))
283        }
284        ComparatorType::BigInt => {
285            let bytes: [u8; 8] = data.try_into().map_err(|_| {
286                Error::InvalidInput(format!("BigInt requires 8 bytes, got {}", data.len()))
287            })?;
288            Ok(Value::BigInt(i64::from_be_bytes(bytes)))
289        }
290        ComparatorType::Counter => {
291            let bytes: [u8; 8] = data.try_into().map_err(|_| {
292                Error::InvalidInput(format!("Counter requires 8 bytes, got {}", data.len()))
293            })?;
294            Ok(Value::Counter(i64::from_be_bytes(bytes)))
295        }
296        ComparatorType::Float32 => {
297            let bytes: [u8; 4] = data.try_into().map_err(|_| {
298                Error::InvalidInput(format!("Float32 requires 4 bytes, got {}", data.len()))
299            })?;
300            Ok(Value::Float32(f32::from_bits(u32::from_be_bytes(bytes))))
301        }
302        ComparatorType::Float => {
303            let bytes: [u8; 8] = data.try_into().map_err(|_| {
304                Error::InvalidInput(format!("Float requires 8 bytes, got {}", data.len()))
305            })?;
306            Ok(Value::Float(f64::from_bits(u64::from_be_bytes(bytes))))
307        }
308        ComparatorType::Text => {
309            let s = String::from_utf8(data.to_vec())
310                .map_err(|e| Error::InvalidInput(format!("Invalid UTF-8 in text value: {}", e)))?;
311            Ok(Value::Text(s))
312        }
313        ComparatorType::Blob => Ok(Value::Blob(data.to_vec())),
314        ComparatorType::Timestamp => {
315            let bytes: [u8; 8] = data.try_into().map_err(|_| {
316                Error::InvalidInput(format!("Timestamp requires 8 bytes, got {}", data.len()))
317            })?;
318            Ok(Value::Timestamp(i64::from_be_bytes(bytes)))
319        }
320        ComparatorType::Date => {
321            let bytes: [u8; 4] = data.try_into().map_err(|_| {
322                Error::InvalidInput(format!("Date requires 4 bytes, got {}", data.len()))
323            })?;
324            let stored = u32::from_be_bytes(bytes);
325            Ok(Value::Date((stored as i32).wrapping_add(i32::MIN)))
326        }
327        ComparatorType::Uuid => {
328            let bytes: [u8; 16] = data.try_into().map_err(|_| {
329                Error::InvalidInput(format!("UUID requires 16 bytes, got {}", data.len()))
330            })?;
331            Ok(Value::Uuid(bytes))
332        }
333        ComparatorType::Custom(name) if name == "time" => {
334            let bytes: [u8; 8] = data.try_into().map_err(|_| {
335                Error::InvalidInput(format!("Time requires 8 bytes, got {}", data.len()))
336            })?;
337            Ok(Value::Time(i64::from_be_bytes(bytes)))
338        }
339        ComparatorType::Custom(name) if name == "inet" => Ok(Value::Inet(data.to_vec())),
340        ComparatorType::Varint => Ok(Value::Varint(data.to_vec())),
341        ComparatorType::Decimal => {
342            if data.len() < 4 {
343                return Err(Error::InvalidInput(format!(
344                    "Decimal requires at least 4 bytes, got {}",
345                    data.len()
346                )));
347            }
348            // Length already validated above
349            let scale_bytes: [u8; 4] = data[..4]
350                .try_into()
351                .map_err(|_| Error::InvalidInput("Decimal scale conversion failed".to_string()))?;
352            let scale = i32::from_be_bytes(scale_bytes);
353            let unscaled = data[4..].to_vec();
354            Ok(Value::Decimal { scale, unscaled })
355        }
356        ComparatorType::Duration => {
357            if data.len() < 16 {
358                return Err(Error::InvalidInput(format!(
359                    "Duration requires 16 bytes, got {}",
360                    data.len()
361                )));
362            }
363            let months = i32::from_be_bytes(data[..4].try_into().map_err(|_| {
364                Error::InvalidInput("Duration months conversion failed".to_string())
365            })?);
366            let days =
367                i32::from_be_bytes(data[4..8].try_into().map_err(|_| {
368                    Error::InvalidInput("Duration days conversion failed".to_string())
369                })?);
370            let nanos = i64::from_be_bytes(data[8..16].try_into().map_err(|_| {
371                Error::InvalidInput("Duration nanos conversion failed".to_string())
372            })?);
373            Ok(Value::Duration {
374                months,
375                days,
376                nanos,
377            })
378        }
379        _ => Err(Error::InvalidInput(format!(
380            "Unsupported comparator for deserialization: {:?}",
381            comparator
382        ))),
383    }
384}

578    /// Drift guard (review follow-up): this module's encoder must stay
579    /// byte-for-byte identical to the write engine's `PartitionKey::to_bytes`,
580    /// because a write persists keys with one and the read-side bloom/BTI prune
581    /// matches with the other. If they ever diverge, a `WHERE pk = ?` could prune
582    /// away the very SSTable holding the row. Values are pre-typed so no coercion
583    /// is involved — we're comparing the raw serializers and framing directly.
584    #[cfg(feature = "write-support")]
585    #[test]
586    fn encoder_matches_write_engine_partition_key_to_bytes() {
587        use crate::storage::write_engine::mutation::PartitionKey;
588
589        let uuid = [
590            0u8, 35, 236, 231, 124, 78, 71, 5, 144, 104, 209, 165, 158, 197, 254, 25,
591        ];
592        // (partition-key column (name, type) list, expected (name, value) list).
593        type Case<'a> = (Vec<(&'a str, &'a str)>, Vec<(&'a str, Value)>);
594        let cases: Vec<Case> = vec![
595            (
596                vec![("id", "text")],
597                vec![("id", Value::Text("k123".to_string()))],
598            ),
599            (vec![("id", "int")], vec![("id", Value::Integer(42))]),
600            (vec![("id", "bigint")], vec![("id", Value::BigInt(-7))]),
601            (vec![("id", "uuid")], vec![("id", Value::Uuid(uuid))]),
602            (
603                vec![("app", "text"), ("metric", "text")],
604                vec![
605                    ("app", Value::Text("a".to_string())),
606                    ("metric", Value::Text("view".to_string())),
607                ],
608            ),
609            (
610                vec![("app", "text"), ("part", "int")],
611                vec![
612                    ("app", Value::Text("svc".to_string())),
613                    ("part", Value::Integer(3)),
614                ],
615            ),
616        ];
617
618        for (pks, values) in cases {
619            let schema = schema_with_pks(&pks);
620            let value_only: Vec<Value> = values.iter().map(|(_, v)| v.clone()).collect();
621            let columns: Vec<(String, Value)> = values
622                .iter()
623                .map(|(n, v)| (n.to_string(), v.clone()))
624                .collect();
625
626            let codec_bytes = encode_partition_key_columns(&value_only, &schema).unwrap();
627            let write_bytes = PartitionKey::new(columns).to_bytes(&schema).unwrap();
628            assert_eq!(
629                codec_bytes, write_bytes,
630                "codec encoder and write-engine PartitionKey::to_bytes disagree for {pks:?}",
631            );
632        }
633    }
