1//! Minimal per-table schema for schema-aware CommitLog mutation decoding
2//! (issue #2389).
3//!
4//! Cassandra's CommitLog mutation stream is only partially self-describing: the
5//! per-partition messaging header carries column **names**, but clustering and
6//! cell **values** are serialized by their CQL type's fixed/variable-length
7//! rule (`AbstractType.writeValue`) with no on-wire type tag. Fixed-length types
8//! (int, bigint, uuid, …) are written with **no length prefix**, so decoding
9//! them requires knowing the type — exactly the schema-aware posture CQLite's
10//! SSTable reader takes (no-heuristics, issue #28: types come from the schema,
11//! never guessed from bytes).
12//!
13//! Without a schema, [`super::CommitLogReader`] still decodes every mutation's
14//! table id, partition key, and column names structurally; a schema unlocks full
15//! clustering/cell value decode.
16
17use std::collections::HashMap;
18
19/// A column's name and CQL type name (e.g. `("age", "int")`).
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub struct ColumnSpec {
22    /// Column name as it appears in the messaging header.
23    pub name: String,
24    /// CQL type name, lowercased (e.g. `int`, `text`, `uuid`).
25    pub type_name: String,
26}
27
28impl ColumnSpec {
29    /// Construct a spec, normalizing the type name to lowercase.
30    pub fn new(name: impl Into<String>, type_name: impl Into<String>) -> Self {
31        Self {
32            name: name.into(),
33            type_name: type_name.into().to_ascii_lowercase(),
34        }
35    }
36}
37
38/// A single table's schema, keyed into a decode by its 16-byte table id.
39#[derive(Debug, Clone, PartialEq, Eq)]
40pub struct CommitLogSchema {
41    /// Keyspace name (informational).
42    pub keyspace: String,
43    /// Table name (informational).
44    pub table: String,
45    /// Partition-key columns in order.
46    pub partition_key: Vec<ColumnSpec>,
47    /// Clustering columns in order.
48    pub clustering: Vec<ColumnSpec>,
49    /// Regular + static columns, looked up by name during cell decode.
50    pub columns: Vec<ColumnSpec>,
51}
52
53impl CommitLogSchema {
54    /// Look up a regular/static column's CQL type by name.
55    pub fn column_type(&self, name: &str) -> Option<&str> {
56        self.columns
57            .iter()
58            .find(|c| c.name == name)
59            .map(|c| c.type_name.as_str())
60    }
61}
62
63/// A set of table schemas keyed by their 16-byte table id.
64pub type SchemaSet = HashMap<[u8; 16], CommitLogSchema>;
65
66/// Fixed serialized length of a CQL type's value, or `None` if variable-length.
67///
68/// Mirrors `AbstractType.valueLengthIfFixed()` for the Cassandra 5.0 types.
69/// Variable-length types (text/blob/varint/decimal/inet/collections/…) return
70/// `None` and are decoded with an unsigned-vint length prefix.
71pub fn cql_fixed_len(type_name: &str) -> Option<usize> {
72    match type_name.trim().to_ascii_lowercase().as_str() {
73        "boolean" => Some(1),
74        "int" | "float" => Some(4),
75        "bigint" | "timestamp" | "double" => Some(8),
76        "uuid" | "timeuuid" => Some(16),
77        // `tinyint`/`smallint`/`date`/`time` are NOT fixed-length. Per Cassandra
78        // 5.0.2 marshal source, `ByteType`(tinyint)/`ShortType`(smallint) extend
79        // `NumberType`, and `SimpleDateType`(date)/`TimeType`(time) extend
80        // `TemporalType`; none of them — nor their parents — override
81        // `AbstractType.valueLengthIfFixed()`, so all four inherit
82        // `VARIABLE_LENGTH == -1` and are written via `writeWithVIntLength`
83        // (unsigned-vint length prefix + bytes), exactly like text/blob.
84        // Classifying them as `Fixed(1/2/4/8)` would read the first value byte
85        // as the vint length and misalign the cursor for every subsequent cell
86        // (mirrors the AUTHORITY NOTE in parser/repair_clustering.rs). They fall
87        // through to `None` below (the vint-length path) and stay simple scalars
88        // on the `is_simple_scalar_type` allowlist.
89        //
90        // `counter` is NOT fixed-8 either: CounterColumnType extends NumberType (not
91        // LongType) and does not override valueLengthIfFixed(), so a counter
92        // cell's on-disk value is a vint-length-prefixed CounterContext blob
93        // (header + shards), not a raw i64. Treating it as fixed-8 silently
94        // misaligned every subsequent field in the partition (roborev
95        // finding, review-first pass). Falls through to `None` below, which
96        // reads it exactly like blob — the vint-length envelope is what
97        // matters for staying aligned; this module doesn't further interpret
98        // the CounterContext's internal shard structure.
99        _ => None,
100    }
101}

103/// Column types this module can decode without misaligning the cursor.
104///
105/// Cassandra's row format serializes a "complex" column (collection/tuple/
106/// UDT/vector) as a categorically different wire structure — an optional
107/// complex-deletion time, then a count-prefixed set of (path, cell) pairs —
108/// not the single simple `Cell` this module's `read_cell` decodes. Treating a
109/// complex column as simple doesn't just read the wrong length, it reads the
110/// wrong SHAPE, silently misaligning every subsequent row/field while still
111/// reporting `rows_decoded == true` (roborev finding, review-first pass: the
112/// module doc already claimed this was surfaced honestly — it was not,
113/// because nothing actually checked for it). Prose-matched, not a byte-level
114/// verification against Cassandra 5.0 source (unavailable in this
115/// environment) — kept conservative (allowlist, not denylist) so an unknown
116/// type name bails rather than being assumed simple.
117pub fn is_simple_scalar_type(type_name: &str) -> bool {
118    let t = type_name.trim().to_ascii_lowercase();
119    matches!(
120        t.as_str(),
121        "boolean"
122            | "tinyint"
123            | "smallint"
124            | "int"
125            | "date"
126            | "float"
127            | "bigint"
128            | "timestamp"
129            | "time"
130            | "double"
131            | "uuid"
132            | "timeuuid"
133            | "text"
134            | "varchar"
135            | "ascii"
136            | "blob"
137            | "varint"
138            | "decimal"
139            | "inet"
140            | "counter"
141    )
142}

144/// Format a 16-byte table id as a canonical UUID string
145/// (`xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx`).
146pub fn format_table_id(id: &[u8; 16]) -> String {
147    let hex: String = id.iter().map(|b| format!("{b:02x}")).collect();
148    format!(
149        "{}-{}-{}-{}-{}",
150        &hex[0..8],
151        &hex[8..12],
152        &hex[12..16],
153        &hex[16..20],
154        &hex[20..32],
155    )
156}
157
158/// Parse a canonical UUID string into 16 bytes. Returns `None` on malformed
159/// input.
160pub fn parse_table_id(s: &str) -> Option<[u8; 16]> {
161    let hex: String = s.chars().filter(|c| *c != '-').collect();
162    // `hex.len()` is a BYTE length; a non-ASCII multi-byte char could pass a
163    // `!= 32` byte-length guard while still slicing across a UTF-8 char
164    // boundary below and panicking — a public API taking untrusted input
165    // must return None, never panic (roborev finding, review-first pass).
166    if !hex.is_ascii() || hex.len() != 32 {
167        return None;
168    }
169    let mut out = [0u8; 16];
170    for (i, byte) in out.iter_mut().enumerate() {
171        *byte = u8::from_str_radix(&hex[i * 2..i * 2 + 2], 16).ok()?;
172    }
173    Some(out)
174}

176#[cfg(test)]
177mod tests {
178    use super::*;

180    #[test]
181    fn fixed_lengths_match_cassandra() {
182        assert_eq!(cql_fixed_len("int"), Some(4));
183        assert_eq!(cql_fixed_len("bigint"), Some(8));
184        assert_eq!(cql_fixed_len("uuid"), Some(16));
185        assert_eq!(cql_fixed_len("boolean"), Some(1));
186        assert_eq!(cql_fixed_len("text"), None);
187        assert_eq!(cql_fixed_len("blob"), None);
188        assert_eq!(cql_fixed_len("varint"), None);
189    }
190
191    #[test]
192    fn table_id_round_trips() {
193        let s = "d6de7150-8448-11f1-bf5a-bf4cbc47cb4a";
194        let id = parse_table_id(s).expect("parse");
195        assert_eq!(format_table_id(&id), s);
196    }
197
198    #[test]
199    fn counter_is_not_fixed_length() {
200        // Regression: CounterColumnType does not override valueLengthIfFixed()
201        // — a counter cell is a vint-length-prefixed CounterContext blob, not
202        // a raw fixed-8 i64 (roborev finding, review-first pass).
203        assert_eq!(cql_fixed_len("counter"), None);
204    }
205
206    #[test]
207    fn tinyint_smallint_date_time_are_not_fixed_length() {
208        // Regression: none of ByteType(tinyint)/ShortType(smallint)/
209        // SimpleDateType(date)/TimeType(time) override valueLengthIfFixed() in
210        // Cassandra 5.0.2 — they inherit VARIABLE_LENGTH and are written with a
211        // vint length prefix (writeWithVIntLength), exactly like text/blob.
212        // Treating them as Fixed(1/2/4/8) reads the first value byte as the vint
213        // length and misaligns every subsequent cell (PR #2797 blocker).
214        for t in ["tinyint", "smallint", "date", "time"] {
215            assert_eq!(cql_fixed_len(t), None, "{t} must be variable-length");
216            // Still a simple scalar — just a variable-length one.
217            assert!(is_simple_scalar_type(t), "{t} stays a simple scalar");
218        }
219    }
220
221    #[test]
222    fn simple_scalar_type_allowlist_excludes_complex_types() {

224            "int", "bigint", "text", "blob", "uuid", "counter", "decimal", "varint", "inet",
225        ] {
226            assert!(is_simple_scalar_type(t), "{t} should be a simple scalar");

233            "tuple<int, text>",
234            "vector<float, 4>",

236        ] {
