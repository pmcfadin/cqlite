1//! Decoded CommitLog mutation representation + the mutation-body decoder
2//! (issue #2389).
3//!
4//! A record body (validated by [`super::frame`]) is a serialized Cassandra
5//! `Mutation` in **messaging** form (`Mutation.serializer` /
6//! `UnfilteredRowIteratorSerializer` with `isForSSTable() == false`). The
7//! grammar below is verified byte-for-byte against a real Cassandra 5.0.2
8//! segment:
9//!
10//! ```text
11//! mutation:  uvint numUpdates, then numUpdates × partition-update
12//! update:    tableId(16), UnfilteredRowIterator
13//! iterator:  writeWithVIntLength(partitionKey), iterFlags(1),
14//!            EncodingStats(3 uvints), [static columns], regular columns,
15//!            [partition deletion], [static row], [row estimate uvint],
16//!            unfiltereds… END_OF_PARTITION(0x01)
17//! row:       flags(1) [ext flags], clustering values, [liveness], [deletion],
18//!            [column subset], cells…
19//! cell:      flags(1), [ts delta], [localDeletion], [ttl], [path], [value]
20//! ```
21//!
22//! This module fully decodes the common insert path (simple columns, all
23//! columns present, no clustering/static/complex/subset/deletion). Constructs
24//! it does not fully model (clustering columns — the `ClusteringPrefix`
25//! presence-header is not decoded, see `decode_rows`'s bail — static rows,
26//! complex/collection columns, column subsets, range tombstone markers,
27//! row/partition deletions) are surfaced honestly: the affected partition's
28//! [`PartitionUpdate::rows_decoded`] is set `false` rather than guessing byte
29//! offsets — each mutation record is independently CRC-framed and
30//! length-delimited, so bailing on one body never disturbs the others.
31

37// ---- UnfilteredRowIterator flags -----------------------------------------
38const ITER_IS_EMPTY: u8 = 0x01;
39const ITER_HAS_PARTITION_DELETION: u8 = 0x04;
40const ITER_HAS_STATIC_ROW: u8 = 0x08;
41const ITER_HAS_ROW_ESTIMATE: u8 = 0x10;
42
43// ---- Unfiltered (row / marker) flags -------------------------------------
44const ROW_END_OF_PARTITION: u8 = 0x01;
45const ROW_IS_MARKER: u8 = 0x02;
46const ROW_HAS_TIMESTAMP: u8 = 0x04;
47const ROW_HAS_TTL: u8 = 0x08;
48const ROW_HAS_DELETION: u8 = 0x10;
49const ROW_HAS_ALL_COLUMNS: u8 = 0x20;
50const ROW_HAS_COMPLEX_DELETION: u8 = 0x40;
51const ROW_EXTENSION_FLAG: u8 = 0x80;
52
53// ---- Cell flags -----------------------------------------------------------
54const CELL_IS_DELETED: u8 = 0x01;
55const CELL_IS_EXPIRING: u8 = 0x02;
56const CELL_HAS_EMPTY_VALUE: u8 = 0x04;
57const CELL_USE_ROW_TIMESTAMP: u8 = 0x08;
58const CELL_USE_ROW_TTL: u8 = 0x10;

60/// A decoded CommitLog mutation: one or more partition updates.
61#[derive(Debug, Clone, PartialEq, Eq)]
62pub struct Mutation {
63    /// The partition updates carried by this mutation (one per table touched).
64    /// May be a PREFIX of the mutation's true update count — see
65    /// [`Mutation::updates_complete`].
66    pub updates: Vec<PartitionUpdate>,
67    /// `false` when a batch mutation declared more partition updates than
68    /// `updates` contains — the loop stops at the first update whose body
69    /// isn't fully consumed (no schema, or an unmodeled construct), since a
70    /// partial decode can't locate the next update's offset without the
71    /// schema. This is the common case for `open()` with no schemas: a
72    /// multi-table batch silently reported only its first update with no
73    /// signal that more existed, until this field was added (roborev
74    /// finding, review-first pass).
75    pub updates_complete: bool,
76}
77
78/// A single-table partition update within a mutation.
79#[derive(Debug, Clone, PartialEq, Eq)]
80pub struct PartitionUpdate {
81    /// 16-byte table id (Cassandra `TableId`).
82    pub table_id: [u8; 16],
83    /// Raw partition-key bytes (schema-typed decoding is the caller's choice).
84    pub partition_key: Vec<u8>,
85    /// Regular column names from the messaging header, in wire order.
86    pub column_names: Vec<String>,
87    /// Whether the partition carries a partition-level deletion.
88    pub has_partition_deletion: bool,
89    /// Fully-decoded rows. Empty when `rows_decoded` is `false`.
90    pub rows: Vec<DecodedRow>,
91    /// `true` when every row/cell was decoded; `false` when a schema was
92    /// unavailable or the partition used a construct this decoder does not
93    /// model (surfaced honestly rather than guessed).
94    pub rows_decoded: bool,
95}
96
97impl PartitionUpdate {
98    /// The table id formatted as a canonical UUID string.
99    pub fn table_id_uuid(&self) -> String {
100        format_table_id(&self.table_id)
101    }
102}

104/// A decoded row within a partition update.
105#[derive(Debug, Clone, PartialEq, Eq)]
106pub struct DecodedRow {
107    /// Clustering-column values in order (raw bytes; empty for a no-clustering
108    /// table).
109    pub clustering: Vec<Vec<u8>>,
110    /// Decoded cells for the row's present columns.
111    pub cells: Vec<DecodedCell>,
112    /// Whether the row itself is a row deletion (tombstone).
113    pub is_row_deletion: bool,
114}

116/// A decoded cell (one column value within a row).
117#[derive(Debug, Clone, PartialEq, Eq)]
118pub struct DecodedCell {
119    /// Column name.
120    pub column: String,
121    /// Raw value bytes, or `None` for a tombstone / empty-value cell.
122    pub value: Option<Vec<u8>>,
123    /// Whether the cell is a tombstone (deleted).
124    pub deleted: bool,
125}

127/// Decode a mutation record body into a [`Mutation`].
128///
129/// `schemas` supplies per-table-id schemas for full clustering/cell decode; a
130/// table with no schema decodes structurally (table id + partition key + column
131/// names) with `rows_decoded == false`.
132///
133/// # Errors
134/// [`Error::CorruptCommitLogFrame`] if the body runs out of bytes mid-field
135/// (should not happen for a CRC-validated record, but is handled defensively —
136/// never panics).
137pub fn decode_mutation(body: &[u8], schemas: &SchemaSet) -> Result<Mutation> {
138    let mut c = Cursor::new(body);
139    let num_updates = c.uvint()?;
140    // Guard against a maliciously huge declared count spinning the loop below:
141    // each update needs at least a few bytes (a 16-byte table id alone), so a
142    // count exceeding the body length is already provably impossible — reject
143    // it outright rather than relying on each iteration's short-read failure
144    // to terminate a `for _ in 0..num_updates` driven by an untrusted u64
145    // (roborev finding — a hang risk even though it fails fast today, review-
146    // first pass).
147    if num_updates > body.len() as u64 {
148        return Err(Error::CorruptCommitLogFrame(format!(
149            "mutation declares {num_updates} updates, impossible for a {}-byte body",
150            body.len()
151        )));
152    }
153    let mut updates = Vec::with_capacity(num_updates.min(1024) as usize);
154    let mut updates_complete = true;
155    for _ in 0..num_updates {
156        // Each update is decoded either fully (cursor left at the next update) or
157        // partially (structural fields only — no schema, or an unmodeled
158        // construct). A partial decode cannot locate the following update's
159        // offset without the schema, so we stop the update loop and return what
160        // we have. The record's frame CRC already proved the bytes are intact;
161        // an under-reported batch is honest, not corruption — updates_complete
162        // makes that honesty visible to the caller instead of a silent
163        // truncation (roborev finding, review-first pass).
164        let (update, consumed_fully) = decode_partition_update(&mut c, schemas)?;
165        updates.push(update);
166        if !consumed_fully {
167            updates_complete = false;
168            break;
169        }
170    }
171    Ok(Mutation {
172        updates,
173        updates_complete,
174    })
175}

177fn decode_partition_update(
178    c: &mut Cursor<'_>,
179    schemas: &SchemaSet,
180) -> Result<(PartitionUpdate, bool)> {
181    let table_id = c.take_array16()?;
182    let pk_len = c.uvint()? as usize;
183    let partition_key = c.take(pk_len)?.to_vec();
184    let iter_flags = c.u8()?;
185
186    let mut update = PartitionUpdate {
187        table_id,
188        partition_key,
189        column_names: Vec::new(),
190        // Set from the authoritative iter_flags bit immediately, before any
191        // early return — has_partition_deletion is a structural fact readable
192        // right here, independent of rows_decoded, so it must not silently
193        // default to false on the static-row/empty-partition paths below.
194        // Those early returns previously left it false even when the
195        // partition genuinely carried a deletion, reporting an
196        // authoritative-looking "not deleted" to callers (CLI JSON, library
197        // consumers) that was simply never checked (roborev finding, review-
198        // first pass).
199        has_partition_deletion: iter_flags & ITER_HAS_PARTITION_DELETION != 0,
200        rows: Vec::new(),
201        rows_decoded: false,
202    };
203
204    if iter_flags & ITER_IS_EMPTY != 0 {
205        // Empty partition (e.g. a partition-only delete carrier). The update body
206        // is exactly tableId + pk + flags — fully consumed.
207        update.rows_decoded = true;
208        return Ok((update, true));
209    }
210
211    // EncodingStats: minTimestamp, minLocalDeletionTime, minTTL (all uvints).
212    let _min_timestamp = c.uvint()?;
213    let _min_ldt = c.uvint()?;
214    let _min_ttl = c.uvint()?;
215
216    let has_static = iter_flags & ITER_HAS_STATIC_ROW != 0;
217    if has_static {
218        // Static rows: NOT modeled. Whether Cassandra's
219        // UnfilteredRowIteratorSerializer writes a static-columns block gated
220        // on this per-partition iterator flag (the prior assumption here), or
221        // unconditionally based on the table's static-column set regardless
222        // of whether THIS iterator instance carries a static row, is
223        // unverified against real Cassandra 5.0 source (unavailable in this
224        // environment — see the appendix-h doc). Bail BEFORE reading any
225        // columns block at all: under either assumption, guessing wrong here
226        // would misalign the cursor and — worse — would leave
227        // update.column_names populated with misparsed bytes that then get
228        // surfaced to callers (including the CLI's structural-only output)
229        // as if authoritative, even though rows_decoded is false (roborev
230        // finding, review-first pass).
231        return Ok((update, false));
232    }
233    update.column_names = read_column_names(c)?;
234
235    // Constructs we do not fully model: bail structurally (honest, no guessing).
236    // The cursor is left mid-body, so this update is NOT fully consumed.
237    if update.has_partition_deletion {
238        return Ok((update, false));
239    }
240
241    if iter_flags & ITER_HAS_ROW_ESTIMATE != 0 {
242        let _row_estimate = c.uvint()?;
243    }
244
245    let schema = match schemas.get(&table_id) {
246        Some(s) => s,
247        None => return Ok((update, false)), // structural-only decode
248    };
249
250    match decode_rows(c, schema, &update.column_names) {
251        Ok(rows) => {
252            update.rows = rows;
253            update.rows_decoded = true;
254            // decode_rows consumed through END_OF_PARTITION.
255            Ok((update, true))
256        }
257        Err(_) => {
258            // A construct beyond this decoder's model was hit; keep the
259            // structural fields and report rows as not decoded.
260            update.rows.clear();
261            update.rows_decoded = false;
262            Ok((update, false))
263        }
264    }
265}

267/// A sentinel error used internally to bail out of the full row decode into the
268/// structural-only path without treating it as a hard corruption error.
269struct Unsupported;
270type RowResult<T> = std::result::Result<T, Unsupported>;
271
272fn decode_rows(
273    c: &mut Cursor<'_>,
274    schema: &CommitLogSchema,
275    column_names: &[String],
276) -> RowResult<Vec<DecodedRow>> {
277    // Clustered tables: NOT modeled. Cassandra's `ClusteringPrefix.Serializer`
278    // writes a `writeUnsignedVInt` PRESENCE HEADER (2 bits per clustering
279    // column, batched per 32 columns: 0=present, 1=empty, 2=null) before the
280    // present/non-empty values — see the SSTable reader's
281    // `row_decoder/row_framing.rs` (issue #213) for the canonical decode of
282    // this exact header. A naive per-column value loop (no header) misreads
283    // the header bytes as the first value's bytes, silently misaligning every
284    // subsequent field. Bail honestly rather than guess, matching this
285    // module's existing posture for other unmodeled constructs (range
286    // tombstone markers, partition/static-row deletions) — decoding it is a
287    // follow-up, not a v1 requirement.
288    if !schema.clustering.is_empty() {
289        return Err(Unsupported);
290    }
291    // Complex columns (collection/tuple/UDT/vector): NOT modeled. Cassandra
292    // serializes a complex column as an entirely different wire shape (an
293    // optional complex-deletion time, then a count-prefixed set of
294    // (cell-path, cell) pairs) — not the single simple Cell this module's
295    // read_cell decodes. The module doc previously claimed this was
296    // "surfaced honestly", but nothing actually checked for it: a schema
297    // with e.g. a `list<text>` column would silently misalign every
298    // subsequent row while still reporting rows_decoded == true (roborev
299    // finding, review-first pass). Checked once here (a schema-level
300    // property, not a per-row one) rather than per column per row.
301    if column_names
302        .iter()
303        .any(|name| !schema.column_type(name).is_some_and(is_simple_scalar_type))
304    {
305        return Err(Unsupported);
306    }
307    let mut rows = Vec::new();
308    loop {
309        let flags = c.u8().map_err(|_| Unsupported)?;
310        if flags & ROW_END_OF_PARTITION != 0 {
311            break;
312        }
313        if flags & ROW_IS_MARKER != 0 {
314            // Range tombstone marker — not modeled.
315            return Err(Unsupported);
316        }
317        if flags & ROW_EXTENSION_FLAG != 0 {
318            let _ext = c.u8().map_err(|_| Unsupported)?;
319        }
320
321        // Clustering values: schema.clustering is guaranteed empty at this
322        // point (decode_rows bails to structural-only above whenever it
323        // isn't), so there is nothing to read here. No dead read-loop is kept
324        // around it — re-enabling clustering support means deliberately
325        // writing the ClusteringPrefix presence-header decode (see the bail
326        // site's comment), not just deleting the guard (roborev finding,
327        // review-first pass).
328        let clustering = Vec::new();
329
330        // Liveness.
331        if flags & ROW_HAS_TIMESTAMP != 0 {
332            let _ts_delta = c.uvint().map_err(|_| Unsupported)?;
333        }
334        if flags & ROW_HAS_TTL != 0 {
335            let _ttl = c.uvint().map_err(|_| Unsupported)?;
336            let _ldt = c.uvint().map_err(|_| Unsupported)?;
337        }
338        let mut is_row_deletion = false;
339        if flags & ROW_HAS_DELETION != 0 {
340            // Row deletion time (two uvints in the messaging delta encoding).
341            let _mfda = c.uvint().map_err(|_| Unsupported)?;
342            let _ldt = c.uvint().map_err(|_| Unsupported)?;
343            is_row_deletion = true;
344        }
345        if flags & ROW_HAS_COMPLEX_DELETION != 0 {
346            return Err(Unsupported);
347        }
348
349        // Column subset — not modeled; only the all-columns case is decoded.
350        if flags & ROW_HAS_ALL_COLUMNS == 0 {
351            return Err(Unsupported);
352        }
353
354        // One simple cell per regular column, in header order.
355        let mut cells = Vec::with_capacity(column_names.len());
356        for name in column_names {
357            let type_name = schema.column_type(name).ok_or(Unsupported)?;
358            let cell = read_cell(c, name, type_name)?;
359            cells.push(cell);
360        }
361
362        rows.push(DecodedRow {
363            clustering,
364            cells,
365            is_row_deletion,
366        });
367    }
368    Ok(rows)
369}

371fn read_cell(c: &mut Cursor<'_>, column: &str, type_name: &str) -> RowResult<DecodedCell> {
372    let flags = c.u8().map_err(|_| Unsupported)?;
373    let deleted = flags & CELL_IS_DELETED != 0;
374    let expiring = flags & CELL_IS_EXPIRING != 0;
375    let empty = flags & CELL_HAS_EMPTY_VALUE != 0;
376
377    if flags & CELL_USE_ROW_TIMESTAMP == 0 {
378        let _ts_delta = c.uvint().map_err(|_| Unsupported)?;
379    }
380    let use_row_ttl = flags & CELL_USE_ROW_TTL != 0;
381    if (deleted || expiring) && !use_row_ttl {
382        let _ldt = c.uvint().map_err(|_| Unsupported)?;
383    }
384    if expiring && !use_row_ttl {
385        let _ttl = c.uvint().map_err(|_| Unsupported)?;
386    }
387
388    // A value is present unless the cell is a tombstone or explicitly empty.
389    let value = if deleted || empty {
390        None
391    } else {
392        Some(read_typed_value(c, type_name)?)
393    };
394
395    Ok(DecodedCell {
396        column: column.to_string(),
397        value,
398        deleted,
399    })
400}

402/// Read one CQL value: fixed-length types have no length prefix; variable
403/// types are unsigned-vint-length-prefixed (`AbstractType.writeValue`).
404fn read_typed_value(c: &mut Cursor<'_>, type_name: &str) -> RowResult<Vec<u8>> {
405    match cql_fixed_len(type_name) {
406        Some(n) => c.take(n).map(|s| s.to_vec()).map_err(|_| Unsupported),
407        None => {
408            let len = c.uvint().map_err(|_| Unsupported)? as usize;
409            c.take(len).map(|s| s.to_vec()).map_err(|_| Unsupported)
410        }
411    }
412}

414/// Read a `Columns` block: `uvint count`, then `count` × vint-length-prefixed
415/// column name bytes.
416fn read_column_names(c: &mut Cursor<'_>) -> Result<Vec<String>> {
417    let count = c.uvint()?;
418    // Guard the declared count against the cursor's remaining bytes before
419    // spinning the loop below `count` times on an untrusted u64: each column
420    // name needs at least its own 1-byte length vint, so a count exceeding the
421    // remaining bytes is provably impossible. Without this, a CRC-valid body of
422    // mostly 0x00 bytes (each decoding as a zero-length name) could turn a
423    // ~128MB body into ~128M Vec<String> entries (~3GB), defeating the <128MB
424    // memory target — the same class as decode_mutation's num_updates guard.
425    let remaining = c.remaining() as u64;
426    if count > remaining {
427        return Err(Error::CorruptCommitLogFrame(format!(
428            "column-names block declares {count} names, impossible for {remaining} remaining bytes"
429        )));
430    }
431    let mut names = Vec::with_capacity(count.min(4096) as usize);
432    for _ in 0..count {
433        let len = c.uvint()? as usize;
434        let bytes = c.take(len)?;
435        names.push(String::from_utf8_lossy(bytes).into_owned());
436    }
437    Ok(names)
438}

440/// A minimal forward-only byte cursor with unsigned-vint support.
441struct Cursor<'a> {
442    bytes: &'a [u8],
443    pos: usize,
444}
445
446impl<'a> Cursor<'a> {
447    fn new(bytes: &'a [u8]) -> Self {
448        Self { bytes, pos: 0 }
449    }
450
451    /// Bytes not yet consumed — an upper bound for any count-prefixed block, so
452    /// a hostile declared count can be rejected before it drives a loop.
453    fn remaining(&self) -> usize {
454        self.bytes.len().saturating_sub(self.pos)
455    }
456
457    fn u8(&mut self) -> Result<u8> {
458        let b = *self.bytes.get(self.pos).ok_or_else(short)?;
459        self.pos += 1;
460        Ok(b)
461    }
462
463    fn take(&mut self, n: usize) -> Result<&'a [u8]> {
464        let end = self.pos.checked_add(n).ok_or_else(short)?;
465        if end > self.bytes.len() {
466            return Err(short());
467        }
468        let out = &self.bytes[self.pos..end];
469        self.pos = end;
470        Ok(out)
471    }
472
473    fn take_array16(&mut self) -> Result<[u8; 16]> {
474        let s = self.take(16)?;
475        let mut a = [0u8; 16];
476        a.copy_from_slice(s);
477        Ok(a)
478    }
479
480    /// Decode a Cassandra unsigned VInt (`writeUnsignedVInt`).
481    ///
482    /// Delegates to [`crate::parser::vint::decode_unsigned`] — the ONE canonical
483    /// read-side VInt decoder (Issue #1638, Epic J/J4), already used by the
484    /// SSTable reader and already fuzzed (`fuzz/fuzz_targets/fuzz_vint.rs`) —
485    /// rather than maintaining a second, independent bit-assembly here. (An
486    /// earlier local reimplementation had a 9-byte shift-overflow bug that the
487    /// canonical decoder already special-cases correctly.)
488    fn uvint(&mut self) -> Result<u64> {
489        let (value, consumed) =
490            crate::parser::vint::decode_unsigned(&self.bytes[self.pos..]).map_err(|_| short())?;
491        self.pos += consumed;
492        Ok(value)
493    }
494}
495
496fn short() -> Error {
497    Error::CorruptCommitLogFrame("mutation body ended mid-field".to_string())
498}

563        // without reading Cassandra's ClusteringPrefix presence-header first
564        // silently misaligns every field after it. Rather than decode wrong
565        // values, the schema-declares-clustering case must bail to the
566        // honest structural-only path, exactly like partition deletions and
567        // static rows already do.
568        let body = hex(ALICE_BODY);
569        let id = parse_table_id("d6de7150-8448-11f1-bf5a-bf4cbc47cb4a").unwrap();
570        let mut schema = users_schema();
571        schema.clustering = vec![ColumnSpec::new("ck", "int")];
572        let mut schemas: SchemaSet = HashMap::new();
573        schemas.insert(id, schema);
574        let m = decode_mutation(&body, &schemas).expect("decode");
575        let u = &m.updates[0];
576        assert!(
577            !u.rows_decoded,
578            "clustered schema must bail to structural-only, never emit misaligned rows"

582        // — only the row body is honestly withheld.

629    fn tinyint_cell_stays_aligned_into_following_int_cell() {
630        // Regression (PR #2797 blocker): tinyint is vint-length-prefixed, not
631        // fixed-1. Before the fix, cql_fixed_len("tinyint") == Some(1) read the
632        // uvint LENGTH byte (0x01) as the value and left the cursor one byte
633        // behind, corrupting the following int cell. Hand-encode one row with a
634        // tinyint cell (value 5) followed by an int cell (value 42) and prove
635        // both decode with the cursor staying aligned across the tinyint.
636        let schema = CommitLogSchema {
637            keyspace: "commitlog_test".into(),
638            table: "gauges".into(),
639            partition_key: vec![ColumnSpec::new("id", "int")],
640            clustering: vec![],
641            columns: vec![
642                ColumnSpec::new("flag", "tinyint"),
643                ColumnSpec::new("count", "int"),
644            ],
645        };
646        let column_names = vec!["flag".to_string(), "count".to_string()];
647
648        let bytes: Vec<u8> = [
649            // Row flags: all columns present, nothing else.
650            ROW_HAS_ALL_COLUMNS,
651            // Cell 1 ("flag", tinyint): use-row-timestamp, then vint-length(1) + 0x05.
652            CELL_USE_ROW_TIMESTAMP,
653            0x01, // uvint length = 1
654            0x05, // the tinyint byte value 5
655            // Cell 2 ("count", int): use-row-timestamp, then fixed-4 big-endian 42.
656            CELL_USE_ROW_TIMESTAMP,
657            0x00,
658            0x00,
659            0x00,
660            0x2A,
661            // Terminator.
662            ROW_END_OF_PARTITION,
663        ]
664        .to_vec();
665
666        let mut c = Cursor::new(&bytes);
667        let rows = match decode_rows(&mut c, &schema, &column_names) {
668            Ok(rows) => rows,
669            Err(Unsupported) => panic!("row with a variable-length tinyint cell must decode"),
670        };
671        assert_eq!(rows.len(), 1);
672        let cells = &rows[0].cells;
673        let flag = cells.iter().find(|c| c.column == "flag").unwrap();
674        assert_eq!(
675            flag.value.as_deref(),
676            Some([0x05].as_ref()),
677            "tinyint value must be the single byte 5, not the vint length"
678        );
679        let count = cells.iter().find(|c| c.column == "count").unwrap();
680        assert_eq!(
681            count.value.as_deref(),
682            Some([0x00, 0x00, 0x00, 0x2A].as_ref()),
683            "int cell must stay aligned after the variable-length tinyint"
684        );
685    }

708    fn static_row_bail_leaves_column_names_empty() {
709        // Minimal partition-update body: 16-byte table id + pk_len=0 +
710        // iter_flags=ITER_HAS_STATIC_ROW + the three EncodingStats uvints (read
711        // unconditionally BEFORE the static check). The static-row path is not
712        // modeled: it must bail to structural-only BEFORE reading any

719        let (update, consumed_fully) =

722            !consumed_fully,
723            "static-row path must not report full consume"
724        );
725        assert!(!update.rows_decoded);
726        assert!(
727            update.column_names.is_empty(),
728            "static-row bail must happen before any column-names read"
729        );
730    }

770    fn hex(s: &str) -> Vec<u8> {
771        (0..s.len())
772            .step_by(2)
773            .map(|i| u8::from_str_radix(&s[i..i + 2], 16).unwrap())
774            .collect()
