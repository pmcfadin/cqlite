1//! Crash-durability barrier for the flush -> WAL-truncate handoff (issue #1392).
2//!
3//! When a memtable flush finishes, the same data lives in two places: the newly
4//! written SSTable components on disk and the still-intact WAL. Truncating the
5//! WAL discards the WAL copy, so it is only safe *after* the SSTable is truly
6//! durable.
7//!
8//! Fsyncing each component file (Data.db, Filter.db, ...) persists the file
9//! *contents*, but on POSIX filesystems it does **not** persist the parent
10//! directory's entries. After a power loss the freshly created dirents can be
11//! missing even though the file contents were flushed — leaving the data in
12//! neither a reachable SSTable nor the (already truncated) WAL.
13//!
14//! This module closes that gap: on every WAL-truncating flush it fsyncs the
15//! **full ancestor chain** from the SSTable's leaf directory up to (and
16//! including) the configured data root, deepest-first, *after* the component
17//! files are synced and *before* the WAL is truncated. It also makes the
18//! truncate-failure path explicit (leave the WAL intact as a durable replay
19//! marker) rather than silently swallowing the error.
20//!
21//! Syncing the whole leaf→data-root chain unconditionally — rather than only
22//! the directories a *given* flush attempt happened to create — closes a
23//! retry/partial-failure hazard (issue #1392): if an earlier attempt created
24//! `data_dir/ks/table` but crashed before its full directory-sync chain
25//! completed, the retry sees those dirs already present, so a
26//! "sync-only-what-I-created" barrier would sync just the leaf and truncate the
27//! WAL while the keyspace/table dirents were still not durable. Directory fsync
28//! is cheap and idempotent, so re-syncing an already-durable dir is harmless
29//! and removes the hazard entirely.
30//!
31//! The [`DurabilityBarrier`] trait is a test seam: production uses
32//! [`RealDurabilityBarrier`], while tests inject fakes that record the ordering
33//! of filesystem operations or fault a specific step.
34

39/// Outcome of a successful [`finalize_flush_durability`] call.
40///
41/// In every variant the SSTable has already been fully written and published
42/// (its components' contents are fsynced and its directory entries persisted).
43/// The variant tells the caller how to reconcile flush state (issue #1392):
44///
45/// * [`FlushDurabilityOutcome::Durable`] — the WAL was truncated, or was left
46///   intact but is still a valid replay marker. The flush is clean; the caller
47///   commits state (clears the memtable, advances the generation) and returns
48///   success.
49/// * [`FlushDurabilityOutcome::WalTruncateFailedAfterCommit`] — the WAL truncate
50///   failed *after* `set_len(0)` had already zeroed the WAL. The WAL is no
51///   longer replayable, so the published SSTable is the **only** durable copy
52///   of the flushed mutations. The caller MUST still commit flush state (clear
53///   the memtable and advance the generation) so a retry writes a NEW
54///   generation instead of overwriting the published SSTable, and then surface
55///   the carried error. Returning the error *without* committing state would
56///   let a retry reuse this generation and `File::create` would overwrite the
57///   sole durable copy — a crash mid-retry would then lose the data entirely.
58#[derive(Debug)]
59pub(crate) enum FlushDurabilityOutcome {
60    /// The handoff completed cleanly; the WAL is truncated or still replayable.
61    Durable,
62    /// The SSTable is durably published but the WAL truncate failed after
63    /// zeroing the WAL. State MUST be committed before the error is surfaced.
64    WalTruncateFailedAfterCommit(Error),
65}

67/// Seam over the two durability-critical operations performed at the end of a
68/// flush. Splitting them behind a trait lets tests assert their ordering and
69/// inject faults without touching the real filesystem.
70pub(crate) trait DurabilityBarrier {
71    /// Fsync `dir` so that directory entries created during the flush (the new
72    /// SSTable's component files) are persisted.
73    fn sync_dir(&self, dir: &Path) -> Result<()>;
74
75    /// Truncate the WAL now that the SSTable is durable.
76    ///
77    /// Returns [`TruncateError`] on failure so the caller can distinguish a
78    /// failure that leaves the WAL intact ([`TruncateError::BeforeMutation`])
79    /// from one that has already zeroed it
80    /// ([`TruncateError::AfterMutation`], issue #1392).
81    fn truncate_wal(&self, wal: &mut WriteAheadLog) -> std::result::Result<(), TruncateError>;
82}

84/// Production barrier: performs real directory fsyncs and WAL truncation.
85#[derive(Debug, Default, Clone, Copy)]
86pub(crate) struct RealDurabilityBarrier;
87
88impl DurabilityBarrier for RealDurabilityBarrier {
89    fn sync_dir(&self, dir: &Path) -> Result<()> {
90        sync_directory(dir)
91    }
92
93    fn truncate_wal(&self, wal: &mut WriteAheadLog) -> std::result::Result<(), TruncateError> {
94        wal.truncate_checked()
95    }
96}

98/// Make a directory's entries durable.
99///
100/// On Unix this opens the directory and calls `fsync` on the resulting
101/// descriptor, which is the portable POSIX way to persist newly created
102/// dirents. On non-Unix targets directory fsync is not available the same way;
103/// the per-file fsyncs performed by the component writers remain the durability
104/// guarantee there, so this is a documented no-op.
105///
106/// Shared crate-wide (issue #1959): the SSTable writer finalize and the
107/// compaction publication rename both fsync their destination directory (and,
108/// on the compaction path, the full leaf→data-root ancestor chain via
109/// [`dirs_to_sync`]) through this helper so the "TOC-existence ⇒ all components
110/// durable" crash-safety invariant holds on the compaction path too, not just
111/// the flush path.
112///
113/// **Unix-only invariant.** As with the flush durability barrier (issue #1392),
114/// the "TOC-existence ⇒ every named component is durable" invariant is upheld on
115/// **Unix only**. On non-Unix targets this function is a documented no-op (see
116/// the non-Unix impl below), so the crash-safety guarantee there degrades to the
117/// per-file `fsync`s the component writers perform — the parent-directory
118/// dirents are not separately persisted. This is a known, pre-existing platform
119/// limitation, not a regression introduced by the compaction path.
120#[cfg(unix)]
121pub(crate) fn sync_directory(dir: &Path) -> Result<()> {
122    let handle = std::fs::File::open(dir).map_err(|e| {
123        Error::Storage(format!(
124            "Failed to open data directory {} for fsync: {e}",
125            dir.display()
126        ))
127    })?;
128    handle.sync_all().map_err(|e| {
129        Error::Storage(format!(
130            "Failed to fsync data directory {}: {e}",
131            dir.display()
132        ))
133    })
134}
135
136#[cfg(not(unix))]
137pub(crate) fn sync_directory(_dir: &Path) -> Result<()> {
138    Ok(())
139}

141/// Like [`std::fs::create_dir_all`], but maps failures to [`Error::Storage`]
142/// and treats a raced/pre-existing directory as success (issue #1392).
143///
144/// Directory *creation* no longer needs to record which ancestors it
145/// materialized: the flush durability barrier fsyncs the entire leaf→data-root
146/// chain unconditionally (see [`dirs_to_sync`]), so there is nothing to track.
147pub(crate) fn create_dir_all(path: &Path) -> Result<()> {
148    if path.as_os_str().is_empty() {
149        return Ok(());
150    }
151    match std::fs::create_dir(path) {
152        Ok(()) => Ok(()),
153        // Raced/pre-existing: another actor already made it, or it existed.
154        Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists && path.is_dir() => Ok(()),
155        // Parent missing: create ancestors first, then retry.
156        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
157            match path.parent() {
158                Some(parent) if !parent.as_os_str().is_empty() => {
159                    create_dir_all(parent)?;
160                }
161                _ => {
162                    return Err(Error::Storage(format!(
163                        "Failed to create directory {}: {e}",
164                        path.display()
165                    )));
166                }
167            }
168            match std::fs::create_dir(path) {
169                Ok(()) => Ok(()),
170                Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists && path.is_dir() => Ok(()),
171                Err(e) => Err(Error::Storage(format!(
172                    "Failed to create directory {}: {e}",
173                    path.display()
174                ))),
175            }
176        }
177        Err(e) => Err(Error::Storage(format!(
178            "Failed to create directory {}: {e}",
179            path.display()
180        ))),
181    }
182}

184/// Compute the ordered list of directories to fsync so that every dirent along
185/// the newly written SSTable's path is persisted before the WAL is truncated
186/// (issue #1392).
187///
188/// Walks from the leaf SSTable directory up to (and including) `data_root`,
189/// deepest-first. This is the whole ancestor chain that could hold a
190/// not-yet-durable dirent for the SSTable — the leaf (its component dirents),
191/// the table directory (the leaf's own entry), the keyspace directory (the
192/// table's entry), and the data root (the keyspace's entry).
193///
194/// Syncing the FULL chain on every flush — regardless of which dirs this
195/// particular attempt created — is deliberate (issue #1392): a retry after a
196/// partially completed earlier attempt sees the ancestors already present, so a
197/// "sync-only-what-I-created" barrier would skip them and truncate the WAL
198/// while they were still not durable. Directory fsync is idempotent, so
199/// re-syncing an already-durable dir is harmless.
200///
201/// The result is ordered deepest-first so each child directory is synced before
202/// its parent — matching the POSIX convention that a parent's fsync only needs
203/// to follow the creation (and here the sync) of the child it references. The
204/// walk never ascends above `data_root`: it stops as soon as it reaches it, and
205/// defensively stops if `leaf` is not under `data_root` at all.
206///
207/// Reused by the background compaction publish path (issue #1959): it fsyncs
208/// this same leaf→data-root chain after its renames, giving the compaction
209/// finalize full symmetry with the flush barrier (so a defensively-created
210/// `sstable_dir` whose parent dirent was never fsynced cannot lose the whole
211/// published directory on a crash).
212pub(crate) fn dirs_to_sync(leaf: &Path, data_root: &Path) -> Vec<PathBuf> {
213    let mut dirs: Vec<PathBuf> = Vec::new();
214    let mut cur = leaf;
215    loop {
216        dirs.push(cur.to_path_buf());
217        if cur == data_root {
218            break;
219        }
220        match cur.parent() {
221            // Only ascend while we stay within the configured data root; never
222            // fsync directories above it.
223            Some(parent) if !parent.as_os_str().is_empty() && parent.starts_with(data_root) => {
224                cur = parent;
225            }
226            _ => break,
227        }
228    }
229    dirs
230}

232/// Complete the durability handoff at the end of a flush.
233///
234/// `data_path` is the SSTable's Data.db path; its parent directory is the leaf
235/// SSTable directory (falling back to `data_dir` if `data_path` has no parent).
236/// `data_dir` is the write engine's configured data root. Ordering is
237/// load-bearing (issue #1392):
238///
239/// 1. fsync the **full ancestor chain** from the leaf SSTable directory up to
240///    (and including) `data_dir`, deepest-first, so every directory entry along
241///    the new SSTable's path is durable.
242/// 2. Only then truncate the WAL — the WAL copy of the data is discarded here.
243///
244/// The full chain is synced on EVERY flush, not just the directories this
245/// attempt created (see [`dirs_to_sync`]). Fsyncing only the leaf — or only the
246/// dirs a given attempt created — would persist the component dirents but could
247/// leave the leaf's (or keyspace's) own entry in its parent unsynced, so a
248/// crash after the WAL truncate could lose the entire new SSTable directory.
249/// This is especially dangerous on a RETRY after a partially completed earlier
250/// attempt: the ancestors already exist, so a "created this attempt" list would
251/// be empty and only the leaf would be synced. Directory fsync is cheap and
252/// idempotent, so re-syncing already-durable ancestors is harmless.
253///
254/// If any directory fsync fails, the error is propagated and the WAL is
255/// **not** truncated, so recovery can still replay from the WAL.
256///
257/// When the WAL is already empty (`wal.size() == 0`) — e.g. a flush under
258/// `Durability::Disabled` or any flush whose mutations were never WAL'd — the
259/// truncate phase is skipped entirely and [`FlushDurabilityOutcome::Durable`]
260/// is returned (issue #1392). There is no recovery marker to preserve or
261/// destroy, so entering the truncate phase would only risk a spurious
262/// after-commit error on an already-durable flush. The directory-sync barrier
263/// above still runs unconditionally.
264///
265/// The WAL truncate is phase-aware (issue #1392):
266///
267/// * [`TruncateError::BeforeMutation`] — the truncate failed without touching
268///   the WAL contents, so the WAL is still intact. The error is surfaced as a
269///   warning and the WAL is deliberately left in place: it remains a durable
270///   replay marker, and replay is idempotent (identical mutations reconcile by
271///   last-write-wins timestamp), so this degrades to a redundant re-flush
272///   rather than data loss.
273/// * [`TruncateError::AfterMutation`] — the truncate failed *after* `set_len(0)`
274///   already zeroed the WAL. The WAL is **no longer** a replayable marker, so
275///   this returns [`FlushDurabilityOutcome::WalTruncateFailedAfterCommit`]
276///   carrying the error. The published SSTable is now the ONLY durable copy of
277///   the flushed data, so the caller must treat the flush as COMMITTED for
278///   state purposes (clear the memtable and advance the generation) BEFORE
279///   surfacing the error — otherwise a retry would reuse this generation and
280///   `File::create` would overwrite the sole durable copy, losing data on a
281///   crash mid-retry (issue #1392).
282///
283/// Every emitted SSTable component's *contents* must already be fsynced by the
284/// component writers (see [`crate::storage::sstable::writer::SSTableWriter::finish`])
285/// before this function runs. Ordering is load-bearing: component-content
286/// fsyncs (inside `writer.finish()`) happen first, then this function fsyncs
287/// the directory, then it truncates the WAL.
288pub(crate) fn finalize_flush_durability(
289    barrier: &dyn DurabilityBarrier,
290    data_path: &Path,
291    data_dir: &Path,
292    wal: &mut WriteAheadLog,
293) -> Result<FlushDurabilityOutcome> {
294    // Step 1: persist the new SSTable's dirents BEFORE dropping the WAL copy.
295    // (The component *contents* were already fsynced inside `writer.finish()`.)
296    // This fsyncs the FULL ancestor chain from the leaf SSTable directory up to
297    // (and including) the data root, deepest-first, so every directory entry
298    // along the SSTable's path is durable — not just the leaf, and regardless
299    // of whether an earlier flush attempt already created some ancestors
300    // (issue #1392).
301    let sstable_dir = data_path.parent().unwrap_or(data_dir);
302    for dir in dirs_to_sync(sstable_dir, data_dir) {
303        barrier.sync_dir(&dir)?;
304    }
305
306    // Step 2: it is now durable-safe to truncate the WAL.
307    //
308    // Skip the truncate phase entirely when the WAL is already empty (issue
309    // #1392): a flush under `Durability::Disabled`, or any flush whose
310    // mutations were never WAL'd, has nothing to truncate — there is no
311    // recovery marker to preserve or destroy. Entering the truncate phase
312    // anyway would let a post-`set_len(0)` seek/sync fault surface as a
313    // spurious `WalTruncateFailedAfterCommit` on a flush that is already fully
314    // durable in the published SSTable. The directory-sync barrier above still
315    // runs unconditionally, so the durability guarantee is unchanged.
316    if wal.size() == 0 {
317        return Ok(FlushDurabilityOutcome::Durable);
318    }
319
320    match barrier.truncate_wal(wal) {
321        Ok(()) => Ok(FlushDurabilityOutcome::Durable),
322        Err(TruncateError::BeforeMutation(e)) => {
323            // The WAL was never mutated; it is still a valid replay marker.
324            tracing::warn!(
325                "WAL truncate failed before mutating the WAL ({e}). Leaving the \
326                 WAL intact as a durable replay marker; the next startup will \
327                 replay it idempotently (last-write-wins), so no data is lost."
328            );
329            Ok(FlushDurabilityOutcome::Durable)
330        }
331        Err(TruncateError::AfterMutation(e)) => {
332            // The WAL was already zeroed by set_len(0) before the failure, so it
333            // is NO LONGER a replayable marker. The published SSTable is now the
334            // ONLY durable copy of these mutations. Report the flush as COMMITTED
335            // (carrying the error) so the caller advances the generation and
336            // clears the memtable BEFORE surfacing the failure — a retry must
337            // write a NEW generation rather than overwrite the published SSTable
338            // (which `File::create` would do at the reused generation), which
339            // would lose the data on a crash mid-retry (issue #1392).
340            Ok(FlushDurabilityOutcome::WalTruncateFailedAfterCommit(
341                Error::Storage(format!(
342                    "WAL truncate failed AFTER zeroing the WAL ({e}); the WAL is \
343                     no longer a replayable recovery marker. The flushed SSTable \
344                     is durable and the generation has been advanced so a retry \
345                     will not overwrite it."
346                )),
347            ))
348        }
349    }
350}

360    /// A single durability-relevant filesystem operation, recorded by the test
361    /// seam to assert ordering (the directory fsync MUST precede the WAL
362    /// truncate).
363    #[derive(Debug, Clone, PartialEq, Eq)]
364    enum FsOp {
365        /// fsync of a directory whose dirents changed during the flush (the leaf
366        /// SSTable directory and/or a newly created ancestor). Carries the path
367        /// so tests can assert *which* directories were synced and in what order.
368        SyncDir(PathBuf),
369        /// Truncation of the WAL to zero length.
370        WalTruncate,
371    }

373    fn test_mutation(id: i32, name: &str) -> Mutation {
374        let table_id = TableId::new("test_ks", "test_table");
375        let pk = PartitionKey::single("id", Value::Integer(id));
376        let ops = vec![CellOperation::Write {
377            column: "name".to_string(),
378            value: Value::Text(name.to_string()),
379        }];
380        Mutation::new(table_id, pk, None, ops, 1234567890, None)
381    }
382
383    /// Records the order of operations; `sync_dir` optionally faults.
384    struct RecordingBarrier {
385        ops: RefCell<Vec<FsOp>>,
386        fail_dir_sync: bool,
387        /// Fault the truncate BEFORE mutating the WAL (WAL stays intact).
388        fail_truncate_before_mutation: bool,
389        /// Fault the truncate AFTER `set_len(0)` has already zeroed the WAL.
390        fail_truncate_after_mutation: bool,
391    }
392
393    impl RecordingBarrier {
394        fn new() -> Self {
395            Self {
396                ops: RefCell::new(Vec::new()),
397                fail_dir_sync: false,
398                fail_truncate_before_mutation: false,
399                fail_truncate_after_mutation: false,
400            }
401        }
402    }
403
404    impl DurabilityBarrier for RecordingBarrier {
405        fn sync_dir(&self, dir: &Path) -> Result<()> {
406            if self.fail_dir_sync {
407                return Err(Error::Storage("injected dir fsync fault".to_string()));
408            }
409            self.ops.borrow_mut().push(FsOp::SyncDir(dir.to_path_buf()));
410            Ok(())
411        }
412
413        fn truncate_wal(&self, wal: &mut WriteAheadLog) -> std::result::Result<(), TruncateError> {
414            if self.fail_truncate_before_mutation {
415                // WAL contents untouched.
416                return Err(TruncateError::BeforeMutation(Error::Storage(
417                    "injected pre-mutation truncate fault".to_string(),
418                )));
419            }
420            if self.fail_truncate_after_mutation {
421                // Model a failure that occurs AFTER set_len(0): actually zero the
422                // WAL (mutating it), then report an after-mutation fault. The WAL
423                // is now empty and is no longer a replayable marker.
424                wal.truncate().map_err(TruncateError::AfterMutation)?;
425                return Err(TruncateError::AfterMutation(Error::Storage(
426                    "injected post-set_len truncate fault".to_string(),
427                )));
428            }
429            self.ops.borrow_mut().push(FsOp::WalTruncate);
430            wal.truncate().map_err(TruncateError::BeforeMutation)
431        }
432    }

436    fn dir_fsync_precedes_wal_truncate() {
437        let dir = TempDir::new().unwrap();
438        let mut wal = WriteAheadLog::create(dir.path()).unwrap();
439        wal.append(&test_mutation(1, "Alice")).unwrap();
440        wal.sync().unwrap();
441

443        let outcome = finalize_flush_durability(

447            &mut wal,

449        .unwrap();

536    // Issue #1392: a WAL truncate that fails AFTER `set_len(0)` has already
537    // zeroed the WAL must NOT be swallowed as "WAL intact". The WAL is no longer
538    // a replayable marker, so finalize returns
539    // `WalTruncateFailedAfterCommit(err)` — the SSTable is committed (so the
540    // caller advances the generation and clears the memtable) but the error is
541    // carried so it is still surfaced to the caller.
542    #[test]
543    fn faulted_truncate_after_mutation_reports_committed_with_error() {
544        let dir = TempDir::new().unwrap();
545        let mut wal = WriteAheadLog::create(dir.path()).unwrap();
546        wal.append(&test_mutation(11, "Dave")).unwrap();
547        wal.sync().unwrap();

550        barrier.fail_truncate_after_mutation = true;
551
552        let outcome = finalize_flush_durability(
553            &barrier,
554            &dir.path().join("nb-1-big-Data.db"),
555            dir.path(),
556            &mut wal,
557        )
558        .expect("a post-mutation truncate failure is COMMITTED, not a hard error");

560            FlushDurabilityOutcome::WalTruncateFailedAfterCommit(e) => e,

586    // Issue #1392 (roborev LOW, FINDING 1): a flush whose WAL is already empty
587    // (e.g. `Durability::Disabled`, or any flush whose mutations were never
588    // WAL'd) must SKIP the WAL truncate phase entirely. The directory-sync
589    // barrier still runs, but `truncate_wal` is never invoked — so a
590    // truncate-phase fault cannot turn an already-durable flush into a spurious
591    // `WalTruncateFailedAfterCommit` error.
592    #[test]
593    fn empty_wal_skips_truncate_phase() {
594        let dir = TempDir::new().unwrap();
595        // A freshly created WAL with nothing appended has size 0.
596        let mut wal = WriteAheadLog::create(dir.path()).unwrap();
597        assert_eq!(wal.size(), 0, "a fresh, unwritten WAL must have size 0");
598
599        // Arm the truncate to fault AFTER mutating the WAL: if the truncate
600        // phase were entered at all, this would surface as
601        // `WalTruncateFailedAfterCommit`. Skipping the phase must avoid it.
602        let mut barrier = RecordingBarrier::new();
603        barrier.fail_truncate_after_mutation = true;
604
605        let outcome = finalize_flush_durability(
606            &barrier,
607            &dir.path().join("nb-1-big-Data.db"),
608            dir.path(),
609            &mut wal,
610        )
611        .expect("an empty-WAL flush must not enter the truncate phase");
612        assert!(
613            matches!(outcome, FlushDurabilityOutcome::Durable),
614            "empty WAL: the truncate phase is skipped, so the flush stays Durable"
615        );
616
617        // The directory was still synced, but the WAL truncate was never
618        // attempted (no WalTruncate op recorded and the armed fault never fired).
619        assert_eq!(
620            barrier.ops.into_inner(),
621            vec![FsOp::SyncDir(dir.path().to_path_buf())],
622            "dir fsync still runs; the truncate phase is skipped for an empty WAL"
623        );
624    }

645    // keyspace/table path, the barrier must fsync the leaf AND every ancestor up
646    // to the data root — deepest first — BEFORE the WAL truncate, so a crash
647    // cannot lose the freshly created directory tree.

653        let mut wal = WriteAheadLog::create(data_dir).unwrap();

656
657        // Leaf SSTable dir: data_dir/test_ks/test_table (both ancestors new).

659        let leaf_dir = ks_dir.join("test_table");

661
662        let barrier = RecordingBarrier::new();
663        finalize_flush_durability(&barrier, &data_path, data_dir, &mut wal).unwrap();
