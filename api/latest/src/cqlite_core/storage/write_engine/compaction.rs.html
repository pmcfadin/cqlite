1//! SSTable compaction execution: K-way merge start + atomic finalize.
2//!
3//! Extracted verbatim from `write_engine/maintenance.rs` (issue #1120, epic
4//! #1116) as a behavior-preserving split. Owns the merge state-machine entry
5//! points driven by `maintenance_step`: `start_merge` (opens the K-way merger +
6//! tmp output writer) and the blocking/async finalizers that atomically publish
7//! the merged SSTable and reclaim the inputs. `WriteEngine`'s fields are
8//! reachable here because this is a sibling module in the same crate.
9

11use super::{merge, KWayMerger, WriteEngine};
12use crate::error::{Error, Result};
13use std::path::PathBuf;
14use std::time::Instant;

16// Test-only crash-injection seam (issue #1393).
17//
18// When set on the thread that drives `finalize_merge_async`, the finalizer
19// aborts *after* the tmp component files have been written but *before* the
20// atomic publication rename — and deliberately leaves the `.compaction-tmp-*`
21// directory on disk, exactly as a real process crash mid-compaction would. The
22// startup orphan sweep is then exercised end-to-end against genuine partial
23// output. It is a `thread_local` (not a global) so parallel tests cannot see
24// each other's injection: the sweep e2e test runs `maintenance_step`
25// synchronously with no ambient runtime, so `block_on_async` polls the
26// finalizer on the very thread that set the flag.
27#[cfg(test)]
28thread_local! {
29    pub(crate) static FAIL_COMPACTION_BEFORE_RENAME: std::cell::Cell<bool> =
30        const { std::cell::Cell::new(false) };
31}

33impl WriteEngine {
34    /// Start a new merge operation (M5.2 helper, Issue #474)
35    ///
36    /// ## Atomicity design
37    ///
38    /// The SSTableWriter is pointed at a temporary directory (`tmp_dir`) that lives
39    /// alongside the final SSTable directory. After `writer.finish()` succeeds, each
40    /// output component is atomically renamed into the final directory. Input files are
41    /// deleted only after all renames complete. This guarantees:
42    ///
43    /// - A crash before renames: tmp files are incomplete; inputs intact.
44    /// - A crash mid-rename: at worst a partial output component exists in the final
45    ///   directory, but the old inputs are still there and the TOC.txt (publication
46    ///   barrier) has not been renamed yet, so the partial output is never discovered
47    ///   by readers scanning for `TOC.txt`.
48    /// - A crash after all renames but before input deletion: a harmless duplicate
49    ///   exists until next compaction.
50    ///
51    /// `purge_safe` (#921 finding 1): `true` only when `input_paths` spans every
52    /// SSTable for the table (a major/full compaction), allowing gc_grace
53    /// tombstone purging without risking resurrection of data shadowed in a
54    /// non-included overlapping SSTable. `false` keeps every tombstone.
55    ///
56    /// `max_purgeable_timestamp` (#935): for a PARTIAL compaction (`purge_safe ==
57    /// false`), the min write timestamp across the non-included overlapping
58    /// SSTables — see [`merge::compute_max_purgeable_timestamp`]. When `Some`, a
59    /// tombstone older than every outside SSTable is purged even in a partial
60    /// compaction; `None` keeps the conservative #921 behavior (no purging).
61    #[tracing::instrument(name = "compaction.start_merge", skip(self, input_paths, max_purgeable_timestamp, outside_paths), fields(inputs = input_paths.len()))]
62    pub(crate) fn start_merge(
63        &mut self,
64        input_paths: Vec<PathBuf>,
65        purge_safe: bool,
66        max_purgeable_timestamp: Option<i64>,
67        outside_paths: Vec<PathBuf>,
68    ) -> Result<()> {
69        log::info!(
70            "Starting compaction merge of {} SSTables",
71            input_paths.len()
72        );
73
74        // gc_grace / gcBefore cutoff (issue #845): the fully-expired drop-set
75        // (issue #1388) needs the same cutoff the merger's purge stage uses, so
76        // compute it up front. `now_secs` is reused below for the merger.
77        let now_secs = std::time::SystemTime::now()
78            .duration_since(std::time::UNIX_EPOCH)
79            .map(|d| d.as_secs() as i64)
80            .unwrap_or(0);
81        let gc_before_secs = merge::compute_gc_before(&self.config.schema, now_secs);
82
83        // Fully-expired SSTable drop (issue #1388): compute the subset of the
84        // selected inputs that are provably fully expired (authoritative
85        // `Statistics.db` `max_deletion_time < gcBefore`, no cell scan) AND
86        // overlap-safe against the outside set (`max_timestamp` below the outside
87        // min write timestamp — the same #935 coarse global bound). For a full
88        // compaction `outside_paths` is empty ⇒ +inf bound ⇒ every fully-expired
89        // input is droppable. These SSTables are EXCLUDED from the merger's input
90        // list below (never read/decoded — the perf win) and reclaimed after the
91        // merged output publishes.
92        let drop_set = merge::fully_expired_sstables(&input_paths, &outside_paths, gc_before_secs);
93        // Subtract the drop-set from the merger inputs (with the all-dropped guard);
94        // shared with the CLI one-shot path.
95        let (input_paths, dropped_whole) = merge::split_merge_and_dropped(&input_paths, drop_set);
96
97        // Measure total bytes read (sum of Data.db file sizes as an approximation)
98        let bytes_read: u64 = input_paths
99            .iter()
100            .map(|p| std::fs::metadata(p).map(|m| m.len()).unwrap_or(0))
101            .sum();
102
103        let output_generation = self.generation;
104
105        // Final SSTable directory: data_dir/keyspace/table/
106        let sstable_dir = self
107            .config
108            .data_dir
109            .join(&self.config.schema.keyspace)
110            .join(&self.config.schema.table);
111
112        // Temporary root: data_dir/.compaction-tmp-{gen}/
113        //
114        // Placing the tmp root inside `data_dir` (not inside `sstable_dir`) keeps
115        // the path simple and guarantees the rename is within the same filesystem.
116        // The SSTableWriter appends `keyspace/table/` internally, so component files
117        // land at: data_dir/.compaction-tmp-{gen}/keyspace/table/nb-{gen}-big-*.db
118        let tmp_dir = self
119            .config
120            .data_dir
121            .join(format!(".compaction-tmp-{}", output_generation));
122
123        // Create the tmp root (SSTableWriter::finish will create the subdirs)
124        std::fs::create_dir_all(&tmp_dir).map_err(|e| {
125            Error::Storage(format!(
126                "Failed to create compaction tmp directory {:?}: {}",
127                tmp_dir, e
128            ))
129        })?;
130
131        // #850: derive the effective compaction schema by reading static-row
132        // presence from the input SSTable headers. If a static column was dropped
133        // from the current schema but an input still carries it, re-add it so the
134        // merger decodes the static cells and the writer emits the static prelude
135        // (header-driven static presence). Byte-identical to `config.schema` when
136        // no such column exists.
137        let mut effective_schema =
138            merge::effective_compaction_schema(&self.config.schema, &input_paths);
139        // #929: normalize bare-name UDT columns to their UserType(...) marshal on
140        // the EFFECTIVE (decode) schema by copying each column's exact UserType
141        // marshal from the input headers. The compaction reader decides complex-
142        // vs-simple from this schema's `data_type` (is_complex_column), so a bare
143        // name would make it misdecode/drop pre-existing per-field UDT cells
144        // (roborev #1009/#1013/#1015/#1017/#1019/#1021). Derived ENTIRELY from the
145        // input headers, so it runs unconditionally (a registry is needed only at
146        // flush time); errors on a mixed encoding rather than corrupting.
147        // `write_schema` derives from this via `for_compaction_output`. The
148        // configured registry handles a UDT column added after the inputs were
149        // written (absent from every input header) so the output is not emitted
150        // as a bare/BytesType column (roborev #1023).
151        merge::apply_udt_marshals_from_inputs(
152            &mut effective_schema,
153            &input_paths,
154            self.config.udt_registry.as_ref(),
155        )?;
156
157        // Create K-way merger.
158        //
159        // gc_grace / gcBefore purging (issue #845): `gcBefore = now -
160        // gc_grace_seconds` and `now_secs` were computed above (also used by the
161        // #1388 fully-expired drop-set). `compute_gc_before` returns `None` when the
162        // table declares no valid `gc_grace_seconds`, disabling purging (and, above,
163        // dropping) — preserving the pre-#845 behavior. `now_secs` is the wall-clock
164        // used for both the purge boundary and TTL evaluation.
165        //
166        // Overlap-safety gate (#921 finding 1): only a major/full compaction
167        // (one that spans every candidate SSTable for the table) may purge
168        // tombstones. A partial compaction sets `purge_safe = false`, which
169        // makes the gc_grace purge stage a strict no-op and cannot resurrect
170        // data shadowed in a non-included overlapping SSTable.
171
172        // #921 finding 2: mirror `compact_sstables`' dropped-column survivor
173        // pre-pass. Decode with `effective_schema` (retains dropped columns so
174        // their cells parse and can be purged), but WRITE with a post-drop
175        // schema: a background compaction that purges the LAST cell/tombstone of
176        // a dropped column must NOT emit that column in the output
177        // SerializationHeader (stale header columns misalign a post-drop reader).
178        // The pre-pass uses the SAME gc cutoff AND `purge_safe` as the write
179        // merger below so the two make IDENTICAL purge decisions — a tombstone
180        // purged in the write pass is also purged here and never counted as a
181        // surviving cell. Reuses the exact helper `compact_sstables` uses.
182        let retained_dropped = if effective_schema.dropped_columns.is_empty() {
183            std::collections::HashSet::new()
184        } else {
185            merge::compute_surviving_dropped_columns(
186                input_paths.clone(),
187                &effective_schema,
188                gc_before_secs,
189                Some(now_secs),
190                purge_safe,
191                max_purgeable_timestamp,
192            )?
193        };
194        // `write_schema` inherits the #929 UDT normalization from the already-
195        // normalized `effective_schema` (for_compaction_output clones columns).
196        let write_schema = effective_schema.for_compaction_output(&retained_dropped);
197
198        // Issue #1234: thread the configured UDT registry onto the merge readers so
199        // a top-level `frozen<UDT>` value decodes structurally during background
200        // compaction instead of erroring out and dropping the partition.
201        let merger = KWayMerger::new_with_gc_and_registry(
202            input_paths.clone(),
203            &effective_schema,
204            gc_before_secs,
205            Some(now_secs),
206            self.config.udt_registry.clone(),
207        )?
208        .with_purge_safe(purge_safe)
209        // #935: overlap-aware purging for a partial compaction. `purge_safe`
210        // overrides this in the merger (full compaction → +inf bound), so passing
211        // the partial bound unconditionally is safe.
212        .with_max_purgeable_timestamp(max_purgeable_timestamp);
213
214        // Repair-state preservation + mixed-state rejection (issue #1021).
215        //
216        // Cassandra partitions compaction candidates by repair state and never
217        // mixes repaired / unrepaired / pending-repair SSTables in one
218        // compaction. CQLite cannot reproduce the repair-boundary tombstone
219        // constraints, so it MUST NOT silently merge across that boundary: read
220        // each input's persisted repair state from its Statistics.db, reject a
221        // mixed set with a typed error, and PRESERVE the single shared state into
222        // the merged output below. Authoritative metadata only (no heuristics).
223        let repair_state = merge::classify_inputs(&input_paths)?;
224
225        // Point the SSTableWriter at the tmp root; it will write to
226        // tmp_dir/keyspace/table/nb-{gen}-big-*.db. Use the post-drop
227        // `write_schema` (NOT `effective_schema`) so fully-purged dropped
228        // columns are stripped from the output header (#921 finding 2).
229        let mut writer = crate::storage::sstable::writer::SSTableWriter::new(
230            tmp_dir.clone(),
231            output_generation,
232            &write_schema,
233        )?;
234        // Compaction output (issue #1222): emit the uncompressed-BIG CRC.db with
235        // Cassandra's compaction-only trailing empty-final-chunk CRC32 = 0. The
236        // flush path leaves this unset, keeping its CRC.db byte-identical.
237        writer.mark_compaction_output();
238
239        // Carry the inputs' shared repair state into the merged output's
240        // Statistics.db (issue #1021). For a normal unrepaired corpus this is the
241        // unrepaired default (a no-op vs the previous hardcoded zeros).
242        writer.set_repair_state(
243            repair_state.repaired_at,
244            repair_state.pending_repair,
245            repair_state.is_transient,
246        );
247
248        // Two-pass compaction (issue #729): compute output FINAL encoding baselines
249        // by reading the minimum values from each input SSTable's Statistics.db.
250        // The output baseline must be ≤ all per-partition values from all inputs.
251        let (baseline_min_ts, baseline_min_ldt, baseline_min_ttl) =
252            merge::compute_baseline_min(&input_paths);
253        writer.pre_seed_encoding_baselines(baseline_min_ts, baseline_min_ldt, baseline_min_ttl);
254
255        // Increment generation for next operation
256        self.generation += 1;
257
258        self.active_merge = Some(ActiveMerge {
259            merger,
260            writer,
261            input_paths,
262            tmp_dir,
263            sstable_dir,
264            rows_merged: 0,
265            bytes_read,
266            started_at: Instant::now(),
267            effective_schema,
268            dropped_whole,
269        });
270
271        Ok(())
272    }
273
274    /// Finalize the active merge - blocking version (M5.2 helper).
275    ///
276    /// Bridges to the async finalizer via [`merge::block_on_async`], which is
277    /// safe to call from within an active Tokio runtime (e.g. the CLI's
278    /// `#[tokio::main]` worker threads). A nested `Handle::block_on` here would
279    /// otherwise panic with "Cannot start a runtime from within a runtime"
280    /// (Issue #587).
281    pub(crate) fn finalize_merge_blocking(&mut self, report: &mut MaintenanceReport) -> Result<()> {
282        merge::block_on_async(self.finalize_merge_async(report))
283    }
284
285    /// Finalize the active merge - async version (M5.2 helper, Issue #474)
286    ///
287    /// ## Atomicity protocol
288    ///
289    /// 1. `writer.finish()` flushes all component files to the tmp directory.
290    /// 2. Each component file is renamed from `tmp_dir/` to `sstable_dir/`.
291    ///    The TOC.txt rename is performed **last** (it is the publication barrier:
292    ///    readers discover SSTables by scanning for `TOC.txt`).
293    /// 3. Only after all renames succeed are the input SSTable files deleted.
294    /// 4. The now-empty tmp directory is removed.
295    ///
296    /// If any step 2 rename fails, the partially-renamed output components are
297    /// cleaned up and an error is returned. The input SSTables remain intact.
298    ///
299    /// Single-boundary error recording (issue #1037): this is an *unrecorded*
300    /// inner helper. Any error returned here escapes through
301    /// `maintenance_step_inner` to the public `maintenance_step`, which wraps the
302    /// whole step in `record_result("compaction", ..)` and counts it exactly
303    /// once. Recording here too would double-count finalize failures.
304    #[tracing::instrument(name = "compaction.finalize", skip(self, report))]
305    async fn finalize_merge_async(&mut self, report: &mut MaintenanceReport) -> Result<()> {
306        let merge = match self.active_merge.take() {
307            Some(m) => m,
308            None => return Ok(()),
309        };
310
311        let input_count = merge.input_paths.len() as u64;
312        let merge_rows = merge.rows_merged;
313        let elapsed = merge.started_at.elapsed();
314        log::info!(
315            "Finalizing compaction merge: {} rows, {:?} elapsed",
316            merge.rows_merged,
317            elapsed
318        );
319
320        // Step 1: Finish writing all components to the tmp directory.
321        // If this fails the tmp directory may contain partial files, but inputs are safe.
322        let tmp_info = match merge.writer.finish().await {
323            Ok(info) => info,
324            Err(e) => {
325                // Clean up tmp directory on failure (best effort)
326                let _ = std::fs::remove_dir_all(&merge.tmp_dir);
327                return Err(Error::Storage(format!(
328                    "Compaction merge write failed (inputs intact): {}",
329                    e
330                )));
331            }
332        };
333
334        log::info!(
335            "Compaction tmp output: {} bytes, {} partitions",
336            tmp_info.data_size,
337            tmp_info.partition_count
338        );
339
340        // Test-only crash injection (issue #1393): simulate a process crash after
341        // the tmp component files are on disk but before the publication rename.
342        // We intentionally do NOT clean up `merge.tmp_dir` — a real crash leaves
343        // it for the startup orphan sweep to reclaim. Inputs are still intact
344        // (no rename or delete has run yet).
345        #[cfg(test)]
346        if FAIL_COMPACTION_BEFORE_RENAME.with(|f| f.get()) {
347            return Err(Error::Storage(
348                "injected compaction crash before publication rename (test seam, issue #1393)"
349                    .to_string(),
350            ));
351        }
352
353        // Step 2: Atomically rename each component from the tmp sub-directory to
354        // the final SSTable directory. Because both directories are under the same
355        // `data_dir`, the rename is within the same filesystem (POSIX atomic).
356        // We rename TOC.txt last because it is the publication barrier.
357        let sstable_dir = &merge.sstable_dir;
358
359        // Ensure the final SSTable directory exists (it normally does, but handle
360        // the edge case where it was created by start_merge only in the tmp path).
361        std::fs::create_dir_all(sstable_dir).map_err(|e| {
362            Error::Storage(format!(
363                "Failed to create SSTable directory {:?}: {}",
364                sstable_dir, e
365            ))
366        })?;
367
368        // Helper: map a tmp component path to its final destination.
369        // tmp_info paths look like: data_dir/.compaction-tmp-N/keyspace/table/nb-N-big-X.db
370        // Final destination:        data_dir/keyspace/table/nb-N-big-X.db
371        let make_rename = |src: &PathBuf| -> Result<(PathBuf, PathBuf)> {
372            let filename = src
373                .file_name()
374                .ok_or_else(|| Error::Storage("Component path has no filename".to_string()))?;
375            let dst = sstable_dir.join(filename);
376            Ok((src.clone(), dst))
377        };
378
379        // Build list of (src, dst) renames. TOC.txt goes last (publication barrier).
380        let mut renames: Vec<(PathBuf, PathBuf)> = Vec::new();
381
382        // Non-TOC components first. Index.db/Summary.db are BIG-only (BTI writers
383        // report them as None — issue #908); only rename them when present.
384        // Filter.db is optional too: a table with a disabled bloom filter
385        // (bloom_filter_fp_chance = 1.0, AlwaysPresentFilter) emits NO Filter.db,
386        // so there is nothing to rename. Including a non-existent path here would
387        // fail the atomic publish (Issue #852).
388        let mut non_toc: Vec<&PathBuf> = vec![&tmp_info.data_path];
389        if let Some(ref p) = tmp_info.index_path {
390            non_toc.push(p);
391        }
392        if let Some(ref p) = tmp_info.filter_path {
393            non_toc.push(p);
394        }
395        if let Some(ref p) = tmp_info.summary_path {
396            non_toc.push(p);
397        }
398        non_toc.push(&tmp_info.stats_path);
399        non_toc.push(&tmp_info.digest_path);
400        // CRC.db is emitted for uncompressed BIG output (issue #1197); rename it
401        // when present so the published SSTable keeps its per-chunk CRC file.
402        if let Some(ref p) = tmp_info.crc_path {
403            non_toc.push(p);
404        }
405        for src in non_toc {
406            renames.push(make_rename(src)?);
407        }
408        if let Some(ref p) = tmp_info.partitions_path {
409            renames.push(make_rename(p)?);
410        }
411        if let Some(ref ci_path) = tmp_info.compression_info_path {
412            renames.push(make_rename(ci_path)?);
413        }
414        // TOC.txt is last (publication barrier)
415        renames.push(make_rename(&tmp_info.toc_path)?);
416
417        // Perform the renames. On failure, remove any already-renamed files so
418        // we don't leave a half-published SSTable, then return the error.
419        // Time the rename/publication-barrier phase (issue #1037).
420        let finalize_start = Instant::now();
421        let mut renamed: Vec<PathBuf> = Vec::with_capacity(renames.len());
422        let mut rename_error: Option<Error> = None;
423
424        for (src, dst) in &renames {
425            match std::fs::rename(src, dst) {
426                Ok(()) => {
427                    log::debug!(
428                        "Renamed {:?} → {:?}",
429                        src.file_name().unwrap_or_default(),
430                        dst.file_name().unwrap_or_default()
431                    );
432                    renamed.push(dst.clone());
433                }
434                Err(e) => {
435                    rename_error = Some(Error::Storage(format!(
436                        "Atomic rename of {:?} to {:?} failed (rolling back, inputs intact): {}",
437                        src, dst, e
438                    )));
439                    break;
440                }
441            }
442        }
443
444        if let Some(err) = rename_error {
445            // Roll back already-renamed files (best effort)
446            for dst in &renamed {
447                let _ = std::fs::remove_file(dst);
448            }
449            // Clean up tmp directory
450            let _ = std::fs::remove_dir_all(&merge.tmp_dir);
451            return Err(err);
452        }
453
454        // Finalize/rename latency in seconds (issue #1037): the atomic
455        // publication barrier just completed successfully.
456        crate::observability::record_histogram(
457            crate::observability::catalog::COMPACTION_FINALIZE_DURATION,
458            finalize_start.elapsed().as_secs_f64(),
459            &[],
460        );
461
462        // Step 3: All renames succeeded. The new SSTable is now visible.
463        // Delete input SSTable files. If deletion fails we log a warning but do
464        // NOT return an error — the merge output is correct.
465        //
466        // Issue #591 (write-while-mapped / Windows policy): the inputs were read
467        // through buffered I/O (never memory-mapped) and the merge has fully
468        // drained every source through its streaming producer by this point
469        // (issue #827), so the merger holds no mapping over them — deleting them
470        // cannot fault with SIGBUS. `delete_sstable_files` removes each
471        // input's TOC.txt first (unpublishing it) and is best-effort on the data
472        // components, so a component still pinned by a concurrent mapped reader on
473        // Windows becomes an invisible orphan reclaimed on the next startup rather
474        // than a hard failure or a source of duplicate rows.
475        for input_path in &merge.input_paths {
476            if let Err(e) = self.delete_sstable_files(input_path) {
477                log::warn!(
478                    "Failed to delete compaction input {:?}: {} \
479                     (merge output is valid; inputs will be re-evaluated next cycle)",
480                    input_path,
481                    e
482                );
483            }
484        }
485
486        // Reclaim the DROPPED-WHOLE SSTables (issue #1388): proven fully expired +
487        // overlap-safe, they were EXCLUDED from the merger (never read) so their
488        // reclamation happens here — AFTER the merged output published — via the
489        // same best-effort component-delete path as the merged inputs. A failure is
490        // an invisible orphan (TOC.txt removed first) reclaimed on next startup.
491        //
492        // This surface already deleted EVERY merge input above, so pass
493        // `already_deleted = merge.input_paths`: in the degenerate all-expired case
494        // the SSTable the all-dropped guard retained as a merge input is ALSO in
495        // `dropped_whole`, and the loop above already reclaimed it — the dedup skips
496        // it here to avoid a double-delete + spurious orphan warning (roborev #1388).
497        crate::storage::write_engine::merge::reclaim_dropped_whole(
498            &merge.dropped_whole,
499            &merge.input_paths,
500            |dropped| {
501                if let Err(e) = self.delete_sstable_files(dropped) {
502                    log::warn!(
503                        "Failed to delete dropped-whole compaction input {:?}: {} \
504                         (merge output is valid; leftover is an invisible orphan)",
505                        dropped,
506                        e
507                    );
508                }
509            },
510        );
511
512        // Step 4: Remove the now-empty tmp directory (best effort).
513        if let Err(e) = std::fs::remove_dir_all(&merge.tmp_dir) {
514            log::debug!(
515                "Failed to remove compaction tmp directory {:?}: {}",
516                merge.tmp_dir,
517                e
518            );
519        }
520
521        // The final Data.db path is in sstable_dir (renamed from tmp)
522        let final_data_path = sstable_dir.join(
523            tmp_info
524                .data_path
525                .file_name()
526                .ok_or_else(|| Error::Storage("Data.db path has no filename".to_string()))?,
527        );
528
529        // Compute total bytes written across ALL output SSTable components (not just Data.db).
530        // We stat the final paths (post-rename) so we measure what was actually persisted.
531        let stat_final = |p: &PathBuf| -> u64 {
532            let filename = p.file_name().unwrap_or_default();
533            std::fs::metadata(sstable_dir.join(filename))
534                .map(|m| m.len())
535                .unwrap_or(0)
536        };
537        // Index.db/Summary.db are BIG-only (BTI reports None — issue #908) and
538        // Filter.db is optional (disabled bloom filter omits it, Issue #852), so
539        // only stat the components that were actually written.
540        let mut byte_paths: Vec<&PathBuf> = vec![&tmp_info.data_path];
541        if let Some(ref p) = tmp_info.index_path {
542            byte_paths.push(p);
543        }
544        if let Some(ref p) = tmp_info.filter_path {
545            byte_paths.push(p);
546        }
547        if let Some(ref p) = tmp_info.summary_path {
548            byte_paths.push(p);
549        }
550        byte_paths.push(&tmp_info.stats_path);
551        byte_paths.push(&tmp_info.digest_path);
552        if let Some(ref p) = tmp_info.crc_path {
553            byte_paths.push(p);
554        }
555        if let Some(ref p) = tmp_info.partitions_path {
556            byte_paths.push(p);
557        }
558        let total_bytes_written: u64 = byte_paths.iter().map(|p| stat_final(p)).sum::<u64>()
559            + tmp_info
560                .compression_info_path
561                .as_ref()
562                .map(stat_final)
563                .unwrap_or(0);
564
565        // Update per-step report
566        report.completed_merges.push(final_data_path);
567        report.bytes_written += total_bytes_written;
568        // Record the dropped-whole set (issue #1388, R4), distinct from the merged
569        // inputs, so the drop decision is assertable from the plan/stats.
570        report
571            .dropped_whole
572            .extend(merge.dropped_whole.iter().cloned());
573
574        // Update cumulative lifetime stats
575        self.cumulative_stats.compactions_completed += 1;
576        self.cumulative_stats.sstables_merged_in += merge.input_paths.len() as u64;
577        self.cumulative_stats.sstables_produced += 1;
578        self.cumulative_stats.bytes_read += merge.bytes_read;
579        self.cumulative_stats.bytes_written += total_bytes_written;
580        self.cumulative_stats.rows_merged += merge.rows_merged;
581        self.cumulative_stats.total_time += elapsed;
582
583        // Compaction completion metrics (issue #1037): full-compaction duration
584        // (pairs with COMPACTION_ROWS_MERGED — emitted incrementally per
585        // maintenance step — for rows/sec throughput), bytes written across all
586        // output components, and SSTables in/out. Rows are NOT re-counted here to
587        // avoid double-counting the per-step COMPACTION_ROWS_MERGED increments.
588        let _ = merge_rows;
589        {
590            use crate::observability::{self as obs, catalog};
591            obs::record_histogram(catalog::COMPACTION_DURATION, elapsed.as_secs_f64(), &[]);
592            obs::add_counter(catalog::COMPACTION_BYTES_WRITTEN, total_bytes_written, &[]);
593            obs::add_counter(catalog::COMPACTION_SSTABLES_IN, input_count, &[]);
594            obs::add_counter(catalog::COMPACTION_SSTABLES_OUT, 1, &[]);
595        }
596
597        log::info!(
598            "Compaction complete: merged {} inputs → 1 output ({} bytes total across all components, {} rows, {:?})",
599            merge.input_paths.len(),
600            total_bytes_written,
601            merge.rows_merged,
602            elapsed
603        );
604
605        Ok(())
606    }
