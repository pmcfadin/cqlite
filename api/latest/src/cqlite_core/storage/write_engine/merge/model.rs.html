1//! Merge data model — the entities that flow through the k-way merge stream.
2//!
3//! Extracted from `merge.rs` (issue #945) as a pure, behavior-preserving
4//! relocation: `MergeEntry`, `RowData`, `CellData`, `ComplexDeletion`,
5//! `MergeStep`, and `MergeStats` plus their builders. The parent `merge`
6//! module re-exports every item, so external `write_engine::merge::MergeEntry`
7//! paths keep resolving unchanged. No logic moved here — see `mod.rs` for the
8//! merge machinery and `reconcile.rs` for the reconciliation kernel.

22/// Entry in the merge stream
23///
24/// Represents a single row from one of the input SSTables. This is the
25/// fundamental unit that flows through the merge heap.
26#[cfg(feature = "write-support")]
27#[derive(Debug, PartialEq, Eq)]
28pub struct MergeEntry {
29    /// Which SSTable this came from (0 = newest)
30    pub run_index: usize,
31    /// Partition key with token
32    pub key: DecoratedKey,
33    /// Clustering key (None for tables without clustering)
34    pub clustering_key: Option<ClusteringKey>,
35    /// Timestamp in microseconds since Unix epoch
36    pub timestamp: i64,
37    /// Row data (live cells or tombstone)
38    pub row_data: RowData,
39    /// Complex (collection / UDT) deletion markers for the multi-cell columns
40    /// of this `(pk, ck)` (issue #886 substrate).
41    ///
42    /// Carried through the merge so the per-cell-path collection/UDT followup
43    /// (#844) and shadow-before-purge (#887) can preserve collection/UDT
44    /// deletion timestamps. `reconcile_cluster` unions and preserves these
45    /// across a cluster, but reconciliation does **not yet consult** them and
46    /// the writer does not yet apply them — defaults to empty, so output is
47    /// byte-unchanged. Population (per-element reader emit) lands in #899.
48    pub complex_deletions: Vec<ComplexDeletion>,
49    /// Range-deletion marker covering a span of clustering keys (issue #886
50    /// substrate).
51    ///
52    /// A first-class slot so range tombstones can flow through the merge stream
53    /// instead of being skipped by the parser; applying them to shadow covered
54    /// cells is the follow-up #846. Carried (and timestamp-max-preserved)
55    /// through `reconcile_cluster` but **not yet consulted** by reconciliation
56    /// or the writer, so output is byte-unchanged. `None` when this entry
57    /// carries no range deletion.
58    pub range_deletion: Option<RangeTombstone>,
59    /// Row-level deletion that COEXISTS with this entry's surviving live cells
60    /// (issue #932).
61    ///
62    /// `Some((deletion_time_micros, local_deletion_time_secs))` when a row
63    /// tombstone whose timestamp is OLDER than the surviving cells must be kept
64    /// alongside `RowData::Live`. `RowData::Live` has no row-deletion slot, so —
65    /// mirroring the carry-only `complex_deletions` / `range_deletion` fields —
66    /// the coexisting deletion rides here instead of forcing a `RowData` change
67    /// across ~100 `RowData::Live` sites. `reconcile_cluster` folds it into the
68    /// winning row deletion and re-attaches it to the emitted live entry;
69    /// `merge_entry_to_mutation` emits it as `Mutation::row_tombstone`. Without
70    /// this, a partial compaction that keeps newer cells would DROP the row
71    /// deletion and let older cells of other columns (in SSTables not part of the
72    /// compaction) resurrect. `None` for a row with no coexisting deletion.
73    pub row_deletion: Option<(i64, i32)>,
74    /// Partition-level deletion `(markedForDeleteAt µs, localDeletionTime s)`
75    /// carried by a synthetic partition-tombstone carrier entry (issue #1072).
76    ///
77    /// `Some(..)` ONLY on the carrier entry the reader emits for a partition
78    /// header that has a tombstone (empty `RowData::Live`, `clustering_key:
79    /// None`). The merge extracts the MAX `markedForDeleteAt` across sources,
80    /// applies it as the OUTERMOST per-cell `<=` shadow floor for the whole
81    /// partition, and re-emits the surviving tombstone via
82    /// `merge_entry_to_mutation` (→ `Mutation::partition_tombstone`). Without
83    /// this a newer partition tombstone in one SSTable failed to shadow older
84    /// live rows in another, resurrecting the deleted partition. `None` for every
85    /// non-carrier entry.
86    pub partition_deletion: Option<(i64, i32)>,
87}

113impl MergeEntry {
114    /// Create a new merge entry.
115    ///
116    /// The carry-only #886 substrate fields (`complex_deletions`,
117    /// `range_deletion`) default to empty/`None`; attach them with
118    /// [`with_complex_deletions`](Self::with_complex_deletions) /
119    /// [`with_range_deletion`](Self::with_range_deletion) once the reader emit
120    /// surfaces them (#899).
121    pub fn new(
122        run_index: usize,
123        key: DecoratedKey,
124        clustering_key: Option<ClusteringKey>,
125        timestamp: i64,
126        row_data: RowData,
127    ) -> Self {
128        Self {
129            run_index,
130            key,
131            clustering_key,
132            timestamp,
133            row_data,
134            complex_deletions: Vec::new(),
135            range_deletion: None,
136            row_deletion: None,
137            partition_deletion: None,
138        }
139    }
140
141    /// Attach complex-deletion markers (issue #886 substrate; carry-only).
142    #[must_use]
143    pub fn with_complex_deletions(mut self, complex_deletions: Vec<ComplexDeletion>) -> Self {
144        self.complex_deletions = complex_deletions;
145        self
146    }
147
148    /// Attach a coexisting row-level deletion (issue #932).
149    ///
150    /// `deletion_time` is `markedForDeleteAt` (microseconds); `ldt` is the
151    /// `localDeletionTime` (GC-clock seconds, `0` = not surfaced). Used when a
152    /// row's surviving cells are newer than a row tombstone that must still be
153    /// preserved so it keeps shadowing older cells in non-compacted SSTables.
154    #[must_use]
155    pub fn with_row_deletion(mut self, deletion_time: i64, ldt: i32) -> Self {
156        self.row_deletion = Some((deletion_time, ldt));
157        self
158    }
159
160    /// Attach a range-deletion marker (issue #886 substrate; carry-only).
161    #[must_use]
162    pub fn with_range_deletion(mut self, range_deletion: RangeTombstone) -> Self {
163        self.range_deletion = Some(range_deletion);
164        self
165    }
166
167    /// Attach a partition-level deletion (issue #1072).
168    ///
169    /// `deletion_time` is `markedForDeleteAt` (microseconds); `ldt` is the
170    /// `localDeletionTime` (GC-clock seconds). Marks this entry as the synthetic
171    /// partition-tombstone carrier so the merge applies the partition floor and
172    /// re-emits the surviving tombstone.
173    #[must_use]
174    pub fn with_partition_deletion(mut self, partition_deletion: (i64, i32)) -> Self {
175        self.partition_deletion = Some(partition_deletion);
176        self
177    }
178
179    /// True when this entry has NO writer-emittable content at all — an empty
180    /// `RowData::Live { cells: vec![] }` with no row tombstone and no carried
181    /// deletion metadata. Routing such an entry to the writer would create a
182    /// PHANTOM live empty (pure-PK) row at timestamp 0 (`DataWriter::merge_row_group`
183    /// treats a no-op mutation as a primary-key insert), so `merge` filters it.
184    ///
185    /// Epic #899 Phase C: a COMPLEX DELETION is consumed by the writer (a real
186    /// `CellOperation::ComplexDeletion` marker), so a complex-deletion-only entry
187    /// is NOT a no-op.
188    ///
189    /// Issue #933: a RANGE DELETION is now ALSO consumed by the compaction writer —
190    /// `merge_entry_to_mutation` threads it onto the mutation's `range_tombstones`,
191    /// which the writer interleaves as on-disk bound markers AND uses to shadow
192    /// covered same-partition rows. A range-deletion-only carrier is therefore NO
193    /// LONGER a no-op and MUST reach the writer (dropping it would resurrect the
194    /// covered rows it shadowed — roborev #959 High #2). In practice
195    /// `reconcile_cluster` never emits a truly-empty entry (one with empty cells,
196    /// no row tombstone, and no carried metadata reconciles to `None`), so this
197    /// guard is now effectively unreachable but kept as a defensive phantom-row
198    /// filter.
199    #[must_use]
200    pub fn is_metadata_only_no_op(&self) -> bool {
201        matches!(&self.row_data, RowData::Live { cells } if cells.is_empty())
202            && self.complex_deletions.is_empty()
203            && self.range_deletion.is_none()
204            && self.row_deletion.is_none()
205            && self.partition_deletion.is_none()
206    }
207
208    /// True when this entry is a synthetic partition-tombstone carrier (issue
209    /// #1072): an empty live row whose only payload is `partition_deletion`.
210    /// Produced by [`Self::build_merge_entry`] from a reader `PartitionDelete`.
211    /// Such an entry produces NO output row (it yields a
212    /// `Mutation::partition_tombstone`, not a clustering row), so the merge must
213    /// not count it toward output row counts — but it MUST reach the writer.
214    #[must_use]
215    pub fn is_partition_delete_carrier(&self) -> bool {
216        self.partition_deletion.is_some()
217    }
218}

220/// Ord implementation for min-heap routing ONLY (not LWW winner selection).
221///
222/// This orders entries so the heap yields them grouped by partition and
223/// clustering key. The actual equal-timestamp Delete-vs-Live winner is chosen
224/// in `merge_partition_rows` (timestamp → liveness → run_index), NOT here.
225///
226/// Order by:
227/// 1. Token (ascending)
228/// 2. Key bytes (ascending, for hash collisions)
229/// 3. Clustering key (ascending, schema-aware)
230/// 4. Run index (ascending) - stable routing tiebreak only
231#[cfg(feature = "write-support")]
232impl Ord for MergeEntry {
233    fn cmp(&self, other: &Self) -> Ordering {
234        // Primary: by token
235        match self.key.token.cmp(&other.key.token) {
236            Ordering::Equal => {
237                // Secondary: by key bytes (hash collision resolution)
238                match self.key.key.cmp(&other.key.key) {
239                    Ordering::Equal => {
240                        // Tertiary: by clustering key
241                        match (&self.clustering_key, &other.clustering_key) {
242                            (None, None) => {
243                                // Quaternary: by run_index (lower = newer)
244                                self.run_index.cmp(&other.run_index)
245                            }
246                            (None, Some(_)) => Ordering::Less,
247                            (Some(_), None) => Ordering::Greater,
248                            (Some(a), Some(b)) => {
249                                // Use fallback Ord (not schema-aware at this level)
250                                // Schema-aware comparison happens during partition merge
251                                match a.cmp(b) {
252                                    Ordering::Equal => {
253                                        // Equal clustering keys: prefer lower run_index
254                                        self.run_index.cmp(&other.run_index)
255                                    }
256                                    other_ord => other_ord,
257                                }
258                            }
259                        }
260                    }
261                    other_ord => other_ord,
262                }
263            }
264            other_ord => other_ord,
265        }
266    }
267}

269#[cfg(feature = "write-support")]
270impl PartialOrd for MergeEntry {
271    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
272        Some(self.cmp(other))
273    }
274}

294/// Cell data with timestamp, optional TTL, and (for complex columns) cell path.
295///
296/// ## Per-cell merge metadata (issue #886 — byte-parity foundation)
297///
298/// To reconcile per-cell and per-element data byte-faithfully (Cassandra
299/// `Cells#reconcile`), the merge entry must carry more than a single row-level
300/// timestamp. The fields below thread that richer state from the reader toward
301/// the followup behaviors (#844 per-cell-path collection/UDT merge, #848
302/// tombstone-vs-expiring TTL tie-break). They are **carried but not yet acted
303/// on** by reconciliation — this struct change is plumbing only and must not
304/// alter output bytes.
305///
306/// Where the reader does not yet surface a value the field is left `None`; the
307/// dependent issues fill it in once the reader is extended.
308#[cfg(feature = "write-support")]
309#[derive(Debug, PartialEq, Eq)]
310pub struct CellData {
311    /// Column name
312    pub column: String,
313    /// Column value
314    pub value: Value,
315    /// Cell timestamp (microseconds)
316    pub timestamp: i64,
317    /// TTL in seconds (None = no expiration)
318    pub ttl: Option<u32>,
319    /// Cell path for a complex (collection / non-frozen UDT) element — the
320    /// serialized element key/index that distinguishes one element of a
321    /// multi-cell column from another (issue #886 substrate).
322    ///
323    /// **Carry-only.** This field is threaded through the merge entry so that
324    /// per-element reconciliation can become byte-faithful, but nothing
325    /// populates or consumes it yet: the reader still collapses collections to
326    /// a single whole-column [`CellData`] and the writer does not read this
327    /// field. Population (per-element reader emit) and consumption (per-path
328    /// merge #844) land in the follow-up #899. `None` for simple cells.
329    pub cell_path: Option<Vec<u8>>,
330    /// Local deletion time in **seconds** since the Unix epoch for this cell
331    /// (the on-disk `localDeletionTime`), used by gc_grace purging and
332    /// expiring-cell tie-breaks (issue #886 substrate).
333    ///
334    /// For an expiring (TTL) cell this is the cell's expiration instant; for a
335    /// cell tombstone it is when the delete was applied.
336    ///
337    /// Populated for complex (collection / UDT) elements on the compaction read
338    /// path (epic #899, Phase C); `None` for simple cells whose LDT the reader
339    /// does not surface, and for live simple cells.
340    pub local_deletion_time: Option<i32>,
341    /// True when this `CellData` represents a single ELEMENT of a non-frozen
342    /// complex column (a list/set member or a map entry), as opposed to a simple
343    /// single-cell column (epic #899, Phase C).
344    ///
345    /// When `true`, [`cell_path`](Self::cell_path) is the element's authoritative
346    /// on-disk path and the merge→mutation step emits a
347    /// [`CellOperation::WriteComplexElement`] (preserving per-element
348    /// ts/ttl/ldt/path) rather than a whole-column `Write`. `false` for every
349    /// simple cell (whole-column collapse no longer happens on the production
350    /// path).
351    ///
352    /// [`CellOperation::WriteComplexElement`]: crate::storage::write_engine::mutation::CellOperation::WriteComplexElement
353    pub is_complex_element: bool,
354    /// Authoritative IS_DELETED (0x01) flag for a complex element (epic #899,
355    /// Phase C). Carried verbatim from [`ComplexElement::is_deleted`] — NOT
356    /// re-derived from value/ttl/ldt shape, so an expiring SET member (empty
357    /// value, ttl + ldt set) is correctly NOT treated as a tombstone
358    /// (no-heuristics mandate). Always `false` for simple cells (a simple cell
359    /// tombstone rides in [`value`](Self::value) as `Value::Tombstone`).
360    ///
361    /// [`ComplexElement::is_deleted`]: crate::storage::sstable::reader::compaction_row::ComplexElement::is_deleted
362    pub is_deleted: bool,
363    /// On-disk HAS_EMPTY_VALUE (0x04) flag for a complex element (epic #899,
364    /// Phase C). `true` for a SET member (whose identity lives in the cell_path)
365    /// and any genuinely empty-value element, so the writer reproduces the same
366    /// on-disk emptiness rather than re-deriving it from the decoded value.
367    /// Always `false` for simple cells.
368    pub has_empty_value: bool,
369}

416/// Complex (collection / non-frozen UDT) deletion marker for one column
417/// (issue #886 substrate).
418///
419/// Cassandra writes a complex-deletion marker ahead of a multi-cell column's
420/// elements to delete every element written at or before `marked_for_delete_at`.
421/// A merged complex deletion is dropped unless it **strictly supersedes** the
422/// active one (Cassandra commit `bd244649`). CQLite currently reduces this to a
423/// boolean and discards the timestamps; this first-class entity preserves them
424/// so per-path merge (#844) and shadow-before-purge (#887) can be byte-faithful.
425///
426/// **Carry-only.** Carried on [`MergeEntry`] (unioned through
427/// `reconcile_cluster`) but not yet populated by the reader or applied during
428/// the merge — that is the follow-up #899/#887.
429#[cfg(feature = "write-support")]
430#[derive(Debug, Clone, PartialEq, Eq)]
431pub struct ComplexDeletion {
432    /// Name of the complex column this deletion covers.
433    pub column: String,
434    /// Deletion timestamp (`markedForDeleteAt`) in microseconds since the epoch.
435    pub marked_for_delete_at: i64,
436    /// Local deletion time in seconds since the epoch.
437    pub local_deletion_time: i32,
438}
