1//! Composite (frozen tuple / UDT / nested frozen collection) collection
2//! key/element handling for [`super`], the merged-read assembler (issue #2339).
3//!
4//! A non-frozen `set` whose element — or `map` whose key — is an opaque composite
5//! keeps that element's IDENTITY in the cell's authoritative `cell_path`, not in
6//! its value (a set cell's value is empty; sstabledump prints `"value":""`). This
7//! module owns the two things the assembler must do with those bytes:
8//!
9//! 1. [`decode_composite`] — DECODE them into a typed `Value::Tuple`/`Value::Udt`/
10//!    nested collection, reusing the canonical structural value deserializer
11//!    (`comparator_value_parsing::parse_value_with_comparator`), the same decoder
12//!    the single-generation read path uses. Never a second structural decoder.
13//! 2. [`compare_composite`] — ORDER them the way Cassandra does.
14//!
15//! ## Why ordering needs its own comparator (measured, not assumed)
16//!
17//! Cassandra writes a complex column's cells in `cellPathComparator()` order,
18//! which for a collection is the declared element/key TYPE's comparator — NOT
19//! unsigned byte order of the serialized form. Verified against Cassandra-written
20//! bytes: in `test_collections.collections_with_udts` partition `c2a32d80…` the
21//! `contacts set<frozen<contact_info>>` cells are on disk in the order
22//! `harrisvalerie@example.net`, `sandra59@example.org` (component-wise text
23//! order), whereas RAW byte comparison of the same two cell paths puts
24//! `sandra59` FIRST — the first component's 4-byte i32-BE LENGTH prefix
25//! (`0x14` vs `0x19`) dominates the byte comparison before any text is reached.
26//! The same inversion shows in `test_nested_udt_keys.nested_udt_keys`, where a
27//! 2-element frozen collection precedes a 1-element one on disk while its
28//! `i32-BE` count prefix would sort it second.
29//!
30//! So the semantics below are transcribed from the pinned `cassandra-5.0.8`
31//! comparators, applied to the DECODED values (which the assembler needs anyway):
32//!
33//! * `TupleType.compareCustom` (`UserType` extends `TupleType`): component-wise
34//!   with each component's own comparator; a NULL component sorts BEFORE a
35//!   non-null one; when one side runs out, the sides are EQUAL iff every
36//!   remaining component of the other is null; an empty value sorts first.
37//! * `CollectionType.compareListOrSet`: element-wise over `min(sizeL, sizeR)`,
38//!   then `Integer.compare(sizeL, sizeR)`.
39//! * `MapType.compareMaps`: `(key, value)`-wise over `min(sizeL, sizeR)`, then
40//!   `Integer.compare(sizeL, sizeR)`.
41//!
42//! Scalar leaves delegate to [`ComparatorType::compare`], resolved from the
43//! DECLARED (schema) leaf type — the single owner of per-type scalar ordering.
44//! There is deliberately NO second scalar-ordering path here: two ordering
45//! authorities for one type is precisely the divergence class issue #2339 exists
46//! to remove, and a schema-driven dispatch is the no-heuristics direction (#28)
47//! where a runtime-`Value`-variant dispatch is not.
48//!
49//! Three leaf types — `varint`, `decimal` and `uuid`/`timeuuid` — have no
50//! Cassandra-compatible ordering in that single owner yet (issue #4063), so
51//! [`compare_composite`] REFUSES an ordering decision on them instead of
52//! inheriting a known-wrong answer; the citations and the reasoning are on
53//! [`divergent_leaf`]. Refusal keeps ONE ordering authority: a locally-correct
54//! comparator here would be the second one, which is the very class above.
55//!
56//! `inet` and `time` need no special case: `types::comparator::custom` orders both
57//! by their serialized form's unsigned byte order, which IS their Cassandra
58//! `ComparisonType.BYTE_ORDER` (fixture-backed against the Cassandra-written
59//! `test_comparator_order` corpus, #3790/#3935).
60//!
61//! Dispatch is on the DECLARED comparator only, never a byte pattern (#28).

72/// Peel `Frozen` wrappers off a comparator.
73pub(super) fn unwrap_frozen_comparator(cmp: &ComparatorType) -> &ComparatorType {
74    match cmp {
75        ComparatorType::Frozen(inner) => unwrap_frozen_comparator(inner),
76        other => other,
77    }
78}

83    match value {

89/// Decode an OPAQUE COMPOSITE collection key/element from its authoritative
90/// `cell_path` identity bytes into a typed `Value` (issue #2339).
91///
92/// Before #2339 the merged-read arm FAILED CLOSED here, which made a correctness
93/// outcome flip on SSTable generation count: one generation decoded via the
94/// single-generation arm, two errored the whole request.
95///
96/// A composite whose declared type never resolved to a STRUCTURE — a `Custom` UDT
97/// reference with no entry in the table's `UdtRegistry` — has no field list to
98/// decode into, so it still fails closed with a clear error naming the column and
99/// the declared type. Emitting opaque bytes instead would break the typed Arrow
100/// builder one layer deeper (roborev 1632).
101pub(super) fn decode_composite(
102    column: &str,
103    kind: &str,
104    bytes: &[u8],
105    cmp: &ComparatorType,
106) -> Result<Value> {
107    reject_trailing_bytes(column, kind, bytes, cmp)?;
108    if let Some(name) = first_unresolved_custom(cmp) {
109        return Err(Error::unsupported_format(format!(
110            "column '{column}': {kind} type '{name}' did not resolve to a structure \
111             — a UDT reference with no definition in the table's UDT registry has no \
112             field list to decode into; failing closed rather than emitting opaque \
113             bytes (issues #28/#2339)"
114        )));
115    }
116    parse_value_with_comparator(bytes, cmp)
117}

119/// Refuse a composite cell path whose framing does not consume the WHOLE slice.
120///
121/// **Why this is required here and not inside the shared decoder (roborev job 60).**
122/// `parse_value_with_comparator`'s tuple/UDT helpers stop once every DECLARED field
123/// is read and ignore whatever follows, because their other callers hand them element
124/// bytes already bounded by an outer length prefix — trailing bytes cannot occur
125/// there. **A cell path has no such outer bound: the whole slice IS the key.** So two
126/// DISTINCT cell paths that share a prefix decoded to the SAME logical map key / set
127/// element, which in a map is a duplicate-key hazard and in a set collapses two
128/// members into one.
129///
130/// The rule is the one `cell_path_key.rs` already states for the single-generation
131/// reader, and it is applied here verbatim so the two arms agree: **`Err` only where
132/// Cassandra's own `validate`/`split` throws** — and `TupleType.validate` at the
133/// pinned `cassandra-5.0.8` tag throws on trailing bytes after a composite's
134/// components. Such input is corrupt on Cassandra's own terms, so refusing it adds no
135/// availability risk for data Cassandra itself would have read.
136///
137/// Only the TOP-LEVEL framing is walked, which is where the unbounded slice is; a
138/// nested value is bounded by its own `i32` length prefix. Field bytes are never
139/// inspected — the walk is over the framing alone (no-heuristics, #28) — and a length
140/// that would run past the end is itself a refusal rather than a panic, since this
141/// slice is attacker-controlled SSTable content.
142fn reject_trailing_bytes(
143    column: &str,
144    kind: &str,
145    bytes: &[u8],
146    cmp: &ComparatorType,
147) -> Result<()> {
148    let arity = match unwrap_frozen_comparator(cmp) {
149        ComparatorType::Tuple(fields) => fields.len(),
150        ComparatorType::Udt {
151            field_comparators, ..
152        } => field_comparators.len(),
153        // Only tuple/UDT framing is fixed-arity and therefore checkable this way.
154        // A frozen collection carries its own leading count, and a scalar leaf is
155        // width-validated by its own decoder.
156        _ => return Ok(()),
157    };
158
159    let mut off = 0usize;
160    for _ in 0..arity {
161        // A composite with FEWER encoded components than declared is legal —
162        // Cassandra's `compareCustom` treats an omitted suffix as all-null — so
163        // running out of input here is not an error, only a stop.
164        if off == bytes.len() {
165            return Ok(());
166        }
167        let Some(len_bytes) = bytes.get(off..off + 4) else {
168            return Err(trailing_error(column, kind, bytes.len(), off));
169        };
170        off += 4;
171        let raw = i32::from_be_bytes([len_bytes[0], len_bytes[1], len_bytes[2], len_bytes[3]]);
172        if raw < 0 {
173            // Negative length == a null component; it consumes no value bytes.
174            continue;
175        }
176        let Some(next) = off.checked_add(raw as usize) else {
177            return Err(trailing_error(column, kind, bytes.len(), off));
178        };
179        if next > bytes.len() {
180            return Err(trailing_error(column, kind, bytes.len(), off));
181        }
182        off = next;
183    }
184
185    if off != bytes.len() {
186        return Err(trailing_error(column, kind, bytes.len(), off));
187    }
188    Ok(())
189}

191fn trailing_error(column: &str, kind: &str, total: usize, consumed: usize) -> Error {
192    Error::corruption(format!(
193        "column '{column}': {kind} cell path is {total} bytes but its declared \
194         composite framing consumes {consumed} — trailing or truncated bytes after the \
195         components. Cassandra's TupleType.validate throws on this, so the key is \
196         corrupt on its own terms; decoding it from a prefix would let two distinct \
197         cell paths collapse to one logical key (issues #28/#2339)"
198    ))
199}

201/// The name of the first `Custom` comparator node ANYWHERE in `cmp`'s tree that
202/// `custom_scalar::decode_custom_scalar` cannot decode, or `None` when every
203/// `Custom` node is one it recognises.
204///
205/// **Why this is RECURSIVE, and why a top-level-only check was a silent-wrong-value
206/// bug (roborev job 52, G2).** The guard in [`decode_composite`] originally tested
207/// only the comparator that `unwrap_frozen_comparator` returns, i.e. the top level.
208/// An unresolved UDT reference NESTED inside a tuple, UDT or collection — say
209/// `set<frozen<tuple<frozen<unregistered_udt>, int>>>` — leaves `Custom` at a nested
210/// position, so the guard passed and the decoder's `_ =>` arm turned that field into
211/// an opaque `Value::Blob`. A multi-generation read then emitted AND SORTED a
212/// plausible-looking wrong value instead of failing closed, which is exactly the
213/// class the no-heuristics mandate forbids (#28; the same shape as #3612's
214/// "the blob fallback must not swallow a recognised-but-unhandled composite").
215///
216/// The admitted set is **`time` / `inet` / `json`** because that is precisely what
217/// `custom_scalar::decode_custom_scalar` matches; every other name falls to its
218/// `_ =>` arm and becomes a `Blob`. Read from that function rather than assumed —
219/// if it gains an arm, this list must gain it too, or a decodable type starts
220/// failing closed.
221fn first_unresolved_custom(cmp: &ComparatorType) -> Option<&str> {
222    match cmp {
223        ComparatorType::Custom(name) => {
224            if matches!(name.as_str(), "time" | "inet" | "json") {
225                None
226            } else {
227                Some(name.as_str())
228            }
229        }
230        ComparatorType::Frozen(inner) => first_unresolved_custom(inner),
231        ComparatorType::List(elem) | ComparatorType::Set(elem) => first_unresolved_custom(elem),
232        ComparatorType::Map(key, val) => {
233            first_unresolved_custom(key).or_else(|| first_unresolved_custom(val))
234        }
235        ComparatorType::Tuple(fields) => fields.iter().find_map(first_unresolved_custom),
236        ComparatorType::Udt {
237            field_comparators, ..
238        } => field_comparators
239            .iter()
240            .find_map(|(_, c)| first_unresolved_custom(c)),
241        _ => None,
242    }
243}

245/// Order two decoded composite keys/elements exactly as Cassandra's type
246/// comparator does (see the module doc for the pinned-source transcription).
247///
248/// `Err` on a value/comparator shape mismatch the declared metadata should never
249/// produce — surfaced rather than silently mis-ordered — and on a scalar leaf whose
250/// central comparator is KNOWN to diverge from Cassandra's (see
251/// [`divergent_leaf`]), which is refused rather than ordered wrongly.
252///
253/// `column` is carried only to NAME the column in those refusals, exactly as the
254/// decode-side refusals in this module do; it never influences the ordering.
255pub(super) fn compare_composite(
256    column: &str,
257    left: &Value,
258    right: &Value,
259    cmp: &ComparatorType,
260) -> Result<Ordering> {
261    let cmp = unwrap_frozen_comparator(cmp);
262    let left = unwrap_frozen_value(left);
263    let right = unwrap_frozen_value(right);
264    match cmp {
265        // `TupleType.compareCustom` — `UserType` extends `TupleType`, so a UDT and
266        // a tuple order identically (component-wise, nulls first).
267        ComparatorType::Tuple(field_cmps) => {
268            let (l, r) = (tuple_fields(left, cmp)?, tuple_fields(right, cmp)?);
269            compare_components(column, &l, &r, field_cmps)
270        }
271        ComparatorType::Udt {
272            field_comparators, ..
273        } => {
274            let (l, r) = (udt_fields(left, cmp)?, udt_fields(right, cmp)?);
275            let field_cmps: Vec<&ComparatorType> =
276                field_comparators.iter().map(|(_, c)| c).collect();
277            compare_components_ref(column, &l, &r, &field_cmps)
278        }
279        // `CollectionType.compareListOrSet`: element-wise, then size.
280        ComparatorType::List(elem) | ComparatorType::Set(elem) => {
281            let (l, r) = (sequence(left, cmp)?, sequence(right, cmp)?);
282            for (a, b) in l.iter().zip(r.iter()) {
283                match compare_composite(column, a, b, elem)? {
284                    Ordering::Equal => {}
285                    other => return Ok(other),
286                }
287            }
288            Ok(l.len().cmp(&r.len()))
289        }
290        // `MapType.compareMaps`: (key, value)-wise, then size.
291        ComparatorType::Map(key_cmp, val_cmp) => {
292            let (l, r) = (entries(left, cmp)?, entries(right, cmp)?);
293            for ((lk, lv), (rk, rv)) in l.iter().zip(r.iter()) {
294                match compare_composite(column, lk, rk, key_cmp)? {
295                    Ordering::Equal => {}
296                    other => return Ok(other),
297                }
298                match compare_composite(column, lv, rv, val_cmp)? {
299                    Ordering::Equal => {}
300                    other => return Ok(other),
301                }
302            }
303            Ok(l.len().cmp(&r.len()))
304        }
305        // Every scalar leaf delegates to the CENTRAL comparator, resolved from the
306        // DECLARED (schema) leaf type. `cmp` here is the frozen-unwrapped declared
307        // comparator for this component, so this is a schema-driven dispatch, never
308        // an inspection of the runtime value (no-heuristics, #28).
309        //
310        // `inet`/`time` come along for free: `types::comparator::custom` orders both
311        // by their serialized form's unsigned byte order — `compare_inet` is a
312        // `[u8]` compare of the raw address and `compare_time` a compare of
313        // `i64::to_be_bytes` — which is `ComparisonType.BYTE_ORDER` verbatim, backed
314        // by the Cassandra-written `test_comparator_order` fixtures (#3790/#3935).
315        //
316        // ONE authority, deliberately. An earlier revision of this arm called the
317        // WRITE path's `collection_order::compare_collection_elements` instead; that
318        // was wrong. It dispatches on the runtime `Value` VARIANT while this one
319        // dispatches on the SCHEMA type, so keeping both would leave two ordering
320        // authorities for one type — the divergence class #2339 exists to remove.
321        //
322        // GAP (tracked as #4063), and what this arm DOES about it: three leaf types
323        // have no Cassandra-compatible ordering in `ComparatorType::compare`, so an
324        // ordering DECISION on one of them is REFUSED here rather than answered
325        // wrongly (see [`divergent_leaf`] for the per-type citations). Returning the
326        // central comparator's answer would emit a plausible-looking collection in
327        // the wrong order — a silent-wrong-value outcome, which is the class #28
328        // forbids and the same shape as the nested-unresolved-UDT bug this module
329        // already fails closed on.
330        //
331        // Fixing the central arms is deliberately NOT done here: it is a convergence
332        // on `collection_order::scalar` (which already implements all three under its
333        // own Cassandra citations) with its own blast radius, tracked as #4063. A
334        // local correct comparator would be a SECOND ordering authority for the same
335        // types — the divergence class #2339 exists to remove — so refusal, not a
336        // second path, is the only option that keeps ONE authority.
337        _ => {
338            if let Some(leaf) = divergent_leaf(cmp) {
339                return Err(divergent_leaf_error(column, leaf));
340            }
341            cmp.compare(left, right)
342        }
343    }
344}

346/// A scalar leaf type whose CENTRAL `ComparatorType::compare` arm is KNOWN not to
347/// implement Cassandra's ordering, with the pinned-source citation for WHY
348/// (issue #4063).
349///
350/// This is a REFUSAL list, not a second comparator: [`compare_composite`] fails
351/// closed for these leaves instead of ordering them. It exists because the
352/// alternative — inheriting `ComparatorType::compare`'s answer — returns a
353/// plausible-looking collection in an order Cassandra would not produce, and a
354/// merged read has no way for the caller to notice.
355///
356/// The divergences, read from `types::comparator` rather than assumed, against the
357/// pinned `cassandra-5.0.8` comparators:
358///
359/// * `varint` — `compare_varint` is `Bytes::cmp`, i.e. raw UNSIGNED bytes, where
360///   Cassandra `IntegerType.compare` orders by SIGNED two's-complement magnitude
361///   (so `-1`, body `0xFF`, sorts ABOVE `0` here — reversed).
362/// * `decimal` — `compare_decimal` normalizes unequal scales by comparing
363///   `format!("{:?}.{}", unscaled, scale)` STRINGS, self-described in source as
364///   "For now, simple string comparison"; Cassandra `DecimalType.compare` compares
365///   NUMERICALLY.
366/// * `uuid` — `compare_uuid` is `Uuid::cmp` (raw bytes), where Cassandra `UUIDType`
367///   compares version first, then a v1 timestamp, then the tail. `CqlType::TimeUuid`
368///   maps onto this SAME `ComparatorType::Uuid` variant, so a `timeuuid` leaf is
369///   refused here too — and Cassandra's `TimeUUIDType` order differs again, so
370///   inheriting would be wrong for both spellings.
371///
372/// **SCOPE — why the refusal is CONFINED to the composite arm, measured not
373/// aesthetic.** The same central defect affects a SCALAR element/key of these types
374/// (`set<uuid>`, `set<decimal>`, `map<varint, ...>`), which
375/// `super::sort_elements_by_cell_path` orders with the same
376/// `ComparatorType::compare`. That arm is deliberately NOT narrowed here:
377///
378/// * For a COMPOSITE, refusing restores the PRE-#2339 outcome exactly — that arm
379///   failed closed for EVERY composite (see [`decode_composite`]) — so the refusal
380///   costs no behaviour that existed before #2339, and it stops #2339 introducing a
381///   NEW silent-wrong-order case.
382/// * For a SCALAR the wrong ordering is long-standing AND in-corpus: the committed
383///   Cassandra-written fixtures declare `uuid_set SET<UUID>` / `decimal_set
384///   SET<DECIMAL>` (`test-data/schemas/collections.cql`) and `related_products
385///   SET<UUID>` (`wide-rows.cql`), so refusing there would convert an ordering
386///   imprecision into an availability failure for data Cassandra reads fine.
387///
388/// Both halves converge when #4063 fixes the central arms — that is the fix, and
389/// this guard is deleted with it (the `#[ignore]`d tripwire test says so too).
390///
391/// Dispatch is on the DECLARED comparator, never on the runtime value (#28).
392/// The first leaf ANYWHERE in `cmp`'s tree whose ordering the merged read path
393/// REFUSES (issue #4063), or `None` when every leaf is orderable.
394///
395/// **Why this is public, and why the bypass arm MUST ask it rather than keep its own
396/// list (issue #3058 + roborev job 116 F1).** `compare_composite` refuses these leaves,
397/// so a MERGED read of such a collection fails closed — while a one-source read that
398/// BYPASSES merging would decode and return it, succeeding. The same query would then
399/// start failing the moment a second SSTable appeared, which is a worse failure than
400/// either arm's own behaviour: it is arm-dependent and therefore unpredictable.
401///
402/// The bypass predicate keys its divergence check on this function so there is ONE
403/// authority for "can the merged arm order this?". A second copy of the leaf list in
404/// `cqlite-flight` is exactly the two-ordering-authorities divergence #2339 exists to
405/// remove, one crate over.
406///
407/// Recursive for the same reason `first_unresolved_custom` is: a refused leaf can sit
408/// at any depth, and a top-level-only check was a real silent-wrong-value bug here
409/// (roborev job 52, G2).
410pub fn first_unorderable_leaf(cmp: &ComparatorType) -> Option<String> {
411    if let Some((leaf, _)) = divergent_leaf(cmp) {
412        return Some(leaf.into_owned());
413    }
414    match cmp {
415        ComparatorType::Frozen(inner) => first_unorderable_leaf(inner),
416        ComparatorType::List(elem) | ComparatorType::Set(elem) => first_unorderable_leaf(elem),
417        ComparatorType::Map(key, val) => {
418            first_unorderable_leaf(key).or_else(|| first_unorderable_leaf(val))
419        }
420        ComparatorType::Tuple(fields) => fields.iter().find_map(first_unorderable_leaf),
421        ComparatorType::Udt {
422            field_comparators, ..
423        } => field_comparators
424            .iter()
425            .find_map(|(_, c)| first_unorderable_leaf(c)),
426        _ => None,
427    }
428}

430fn divergent_leaf(cmp: &ComparatorType) -> Option<(Cow<'_, str>, &'static str)> {
431    match cmp {
432        ComparatorType::Varint => Some((
433            Cow::Borrowed("varint"),
434            "IntegerType (signed two's-complement magnitude), where compare_varint is raw \
435             unsigned byte order",
436        )),
437        ComparatorType::Decimal => Some((
438            Cow::Borrowed("decimal"),
439            "DecimalType (numeric across scales), where compare_decimal falls back to a \
440             formatted-string comparison for unequal scales",
441        )),
442        ComparatorType::Uuid => Some((
443            Cow::Borrowed("uuid/timeuuid"),
444            "UUIDType (version, then v1 timestamp, then tail), where compare_uuid is raw \
445             byte order",
446        )),
447        // `Custom` is the SAME defect class arriving by a different door, and the
448        // decode-side guard cannot catch it. `first_unresolved_custom` deliberately
449        // ADMITS every name `decode_custom_scalar` can decode — `time`, `inet` AND
450        // `json` — so a `json` leaf decodes, passes that guard, and reaches here.
451        // `custom::compare` then handles only `inet`/`time` and falls to a
452        // DISPLAY-STRING comparison for everything else, which is the same
453        // "string comparison for now" shape refused above for `decimal`.
454        //
455        // The predicate is `supports_ordering()`, not a name list, because that IS
456        // the codebase's own answer to "does this type order?" — it reports FALSE
457        // for `json` while this path was ordering it anyway. Deriving the refusal
458        // from it means a future `decode_custom_scalar` arm cannot silently become
459        // orderable: it starts REFUSING until someone gives it a real comparator,
460        // which is the direction `first_unresolved_custom`'s own doc asks for.
461        // `inet`/`time` report TRUE (fixture-backed by the Cassandra-written
462        // `test_comparator_order` corpus) and are unaffected.
463        ComparatorType::Custom(name) if !cmp.supports_ordering() => Some((
464            Cow::Owned(format!("custom '{name}'")),
465            "no implemented ordering at all — `custom::compare` falls back to a \
466             DISPLAY-STRING comparison for this name, and `ComparatorType::\
467             supports_ordering` reports false for it",
468        )),
469        // The `Json` VARIANT is the same string-comparison defect one enum arm over
470        // (`compare_json` serialises both sides and compares the strings). No
471        // `CqlType` maps to it today, so this arm is defence in depth rather than a
472        // reachable case — cheap, and it cannot become reachable silently.
473        ComparatorType::Json => Some((
474            Cow::Borrowed("json"),
475            "no Cassandra ordering — `compare_json` compares serialised JSON \
476             strings",
477        )),
478        _ => None,
479    }
480}

482/// The refusal [`compare_composite`] returns for a [`divergent_leaf`].
483///
484/// Names the COLUMN, the offending LEAF TYPE and issue #4063, in the same style as
485/// this module's decode-side refusals, so an operator can tell which column and
486/// which type to look at without reading source.
487fn divergent_leaf_error(column: &str, (leaf, citation): (Cow<'_, str>, &'static str)) -> Error {
488    Error::unsupported_format(format!(
489        "column '{column}': refusing to ORDER a composite collection key/element \
490         whose '{leaf}' leaf has no Cassandra-compatible ordering yet — the central \
491         ComparatorType::compare arm for '{leaf}' diverges from Cassandra's {citation} \
492         (tracked as issue #4063). Failing closed rather than returning a \
493         plausible-looking collection in the wrong order (issues #28/#2339)"
494    ))
495}
496
497/// `TupleType.compareCustom`'s component loop over already-decoded components.
498///
499/// * a NULL component sorts BEFORE a non-null one (`sizeL < 0` ⇒ `-1`);
500/// * when one side runs out of components, the two are EQUAL iff every remaining
501///   component of the other side is NULL, else the side with content is GREATER;
502/// * an EMPTY value therefore sorts first, matching the `isEmpty` pre-check.
503fn compare_components(
504    column: &str,
505    left: &[Value],
506    right: &[Value],
507    field_cmps: &[ComparatorType],
508) -> Result<Ordering> {
509    let refs: Vec<&ComparatorType> = field_cmps.iter().collect();
510    compare_components_ref(column, left, right, &refs)
511}
512
513fn compare_components_ref(
514    column: &str,
515    left: &[Value],
516    right: &[Value],
517    field_cmps: &[&ComparatorType],
518) -> Result<Ordering> {
519    let common = left.len().min(right.len()).min(field_cmps.len());
520    for i in 0..common {
521        let (l, r) = (&left[i], &right[i]);
522        match (l.is_null(), r.is_null()) {
523            (true, true) => continue,
524            (true, false) => return Ok(Ordering::Less),
525            (false, true) => return Ok(Ordering::Greater),
526            (false, false) => {}
527        }
528        match compare_composite(column, l, r, field_cmps[i])? {
529            Ordering::Equal => {}
530            other => return Ok(other),
531        }
532    }
533    let l_rest_null = left[common.min(left.len())..].iter().all(Value::is_null);
534    let r_rest_null = right[common.min(right.len())..].iter().all(Value::is_null);
535    match (l_rest_null, r_rest_null) {
536        (true, true) => Ok(Ordering::Equal),
537        (true, false) => Ok(Ordering::Less),
538        (false, true) => Ok(Ordering::Greater),
539        // Both sides still carry non-null components beyond the comparable
540        // prefix; only reachable when `field_cmps` is SHORTER than both decoded
541        // component lists, i.e. the declared type contradicts the bytes.
542        (false, false) => Ok(left.len().cmp(&right.len())),
543    }
544}
545
546fn shape_error(value: &Value, cmp: &ComparatorType) -> Error {
547    Error::Schema(format!(
548        "composite collection key/element: decoded {value:?} does not match the \
549         declared comparator {cmp:?}"
550    ))
551}
552
553fn tuple_fields(value: &Value, cmp: &ComparatorType) -> Result<Vec<Value>> {
554    match value {
555        Value::Tuple(fields) => Ok(fields.clone()),
556        other => Err(shape_error(other, cmp)),
557    }
558}
559
560fn udt_fields(value: &Value, cmp: &ComparatorType) -> Result<Vec<Value>> {
561    match value {
562        Value::Udt(udt) => Ok(udt
563            .fields
564            .iter()
565            .map(|f| f.value.clone().unwrap_or(Value::Null))
566            .collect()),
567        other => Err(shape_error(other, cmp)),
568    }
569}
570
571fn sequence(value: &Value, cmp: &ComparatorType) -> Result<Vec<Value>> {
572    match value {
573        Value::List(items) | Value::Set(items) => Ok(items.clone()),
574        other => Err(shape_error(other, cmp)),
575    }
576}
577
578fn entries(value: &Value, cmp: &ComparatorType) -> Result<Vec<(Value, Value)>> {
579    match value {
580        Value::Map(pairs) => Ok(pairs.clone()),
581        other => Err(shape_error(other, cmp)),
582    }
