1//! Producer-thread gauge for the k-way merge (issue #2316).
2//!
3//! Backs the [`crate::observability::catalog::MERGE_PRODUCER_THREADS`] gauge with
4//! a process-global live count of merge producer OS threads. The count is
5//! incremented BEFORE the OS thread is spawned ([`spawned`], called from
6//! `SSTableRowIteratorAdapter::open`) so the increment happens-before any possible
7//! decrement (correct-by-construction: the decrementing `ProducerThreadGuard` is
8//! created first thing inside the child thread, so a fast-exiting producer can
9//! never race its own decrement against the increment). If the spawn itself then
10//! fails ([`rollback`], called from the same `open` on a `Builder::spawn` `Err`),
11//! the increment is rolled back so it never leaks for a thread that never
12//! started. The guard decrements when the producer thread exits (even on panic).
13//! The gauge is re-recorded on each change, so it RISES to the `O(M)` producer
14//! count during a merge and RETURNS to baseline once the producers exit — making
15//! the previously-invisible per-merge thread cost observable on a loaded node.
16
17use std::sync::atomic::{AtomicI64, Ordering};
18
19use crate::observability;
20
21static LIVE: AtomicI64 = AtomicI64::new(0);
22
23fn record(live: i64) {
24    observability::record_gauge(observability::catalog::MERGE_PRODUCER_THREADS, live, &[]);
25}
26
27fn decrement() {
28    record(LIVE.fetch_sub(1, Ordering::SeqCst) - 1);
29}
30
31/// Account a just-about-to-spawn producer thread on the gauge, BEFORE
32/// `std::thread::Builder::spawn` is called. Balanced by exactly one of:
33/// [`ProducerThreadGuard`]'s drop (the thread started and later exited), or
34/// [`rollback`] (the thread never actually started).
35pub(super) fn spawned() {
36    record(LIVE.fetch_add(1, Ordering::SeqCst) + 1);
37}
38
39/// Undo a [`spawned`] increment for a producer that failed to actually start
40/// (`Builder::spawn` returned `Err`) — the increment must not leak for a thread
41/// that never ran (and thus never gets a `ProducerThreadGuard` decrement).
42pub(super) fn rollback() {
43    decrement();
44}
45
46/// RAII guard that decrements the live producer-thread count (and re-records the
47/// gauge) when a producer thread exits — even on panic. Created as the FIRST act
48/// inside the spawned producer thread, so it is guaranteed to run for every
49/// thread that actually started (pairing exactly one decrement per [`spawned`]
50/// increment that was not rolled back).
51pub(super) struct ProducerThreadGuard;
52
53impl Drop for ProducerThreadGuard {
54    fn drop(&mut self) {
55        decrement();
56    }
57}
