1//! Fully-expired SSTable drop classification (issue #1388).
2//!
3//! Parity with Cassandra `CompactionController.getFullyExpiredSSTables`: before a
4//! compaction rewrites anything, classify the subset of input SSTables that are
5//! entirely past `gcBefore` AND provably shadow nothing outside the compaction
6//! set, so they can be DROPPED WHOLE (excluded from the K-way merge and reclaimed
7//! after the output publishes) instead of read, merged, and re-serialized.
8//!
9//! The classification is **metadata-only** — a single integer comparison against
10//! the authoritative `Statistics.db` `TimestampStatistics.max_deletion_time`
11//! (Cassandra `StatsMetadata.maxLocalDeletionTime`) plus the existing #935
12//! overlap bound (`compute_max_purgeable_timestamp`). It NEVER reads/decodes/scans
13//! a candidate's `Data.db` (no-heuristics mandate, issue #28): that is both the
14//! whole point of the optimization (dropping must not pay a scan) and the
15//! no-heuristics-compliant path. Trust in `Statistics.db` is already load-bearing
16//! across the write engine (`compute_baseline_min`, `compute_max_purgeable_timestamp`).
17//!
18//! The whole module is gated `#[cfg(feature = "write-support")]` at its `mod`
19//! declaration in the parent, so no inner `#![cfg]` is needed here.
20

26/// Classify a candidate SSTable as *fully expired* for a compaction with cutoff
27/// `gc_before_secs` from AUTHORITATIVE `Statistics.db` metadata ONLY.
28///
29/// A candidate is fully expired iff EVERY cell/tombstone in it has
30/// `localDeletionTime < gcBefore`. The single authoritative field that proves this
31/// is `TimestampStatistics.max_deletion_time` (Cassandra
32/// `StatsMetadata.maxLocalDeletionTime`): if the MAXIMUM local-deletion-time in the
33/// SSTable is below `gcBefore`, all of them are. This is exactly Cassandra's
34/// `getFullyExpiredSSTables` predicate
35/// (`sstable.getSSTableMetadata().maxLocalDeletionTime < gcBefore`).
36///
37/// The LIVE / `NO_DELETION_TIME` sentinel (an SSTable holding any live, non-TTL
38/// data) surfaces from the parser as `i64::MAX`, which is never `< gcBefore`, so
39/// such an SSTable is correctly NOT classified fully expired — the sentinel falls
40/// out of the comparison naturally with no special-casing.
41pub(super) fn is_fully_expired(stats: &TimestampStatistics, gc_before_secs: i64) -> bool {
42    stats.max_deletion_time < gc_before_secs
43}

45/// The authoritative newest write timestamp of an SSTable, or `None` when it is
46/// UNAVAILABLE (#1729 fail-closed `i64::MIN` sentinel).
47///
48/// Mirrors [`crate::storage::sstable::statistics_reader::StatisticsReader::max_timestamp`]:
49/// a drop/GC gate must never treat an unknown newest write as `i64::MIN` (which
50/// would compare below any overlap bound and wrongly permit a drop). Returning
51/// `None` here forces the overlap gate to FAIL CLOSED (retain the candidate).
52fn authoritative_max_timestamp(stats: &TimestampStatistics) -> Option<i64> {
53    match stats.max_timestamp {
54        i64::MIN => None,
55        v => Some(v),
56    }
57}

59/// Read a candidate SSTable's `TimestampStatistics` from its sibling
60/// `Statistics.db`. Returns `None` when the file is absent or fails to parse — the
61/// conservative signal that its expiry/timestamp facts are UNKNOWN, so the
62/// candidate must NOT be dropped (matches `compute_max_purgeable_timestamp`
63/// degradation). Authoritative metadata only (no cell scan).
64fn read_timestamp_stats(data_path: &Path) -> Option<TimestampStatistics> {
65    let stats_path = stats_path_for(data_path);
66    if !stats_path.exists() {
67        return None;
68    }
69    let stats_bytes = match std::fs::read(&stats_path) {
70        Ok(b) => b,
71        Err(e) => {
72            log::warn!(
73                "Could not read Statistics.db {:?} for fully-expired check: {}",
74                stats_path,
75                e
76            );
77            return None;
78        }
79    };
80    match crate::parser::enhanced_statistics_parser::parse_statistics_with_fallback(
81        &stats_bytes,
82        None,
83    ) {
84        Ok((_, sstable_stats)) => Some(sstable_stats.timestamp_stats),
85        Err(e) => {
86            log::warn!(
87                "Could not parse Statistics.db {:?} for fully-expired check: {:?}",
88                stats_path,
89                e
90            );
91            None
92        }
93    }
94}

96/// Compute the subset of `input_paths` that may be DROPPED WHOLE for this
97/// compaction (parity with Cassandra `CompactionController.getFullyExpiredSSTables`).
98///
99/// A candidate is included in the drop-set iff BOTH hold, from authoritative
100/// `Statistics.db` metadata only (no cell scan):
101///
102/// 1. **Fully expired** — `max_deletion_time < gc_before_secs` (see
103///    [`is_fully_expired`]). Since #1728 the writer stamps live cells with the
104///    `NO_DELETION_TIME` sentinel, so a MIXED SSTable (an old tombstone below
105///    `gcBefore` PLUS any live non-TTL cell) finalizes `max_deletion_time` as the
106///    `i32::MAX` live sentinel (parsed back as `i64::MAX`), which is never
107///    `< gc_before_secs`. Such a mixed SSTable is therefore correctly NOT
108///    fully-expired — closing the former data-loss gap (roborev F1) with the now
109///    authoritative `max_local_deletion_time`.
110/// 2. **Overlap-safe** — its authoritative `max_timestamp` is STRICTLY LESS THAN
111///    the minimum write timestamp across every overlapping SSTable that is NOT
112///    itself fully-expired: both the OUTSIDE overlapping SSTables (`outside_paths`)
113///    AND the NON-EXPIRED inputs of THIS compaction. If the candidate's newest
114///    write predates all of them, nothing it holds (tombstone or data) can shadow
115///    live data that would otherwise survive, so dropping it can never resurrect
116///    data. Mirrors Cassandra `CompactionController.getFullyExpiredSSTables`, which
117///    folds the min write timestamp of the non-fully-expired *compacting* SSTables
118///    into the same overlap bound as the non-compacting overlapping ones. The
119///    non-compacting bound reuses [`compute_max_purgeable_timestamp`] (the coarse
120///    global-`min_timestamp` #935 gate; OQ-2 → (A), key-range precision deferred).
121///    Since #1729 the candidate's `max_timestamp` may be the `i64::MIN` UNAVAILABLE
122///    sentinel; a candidate whose newest write is unknown CANNOT be proven to
123///    predate the overlap bound, so it FAILS CLOSED (is retained, never dropped) —
124///    closing the former resurrection gap (roborev F2).
125///
126/// A non-expired INPUT is included in the bound because dropping a fully-expired
127/// SSTable whose tombstone shadows an OLDER live cell in a *co-compacting* input
128/// would resurrect that cell: the tombstone (dropped) no longer purges it, and the
129/// co-input's live cell is merged into the output. (A fully-expired co-input is NOT
130/// folded in: its own data is past `gcBefore` and would be purged anyway, so a
131/// tombstone shadowing it can never resurrect surviving data.)
132///
133/// Conservatism (never resurrect data, never drop live data):
134/// - `gc_before_secs == None` (invalid/absent gc_grace disables purging) ⇒ empty
135///   drop-set.
136/// - An empty `outside_paths` AND no non-expired inputs (e.g. a FULL/major
137///   compaction of only-expired SSTables) ⇒ the bound is `+inf` ⇒ every candidate
138///   is droppable.
139/// - A non-empty `outside_paths` whose bound is UNKNOWN
140///   (`compute_max_purgeable_timestamp` returned `None` because an outside
141///   `Statistics.db` was unreadable) ⇒ empty drop-set (cannot prove safety).
142/// - A candidate whose own `Statistics.db` is absent/unreadable ⇒ not droppable
143///   (it is treated as non-expired, and its `min_timestamp` is unknown, so it also
144///   cannot lower the bound — the whole drop-set is disabled to stay safe).
145///
146/// The returned paths are a subset of `input_paths`, in input order.
147pub fn fully_expired_sstables(
148    input_paths: &[PathBuf],
149    outside_paths: &[PathBuf],
150    gc_before_secs: Option<i64>,
151) -> Vec<PathBuf> {
152    // No cutoff ⇒ purging is disabled ⇒ never drop (conservative, matches
153    // compute_gc_before degradation).
154    let Some(gc_before) = gc_before_secs else {
155        return Vec::new();
156    };
157
158    // Read each input's timestamp stats ONCE. A candidate whose own Statistics.db
159    // is unreadable is treated as non-expired (not droppable) AND, because its
160    // min_timestamp is then unknown, it cannot be safely folded into the overlap
161    // bound — so an unreadable INPUT disables the whole drop-set (conservative).
162    let mut input_stats: Vec<(&PathBuf, Option<TimestampStatistics>)> =
163        Vec::with_capacity(input_paths.len());
164    for p in input_paths {
165        input_stats.push((p, read_timestamp_stats(p)));
166    }
167
168    // Outside overlap bound (non-compacting overlapping SSTables). An empty outside
169    // set contributes +inf; a non-empty set with an UNKNOWN bound (an outside
170    // Statistics.db unreadable) ⇒ cannot prove safety ⇒ drop nothing.
171    let outside_bound: Option<i64> = if outside_paths.is_empty() {
172        Some(i64::MAX)
173    } else {
174        match compute_max_purgeable_timestamp(outside_paths) {
175            Some(bound) => Some(bound),
176            None => return Vec::new(),
177        }
178    };
179
180    classify_drop_set(&input_stats, outside_bound, gc_before)
181}

183/// Pure drop-set classifier over already-read stats (no I/O), shared by
184/// [`fully_expired_sstables`] and its regression tests.
185///
186/// `outside_bound` is the overlap bound from the OUTSIDE (non-compacting) SSTables:
187/// `Some(i64::MAX)` for an empty outside set (+inf), `Some(b)` for a known bound,
188/// and callers must have already returned an empty drop-set for an UNKNOWN outside
189/// bound before calling this. It is here typed `Option` only so a caller can pass a
190/// resolved bound; a `None` here fails closed (empty drop-set).
191///
192/// Returns the subset of paths (in order) that may be dropped whole. See
193/// [`fully_expired_sstables`] for the full contract.
194fn classify_drop_set(
195    input_stats: &[(&PathBuf, Option<TimestampStatistics>)],
196    outside_bound: Option<i64>,
197    gc_before: i64,
198) -> Vec<PathBuf> {
199    let Some(mut overlap_bound) = outside_bound else {
200        return Vec::new();
201    };
202
203    // Fold in the min write timestamp of every NON-EXPIRED input of THIS
204    // compaction (Cassandra parity). A fully-expired input does not constrain the
205    // bound (its own data is past gcBefore and purged anyway). An input whose
206    // stats could not be read is NON-expired with an UNKNOWN min_timestamp, which
207    // we cannot fold safely ⇒ disable the drop-set.
208    for (_, stats) in input_stats {
209        match stats {
210            Some(s) if is_fully_expired(s, gc_before) => {}
211            Some(s) => overlap_bound = overlap_bound.min(s.min_timestamp),
212            None => return Vec::new(),
213        }
214    }
215
216    input_stats
217        .iter()
218        .filter_map(|(path, stats)| {
219            let stats = stats.as_ref()?;
220            // Fully expired AND provably shadows nothing that would otherwise
221            // survive (its newest write predates every non-expired overlap).
222            // #1729: an UNAVAILABLE max_timestamp (i64::MIN sentinel) cannot be
223            // proven to predate the bound ⇒ fail closed (retain, never drop).
224            let candidate_max_ts = authoritative_max_timestamp(stats)?;
225            if is_fully_expired(stats, gc_before) && candidate_max_ts < overlap_bound {
226                Some((*path).clone())
227            } else {
228                None
229            }
230        })
231        .collect()
232}

234/// Split `input_paths` into `(merge_inputs, dropped_whole)` given a `drop_set`
235/// (a subset of `input_paths`, e.g. from [`fully_expired_sstables`]).
236///
237/// `merge_inputs` is `input_paths` minus the drop-set — the SSTables the K-way
238/// merger will actually read. `dropped_whole` is the FULL drop-set: the SSTables
239/// proven fully expired + overlap-safe, to be reported and reclaimed after
240/// publish.
241///
242/// Degenerate all-dropped guard: the merger requires at least one input, so when
243/// the drop-set is EVERY input (a major compaction of an only-expired input set),
244/// ONE dropped SSTable is ALSO retained in `merge_inputs` (its rows purge to empty
245/// through the normal merge path) so the merger has a source. That retained
246/// SSTable stays in `dropped_whole` too, so the drop is reported honestly and
247/// reclaimed on both surfaces (roborev #1388 Medium): the reclaim loops dedupe
248/// against `merge_inputs` (which the core WriteEngine path deletes anyway) so the
249/// retained SSTable is deleted exactly once. Only the retained SSTable pays a read
250/// cost; the rest are still excluded from the merge.
251///
252/// Shared by both compaction surfaces so the exclusion + all-dropped guard live in
253/// one place (issue #1388).
254pub(crate) fn split_merge_and_dropped(
255    input_paths: &[PathBuf],
256    drop_set: Vec<PathBuf>,
257) -> (Vec<PathBuf>, Vec<PathBuf>) {
258    let dropped_lookup: std::collections::HashSet<&PathBuf> = drop_set.iter().collect();
259    let mut merge_inputs: Vec<PathBuf> = input_paths
260        .iter()
261        .filter(|p| !dropped_lookup.contains(*p))
262        .cloned()
263        .collect();
264    if !merge_inputs.is_empty() {
265        return (merge_inputs, drop_set);
266    }
267    // Everything was dropped: retain one for the merger. It STAYS in the drop-set
268    // so the drop is reported/reclaimed honestly; reclaim sites dedupe against
269    // `merge_inputs` to delete it exactly once (roborev #1388 Medium).
270    if let Some(retained) = drop_set.first() {
271        merge_inputs.push(retained.clone());
272    }
273    (merge_inputs, drop_set)
274}

276/// Reclaim (best-effort component-delete) each SSTable in `dropped_whole` that the
277/// caller is NOT already deleting via another path, calling `delete` on the
278/// survivors. The degenerate all-dropped guard retains one dropped SSTable in BOTH
279/// `dropped_whole` and `merge_inputs` (see [`split_merge_and_dropped`]), so it is
280/// reclaimed exactly once:
281///
282/// - Core WriteEngine surface: it deletes ALL merge inputs separately, so it
283///   passes `already_deleted = merge_inputs` here to skip the retained one (avoid a
284///   double-delete + spurious orphan warning).
285/// - CLI one-shot surface: it deletes NO merge inputs (the operator owns the input
286///   dir; output lands in a separate `--output` dir), so it passes an EMPTY
287///   `already_deleted` and the retained SSTable IS reclaimed here — closing the
288///   former "all-expired input left on disk" gap (roborev #1388 Medium).
289///
290/// `delete` is the surface's own best-effort delete (logs, never errors), so this
291/// helper only decides the set to reclaim.
292pub(crate) fn reclaim_dropped_whole<F: FnMut(&Path)>(
293    dropped_whole: &[PathBuf],
294    already_deleted: &[PathBuf],
295    mut delete: F,
296) {
297    let skip: std::collections::HashSet<&PathBuf> = already_deleted.iter().collect();
298    for dropped in dropped_whole {
299        if skip.contains(dropped) {
300            continue;
301        }
302        delete(dropped);
303    }
304}

493    /// Roborev F2 (former resurrection) regression: the overlap gate compares the
494    /// candidate's TRUE NEWEST write (`max_timestamp`, #1729-authoritative), not its
495    /// `min_timestamp`. A candidate whose `min_timestamp < overlap_bound <
496    /// max_timestamp` (an old tombstone min but a NEWER tombstone max) is RETAINED —
497    /// its newest write does NOT predate the outside data, so dropping it could
498    /// resurrect data. A gate that mistakenly used `min_timestamp` would drop it.
499    #[test]

517    /// Roborev F2 fail-closed: since #1729 a candidate's `max_timestamp` may be the
518    /// `i64::MIN` UNAVAILABLE sentinel. An unknown newest write cannot be proven to
519    /// predate the overlap bound, so the candidate is RETAINED (never dropped) even
520    /// though it is fully expired and its (unavailable) sentinel would compare below
521    /// any real bound. Exercised through the pure classifier so the in-memory
522    /// sentinel is honest (the on-disk StatisticsWriter finalize normalizes
523    /// `i64::MIN → 0`, so it cannot round-trip the sentinel).
524    #[test]
525    fn overlap_gate_fails_closed_on_unavailable_max_timestamp() {
526        let cand: PathBuf = PathBuf::from("nb-1-big-Data.db");
527        // Fully expired (LDT 500 < 1_000) but max_timestamp is the #1729 unavailable
528        // sentinel (i64::MIN). A naive `sentinel < bound` comparison would wrongly
529        // permit a drop; the authoritative accessor forces a retain.
530        let stats = ts_stats(500, i64::MIN);
531        let input_stats = [(&cand, Some(stats))];
532        // Concrete, above-sentinel outside bound (5_000).
533        let dropped = classify_drop_set(&input_stats, Some(5_000), 1_000);
534        assert!(
535            dropped.is_empty(),
536            "an UNAVAILABLE (i64::MIN) max_timestamp must fail closed: retain, never drop (#1729 F2)"
537        );
538
539        // Control: the SAME candidate with a real, below-bound max_timestamp IS
540        // droppable, proving it is the sentinel (not some other field) that retains.
541        let live_max = ts_stats(500, 4_000);
542        let input_stats_live = [(&cand, Some(live_max))];
543        assert_eq!(
544            classify_drop_set(&input_stats_live, Some(5_000), 1_000),
545            vec![cand.clone()],
546            "a concrete below-bound max_timestamp on the same candidate is droppable"
547        );
548    }

605    /// [`reclaim_dropped_whole`] deletes only the drops NOT already deleted by the
606    /// caller: the core surface (which deletes all merge inputs) skips the retained
607    /// one; the CLI surface (empty `already_deleted`) reclaims every drop.
608    #[test]
609    fn reclaim_dedups_against_already_deleted() {
610        let a = PathBuf::from("nb-1-big-Data.db");
611        let b = PathBuf::from("nb-2-big-Data.db");
612        let dropped_whole = [a.clone(), b.clone()];
613
614        // Core surface: `b` is the retained merge input (already deleted) ⇒ only `a`.
615        let mut core_deleted = Vec::new();
616        reclaim_dropped_whole(&dropped_whole, &[b.clone()], |p| {
617            core_deleted.push(p.to_path_buf())
618        });
619        assert_eq!(
620            core_deleted,
621            vec![a.clone()],
622            "core surface skips the retained merge input to avoid a double-delete"
623        );
624
625        // CLI surface: nothing pre-deleted ⇒ both reclaimed here.
626        let mut cli_deleted = Vec::new();
627        reclaim_dropped_whole(&dropped_whole, &[], |p| cli_deleted.push(p.to_path_buf()));
628        assert_eq!(
629            cli_deleted,
630            vec![a, b],
631            "CLI surface reclaims every drop (it deletes no merge inputs otherwise)"
632        );
633    }
