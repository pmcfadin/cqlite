1//! Partition-scoped tombstone-carrier pre-scan (issue #1668, stage 1).
2//!
3//! `merge_partition_rows` (in `mod.rs`) buffers a whole partition and, in a
4//! single pass, splits three PARTITION-LEVEL markers out of the per-`(pk, ck)`
5//! row stream before per-cluster reconciliation:
6//!
7//!   * **range-tombstone carriers** (issue #933) — empty live rows carrying a
8//!     [`RangeTombstone`] spanning a clustering range;
9//!   * **partition-deletion carriers** (issue #1072) — synthetic empty live
10//!     rows whose only payload is `partition_deletion`. The MAX
11//!     `markedForDeleteAt` across sources is the partition's outermost shadow
12//!     floor.
13//!
14//! Per the #933/#846/#959 correctness invariants, these carriers must be
15//! collected across the WHOLE partition BEFORE per-cluster shadowing
16//! (`apply_range_shadowing` / `apply_partition_shadowing`) can run — a single
17//! cluster cannot be shadowed until the full coalesced range set is known.
18//!
19//! This module extracts that carrier collection into a standalone, pure,
20//! independently testable pre-scan ([`scan_partition_carriers`]). It is a
21//! behavior-preserving refactor: the whole-partition buffering path in
22//! `merge_partition_rows` is unchanged and still the only production path — it
23//! now calls this helper for the carrier set, then buffers the remaining
24//! (non-carrier) rows exactly as before. This pre-scan is the seed for the
25//! streaming second pass in later stages (#1668 stages 2+), which will apply
26//! per-cluster shadowing over the coalesced carriers without re-buffering the
27//! whole partition.
28

34/// The partition-scoped tombstone markers extracted from a partition's buffered
35/// merge entries by [`scan_partition_carriers`].
36///
37/// These are the markers that span a clustering range or the whole partition —
38/// they are NOT per-`(pk, ck)` rows and are kept OUT of the clustered-row
39/// stream. `merge_partition_rows` coalesces `range_tombstones`, applies
40/// `max_partition_deletion` as the outermost per-cell shadow floor, and
41/// re-emits the survivors so the writer persists the markers.
42#[cfg(feature = "write-support")]
43#[derive(Debug, Default)]
44pub(super) struct PartitionCarriers {
45    /// Range-tombstone carriers, one `(key, range)` per input marker, in
46    /// first-seen (heap) order. Coalesced into a non-overlapping canonical
47    /// sequence by the caller (issue #933 / #959).
48    pub range_tombstones: Vec<(DecoratedKey, RangeTombstone)>,
49    /// MAX partition deletion `(markedForDeleteAt µs, localDeletionTime s)`
50    /// across all sources — the partition's outermost shadow floor (issue
51    /// #1072). `None` when no partition-deletion carrier is present.
52    pub max_partition_deletion: Option<(i64, i32)>,
53    /// The [`DecoratedKey`] of the first-seen partition-deletion carrier, kept
54    /// so the surviving partition tombstone can be re-emitted (issue #1072).
55    /// `None` when no partition-deletion carrier is present.
56    pub partition_delete_key: Option<DecoratedKey>,
57}
58
59/// True when this entry is a range-tombstone carrier (issue #933): an empty
60/// live row whose only payload is a [`RangeTombstone`] spanning a clustering
61/// range — no cells, no row deletion, no complex deletions. Such a marker is
62/// partition-level (spans many `(pk, ck)`), NOT a single reconcilable row, so
63/// it must be split out of the clustered-row stream.
64#[cfg(feature = "write-support")]
65pub(super) fn is_range_marker_carrier(entry: &MergeEntry) -> bool {
66    entry.range_deletion.is_some()
67        && entry.complex_deletions.is_empty()
68        && entry.row_deletion.is_none()
69        && matches!(&entry.row_data, RowData::Live { cells } if cells.is_empty())
70}

72/// Pre-scan a partition's buffered merge entries and extract the
73/// partition-scoped tombstone carriers ([`PartitionCarriers`]).
74///
75/// This is a light, read-only first pass that touches ONLY the (typically few)
76/// range/partition-tombstone carriers — it borrows `rows` and does not consume
77/// them, so `merge_partition_rows` can still buffer the non-carrier rows into
78/// `clustered_rows` afterward. It reproduces exactly the carrier-splitting logic
79/// that previously lived inline in `merge_partition_rows`:
80///
81///   * A partition-deletion carrier (checked FIRST, matching the original
82///     precedence) contributes its `(mfda, ldt)` to `max_partition_deletion`,
83///     keeping the MAX `markedForDeleteAt` (ties keep the first-seen), and
84///     records the first-seen key in `partition_delete_key`.
85///   * Otherwise a range-marker carrier ([`is_range_marker_carrier`])
86///     contributes its `(key, range)` to `range_tombstones` in first-seen order.
87///   * Every other entry is a normal `(pk, ck)` row and is ignored here (the
88///     caller buffers it into `clustered_rows`).
89#[cfg(feature = "write-support")]
90pub(super) fn scan_partition_carriers(rows: &[MergeEntry]) -> PartitionCarriers {
91    let mut carriers = PartitionCarriers::default();
92
93    for row in rows {
94        // Partition-deletion carriers take precedence over range markers,
95        // exactly as the original inline pass did.
96        if row.is_partition_delete_carrier() {
97            if let Some((mfda, ldt)) = row.partition_deletion {
98                carriers
99                    .partition_delete_key
100                    .get_or_insert_with(|| row.key.clone());
101                match carriers.max_partition_deletion {
102                    Some((cur_mfda, _)) if cur_mfda >= mfda => {}
103                    _ => carriers.max_partition_deletion = Some((mfda, ldt)),
104                }
105                continue;
106            }
107        }
108        if is_range_marker_carrier(row) {
109            if let Some(rt) = row.range_deletion.clone() {
110                carriers.range_tombstones.push((row.key.clone(), rt));
111            }
112        }
113    }
114
115    carriers
116}

118#[cfg(all(test, feature = "write-support"))]
119mod tests {
120    use super::*;
121    use crate::storage::write_engine::merge::CellData;
122    use crate::storage::write_engine::mutation::ClusteringBound;
123    use crate::types::Value;
124
125    fn key(token: i64) -> DecoratedKey {
126        DecoratedKey::new(token, vec![token as u8])
127    }
128
129    fn range_tombstone(deletion_time: i64, local_deletion_time: i32) -> RangeTombstone {
130        RangeTombstone {
131            start: ClusteringBound::Bottom,
132            end: ClusteringBound::Top,
133            deletion_time,
134            local_deletion_time,
135        }
136    }
137
138    /// An ordinary live `(pk, ck)` row — must be ignored by the pre-scan.
139    fn live_row(token: i64) -> MergeEntry {
140        MergeEntry::new(
141            0,
142            key(token),
143            None,
144            100,
145            RowData::Live {
146                cells: vec![CellData::new("c".to_string(), Value::Integer(1), 100)],
147            },
148        )
149    }
150
151    /// A range-tombstone carrier: empty live row + a range deletion.
152    fn range_carrier(token: i64, deletion_time: i64, ldt: i32) -> MergeEntry {
153        MergeEntry::new(
154            0,
155            key(token),
156            None,
157            deletion_time,
158            RowData::Live { cells: Vec::new() },
159        )
160        .with_range_deletion(range_tombstone(deletion_time, ldt))
161    }
162
163    /// A partition-deletion carrier: empty live row + a partition deletion.
164    fn partition_carrier(token: i64, mfda: i64, ldt: i32) -> MergeEntry {
165        MergeEntry::new(
166            0,
167            key(token),
168            None,
169            mfda,
170            RowData::Live { cells: Vec::new() },
171        )
172        .with_partition_deletion((mfda, ldt))
173    }
174
175    #[test]
176    fn empty_input_yields_no_carriers() {
177        let carriers = scan_partition_carriers(&[]);
178        assert!(carriers.range_tombstones.is_empty());
179        assert_eq!(carriers.max_partition_deletion, None);
180        assert!(carriers.partition_delete_key.is_none());
181    }
182
183    #[test]
184    fn plain_rows_yield_no_carriers() {
185        let rows = vec![live_row(1), live_row(1), live_row(1)];
186        let carriers = scan_partition_carriers(&rows);
187        assert!(carriers.range_tombstones.is_empty());
188        assert_eq!(carriers.max_partition_deletion, None);
189        assert!(carriers.partition_delete_key.is_none());
190    }
191
192    #[test]
193    fn extracts_range_carriers_in_first_seen_order() {
194        let rows = vec![
195            range_carrier(1, 200, 20),
196            live_row(1),
197            range_carrier(1, 100, 10),
198        ];
199        let carriers = scan_partition_carriers(&rows);
200        assert_eq!(carriers.range_tombstones.len(), 2);
201        // First-seen (heap) order preserved: 200 before 100.
202        assert_eq!(carriers.range_tombstones[0].1.deletion_time, 200);
203        assert_eq!(carriers.range_tombstones[1].1.deletion_time, 100);
204        // Range carriers do not populate the partition-deletion floor.
205        assert_eq!(carriers.max_partition_deletion, None);
206        assert!(carriers.partition_delete_key.is_none());
207    }
208
209    #[test]
210    fn keeps_max_partition_deletion_across_carriers() {
211        // Out-of-order mfda: 50, then 300 (the max), then 200 — max wins.
212        let rows = vec![
213            partition_carrier(7, 50, 5),
214            live_row(7),
215            partition_carrier(7, 300, 30),
216            partition_carrier(7, 200, 20),
217        ];
218        let carriers = scan_partition_carriers(&rows);
219        assert_eq!(carriers.max_partition_deletion, Some((300, 30)));
220        // First-seen carrier's key is retained.
221        assert_eq!(carriers.partition_delete_key, Some(key(7)));
222        assert!(carriers.range_tombstones.is_empty());
223    }
224
225    #[test]
226    fn equal_mfda_keeps_first_seen_local_deletion_time() {
227        // On an mfda tie the FIRST-seen (mfda, ldt) is kept (>= guard).
228        let rows = vec![partition_carrier(3, 100, 11), partition_carrier(3, 100, 22)];
229        let carriers = scan_partition_carriers(&rows);
230        assert_eq!(carriers.max_partition_deletion, Some((100, 11)));
231    }
232
233    #[test]
234    fn mixed_carriers_and_rows_split_correctly() {
235        let rows = vec![
236            live_row(5),
237            range_carrier(5, 400, 40),
238            partition_carrier(5, 250, 25),
239            live_row(5),
240            range_carrier(5, 150, 15),
241        ];
242        let carriers = scan_partition_carriers(&rows);
243        assert_eq!(carriers.range_tombstones.len(), 2);
244        assert_eq!(carriers.range_tombstones[0].1.deletion_time, 400);
245        assert_eq!(carriers.range_tombstones[1].1.deletion_time, 150);
246        assert_eq!(carriers.max_partition_deletion, Some((250, 25)));
247        assert_eq!(carriers.partition_delete_key, Some(key(5)));
248    }
249
250    #[test]
251    fn is_range_marker_carrier_rejects_non_empty_live_row() {
252        // A row with a range deletion AND live cells is NOT a pure carrier.
253        let entry = live_row(1).with_range_deletion(range_tombstone(100, 10));
254        assert!(!is_range_marker_carrier(&entry));
255        // The pure empty-live carrier is recognized.
256        assert!(is_range_marker_carrier(&range_carrier(1, 100, 10)));
257    }
