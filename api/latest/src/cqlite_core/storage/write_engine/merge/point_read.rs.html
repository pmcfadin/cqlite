1//! Single-partition point-read merge assembly (issue #2207, Stage 1/2 core side).
2//!
3//! Builds a [`KWayMerger`] over ONE run per candidate SSTable, each yielding only
4//! the target partition's entries, so the Flight `do_get` point path reconciles
5//! through the SAME merge as the full scan — byte-identically, only over fewer
6//! partitions (design candidate (c)).
7//!
8//! Per candidate the core primitive
9//! [`SSTableReader::read_single_partition_for_compaction`] decides:
10//! - `DefinitelyAbsent` → the candidate is pruned (the presence oracle already
11//!   incremented `cqlite.read.sstables_pruned`); no run is built.
12//! - `Rows(rows)` → a seeked run yielding exactly those compaction rows.
13//! - `IndexUnavailable` → a fail-safe run that scans this ONE SSTable's whole
14//!   compaction stream and forwards only the target partition's entries — never
15//!   skipping a candidate that might hold the key (#2295 Data.db-only shape).
16//!
17//! Run index (LWW tie-break rank) is the candidate's position in the input
18//! `paths` list, identical to the full-scan merger, so reconciliation ties break
19//! the same way.
20

49impl KWayMerger {
50    /// Build a k-way merger from pre-constructed run iterators (issue #2207).
51    ///
52    /// Unlike [`KWayMerger::new`], which opens each input SSTable and streams its
53    /// WHOLE compaction scan, this accepts runs the caller has already scoped —
54    /// the single-partition point-read path hands one run per candidate SSTable,
55    /// each yielding ONLY the target partition's entries (a seeked `Vec` or a
56    /// key-filtered stream). The reconciliation, heap, and per-partition merge are
57    /// IDENTICAL to the full-scan path — only the inputs are narrower — so the
58    /// point path reconciles byte-identically to the scan.
59    ///
60    /// `runs` must be non-empty and ordered newest-to-oldest (run index = LWW
61    /// tie-break rank), exactly as [`KWayMerger::new`]'s `input_paths` are.
62    pub fn from_row_iterators(
63        runs: Vec<Box<dyn SSTableRowIterator>>,
64        schema: &TableSchema,
65    ) -> Result<Self> {
66        if runs.is_empty() {
67            return Err(Error::InvalidInput(
68                "K-way merge requires at least one input run".to_string(),
69            ));
70        }
71        schema.validate_dropped_columns()?;
72        // Child modules see the parent's private fields; build the merger in the
73        // same shape as `new_with_gc_and_registry_cancellable` (no purge params —
74        // the point path never compacts).
75        let runs = runs.into_iter().map(RunReader::new).collect();
76        Ok(Self {
77            runs,
78            heap: BinaryHeap::new(),
79            current_partition: None,
80            schema: schema.clone(),
81            // Issue #1668, stage 5c-i: `Arc`-wrapped clone of `schema`, used
82            // only by the heap's schema-aware comparator (see the field doc).
83            // (Semantic-merge fix: #1668 landed on main between #2207's branch
84            // point and its merge; this constructor predates the field.)
85            schema_arc: std::sync::Arc::new(schema.clone()),
86            gc_before_secs: None,
87            now_secs: None,
88            purge_safe: false,
89            max_purgeable_timestamp: None,
90            // Issue #2765: the point-read builders open their fail-safe adapters
91            // (and snapshot the shared capacity) OUTSIDE this constructor, then
92            // attach the matching slot guard via `with_egress_slot`. A merger
93            // built directly from pre-supplied runs registers no slot.
94            _egress_slot: None,
95        })
96    }
97}

99/// A run backed by a pre-built `Vec` of merge entries — the seeked single
100/// partition. Yields each entry once, in order, then reports exhaustion.
101///
102/// Only the default `not(tombstones)` build seeks (the alternate build always
103/// scan-falls-back), so this is dead there.
104#[cfg_attr(feature = "tombstones", allow(dead_code))]
105struct VecRun {
106    entries: VecDeque<MergeEntry>,
107}
108
109impl SSTableRowIterator for VecRun {
110    fn next(&mut self) -> Option<Result<MergeEntry>> {
111        self.entries.pop_front().map(Ok)
112    }
113}

115/// A run that forwards ONLY the target partitions' entries from an SSTable's full
116/// compaction stream — the index-unavailable fail-safe. Byte-identical to the
117/// full scan's contribution for those partitions (same stream, same decode), just
118/// with every other partition dropped before the merge sees it.
119struct SinglePartitionFilterRun {
120    inner: SSTableRowIteratorAdapter,
121    keys: std::collections::HashSet<Vec<u8>>,
122}
123
124impl SSTableRowIterator for SinglePartitionFilterRun {
125    /// Issue #2765: delegate the egress-capacity observation to the wrapped
126    /// adapter so the point-read fail-safe channel is observable by wiring tests.
127    #[cfg(test)]
128    fn egress_channel_capacity(&self) -> Option<usize> {
129        self.inner.egress_channel_capacity()
130    }
131
132    /// Issue #2820: delegate the ROW snapshot too, so the wiring test can assert
133    /// the ROWS→MESSAGES conversion on this seam as well as on the two full-scan
134    /// construction sites.
135    #[cfg(test)]
136    fn egress_rows_capacity(&self) -> Option<usize> {
137        self.inner.egress_rows_capacity()
138    }
139
140    fn next(&mut self) -> Option<Result<MergeEntry>> {
141        loop {
142            match self.inner.next() {
143                Some(Ok(entry)) => {
144                    if self.keys.contains(&entry.key.key) {
145                        return Some(Ok(entry));
146                    }
147                    // A non-target partition: drop it and pull the next entry.
148                }
149                // An error or exhaustion propagates unchanged (a cancelled scan
150                // stays a distinct `Error::Cancelled`, issue #2264).
151                other => return other,
152            }
153        }
154    }
155}

157/// What one candidate SSTable reported about ONE requested key's on-disk size —
158/// the byte-weight half of the partition access-distribution probe (issue #2827).
159///
160/// The probe sites supply byte weights ONLY; they never count accesses. Counting
161/// here would multiply a partition's repeat count by the number of generations
162/// holding it and manufacture concentration the workload does not have, which is a
163/// bias toward "build the cache". The single access is recorded once per logical
164/// point read by the builder below.
165/// Only the default `not(tombstones)` seek path resolves per-key sizes (the
166/// alternate build always falls back to a whole-file scan and produces no notes),
167/// so the variants are dead there — same shape as [`PathProbe`].
168#[cfg_attr(feature = "tombstones", allow(dead_code))]
169#[derive(Clone, Copy, Debug, PartialEq, Eq)]
170enum KeySizeNote {
171    /// This SSTable does not hold the key, so it contributes nothing either way.
172    NotHeld,
173    /// This SSTable holds the key and its index recorded an authoritative on-disk
174    /// size for it.
175    Sized(u32),
176    /// This SSTable holds the key and its on-disk extent was MEASURED as the
177    /// partition's successor gap.
178    Measured(u64),
179    /// This SSTable holds (or may hold) the key but no authoritative extent is
180    /// available — e.g. a fail-safe whole-file scan resolves no per-partition
181    /// layout, or no data-section length is known. Never estimated.
182    Unsized,
183}

185/// Whether a merger builder records the #2827 partition accesses itself.
186///
187/// The reader-based builder has TWO callers with opposite needs, so this cannot be a
188/// property of the function:
189///
190/// - the **core executor's** multi-generation targeted read reaches it through
191///   `generation_merge::seek_merge_generations_for_read`, and the executor already
192///   records the access at its own logical boundary
193///   (`StorageEngine::scan_partition_clustering`) — recording here too would count
194///   one logical read twice;
195/// - the **Flight warm point path** reaches it directly, and this IS its logical
196///   point-read boundary, so recording must happen here or not at all.
197///
198/// Counting stays once per logical partition read either way. Nothing is ever
199/// recorded at a per-SSTable probe site, which would multiply a repeat count by the
200/// number of generations holding the key.
201#[derive(Clone, Copy, Debug, PartialEq, Eq)]
202pub enum PointAccessRecording {
203    /// This call is the logical point-read boundary: record one access per key.
204    Record,
205    /// The caller records at its own logical boundary; do not record here.
206    CallerRecords,
207}

209/// A candidate probe plus, when the probe is enabled, one [`KeySizeNote`] per
210/// requested key in the requested order.
211struct ProbeOutcome {
212    probe: PathProbe,
213    /// Empty when the probe is disabled — the default, so this costs nothing.
214    notes: Vec<KeySizeNote>,
215}

217/// Per-candidate probe outcome across ALL requested keys. `Seeked`/`Empty` are
218/// produced only by the default `not(tombstones)` seek path (the alternate build
219/// always returns `NeedsScan`), so they are dead under `tombstones`.
220#[cfg_attr(feature = "tombstones", allow(dead_code))]
221enum PathProbe {
222    /// No requested key is present in this SSTable (all definite-absent / empty).
223    Empty,
224    /// The seeked compaction rows for every present key (combined; same generation
225    /// so they share one run index).
226    Seeked(Vec<CompactionRow>),
227    /// This SSTable has no usable index — scan it, filtered to the requested keys.
228    NeedsScan,
229}

231/// Assemble a [`KWayMerger`] that reads ONLY the target `keys` across the
232/// candidate `paths` (issue #2207).
233///
234/// Returns `Ok(None)` when no candidate holds any target key (every one a definite
235/// presence-oracle negative / prefix-collision) — the caller streams zero rows.
236/// Otherwise returns a merger the caller drives through its existing reconciliation
237/// loop; it emits one `MergeStep::Partition` per distinct present key, reconciled
238/// across generations exactly as the full-scan merge would.
239///
240/// `paths` must be the same (token-pruned) candidate list, in the same order, the
241/// full-scan path would merge — run index is the position here, so LWW tie-breaks
242/// match. `keys` are raw partition-key bytes (as `PartitionKey::to_bytes`
243/// produces). `scan_cancel` is threaded as a PER-CALL token into every probe/scan
244/// (issue #2346) so a client disconnect abandons an in-flight fail-safe scan
245/// promptly (#2264).
246pub fn build_single_partition_merger(
247    paths: Vec<PathBuf>,
248    keys: &[Vec<u8>],
249    schema: &TableSchema,
250    scan_cancel: ScanCancel,
251) -> Result<Option<KWayMerger>> {
252    build_single_partition_merger_with_registry(paths, keys, schema, None, scan_cancel)
253}

255/// Like [`build_single_partition_merger`], but threads an authoritative
256/// [`UdtRegistry`] onto every candidate reader (issue #2349) so a `frozen<UDT>`
257/// cell inside a collection decodes structurally instead of as opaque bytes — the
258/// cold point-read analogue of the full-scan
259/// [`KWayMerger::new_with_gc_and_registry_cancellable`]. Passing `None` is
260/// byte-identical to the registry-free path. Used by the Flight point-read route so
261/// point and full-scan reads never diverge (issue #1918 differential lane).
262pub fn build_single_partition_merger_with_registry(
263    paths: Vec<PathBuf>,
264    keys: &[Vec<u8>],
265    schema: &TableSchema,
266    udt_registry: Option<&UdtRegistry>,
267    scan_cancel: ScanCancel,
268) -> Result<Option<KWayMerger>> {
269    schema.validate_dropped_columns()?;
270    if keys.is_empty() {
271        return Ok(None);
272    }
273
274    // Canonicalize the requested key set (roborev suggestion, issue #2207):
275    // dedup identical raw keys — a `pk IN (5, 5)` ticket must not double-seek the
276    // same partition — and sort by Murmur3 token, the natural probing order
277    // (matches the ascending order `MergeEntry::Ord` requires each run to
278    // yield). Correctness does not depend on this order (each candidate's
279    // combined entries are sorted again below before wrapping in `VecRun`), but
280    // canonicalizing up front avoids redundant work and reads more naturally.
281    let mut canonical: Vec<Vec<u8>> = keys.to_vec();
282    canonical.sort_by_key(|k| crate::util::cassandra_murmur3::cassandra_murmur3_token(k));
283    canonical.dedup();
284    let keys: &[Vec<u8>] = &canonical;
285
286    let key_set: std::collections::HashSet<Vec<u8>> = keys.iter().cloned().collect();
287
288    // Issue #2765 (LAZY registration): a point read whose candidates ALL seek
289    // (`PathProbe::Seeked` → in-memory `VecRun`s, ZERO egress channels) buffers
290    // NOTHING, so it must NOT occupy an active-merge slot — else high-QPS Flight
291    // point reads would inflate the process-global count and throttle a
292    // concurrent channel-backed compaction/full-scan toward `MIN_CAP`. Defer
293    // `begin_merge` to the FIRST `NeedsScan` egress channel and memoize its
294    // snapshot so every channel in THIS merge still shares ONE capacity.
295    let mut egress: Option<(usize, egress_budget::ActiveMergeGuard)> = None;
296
297    let mut runs: Vec<Box<dyn SSTableRowIterator>> = Vec::new();
298    // Byte weights for the partition access-distribution probe (issue #2827), one
299    // accumulator per requested key, summed across the candidates that resolved it.
300    // Allocated only when the probe is enabled (it is OFF by default).
301    // The path-based builder IS the cold Flight point path's logical boundary, so it
302    // always records. Extent measurement is nonetheless gated on the probe being on:
303    // it is real work (an index materialize + a successor resolution per key per
304    // generation) and must never be paid by a process that asked for no telemetry.
305    let collect = partition_access::enabled();
306    let mut weights: Vec<partition_access::AccessWeightBuilder> = if collect {
307        vec![partition_access::AccessWeightBuilder::new(); keys.len()]
308    } else {
309        Vec::new()
310    };
311    for (run_index, path) in paths.iter().enumerate() {
312        // Cooperative cancellation (#2264): honour a cancel BEFORE each candidate's
313        // seek/open, so a disconnected point read does no further per-SSTable work.
314        scan_cancel.check()?;
315
316        let ProbeOutcome { probe, notes } = probe_path(
317            path,
318            keys,
319            schema,
320            udt_registry,
321            scan_cancel.clone(),
322            collect,
323        )?;
324        fold_size_notes(&mut weights, &notes);
325        match probe {
326            PathProbe::Empty => {
327                // Pruned / absent for every key. No run.
328            }
329            PathProbe::Seeked(rows) => {
330                if rows.is_empty() {
331                    continue;
332                }
333                let mut entries: Vec<MergeEntry> = Vec::with_capacity(rows.len());
334                for row in rows {
335                    // Issue #2096: one merge entry decoded from `Data.db` by the
336                    // SEEK path (this candidate held the target partition). Counting
337                    // per built entry — the same unit the full-scan run counts per
338                    // `Ok(entry)` — keeps `merge_run_entries_decoded` apples-to-
339                    // apples between the two paths.
340                    crate::storage::sstable::work_counters::add_merge_run_entry_decoded();
341                    entries.push(SSTableRowIteratorAdapter::build_merge_entry(
342                        run_index, row, schema,
343                    )?);
344                }
345                // BLOCKER fix (roborev, issue #2207): `rows` was accumulated in
346                // REQUESTED-KEY order (IN-list / Or order), not token order. Every
347                // `SSTableRowIterator` MUST yield entries in ascending `MergeEntry`
348                // order (token, key, clustering) — `refill_heap` buffers only ONE
349                // entry per run at a time and relies on that invariant (mod.rs
350                // `step`/`refill_heap`). An out-of-order run causes `step()` to
351                // silently split one run's contribution across two heap pops,
352                // duplicating rows and breaking cross-generation reconciliation
353                // (a newer generation's overwrite/tombstone no longer shadows the
354                // older one). Sort once, here, before wrapping in `VecRun`.
355                entries.sort();
356                runs.push(Box::new(VecRun {
357                    entries: entries.into(),
358                }));
359            }
360            PathProbe::NeedsScan => {
361                // Fail-safe: scan this ONE SSTable, forwarding only the target
362                // partitions. The missing index costs speed, never correctness.
363                // Snapshot the shared capacity on the FIRST egress channel only.
364                let channel_capacity = egress.get_or_insert_with(egress_budget::begin_merge).0;
365                let adapter = SSTableRowIteratorAdapter::open(
366                    path,
367                    run_index,
368                    schema,
369                    udt_registry.cloned(),
370                    scan_cancel.clone(),
371                    channel_capacity,
372                    // Issue #1704: no recording seam encloses this fail-safe scan, so
373                    // a failed input open must still report itself. See
374                    // `merge::constructors`' module doc for why the caller states it.
375                    crate::storage::sstable::reader::OpenErrorReporting::SelfReported,
376                )?;
377                runs.push(Box::new(SinglePartitionFilterRun {
378                    inner: adapter,
379                    keys: key_set.clone(),
380                }));
381            }
382        }
383    }
384
385    // ONE access per LOGICAL partition read, recorded after every candidate has
386    // reported (issue #2827). This is the point-read path's logical boundary: the
387    // key list is canonical (deduplicated), so `pk IN (5, 5)` is one access, and a
388    // key present in seven generations is still one access carrying the SUM of the
389    // per-generation on-disk sizes. Recorded even when no candidate held the key —
390    // a miss is a real access, and omitting it would understate the singleton
391    // bucket. A no-op when the probe is disabled.
392    record_logical_accesses(schema, keys, weights);
393
394    if runs.is_empty() {
395        // No candidate produced a run; any snapshot guard drops here.
396        return Ok(None);
397    }
398    let merger = KWayMerger::from_row_iterators(runs, schema)?;
399    // Attach the slot ONLY if a `NeedsScan` egress channel was actually opened;
400    // an all-`Seeked` (channel-less) point read registers no slot.
401    Ok(Some(match egress {
402        Some((_, guard)) => merger.with_egress_slot(guard),
403        None => merger,
404    }))
405}

407/// Reader-based analogue of [`build_single_partition_merger`] (issue #2346):
408/// assembles a [`KWayMerger`] that reads ONLY the target `keys`, but across
409/// already-open, possibly-SHARED `readers` instead of `paths` — the seam a
410/// future cached-reader caller (e.g. a Flight warm-handle registry) uses to
411/// avoid a fresh per-request reader-open/Index/Summary/Statistics/bloom parse.
412///
413/// Semantics mirror `build_single_partition_merger` exactly (same
414/// canonicalization, same three-way per-candidate probe via [`probe_reader`],
415/// same [`SinglePartitionFilterRun`] fail-safe, same run-index ordering) — only
416/// WHO opens/owns the `SSTableReader` differs. `readers` must be the same
417/// (token-pruned) candidate list, in the same order, the full-scan reader-based
418/// merger ([`KWayMerger::new_from_readers`]) would merge.
419/// `recording` (issue #2827) governs ONLY whether this call records one partition
420/// access per requested key; it changes no rows, no ordering and no probe behaviour.
421/// It is an explicit caller decision because this builder's two callers sit at
422/// different levels — see [`PointAccessRecording`].
423pub fn build_single_partition_merger_from_readers(
424    readers: Vec<Arc<SSTableReader>>,
425    keys: &[Vec<u8>],
426    schema: &TableSchema,
427    scan_cancel: ScanCancel,
428    recording: PointAccessRecording,
429) -> Result<Option<KWayMerger>> {
430    schema.validate_dropped_columns()?;
431    if keys.is_empty() {
432        return Ok(None);
433    }
434
435    let mut canonical: Vec<Vec<u8>> = keys.to_vec();
436    canonical.sort_by_key(|k| crate::util::cassandra_murmur3::cassandra_murmur3_token(k));
437    canonical.dedup();
438    let keys: &[Vec<u8>] = &canonical;
439
440    let key_set: std::collections::HashSet<Vec<u8>> = keys.iter().cloned().collect();
441
442    // Issue #2765 (LAZY registration, see the path-based builder above): occupy an
443    // active-merge slot ONLY when a `NeedsScan` egress channel is actually opened;
444    // an all-`Seeked` point read buffers nothing and registers no slot. The
445    // snapshot is memoized so every channel in THIS merge shares ONE capacity.
446    let mut egress: Option<(usize, egress_budget::ActiveMergeGuard)> = None;
447
448    // `CallerRecords` must measure NOTHING. Measuring an extent costs an index
449    // materialize plus an O(N) `Index.db` successor scan per key per generation, and
450    // under `CallerRecords` the result is discarded and re-derived by the caller —
451    // a per-read O(N) scan whose output is thrown away.
452    let collect = recording == PointAccessRecording::Record && partition_access::enabled();
453    let mut weights: Vec<partition_access::AccessWeightBuilder> = if collect {
454        vec![partition_access::AccessWeightBuilder::new(); keys.len()]
455    } else {
456        Vec::new()
457    };
458
459    let mut runs: Vec<Box<dyn SSTableRowIterator>> = Vec::new();
460    for (run_index, reader) in readers.into_iter().enumerate() {
461        // Cooperative cancellation (#2264): honour a cancel BEFORE each candidate's
462        // probe, so a disconnected point read does no further per-SSTable work.
463        scan_cancel.check()?;
464
465        let ProbeOutcome { probe, notes } = probe_reader(
466            Arc::clone(&reader),
467            keys,
468            schema,
469            scan_cancel.clone(),
470            collect,
471        )?;
472        fold_size_notes(&mut weights, &notes);
473        match probe {
474            PathProbe::Empty => {
475                // Pruned / absent for every key. No run.
476            }
477            PathProbe::Seeked(rows) => {
478                if rows.is_empty() {
479                    continue;
480                }
481                let mut entries: Vec<MergeEntry> = Vec::with_capacity(rows.len());
482                for row in rows {
483                    // Issue #2096: one merge entry decoded from `Data.db` by the
484                    // SEEK path (this candidate held the target partition), same
485                    // accounting as the path-based builder above.
486                    crate::storage::sstable::work_counters::add_merge_run_entry_decoded();
487                    entries.push(SSTableRowIteratorAdapter::build_merge_entry(
488                        run_index, row, schema,
489                    )?);
490                }
491                // See `build_single_partition_merger`'s identical sort — every
492                // `SSTableRowIterator` MUST yield ascending `MergeEntry` order.
493                entries.sort();
494                runs.push(Box::new(VecRun {
495                    entries: entries.into(),
496                }));
497            }
498            PathProbe::NeedsScan => {
499                // Fail-safe: scan this ONE shared reader's compaction stream,
500                // forwarding only the target partitions — never a fresh
501                // path-based open (issue #2346's whole point). `reader` (the
502                // ORIGINAL, not the clone `probe_reader` consumed above) is
503                // still available here.
504                // Point-read fail-safe: a specific-key filter, not a range scan —
505                // no token bound is pushed (issue #2412; the key set bounds it).
506                // Snapshot the shared capacity on the FIRST egress channel only.
507                let channel_capacity = egress.get_or_insert_with(egress_budget::begin_merge).0;
508                let adapter = SSTableRowIteratorAdapter::open_from_reader(
509                    reader,
510                    run_index,
511                    schema,
512                    scan_cancel.clone(),
513                    None,
514                    channel_capacity,
515                )?;
516                runs.push(Box::new(SinglePartitionFilterRun {
517                    inner: adapter,
518                    keys: key_set.clone(),
519                }));
520            }
521        }
522    }
523
524    // ONE access per LOGICAL partition read, when this call is the boundary
525    // (issue #2827). Recorded even when no candidate held the key — a miss is a real
526    // access, and omitting it would understate the singleton bucket.
527    if collect {
528        record_logical_accesses(schema, keys, weights);
529    }
530
531    if runs.is_empty() {
532        // No candidate produced a run; any snapshot guard drops here.
533        return Ok(None);
534    }
535    let merger = KWayMerger::from_row_iterators(runs, schema)?;
536    // Attach the slot ONLY if a `NeedsScan` egress channel was actually opened.
537    Ok(Some(match egress {
538        Some((_, guard)) => merger.with_egress_slot(guard),
539        None => merger,
540    }))
541}

543/// Probe an ALREADY-OPEN `reader` for every requested key via the core seek
544/// primitive (default `not(tombstones)` build) — the shared core BOTH
545/// [`probe_path`] (path-based, opens its own reader then delegates here) and
546/// [`build_single_partition_merger_from_readers`] (reader-based) use, so the
547/// two never diverge (issue #2346).
548///
549/// A single SSTable's index availability is a property of the SSTable (not the
550/// key), so the FIRST `IndexUnavailable` short-circuits to [`PathProbe::NeedsScan`]
551/// (scan the whole file, filtered). Otherwise every present key's seeked rows are
552/// concatenated (they share this SSTable's generation / run index).
553#[cfg(not(feature = "tombstones"))]
554async fn probe_reader_async(
555    reader: &SSTableReader,
556    keys: &[Vec<u8>],
557    schema: &TableSchema,
558    scan_cancel: &ScanCancel,
559    collect_notes: bool,
560) -> Result<ProbeOutcome> {
561    let mut notes: Vec<KeySizeNote> = Vec::new();
562    let mut rows: Vec<CompactionRow> = Vec::new();
563    for key in keys {
564        match reader
565            .read_single_partition_for_compaction(key, Some(schema), scan_cancel)
566            .await?
567        {
568            SinglePartitionCompaction::DefinitelyAbsent => {
569                if collect_notes {
570                    notes.push(KeySizeNote::NotHeld);
571                }
572            }
573            SinglePartitionCompaction::Rows(mut r) => {
574                if collect_notes {
575                    notes.push(resolved_size_note(reader, key).await);
576                }
577                rows.append(&mut r)
578            }
579            // Index availability is per-SSTable, not per-key: fall back to
580            // scanning the whole file once, filtered to the key set.
581            SinglePartitionCompaction::IndexUnavailable => {
582                return Ok(ProbeOutcome {
583                    probe: PathProbe::NeedsScan,
584                    // A whole-file fail-safe scan resolves no per-partition size,
585                    // so EVERY key this SSTable might hold is unpriceable through
586                    // it. Reported as unavailable rather than skipped, so the
587                    // window's byte total is visibly incomplete instead of quietly
588                    // short (issue #2827, design D6).
589                    notes: if collect_notes {
590                        vec![KeySizeNote::Unsized; keys.len()]
591                    } else {
592                        Vec::new()
593                    },
594                });
595            }
596        }
597    }
598    let probe = if rows.is_empty() {
599        PathProbe::Empty
600    } else {
601        PathProbe::Seeked(rows)
602    };
603    Ok(ProbeOutcome { probe, notes })
604}

606/// This SSTable's authoritative on-disk extent for `key`.
607///
608/// The partition's location comes from what the read that just ran already resolved
609/// into the process-global key→partition-offset cache (issue #2059's
610/// `PartitionLoc`), so the instrument never re-drives a lookup and can never
611/// perturb the read path's own telemetry. The EXTENT is then measured as the
612/// successor gap — `[data_offset, successor_offset)`, bounding to the authoritative
613/// uncompressed data-section length for the last partition — which is the same
614/// authoritative index-layout bound the single-partition seek uses to size its
615/// decompression window.
616///
617/// Nothing is ever estimated. Where no extent is resolvable the note is
618/// [`KeySizeNote::Unsized`]: the access is still counted, and it contributes zero
619/// bytes under `size_source = unavailable`.
620///
621/// A key-cache MISS is NOT treated as "no extent". This function is only reached
622/// when the probe returned rows for the key, so the partition IS held; the offset is
623/// then resolved authoritatively, exactly as the core path's resolver does. Pricing
624/// the same partition differently on the two paths — measured on one, unpriceable on
625/// the other — would make a window's byte total depend on which executor served the
626/// read.
627#[cfg(not(feature = "tombstones"))]
628async fn resolved_size_note(reader: &SSTableReader, key: &[u8]) -> KeySizeNote {
629    // An index-recorded size, if one ever exists. No Cassandra 5.0 index records
630    // one, so in practice this never fires.
631    if let Some(loc) = reader.key_cache_get(key) {
632        if loc.data_size > 0 {
633            return KeySizeNote::Sized(loc.data_size);
634        }
635        if let Ok(Some(gap)) = reader.measure_partition_extent(loc.data_offset, key).await {
636            return KeySizeNote::Measured(gap);
637        }
638        return KeySizeNote::Unsized;
639    }
640    // Cache miss on a partition the read just returned rows for: resolve the offset
641    // rather than declaring it unpriceable.
642    use crate::storage::sstable::reader::partition_successor::PartitionResidency;
643    match reader.partition_residency(key).await {
644        PartitionResidency::HeldAt(offset) => {
645            match reader.measure_partition_extent(offset, key).await {
646                Ok(Some(gap)) => KeySizeNote::Measured(gap),
647                Ok(None) | Err(_) => KeySizeNote::Unsized,
648            }
649        }
650        // `NotHeld` is unreachable here (the read returned rows for this key) and
651        // `Unknown` is genuinely unpriceable; both fail closed.
652        PartitionResidency::NotHeld | PartitionResidency::Unknown => KeySizeNote::Unsized,
653    }
654}

656/// Fold one candidate's per-key size notes into the per-key weight accumulators.
657///
658/// A no-op when the probe is disabled (both vectors are empty).
659fn fold_size_notes(weights: &mut [partition_access::AccessWeightBuilder], notes: &[KeySizeNote]) {
660    for (w, note) in weights.iter_mut().zip(notes.iter()) {
661        match note {
662            KeySizeNote::NotHeld => {}
663            KeySizeNote::Sized(n) => w.note_sized(*n),
664            KeySizeNote::Measured(n) => w.note_measured(*n),
665            KeySizeNote::Unsized => w.note_unsized(),
666        }
667    }
668}

670/// Record ONE logical partition access per requested key (issue #2827).
671///
672/// A no-op when the probe is disabled — `weights` is empty then, so this does not
673/// even iterate.
674fn record_logical_accesses(
675    schema: &TableSchema,
676    keys: &[Vec<u8>],
677    weights: Vec<partition_access::AccessWeightBuilder>,
678) {
679    // The table is part of the entry identity — see `TableScope`. The schema is the
680    // authority here (the builders are driven per-table), so no formatting or
681    // allocation is needed on this path.
682    let scope = partition_access::TableScope::new(&schema.keyspace, &schema.table);
683    for (key, weight) in keys.iter().zip(weights) {
684        partition_access::record_partition_access(scope, key, weight.finish());
685    }
686}

688/// Open `path` ONCE and probe it for every requested key, delegating to
689/// [`probe_reader_async`] (issue #2346) once the reader is open.
690///
691/// Runs the async open + seek on the shared bridge runtime. The reader is opened
692/// with buffered I/O (never mmap/direct — the file may be deleted after the
693/// read), matching the full-scan producer's file-lifetime contract (issue #591).
694#[cfg(not(feature = "tombstones"))]
695fn probe_path(
696    path: &Path,
697    keys: &[Vec<u8>],
698    schema: &TableSchema,
699    udt_registry: Option<&UdtRegistry>,
700    scan_cancel: ScanCancel,
701    collect_notes: bool,
702) -> Result<ProbeOutcome> {
703    let path = path.to_path_buf();
704    let schema = schema.clone();
705    let keys: Vec<Vec<u8>> = keys.to_vec();
706    let udt_registry = udt_registry.cloned();
707    block_on_async(async move {
708        let mut config = Config::default();
709        config.storage.use_mmap = false;
710        config.storage.disk_access_mode = DiskAccessMode::Buffered;
711        let platform = Arc::new(Platform::new(&config).await?);
712        let mut reader = SSTableReader::open(&path, &config, platform).await?;
713        // Issue #2349: decode a `frozen<UDT>`-in-collection cell structurally on the
714        // point-read seek path too, matching the full-scan cold reader.
715        if let Some(registry) = udt_registry {
716            reader.set_udt_registry(registry);
717        }
718        probe_reader_async(&reader, &keys, &schema, &scan_cancel, collect_notes).await
719    })
720}

722/// Probe an ALREADY-OPEN, possibly-SHARED `reader` (issue #2346) — the
723/// reader-based analogue of [`probe_path`], used by
724/// [`build_single_partition_merger_from_readers`]. `reader` is an `Arc` clone
725/// (the caller keeps the original for the `NeedsScan` fail-safe path), moved
726/// into the bridged future so no borrow needs to outlive this call.
727#[cfg(not(feature = "tombstones"))]
728fn probe_reader(
729    reader: Arc<SSTableReader>,
730    keys: &[Vec<u8>],
731    schema: &TableSchema,
732    scan_cancel: ScanCancel,
733    collect_notes: bool,
734) -> Result<ProbeOutcome> {
735    let schema = schema.clone();
736    let keys: Vec<Vec<u8>> = keys.to_vec();
737    block_on_async(async move {
738        probe_reader_async(&reader, &keys, &schema, &scan_cancel, collect_notes).await
739    })
740}

742/// Tombstones-build fallback: the single-partition SEEK machinery
743/// (`successor_partition_offset`, `point_read_whole_section`, …) is
744/// `not(tombstones)` only, so under the alternate `tombstones` feature every
745/// candidate degrades to a full scan filtered to the key set — correct, just
746/// without the seek speedup. Keeps the public builder API identical across builds.
747#[cfg(feature = "tombstones")]
748fn probe_path(
749    _path: &Path,
750    _keys: &[Vec<u8>],
751    _schema: &TableSchema,
752    _udt_registry: Option<&UdtRegistry>,
753    _scan_cancel: ScanCancel,
754    _collect_notes: bool,
755) -> Result<ProbeOutcome> {
756    Ok(ProbeOutcome {
757        probe: PathProbe::NeedsScan,
758        notes: Vec::new(),
759    })
760}
