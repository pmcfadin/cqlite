1//! Single-partition point-read merge assembly (issue #2207, Stage 1/2 core side).
2//!
3//! Builds a [`KWayMerger`] over ONE run per candidate SSTable, each yielding only
4//! the target partition's entries, so the Flight `do_get` point path reconciles
5//! through the SAME merge as the full scan — byte-identically, only over fewer
6//! partitions (design candidate (c)).
7//!
8//! Per candidate the core primitive
9//! [`SSTableReader::read_single_partition_for_compaction`] decides:
10//! - `DefinitelyAbsent` → the candidate is pruned (the presence oracle already
11//!   incremented `cqlite.read.sstables_pruned`); no run is built.
12//! - `Rows(rows)` → a seeked run yielding exactly those compaction rows.
13//! - `IndexUnavailable` → a fail-safe run that scans this ONE SSTable's whole
14//!   compaction stream and forwards only the target partition's entries — never
15//!   skipping a candidate that might hold the key (#2295 Data.db-only shape).
16//!
17//! Run index (LWW tie-break rank) is the candidate's position in the input
18//! `paths` list, identical to the full-scan merger, so reconciliation ties break
19//! the same way.
20

44impl KWayMerger {
45    /// Build a k-way merger from pre-constructed run iterators (issue #2207).
46    ///
47    /// Unlike [`KWayMerger::new`], which opens each input SSTable and streams its
48    /// WHOLE compaction scan, this accepts runs the caller has already scoped —
49    /// the single-partition point-read path hands one run per candidate SSTable,
50    /// each yielding ONLY the target partition's entries (a seeked `Vec` or a
51    /// key-filtered stream). The reconciliation, heap, and per-partition merge are
52    /// IDENTICAL to the full-scan path — only the inputs are narrower — so the
53    /// point path reconciles byte-identically to the scan.
54    ///
55    /// `runs` must be non-empty and ordered newest-to-oldest (run index = LWW
56    /// tie-break rank), exactly as [`KWayMerger::new`]'s `input_paths` are.
57    pub fn from_row_iterators(
58        runs: Vec<Box<dyn SSTableRowIterator>>,
59        schema: &TableSchema,
60    ) -> Result<Self> {
61        if runs.is_empty() {
62            return Err(Error::InvalidInput(
63                "K-way merge requires at least one input run".to_string(),
64            ));
65        }
66        schema.validate_dropped_columns()?;
67        // Child modules see the parent's private fields; build the merger in the
68        // same shape as `new_with_gc_and_registry_cancellable` (no purge params —
69        // the point path never compacts).
70        let runs = runs.into_iter().map(RunReader::new).collect();
71        Ok(Self {
72            runs,
73            heap: BinaryHeap::new(),
74            current_partition: None,
75            schema: schema.clone(),
76            // Issue #1668, stage 5c-i: `Arc`-wrapped clone of `schema`, used
77            // only by the heap's schema-aware comparator (see the field doc).
78            // (Semantic-merge fix: #1668 landed on main between #2207's branch
79            // point and its merge; this constructor predates the field.)
80            schema_arc: std::sync::Arc::new(schema.clone()),
81            gc_before_secs: None,
82            now_secs: None,
83            purge_safe: false,
84            max_purgeable_timestamp: None,
85            // Issue #2765: the point-read builders open their fail-safe adapters
86            // (and snapshot the shared capacity) OUTSIDE this constructor, then
87            // attach the matching slot guard via `with_egress_slot`. A merger
88            // built directly from pre-supplied runs registers no slot.
89            _egress_slot: None,
90            trace: NoTrace,
91        })
92    }
93}

95/// A run that forwards ONLY the target partitions' entries from an SSTable's full
96/// compaction stream — the index-unavailable fail-safe. Byte-identical to the
97/// full scan's contribution for those partitions (same stream, same decode), just
98/// with every other partition dropped before the merge sees it.
99struct SinglePartitionFilterRun {
100    inner: SSTableRowIteratorAdapter,
101    keys: std::collections::HashSet<Vec<u8>>,
102}
103
104impl SSTableRowIterator for SinglePartitionFilterRun {
105    /// Issue #2765: delegate the egress-capacity observation to the wrapped
106    /// adapter so the point-read fail-safe channel is observable by wiring tests.
107    #[cfg(test)]
108    fn egress_channel_capacity(&self) -> Option<usize> {
109        self.inner.egress_channel_capacity()
110    }
111
112    /// Issue #2820: delegate the ROW snapshot too, so the wiring test can assert
113    /// the ROWS→MESSAGES conversion on this seam as well as on the two full-scan
114    /// construction sites.
115    #[cfg(test)]
116    fn egress_rows_capacity(&self) -> Option<usize> {
117        self.inner.egress_rows_capacity()
118    }
119
120    fn next(&mut self) -> Option<Result<MergeEntry>> {
121        loop {
122            match self.inner.next() {
123                Some(Ok(entry)) => {
124                    if self.keys.contains(&entry.key.key) {
125                        return Some(Ok(entry));
126                    }
127                    // A non-target partition: drop it and pull the next entry.
128                }
129                // An error or exhaustion propagates unchanged (a cancelled scan
130                // stays a distinct `Error::Cancelled`, issue #2264).
131                other => return other,
132            }
133        }
134    }
135}

137/// What one candidate SSTable reported about ONE requested key's on-disk size —
138/// the byte-weight half of the partition access-distribution probe (issue #2827).
139///
140/// The probe sites supply byte weights ONLY; they never count accesses. Counting
141/// here would multiply a partition's repeat count by the number of generations
142/// holding it and manufacture concentration the workload does not have, which is a
143/// bias toward "build the cache". The single access is recorded once per logical
144/// point read by the builder below.
145/// Only the default `not(tombstones)` seek path resolves per-key sizes (the
146/// alternate build always falls back to a whole-file scan and produces no notes),
147/// so the variants are dead there — same shape as [`PathProbe`].
148#[cfg_attr(feature = "tombstones", allow(dead_code))]
149#[derive(Clone, Copy, Debug, PartialEq, Eq)]
150enum KeySizeNote {
151    /// This SSTable does not hold the key, so it contributes nothing either way.
152    NotHeld,
153    /// This SSTable holds the key and its index recorded an authoritative on-disk
154    /// size for it.
155    Sized(u32),
156    /// This SSTable holds the key and its on-disk extent was MEASURED as the
157    /// partition's successor gap.
158    Measured(u64),
159    /// This SSTable holds (or may hold) the key but no authoritative extent is
160    /// available — e.g. a fail-safe whole-file scan resolves no per-partition
161    /// layout, or no data-section length is known. Never estimated.
162    Unsized,
163}
164
165/// Whether a merger builder records the #2827 partition accesses itself.
166///
167/// The reader-based builder has TWO callers with opposite needs, so this cannot be a
168/// property of the function:
169///
170/// - the **core executor's** multi-generation targeted read reaches it through
171///   `generation_merge::seek_merge_generations_for_read`, and the executor already
172///   records the access at its own logical boundary
173///   (`StorageEngine::scan_partition_clustering`) — recording here too would count
174///   one logical read twice;
175/// - the **Flight warm point path** reaches it directly, and this IS its logical
176///   point-read boundary, so recording must happen here or not at all.
177///
178/// Counting stays once per logical partition read either way. Nothing is ever
179/// recorded at a per-SSTable probe site, which would multiply a repeat count by the
180/// number of generations holding the key.
181#[derive(Clone, Copy, Debug, PartialEq, Eq)]
182pub enum PointAccessRecording {
183    /// This call is the logical point-read boundary: record one access per key.
184    Record,
185    /// The caller records at its own logical boundary; do not record here.
186    CallerRecords,
187}

189/// A candidate probe plus, when the probe is enabled, one [`KeySizeNote`] per
190/// requested key in the requested order.
191struct ProbeOutcome {
192    probe: PathProbe,
193    /// Empty when the probe is disabled — the default, so this costs nothing.
194    notes: Vec<KeySizeNote>,
195}

197/// Per-candidate probe outcome across ALL requested keys. `Seeked`/`Empty` are
198/// produced only by the default `not(tombstones)` seek path (the alternate build
199/// always returns `NeedsScan`), so they are dead under `tombstones`.
200#[cfg_attr(feature = "tombstones", allow(dead_code))]
201enum PathProbe {
202    /// No requested key is present in this SSTable (all definite-absent / empty).
203    Empty,
204    /// The seeked compaction rows for every present key (combined; same generation
205    /// so they share one run index).
206    Seeked(Vec<CompactionRow>),
207    /// This SSTable has no usable index — scan it, filtered to the requested keys.
208    NeedsScan,
209}

211/// Assemble a [`KWayMerger`] that reads ONLY the target `keys` across the
212/// candidate `paths` (issue #2207).
213///
214/// Returns `Ok(None)` when no candidate holds any target key (every one a definite
215/// presence-oracle negative / prefix-collision) — the caller streams zero rows.
216/// Otherwise returns a merger the caller drives through its existing reconciliation
217/// loop; it emits one `MergeStep::Partition` per distinct present key, reconciled
218/// across generations exactly as the full-scan merge would.
219///
220/// `paths` must be the same (token-pruned) candidate list, in the same order, the
221/// full-scan path would merge — run index is the position here, so LWW tie-breaks
222/// match. `keys` are raw partition-key bytes (as `PartitionKey::to_bytes`
223/// produces). `scan_cancel` is threaded as a PER-CALL token into every probe/scan
224/// (issue #2346) so a client disconnect abandons an in-flight fail-safe scan
225/// promptly (#2264).
226pub fn build_single_partition_merger(
227    paths: Vec<PathBuf>,
228    keys: &[Vec<u8>],
229    schema: &TableSchema,
230    scan_cancel: ScanCancel,
231) -> Result<Option<KWayMerger>> {
232    build_single_partition_merger_with_registry(paths, keys, schema, None, scan_cancel)
233}

251        keys,
252        schema,

254        scan_cancel,
255        NoTrace,

296/// Shared path-builder implementation.  The public untraced wrapper passes a
297/// `NoTrace`; the explain path passes its caller-owned sink.  Probe failures
298/// return before the sink can escape, so a failed build cannot expose a
299/// partially populated trail.
300fn build_single_partition_merger_with_registry_and_sink<S: TraceSink>(
301    paths: Vec<PathBuf>,
302    keys: &[Vec<u8>],
303    schema: &TableSchema,
304    udt_registry: Option<&UdtRegistry>,
305    scan_cancel: ScanCancel,
306    mut trace: S,
307) -> Result<Option<KWayMerger<S>>> {
308    schema.validate_dropped_columns()?;
309    if keys.is_empty() {
310        return Ok(None);
311    }
312
313    // Canonicalize the requested key set (roborev suggestion, issue #2207):
314    // dedup identical raw keys — a `pk IN (5, 5)` ticket must not double-seek the
315    // same partition — and sort by Murmur3 token, the natural probing order
316    // (matches the ascending order `MergeEntry::Ord` requires each run to
317    // yield). Correctness does not depend on this order (each candidate's
318    // combined entries are sorted again below before wrapping in `VecRun`), but
319    // canonicalizing up front avoids redundant work and reads more naturally.
320    let mut canonical: Vec<Vec<u8>> = keys.to_vec();
321    canonical.sort_by_key(|k| crate::util::cassandra_murmur3::cassandra_murmur3_token(k));
322    canonical.dedup();
323    let keys: &[Vec<u8>] = &canonical;
324
325    let key_set: std::collections::HashSet<Vec<u8>> = keys.iter().cloned().collect();
326
327    // Issue #2765 (LAZY registration): a point read whose candidates ALL seek
328    // (`PathProbe::Seeked` → in-memory `VecRun`s, ZERO egress channels) buffers
329    // NOTHING, so it must NOT occupy an active-merge slot — else high-QPS Flight
330    // point reads would inflate the process-global count and throttle a
331    // concurrent channel-backed compaction/full-scan toward `MIN_CAP`. Defer
332    // `begin_merge` to the FIRST `NeedsScan` egress channel and memoize its
333    // snapshot so every channel in THIS merge still shares ONE capacity.
334    let mut egress: Option<(usize, egress_budget::ActiveMergeGuard)> = None;
335
336    // #4629: `runs[i]` MUST be candidate `i`. `MergeEntry::run_index` is the candidate's position in the
337    // caller's list, and `KWayMerger::refill_heap` indexes `runs` by it, so a candidate that holds none of
338    // the keys keeps an empty slot instead of being dropped. `produced_entries` still lets an all-empty
339    // read report `None`.
340    let mut runs: Vec<Box<dyn SSTableRowIterator>> = Vec::new();
341    let mut produced_entries = false;
342    // Byte weights for the partition access-distribution probe (issue #2827), one
343    // accumulator per requested key, summed across the candidates that resolved it.
344    // Allocated only when the probe is enabled (it is OFF by default).
345    // The path-based builder IS the cold Flight point path's logical boundary, so it
346    // always records. Extent measurement is nonetheless gated on the probe being on:
347    // it is real work (an index materialize + a successor resolution per key per
348    // generation) and must never be paid by a process that asked for no telemetry.
349    let collect = partition_access::enabled();
350    let mut weights: Vec<partition_access::AccessWeightBuilder> = if collect {
351        vec![partition_access::AccessWeightBuilder::new(); keys.len()]
352    } else {
353        Vec::new()
354    };
355    for (run_index, path) in paths.iter().enumerate() {
356        // Cooperative cancellation (#2264): honour a cancel BEFORE each candidate's
357        // seek/open, so a disconnected point read does no further per-SSTable work.
358        scan_cancel.check()?;
359
360        let ProbeOutcome { probe, notes } = probe_path(
361            path,
362            keys,
363            schema,
364            udt_registry,
365            scan_cancel.clone(),
366            collect,
367        )?;
368        fold_size_notes(&mut weights, &notes);
369        trace.generation_probe(run_index, trace_probe_outcome(&probe));
370        match probe {
371            PathProbe::Empty => {
372                // Pruned / absent for every key: an EMPTY run keeps its slot (#4629).
373                runs.push(empty_run());
374            }
375            PathProbe::Seeked(rows) => {
376                if rows.is_empty() {
377                    runs.push(empty_run());
378                    continue;
379                }
380                produced_entries = true;
381                let mut entries: Vec<MergeEntry> = Vec::with_capacity(rows.len());
382                for row in rows {
383                    // Issue #2096: one merge entry decoded from `Data.db` by the
384                    // SEEK path (this candidate held the target partition). Counting
385                    // per built entry — the same unit the full-scan run counts per
386                    // `Ok(entry)` — keeps `merge_run_entries_decoded` apples-to-
387                    // apples between the two paths.
388                    crate::storage::sstable::work_counters::add_merge_run_entry_decoded();
389                    entries.push(SSTableRowIteratorAdapter::build_merge_entry(
390                        run_index, row, schema,
391                    )?);
392                }
393                // BLOCKER fix (roborev, issue #2207): `rows` was accumulated in
394                // REQUESTED-KEY order (IN-list / Or order), not token order. Every
395                // `SSTableRowIterator` MUST yield entries in ascending SCHEMA-AWARE heap
396                // order (token, key, clustering — `schema_order::schema_entry_cmp`) — `refill_heap` buffers only ONE
397                // entry per run at a time and relies on that invariant (mod.rs
398                // `step`/`refill_heap`). An out-of-order run causes `step()` to
399                // silently split one run's contribution across two heap pops,
400                // duplicating rows and breaking cross-generation reconciliation
401                // (a newer generation's overwrite/tombstone no longer shadows the
402                // older one). Sort once, here, before wrapping in `VecRun`.
403                super::schema_order::sort_entries_schema_ordered(&mut entries, schema);
404                runs.push(Box::new(VecRun {
405                    entries: entries.into(),
406                }));
407            }
408            PathProbe::NeedsScan => {
409                // Fail-safe: scan this ONE SSTable, forwarding only the target
410                // partitions. The missing index costs speed, never correctness.
411                // Snapshot the shared capacity on the FIRST egress channel only.
412                let channel_capacity = egress.get_or_insert_with(egress_budget::begin_merge).0;
413                let adapter = SSTableRowIteratorAdapter::open(
414                    path,
415                    run_index,
416                    schema,
417                    udt_registry.cloned(),
418                    scan_cancel.clone(),
419                    channel_capacity,
420                    // Issue #1704: no recording seam encloses this fail-safe scan, so
421                    // a failed input open must still report itself. See
422                    // `merge::constructors`' module doc for why the caller states it.
423                    crate::storage::sstable::reader::OpenErrorReporting::SelfReported,
424                )?;
425                produced_entries = true;
426                runs.push(Box::new(SinglePartitionFilterRun {
427                    inner: adapter,
428                    keys: key_set.clone(),
429                }));
430            }
431        }
432    }
433
434    // ONE access per LOGICAL partition read, recorded after every candidate has
435    // reported (issue #2827). This is the point-read path's logical boundary: the
436    // key list is canonical (deduplicated), so `pk IN (5, 5)` is one access, and a
437    // key present in seven generations is still one access carrying the SUM of the
438    // per-generation on-disk sizes. Recorded even when no candidate held the key —
439    // a miss is a real access, and omitting it would understate the singleton
440    // bucket. A no-op when the probe is disabled.
441    record_logical_accesses(schema, keys, weights);
442
443    if !produced_entries {
444        // No candidate produced a run; any snapshot guard drops here.
445        return Ok(None);
446    }
447    let merger = KWayMerger::from_row_iterators(runs, schema)?;
448    // Attach the slot ONLY if a `NeedsScan` egress channel was actually opened;
449    // an all-`Seeked` (channel-less) point read registers no slot.
450    let merger = match egress {
451        Some((_, guard)) => merger.with_egress_slot(guard),
452        None => merger,
453    };
454    Ok(Some(merger.with_trace_sink(trace)))
455}

457/// Reader-based analogue of [`build_single_partition_merger`] (issue #2346):
458/// assembles a [`KWayMerger`] that reads ONLY the target `keys`, but across
459/// already-open, possibly-SHARED `readers` instead of `paths` — the seam a
460/// future cached-reader caller (e.g. a Flight warm-handle registry) uses to
461/// avoid a fresh per-request reader-open/Index/Summary/Statistics/bloom parse.
462///
463/// Semantics mirror `build_single_partition_merger` exactly (same
464/// canonicalization, same three-way per-candidate probe via [`probe_reader`],
465/// same [`SinglePartitionFilterRun`] fail-safe, same run-index ordering) — only
466/// WHO opens/owns the `SSTableReader` differs. `readers` must be the same
467/// (token-pruned) candidate list, in the same order, the full-scan reader-based
468/// merger ([`KWayMerger::new_from_readers`]) would merge.
469/// `recording` (issue #2827) governs ONLY whether this call records one partition
470/// access per requested key; it changes no rows, no ordering and no probe behaviour.
471/// It is an explicit caller decision because this builder's two callers sit at
472/// different levels — see [`PointAccessRecording`].
473pub fn build_single_partition_merger_from_readers(
474    readers: Vec<Arc<SSTableReader>>,
475    keys: &[Vec<u8>],
476    schema: &TableSchema,
477    scan_cancel: ScanCancel,
478    recording: PointAccessRecording,
479) -> Result<Option<KWayMerger>> {
480    schema.validate_dropped_columns()?;
481    if keys.is_empty() {
482        return Ok(None);
483    }
484
485    let mut canonical: Vec<Vec<u8>> = keys.to_vec();
486    canonical.sort_by_key(|k| crate::util::cassandra_murmur3::cassandra_murmur3_token(k));
487    canonical.dedup();
488    let keys: &[Vec<u8>] = &canonical;
489
490    let key_set: std::collections::HashSet<Vec<u8>> = keys.iter().cloned().collect();
491
492    // Issue #2765 (LAZY registration, see the path-based builder above): occupy an
493    // active-merge slot ONLY when a `NeedsScan` egress channel is actually opened;
494    // an all-`Seeked` point read buffers nothing and registers no slot. The
495    // snapshot is memoized so every channel in THIS merge shares ONE capacity.
496    let mut egress: Option<(usize, egress_budget::ActiveMergeGuard)> = None;
497
498    // `CallerRecords` must measure NOTHING. Measuring an extent costs an index
499    // materialize plus an O(N) `Index.db` successor scan per key per generation, and
500    // under `CallerRecords` the result is discarded and re-derived by the caller —
501    // a per-read O(N) scan whose output is thrown away.
502    let collect = recording == PointAccessRecording::Record && partition_access::enabled();
503    let mut weights: Vec<partition_access::AccessWeightBuilder> = if collect {
504        vec![partition_access::AccessWeightBuilder::new(); keys.len()]
505    } else {
506        Vec::new()
507    };
508
509    // #4629: `runs[i]` MUST be candidate `i`. `MergeEntry::run_index` is the candidate's position in the
510    // caller's list, and `KWayMerger::refill_heap` indexes `runs` by it, so a candidate that holds none of
511    // the keys keeps an empty slot instead of being dropped. `produced_entries` still lets an all-empty
512    // read report `None`.
513    let mut runs: Vec<Box<dyn SSTableRowIterator>> = Vec::new();
514    let mut produced_entries = false;
515    for (run_index, reader) in readers.into_iter().enumerate() {
516        // Cooperative cancellation (#2264): honour a cancel BEFORE each candidate's
517        // probe, so a disconnected point read does no further per-SSTable work.
518        scan_cancel.check()?;
519
520        let ProbeOutcome { probe, notes } = probe_reader(
521            Arc::clone(&reader),
522            keys,
523            schema,
524            scan_cancel.clone(),
525            collect,
526        )?;
527        fold_size_notes(&mut weights, &notes);
528        match probe {
529            PathProbe::Empty => {
530                // Pruned / absent for every key: an EMPTY run keeps its slot (#4629).
531                runs.push(empty_run());
532            }
533            PathProbe::Seeked(rows) => {
534                if rows.is_empty() {
535                    runs.push(empty_run());
536                    continue;
537                }
538                produced_entries = true;
539                let mut entries: Vec<MergeEntry> = Vec::with_capacity(rows.len());
540                for row in rows {
541                    // Issue #2096: one merge entry decoded from `Data.db` by the
542                    // SEEK path (this candidate held the target partition), same
543                    // accounting as the path-based builder above.
544                    crate::storage::sstable::work_counters::add_merge_run_entry_decoded();
545                    entries.push(SSTableRowIteratorAdapter::build_merge_entry(
546                        run_index, row, schema,
547                    )?);
548                }
549                // See `build_single_partition_merger`'s identical sort — every
550                // `SSTableRowIterator` MUST yield ascending `MergeEntry` order.
551                super::schema_order::sort_entries_schema_ordered(&mut entries, schema);
552                runs.push(Box::new(VecRun {
553                    entries: entries.into(),
554                }));
555            }
556            PathProbe::NeedsScan => {
557                // Fail-safe: scan this ONE shared reader's compaction stream,
558                // forwarding only the target partitions — never a fresh
559                // path-based open (issue #2346's whole point). `reader` (the
560                // ORIGINAL, not the clone `probe_reader` consumed above) is
561                // still available here.
562                // Point-read fail-safe: a specific-key filter, not a range scan —
563                // no token bound is pushed (issue #2412; the key set bounds it).
564                // Snapshot the shared capacity on the FIRST egress channel only.
565                let channel_capacity = egress.get_or_insert_with(egress_budget::begin_merge).0;
566                let adapter = SSTableRowIteratorAdapter::open_from_reader(
567                    reader,
568                    run_index,
569                    schema,
570                    scan_cancel.clone(),
571                    None,
572                    channel_capacity,
573                )?;
574                produced_entries = true;
575                runs.push(Box::new(SinglePartitionFilterRun {
576                    inner: adapter,
577                    keys: key_set.clone(),
578                }));
579            }
580        }
581    }
582
583    // ONE access per LOGICAL partition read, when this call is the boundary
584    // (issue #2827). Recorded even when no candidate held the key — a miss is a real
585    // access, and omitting it would understate the singleton bucket.
586    if collect {
587        record_logical_accesses(schema, keys, weights);
588    }
589
590    if !produced_entries {
591        // No candidate produced a run; any snapshot guard drops here.
592        return Ok(None);
593    }
594    let merger = KWayMerger::from_row_iterators(runs, schema)?;
595    // Attach the slot ONLY if a `NeedsScan` egress channel was actually opened.
596    Ok(Some(match egress {
597        Some((_, guard)) => merger.with_egress_slot(guard),
598        None => merger,
599    }))
600}

602/// Probe an ALREADY-OPEN `reader` for every requested key via the core seek
603/// primitive (default `not(tombstones)` build) — the shared core BOTH
604/// [`probe_path`] (path-based, opens its own reader then delegates here) and
605/// [`build_single_partition_merger_from_readers`] (reader-based) use, so the
606/// two never diverge (issue #2346).
607///
608/// A single SSTable's index availability is a property of the SSTable (not the
609/// key), so the FIRST `IndexUnavailable` short-circuits to [`PathProbe::NeedsScan`]
610/// (scan the whole file, filtered). Otherwise every present key's seeked rows are
611/// concatenated (they share this SSTable's generation / run index).
612#[cfg(not(feature = "tombstones"))]
613async fn probe_reader_async(
614    reader: &SSTableReader,
615    keys: &[Vec<u8>],
616    schema: &TableSchema,
617    scan_cancel: &ScanCancel,
618    collect_notes: bool,
619) -> Result<ProbeOutcome> {
620    let mut notes: Vec<KeySizeNote> = Vec::new();
621    let mut rows: Vec<CompactionRow> = Vec::new();
622    for key in keys {
623        match reader
624            .read_single_partition_for_compaction(key, Some(schema), scan_cancel)
625            .await?
626        {
627            SinglePartitionCompaction::DefinitelyAbsent => {
628                if collect_notes {
629                    notes.push(KeySizeNote::NotHeld);
630                }
631            }
632            SinglePartitionCompaction::Rows(mut r) => {
633                if collect_notes {
634                    notes.push(resolved_size_note(reader, key).await);
635                }
636                rows.append(&mut r)
637            }
638            // Index availability is per-SSTable, not per-key: fall back to
639            // scanning the whole file once, filtered to the key set.
640            SinglePartitionCompaction::IndexUnavailable => {
641                return Ok(ProbeOutcome {
642                    probe: PathProbe::NeedsScan,
643                    // A whole-file fail-safe scan resolves no per-partition size,
644                    // so EVERY key this SSTable might hold is unpriceable through
645                    // it. Reported as unavailable rather than skipped, so the
646                    // window's byte total is visibly incomplete instead of quietly
647                    // short (issue #2827, design D6).
648                    notes: if collect_notes {
649                        vec![KeySizeNote::Unsized; keys.len()]
650                    } else {
651                        Vec::new()
652                    },
653                });
654            }
655        }
656    }
657    let probe = if rows.is_empty() {
658        PathProbe::Empty
659    } else {
660        PathProbe::Seeked(rows)
661    };
662    Ok(ProbeOutcome { probe, notes })
663}

665/// This SSTable's authoritative on-disk extent for `key`.
666///
667/// The partition's location comes from what the read that just ran already resolved
668/// into the process-global key→partition-offset cache (issue #2059's
669/// `PartitionLoc`), so the instrument never re-drives a lookup and can never
670/// perturb the read path's own telemetry. The EXTENT is then measured as the
671/// successor gap — `[data_offset, successor_offset)`, bounding to the authoritative
672/// uncompressed data-section length for the last partition — which is the same
673/// authoritative index-layout bound the single-partition seek uses to size its
674/// decompression window.
675///
676/// Nothing is ever estimated. Where no extent is resolvable the note is
677/// [`KeySizeNote::Unsized`]: the access is still counted, and it contributes zero
678/// bytes under `size_source = unavailable`.
679///
680/// A key-cache MISS is NOT treated as "no extent". This function is only reached
681/// when the probe returned rows for the key, so the partition IS held; the offset is
682/// then resolved authoritatively, exactly as the core path's resolver does. Pricing
683/// the same partition differently on the two paths — measured on one, unpriceable on
684/// the other — would make a window's byte total depend on which executor served the
685/// read.
686#[cfg(not(feature = "tombstones"))]
687async fn resolved_size_note(reader: &SSTableReader, key: &[u8]) -> KeySizeNote {
688    // An index-recorded size, if one ever exists. No Cassandra 5.0 index records
689    // one, so in practice this never fires.
690    if let Some(loc) = reader.key_cache_get(key) {
691        if loc.data_size > 0 {
692            return KeySizeNote::Sized(loc.data_size);
693        }
694        if let Ok(Some(gap)) = reader.measure_partition_extent(loc.data_offset, key).await {
695            return KeySizeNote::Measured(gap);
696        }
697        return KeySizeNote::Unsized;
698    }
699    // Cache miss on a partition the read just returned rows for: resolve the offset
700    // rather than declaring it unpriceable.
701    use crate::storage::sstable::reader::partition_successor::PartitionResidency;
702    match reader.partition_residency(key).await {
703        PartitionResidency::HeldAt(offset) => {
704            match reader.measure_partition_extent(offset, key).await {
705                Ok(Some(gap)) => KeySizeNote::Measured(gap),
706                Ok(None) | Err(_) => KeySizeNote::Unsized,
707            }
708        }
709        // `NotHeld` is unreachable here (the read returned rows for this key) and
710        // `Unknown` is genuinely unpriceable; both fail closed.
711        PartitionResidency::NotHeld | PartitionResidency::Unknown => KeySizeNote::Unsized,
712    }
713}

715/// Fold one candidate's per-key size notes into the per-key weight accumulators.
716///
717/// A no-op when the probe is disabled (both vectors are empty).
718fn fold_size_notes(weights: &mut [partition_access::AccessWeightBuilder], notes: &[KeySizeNote]) {
719    for (w, note) in weights.iter_mut().zip(notes.iter()) {
720        match note {
721            KeySizeNote::NotHeld => {}
722            KeySizeNote::Sized(n) => w.note_sized(*n),
723            KeySizeNote::Measured(n) => w.note_measured(*n),
724            KeySizeNote::Unsized => w.note_unsized(),
725        }
726    }
727}

729/// Record ONE logical partition access per requested key (issue #2827).
730///
731/// A no-op when the probe is disabled — `weights` is empty then, so this does not
732/// even iterate.
733fn record_logical_accesses(
734    schema: &TableSchema,
735    keys: &[Vec<u8>],
736    weights: Vec<partition_access::AccessWeightBuilder>,
737) {
738    // The table is part of the entry identity — see `TableScope`. The schema is the
739    // authority here (the builders are driven per-table), so no formatting or
740    // allocation is needed on this path.
741    let scope = partition_access::TableScope::new(&schema.keyspace, &schema.table);
742    for (key, weight) in keys.iter().zip(weights) {
743        partition_access::record_partition_access(scope, key, weight.finish());
744    }
745}

754/// The trail's per-generation probe outcome for a [`PathProbe`] (issue #4193 review L5).
755///
756/// A `Seeked` probe with NO rows is an authoritative "this generation does not hold the key"
757/// (a prefix-collision candidate decoded and rejected, see `SinglePartitionCompaction::Rows`),
758/// so it is reported `Absent` — never `Hit` — exactly as the merge treats it (no run is added).
759fn trace_probe_outcome(probe: &PathProbe) -> TraceProbeOutcome {
760    match probe {
761        PathProbe::Empty => TraceProbeOutcome::Absent,
762        PathProbe::Seeked(rows) if rows.is_empty() => TraceProbeOutcome::Absent,
763        PathProbe::Seeked(_) => TraceProbeOutcome::Hit,
764        PathProbe::NeedsScan => TraceProbeOutcome::Scanned,
765    }
766}
