2//! non-overlapping canonical sequence (issue #933 / roborev #959) and test
3//! clustering-key containment against a single range tombstone.

6//! rule) — pure relocation, no behavior change.

15/// A cut position on the clustering axis, used to coalesce range tombstones
16/// into a NON-OVERLAPPING canonical sequence (issue #933 / roborev #959 High
17/// #1).
18///
19/// The writer emits each [`RangeTombstone`] as an INDEPENDENT open/close marker
20/// pair, sorted by clustering position; the reader pairs markers using a single
21/// `pending_range_start`. Overlapping or nested ranges with different bounds
22/// would therefore mis-pair on read-back (e.g. `start[1,5] start[2,3] end[2,3]
23/// end[1,5]` resurfaces as `[2,3]` and `[Bottom,5]`), corrupting the persisted
24/// deletion ranges. Coalescing the cross-SSTable union into disjoint ranges
25/// before re-emission mirrors Cassandra's `RangeTombstoneList` invariant
26/// (on-disk range tombstones within a partition never overlap).
27///
28/// A range `[start, end]` is modelled as a closed interval over these cut
29/// positions. [`Self::At`] is the infinitesimal point just before (`after =
30/// false`) or just after (`after = true`) a clustering prefix; an open bound is
31/// [`Self::Bottom`] / [`Self::Top`]. The `after` flag at a SHORTER prefix sorts
32/// relative to all longer extensions (Cassandra's kind-weighted prefix
33/// ordering), so a prefix end bound `[ck1]` correctly covers every `[ck1, *]`.
34#[cfg(feature = "write-support")]
35#[derive(Clone)]
36pub(in super::super) enum RangeCut {
37    /// Before all clustering keys (start of partition).
38    Bottom,
39    /// Infinitesimally before (`after == false`) or after (`after == true`) the
40    /// clustering prefix.
41    At {
42        /// Clustering prefix (possibly shorter than the full arity).
43        key: ClusteringKey,
44        /// `false` = just before the prefix and all its extensions; `true` =
45        /// just after them.
46        after: bool,
47    },
48    /// After all clustering keys (end of partition).
49    Top,
50}

52impl<S: TraceSink> KWayMerger<S> {
53    /// Coalesce range tombstones into a NON-OVERLAPPING canonical sequence per
54    /// partition, the winning (newest) deletion time per covered segment (issue
55    /// #933 / roborev #959 High #1).
56    ///
57    /// Each input SSTable's range tombstones are individually non-overlapping
58    /// (Cassandra's `RangeTombstoneList` invariant), but the cross-SSTable union
59    /// gathered during compaction can overlap with different bounds. The writer
60    /// emits each retained range as an independent open/close marker pair and the
61    /// reader pairs them with a single `pending_range_start`, so OVERLAPPING
62    /// re-emitted ranges would mis-pair on read-back and corrupt the persisted
63    /// ranges. Splitting the union into disjoint segments (each carrying the max
64    /// `markedForDeleteAt` covering it) keeps the on-disk markers a clean
65    /// alternating open/close sequence. This also subsumes the prior
66    /// identical-bounds dedup.
67    pub(in super::super) fn coalesce_range_tombstones(
68        rts: &mut Vec<(DecoratedKey, RangeTombstone)>,
69        schema: &TableSchema,
70    ) {
71        // Group by partition-key bytes, preserving the first-seen DecoratedKey as
72        // the representative for each group (token + raw bytes are identical
73        // across the group).
74        let mut groups: Vec<(DecoratedKey, Vec<RangeTombstone>)> = Vec::new();
75        for (key, rt) in rts.drain(..) {
76            if let Some((_, ranges)) = groups.iter_mut().find(|(k, _)| k.key == key.key) {
77                ranges.push(rt);
78            } else {
79                groups.push((key, vec![rt]));
80            }
81        }
82
83        let mut out: Vec<(DecoratedKey, RangeTombstone)> = Vec::new();
84        for (key, ranges) in groups {
85            for rt in Self::coalesce_partition_range_tombstones(ranges, schema) {
86                out.push((key.clone(), rt));
87            }
88        }
89        *rts = out;
90    }

92    /// Coalesce the range tombstones of a SINGLE partition into a disjoint,
93    /// clustering-sorted sequence (helper for [`Self::coalesce_range_tombstones`]).
94    pub(in super::super) fn coalesce_partition_range_tombstones(
95        ranges: Vec<RangeTombstone>,
96        schema: &TableSchema,
97    ) -> Vec<RangeTombstone> {
98        if ranges.len() <= 1 {
99            return ranges;
100        }
101
102        // Model each range as a closed interval [start_cut, end_cut] with its
103        // (mfda, ldt).
104        let items: Vec<(RangeCut, RangeCut, i64, i32)> = ranges
105            .iter()
106            .map(|rt| {
107                (
108                    Self::range_start_cut(&rt.start),
109                    Self::range_end_cut(&rt.end),
110                    rt.deletion_time,
111                    rt.local_deletion_time,
112                )
113            })
114            .collect();
115
116        // Distinct, sorted cut positions (the candidate segment boundaries).
117        let mut cuts: Vec<RangeCut> = Vec::with_capacity(items.len() * 2);
118        for (s, e, _, _) in &items {
119            cuts.push(s.clone());
120            cuts.push(e.clone());
121        }
122        cuts.sort_by(|a, b| Self::cut_cmp(a, b, schema));
123        cuts.dedup_by(|a, b| Self::cut_cmp(a, b, schema) == Ordering::Equal);
124
125        // For each elementary gap (cuts[i], cuts[i+1]), the winning deletion is
126        // the max `markedForDeleteAt` among ranges whose closed interval fully
127        // contains the gap (start <= lo AND hi <= end).
128        let mut segs: Vec<(RangeCut, RangeCut, i64, i32)> = Vec::new();
129        for window in cuts.windows(2) {
130            let (lo, hi) = (&window[0], &window[1]);
131            let mut best: Option<(i64, i32)> = None;
132            for (s, e, mfda, ldt) in &items {
133                let covers = Self::cut_cmp(s, lo, schema) != Ordering::Greater
134                    && Self::cut_cmp(hi, e, schema) != Ordering::Greater;
135                if !covers {
136                    continue;
137                }
138                best = Some(match best {
139                    Some((bm, bl)) if bm > *mfda || (bm == *mfda && bl >= *ldt) => (bm, bl),
140                    _ => (*mfda, *ldt),
141                });
142            }
143            if let Some((mfda, ldt)) = best {
144                segs.push((lo.clone(), hi.clone(), mfda, ldt));
145            }
146        }
147
148        // Merge adjacent segments that share a boundary AND the same deletion
149        // (minimal fragmentation); a gap with no covering range breaks the run.
150        let mut merged: Vec<(RangeCut, RangeCut, i64, i32)> = Vec::new();
151        for seg in segs {
152            if let Some(last) = merged.last_mut() {
153                if last.2 == seg.2
154                    && last.3 == seg.3
155                    && Self::cut_cmp(&last.1, &seg.0, schema) == Ordering::Equal
156                {
157                    last.1 = seg.1;
158                    continue;
159                }
160            }
161            merged.push(seg);
162        }
163
164        merged
165            .into_iter()
166            .map(|(lo, hi, mfda, ldt)| RangeTombstone {
167                start: Self::cut_to_start_bound(lo),
168                end: Self::cut_to_end_bound(hi),
169                deletion_time: mfda,
170                local_deletion_time: ldt,
171            })
172            .collect()
173    }

175    /// Total order of two cut positions on the clustering axis (schema-aware,
176    /// honoring per-column ASC/DESC and Cassandra's kind-weighted prefix
177    /// ordering). See [`RangeCut`].
178    pub(in super::super) fn cut_cmp(a: &RangeCut, b: &RangeCut, schema: &TableSchema) -> Ordering {
179        match (a, b) {
180            (RangeCut::Bottom, RangeCut::Bottom) => Ordering::Equal,
181            (RangeCut::Bottom, _) => Ordering::Less,
182            (_, RangeCut::Bottom) => Ordering::Greater,
183            (RangeCut::Top, RangeCut::Top) => Ordering::Equal,
184            (RangeCut::Top, _) => Ordering::Greater,
185            (_, RangeCut::Top) => Ordering::Less,
186            (RangeCut::At { key: ka, after: aa }, RangeCut::At { key: kb, after: ab }) => {
187                // Compare the common prefix only; a shorter prefix's `after` flag
188                // decides its position relative to every longer extension.
189                let l = ka.columns.len().min(kb.columns.len());
190                let ta = ClusteringKey {
191                    columns: ka.columns[..l].to_vec(),
192                };
193                let tb = ClusteringKey {
194                    columns: kb.columns[..l].to_vec(),
195                };
196                let ord = ta.compare(&tb, schema).unwrap_or_else(|_| ta.cmp(&tb));
197                if ord != Ordering::Equal {
198                    return ord;
199                }
200                match ka.columns.len().cmp(&kb.columns.len()) {
201                    Ordering::Equal => aa.cmp(ab),
202                    // `a` is the shorter prefix: just-after sorts past every
203                    // extension of it, just-before sorts ahead of all of them.
204                    Ordering::Less => {
205                        if *aa {
206                            Ordering::Greater
207                        } else {
208                            Ordering::Less
209                        }
210                    }
211                    // `b` is the shorter prefix (mirror image).
212                    Ordering::Greater => {
213                        if *ab {
214                            Ordering::Less
215                        } else {
216                            Ordering::Greater
217                        }
218                    }
219                }
220            }
221        }
222    }

224    /// Left edge (cut) of a range's start bound.
225    pub(in super::super) fn range_start_cut(
226        bound: &crate::storage::write_engine::mutation::ClusteringBound,
227    ) -> RangeCut {
228        use crate::storage::write_engine::mutation::ClusteringBound;
229        match bound {
230            ClusteringBound::Inclusive(ck) => RangeCut::At {
231                key: ck.clone(),
232                after: false,
233            },
234            ClusteringBound::Exclusive(ck) => RangeCut::At {
235                key: ck.clone(),
236                after: true,
237            },
238            ClusteringBound::Bottom => RangeCut::Bottom,
239            ClusteringBound::Top => RangeCut::Top,
240        }
241    }

262    /// Whether `outer` fully covers the canonical `inner` range. This is used
263    /// only to carry the source generation across range coalescing; the write
264    /// path's shadowing decision continues to use the authoritative per-cell
265    /// containment check below.
266    pub(in super::super) fn range_tombstone_contains_range(
267        outer: &RangeTombstone,
268        inner: &RangeTombstone,
269        schema: &TableSchema,
270    ) -> bool {
271        let outer_start = Self::range_start_cut(&outer.start);
272        let inner_start = Self::range_start_cut(&inner.start);
273        let inner_end = Self::range_end_cut(&inner.end);
274        let outer_end = Self::range_end_cut(&outer.end);
275        Self::cut_cmp(&outer_start, &inner_start, schema) != Ordering::Greater
276            && Self::cut_cmp(&inner_end, &outer_end, schema) != Ordering::Greater
277    }

279    /// Convert a left-edge cut back into a start [`ClusteringBound`].
280    pub(in super::super) fn cut_to_start_bound(
281        cut: RangeCut,
282    ) -> crate::storage::write_engine::mutation::ClusteringBound {
283        use crate::storage::write_engine::mutation::ClusteringBound;
284        match cut {
285            RangeCut::Bottom => ClusteringBound::Bottom,
286            RangeCut::At { key, after: false } => ClusteringBound::Inclusive(key),
287            RangeCut::At { key, after: true } => ClusteringBound::Exclusive(key),
288            RangeCut::Top => ClusteringBound::Top,
289        }
290    }

304    /// Whether a range tombstone's clustering range covers `ck`, comparing bounds
305    /// SCHEMA-AWARE (honoring per-column ASC/DESC via [`ClusteringKey::compare`],
306    /// NOT the schema-agnostic `cmp`) — issue #933 / roborev #959 Medium #3.
307    ///
308    /// Bounds may be a PREFIX shorter than the full clustering arity; comparing
309    /// only the bound's components (via [`ClusteringKey::compare`], which treats an
310    /// absent trailing component as a first-sorting NULL) yields the correct
311    /// containment for the `DELETE WHERE pk=? AND ck1=?` prefix case.
312    pub(in super::super) fn range_tombstone_covers_ck(
313        ck: &ClusteringKey,
314        rt: &RangeTombstone,
315        schema: &TableSchema,
316    ) -> bool {
317        // Issue #1669: count coverage comparisons so a bound test can prove the
318        // binary search stays O(rows) — one candidate per row — instead of the
319        // former O(rows × ranges) linear scan. Vanishes in production builds.
320        #[cfg(test)]
321        crate::storage::sstable::work_counters::range_coverage_scope::record();
322        use crate::storage::write_engine::mutation::ClusteringBound;
323
324        // Compare `ck` against a bound key over the bound's component count so a
325        // prefix bound only compares its present components.
326        let cmp = |bound: &ClusteringKey| -> Ordering {
327            let n = bound.columns.len();
328            let truncated = ClusteringKey {
329                columns: ck.columns.iter().take(n).cloned().collect(),
330            };
331            truncated
332                .compare(bound, schema)
333                .unwrap_or_else(|_| truncated.cmp(bound))
334        };
335
336        let after_start = match &rt.start {
337            ClusteringBound::Inclusive(b) => cmp(b) != Ordering::Less,
338            ClusteringBound::Exclusive(b) => cmp(b) == Ordering::Greater,
339            ClusteringBound::Bottom => true,
340            ClusteringBound::Top => false,
341        };
342        let before_end = match &rt.end {
343            ClusteringBound::Inclusive(b) => cmp(b) != Ordering::Greater,
344            ClusteringBound::Exclusive(b) => cmp(b) == Ordering::Less,
345            ClusteringBound::Top => true,
346            ClusteringBound::Bottom => false,
347        };
348        after_start && before_end
349    }

351    /// Whether a coalesced range tombstone's END bound lies strictly BEFORE `ck`
352    /// on the clustering axis — i.e. `ck` is beyond the range's end, so the range
353    /// cannot cover it. This is exactly the negation of the `before_end` test in
354    /// [`Self::range_tombstone_covers_ck`], kept in lock-step with it.
355    ///
356    /// It is the monotonic predicate the #1669 binary search feeds to
357    /// [`slice::partition_point`]: because `coalesce_range_tombstones` yields a
358    /// per-partition sequence sorted by start bound and DISJOINT, the ranges are
359    /// also sorted by end bound, so `range_end_before_ck` is `true` for every
360    /// range wholly before `ck` and `false` thereafter — a clean partition point.
361    /// The first `false` range is the ONLY candidate that can contain `ck`
362    /// (disjointness ⇒ at most one covers it).
363    ///
364    /// Deliberately does NOT bump the `range_coverage_scope` counter: it is the
365    /// cheap `O(log ranges)` search step, distinct from the single authoritative
366    /// `range_tombstone_covers_ck` containment check the counter measures.
367    pub(in super::super) fn range_end_before_ck(
368        ck: &ClusteringKey,
369        rt: &RangeTombstone,
370        schema: &TableSchema,
371    ) -> bool {
372        use crate::storage::write_engine::mutation::ClusteringBound;
373
374        // Same prefix-aware comparison as `range_tombstone_covers_ck` (compare
375        // `ck` against the bound over the bound's component count).
376        let cmp = |bound: &ClusteringKey| -> Ordering {
377            let n = bound.columns.len();
378            let truncated = ClusteringKey {
379                columns: ck.columns.iter().take(n).cloned().collect(),
380            };
381            truncated
382                .compare(bound, schema)
383                .unwrap_or_else(|_| truncated.cmp(bound))
384        };
385
386        match &rt.end {
387            // Negation of the `before_end` arms in `range_tombstone_covers_ck`.
388            ClusteringBound::Inclusive(b) => cmp(b) == Ordering::Greater,
389            ClusteringBound::Exclusive(b) => cmp(b) != Ordering::Less,
390            ClusteringBound::Top => false,
391            ClusteringBound::Bottom => true,
392        }
393    }
