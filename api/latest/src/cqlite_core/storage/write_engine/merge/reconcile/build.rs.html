1//! Step 4 of the reconcile kernel — build the merged `MergeEntry` (split out of `reconcile.rs`
2//! under the campsite rule, epic #1116). Unchanged by the split except issue #4581: a
3//! clustering-less row whose only content is its primary-key liveness marker is emitted (with the
4//! marker's real timestamp) on reads AND on compaction, and a marker newer than a surviving row
5//! deletion keeps the row live alongside that deletion.
6
7use std::collections::HashSet;
8
9use super::super::trace::{NoTrace, TraceSink};
10use super::super::{CellData, KWayMerger, MergeEntry, PurgeCounts, RowData};
11use super::{ReconcileState, ReconcileTraceMetadata};
12
13impl<S: TraceSink> ReconcileState<S> {
14    /// Step 4 — build the merged result, applying the phantom key-only-row guard
15    /// (#921 finding 3): recompute `purged_to_empty` AFTER the gc-grace
16    /// cell-tombstone purge, so a CLUSTERED row whose only non-key data was a
17    /// purgeable cell tombstone emits NOTHING instead of a phantom key-only live
18    /// row. A row that was always key-only (a genuine row marker) keeps
19    /// `had_data_before == false` and is preserved.
20    ///
21    /// Returns `None` for an empty group (the original `let key = key?`
22    /// early-out) or a truly absent row.
23    #[allow(dead_code)]
24    pub(in crate::storage::write_engine::merge) fn build(
25        self,
26        purges: &mut PurgeCounts,
27        now: Option<i64>,
28    ) -> Option<MergeEntry> {
29        self.build_with_trace_metadata(purges, now).0
30    }
31
32    /// Build the merged row and return the trace-only source map alongside it.
33    /// The map is moved out before the behavior-preserving build body consumes
34    /// the state, so untraced builds never clone cell keys or allocate a side
35    /// channel.
36    pub(in crate::storage::write_engine::merge) fn build_with_trace_metadata(
37        mut self,
38        purges: &mut PurgeCounts,
39        now: Option<i64>,
40    ) -> (Option<MergeEntry>, ReconcileTraceMetadata) {
41        let winner_runs = self.winner_runs.take();
42        let built = self.build_inner(purges, now);
43        (built, ReconcileTraceMetadata { winner_runs })
44    }
45
46    fn build_inner(self, purges: &mut PurgeCounts, now: Option<i64>) -> Option<MergeEntry> {
47        let ReconcileState {
48            clustering_key,
49            key,
50            run_index,
51            row_del,
52            purged_row_del,
53            row_del_ldt,
54            complex_deletions,
55            range_deletion,
56            mut surviving,
57            had_data_before,
58            row_liveness,
59            marker_purged,
60            ..
61        } = self;
62
63        let key = key?; // empty group => nothing to emit
64
65        let ck_names: HashSet<&str> = clustering_key
66            .as_ref()
67            .map(|ck| ck.columns.iter().map(|(n, _)| n.as_str()).collect())
68            .unwrap_or_default();
69        let is_data_cell = |cell: &CellData| !ck_names.contains(cell.column.as_str());
70        // #4545: key pseudo-cells survive a row deletion only beside LIVE data or a still-live newer
71        // marker (decided here, after expiry/purge; a gc-purged deletion still sheds them).
72        if let Some(d) = row_del.or(purged_row_del) {
73            super::super::model::shed_key_cells(&mut surviving, &ck_names, &row_liveness, d, now);
74        }
75        let has_data_after = surviving.iter().any(is_data_cell);
76        let purged_to_empty = had_data_before && !has_data_after;
77        // #4581: a purged expired marker leaves a clustered key-only row (its clustering
78        // pseudo-cells) with no liveness and no data: `BTreeRow.purge` -> null.
79        let marker_gone = marker_purged && !has_data_after;
80        drop(ck_names);
81
82        // Issue #2163 (roborev r7): a cell tombstone (simple or complex-element,
83        // the SAME predicate `purge_gc_grace`'s retain closure uses) that
84        // SURVIVES into `surviving` — whether because gc-grace/overlap kept it,
85        // or because purging was inactive for this merge entirely (no
86        // `gc_before_secs`, e.g. an unbounded partial compaction) — is a marker
87        // RETAINED into the output, matching the row/range/partition tombstone
88        // "emitted" contract established above. Counted here from the FINAL
89        // surviving set (not inside `purge_gc_grace`'s conditional retain) so a
90        // merge with purging disabled still counts its retained cell
91        // tombstones; a cell tombstone always satisfies `is_data_cell` (it is
92        // never a clustering-key pseudo-cell), so its presence here already
93        // implies `purged_to_empty == false` and the row below is genuinely
94        // emitted with it intact.
95        let retained_cell_tombstones = surviving
96            .iter()
97            .filter(|cell| KWayMerger::<NoTrace>::is_cell_tombstone(cell) || cell.is_deleted)
98            .count() as u64;
99        purges.emitted += retained_cell_tombstones;
100
101        // Attach the carried deletion metadata to whichever entry is emitted so it
102        // is not dropped by reconciliation (#886 plumbing preservation).
103        let has_carried_metadata = !complex_deletions.is_empty() || range_deletion.is_some();
104        // #4581: a clustering-less row whose marker survived shadowing and purging is a live row
105        // whatever else happened to its cells (`Row.Merger.merge` returns null only when liveness,
106        // deletion AND data are all empty, Row.java:778; `BTreeRow.purge` keeps a live marker,
107        // BTreeRow.java:464). A clustered row's marker rides its clustering pseudo-cells instead.
108        let marker_row = clustering_key.is_none() && row_liveness.has_marker;
109
110        // Emit a live row only when real data survives. `!purged_to_empty`
111        // suppresses a clustered row whose only data was a dropped column (phantom
112        // key-only row); a genuine row marker (always key-only) has
113        // `had_data_before == false` so it is preserved.
114        let built = if (!surviving.is_empty() && !purged_to_empty && !marker_gone) || marker_row {
115            // Newest surviving cell; a marker-only row (no cells) takes the marker's own timestamp
116            // (never 0 — the writer would emit ts=0, #4581).
117            let row_ts = surviving
118                .iter()
119                .map(|c| c.timestamp)
120                .max()
121                .or(row_liveness.marker_timestamp)
122                .unwrap_or(0);
123            let live = MergeEntry::new(
124                run_index,
125                key,
126                clustering_key,
127                row_ts,
128                RowData::Live { cells: surviving },
129            );
130            // Issue #932: a row deletion that survived purging COEXISTS with the
131            // surviving (strictly-newer) cells. Carry it on the live entry so the
132            // merge→mutation step emits a `HAS_DELETION` row holding both — the row
133            // deletion keeps shadowing older cells of OTHER columns living in
134            // SSTables not part of a partial compaction, preventing resurrection.
135            // Step 3 already dropped the cells this deletion covers (ts <= row_del),
136            // so the deletion shadows nothing within `surviving`.
137            Some(match row_del {
138                Some(deletion_time) => {
139                    // Issue #2163: this row-tombstone marker is RETAINED alongside
140                    // newer live cells (the common "retained-but-coexisting" case,
141                    // NOT the sole-output case below) — count it as emitted too, so
142                    // `tombstones_emitted` covers every marker that survives
143                    // reconciliation into the output, not only the row-absent case.
144                    purges.emitted += 1;
145                    live.with_row_deletion(deletion_time, row_del_ldt)
146                }
147                None => live,
148            })
149        } else if let Some(deletion_time) = row_del {
150            // No surviving data. If a row tombstone exists, keep the row shadowed
151            // so downstream still emits the deletion (preserves #505/#498 absence).
152            // Issue #2163: this row-tombstone marker is RETAINED into the output —
153            // count it as emitted (a marker carried forward, not purged).
154            purges.emitted += 1;
155            Some(MergeEntry::new(
156                run_index,
157                key,
158                clustering_key,
159                deletion_time,
160                RowData::Tombstone {
161                    deletion_time,
162                    // Preserve the source LDT paired with the winning deletion_time
163                    // (#873) instead of resetting it to 0; the writer encodes it
164                    // verbatim and gc_grace decisions stay faithful.
165                    local_deletion_time: row_del_ldt,
166                },
167            ))
168        } else if has_carried_metadata {
169            // No data/row tombstone, but carried deletion metadata (#844/#846/#899): an empty `Live` row.
170            Some(MergeEntry::new(
171                run_index,
172                key,
173                clustering_key,
174                0,
175                RowData::Live { cells: vec![] },
176            ))
177        } else {
178            // Truly empty/absent row.
179            None
180        };
181
182        built.map(|entry| {
183            // Issue #2374/#2789: carry the reconciled row-marker liveness onto the
184            // emitted entry for the read path's visibility decision (carry-only).
185            let entry = entry.with_row_liveness(row_liveness);
186            let entry = if complex_deletions.is_empty() {
187                entry
188            } else {
189                entry.with_complex_deletions(complex_deletions)
190            };
191            match range_deletion {
192                Some(rd) => entry.with_range_deletion(rd),
193                None => entry,
194            }
195        })
196    }
