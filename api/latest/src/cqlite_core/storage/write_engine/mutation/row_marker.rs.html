1//! The authoritative primary-key liveness marker of a COMPACTED row (issue #4581).
2//!
3//! A flush derives a row's liveness from the cells it writes. A compaction instead READS the
4//! marker off the inputs, and it must re-emit exactly that marker: its own write timestamp, its
5//! TTL and its on-disk `localExpirationTime` (Cassandra `LivenessInfo`, `cassandra-5.0.8`
6//! `db/LivenessInfo.java:42-72`; serialized by `UnfilteredSerializer.serializeRowBody`). Deriving
7//! them from the cells (or from the compaction wall clock) is what wrote a marker-only row with
8//! ts=0 and no TTL.
9
10use crate::storage::sstable::reader::compaction_row::RowLiveness;
11
12/// Primary-key liveness marker carried verbatim onto a [`Mutation`](super::Mutation).
13#[derive(Debug, Clone, Copy, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
14pub struct RowMarker {
15    /// The marker's write timestamp (µs), `LivenessInfo.timestamp()`.
16    pub timestamp_micros: i64,
17    /// `LivenessInfo.ttl()` in seconds; `None` for a marker with no TTL.
18    pub ttl_seconds: Option<u32>,
19    /// `LivenessInfo.localExpirationTime()` (GC-clock seconds, the unsigned on-disk value carried
20    /// as a wrapped `i32`). `None` when the marker has no TTL, or the source did not surface it
21    /// (the writer then keeps its historical `now + ttl` derivation).
22    pub local_expiration_time: Option<i32>,
23}
24
25/// Who decides a mutation's primary-key liveness marker (#4581).
26///
27/// A flush or WAL replay derives it from the cells it writes ([`MarkerMode::Derive`]). A
28/// compaction READS the reconciled marker off the inputs and re-emits exactly that, including
29/// "no marker": an UPDATE-created row, a marker purged by `purge_marker` or a marker shadowed by
30/// a deletion all serialize `LivenessInfo.EMPTY` (no `HAS_TIMESTAMP`), however new the surviving
31/// cells are (`UnfilteredSerializer.serializeRowBody` sets the flags only when
32/// `!pkLiveness.isEmpty()`; `BTreeRow.purge` replaces a purgeable marker with `EMPTY`).
33#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
34pub enum MarkerMode {
35    /// The writer derives liveness from the mutation's cells (flush / WAL).
36    #[default]
37    Derive,
38    /// The compaction path's reconciled marker, `None` meaning the row has no marker. The writer
39    /// never derives liveness from cells for such a mutation.
40    Explicit(Option<RowMarker>),
41}
42
43impl MarkerMode {
44    /// The explicit marker, if this mode carries one.
45    pub fn marker(self) -> Option<RowMarker> {
46        match self {
47            Self::Derive => None,
48            Self::Explicit(m) => m,
49        }
50    }
51
52    /// True when the compaction path decided the marker (possibly "none").
53    pub fn is_explicit(self) -> bool {
54        matches!(self, Self::Explicit(_))
55    }
56}
57
58impl RowMarker {
59    /// The marker a reconciled row carries, or `None` when it has no marker (an UPDATE-created
60    /// row) or the source did not surface the marker's write timestamp. Authoritative on-disk
61    /// fields only (no-heuristics, #28).
62    pub fn from_liveness(liveness: &RowLiveness) -> Option<Self> {
63        let timestamp_micros = liveness.marker_timestamp.filter(|_| liveness.has_marker)?;
64        let ttl_seconds = liveness
65            .ttl_seconds
66            .and_then(|t| u32::try_from(t).ok())
67            .filter(|t| *t != 0);
68        Some(Self {
69            timestamp_micros,
70            ttl_seconds,
71            local_expiration_time: ttl_seconds
72                .and(liveness.expires_at_seconds)
73                .map(|e| e as u32 as i32),
74        })
75    }
76}
