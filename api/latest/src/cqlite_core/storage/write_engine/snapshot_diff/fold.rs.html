21pub(super) fn fmt_micros(micros: i64) -> String {
22    chrono::DateTime::<chrono::Utc>::from_timestamp_micros(micros)
23        .map(|date| date.to_rfc3339_opts(chrono::SecondsFormat::AutoSi, true))
24        .unwrap_or_else(|| micros.to_string())
25}
26
27pub(super) fn fmt_secs(secs: i64) -> String {
28    chrono::DateTime::<chrono::Utc>::from_timestamp(secs, 0)
29        .map(|date| date.to_rfc3339_opts(chrono::SecondsFormat::Secs, true))
30        .unwrap_or_else(|| secs.to_string())
31}

33pub(super) fn fmt_clustering(key: &ClusteringKey) -> String {
34    key.columns
35        .iter()
36        .map(|(_, value)| ValueFormatter::format_value(value))
37        .collect::<Vec<_>>()
38        .join(", ")
39}

41fn fmt_bound(bound: &ClusteringBound) -> String {
42    match bound {
43        ClusteringBound::Inclusive(key) => format!("inclusive:{}", fmt_clustering(key)),
44        ClusteringBound::Exclusive(key) => format!("exclusive:{}", fmt_clustering(key)),
45        ClusteringBound::Bottom => "bottom".to_string(),
46        ClusteringBound::Top => "top".to_string(),
47    }
48}

50fn kind_name(kind: TombstoneKind) -> &'static str {
51    match kind {
52        TombstoneKind::Partition => "partition",
53        TombstoneKind::Range => "range",
54        TombstoneKind::Row => "row",
55        TombstoneKind::Cell => "cell",
56        TombstoneKind::Collection => "collection",
57    }
58}

60/// Total order on optional clustering keys: static (`None`) first, then schema order.
61pub(super) fn cmp_clustering(
62    a: &Option<ClusteringKey>,
63    b: &Option<ClusteringKey>,
64    schema: &TableSchema,
65) -> Ordering {
66    match (a, b) {
67        (None, None) => Ordering::Equal,
68        (None, Some(_)) => Ordering::Less,
69        (Some(_), None) => Ordering::Greater,
70        (Some(a), Some(b)) => a.compare(b, schema).unwrap_or(Ordering::Equal),
71    }
72}

74/// Identity of a tombstone for the both-sides test (spec R3): kind, scope, deletion time and local
75/// deletion time. A range tombstone's scope is its span; the `clustering` of the entry that
76/// happened to carry it is not part of its identity. `run_index` and `droppable_at_now` are
77/// per-side facts and are not part of it either.
78fn tombstone_identity(tombstone: &TombstoneRecord) -> String {
79    let clustering = if tombstone.kind == TombstoneKind::Range {
80        None
81    } else {
82        tombstone.clustering.as_ref()
83    };
84    // `Debug` is infallible and exact for these plain-data types, so two distinct tombstones can
85    // never collapse onto one identity (a fallible serializer defaulting to "" could).
86    format!(
87        "{:?}",
88        (
89            kind_name(tombstone.kind),
90            clustering,
91            &tombstone.column,
92            &tombstone.cell_path,
93            &tombstone.range_start,
94            &tombstone.range_end,
95            tombstone.deletion_time,
96            tombstone.local_deletion_time,
97        )
98    )
99}

101fn distinct_tombstones(trail: &SideTrail) -> BTreeMap<String, &TombstoneRecord> {
102    let mut out = BTreeMap::new();
103    for tombstone in &trail.tombstones {
104        out.entry(tombstone_identity(tombstone))
105            .or_insert(tombstone);
106    }
107    out
108}

110fn winners(trail: &SideTrail) -> Vec<&CellDecision> {
111    trail
112        .cells
113        .iter()
114        .filter(|cell| cell.verdict == Verdict::Winner)
115        .collect()
116}

118fn tombstone_entry(
119    tombstone: &TombstoneRecord,
120    holder: Side,
121    holder_winners: &[&CellDecision],
122    other_winners: &[&CellDecision],
123    gc_before: i64,
124    schema: &TableSchema,
125) -> crate::Result<TombstoneDiffEntry> {
126    let mut shadowed: Vec<(Option<ClusteringKey>, ShadowedCell)> = Vec::new();
127    for cell in other_winners {
128        if !shadows(tombstone, cell, schema)? {
129            continue;
130        }
131        // Not at risk when the holder itself holds a live cell at the same key that wins the merge
132        // against the remote value (`Cells.resolveRegular`): the remote cannot resurrect.
133        let mut superseded = false;
134        for own in holder_winners.iter().filter(|own| {
135            own.clustering == cell.clustering
136                && own.column == cell.column
137                && own.cell_path == cell.cell_path
138        }) {
139            superseded |= super::cells::holder_supersedes(own, cell)?;
140        }
141        if !superseded {
142            shadowed.push((
143                cell.clustering.clone(),
144                ShadowedCell {
145                    clustering: cell.clustering.as_ref().map(fmt_clustering),
146                    column: cell.column.clone(),
147                    cell_path: cell.cell_path.as_ref().map(hex::encode),
148                    side: holder.other(),
149                },
150            ));
151        }
152    }
153    shadowed.sort_by(|x, y| {
154        cmp_clustering(&x.0, &y.0, schema).then_with(|| {
155            x.1.column
156                .cmp(&y.1.column)
157                .then_with(|| x.1.cell_path.cmp(&y.1.cell_path))
158        })
159    });
160    let mut shadowed_cells: Vec<ShadowedCell> =
161        shadowed.into_iter().map(|(_, cell)| cell).collect();
162    shadowed_cells.dedup();
163    let resurrection_risk = if shadowed_cells.is_empty() {
164        ResurrectionRisk {
165            status: "no",
166            phase: None,
167            shadowed_cells,
168        }
169    } else {
170        ResurrectionRisk {
171            status: "yes",
172            phase: Some(gc_phase(tombstone.local_deletion_time, gc_before)),
173            shadowed_cells,
174        }
175    };
176    let is_range = tombstone.kind == TombstoneKind::Range;
177    Ok(TombstoneDiffEntry {
178        kind: kind_name(tombstone.kind),
179        clustering: if is_range {
180            None
181        } else {
182            tombstone.clustering.as_ref().map(fmt_clustering)
183        },
184        column: tombstone.column.clone(),
185        cell_path: tombstone.cell_path.as_ref().map(hex::encode),
186        range_start: tombstone.range_start.as_ref().map(fmt_bound),
187        range_end: tombstone.range_end.as_ref().map(fmt_bound),
188        side: holder,
189        deletion_time: fmt_micros(tombstone.deletion_time),
190        local_deletion_time: fmt_secs(i64::from(tombstone.local_deletion_time as u32)),
191        resurrection_risk,
192    })
193}

195/// Compare the two sides' tombstones (spec R3) and classify each one-sided tombstone's
196/// resurrection risk against the OTHER side's winners (spec R4 / R4A / R5).
197fn fold_tombstones(
198    a: &SideTrail,
199    b: &SideTrail,
200    gc_before: i64,
201    schema: &TableSchema,
202) -> crate::Result<Vec<TombstoneDiffEntry>> {
203    let a_set = distinct_tombstones(a);
204    let b_set = distinct_tombstones(b);
205    let (a_winners, b_winners) = (winners(a), winners(b));
206    let mut out: Vec<(&TombstoneRecord, TombstoneDiffEntry)> = Vec::new();
207    for (identity, tombstone) in &a_set {
208        if !b_set.contains_key(identity) {
209            out.push((
210                tombstone,
211                tombstone_entry(
212                    tombstone,
213                    Side::A,
214                    &a_winners,
215                    &b_winners,
216                    gc_before,
217                    schema,
218                )?,
219            ));
220        }
221    }
222    for (identity, tombstone) in &b_set {
223        if !a_set.contains_key(identity) {
224            out.push((
225                tombstone,
226                tombstone_entry(
227                    tombstone,
228                    Side::B,
229                    &b_winners,
230                    &a_winners,
231                    gc_before,
232                    schema,
233                )?,
234            ));
235        }
236    }
237    // Deterministic, label-independent order (so a swapped diff lists them identically).
238    out.sort_by(|(x, _), (y, _)| {
239        kind_rank(x.kind)
240            .cmp(&kind_rank(y.kind))
241            .then_with(|| cmp_clustering(&x.clustering, &y.clustering, schema))
242            .then_with(|| x.column.cmp(&y.column))
243            .then_with(|| x.deletion_time.cmp(&y.deletion_time))
244            .then_with(|| x.local_deletion_time.cmp(&y.local_deletion_time))
245            .then_with(|| tombstone_identity(x).cmp(&tombstone_identity(y)))
246    });
247    Ok(out.into_iter().map(|(_, entry)| entry).collect())
248}
249
250fn kind_rank(kind: TombstoneKind) -> u8 {
251    match kind {
252        TombstoneKind::Partition => 0,
253        TombstoneKind::Range => 1,
254        TombstoneKind::Row => 2,
255        TombstoneKind::Collection => 3,
256        TombstoneKind::Cell => 4,
257    }
258}

260/// Fold one partition's two trails into its [`PartitionDiff`].
261pub(super) fn fold_partition(
262    partition_key: String,
263    a: &SideTrail,
264    b: &SideTrail,
265    gc_before: i64,
266    schema: &TableSchema,
267) -> std::result::Result<PartitionDiff, DiffError> {
268    let cells = super::cells::fold_cells(&partition_key, a, b, schema)?;
269    let tombstones = fold_tombstones(a, b, gc_before, schema)?;
270    Ok(PartitionDiff {
271        partition_key,
272        cells,
273        tombstones,
274    })
275}

277impl DiffSummary {
278    /// Count one folded partition into the running summary.
279    pub(super) fn add(&mut self, partition: &PartitionDiff) {
280        self.partitions_compared += 1;
281        for cell in &partition.cells {
282            match cell.status {
283                CellStatus::Equal => self.equal += 1,
284                CellStatus::OnlyA => self.only_a += 1,
285                CellStatus::OnlyB => self.only_b += 1,
286                CellStatus::NewerInA => self.newer_in_a += 1,
287                CellStatus::NewerInB => self.newer_in_b += 1,
288                CellStatus::Conflict => self.conflicts += 1,
289            }
290        }
291        for tombstone in &partition.tombstones {
292            match tombstone.side {
293                Side::A => self.tombstones_only_a += 1,
294                Side::B => self.tombstones_only_b += 1,
295            }
296            match tombstone.resurrection_risk.phase {
297                Some("before-gc-grace") => self.resurrection_risk_before_gc_grace += 1,
298                Some("past-gc-grace") => self.resurrection_risk_past_gc_grace += 1,
299                _ => {}
300            }
301        }
302    }
