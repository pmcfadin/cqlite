1//! `CRC.db` writer — per-chunk CRC32 for uncompressed BIG SSTables.
2//!
3//! Cassandra 5.0 writes a `CRC.db` component for every **uncompressed** BIG
4//! (`nb`) SSTable, alongside (not instead of) `Digest.crc32`. Compressed tables
5//! carry per-chunk CRCs inline after each compressed chunk and describe them in
6//! `CompressionInfo.db`, so they have no `CRC.db`. BTI (`da`) tables likewise do
7//! not emit `CRC.db` (verified against the Cassandra-written `da` fixtures).
8//!
9//! # On-disk format (oracle: Cassandra 5.0)
10//!
11//! Produced by `ChecksummedSequentialWriter` via `ChecksumWriter`
12//! (`org.apache.cassandra.io.util.ChecksumWriter` /
13//! `ChecksummedSequentialWriter`):
14//!
15//! ```text
16//! [chunk size : 4 bytes, signed int, big-endian]   <- data writer buffer.capacity()
17//! [CRC32 chunk 0 : 4 bytes, big-endian]
18//! [CRC32 chunk 1 : 4 bytes, big-endian]
19//! ...
20//! ```
21//!
22//! - **Chunk size** is the data `SequentialWriter`'s `buffer.capacity()`, which
23//!   defaults to `64 * 1024` (`SequentialWriterOption.Builder.bufferSize`). The
24//!   real Cassandra fixtures store `0x00010000` (65536). `ChecksumWriter`
25//!   serializes it with `DataOutput.writeInt` (big-endian).
26//! - Each **CRC32** covers exactly one chunk of the **raw (uncompressed)**
27//!   Data.db bytes — `flushData()` checksums `buffer.position()` bytes per buffer
28//!   flush. The algorithm is `java.util.zip.CRC32` (IEEE / `crc32fast`); the
29//!   value is written as a big-endian `i32`/`u32` via `DataOutput.writeInt`.
30//! - The per-chunk CRC values are **not** folded into the data file's
31//!   `Digest.crc32` (`checksumIncrementalResult = false` for the uncompressed
32//!   path), so `Digest.crc32` remains the CRC32 over the raw data bytes only.
33//!   For a single-chunk file the lone `CRC.db` entry therefore equals the
34//!   `Digest.crc32` value.
35//!
36//! # Flush vs compaction: the trailing empty-chunk CRC32 (issue #1222)
37//!
38//! Cassandra's **compaction** write path (`CompactionAwareWriter` →
39//! `SSTableRewriter` → `BigTableWriter`) flushes the data `SequentialWriter` once
40//! more at close after the last real chunk, checksumming a ZERO-length buffer.
41//! `java.util.zip.CRC32` over zero bytes is `0`, so the compacted `CRC.db` carries
42//! one extra trailing `00000000` group after the last real per-chunk CRC32. The
43//! **flush** path does NOT do this (verified against the #1190 flush goldens,
44//! whose `CRC.db` ends exactly on the last real chunk CRC). The writer therefore
45//! takes an explicit [`CrcTrailer`] so the compaction path can request the
46//! trailing empty-chunk CRC32 without changing the byte-identical flush output.
47//!
48//! # Seek formula (read side)
49//!
50//! To locate the CRC for byte `offset` in Data.db:
51//! `crc_file_pos = (offset / chunk_size) * 4 + 4`
52//! (`DataIntegrityMetadata.ChecksumValidator`).
53

62/// Default uncompressed CRC chunk size, matching Cassandra's
63/// `SequentialWriterOption.Builder` default `bufferSize` of `64 * 1024`.
64///
65/// This is the value stored in the 4-byte `CRC.db` header and the block size
66/// over which each per-chunk CRC32 is computed.
67pub const CRC_CHUNK_SIZE: usize = 64 * 1024;
68
69/// Whether the `CRC.db` carries Cassandra's compaction-only trailing
70/// empty-final-chunk CRC32 (issue #1222).
71///
72/// The two SSTable write paths produce byte-different `CRC.db` tails:
73/// - [`CrcTrailer::None`] — the **flush** path: the file ends on the last real
74///   per-chunk CRC32. This is the byte-identical match for the #1190 flush
75///   goldens and MUST stay unchanged.
76/// - [`CrcTrailer::EmptyFinalChunk`] — the **compaction** path: Cassandra's
77///   `CompactionAwareWriter` flushes the data writer once more at close over a
78///   zero-length buffer, appending one trailing `00000000` (`CRC32` of zero
79///   bytes). This is the byte-identical match for the #1017 compacted goldens.
80#[derive(Debug, Clone, Copy, PartialEq, Eq)]
81pub(crate) enum CrcTrailer {
82    /// No trailing chunk — the flush path (default).
83    None,
84    /// Append one trailing empty-final-chunk `CRC32 = 0` — the compaction path.
85    EmptyFinalChunk,
86}

88/// Assemble the on-disk `CRC.db` bytes from a slice of per-chunk CRC32 values.
89///
90/// This is the single source of truth for the `CRC.db` byte layout (issue
91/// #1663): a big-endian `i32` chunk-size header followed by one big-endian `u32`
92/// CRC32 per chunk, plus the issue-#1222 compaction trailer rule. Both the
93/// streaming write path (checksums accumulated during the write — see
94/// [`StreamingCrc`]) and the re-read [`build_crc_bytes`] oracle route their
95/// chunk CRCs through here, so the two paths are provably byte-identical.
96///
97/// Trailer rule (issue #1222):
98/// - [`CrcTrailer::None`] (flush path) appends nothing after the last chunk CRC.
99/// - [`CrcTrailer::EmptyFinalChunk`] (compaction path) appends ONE trailing
100///   `CRC32 = 0` (`00000000`) — Cassandra's compaction close-time zero-length
101///   buffer flush — but ONLY when at least one real chunk CRC was produced
102///   (non-empty `Data.db`). An empty `Data.db` stays header-only for BOTH
103///   trailers (there is no Cassandra golden for the empty-compaction case).
104pub(crate) fn assemble_crc_bytes(chunk_crcs: &[u32], trailer: CrcTrailer) -> Vec<u8> {
105    // header (4) + one u32 per chunk + optional trailing empty-chunk CRC (4).
106    let mut out = Vec::with_capacity(4 + chunk_crcs.len() * 4 + 4);
107    // Header: chunk size as a big-endian signed 32-bit int (DataOutput.writeInt).
108    out.extend_from_slice(&(CRC_CHUNK_SIZE as i32).to_be_bytes());
109    for crc in chunk_crcs {
110        out.extend_from_slice(&crc.to_be_bytes());
111    }
112    // Compaction-only trailing empty-final-chunk CRC32 (issue #1222): CRC32 of a
113    // zero-length buffer is 0. Gated on a real chunk having been produced so an
114    // empty Data.db keeps the documented header-only output for both trailers.
115    if matches!(trailer, CrcTrailer::EmptyFinalChunk) && !chunk_crcs.is_empty() {
116        out.extend_from_slice(&0u32.to_be_bytes());
117    }
118    out
119}

121/// Incremental `CRC.db` + `Digest.crc32` accumulator for the streaming write
122/// path (issue #1663).
123///
124/// The streaming Data.db writer feeds every byte it writes to disk through
125/// [`update`](Self::update) exactly once, in write order. This computes the
126/// whole-file CRC32 (the `Digest.crc32` value) and the per-chunk CRC32s
127/// (`CRC.db`) as the data streams by, so `finish()` never has to re-read the
128/// finished `Data.db` to checksum it. The chunk boundaries are fixed
129/// `CRC_CHUNK_SIZE` blocks of the RAW Data.db bytes and span partition
130/// boundaries — identical to what the re-read [`build_crc_bytes`] oracle
131/// produces — because `update` carries `chunk_filled` across calls.
132#[derive(Debug, Default)]
133pub(crate) struct StreamingCrc {
134    /// Whole-file CRC32 hasher (the `Digest.crc32` value).
135    whole: crc32fast::Hasher,
136    /// CRC32 hasher for the chunk currently being filled.
137    chunk: crc32fast::Hasher,
138    /// Bytes already fed into `chunk` toward the current `CRC_CHUNK_SIZE` block.
139    chunk_filled: usize,
140    /// Finalized per-chunk CRC32s, in order.
141    chunk_crcs: Vec<u32>,
142}
143
144impl StreamingCrc {
145    /// Create an empty accumulator.
146    pub(crate) fn new() -> Self {
147        Self::default()
148    }
149
150    /// Feed the next run of just-written Data.db bytes, in write order.
151    ///
152    /// Updates the whole-file hasher and slices `bytes` into the fixed
153    /// `CRC_CHUNK_SIZE` chunk grid, finalizing a chunk CRC every time a full
154    /// block is completed. `chunk_filled` carries across calls so a chunk that
155    /// straddles two `update` calls (i.e. two partition flushes) is checksummed
156    /// as one block — matching `build_crc_bytes`'s chunk boundaries exactly.
157    pub(crate) fn update(&mut self, bytes: &[u8]) {
158        self.whole.update(bytes);
159        let mut remaining = bytes;
160        while !remaining.is_empty() {
161            let take = (CRC_CHUNK_SIZE - self.chunk_filled).min(remaining.len());
162            self.chunk.update(&remaining[..take]);
163            self.chunk_filled += take;
164            remaining = &remaining[take..];
165            if self.chunk_filled == CRC_CHUNK_SIZE {
166                let done = std::mem::replace(&mut self.chunk, crc32fast::Hasher::new());
167                self.chunk_crcs.push(done.finalize());
168                self.chunk_filled = 0;
169            }
170        }
171    }
172
173    /// Finalize the accumulator, returning `(digest_crc32, chunk_crcs)`.
174    ///
175    /// `digest_crc32` is the whole-file CRC32 over every fed byte (the
176    /// `Digest.crc32` value). `chunk_crcs` is one CRC32 per `CRC_CHUNK_SIZE`
177    /// block, including a final short chunk if `chunk_filled > 0`. Pass
178    /// `chunk_crcs` to [`assemble_crc_bytes`] to obtain the `CRC.db` bytes.
179    pub(crate) fn finalize(mut self) -> (u32, Vec<u32>) {
180        if self.chunk_filled > 0 {
181            self.chunk_crcs.push(self.chunk.finalize());
182        }
183        (self.whole.finalize(), self.chunk_crcs)
184    }
185}

187/// Compute the `CRC.db` bytes for an uncompressed `Data.db` by RE-READING it.
188///
189/// Streams `data_path` in `CRC_CHUNK_SIZE` blocks (bounded memory, independent
190/// of the total Data.db size) to build the per-chunk CRC32s, then delegates the
191/// byte layout to [`assemble_crc_bytes`] so this re-read oracle and the
192/// streaming [`StreamingCrc`] path are provably byte-identical.
193///
194/// Retained (issue #1663) as the golden oracle for the incremental path's
195/// byte-parity tests and to build a `CRC.db` for the reader-side tests; the
196/// production write path no longer re-reads Data.db and does NOT call this.
197/// Each invocation bumps the [`data_db_checksum_full_reads`] work counter so a
198/// regression that reintroduces the full-file re-read is caught.
199///
200/// [`data_db_checksum_full_reads`]: crate::storage::sstable::work_counters::data_db_checksum_full_reads
201#[cfg(test)]
202pub(super) async fn build_crc_bytes(data_path: &Path, trailer: CrcTrailer) -> Result<Vec<u8>> {
203    use tokio::io::AsyncReadExt;
204
205    let mut file = tokio::fs::File::open(data_path).await?;
206    // Full re-read of Data.db for checksums — count it (issue #1663).
207    crate::storage::sstable::work_counters::add_data_db_checksum_full_read();
208    let mut chunk_crcs = Vec::new();
209    let mut buffer = vec![0u8; CRC_CHUNK_SIZE];
210    let mut filled = 0usize;
211    loop {
212        // Fill up to a full chunk before checksumming so each CRC covers a
213        // complete CRC_CHUNK_SIZE block (except the final, short chunk), exactly
214        // as Cassandra's per-buffer flushData() does. A single short read does
215        // not imply EOF, so accumulate until the chunk is full or EOF is hit.
216        let n = file.read(&mut buffer[filled..]).await?;
217        if n == 0 {
218            if filled > 0 {
219                let mut hasher = crc32fast::Hasher::new();
220                hasher.update(&buffer[..filled]);
221                chunk_crcs.push(hasher.finalize());
222            }
223            break;
224        }
225        filled += n;
226        if filled == CRC_CHUNK_SIZE {
227            let mut hasher = crc32fast::Hasher::new();
228            hasher.update(&buffer[..filled]);
229            chunk_crcs.push(hasher.finalize());
230            filled = 0;
231        }
232    }
233
234    Ok(assemble_crc_bytes(&chunk_crcs, trailer))
235}

237/// Write a `CRC.db` component by re-reading `data_path` (test-only oracle,
238/// issue #1663).
239///
240/// The production write path assembles `CRC.db` from streaming-accumulated chunk
241/// CRCs (see [`assemble_crc_bytes`] / [`StreamingCrc`]) and no longer calls this;
242/// it is retained to build `CRC.db` fixtures for the reader-side tests.
243///
244/// `trailer` selects the flush vs compaction tail (issue #1222): flush callers
245/// pass [`CrcTrailer::None`]; compaction callers pass
246/// [`CrcTrailer::EmptyFinalChunk`].
247#[cfg(test)]
248pub(crate) async fn write_crc_db(
249    data_path: &Path,
250    crc_path: PathBuf,
251    trailer: CrcTrailer,
252) -> Result<PathBuf> {
253    let bytes = build_crc_bytes(data_path, trailer).await?;
254    tokio::fs::write(&crc_path, bytes).await?;
255    Ok(crc_path)
256}

392    // byte-identical to the re-read oracle for BOTH the flush (`CrcTrailer::None`)
393    // and compaction (`CrcTrailer::EmptyFinalChunk`) tails, across single-chunk,

399    /// Feed `payload` through a `StreamingCrc` in `split`-sized runs (simulating
400    /// per-partition flushes), then assert the assembled `CRC.db` bytes and the
401    /// whole-file digest equal the re-read oracle (`build_crc_bytes` /
402    /// `SSTableWriter::compute_crc32`) over a file holding the same bytes.
403    async fn assert_incremental_matches_reread(payload: &[u8], trailer: CrcTrailer, split: usize) {
404        use crate::storage::sstable::writer::SSTableWriter;
405
406        // Incremental path: accumulate as the bytes "stream" by in `split` runs.
407        let mut acc = StreamingCrc::new();
408        for run in payload.chunks(split.max(1)) {
409            acc.update(run);
410        }
411        let (inc_digest, inc_chunk_crcs) = acc.finalize();
412        let inc_crc_db = assemble_crc_bytes(&inc_chunk_crcs, trailer);
413
414        // Re-read oracle over a file holding the identical bytes.
415        let dir = tempfile::tempdir().expect("tempdir");
416        let data = dir.path().join("Data.db");
417        tokio::fs::write(&data, payload).await.expect("write data");
418        let reread_crc_db = build_crc_bytes(&data, trailer)
419            .await
420            .expect("build_crc_bytes");
421        let reread_digest = SSTableWriter::compute_crc32(&data)
422            .await
423            .expect("compute_crc32");
424
425        assert_eq!(
426            inc_crc_db, reread_crc_db,
427            "incremental CRC.db must equal re-read oracle (len={}, trailer={trailer:?}, split={split})",
428            payload.len()
429        );
430        assert_eq!(
431            inc_digest,
432            reread_digest,
433            "incremental Digest.crc32 must equal re-read oracle (len={}, split={split})",
434            payload.len()
435        );
436    }
