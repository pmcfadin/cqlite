1//! SSTable writer components for producing Cassandra 5.0-compatible SSTables
2//!
3//! This module coordinates the generation of all SSTable components:
4//! - Data.db: Row data with partition and clustering ordering
5//! - Index.db: Partition index for fast lookups
6//! - Filter.db: Bloom filter for existence checks
7//! - Statistics.db: Metadata for delta encoding
8//! - Summary.db: Sampled index entries
9//! - CompressionInfo.db: Compression metadata (only when compressed)
10//! - TOC.txt: Component manifest (publication barrier)
11//!
12//! Component generation order is critical (see M5 Council Recommendation):
13//! 1. Statistics.db (provides delta encoding baseline)
14//! 2. Data.db + Index.db (single pass, track offsets)
15//! 3. Summary.db (sample Index.db entries)
16//! 4. Filter.db (finalize Bloom filter)
17//! 5. CompressionInfo.db (only when compressed)
18//! 6. Digest.crc32
19//! 7. TOC.txt (makes SSTable visible)
20//!
21//! TODO: Implementation in M5.0-7 through M5.0-13

101/// On-disk index format emitted by [`SSTableWriter`].
102///
103/// Issue #766 (epic #762, writer fidelity D4). The default is [`Big`], which
104/// produces the legacy `Index.db`/`Summary.db` partition index and is byte-for-byte
105/// unchanged from before this option existed. [`Bti`] additionally writes a BTI
106/// `Partitions.db` trie (phase 1) so partition lookups can resolve `Data.db`
107/// offsets via the trie reader.
108///
109/// [`Big`]: SSTableFormat::Big
110/// [`Bti`]: SSTableFormat::Bti
111#[cfg(feature = "write-support")]
112#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
113pub enum SSTableFormat {
114    /// Legacy BIG format: `Index.db` + `Summary.db`. **Default.**
115    #[default]
116    Big,
117    /// Cassandra-canonical BTI format (`da-<gen>-bti-*`).
118    ///
119    /// Issue #908 (epic #872). Emits a true BTI component set:
120    /// - `Data.db` — the partition/row serialization is **identical** to BIG in
121    ///   Cassandra 5 (the BTI format shares `BigTableWriter`'s row encoding; only
122    ///   the descriptor/version and index components differ — see
123    ///   `docs/sstables-definitive-guide/chapters/17-bti-formats.md`, which lists
124    ///   `Data.db` as a *common component retained*). The bytes are therefore the
125    ///   same as BIG; only the filename descriptor changes.
126    /// - `Partitions.db` — partition trie (replaces `Index.db`/`Summary.db`).
127    /// - `Rows.db` — within-partition row-index trie (issue #910). WIDE
128    ///   partitions (`>= 2` × 64 KiB column-index blocks) get a per-partition
129    ///   row index and a positive `RowsOffset` in their partition-trie leaf;
130    ///   NARROW partitions keep a direct (negative) `Data.db` offset. `Rows.db`
131    ///   is always emitted for a BTI SSTable — possibly 0 bytes when no partition
132    ///   is wide, matching the real `da` fixtures.
133    /// - `Statistics.db`, `Filter.db`, `Digest.crc32`, `TOC.txt`.
134    ///
135    /// **No `Index.db` and no `Summary.db`** — those are BIG-only components; BTI
136    /// resolves partitions through the trie. All components use the `da` version
137    /// letter and `bti` format segment.
138    ///
139    /// An **empty** BTI SSTable (zero partitions) is refused by `finish()`: a
140    /// `da` SSTable requires a readable `Partitions.db` (8-byte root footer),
141    /// which has no valid zero-partition form.
142    Bti,
143}

198/// SSTable writer coordinator
199///
200/// Orchestrates the generation of all SSTable components in the correct order.
201/// Produces valid Cassandra 5.0 BIG format SSTables.
202///
203/// # Write Order
204///
205/// Components are written in the following critical order:
206/// 1. Statistics.db - Provides delta encoding baseline (FIRST)
207/// 2. Data.db - Main partition/row data
208/// 3. Index.db - Partition index (uses Data.db offsets)
209/// 4. Filter.db - Bloom filter
210/// 5. Summary.db - Sampled index entries
211/// 6. CompressionInfo.db - Compression metadata (only when compressed)
212/// 7. Digest.crc32 - Data.db checksum
213/// 8. TOC.txt - Table of contents (LAST, publication barrier)
214///
215/// # File Naming
216///
217/// All components follow the pattern: `nb-{generation}-big-{Component}.db`
218/// Example: `nb-1-big-Data.db`, `nb-1-big-Index.db`
219///
220/// # Partition Ordering
221///
222/// Partitions MUST be written in Murmur3 token order (caller responsibility).
223/// The writer validates token ordering on each `write_partition()` call.
224///
225/// # Example
226///
227/// ```rust,ignore
228/// use cqlite_core::storage::sstable::writer::SSTableWriter;
229/// use cqlite_core::storage::write_engine::mutation::{Mutation, DecoratedKey};
230/// use cqlite_core::schema::TableSchema;
231///
232/// // Create schema
233/// let schema = TableSchema::from_json("...")?;
234///
235/// // Create writer
236/// let mut writer = SSTableWriter::new(
237///     PathBuf::from("data/ks/table"),
238///     1,  // generation
239///     &schema
240/// )?;
241///
242/// // Write partitions (MUST be in token order)
243/// let key = DecoratedKey::new(token, key_bytes);
244/// let mutations = vec![/* ... */];
245/// writer.write_partition(key, mutations)?;
246///
247/// // Finish writing
248/// let info = writer.finish().await?;
249/// println!("Wrote SSTable with {} partitions", info.partition_count);
250/// ```
251#[cfg(feature = "write-support")]
252#[derive(Debug)]
253pub struct SSTableWriter {
254    /// SSTable output directory: output_dir/keyspace/table/
255    sstable_dir: PathBuf,
256    /// SSTable generation number
257    generation: u64,
258    /// Table schema for column metadata
259    schema: TableSchema,
260    /// Statistics metadata (collected during writes)
261    stats: StatisticsMetadata,
262    /// Data.db writer
263    data_writer: DataWriter,
264    /// Index.db writer
265    index_writer: IndexWriter,
266    /// Filter.db writer
267    filter_writer: Option<FilterWriter>,
268    /// Summary.db writer
269    summary_writer: SummaryWriter,
270    /// Last token written (for ordering validation)
271    last_token: Option<i64>,
272    /// Number of partitions written
273    partition_count: usize,
274    /// Summary sampling counter (sample every N entries)
275    summary_sample_counter: usize,
276    /// Sampling interval for Summary.db (default: 128)
277    summary_sample_interval: usize,
278    /// Whether encoding baselines have been pre-seeded via `pre_seed_encoding_baselines`.
279    ///
280    /// When `true`, `write_partition` skips the incremental `data_writer.update_stats()`
281    /// call so the pre-computed final baselines are not overwritten by an intermediate
282    /// (and potentially higher) baseline from an earlier partition.
283    ///
284    /// Issue #729: two-pass flush baseline fix.
285    baselines_locked: bool,
286    /// Index format to emit (issue #766). Defaults to [`SSTableFormat::Big`].
287    format: SSTableFormat,
288    /// BTI partition trie accumulator. Populated only when `format` is
289    /// [`SSTableFormat::Bti`]; `None` otherwise so the BIG path allocates nothing.
290    partitions_trie: Option<partitions_writer::PartitionsTrieWriter>,
291    /// BTI per-partition pending payloads (issue #910). Populated only for
292    /// [`SSTableFormat::Bti`]. The partition-trie leaf payload (direct
293    /// `Data.db` offset vs `Rows.db` `RowsOffset`) cannot be finalized until
294    /// `Rows.db` is serialized in [`Self::finish`] (the `RowsOffset` is the
295    /// `TrieIndexEntry` position), so we defer the decision: each entry records
296    /// the partition's raw key, its `Data.db` offset, and — for WIDE partitions
297    /// (>= 2 column-index blocks) — the row-index blocks. `None` for BIG so that
298    /// path allocates nothing. See [`bti_state::PendingBtiPartition`].
299    bti_pending: Option<Vec<bti_state::PendingBtiPartition>>,
300    /// Whether this writer produces a COMPACTION output (issue #1222).
301    ///
302    /// `false` for the default FLUSH path. When `true`, [`Self::finish`] emits
303    /// the uncompressed-BIG `CRC.db` with Cassandra's compaction-only trailing
304    /// empty-final-chunk `CRC32 = 0` (see [`crc_writer::CrcTrailer`]); the flush
305    /// path's `CRC.db` is byte-for-byte unchanged. Set via
306    /// [`Self::mark_compaction_output`] by the two compaction producers
307    /// (`compact_sstables` and the WriteEngine background compactor).
308    is_compaction_output: bool,
309}

430    /// Create a new SSTable writer selecting the on-disk index `format` and an
431    /// optional [`UdtRegistry`] for bare-UDT resolution (issue #766 + #929).
432    pub fn with_format_and_registry(
433        output_dir: PathBuf,
434        generation: u64,
435        schema: &TableSchema,
436        expected_partitions: usize,
437        format: SSTableFormat,
438        registry: Option<&UdtRegistry>,
439    ) -> Result<Self> {
440        // Issue #929: resolve TOP-LEVEL bare UDT column names to their marshal
441        // form on the schema copy the writer holds, so the existing complex-cell
442        // path treats them as multi-cell UDTs. No registry => no rewrite.
443        let mut schema = schema.clone();
444        if let Some(registry) = registry {
445            crate::storage::sstable::writer::data_writer::normalize_schema_udts(
446                &mut schema,
447                registry,
448            );
449        }
450        let schema = &schema;
451
452        // Initialize statistics metadata with sentinel values
453        let mut stats = StatisticsMetadata::new();
454        // Pre-set min values to reasonable defaults (will be updated during writes)
455        stats.min_timestamp = i64::MAX;
456        stats.min_ttl = i32::MAX;
457        stats.min_local_deletion_time = i32::MAX;
458
459        // Compute the SSTable output directory: output_dir/keyspace/table/
460        // This ensures the reader's extract_table_name() can map files to table names.
461        let sstable_dir = output_dir.join(&schema.keyspace).join(&schema.table);
462
463        // Create Data.db writer in streaming mode (Issue #492): each partition is
464        // flushed to the Data.db file as it is written, bounding peak memory to a
465        // single partition instead of buffering the whole component. The file is
466        // opened lazily on the first partition (creating sstable_dir as needed).
467        let data_path = Self::component_path_for(&sstable_dir, generation, format, "Data.db");
468        // Issue #1741: `da` (BTI) SSTables must serialize the partition-level
469        // `DeletionTime` in the oa layout so the reader's `hasUIntDeletionTime`
470        // branch reads the LIVE sentinel correctly (a legacy `nb` layout in a `da`
471        // file is misread as a partition tombstone). `nb` (BIG) keeps the legacy
472        // layout, byte-identical to before.
473        let data_writer = DataWriter::with_sink(stats.clone(), data_path)
474            .with_oa_partition_deletion(matches!(format, SSTableFormat::Bti));
475
476        // Index.db writer.
477        //
478        // BIG (Issue #753): streaming mode flushes each entry straight to
479        // `Index.db` as it arrives, keeping only the current entry's bytes in
480        // memory (O(1) in partition count).
481        //
482        // BTI (Issue #908): the BTI format has no `Index.db` — partition lookups
483        // go through the `Partitions.db` trie instead. We use a *counting-only*
484        // `IndexWriter` (no sink, so no file is created) purely to compute the
485        // per-partition index offsets and promoted-block bookkeeping reused by
486        // the write path. Counting mode serializes each entry into a scratch buffer
487        // only to measure it, then clears the scratch, so peak memory stays
488        // O(one entry). (The prior in-memory mode retained every serialized entry
489        // forever — a full, never-emitted in-memory `Index.db` that defeated the
490        // streaming memory budget on large BTI writes.) Nothing is persisted and no
491        // `Index.db` is emitted.
492        let index_writer = match format {
493            SSTableFormat::Big => {
494                let index_path =
495                    Self::component_path_for(&sstable_dir, generation, format, "Index.db");
496                IndexWriter::with_sink(index_path)
497            }
498            SSTableFormat::Bti => IndexWriter::counting(),
499        };
500
501        // Create Filter.db writer. The false-positive chance comes from the
502        // table's `bloom_filter_fp_chance` (Issue #852): thread the schema's
503        // actual value through instead of hardcoding 0.01. A value of exactly
504        // 1.0 disables the filter (Cassandra's AlwaysPresentFilter) and the
505        // FilterWriter then emits no Filter.db component; the default when the
506        // schema does not specify one remains Cassandra's 0.01.
507        let filter_path = Self::component_path_for(&sstable_dir, generation, format, "Filter.db");
508        let fp_chance = Self::bloom_filter_fp_chance(schema);
509        let filter_writer = Some(FilterWriter::new(
510            filter_path,
511            expected_partitions.max(1),
512            fp_chance,
513        )?);
514
515        // Create Summary.db writer (sample every 128 entries per Cassandra default)
516        let summary_sample_interval = 128;
517        let summary_writer = SummaryWriter::new(summary_sample_interval as u32);
518
519        // BTI phase 1 (issue #766): only allocate the partition-trie accumulator
520        // when the BTI format is selected, so the default BIG path is unchanged.
521        let partitions_trie = match format {
522            SSTableFormat::Big => None,
523            SSTableFormat::Bti => Some(partitions_writer::PartitionsTrieWriter::new()),
524        };
525        // BTI (issue #910): defer partition-trie payloads until Rows.db is built.
526        let bti_pending = match format {
527            SSTableFormat::Big => None,
528            SSTableFormat::Bti => Some(Vec::new()),
529        };
530
531        Ok(Self {
532            sstable_dir,
533            generation,
534            schema: schema.clone(),
535            stats,
536            data_writer,
537            index_writer,
538            filter_writer,
539            summary_writer,
540            last_token: None,
541            partition_count: 0,
542            summary_sample_counter: 0,
543            summary_sample_interval,
544            baselines_locked: false,
545            format,
546            partitions_trie,
547            bti_pending,
548            is_compaction_output: false,
549        })
550    }

557    /// Mark this writer as producing a COMPACTION output (issue #1222).
558    ///
559    /// Cassandra's compaction write path appends one trailing empty-final-chunk
560    /// `CRC32 = 0` to the uncompressed-BIG `CRC.db` (its close-time zero-length

570    /// Write a partition (partition key + all mutations)
571    ///
572    /// # Arguments
573    ///
574    /// * `key` - DecoratedKey (token + raw partition key bytes)
575    /// * `mutations` - All mutations for this partition (must be in clustering order)
576    ///
577    /// # Returns
578    ///
579    /// Ok(()) on success, or an error if:
580    /// - Partitions are not in token order
581    /// - Schema validation fails
582    /// - I/O error occurs
583    ///
584    /// # Ordering Requirement
585    ///
586    /// Partitions MUST be written in ascending token order. This method validates
587    /// ordering and returns an error if violated.
588    ///
589    /// # Example
590    ///
591    /// ```rust,ignore
592    /// let key = DecoratedKey::new(12345, vec![0x00, 0x01, 0x02]);
593    /// let mutations = vec![
594    ///     Mutation::new(/* ... */)
595    /// ];
596    /// writer.write_partition(key, mutations)?;
597    /// ```
598    #[tracing::instrument(name = "writer.write_partition", skip(self, key, mutations))]
599    pub fn write_partition(&mut self, key: DecoratedKey, mutations: Vec<Mutation>) -> Result<()> {
600        // Validate token ordering
601        if let Some(last_token) = self.last_token {
602            if key.token <= last_token {
603                return Err(Error::InvalidInput(format!(
604                    "Partitions must be written in token order: got token {} after {}",
605                    key.token, last_token
606                )));
607            }
608        }
609        self.last_token = Some(key.token);
610
611        // Record the SSTable key range (lowest = first seen, highest = last) for
612        // the `da`-format StatsMetadata `hasKeyRange` fields. Partitions arrive in
613        // ascending token order (validated above), so first/last fall out for
614        // free. Harmless for BIG (the legacy STATS body ignores these fields).
615        self.stats.update_key_range(&key.key);
616
617        // Sort mutations by clustering key (Cassandra requires sorted rows within partitions)
618        let mut mutations = mutations;
619        mutations.sort_by(|a, b| match (&a.clustering_key, &b.clustering_key) {
620            (None, None) => std::cmp::Ordering::Equal,
621            (None, Some(_)) => std::cmp::Ordering::Less,
622            (Some(_), None) => std::cmp::Ordering::Greater,
623            (Some(ck_a), Some(ck_b)) => ck_a
624                .compare(ck_b, &self.schema)
625                .unwrap_or_else(|_| ck_a.cmp(ck_b)),
626        });
627
628        // Update statistics from mutations
629        for mutation in &mutations {
630            self.stats.update_timestamp(mutation.timestamp_micros);
631            // Issue #1018: simple `Write`/`WriteWithTtl`/`Delete` cells may carry
632            // their OWN (lower) per-cell timestamps in
633            // `Mutation::cell_write_timestamps` (a live cell's writetime OR a cell
634            // tombstone's markedForDeleteAt) and are emitted with an explicit
635            // `min_timestamp` delta. On the non-preseeded (incremental) write path,
636            // fold every per-cell timestamp into the stats BEFORE emitting cells so
637            // `min_timestamp` can never exceed an emitted cell's actual timestamp
638            // (which would underflow the unsigned-VInt delta). Mirrors the pre-pass
639            // fold in `compute_mutations_baseline_stats`.
640            if let Some(cell_ts) = &mutation.cell_write_timestamps {
641                for ts in cell_ts.values() {
642                    self.stats.update_timestamp(*ts);
643                }
644            }
645            if let Some(ttl) = mutation.ttl_seconds {
646                self.stats.update_ttl(ttl as i32);
647                let now_seconds = std::time::SystemTime::now()
648                    .duration_since(std::time::UNIX_EPOCH)
649                    .map(|d| d.as_secs() as i32)
650                    .unwrap_or(0);
651                let local_deletion_time = now_seconds.saturating_add(ttl as i32);
652                self.stats.update_local_deletion_time(local_deletion_time);
653            }
654            // Track local deletion times for tombstones and TTL cells.
655            // Issue #764: row/cell tombstones use the caller-supplied
656            // `local_deletion_time` when present, else the timestamp-derived
657            // value (`effective_local_deletion_time`).
658            for op in &mutation.operations {
659                match op {
660                    crate::storage::write_engine::mutation::CellOperation::WriteWithTtl {
661                        ttl_seconds,
662                        ..
663                    } => {
664                        // Track TTL
665                        self.stats.update_ttl(*ttl_seconds as i32);
666                        // CRITICAL: TTL cells need local_deletion_time tracked
667                        // local_deletion_time = now + ttl
668                        let now_seconds = std::time::SystemTime::now()
669                            .duration_since(std::time::UNIX_EPOCH)
670                            .map(|d| d.as_secs() as i32)
671                            .unwrap_or(0);
672                        let local_deletion_time = now_seconds.saturating_add(*ttl_seconds as i32);
673                        self.stats.update_local_deletion_time(local_deletion_time);
674                    }
675                    op @ (crate::storage::write_engine::mutation::CellOperation::Delete { .. }
676                    | crate::storage::write_engine::mutation::CellOperation::DeleteRow) => {
677                        // Issue #764 / #921 finding 2: record the EXACT LDT the
678                        // tombstone is emitted with. A `Delete` carrying a per-cell
679                        // `local_deletion_time: Some(L)` is stamped with `L`
680                        // verbatim by DataWriter; everything else derives the LDT
681                        // from the mutation. Reuse `op_cell_local_deletion_time`
682                        // (the emit path's helper) so stats and the bytes written
683                        // to Data.db agree exactly — a per-cell `L` below the
684                        // mutation-derived value would otherwise underflow the
685                        // delta, and one above it would leave min/max wrong.
686                        let local_deletion_time =
687                            crate::storage::sstable::writer::data_writer::op_cell_local_deletion_time(
688                                op, mutation,
689                            );
690                        self.stats.update_local_deletion_time(local_deletion_time);
691                    }
692                    // Issue #887: a `ComplexDeletion` marker is physically written
693                    // with its OWN `marked_for_delete_at` / `local_deletion_time`
694                    // (DataWriter delta-encodes both against `min_timestamp` /
695                    // `min_local_deletion_time`). The mutation's row timestamp does
696                    // NOT cover the marker — for a tombstone-only row whose marker
697                    // strictly supersedes the row tombstone, `marked_for_delete_at`
698                    // can lie ABOVE the row timestamp and the marker LDT below the
699                    // row's LDT. Fold both into the stats so `max_timestamp` /
700                    // `min_local_deletion_time` reflect what was actually emitted to
701                    // Data.db (LIVE sentinels are filtered inside the `update_*`
702                    // chokepoints, issue #851).
703                    crate::storage::write_engine::mutation::CellOperation::ComplexDeletion {
704                        marked_for_delete_at,
705                        local_deletion_time,
706                        ..
707                    } => {
708                        self.stats.update_timestamp(*marked_for_delete_at);
709                        self.stats.update_local_deletion_time(*local_deletion_time);
710                    }
711                    // Issue #887: a per-element complex cell carries its OWN explicit
712                    // timestamp/ttl/local_deletion_time (DataWriter delta-encodes each
713                    // against the SSTable baselines). The element timestamp can differ
714                    // from the row liveness timestamp, and a deleted/expiring element
715                    // supplies an LDT/TTL the row-level accumulation never sees. Fold
716                    // them all in so the baselines cover every byte emitted by
717                    // `write_complex_element_cell`.
718                    crate::storage::write_engine::mutation::CellOperation::WriteComplexElement {
719                        timestamp_micros,
720                        ttl_seconds,
721                        local_deletion_time,
722                        is_deleted,
723                        ..
724                    } => {
725                        self.stats.update_timestamp(*timestamp_micros);
726                        if let Some(ttl) = ttl_seconds {
727                            self.stats.update_ttl(*ttl as i32);
728                        }
729                        if let Some(ldt) = local_deletion_time {
730                            self.stats.update_local_deletion_time(*ldt);
731                        }
732                        // Issue #1728 (roborev finding 2): a LIVE complex element
733                        // (not deleted, not expiring, no explicit LDT) carries
734                        // Cassandra's `NO_DELETION_TIME` sentinel just like a simple
735                        // live `Write`, so it must lift `maxLocalDeletionTime` to the
736                        // sentinel in a mixed SSTable. Authoritative liveness comes
737                        // from the verbatim `is_deleted` flag (never re-derived from
738                        // `value`, per the no-heuristics mandate / roborev #885).
739                        if !*is_deleted && ttl_seconds.is_none() && local_deletion_time.is_none() {
740                            self.stats.note_live_local_deletion_time();
741                        }
742                    }
743                    // Issue #1728: a live, non-TTL `Write` cell carries Cassandra's
744                    // `Cell.NO_DELETION_TIME` (`Integer.MAX_VALUE`) as its
745                    // localDeletionTime. Cassandra's MetadataCollector folds that
746                    // sentinel into `maxLocalDeletionTime`, so any SSTable holding a
747                    // live non-TTL cell finalizes `maxLocalDeletionTime` == the live
748                    // sentinel — including a MIXED SSTable that also carries older
749                    // tombstones with a smaller real LDT. The dedicated chokepoint
750                    // raises the max ALONE (never poisoning minLocalDeletionTime or
751                    // the tombstone drop-time histogram, issue #851).
752                    //
753                    // Two cases must NOT fold the live sentinel (roborev #1728):
754                    //   * `mutation.ttl_seconds.is_some()` — a plain `Write` under a
755                    //     row-level TTL is emitted as an EXPIRING cell with a real
756                    //     `now + ttl` LDT (data_writer::rows), which the mutation-TTL
757                    //     block above (`if let Some(ttl) = mutation.ttl_seconds`)
758                    //     already folded; overwriting max with the sentinel would
759                    //     clobber that real expiring LDT.
760                    //   * `value == Null` — a null `Write` is skipped by
761                    //     `write_merged_cells` (no cell is emitted), so it confers no
762                    //     LDT at all.
763                    crate::storage::write_engine::mutation::CellOperation::Write { value, .. } => {
764                        if mutation.ttl_seconds.is_none()
765                            && !matches!(value, crate::types::Value::Null)
766                        {
767                            self.stats.note_live_local_deletion_time();
768                        }
769                    }
770                }
771            }
772            // Track stats for partition tombstones
773            if let Some(pt) = &mutation.partition_tombstone {
774                self.stats.update_timestamp(pt.deletion_time);
775                self.stats
776                    .update_local_deletion_time(pt.local_deletion_time);
777                // Record the partition-level deletion marker for the `da`-format
778                // StatsMetadata.hasPartitionLevelDeletions field. Authoritative:
779                // the mutation explicitly carries a partition tombstone.
780                self.stats.mark_partition_level_deletion();
781            }
782
783            // Track stats for range tombstones
784            for rt in &mutation.range_tombstones {
785                self.stats.update_timestamp(rt.deletion_time);
786                self.stats
787                    .update_local_deletion_time(rt.local_deletion_time);
788            }
789
790            // Issue #1721: a decoupled row tombstone (#932
791            // `Mutation::row_tombstone = Some((deletion_time, ldt))`) is emitted
792            // by DataWriter as a `HAS_DELETION` row stamped with its OWN
793            // `(deletion_time, ldt)` — DECOUPLED from `timestamp_micros`, so the
794            // per-cell/mutation folds above never see it. Fold both through the
795            // same `update_*` chokepoints as `partition_tombstone` so
796            // `min_local_deletion_time` covers the tombstone's `ldt`; otherwise
797            // the below-baseline guard in data_writer/rows.rs (static sibling
798            // static_rows.rs) rejects the row when `ldt` sits below the
799            // incremental baseline. LIVE sentinels are filtered inside the
800            // chokepoints (#851), so route the raw values straight through.
801            if let Some((deletion_time, ldt)) = mutation.row_tombstone {
802                self.stats.update_timestamp(deletion_time);
803                self.stats.update_local_deletion_time(ldt);
804            }
805
806            // Issue #851: row_count (totalRows) and column_count
807            // (totalColumnsSet) are NOT re-derived per-mutation here. The two
808            // previous attempts re-grouped rows in this loop and kept diverging
809            // from `DataWriter::merge_row_group`, which is what actually emits
810            // rows/cells to Data.db (e.g. it drops partition/clustering-key
811            // columns from cells, and merges static ops from ALL mutations into
812            // a single static prelude that is a SEPARATE row from the clustering
813            // row). Instead, the emitter returns `PartitionEmitCounts` and we add
814            // them below (see the `write_partition_with_index_blocks` call) so
815            // the stats can never drift from what was physically written.
816        }
817
818        // Update DataWriter's stats before writing, unless baselines were
819        // pre-seeded for the whole SSTable (issue #729 two-pass flush).
820        // When baselines are locked, the DataWriter already holds the final
821        // minimum values computed over ALL partitions; overwriting them with
822        // the incrementally-growing stats of this partition would raise the
823        // baseline and corrupt delta encoding for earlier partitions.
824        if !self.baselines_locked {
825            self.data_writer.update_stats_from_metadata(&self.stats);
826        }
827
828        // Extract partition tombstone and range tombstones from mutations.
829        // The tombstone can arrive on ANY mutation of the partition (a DELETE
830        // typically follows earlier INSERTs), so scan all of them and keep the
831        // newest deletion (Issue #716: taking only the first mutation dropped
832        // the tombstone and left the partition header LIVE).
833        let partition_tombstone = mutations
834            .iter()
835            .filter_map(|m| m.partition_tombstone.as_ref())
836            .max_by_key(|pt| pt.deletion_time);
837
838        // Collect all range tombstones from mutations
839        let range_tombstones: Vec<_> = mutations
840            .iter()
841            .flat_map(|m| m.range_tombstones.iter())
842            .cloned()
843            .collect();
844
845        // Write partition to Data.db, collecting promoted index blocks for wide partitions.
846        // Wide partitions (≥ 64 KiB of row data) get a non-zero promoted index so Cassandra
847        // can seek directly to a clustering-key range without reading the full partition.
848        let (data_offset, promoted_blocks, emit_counts) =
849            self.data_writer.write_partition_with_index_blocks(
850                &key,
851                &mutations,
852                &self.schema,
853                partition_tombstone,
854                &range_tombstones,
855            )?;
856
857        // Issue #851: Statistics' totalRows / totalColumnsSet are fed directly
858        // from what `DataWriter` physically emitted (the single source of truth),
859        // so they cannot drift from Data.db. The empty static-row prelude and
860        // range tombstone markers are already excluded by the emitter, matching
861        // Cassandra `Row.isEmpty()` / `Row.columnCount()`.
862        self.stats.row_count += emit_counts.rows;
863        self.stats.column_count += emit_counts.columns;
864
865        // Issue #1327: record this partition's serialized Data.db size and cell
866        // count into the estimatedPartitionSize / estimatedCellPerPartitionCount
867        // EstimatedHistograms (one observation per partition). `data_offset` is
868        // this partition's start offset in Data.db and `data_writer.position()`
869        // is the running Data.db length after the write, so their difference is
870        // the exact serialized partition byte length — the value Cassandra's
871        // MetadataCollector records. Σ estimatedPartitionSize bucket counts then
872        // equals the SSTable partition count for the authoritative read-side
873        // decode (`read_table_counts`, issue #944).
874        let partition_serialized_size = self.data_writer.position().saturating_sub(data_offset);
875        self.stats
876            .record_partition(partition_serialized_size, emit_counts.columns);
877
878        // Add partition to Index.db and get entry info.
879        // Pass promoted blocks (writer gates on >= 2 blocks before emitting payload).
880        // IMPORTANT: Capture index_offset AFTER the entry is written to Index.db.
881        let entry_info =
882            self.index_writer
883                .add_partition_with_promoted(&key, data_offset, &promoted_blocks)?;
884
885        // Add partition key to Filter.db
886        if let Some(ref mut filter) = self.filter_writer {
887            filter.add_key(&key);
888        }
889
890        // BTI (issue #766 / #910): defer this partition's Partitions.db trie
891        // payload (a no-op for BIG). The wide/narrow gate, OSS50-separator
892        // construction, and the deferred-finalization rationale live in
893        // [`Self::queue_bti_partition`] (see `bti_state`).
894        self.queue_bti_partition(&key, data_offset, &promoted_blocks, partition_tombstone);
895
896        // Track every partition for first_key / last_key / total_partition_count.
897        // These fields must cover the full SSTable, not just sampled entries.
898        // (Issue #666: first/last keys in Summary.db must span the whole SSTable
899        //  so Cassandra's range queries cover all partitions.)
900        self.summary_writer.note_partition(&key);
901
902        // Sample for Summary.db (every Nth entry, where N = summary_sample_interval)
903        // CRITICAL: Use the actual index_offset from entry_info, not an estimate
904        if self.summary_sample_counter % self.summary_sample_interval == 0 {
905            self.summary_writer
906                .add_entry(&key, entry_info.index_offset)?;
907        }
908
909        self.summary_sample_counter += 1;
910        self.partition_count += 1;
911        self.stats.increment_partition_count();
912
913        // Partitions-written counter (issue #1036). One per successful partition.
914        crate::observability::add_counter(crate::observability::catalog::WRITE_PARTITIONS, 1, &[]);
915
916        Ok(())
917    }

976    /// Compute the encoding baseline stats (min values only) from a slice of mutations.
977    ///
978    /// Used for the pre-pass before `write_partition` is called (issue #729
979    /// two-pass flush).  Returns `(min_timestamp, min_local_deletion_time, min_ttl)`.
980    /// Sentinel value `i64::MAX` / `i32::MAX` is returned for each field when no
981    /// relevant data is found in the slice (caller should handle via `.min()`
982    /// accumulation and then pass the final result to `pre_seed_encoding_baselines`).
983    pub fn compute_mutations_baseline_stats(mutations_slice: &[Mutation]) -> (i64, i32, i32) {
984        let mut min_timestamp = i64::MAX;
985        let mut min_ldt = i32::MAX;
986        let mut min_ttl = i32::MAX;
987
988        for mutation in mutations_slice {
989            min_timestamp = min_timestamp.min(mutation.timestamp_micros);
990
991            // Issue #1018: a simple `Write`/`WriteWithTtl`/`Delete` cell may carry
992            // its OWN (lower) per-cell timestamp in
993            // `Mutation::cell_write_timestamps` — a live cell's writetime OR a cell
994            // tombstone's markedForDeleteAt (the compaction merge→mutation path
995            // records it when it differs from the row's `timestamp_micros`). The
996            // DataWriter emits that cell's explicit timestamp as a `min_timestamp`
997            // delta, so the pre-seeded baseline must cover EVERY per-cell timestamp
998            // — otherwise `min_timestamp` could be pre-seeded ABOVE an emitted
999            // cell's actual (lower) timestamp and the unsigned-VInt delta
1000            // underflows/wraps. Fold them all in here, mirroring the per-cell
1001            // timestamp threading in `rows.rs` / `encoding.rs`.
1002            if let Some(cell_ts) = &mutation.cell_write_timestamps {
1003                for ts in cell_ts.values() {
1004                    min_timestamp = min_timestamp.min(*ts);
1005                }
1006            }
1007
1008            for op in &mutation.operations {
1009                match op {
1010                    crate::storage::write_engine::mutation::CellOperation::WriteWithTtl {
1011                        ttl_seconds,
1012                        ..
1013                    } => {
1014                        let ttl = *ttl_seconds as i32;
1015                        if ttl > 0 {
1016                            min_ttl = min_ttl.min(ttl);
1017                            let now_seconds = std::time::SystemTime::now()
1018                                .duration_since(std::time::UNIX_EPOCH)
1019                                .map(|d| d.as_secs() as i32)
1020                                .unwrap_or(0);
1021                            let ldt = now_seconds.saturating_add(ttl);
1022                            min_ldt = min_ldt.min(ldt);
1023                        }
1024                    }
1025                    op @ (crate::storage::write_engine::mutation::CellOperation::Delete { .. }
1026                    | crate::storage::write_engine::mutation::CellOperation::DeleteRow) => {
1027                        // Issue #764 / #921 finding 2: the encoding baseline must
1028                        // match the LDT the row/cell tombstone will ACTUALLY be
1029                        // written with, else the delta underflows. A `Delete` with
1030                        // a per-cell `local_deletion_time: Some(L)` is emitted with
1031                        // `L` verbatim; reuse the emit path's
1032                        // `op_cell_local_deletion_time` helper so the pre-seeded
1033                        // baseline always covers the smallest LDT actually written.
1034                        let ldt =
1035                            crate::storage::sstable::writer::data_writer::op_cell_local_deletion_time(
1036                                op, mutation,
1037                            );
1038                        min_ldt = min_ldt.min(ldt);
1039                    }
1040                    // Issue #887: the pre-seeded baseline path must fold the SAME
1041                    // marker timestamps/LDTs the DataWriter delta-encodes (it
1042                    // subtracts `min_timestamp` / `min_local_deletion_time` from
1043                    // `marked_for_delete_at` and the element LDT). A
1044                    // `ComplexDeletion`/`WriteComplexElement` carrying a timestamp or
1045                    // LDT BELOW the mutation's own values would make the delta
1046                    // underflow when baselines are locked — exactly what #729's
1047                    // two-pass flush is meant to prevent. Mirror the non-pre-seeded
1048                    // accumulation in `write_partition`.
1049                    crate::storage::write_engine::mutation::CellOperation::ComplexDeletion {
1050                        marked_for_delete_at,
1051                        local_deletion_time,
1052                        ..
1053                    } => {
1054                        // Exclude LIVE / NO_DELETION sentinels exactly as the
1055                        // `update_*` chokepoints do (issue #851), so a sentinel
1056                        // marker cannot drag the min baselines to `i64::MIN` /
1057                        // `i32::MAX`.
1058                        if *marked_for_delete_at != i64::MIN && *marked_for_delete_at != i64::MAX {
1059                            min_timestamp = min_timestamp.min(*marked_for_delete_at);
1060                        }
1061                        if *local_deletion_time != i32::MAX {
1062                            min_ldt = min_ldt.min(*local_deletion_time);
1063                        }
1064                    }
1065                    crate::storage::write_engine::mutation::CellOperation::WriteComplexElement {
1066                        timestamp_micros,
1067                        ttl_seconds,
1068                        local_deletion_time,
1069                        ..
1070                    } => {
1071                        if *timestamp_micros != i64::MIN && *timestamp_micros != i64::MAX {
1072                            min_timestamp = min_timestamp.min(*timestamp_micros);
1073                        }
1074                        if let Some(ttl) = ttl_seconds {
1075                            let ttl = *ttl as i32;
1076                            if ttl > 0 {
1077                                min_ttl = min_ttl.min(ttl);
1078                            }
1079                        }
1080                        if let Some(ldt) = local_deletion_time {
1081                            if *ldt != i32::MAX {
1082                                min_ldt = min_ldt.min(*ldt);
1083                            }
1084                        }
1085                    }
1086                    crate::storage::write_engine::mutation::CellOperation::Write { .. } => {}
1087                }
1088            }
1089
1090            // Partition-level TTL (top-level ttl_seconds on the Mutation)
1091            if let Some(ttl) = mutation.ttl_seconds {
1092                let ttl = ttl as i32;
1093                if ttl > 0 {
1094                    min_ttl = min_ttl.min(ttl);
1095                    let now_seconds = std::time::SystemTime::now()
1096                        .duration_since(std::time::UNIX_EPOCH)
1097                        .map(|d| d.as_secs() as i32)
1098                        .unwrap_or(0);
1099                    let ldt = now_seconds.saturating_add(ttl);
1100                    min_ldt = min_ldt.min(ldt);
1101                }
1102            }
1103
1104            if let Some(pt) = &mutation.partition_tombstone {
1105                min_timestamp = min_timestamp.min(pt.deletion_time);
1106                min_ldt = min_ldt.min(pt.local_deletion_time);
1107            }
1108
1109            for rt in &mutation.range_tombstones {
1110                min_timestamp = min_timestamp.min(rt.deletion_time);
1111                min_ldt = min_ldt.min(rt.local_deletion_time);
1112            }
1113
1114            // Issue #1721: a decoupled row tombstone (#932
1115            // `Mutation::row_tombstone = Some((deletion_time, ldt))`) is emitted by
1116            // DataWriter as a `HAS_DELETION` row stamped with its OWN
1117            // `(deletion_time, ldt)` — DECOUPLED from `timestamp_micros`, so the
1118            // folds above never see it. The pre-seeded flush path locks THIS
1119            // baseline; without folding the row tombstone, `min_ldt` stays
1120            // `i32::MAX` and the below-baseline guard in data_writer/rows.rs rejects
1121            // the row (and the `deletion_time` delta underflows against
1122            // `min_timestamp`). Mirror the `partition_tombstone` fold; LIVE
1123            // sentinels never reach this field.
1124            if let Some((deletion_time, ldt)) = mutation.row_tombstone {
1125                min_timestamp = min_timestamp.min(deletion_time);
1126                min_ldt = min_ldt.min(ldt);
1127            }
1128        }
1129
1130        (min_timestamp, min_ldt, min_ttl)
1131    }
