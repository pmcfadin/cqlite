1//! SSTable writer components for producing Cassandra 5.0-compatible SSTables
2//!
3//! This module coordinates the generation of all SSTable components:
4//! - Data.db: Row data with partition and clustering ordering
5//! - Index.db: Partition index for fast lookups
6//! - Filter.db: Bloom filter for existence checks
7//! - Statistics.db: Metadata for delta encoding
8//! - Summary.db: Sampled index entries
9//! - CompressionInfo.db: Compression metadata (only when compressed)
10//! - TOC.txt: Component manifest (publication barrier)
11//!
12//! Component generation order is critical (see M5 Council Recommendation):
13//! 1. Statistics.db (provides delta encoding baseline)
14//! 2. Data.db + Index.db (single pass, track offsets)
15//! 3. Summary.db (sample Index.db entries)
16//! 4. Filter.db (finalize Bloom filter)
17//! 5. CompressionInfo.db (only when compressed)
18//! 6. Digest.crc32
19//! 7. TOC.txt (makes SSTable visible)
20//!
21//! TODO: Implementation in M5.0-7 through M5.0-13
22

106/// SSTable writer coordinator
107///
108/// Orchestrates the generation of all SSTable components in the correct order.
109/// Produces valid Cassandra 5.0 BIG format SSTables.
110///
111/// # Write Order
112///
113/// Components are written in the following critical order:
114/// 1. Statistics.db - Provides delta encoding baseline (FIRST)
115/// 2. Data.db - Main partition/row data
116/// 3. Index.db - Partition index (uses Data.db offsets)
117/// 4. Filter.db - Bloom filter
118/// 5. Summary.db - Sampled index entries
119/// 6. CompressionInfo.db - Compression metadata (only when compressed)
120/// 7. Digest.crc32 - Data.db checksum
121/// 8. TOC.txt - Table of contents (LAST, publication barrier)
122///
123/// # File Naming
124///
125/// All components follow the pattern: `nb-{generation}-big-{Component}.db`
126/// Example: `nb-1-big-Data.db`, `nb-1-big-Index.db`
127///
128/// # Partition Ordering
129///
130/// Partitions MUST be written in Murmur3 token order (caller responsibility).
131/// The writer validates token ordering on each `write_partition()` call.
132///
133/// # Example
134///
135/// ```rust,ignore
136/// use cqlite_core::storage::sstable::writer::SSTableWriter;
137/// use cqlite_core::storage::write_engine::mutation::{Mutation, DecoratedKey};
138/// use cqlite_core::schema::TableSchema;
139///
140/// // Create schema
141/// let schema = TableSchema::from_json("...")?;
142///
143/// // Create writer
144/// let mut writer = SSTableWriter::new(
145///     PathBuf::from("data/ks/table"),
146///     1,  // generation
147///     &schema
148/// )?;
149///
150/// // Write partitions (MUST be in token order)
151/// let key = DecoratedKey::new(token, key_bytes);
152/// let mutations = vec![/* ... */];
153/// writer.write_partition(key, mutations)?;
154///
155/// // Finish writing
156/// let info = writer.finish().await?;
157/// println!("Wrote SSTable with {} partitions", info.partition_count);
158/// ```
159#[cfg(feature = "write-support")]
160#[derive(Debug)]
161pub struct SSTableWriter {
162    /// SSTable output directory: output_dir/keyspace/table/
163    sstable_dir: PathBuf,
164    /// SSTable generation number
165    generation: u64,
166    /// Table schema for column metadata
167    schema: TableSchema,
168    /// Statistics metadata (collected during writes)
169    stats: StatisticsMetadata,
170    /// Data.db writer
171    data_writer: DataWriter,
172    /// Index.db writer
173    index_writer: IndexWriter,
174    /// Filter.db writer
175    filter_writer: Option<FilterWriter>,
176    /// Summary.db writer
177    summary_writer: SummaryWriter,
178    /// Last token written (for ordering validation)
179    last_token: Option<i64>,
180    /// Number of partitions written
181    partition_count: usize,
182    /// Summary sampling counter (sample every N entries)
183    summary_sample_counter: usize,
184    /// Sampling interval for Summary.db (default: 128)
185    summary_sample_interval: usize,
186}

190    /// Create a new SSTable writer
191    ///
192    /// # Arguments
193    ///
194    /// * `output_dir` - Directory where SSTable files will be written
195    /// * `generation` - SSTable generation number (e.g., 1, 2, 3...)
196    /// * `schema` - Table schema for column metadata
197    ///
198    /// # Returns
199    ///
200    /// A new SSTableWriter ready to accept partitions.
201    ///
202    /// # Example
203    ///
204    /// ```rust,ignore
205    /// let writer = SSTableWriter::new(
206    ///     PathBuf::from("data/test_ks/users"),
207    ///     1,
208    ///     &schema
209    /// )?;
210    /// ```
211    pub fn new(output_dir: PathBuf, generation: u64, schema: &TableSchema) -> Result<Self> {
212        Self::with_expected_partitions(output_dir, generation, schema, 128)
213    }

215    /// Create a new SSTable writer with an expected partition count hint
216    ///
217    /// The expected count is used to size the Bloom filter optimally.
218    pub fn with_expected_partitions(
219        output_dir: PathBuf,
220        generation: u64,
221        schema: &TableSchema,
222        expected_partitions: usize,
223    ) -> Result<Self> {
224        // Initialize statistics metadata with sentinel values
225        let mut stats = StatisticsMetadata::new();
226        // Pre-set min values to reasonable defaults (will be updated during writes)
227        stats.min_timestamp = i64::MAX;
228        stats.min_ttl = i32::MAX;
229        stats.min_local_deletion_time = i32::MAX;
230
231        // Compute the SSTable output directory: output_dir/keyspace/table/
232        // This ensures the reader's extract_table_name() can map files to table names.
233        let sstable_dir = output_dir.join(&schema.keyspace).join(&schema.table);
234
235        // Create Data.db writer in streaming mode (Issue #492): each partition is
236        // flushed to the Data.db file as it is written, bounding peak memory to a
237        // single partition instead of buffering the whole component. The file is
238        // opened lazily on the first partition (creating sstable_dir as needed).
239        let data_path = Self::component_path(&sstable_dir, generation, "Data.db");
240        let data_writer = DataWriter::with_sink(stats.clone(), data_path);
241
242        // Create Index.db writer
243        let index_writer = IndexWriter::new();
244
245        // Create Filter.db writer (1% false positive rate by default)
246        let filter_path = Self::component_path(&sstable_dir, generation, "Filter.db");
247        let filter_writer = Some(FilterWriter::new(
248            filter_path,
249            expected_partitions.max(1),
250            0.01,
251        )?);
252
253        // Create Summary.db writer (sample every 128 entries per Cassandra default)
254        let summary_sample_interval = 128;
255        let summary_writer = SummaryWriter::new(summary_sample_interval as u32);
256
257        Ok(Self {
258            sstable_dir,
259            generation,
260            schema: schema.clone(),
261            stats,
262            data_writer,
263            index_writer,
264            filter_writer,
265            summary_writer,
266            last_token: None,
267            partition_count: 0,
268            summary_sample_counter: 0,
269            summary_sample_interval,
270        })
271    }

273    /// Write a partition (partition key + all mutations)
274    ///
275    /// # Arguments
276    ///
277    /// * `key` - DecoratedKey (token + raw partition key bytes)
278    /// * `mutations` - All mutations for this partition (must be in clustering order)
279    ///
280    /// # Returns
281    ///
282    /// Ok(()) on success, or an error if:
283    /// - Partitions are not in token order
284    /// - Schema validation fails
285    /// - I/O error occurs
286    ///
287    /// # Ordering Requirement
288    ///
289    /// Partitions MUST be written in ascending token order. This method validates
290    /// ordering and returns an error if violated.
291    ///
292    /// # Example
293    ///
294    /// ```rust,ignore
295    /// let key = DecoratedKey::new(12345, vec![0x00, 0x01, 0x02]);
296    /// let mutations = vec![
297    ///     Mutation::new(/* ... */)
298    /// ];
299    /// writer.write_partition(key, mutations)?;
300    /// ```
301    pub fn write_partition(&mut self, key: DecoratedKey, mutations: Vec<Mutation>) -> Result<()> {
302        // Validate token ordering
303        if let Some(last_token) = self.last_token {
304            if key.token <= last_token {
305                return Err(Error::InvalidInput(format!(
306                    "Partitions must be written in token order: got token {} after {}",
307                    key.token, last_token
308                )));
309            }
310        }
311        self.last_token = Some(key.token);
312
313        // Sort mutations by clustering key (Cassandra requires sorted rows within partitions)
314        let mut mutations = mutations;
315        mutations.sort_by(|a, b| match (&a.clustering_key, &b.clustering_key) {
316            (None, None) => std::cmp::Ordering::Equal,
317            (None, Some(_)) => std::cmp::Ordering::Less,
318            (Some(_), None) => std::cmp::Ordering::Greater,
319            (Some(ck_a), Some(ck_b)) => ck_a
320                .compare(ck_b, &self.schema)
321                .unwrap_or_else(|_| ck_a.cmp(ck_b)),
322        });
323
324        // Update statistics from mutations
325        for mutation in &mutations {
326            self.stats.update_timestamp(mutation.timestamp_micros);
327            if let Some(ttl) = mutation.ttl_seconds {
328                self.stats.update_ttl(ttl as i32);
329                let now_seconds = std::time::SystemTime::now()
330                    .duration_since(std::time::UNIX_EPOCH)
331                    .map(|d| d.as_secs() as i32)
332                    .unwrap_or(0);
333                let local_deletion_time = now_seconds.saturating_add(ttl as i32);
334                self.stats.update_local_deletion_time(local_deletion_time);
335            }
336            // Track local deletion times for tombstones and TTL cells
337            // TODO(Issue #401): Get proper local_deletion_time from Mutation struct
338            for op in &mutation.operations {
339                match op {
340                    crate::storage::write_engine::mutation::CellOperation::WriteWithTtl {
341                        ttl_seconds,
342                        ..
343                    } => {
344                        // Track TTL
345                        self.stats.update_ttl(*ttl_seconds as i32);
346                        // CRITICAL: TTL cells need local_deletion_time tracked
347                        // local_deletion_time = now + ttl
348                        let now_seconds = std::time::SystemTime::now()
349                            .duration_since(std::time::UNIX_EPOCH)
350                            .map(|d| d.as_secs() as i32)
351                            .unwrap_or(0);
352                        let local_deletion_time = now_seconds.saturating_add(*ttl_seconds as i32);
353                        self.stats.update_local_deletion_time(local_deletion_time);
354                    }
355                    crate::storage::write_engine::mutation::CellOperation::Delete { .. }
356                    | crate::storage::write_engine::mutation::CellOperation::DeleteRow => {
357                        // Derive local_deletion_time from timestamp (workaround)
358                        let local_deletion_time = (mutation.timestamp_micros / 1_000_000) as i32;
359                        self.stats.update_local_deletion_time(local_deletion_time);
360                    }
361                    _ => {}
362                }
363            }
364            // Track stats for partition tombstones
365            if let Some(pt) = &mutation.partition_tombstone {
366                self.stats.update_timestamp(pt.deletion_time);
367                self.stats
368                    .update_local_deletion_time(pt.local_deletion_time);
369            }
370
371            // Track stats for range tombstones
372            for rt in &mutation.range_tombstones {
373                self.stats.update_timestamp(rt.deletion_time);
374                self.stats
375                    .update_local_deletion_time(rt.local_deletion_time);
376            }
377
378            self.stats.increment_row_count();
379            self.stats
380                .add_column_count(mutation.operations.len() as u64);
381        }
382
383        // Update DataWriter's stats before writing
384        // This ensures delta encoding uses the correct baselines
385        self.data_writer.update_stats(self.stats.clone());
386
387        // Extract partition tombstone and range tombstones from mutations.
388        // The tombstone can arrive on ANY mutation of the partition (a DELETE
389        // typically follows earlier INSERTs), so scan all of them and keep the
390        // newest deletion (Issue #716: taking only the first mutation dropped
391        // the tombstone and left the partition header LIVE).
392        let partition_tombstone = mutations
393            .iter()
394            .filter_map(|m| m.partition_tombstone.as_ref())
395            .max_by_key(|pt| pt.deletion_time);
396
397        // Collect all range tombstones from mutations
398        let range_tombstones: Vec<_> = mutations
399            .iter()
400            .flat_map(|m| m.range_tombstones.iter())
401            .cloned()
402            .collect();
403
404        // Write partition to Data.db and get offset
405        let data_offset = self.data_writer.write_partition(
406            &key,
407            &mutations,
408            &self.schema,
409            partition_tombstone,
410            &range_tombstones,
411        )?;
412
413        // Add partition to Index.db and get entry info
414        // IMPORTANT: Capture index_offset AFTER the entry is written to Index.db
415        let entry_info = self.index_writer.add_partition(&key, data_offset)?;
416
417        // Add partition key to Filter.db
418        if let Some(ref mut filter) = self.filter_writer {
419            filter.add_key(&key);
420        }
421
422        // Track every partition for first_key / last_key / total_partition_count.
423        // These fields must cover the full SSTable, not just sampled entries.
424        // (Issue #666: first/last keys in Summary.db must span the whole SSTable
425        //  so Cassandra's range queries cover all partitions.)
426        self.summary_writer.note_partition(&key);
427
428        // Sample for Summary.db (every Nth entry, where N = summary_sample_interval)
429        // CRITICAL: Use the actual index_offset from entry_info, not an estimate
430        if self.summary_sample_counter % self.summary_sample_interval == 0 {
431            self.summary_writer
432                .add_entry(&key, entry_info.index_offset)?;
433        }
434
435        self.summary_sample_counter += 1;
436        self.partition_count += 1;
437        self.stats.increment_partition_count();
438
439        Ok(())
440    }

442    /// Finish writing all components and return SSTable information
443    ///
444    /// This method:
445    /// 1. Finalizes statistics metadata
446    /// 2. Writes all component files in the correct order
447    /// 3. Computes checksums
448    /// 4. Writes TOC.txt (publication barrier)
449    /// 5. Returns SSTableInfo with file paths and metadata
450    ///
451    /// # Returns
452    ///
453    /// SSTableInfo containing paths to all written files and metadata.
454    ///
455    /// # Errors
456    ///
457    /// Returns error if any component write fails.
458    ///
459    /// # Example
460    ///
461    /// ```rust,ignore
462    /// let info = writer.finish().await?;
463    /// println!("SSTable written to {}", info.data_path.display());
464    /// ```
465    pub async fn finish(mut self) -> Result<SSTableInfo> {
466        // Create keyspace/table subdirectory structure so the reader can
467        // extract the table name from the parent directory path. Owned clone so
468        // `finish_streaming()` can move `self.data_writer` out below.
469        let sstable_dir = self.sstable_dir.clone();
470        let sstable_dir = sstable_dir.as_path();
471        tokio::fs::create_dir_all(sstable_dir).await?;
472
473        // Finalize statistics metadata (normalize sentinel values)
474        self.stats.finalize();
475
476        // 1. Write Statistics.db (FIRST - provides delta baseline)
477        let stats_path = Self::component_path(sstable_dir, self.generation, "Statistics.db");
478        let stats_writer = StatisticsWriter::new(stats_path.clone());
479        stats_writer.write(&self.stats, Some(&self.schema))?;
480
481        // 2. Finalize Data.db (Issue #492)
482        // The DataWriter has been streaming each partition to disk as it was
483        // written, so there is no whole-file buffer to write here. `finish_streaming`
484        // flushes and fsyncs the sink and returns the total byte size. If no
485        // partitions were written, lazily ensure an (empty) Data.db file exists so
486        // the downstream Digest CRC re-read and TOC publication remain valid.
487        let data_path = Self::component_path(sstable_dir, self.generation, "Data.db");
488        let data_size = self.data_writer.finish_streaming()?;
489        if data_size == 0 && !data_path.exists() {
490            tokio::fs::write(&data_path, b"").await?;
491        }
492
493        // 3. Write Index.db
494        let index_path = Self::component_path(sstable_dir, self.generation, "Index.db");
495        let index_bytes = self.index_writer.finish()?;
496        tokio::fs::write(&index_path, index_bytes).await?;
497
498        // 4. Write Filter.db (path already set in constructor using sstable_dir)
499        let filter_path = Self::component_path(sstable_dir, self.generation, "Filter.db");
500        if let Some(filter_writer) = self.filter_writer {
501            filter_writer.finish().await?;
502        }
503
504        // 5. Write Summary.db
505        let summary_path = Self::component_path(sstable_dir, self.generation, "Summary.db");
506        let summary_bytes = self.summary_writer.finish()?;
507        tokio::fs::write(&summary_path, summary_bytes).await?;
508
509        // 5.5. CompressionInfo.db is omitted for uncompressed data.
510        // Real Cassandra 5 SSTables do not include CompressionInfo.db when
511        // data is uncompressed. The compression_info_writer module is retained
512        // for future compressed SSTable support.
513
514        // 6. Write Digest.crc32 (compute CRC32 of Data.db)
515        let digest_path = Self::component_path(sstable_dir, self.generation, "Digest.crc32");
516        let digest_writer = DigestWriter::new(digest_path.clone());
517        let crc32_value = Self::compute_crc32(&data_path).await?;
518        digest_writer.write(crc32_value)?;
519
520        // 7. Write TOC.txt (LAST - publication barrier)
521        let toc_path = Self::component_path(sstable_dir, self.generation, "TOC.txt");
522        let toc_writer = TocWriter::new(toc_path.clone());
523        let components = vec![
524            ComponentEntry::new(crate::storage::sstable::directory::types::SSTableComponent::Data),
525            ComponentEntry::new(crate::storage::sstable::directory::types::SSTableComponent::Index),
526            ComponentEntry::new(
527                crate::storage::sstable::directory::types::SSTableComponent::Filter,
528            ),
529            ComponentEntry::new(
530                crate::storage::sstable::directory::types::SSTableComponent::Summary,
531            ),
532            ComponentEntry::new(
533                crate::storage::sstable::directory::types::SSTableComponent::Statistics,
534            ),
535            ComponentEntry::new(
536                crate::storage::sstable::directory::types::SSTableComponent::Digest,
537            ),
538        ];
539        toc_writer.write(&components)?;
540
541        Ok(SSTableInfo {
542            data_path,
543            index_path,
544            filter_path,
545            summary_path,
546            stats_path,
547            compression_info_path: None,
548            toc_path,
549            digest_path,
550            partition_count: self.partition_count,
551            data_size,
552        })
553    }

555    /// Build component file path
556    fn component_path(output_dir: &Path, generation: u64, component: &str) -> PathBuf {
557        let filename = format!("nb-{}-big-{}", generation, component);
558        output_dir.join(filename)
559    }

561    /// Compute CRC32 checksum of a file
562    async fn compute_crc32(file_path: &PathBuf) -> Result<u32> {
563        let data = tokio::fs::read(file_path).await?;
564        let mut hasher = crc32fast::Hasher::new();
565        hasher.update(&data);
566        Ok(hasher.finalize())
567    }

687            .into_iter()

694

697        }
698
699        let info = writer.finish().await.unwrap();
700
701        assert_eq!(info.partition_count, 3);
702        assert!(info.data_size > 0);
