1//! BTI (`da`) write-path state and helpers for [`SSTableWriter`].
2//!
3//! The BTI format (issue #766 / #908 / #910) replaces the legacy BIG
4//! `Index.db`/`Summary.db` partition index with a `Partitions.db` trie and a
5//! within-partition `Rows.db` row-index trie. Because a wide partition's
6//! `Rows.db` `RowsOffset` is only known after `Rows.db` is serialized in
7//! [`SSTableWriter::finish`], each partition's trie payload is deferred:
8//! [`PendingBtiPartition`] records the raw key, the `Data.db` offset, and — for
9//! wide partitions — the OSS50 row-index blocks. The two helpers here own the
10//! BTI-specific portions of the write path so the BIG path stays isolated:
11//!
12//! - [`SSTableWriter::queue_bti_partition`] is called per partition from
13//!   `write_partition` to defer the trie payload.
14//! - [`SSTableWriter::write_bti_components`] is called from `finish` to
15//!   serialize `Rows.db` + `Partitions.db`.
16//!
17//! All bytes are byte-for-byte identical to the prior inline implementation;
18//! this module only relocates the BTI write path out of `mod.rs`.
19
20use super::{partitions_writer, PromotedIndexBlock, SSTableWriter};
21use crate::error::{Error, Result};
22use crate::storage::write_engine::mutation::{DecoratedKey, PartitionTombstone};
23use std::path::PathBuf;
24
25/// A deferred BTI partition payload (issue #910).
26///
27/// Narrow partitions (`row_index` is `None`) get a direct `Data.db` offset in
28/// the partition trie; wide partitions get a `Rows.db` `TrieIndexEntry` and a
29/// positive `RowsOffset` once `Rows.db` is serialized.
30#[derive(Debug)]
31pub(super) struct PendingBtiPartition {
32    /// Raw on-disk partition-key bytes.
33    pub(super) raw_key: Vec<u8>,
34    /// Partition's absolute `Data.db` start offset.
35    pub(super) data_offset: u64,
36    /// `Some` for a wide partition: its row-index blocks + partition deletion.
37    /// `None` for a narrow partition (direct `Data.db` offset).
38    pub(super) row_index: Option<PendingRowIndex>,
39}
40
41/// The row-index payload of a wide BTI partition, queued for `Rows.db`.
42#[derive(Debug)]
43pub(super) struct PendingRowIndex {
44    /// Per-block OSS50 separators + within-partition offsets.
45    pub(super) blocks: Vec<partitions_writer::RowIndexBlock>,
46    /// Partition-level deletion `(local_deletion_time, marked_for_delete_at)`,
47    /// or `None` for LIVE.
48    pub(super) partition_deletion: Option<(i32, i64)>,
49}
50
51impl SSTableWriter {
52    /// Defer this partition's `Partitions.db` trie payload (BTI, issue #766 / #910).
53    ///
54    /// Called per partition from `write_partition` only when the writer is in
55    /// the BTI format (`self.bti_pending` is `Some`); a no-op for BIG. The
56    /// payload is a direct `Data.db` offset for a NARROW partition (< 2
57    /// column-index blocks) or a `Rows.db` `RowsOffset` for a WIDE partition
58    /// (>= 2 blocks). The `RowsOffset` is only known after `Rows.db` is
59    /// serialized in `finish()`, so we record the raw key, the `Data.db`
60    /// offset, and — for wide partitions — the OSS50 row-index separators here,
61    /// and finalize both tries at `finish()`. The wide gate (>= 2 blocks)
62    /// mirrors `RowIndexEntry.create()` /
63    /// `IndexWriter::add_partition_with_promoted` and guide ch.17.
64    pub(super) fn queue_bti_partition(
65        &mut self,
66        key: &DecoratedKey,
67        data_offset: u64,
68        promoted_blocks: &[PromotedIndexBlock],
69        partition_tombstone: Option<&PartitionTombstone>,
70    ) {
71        let Some(ref mut pending) = self.bti_pending else {
72            return;
73        };
74        let row_index = if promoted_blocks.len() >= 2 {
75            // Build OSS50-separator row-index blocks. A block lacking an
76            // OSS50 separator (marker-led, or no clustering key) cannot be
77            // placed in the trie; if ANY block lacks one we fall back to a
78            // direct Data.db offset for the whole partition rather than emit
79            // an unreadable separator (no-heuristics: never guess bytes).
80            let mut blocks = Vec::with_capacity(promoted_blocks.len());
81            let mut all_have_sep = true;
82            for b in promoted_blocks {
83                match &b.oss50_separator {
84                    Some(sep) if !sep.is_empty() => {
85                        blocks.push(partitions_writer::RowIndexBlock {
86                            separator_key: sep.clone(),
87                            block_offset: b.offset,
88                            open_marker: None,
89                        });
90                    }
91                    _ => {
92                        all_have_sep = false;
93                        break;
94                    }
95                }
96            }
97            // Separators must be strictly ascending and unique for the trie.
98            let strictly_ascending = blocks
99                .windows(2)
100                .all(|w| w[0].separator_key < w[1].separator_key);
101            if all_have_sep && strictly_ascending && !blocks.is_empty() {
102                let partition_deletion =
103                    partition_tombstone.map(|pt| (pt.local_deletion_time, pt.deletion_time));
104                Some(PendingRowIndex {
105                    blocks,
106                    partition_deletion,
107                })
108            } else {
109                None
110            }
111        } else {
112            None
113        };
114        pending.push(PendingBtiPartition {
115            raw_key: key.key.clone(),
116            data_offset,
117            row_index,
118        });
119    }
120
121    /// Serialize the BTI `Rows.db` + `Partitions.db` components (issue #766 /
122    /// #908 / #910).
123    ///
124    /// Returns `(partitions_path, rows_path)`; both `None` when this is a BIG
125    /// writer (no deferred BTI payloads). For BIG nothing is written, keeping
126    /// the default path byte-for-byte unchanged.
127    ///
128    /// Order matters: `Rows.db` is serialized FIRST so each wide partition's
129    /// `TrieIndexEntry` offset (`RowsOffset`) is known, then the partition trie
130    /// leaves store either that positive `RowsOffset` (wide) or the negative
131    /// direct `Data.db` offset (narrow). Cassandra always emits a `Rows.db`
132    /// component for a BTI SSTable, even a 0-byte one when no partition is wide
133    /// (verified against the real `simple_table`/`collection_table`/`ttl_table`
134    /// `da-2-bti-Rows.db` fixtures, all 0 bytes yet listed in the TOC).
135    ///
136    /// Finding 2 (roborev #908): an EMPTY BTI SSTable (no partitions) cannot
137    /// produce a readable `Partitions.db` — a zero-byte trie has no 8-byte root
138    /// footer, so the BTI reader rejects it, and a `da` SSTable that omits
139    /// `Partitions.db` is unreadable. Rather than publish an unreadable artifact
140    /// we REFUSE to finish an empty BTI SSTable with a clear error. (A narrow
141    /// non-empty table still publishes a valid trie + a 0-byte `Rows.db`,
142    /// matching Cassandra.)
143    ///
144    /// Takes the deferred state by value (`bti_pending` / `partitions_trie`)
145    /// rather than borrowing `&mut self`, so `finish` can call it after
146    /// `summary_writer.finish()` has partially moved `self`.
147    pub(super) async fn write_bti_components(
148        bti_pending: Option<Vec<PendingBtiPartition>>,
149        partitions_trie: Option<partitions_writer::PartitionsTrieWriter>,
150        cpath: impl Fn(&str) -> PathBuf,
151    ) -> Result<(Option<PathBuf>, Option<PathBuf>)> {
152        let Some(pending) = bti_pending else {
153            return Ok((None, None));
154        };
155
156        if pending.is_empty() {
157            return Err(Error::InvalidInput(
158                "cannot publish an empty BTI SSTable: a `da` SSTable requires a readable \
159                 Partitions.db trie (with an 8-byte root footer), which has no valid \
160                 zero-partition form. Write at least one partition, or use the BIG format \
161                 for empty SSTables."
162                    .to_string(),
163            ));
164        }
165
166        // 1. Serialize Rows.db from the wide partitions, recovering each
167        //    wide partition's RowsOffset (in pending-order of wide entries).
168        let mut rows_writer = partitions_writer::RowsTrieWriter::new();
169        for p in &pending {
170            if let Some(ri) = &p.row_index {
171                rows_writer.add_partition_row_index(
172                    &p.raw_key,
173                    p.data_offset,
174                    ri.blocks.clone(),
175                    ri.partition_deletion,
176                );
177            }
178        }
179        let (rows_bytes, rows_offsets) = rows_writer.finish()?;
180
181        // 2. Build the partition trie: wide partitions get their positive
182        //    RowsOffset, narrow partitions keep the negative DataOffset.
183        let mut trie = partitions_trie.unwrap_or_default();
184        let mut wide_idx = 0usize;
185        for p in &pending {
186            if p.row_index.is_some() {
187                let rows_offset = rows_offsets[wide_idx];
188                wide_idx += 1;
189                trie.add_partition_with_payload(
190                    &p.raw_key,
191                    partitions_writer::PartitionPayload::RowsOffset(rows_offset),
192                );
193            } else {
194                trie.add_partition_with_payload(
195                    &p.raw_key,
196                    partitions_writer::PartitionPayload::DataOffset(p.data_offset),
197                );
198            }
199        }
200        let partitions_bytes = trie.finish()?;
201
202        // Partitions.db must be non-empty here (pending is non-empty).
203        let part_path = cpath("Partitions.db");
204        tokio::fs::write(&part_path, partitions_bytes).await?;
205        // fsync Partitions.db contents (issue #1392): tokio::fs::write does not
206        // fsync, so the bytes would only reach the page cache.
207        Self::fsync_component(&part_path).await?;
208
209        // Rows.db is ALWAYS emitted for BTI (possibly 0 bytes).
210        let rows_path = cpath("Rows.db");
211        tokio::fs::write(&rows_path, rows_bytes).await?;
212        Self::fsync_component(&rows_path).await?;
213
214        Ok((Some(part_path), Some(rows_path)))
215    }
