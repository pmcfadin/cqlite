1//! Promoted-index (`IndexInfo`) `ClusteringPrefix` serialization (Issue #1186).
2//!
3//! Extracted verbatim from `encoding.rs` to keep that file under the campsite-rule
4//! file-size threshold (epic #1116). No emitted bytes change. `use super::*`
5//! provides the crate imports, the bound-kind constants (`INCL_START_BOUND` etc.,
6//! private to `mod.rs`), and the sibling helper `serialize_clustering_prefix_to_vec`
7//! (re-exported from `encoding.rs`).
8//!
9//! Cassandra serializes a promoted-index `firstName`/`lastName` via
10//! `ClusteringPrefix.serializer.serialize` (`ClusteringPrefix.java:462-475`), which
11//! dispatches on the prefix kind: a row clustering goes through
12//! `Clustering.serializer` (`[kind][values-without-size]`); a range-tombstone marker
13//! bound goes through `ClusteringBoundOrBoundary.serializer`
14//! (`[kind][u16 size][values-without-size]`). The two forms are NOT interchangeable —
15//! only the bound form carries the `writeShort(size)`.
16

19/// `ClusteringPrefix.Kind.CLUSTERING.ordinal()` in Cassandra 5.0
20/// (`org.apache.cassandra.db.ClusteringPrefix.Kind`). A row's full clustering key
21/// (a promoted-index `firstName`/`lastName`) is always kind `CLUSTERING`; range
22/// bounds use the other ordinals (e.g. `EXCL_END_INCL_START_BOUNDARY = 2`,
23/// `INCL_END_EXCL_START_BOUNDARY = 5`).
24pub(crate) const CLUSTERING_PREFIX_KIND_CLUSTERING: u8 = 4;
25
26/// Serialize a `ClusteringKey` as the promoted-index (`IndexInfo`) `ClusteringPrefix`
27/// byte sequence (Issue #1186).
28///
29/// This is **NOT** the same as the Data.db row clustering prefix
30/// ([`serialize_clustering_prefix_to_vec`]). Cassandra serializes a Data.db row's
31/// clustering via the values-only `Clustering.serializer` (no kind byte), but it
32/// serializes a promoted-index `firstName`/`lastName` via
33/// `ClusteringPrefix.serializer.serialize`, which prepends a **leading kind byte**
34/// (`Kind.ordinal()`). For a full clustering key that kind is always `CLUSTERING`
35/// (`= 4`). Format:
36///
37/// ```text
38/// [kind: 1 byte = 0x04 (CLUSTERING)]
39/// [header: unsigned VInt]            ← 2 bits per column: 00=present, 10=null
40/// [value bytes…]                     ← type-specific bytes for each PRESENT column
41/// ```
42///
43/// For a single `int` clustering this is the Cassandra-exact 6 bytes
44/// `04 00 <4-byte big-endian int>`, matching the real
45/// `test_big.wide_partition` `Index.db` fixture (verified byte-for-byte).
46///
47/// Returns `Err` if a clustering column type is unknown (the caller falls back to
48/// `[kind, 0x00]` — an empty `Clustering` — in that case).
49pub(super) fn serialize_clustering_prefix_for_index(
50    clustering_key: &ClusteringKey,
51    schema: &TableSchema,
52) -> Result<Vec<u8>> {
53    let values = serialize_clustering_prefix_to_vec(clustering_key, schema)?;
54    let mut buf = Vec::with_capacity(values.len() + 1);
55    buf.push(CLUSTERING_PREFIX_KIND_CLUSTERING);
56    buf.extend_from_slice(&values);
57    Ok(buf)
58}

60/// The empty-clustering promoted-index `ClusteringPrefix`: a `Clustering` of kind
61/// `CLUSTERING` with no columns (Issue #1186). Used for no-clustering rows and
62/// range-bound fallbacks where no per-row clustering values are available. Equals
63/// `[0x04 (CLUSTERING)][0x00 (empty values header)]`.
64pub(super) fn empty_clustering_prefix_for_index() -> Vec<u8> {
65    vec![CLUSTERING_PREFIX_KIND_CLUSTERING, 0x00]
66}

68/// Serialize a range-tombstone **marker** bound as its promoted-index
69/// (`IndexInfo`) `ClusteringPrefix` byte sequence (Issue #1186 roborev MEDIUM).
70///
71/// A marker name uses its **actual bound kind** ordinal, NOT `CLUSTERING`, via
72/// Cassandra's `ClusteringBoundOrBoundary.Serializer.serialize`
73/// (`ClusteringBoundOrBoundary.java:103-108`) — a **different wire format** from
74/// `Clustering.serializer`: kind byte, then a 2-byte big-endian `size` short, then
75/// the values-without-size blob, and NOTHING after the size short when `size == 0`
76/// (`serializeValuesWithoutSize` is a no-op for an empty bound):
77///
78/// ```text
79/// [kind: 1 byte]   ← INCL_START_BOUND=1 / EXCL_END_BOUND=0 / INCL_END_BOUND=6 / EXCL_START_BOUND=7
80/// [size: u16 BE]   ← bound.size() (number of clustering values)
81/// [header: VInt]   ← only when size > 0; 2 bits per column: 00=present, 10=null
82/// [value bytes…]   ← only when size > 0; type-specific bytes for each PRESENT column
83/// ```
84///
85/// This mirrors the on-disk marker bytes ([`DataWriter::write_range_bound`]) minus
86/// the leading `IS_MARKER` flag. An open-ended bound (`Bottom`/`Top`, size 0) is
87/// therefore the 3 bytes `[kind][00][00]` — NOT `[kind]` and NOT `[kind][00]`.
88/// (Issue #1186 roborev MEDIUM: the pre-fix code emitted the `Clustering.serializer`
89/// form — a single `0x00` for empty bounds, no size short for non-empty — never the
90/// bound form.) Returns `Err` only if a clustering column type is unknown (the
91/// caller falls back via [`marker_bound_prefix_for_index`]).
92pub(super) fn serialize_marker_bound_prefix_for_index(
93    bound: &ClusteringBound,
94    is_open: bool,
95    schema: &TableSchema,
96) -> Result<Vec<u8>> {
97    let (kind, clustering) = marker_bound_kind(bound, is_open);
98    let mut buf = Vec::new();
99    buf.push(kind);
100    match clustering {
101        Some(ck) => {
102            let size = ck.columns.len();
103            if size > u16::MAX as usize {
104                return Err(crate::error::Error::InvalidInput(format!(
105                    "Range tombstone bound has too many clustering values: {size}"
106                )));
107            }
108            // 2-byte BE size short, then the values-without-size blob.
109            buf.extend_from_slice(&(size as u16).to_be_bytes());
110            buf.extend_from_slice(&serialize_clustering_prefix_to_vec(ck, schema)?);
111        }
112        // Empty bound (Bottom/Top): size short = 0, no header/values follow.
113        None => buf.extend_from_slice(&0u16.to_be_bytes()),
114    }
115    Ok(buf)
116}

118/// The bound-kind-aware empty-prefix fallback for a marker (Issue #1186): the
119/// marker's correct `Kind.ordinal()` byte followed by a 2-byte big-endian `size`
120/// short of `0` (an empty `ClusteringBound`), matching the bound serializer's
121/// no-values form. Used when the marker's clustering values cannot be encoded.
122pub(super) fn marker_bound_prefix_for_index(bound: &ClusteringBound, is_open: bool) -> Vec<u8> {
123    let (kind, _) = marker_bound_kind(bound, is_open);
124    let mut buf = Vec::with_capacity(3);
125    buf.push(kind);
126    buf.extend_from_slice(&0u16.to_be_bytes());
127    buf
128}

130/// Serialize a range-tombstone **boundary** marker as its promoted-index
131/// (`IndexInfo`) `ClusteringPrefix` byte sequence (issue #1220). A boundary uses
132/// the same `ClusteringBoundOrBoundary.Serializer` wire form as a bound — a kind
133/// byte (2 or 5), a 2-byte big-endian `size` short, then the values-without-size
134/// blob — but its kind is the BOUNDARY ordinal and it always carries a concrete
135/// clustering value (never the empty `Bottom`/`Top` form). Returns `Err` only if a
136/// clustering column type is unknown (the caller falls back via
137/// [`boundary_prefix_for_index_fallback`]).
138pub(super) fn serialize_boundary_prefix_for_index(
139    boundary_kind: u8,
140    clustering: &ClusteringKey,
141    schema: &TableSchema,
142) -> Result<Vec<u8>> {
143    let size = clustering.columns.len();
144    if size > u16::MAX as usize {
145        return Err(crate::error::Error::InvalidInput(format!(
146            "Range tombstone boundary has too many clustering values: {size}"
147        )));
148    }
149    let mut buf = Vec::new();
150    buf.push(boundary_kind);
151    buf.extend_from_slice(&(size as u16).to_be_bytes());
152    buf.extend_from_slice(&serialize_clustering_prefix_to_vec(clustering, schema)?);
153    Ok(buf)
154}

156/// Fallback promoted-index prefix for a boundary whose clustering values cannot be
157/// encoded: the boundary kind byte followed by a `0` size short (an empty bound
158/// form). Mirrors [`marker_bound_prefix_for_index`] for the boundary case.
159pub(super) fn boundary_prefix_for_index_fallback(boundary_kind: u8) -> Vec<u8> {
160    let mut buf = Vec::with_capacity(3);
161    buf.push(boundary_kind);
162    buf.extend_from_slice(&0u16.to_be_bytes());
163    buf
164}

166/// Select the `ClusteringPrefix.Kind` ordinal and clustering values for a marker
167/// bound — the SINGLE source of truth shared by the serializer and the fallback.
168///
169/// Mirrors the `(is_open, bound)` match in [`DataWriter::write_range_bound`]
170/// exactly so the promoted-index kind byte equals the on-disk marker kind byte.
171fn marker_bound_kind(bound: &ClusteringBound, is_open: bool) -> (u8, Option<&ClusteringKey>) {
172    match (is_open, bound) {
173        (true, ClusteringBound::Inclusive(ck)) => (INCL_START_BOUND, Some(ck)),
174        (true, ClusteringBound::Exclusive(ck)) => (EXCL_START_BOUND, Some(ck)),
175        (false, ClusteringBound::Inclusive(ck)) => (INCL_END_BOUND, Some(ck)),
176        (false, ClusteringBound::Exclusive(ck)) => (EXCL_END_BOUND, Some(ck)),
177        (true, ClusteringBound::Bottom | ClusteringBound::Top) => (INCL_START_BOUND, None),
178        (false, ClusteringBound::Bottom | ClusteringBound::Top) => (INCL_END_BOUND, None),
179    }
180}
