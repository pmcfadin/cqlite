1//! Clustering-row grouping and shadow-floor resolution — the ONE place a
2//! partition's clustering rows are selected, grouped and shadow-gated.
3//!
4//! Extracted by issue #4246 (roborev round-8 finding 2). Six call sites had
5//! grown textually-identical copies of the shadow-floor derivation, three of
6//! them wrapped in an identical filter-then-adjacency-group loop:
7//!
8//! | site | what it does with the group |
9//! |------|------------------------------|
10//! | `rows.rs::merge_clustering_rows` | EMITS the row (`merge_row_group`) — the authority |
11//! | `writer/mod.rs::write_partition` | folds the group into persisted `StatisticsMetadata` |
12//! | `writer/mod.rs::compute_mutations_baseline_stats` | folds the group into the #729 encoding baseline |
13//! | `streaming_partition.rs::feed_row` / `incremental_partition.rs::feed_row` | emit one already-reconciled row |
14//! | `merge/mod.rs` + `maintenance.rs` `PartitionEnd` drains | fold one already-reconciled row |
15//!
16//! Issue #4246's whole thesis is that the stats fold must make the SAME
17//! decision the emitter makes. Copies of that decision can drift; a shared
18//! function cannot. Sharing it here is therefore not merely tidiness — it is
19//! the structural form of the property.
20//!
21//! LOAD-BEARING (do not "simplify" away): [`clustering_row_mutations`]'s
22//! `!is_static_row_mutation` filter is what keeps the clustering-group fold
23//! and the separate static-carrier fold DISJOINT. Drop it and a mutation
24//! classified both ways is folded twice — and
25//! `StatisticsMetadata::update_local_deletion_time` is not idempotent, so a
26//! double fold inflates the persisted `estimatedTombstoneDropTime` histogram
27//! Cassandra derives compaction scheduling from.
28
29use super::{is_static_row_mutation, range_tombstone_covers};
30use crate::schema::TableSchema;
31use crate::storage::write_engine::mutation::{ClusteringKey, Mutation, RangeTombstone};
32
33/// A partition's CLUSTERING-ROW mutations, in input order: every mutation
34/// EXCEPT the static-row carriers, whose cells live in the static-row
35/// prelude instead (`is_static_row_mutation`).
36///
37/// See this module's doc comment for why the exclusion is load-bearing.
38pub(crate) fn clustering_row_mutations<'a>(
39    mutations: impl IntoIterator<Item = &'a Mutation>,
40    schema: &TableSchema,
41) -> Vec<&'a Mutation> {
42    mutations
43        .into_iter()
44        .filter(|m| !is_static_row_mutation(m, schema))
45        .collect()
46}

48/// The shadow floor applying to `clustering_key`: the partition tombstone's
49/// deletion timestamp raised by every range tombstone that COVERS that key.
50/// `None` when nothing shadows it.
51///
52/// Mutations at or before this floor are already covered by a
53/// partition/range tombstone, so they are never emitted (and must therefore
54/// never reach persisted statistics either).
55pub(crate) fn resolve_shadow_floor(
56    partition_floor: Option<i64>,
57    range_tombstones: &[RangeTombstone],
58    clustering_key: Option<&ClusteringKey>,
59    schema: &TableSchema,
60) -> Option<i64> {
61    let mut shadow_floor = partition_floor;
62    for rt in range_tombstones {
63        if range_tombstone_covers(rt, clustering_key, schema) {
64            shadow_floor = Some(shadow_floor.map_or(rt.deletion_time, |f| f.max(rt.deletion_time)));
65        }
66    }
67    shadow_floor
68}

70/// Visit each CLUSTERING-KEY GROUP of `row_mutations` (adjacent mutations
71/// sharing a clustering key) together with the group's resolved shadow floor.
72///
73/// `row_mutations` must come from [`clustering_row_mutations`] and must
74/// already be sorted by clustering key — grouping is by ADJACENCY, exactly as
75/// `merge_clustering_rows` has always done, so an unsorted slice silently
76/// splits one logical group in two. The one caller whose input is not
77/// caller-sorted (`compute_mutations_baseline_stats`, reading a
78/// memtable-internal slice) sorts it explicitly before calling.
79pub(crate) fn for_each_clustering_row_group<'a, F>(
80    row_mutations: &[&'a Mutation],
81    schema: &TableSchema,
82    partition_floor: Option<i64>,
83    range_tombstones: &[RangeTombstone],
84    mut visit: F,
85) where
86    F: FnMut(&[&'a Mutation], Option<i64>),
87{
88    let mut start = 0;
89    while start < row_mutations.len() {
90        let mut end = start + 1;
91        while end < row_mutations.len()
92            && row_mutations[end].clustering_key == row_mutations[start].clustering_key
93        {
94            end += 1;
95        }
96        let group = &row_mutations[start..end];
97        let shadow_floor = resolve_shadow_floor(
98            partition_floor,
99            range_tombstones,
100            group[0].clustering_key.as_ref(),
101            schema,
102        );
103        visit(group, shadow_floor);
104        start = end;
105    }
106}
