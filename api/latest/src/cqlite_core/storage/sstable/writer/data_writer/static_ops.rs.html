1//! Incremental static-column last-write-wins tracker (issue #1668, stage
2//! 5c-ii).
3//!
4//! `collect_static_operations` (`encoding.rs`) used to scan a whole
5//! `&[Mutation]` slice in one pass, folding each static-column op into a
6//! `best: HashMap<String, StaticMergedOp>` last-write-wins map. That fold is
7//! ALREADY incremental in spirit — nothing about it actually needs the whole
8//! slice materialized upfront, it just happened to be written as a `for`
9//! loop over a slice. This module extracts the SAME per-mutation logic into
10//! [`StaticOpsTracker`]: a running tracker with `feed()` (one `Mutation` at a
11//! time) and `finish()` (the same `Vec<StaticMergedOp>` the whole-slice
12//! version returns), so a FUTURE incremental writer entry point (stage
13//! 5c-iv) can feed it one cluster group at a time instead of requiring a
14//! fully-materialized `Vec<Mutation>` first.
15//!
16//! `collect_static_operations` itself is now a THIN WRAPPER over this
17//! tracker — same signature, same behavior, byte-identical output — so the
18//! CURRENT production call site (`DataWriter::write_partition_with_index_blocks`)
19//! is completely unaffected by this stage; only the internal mechanism
20//! changed shape.
21//!
22//! ## Order-dependence note (why `feed()` order must match today's slice order)
23//!
24//! The last-write-wins tie-break is `>=`, not `>` (see [`StaticOpsTracker::feed`]):
25//! at an EXACT timestamp tie, the LATER-fed candidate wins. This is
26//! observable, not incidental — `collect_static_operations` iterates
27//! `mutations` in whatever order the caller's slice presents them, and
28//! `write_partition` sorts that slice by clustering key before this runs. A
29//! future incremental caller (5c-iv) MUST feed cluster groups in the SAME
30//! order `write_partition`'s sort would have produced for this to remain
31//! byte-identical — not resolved here (this stage only proves the TRACKER
32//! itself reproduces today's whole-slice result when fed in the SAME order).

36/// Running last-write-wins tracker for static-column operations (issue
37/// #1668, stage 5c-ii). See the module doc for the order-dependence note.
38#[derive(Default)]
39pub(crate) struct StaticOpsTracker {
40    /// column_name → winning `StaticMergedOp` seen so far.
41    best: std::collections::HashMap<String, StaticMergedOp>,
42}
43
44impl StaticOpsTracker {
45    /// A fresh tracker with no candidates yet.
46    pub(crate) fn new() -> Self {
47        Self::default()
48    }
49
50    /// Fold one mutation's static-column operations into the running
51    /// last-write-wins map. Mirrors `collect_static_operations`'s per-
52    /// mutation loop body EXACTLY (mutation-level and per-cell shadow-floor
53    /// skips, per-cell writetime resolution, `>=` tie-break).
54    pub(crate) fn feed(
55        &mut self,
56        mutation: &Mutation,
57        schema: &TableSchema,
58        shadow_floor: Option<i64>,
59    ) {
60        if shadow_floor.is_some_and(|floor| mutation.timestamp_micros <= floor) {
61            return;
62        }
63        for op in &mutation.operations {
64            if !is_static_operation(op, schema) {
65                continue;
66            }
67            let col_name = match op {
68                crate::storage::write_engine::mutation::CellOperation::Write { column, .. }
69                | crate::storage::write_engine::mutation::CellOperation::WriteWithTtl {
70                    column,
71                    ..
72                }
73                | crate::storage::write_engine::mutation::CellOperation::Delete {
74                    column, ..
75                } => column.clone(),
76                // Per-element complex ops are not produced for STATIC complex
77                // columns by the current capability — they flow through the
78                // regular-row per-element path. Skip them here defensively,
79                // matching `collect_static_operations`.
80                crate::storage::write_engine::mutation::CellOperation::WriteComplexElement {
81                    ..
82                }
83                | crate::storage::write_engine::mutation::CellOperation::ComplexDeletion {
84                    ..
85                } => continue,
86                crate::storage::write_engine::mutation::CellOperation::DeleteRow => continue,
87            };
88            let candidate_ts = op_cell_write_timestamp(op, mutation);
89            if shadow_floor.is_some_and(|floor| candidate_ts <= floor)
90                && matches!(
91                    op,
92                    crate::storage::write_engine::mutation::CellOperation::Write { .. }
93                        | crate::storage::write_engine::mutation::CellOperation::WriteWithTtl { .. }
94                        | crate::storage::write_engine::mutation::CellOperation::Delete { .. }
95                )
96            {
97                continue;
98            }
99            let candidate = StaticMergedOp {
100                cell_local_deletion_time: op_cell_local_deletion_time(op, mutation),
101                op: op.clone(),
102                timestamp_micros: candidate_ts,
103                row_ttl_seconds: mutation.ttl_seconds,
104            };
105            match self.best.entry(col_name) {
106                std::collections::hash_map::Entry::Vacant(entry) => {
107                    entry.insert(candidate);
108                }
109                std::collections::hash_map::Entry::Occupied(mut entry) => {
110                    if candidate.timestamp_micros >= entry.get().timestamp_micros {
111                        entry.insert(candidate);
112                    }
113                }
114            }
115        }
116    }
117
118    /// Consume the tracker and return the resolved static operations, one
119    /// per distinct static column, in an unspecified order (the writer sorts
120    /// them by schema column order when building the row body) — matches
121    /// `collect_static_operations`'s existing contract exactly.
122    pub(crate) fn finish(self) -> Vec<StaticMergedOp> {
123        self.best.into_values().collect()
124    }
125}

172    fn static_write(column: &str, value: Value, ts: i64) -> Mutation {
173        let table_id = TableId::new("ks", "t");
174        let pk = PartitionKey::single("id", Value::Integer(1));
175        let ops = vec![CellOperation::Write {
176            column: column.to_string(),
177            value,
178        }];
179        Mutation::new(table_id, pk, None, ops, ts, None)
180    }
181
182    fn static_op_text(ops: &[StaticMergedOp], column: &str) -> Option<String> {
183        ops.iter().find_map(|m| match &m.op {
184            CellOperation::Write {
185                column: c,
186                value: Value::Text(s),
187            } if c == column => Some(s.clone()),
188            _ => None,
189        })
190    }

192    /// THE proof: feeding mutations ONE AT A TIME via `StaticOpsTracker`
193    /// resolves the SAME winner as `collect_static_operations`'s whole-slice
194    /// scan, for a fixture with a genuine last-write-wins conflict — checked
195    /// against an INDEPENDENTLY reasoned expected winner (the strictly
196    /// highest timestamp), not just old-path-equals-new-path.
197    #[test]
198    fn tracker_matches_whole_slice_for_lww_conflict() {
199        let schema = schema_with_static();
200        let mutations = vec![
201            static_write("region", Value::Text("us-east".to_string()), 100),
202            static_write("region", Value::Text("us-west".to_string()), 300), // wins: highest ts
203            static_write("region", Value::Text("eu-west".to_string()), 200),
204            static_write("quota", Value::Integer(10), 150),
205        ];
206
207        // Whole-slice (today's production path, unchanged).
208        let whole_slice = collect_static_operations(&mutations, &schema, None);
209
210        // Incremental: fed one mutation at a time, in the SAME arrival order.
211        let mut tracker = StaticOpsTracker::new();
212        for m in &mutations {
213            tracker.feed(m, &schema, None);
214        }
215        let incremental = tracker.finish();
216
217        // Independent expectation: "us-west" (ts=300) is strictly the
218        // highest timestamp among the "region" candidates.
219        assert_eq!(
220            static_op_text(&whole_slice, "region"),
221            Some("us-west".to_string()),
222            "whole-slice path must resolve the strictly-highest-timestamp winner"
223        );
224        assert_eq!(
225            static_op_text(&incremental, "region"),
226            Some("us-west".to_string()),
227            "incremental tracker must resolve the SAME strictly-highest-timestamp winner"
228        );
229        assert_eq!(
230            whole_slice.len(),
231            incremental.len(),
232            "both paths must resolve the same NUMBER of distinct static columns"
233        );
234    }
235
236    /// Tie-break proof: at an EXACT timestamp tie, the LATER-fed candidate
237    /// wins (the `>=` comparison) — proven independently by construction
238    /// (the fixture's SECOND write at the tied timestamp is the one that
239    /// must survive), and confirmed identical between the incremental
240    /// tracker and the whole-slice path.
241    #[test]
242    fn tracker_matches_whole_slice_for_exact_timestamp_tie() {
243        let schema = schema_with_static();
244        let mutations = vec![
245            static_write("region", Value::Text("first".to_string()), 500),
246            static_write("region", Value::Text("second".to_string()), 500), // later-fed at the SAME ts: wins
247        ];
248
249        let whole_slice = collect_static_operations(&mutations, &schema, None);
250        let mut tracker = StaticOpsTracker::new();
251        for m in &mutations {
252            tracker.feed(m, &schema, None);
253        }
254        let incremental = tracker.finish();
255
256        assert_eq!(
257            static_op_text(&whole_slice, "region"),
258            Some("second".to_string()),
259            "at an exact timestamp tie, the LATER mutation in arrival order must win"
260        );
261        assert_eq!(
262            static_op_text(&incremental, "region"),
263            Some("second".to_string()),
264            "incremental tracker must resolve the tie identically to the whole-slice path"
265        );
266    }

268    /// Shadow-floor proof: a mutation at or before the partition-tombstone
269    /// floor contributes NOTHING, in both paths.
270    #[test]
271    fn tracker_matches_whole_slice_with_shadow_floor() {
272        let schema = schema_with_static();
273        let mutations = vec![
274            static_write("region", Value::Text("shadowed".to_string()), 100), // <= floor
275            static_write("region", Value::Text("survivor".to_string()), 300), // > floor
276        ];
277        let shadow_floor = Some(200i64);
278
279        let whole_slice = collect_static_operations(&mutations, &schema, shadow_floor);
280        let mut tracker = StaticOpsTracker::new();
281        for m in &mutations {
282            tracker.feed(m, &schema, shadow_floor);
283        }
284        let incremental = tracker.finish();

286        assert_eq!(
287            static_op_text(&whole_slice, "region"),
288            Some("survivor".to_string())
289        );
290        assert_eq!(
291            static_op_text(&incremental, "region"),
292            Some("survivor".to_string()),
293            "incremental tracker must apply the SAME shadow-floor gate"
294        );
295    }
