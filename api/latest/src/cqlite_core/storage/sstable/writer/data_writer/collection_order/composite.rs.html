1//! Recursive comparators for COMPOSITE SET-element / MAP-key types (issue #1296).
2//!
3//! Follow-up to the scalar comparator of #1275: for composite element/key types
4//! Cassandra does NOT compare raw serialized bytes — it recurses into each
5//! field/element's own `AbstractType.compare`. This module mirrors those
6//! `compareCustom` implementations exactly, recursing back into the shared
7//! dispatcher [`super::compare_collection_elements`] so the SCALAR leaves reuse
8//! the #1275 signed/float/varint/decimal/uuid logic (no reimplementation).
9//!
10//! Oracle: Cassandra 5.0 `org.apache.cassandra.db.marshal`
11//!   * `TupleType.compareCustom` / `UserType` (extends `TupleType`):
12//!     field-by-field in declared order. Each field is `[int32 len][bytes]` where
13//!     `len < 0` is a null; a null field sorts BEFORE a non-null one; two nulls
14//!     at the same position are equal and comparison continues. The first
15//!     non-equal field decides. When one tuple runs out of fields first (a
16//!     prefix), the shorter sorts first.
17//!   * `SetType.compareCustom` (frozen): a frozen set is stored element-SORTED, so
18//!     we compare the SORTED element sequences (sort each side with the same
19//!     recursive comparator FIRST), then the shorter collection sorts first. The
20//!     sort-first step is what canonicalizes nested inner sets BOTTOM-UP so the
21//!     outer ordering matches the (sorted) inner bytes actually written (#1296).
22//!   * `ListType.compareCustom` (frozen): INSERTION order preserved — element-by-
23//!     element with NO sort, then the shorter list sorts first.
24//!   * `MapType.compareCustom` (frozen): a frozen map is stored KEY-SORTED, so we
25//!     compare the KEY-SORTED entry sequences (sort each side by key FIRST), then
26//!     per entry compare KEY then VALUE, then the shorter map sorts first.
27//!
28//! The decision is driven ENTIRELY by the `Value` variant — which carries the
29//! authoritative CQL type metadata (a `Value::Tuple`/`Value::Udt`/nested
30//! `Value::Set`/`Value::List`/`Value::Map`) — never by inspecting raw bytes
31//! (no-heuristics, issue #28). A composite whose element/field variant is unknown
32//! or mixed bottoms out at the dispatcher's serialized-byte fallback.
33

38/// Is this field-slot value a CQL null for comparison purposes? Both the explicit
39/// `Value::Null` placeholder (tuple slots) and a missing UDT field are nulls.
40fn is_null(v: &Value) -> bool {
41    matches!(v, Value::Null)
42}
43
44/// `TupleType.compareCustom`: compare two tuples field-by-field in positional
45/// order. A null field sorts before a non-null one; equal-position nulls continue;
46/// the first non-equal field decides; a shorter tuple that is a prefix sorts first.
47pub(super) fn compare_tuple(a: &[Value], b: &[Value]) -> Ordering {
48    let common = a.len().min(b.len());
49    for i in 0..common {
50        // `get` is always Some here (i < common <= len); fall back to Null to stay
51        // panic-free without an unwrap if that invariant ever changes.
52        let ea = a.get(i).unwrap_or(&Value::Null);
53        let eb = b.get(i).unwrap_or(&Value::Null);
54        match (is_null(ea), is_null(eb)) {
55            (true, true) => continue,
56            (true, false) => return Ordering::Less,
57            (false, true) => return Ordering::Greater,
58            (false, false) => match compare_collection_elements(ea, eb) {
59                Ordering::Equal => continue,
60                other => return other,
61            },
62        }
63    }
64    // Prefix rule: the tuple that ran out of fields first sorts first.
65    a.len().cmp(&b.len())
66}

68/// `UserType.compareCustom` (UserType extends TupleType): compare two UDT values
69/// field-by-field in DECLARED order. A missing field value (`None`) — or an
70/// explicit `Value::Null` — is a null, sorting before a non-null; the first
71/// non-equal field decides; a shorter field list that is a prefix sorts first.
72pub(super) fn compare_udt(a: &UdtValue, b: &UdtValue) -> Ordering {
73    let common = a.fields.len().min(b.fields.len());
74    for i in 0..common {
75        let va = a.fields.get(i).and_then(|f| f.value.as_ref());
76        let vb = b.fields.get(i).and_then(|f| f.value.as_ref());
77        // A `None` field value OR a `Some(Value::Null)` both count as null.
78        let na = va.is_none_or(is_null);
79        let nb = vb.is_none_or(is_null);
80        match (na, nb) {
81            (true, true) => continue,
82            (true, false) => return Ordering::Less,
83            (false, true) => return Ordering::Greater,
84            (false, false) => {
85                // Both non-null by the check above, so `va`/`vb` are `Some`.
86                match (va, vb) {
87                    (Some(x), Some(y)) => match compare_collection_elements(x, y) {
88                        Ordering::Equal => continue,
89                        other => return other,
90                    },
91                    // Unreachable given na/nb above; keep the sort total.
92                    _ => continue,
93                }
94            }
95        }
96    }
97    a.fields.len().cmp(&b.fields.len())
98}

100/// Element-wise compare of two ALREADY-ORDERED ref sequences: the first non-equal
101/// element decides; otherwise the shorter (prefix) sequence sorts first.
102fn compare_ordered_seq(a: &[&Value], b: &[&Value]) -> Ordering {
103    let common = a.len().min(b.len());
104    for i in 0..common {
105        // `i < common <= len` so `get` is always Some; fall back to Null to stay
106        // panic-free without an unwrap if that invariant ever changes.
107        let ea = a.get(i).copied().unwrap_or(&Value::Null);
108        let eb = b.get(i).copied().unwrap_or(&Value::Null);
109        match compare_collection_elements(ea, eb) {
110            Ordering::Equal => continue,
111            other => return other,
112        }
113    }
114    a.len().cmp(&b.len())
115}

117/// `SetType.compareCustom` (frozen): a `frozen<set<T>>` serializes its elements in
118/// the element-type comparator's order (see `encoding.rs` Set arm), so its
119/// canonical on-disk form is element-SORTED. We therefore compare the SORTED
120/// element sequences: sort each side with the SAME recursive comparator FIRST,
121/// then element-by-element with a shorter-first tiebreak. Sorting first is what
122/// makes nested frozen sets order correctly BOTTOM-UP — the inner set is
123/// canonicalized before the outer set orders by it — even when the in-memory
124/// `Value::Set` is provided with its elements in a non-canonical order (#1296).
125pub(super) fn compare_set(a: &[Value], b: &[Value]) -> Ordering {
126    let mut sa: Vec<&Value> = a.iter().collect();
127    let mut sb: Vec<&Value> = b.iter().collect();
128    // PERF (correctness-first, #1296): each pairwise call re-sorts both inner
129    // sets, so an OUTER sort that compares this set O(N log N) times re-sorts the
130    // same inner collections O(N log N) times — correct but redundant. A future
131    // optimization is to canonicalize (sort) each element ONCE before the outer
132    // sort; intentionally out of scope here (no behavioral change).
133    sa.sort_by(|x, y| compare_collection_elements(x, y));
134    sb.sort_by(|x, y| compare_collection_elements(x, y));
135    compare_ordered_seq(&sa, &sb)
136}

138/// `ListType.compareCustom` (frozen): a `frozen<list<T>>` serializes its elements
139/// in INSERTION order (no sort), so compare element-by-element in stored order
140/// with a shorter-first tiebreak — NEVER sorted.
141pub(super) fn compare_list(a: &[Value], b: &[Value]) -> Ordering {
142    let sa: Vec<&Value> = a.iter().collect();
143    let sb: Vec<&Value> = b.iter().collect();
144    compare_ordered_seq(&sa, &sb)
145}

147/// `MapType.compareCustom` (frozen): a `frozen<map<K,V>>` serializes its entries in
148/// the KEY-type comparator's order (see `encoding.rs` Map arm), so its canonical
149/// on-disk form is KEY-SORTED. We compare the KEY-SORTED entry sequences: sort each
150/// side's entries by key with the SAME recursive comparator FIRST, then per entry
151/// compare KEY then VALUE, with a shorter-first tiebreak. Sorting first makes
152/// nested frozen maps order correctly BOTTOM-UP even when the in-memory
153/// `Value::Map` is provided with its entries in a non-canonical order (#1296).
154pub(super) fn compare_map(a: &[(Value, Value)], b: &[(Value, Value)]) -> Ordering {
155    let mut sa: Vec<&(Value, Value)> = a.iter().collect();
156    let mut sb: Vec<&(Value, Value)> = b.iter().collect();
157    // PERF (correctness-first, #1296): each pairwise call re-sorts both inner
158    // maps by key, so an OUTER sort that compares this map O(N log N) times
159    // re-sorts the same inner collections O(N log N) times — correct but
160    // redundant. A future optimization is to canonicalize (key-sort) each map
161    // ONCE before the outer sort; intentionally out of scope here (no behavioral
162    // change).
163    sa.sort_by(|x, y| compare_collection_elements(&x.0, &y.0));
164    sb.sort_by(|x, y| compare_collection_elements(&x.0, &y.0));
165    let common = sa.len().min(sb.len());
166    for i in 0..common {
167        let (ka, val_a) = match sa.get(i) {
168            Some(e) => (&e.0, &e.1),
169            None => (&Value::Null, &Value::Null),
170        };
171        let (kb, val_b) = match sb.get(i) {
172            Some(e) => (&e.0, &e.1),
173            None => (&Value::Null, &Value::Null),
174        };
175        match compare_collection_elements(ka, kb) {
176            Ordering::Equal => {}
177            other => return other,
178        }
179        match compare_collection_elements(val_a, val_b) {
180            Ordering::Equal => continue,
181            other => return other,
182        }
183    }
184    sa.len().cmp(&sb.len())
185}

430    /// `frozen<list<T>>` PRESERVES INSERTION ORDER (it is NOT sorted) — the most
431    /// error-prone part of the contract: sets/maps canonicalize (sort) but lists
432    /// do not. `[3,1]` vs `[1,3]` is decided POSITIONALLY by element 0 (3 vs 1) →
433    /// `Greater`. If `compare_list` sorted (like a set), both would canonicalize to
434    /// `[1,3]` and the result would be `Equal` — so a `Greater`/`Less` (non-equal)
435    /// result is exactly what proves no sort happens. The SAME elements as a SET
436    /// DO compare equal, pinning the list-vs-set distinction.
437    #[test]
438    fn frozen_list_preserves_insertion_order_not_sorted() {
439        let l_3_1 = vec![Value::Integer(3), Value::Integer(1)];
440        let l_1_3 = vec![Value::Integer(1), Value::Integer(3)];
441        // List: positional — element 0 (3 vs 1) decides → Greater (NOT sorted).
442        assert_eq!(compare_list(&l_3_1, &l_1_3), Ordering::Greater);
443        assert_eq!(compare_list(&l_1_3, &l_3_1), Ordering::Less);
444        // Via the dispatcher (Value::List arm) — same positional result.
445        assert_eq!(
446            compare_collection_elements(&Value::List(l_3_1.clone()), &Value::List(l_1_3.clone())),
447            Ordering::Greater
448        );
449        // CONTRAST: the SAME elements as a SET canonicalize (sort) to {1,3} on both
450        // sides → Equal. A list of the same elements is NOT equal, proving lists are
451        // ordered element-wise positionally while sets are reordered.
452        assert_eq!(
453            compare_collection_elements(&Value::Set(l_3_1.clone()), &Value::Set(l_1_3.clone())),
454            Ordering::Equal
455        );
456        assert_ne!(
457            compare_collection_elements(&Value::List(l_3_1), &Value::List(l_1_3)),
458            Ordering::Equal
459        );
460    }

462    /// `frozen<list<T>>` compared ELEMENT-WISE POSITIONALLY (not reordered): when
463    /// element 0 ties, element 1 decides in stored position. `[1,3]` < `[1,9]` on
464    /// element 1 (3 < 9), and the prefix `[1]` < `[1,0]` (shorter-first tiebreak) —
465    /// note `[1,0]` would sort to `[0,1]` and flip if lists were sorted, so the
466    /// `[1] < [1,0]` result also confirms positional (un-sorted) comparison.

470        let l_1_9 = vec![Value::Integer(1), Value::Integer(9)];
471        assert_eq!(compare_list(&l_1_3, &l_1_9), Ordering::Less);
