1//! The two CELL-PATH-BEARING multicell collection cell writers — SET and MAP.
2//!
3//! Campsite split out of `complex.rs` (epic #1116) by issue #4106, which had to
4//! grow `write_set_complex_cells` and could not grow a file already ~200 lines
5//! over the ratchet ceiling. The responsibility line is Cassandra's own: these
6//! are the two collections whose CELL PATH carries a component of the declared
7//! type (`nameComparator()` — the keys type for a `MapType`, the elements type
8//! for a `SetType`, `schema/ColumnMetadata.java:457-467` at `cassandra-5.0.8`),
9//! so both serialize their path through the schema-aware
10//! [`super::cell_path`] and both must thread the column's declared type down to
11//! reach it. `write_list_complex_cells` stays in `complex.rs`: a LIST's cell path
12//! is a generated 16-byte TimeUUID, not a declared-type component, so it needs no
13//! declared type and shares nothing with these two.
14//!
15//! `use super::*` pulls the shared writer types, serialization/schema helpers,
16//! flag constants and crate imports re-exported from `data_writer/mod.rs`. No
17//! emitted bytes change from the move.
18
19use super::*;
20
21impl DataWriter {
22    /// Write SET complex cells.
23    ///
24    /// SET elements: cell_path = serialized element value, cell value = empty (HAS_EMPTY_VALUE).
25    /// Elements are ordered by the element type's Cassandra `SetType` comparator (#1275).
26    ///
27    /// `set_data_type` is the COLUMN's DECLARED type, threaded down for exactly
28    /// the reason the MAP sibling threads its own: the cell path is the one write
29    /// position where an empty-buffer sentinel is legal, and legality there
30    /// depends on the declared ELEMENT type, which a bare `Value` cannot supply
31    /// (#4106; the shared doctrine is in [`super::cell_path`]'s module header).
32    pub(super) fn write_set_complex_cells(
33        &self,
34        buf: &mut Vec<u8>,
35        value: &Value,
36        set_data_type: &str,
37        timestamp_micros: i64,
38        ttl_seconds: Option<u32>,
39        now_seconds: i32,
40    ) -> Result<()> {
41        let elements = match value {
42            Value::Set(elements) => elements,
43            _ => {
44                return Err(Error::InvalidInput(format!(
45                    "Expected Set value for complex SET column, got {:?}",
46                    value
47                )))
48            }
49        };
50
51        // Order by the element type's Cassandra `SetType` comparator (#1275, see
52        // collection_order: SIGNED numerics, unsigned-byte otherwise) decided from
53        // the element `Value`s. Null is rejected by the cell-path serializer.
54        let mut ordered: Vec<&Value> = elements.iter().collect();
55        ordered.sort_by(|a, b| compare_collection_elements(a, b));
56        let mut serialized: Vec<Vec<u8>> = Vec::with_capacity(ordered.len());
57        for element in &ordered {
58            let mut path = Vec::new();
59            serialize_set_cell_path_element_into(element, set_data_type, &mut path)?;
60            serialized.push(path);
61        }
62
63        encode_unsigned(serialized.len() as u64, buf); // cell count
64        for path_bytes in &serialized {
65            // Cell header: flags + optional TTL fields
66            self.write_complex_cell_header(
67                buf,
68                CELL_HAS_EMPTY_VALUE,
69                timestamp_micros,
70                ttl_seconds,
71                now_seconds,
72            )?;
73
74            // Cell path: serialized element value
75            encode_unsigned(path_bytes.len() as u64, buf);
76            buf.extend_from_slice(path_bytes);
77            // No value bytes (HAS_EMPTY_VALUE flag set)
78        }
79
80        Ok(())
81    }
82
83    /// Write MAP complex cells.
84    ///
85    /// MAP entries: cell_path = serialized key, cell value = serialized value.
86    /// Entries are sorted by their serialized key byte representation for Cassandra compatibility.
87    ///
88    /// `map_data_type` is the COLUMN's DECLARED type (e.g. `map<int, int>`),
89    /// threaded down so the cell path can admit an empty-buffer sentinel against
90    /// the declared KEY type (#3805; doctrine in [`super::cell_path`]). The cell
91    /// VALUE deliberately keeps the type-blind [`serialize_value_into`], which
92    /// REFUSES a sentinel: a zero-byte map VALUE is not a sentinel, it is the
93    /// empty value of the value type — or, with `HAS_EMPTY_VALUE`, a null.
94    pub(super) fn write_map_complex_cells(
95        &self,
96        buf: &mut Vec<u8>,
97        value: &Value,
98        map_data_type: &str,
99        timestamp_micros: i64,
100        ttl_seconds: Option<u32>,
101        now_seconds: i32,
102    ) -> Result<()> {
103        let entries = match value {
104            Value::Map(entries) => entries,
105            _ => {
106                return Err(Error::InvalidInput(format!(
107                    "Expected Map value for complex MAP column, got {:?}",
108                    value
109                )))
110            }
111        };
112
113        // Order by the KEY type's Cassandra `MapType` comparator (#1275, see
114        // collection_order: SIGNED numerics so negative keys sort -1 before 0/1,
115        // unsigned-byte otherwise) from the key `Value`s. Null keys rejected inline.
116        let mut ordered: Vec<&(Value, Value)> = entries.iter().collect();
117        ordered.sort_by(|a, b| compare_collection_elements(&a.0, &b.0));
118
119        // Reusable per-entry scratch (issue #1672): one alloc for the whole map,
120        // not a Vec-of-Vecs holding every key/value.
121        encode_unsigned(ordered.len() as u64, buf); // cell count
122        let mut key_scratch = Vec::new();
123        let mut val_scratch = Vec::new();
124        for (key, val) in ordered {
125            if matches!(key, Value::Null) {
126                return Err(Error::InvalidInput(
127                    "MAP keys cannot be null (CQL semantics)".to_string(),
128                ));
129            }
130
131            // Cell header: flags + optional TTL fields
132            self.write_complex_cell_header(buf, 0, timestamp_micros, ttl_seconds, now_seconds)?;
133
134            // Cell path: serialized key. SCHEMA-AWARE, because this is the one
135            // position an empty-buffer sentinel may occupy (issue #3805) and its
136            // tag must be validated against the DECLARED key type.
137            key_scratch.clear();
138            serialize_map_cell_path_key_into(key, map_data_type, &mut key_scratch)?;
139            encode_unsigned(key_scratch.len() as u64, buf);
140            buf.extend_from_slice(&key_scratch);
141
142            // Cell value: serialized value
143            val_scratch.clear();
144            serialize_value_into(val, &mut val_scratch)?;
145            encode_unsigned(val_scratch.len() as u64, buf);
146            buf.extend_from_slice(&val_scratch);
147        }
148
149        Ok(())
150    }
