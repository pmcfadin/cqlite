10/// One unfiltered to emit inside a partition body, in clustering order: a regular
11/// row, a single range-tombstone BOUND marker, or a coalesced range-tombstone
12/// BOUNDARY marker (issue #1220).
13///
14/// Lifted to module scope (from the two per-function local enums it replaced) so
15/// the marker-coalescing pass ([`coalesce_boundaries`]) is shared verbatim by both
16/// the plain [`DataWriter::write_partition`] and the wide-partition
17/// [`DataWriter::write_partition_with_index_blocks`] emitters — they must produce
18/// byte-identical bodies, boundary coalescing included.
19enum PartitionItem<'a> {
20    Row(RowWrite<'a>),
21    Marker {
22        bound: &'a ClusteringBound,
23        is_open: bool,
24        deletion_time: i64,
25        local_deletion_time: i32,
26    },
27    /// A coalesced boundary: closes the previous range and opens the next at the
28    /// same `clustering` point. `end_*` is the closing range's deletion (primary),
29    /// `start_*` is the opening range's deletion (secondary).
30    Boundary {
31        kind: u8,
32        clustering: &'a ClusteringKey,
33        end_deletion_time: i64,
34        end_local_deletion_time: i32,
35        start_deletion_time: i64,
36        start_local_deletion_time: i32,
37    },
38}

40/// The Cassandra `ClusteringPrefix.Kind` ordinal for a range-tombstone BOUND.
41/// Used purely as a sort tiebreak (see [`sort_class`]); the canonical ordering is
42/// the one decoded on the read path (`v5_compressed_legacy/block_emit.rs`) and the
43/// constants in `data_writer/mod.rs`:
44///   0 EXCL_END_BOUND · 1 INCL_START_BOUND · 6 INCL_END_BOUND · 7 EXCL_START_BOUND.
45fn bound_kind_ordinal(bound: &ClusteringBound, is_open: bool) -> u8 {
46    match (is_open, bound) {
47        (true, ClusteringBound::Inclusive(_)) => INCL_START_BOUND, // 1
48        (false, ClusteringBound::Exclusive(_)) => EXCL_END_BOUND,  // 0
49        (false, ClusteringBound::Inclusive(_)) => INCL_END_BOUND,  // 6
50        (true, ClusteringBound::Exclusive(_)) => EXCL_START_BOUND, // 7
51        // Open-ended bounds are positioned by `class` (Bottom/Top), never reach a
52        // value-level tiebreak; report their side's kind for completeness.
53        (true, ClusteringBound::Bottom | ClusteringBound::Top) => INCL_START_BOUND,
54        (false, ClusteringBound::Bottom | ClusteringBound::Top) => INCL_END_BOUND,
55    }
56}

58/// Sort key for clustering-ordered emission: `(position class, clustering values,
59/// bound weight, kind ordinal)`. class: -1 = before all rows (Bottom), 0 =
60/// positioned by clustering values, 1 = after all rows (Top). The weight orders a
61/// marker relative to a row at the SAME clustering value
62/// (`ClusteringPrefix.Kind.comparedToClustering`): an inclusive-start /
63/// exclusive-end bound sorts before the row, an inclusive-end / exclusive-start
64/// bound after it.
65///
66/// The final element is the Cassandra `ClusteringPrefix.Kind` ordinal, used as a
67/// strict tiebreak so that — at an equal clustering point and equal weight — the
68/// CLOSING bound of one range always sorts immediately before the matching OPENING
69/// bound of the next, regardless of the order the `range_tombstones` arrived in
70/// (issue #1220, roborev finding 2). This mirrors Cassandra's comparator, whose
71/// distinct Kind ordinals disambiguate co-located bounds: EXCL_END(0) < INCL_START(1)
72/// for a kind-2 boundary and INCL_END(6) < EXCL_START(7) for a kind-5 boundary —
73/// close before open in both cases. Without it, two co-located bounds carry the same
74/// `weight` and a stable sort would leave them in input order, so a reversed input
75/// vector would emit an open-before-close pair that [`coalesce_boundaries`] cannot
76/// merge (parity loss + a non-Cassandra marker order).
77///
78/// Rows take the CLUSTERING ordinal (4); since a row's `weight` (0) already differs
79/// from every bound's weight (-1/+1) at the same clustering, the ordinal only ever
80/// breaks ties between two bounds. Boundaries are produced only AFTER this sort (by
81/// [`coalesce_boundaries`]), so the `Boundary` arm is never exercised during
82/// sorting; it is provided for exhaustiveness and positions the boundary at its
83/// clustering value.
84fn sort_class<'a, 'b>(item: &'b PartitionItem<'a>) -> (i8, Option<&'b ClusteringKey>, i8, u8) {
85    match item {
86        PartitionItem::Row(row) => (0, row.clustering_key, 0, CLUSTERING),
87        PartitionItem::Marker { bound, is_open, .. } => {
88            let kind = bound_kind_ordinal(bound, *is_open);
89            match bound {
90                ClusteringBound::Inclusive(ck) => {
91                    (0, Some(ck), if *is_open { -1 } else { 1 }, kind)
92                }
93                ClusteringBound::Exclusive(ck) => {
94                    (0, Some(ck), if *is_open { 1 } else { -1 }, kind)
95                }
96                ClusteringBound::Bottom => (-1, None, 0, kind),
97                ClusteringBound::Top => (1, None, 0, kind),
98            }
99        }
100        PartitionItem::Boundary {
101            kind, clustering, ..
102        } => (0, Some(clustering), 0, *kind),
103    }
104}

106/// Sort `items` into clustering order (rows + bound markers), schema-aware.
107fn sort_partition_items(items: &mut [PartitionItem], schema: &TableSchema) {
108    items.sort_by(|a, b| {
109        let (class_a, ck_a, weight_a, kind_a) = sort_class(a);
110        let (class_b, ck_b, weight_b, kind_b) = sort_class(b);
111        class_a
112            .cmp(&class_b)
113            .then_with(|| match (ck_a, ck_b) {
114                (Some(x), Some(y)) => x.compare(y, schema).unwrap_or_else(|_| x.cmp(y)),
115                _ => std::cmp::Ordering::Equal,
116            })
117            .then(weight_a.cmp(&weight_b))
118            .then(kind_a.cmp(&kind_b))
119    });
120}

122/// The BOUNDARY kind for an adjacent close+open bound pair at the same clustering
123/// point, or `None` when the two bounds do NOT form a boundary (issue #1220).
124///
125/// Cassandra coalesces two adjacent range tombstones into a single boundary only
126/// when the closing bound and the opening bound meet at the same clustering value
127/// with COMPLEMENTARY inclusivity (exactly one of them inclusive) — otherwise the
128/// ranges either overlap (both inclusive) or leave a gap (both exclusive) and stay
129/// separate bounds. Returns the boundary kind ordinal plus the shared clustering.
130fn boundary_kind_for<'a>(
131    close: &'a ClusteringBound,
132    open: &'a ClusteringBound,
133    schema: &TableSchema,
134) -> Option<(u8, &'a ClusteringKey)> {
135    let same = |a: &ClusteringKey, b: &ClusteringKey| {
136        a.compare(b, schema)
137            .map(|o| o == std::cmp::Ordering::Equal)
138            .unwrap_or(false)
139    };
140    match (close, open) {
141        (ClusteringBound::Exclusive(c1), ClusteringBound::Inclusive(c2)) if same(c1, c2) => {
142            Some((EXCL_END_INCL_START_BOUNDARY, c2))
143        }
144        (ClusteringBound::Inclusive(c1), ClusteringBound::Exclusive(c2)) if same(c1, c2) => {
145            Some((INCL_END_EXCL_START_BOUNDARY, c1))
146        }
147        _ => None,
148    }
149}

151/// Coalesce each adjacent (close BOUND, open BOUND) pair that meets at the same
152/// clustering point with complementary inclusivity into a single BOUNDARY item
153/// (issue #1220). After [`sort_partition_items`], the closing bound of one range
154/// and the opening bound of the next sit at the same sort position (equal class,
155/// clustering and weight) with the close first, so no row can fall between them;
156/// this single forward pass merges them into one two-deletion-time boundary.
157fn coalesce_boundaries<'a>(
158    items: Vec<PartitionItem<'a>>,
159    schema: &TableSchema,
160) -> Vec<PartitionItem<'a>> {
161    let mut out: Vec<PartitionItem<'a>> = Vec::with_capacity(items.len());
162    let mut iter = items.into_iter().peekable();
163    while let Some(item) = iter.next() {
164        if let PartitionItem::Marker {
165            bound: close_bound,
166            is_open: false,
167            deletion_time: end_dt,
168            local_deletion_time: end_ldt,
169        } = &item
170        {
171            if let Some(PartitionItem::Marker {
172                bound: open_bound,
173                is_open: true,
174                deletion_time: start_dt,
175                local_deletion_time: start_ldt,
176            }) = iter.peek()
177            {
178                if let Some((kind, clustering)) = boundary_kind_for(close_bound, open_bound, schema)
179                {
180                    let boundary = PartitionItem::Boundary {
181                        kind,
182                        clustering,
183                        end_deletion_time: *end_dt,
184                        end_local_deletion_time: *end_ldt,
185                        start_deletion_time: *start_dt,
186                        start_local_deletion_time: *start_ldt,
187                    };
188                    // The open marker's fields are already copied into `boundary`;
189                    // drop it from the stream.
190                    iter.next();
191                    out.push(boundary);
192                    continue;
193                }
194            }
195        }
196        out.push(item);
197    }
198    out
199}

201impl DataWriter {
202    /// Write a complete partition (partition key + all rows)
203    ///
204    /// # Arguments
205    /// * `key` - Decorated partition key (token + raw bytes)
206    /// * `mutations` - All mutations for this partition (must be in clustering order)
207    /// * `schema` - Table schema for column metadata
208    /// * `partition_tombstone` - Optional partition-level tombstone
209    /// * `range_tombstones` - Range tombstones for this partition (must be in clustering order)
210    ///
211    /// # Returns
212    /// File offset where this partition starts (for Index.db)
213    pub fn write_partition(
214        &mut self,
215        key: &DecoratedKey,
216        mutations: &[Mutation],
217        schema: &TableSchema,
218        partition_tombstone: Option<&PartitionTombstone>,
219        range_tombstones: &[RangeTombstone],
220    ) -> Result<u64> {
221        // File offset of this partition = bytes already flushed (`position`) plus
222        // whatever is currently buffered. In streaming mode `buffer` is empty at
223        // the start of each partition (flushed + cleared by the previous call),
224        // so this is `position`; in in-memory mode `position` is 0 and `buffer`
225        // holds all prior partitions, matching the legacy `buffer.len()`.
226        let partition_offset = self.position + self.buffer.len() as u64;
227
228        // Write partition header (with optional tombstone)
229        let header_start = self.buffer.len();
230        self.write_partition_header(key, partition_tombstone)?;
231        let mut prev_unfiltered_size = (self.buffer.len() - header_start) as u64;
232
233        // SSTables must be internally reconciled: Cassandra's read path and
234        // compaction only reconcile rows against deletions from OTHER sources
235        // (memtables / other sstables) — a row shadowed by a partition or
236        // range tombstone in the SAME sstable is served live. Cassandra's own
237        // flush drops shadowed data, and so must we (Issue #716/#717).
238        // `partition_floor` is the shadow timestamp from the partition
239        // tombstone; per-row floors additionally account for covering range
240        // tombstones.
241        let partition_floor = partition_tombstone.map(|pt| pt.deletion_time);
242
243        // Cassandra's SerializationHeader.hasStatic() returns true whenever the schema
244        // declares any static column — and both the writer and reader unconditionally
245        // emit/consume a static-row prelude in that case.  We must do the same.
246        let schema_has_static = schema.columns.iter().any(|c| c.is_static);
247
248        if schema_has_static {
249            // Collect static-column operations from ALL mutations in this partition,
250            // regardless of whether the mutation also carries a clustering key.
251            // Last-write-wins by timestamp_micros when the same column appears twice.
252            // Static cells shadowed by the partition tombstone are dropped.
253            let merged = collect_static_operations(mutations, schema, partition_floor);
254
255            if merged.is_empty() {
256                // Schema declares statics but this partition writes none.
257                // Cassandra still expects the prelude; emit the minimal empty form.
258                //
259                // Issue #821 (finding #2): a static row hard-codes
260                // previousUnfilteredSize = 0 and does NOT become the "previous
261                // unfiltered" for the chain. The running value (the partition
262                // header size) is carried forward by ADDING the static row's
263                // bytes, so the first regular row sees its own offset from the
264                // partition start (verified against real Cassandra "nb"
265                // SSTables: header + static_row_size).
266                let static_size = self.write_empty_static_row(0, schema)? as u64;
267                prev_unfiltered_size += static_size;
268            } else {
269                // Issue #1018 (roborev HIGH): derive the static row's liveness
270                // timestamp + TTL from the SURVIVING merged ops (each carrying its
271                // own per-cell writetime after the per-cell shadow floor), NOT from
272                // the mutations that merely cleared the floor on their row max — a
273                // static write whose per-cell ts is `<= partition_floor` is already
274                // dropped from `merged`, so it cannot leak its writetime here.
275                // `!merged.is_empty()` guarantees `Some`.
276                let (latest_ts, ttl) = static_liveness_from_ops(&merged).unwrap_or((
277                    mutations.first().map(|m| m.timestamp_micros).unwrap_or(0),
278                    None,
279                ));
280
281                // Issue #764: pass the per-op merged static ops (each carrying its
282                // own originating timestamp + local_deletion_time) so a surviving
283                // older static delete keeps its own LDT instead of inheriting the
284                // newest static mutation's value.
285                //
286                // Issue #821 (finding #2): hard-code prev_size = 0 for the static
287                // row and carry the running chain value forward by adding the
288                // static row's serialized bytes (see comment above).
289                let (static_size, _static_cells) =
290                    self.write_static_row_with_prev_size(&merged, latest_ts, ttl, schema, 0)?;
291                prev_unfiltered_size += static_size as u64;
292            }
293        }
294
295        // Merge all mutations sharing a clustering key into a single row each
296        // (Issue #716/#717: writing one row per mutation produced duplicate
297        // rows with equal clustering — e.g. an INSERT row plus a phantom
298        // tombstone-carrier row — which is invalid in the OA format). Rows
299        // shadowed by the partition tombstone or a covering range tombstone
300        // are dropped during the merge.
301        let rows = self.merge_clustering_rows(
302            mutations,
303            schema,
304            schema_has_static,
305            partition_floor,
306            range_tombstones,
307        );
308
309        // Interleave merged rows with range tombstone bound markers in
310        // clustering order. Cassandra requires every unfiltered (row or
311        // marker) of a partition to appear in clustering order; with equal
312        // clustering values, inclusive-start/exclusive-end bounds sort before
313        // the row and inclusive-end/exclusive-start bounds sort after it
314        // (ClusteringPrefix.Kind.comparedToClustering).
315        let mut items: Vec<PartitionItem> = rows.into_iter().map(PartitionItem::Row).collect();
316        for rt in range_tombstones {
317            items.push(PartitionItem::Marker {
318                bound: &rt.start,
319                is_open: true,
320                deletion_time: rt.deletion_time,
321                local_deletion_time: rt.local_deletion_time,
322            });
323            items.push(PartitionItem::Marker {
324                bound: &rt.end,
325                is_open: false,
326                deletion_time: rt.deletion_time,
327                local_deletion_time: rt.local_deletion_time,
328            });
329        }
330
331        sort_partition_items(&mut items, schema);
332        // Issue #1220: coalesce each adjacent close+open bound pair sharing a
333        // boundary point into a single two-deletion-time BOUNDARY marker, exactly
334        // as Cassandra emits them.
335        let items = coalesce_boundaries(items, schema);
336
337        for item in items {
338            prev_unfiltered_size = match item {
339                PartitionItem::Row(row) => {
340                    let (bytes, _cells) =
341                        self.write_merged_row_with_prev_size(&row, schema, prev_unfiltered_size)?;
342                    bytes as u64
343                }
344                PartitionItem::Marker {
345                    bound,
346                    is_open,
347                    deletion_time,
348                    local_deletion_time,
349                } => self.write_range_bound(
350                    bound,
351                    is_open,
352                    deletion_time,
353                    local_deletion_time,
354                    schema,
355                    prev_unfiltered_size,
356                )? as u64,
357                PartitionItem::Boundary {
358                    kind,
359                    clustering,
360                    end_deletion_time,
361                    end_local_deletion_time,
362                    start_deletion_time,
363                    start_local_deletion_time,
364                } => self.write_range_boundary(
365                    kind,
366                    clustering,
367                    end_deletion_time,
368                    end_local_deletion_time,
369                    start_deletion_time,
370                    start_local_deletion_time,
371                    schema,
372                    prev_unfiltered_size,
373                )? as u64,
374            };
375        }
376
377        // Write end-of-partition marker
378        self.buffer.push(END_OF_PARTITION);
379
380        // Streaming mode: flush this partition to disk and clear the scratch so
381        // only one partition is ever resident in memory. No-op in memory mode.
382        self.flush_partition()?;
383
384        Ok(partition_offset)
385    }
386
387    /// Write a complete partition and collect promoted index blocks for wide partitions.
388    ///
389    /// Same as [`write_partition`] but also tracks `IndexInfo` block boundaries
390    /// for partitions whose uncompressed data exceeds `COLUMN_INDEX_SIZE_BYTES` (64 KiB).
391    /// The returned `Vec<PromotedIndexBlock>` contains **one entry per 64 KiB block**;
392    /// the caller (SSTableWriter) passes this to `IndexWriter::add_partition_with_promoted`.
393    ///
394    /// Block sampling rules (mirrors `BigFormatPartitionWriter`):
395    /// - A new block is opened whenever the bytes written since the last boundary ≥ 64 KiB.
396    /// - The last open block is closed when `END_OF_PARTITION` is reached.
397    /// - Only rows with clustering keys contribute `firstName`/`lastName`; tables without
398    ///   clustering keys produce blocks with empty prefix bytes (`[0x00]` = header-only VInt).
399    /// - A promoted index is only meaningful when 2+ blocks result; the caller checks this.
400    pub fn write_partition_with_index_blocks(
401        &mut self,
402        key: &DecoratedKey,
403        mutations: &[Mutation],
404        schema: &TableSchema,
405        partition_tombstone: Option<&PartitionTombstone>,
406        range_tombstones: &[RangeTombstone],
407    ) -> Result<(u64, Vec<PromotedIndexBlock>, PartitionEmitCounts)> {
408        // File offset of this partition
409        let partition_offset = self.position + self.buffer.len() as u64;
410
411        // Issue #851: tally the rows/cells actually emitted below so Statistics
412        // is fed from the single source of truth (this emitter) instead of a
413        // parallel re-derivation that kept diverging from Data.db.
414        let mut emit = PartitionEmitCounts::default();
415
416        // Note the absolute buffer position at the start of partition data.
417        // For streaming mode this is always `self.position` (buffer is empty at start).
418        // For in-memory mode this is `self.buffer.len()` (position == 0).
419        let partition_buf_start = self.buffer.len();
420
421        // Write partition header (with optional tombstone)
422        let header_start = self.buffer.len();
423        self.write_partition_header(key, partition_tombstone)?;
424        let mut prev_unfiltered_size = (self.buffer.len() - header_start) as u64;
425
426        let partition_floor = partition_tombstone.map(|pt| pt.deletion_time);
427        let schema_has_static = schema.columns.iter().any(|c| c.is_static);
428
429        if schema_has_static {
430            let merged = collect_static_operations(mutations, schema, partition_floor);
431
432            if merged.is_empty() {
433                // Issue #821 (finding #2): static row hard-codes prev_size = 0 and
434                // is skipped by the prev-size chain; carry the running value
435                // forward by adding the static row's bytes (see the non-indexed
436                // path in `write_partition` for the rationale + Cassandra anchor).
437                let static_size = self.write_empty_static_row(0, schema)? as u64;
438                prev_unfiltered_size += static_size;
439            } else {
440                // Issue #1018 (roborev HIGH): derive liveness/TTL from the SURVIVING
441                // merged ops (per-cell writetimes after the shadow floor), not from
442                // mutations that only cleared the floor on their row max. See the
443                // non-indexed path above for the rationale.
444                let (latest_ts, ttl) = static_liveness_from_ops(&merged).unwrap_or((
445                    mutations.first().map(|m| m.timestamp_micros).unwrap_or(0),
446                    None,
447                ));
448
449                // Issue #764: same per-op LDT preservation as the non-indexed path.
450                // Issue #821 (finding #2): prev_size = 0 for the static row; chain
451                // carries forward by adding the static row's serialized bytes.
452                let (static_size, static_cells) =
453                    self.write_static_row_with_prev_size(&merged, latest_ts, ttl, schema, 0)?;
454                prev_unfiltered_size += static_size as u64;
455                // A non-empty static prelude is one physical row. Issue #851
456                // (review): count the static cells the writer ACTUALLY serialized
457                // (returned by the write path), not `merged.len()` — a merged
458                // static null write is skipped by `write_static_cells`, so it
459                // must not be counted as an emitted column.
460                emit.rows += 1;
461                emit.columns += static_cells;
462            }
463        }
464
465        let rows = self.merge_clustering_rows(
466            mutations,
467            schema,
468            schema_has_static,
469            partition_floor,
470            range_tombstones,
471        );
472
473        let mut items: Vec<PartitionItem> = rows.into_iter().map(PartitionItem::Row).collect();
474        for rt in range_tombstones {
475            items.push(PartitionItem::Marker {
476                bound: &rt.start,
477                is_open: true,
478                deletion_time: rt.deletion_time,
479                local_deletion_time: rt.local_deletion_time,
480            });
481            items.push(PartitionItem::Marker {
482                bound: &rt.end,
483                is_open: false,
484                deletion_time: rt.deletion_time,
485                local_deletion_time: rt.local_deletion_time,
486            });
487        }
488
489        sort_partition_items(&mut items, schema);
490        // Issue #1220: coalesce adjacent close+open bound pairs into BOUNDARY
491        // markers so the wide-partition body matches the plain path byte-for-byte.
492        let items = coalesce_boundaries(items, schema);
493
494        // ── Promoted index block tracking ────────────────────────────────
495        // Mirrors Cassandra's BigFormatPartitionWriter:
496        // - Start block at position 0 (immediately after partition header).
497        // - Close block + open a new one whenever accumulated bytes ≥ 64 KiB.
498        // - The last block is closed at END_OF_PARTITION.
499        //
500        // `block_start_buf_offset`: absolute position in `self.buffer` where
501        //   the current block began (relative to `partition_buf_start`).
502        // `current_block_first_ck`: serialized ClusteringPrefix of the first item
503        //   in the current block (empty bytes → "header only" VInt 0x00 for no-CK tables).
504        // `current_block_last_ck`: serialized ClusteringPrefix of the most-recent item.
505
506        let mut blocks: Vec<PromotedIndexBlock> = Vec::new();
507        let mut block_start_buf_offset = self.buffer.len() - partition_buf_start;
508        let mut current_block_first_ck: Option<Vec<u8>> = None;
509        let mut current_block_last_ck: Option<Vec<u8>> = None;
510        // OSS50 byte-comparable separator (first unfiltered's clustering) for the
511        // BTI Rows.db row-index trie (issue #910). `None` until the first item.
512        let mut current_block_oss50: Option<Option<Vec<u8>>> = None;
513
514        for item in items {
515            // Promoted-index `firstName`/`lastName` ClusteringPrefix bytes for this
516            // item (Issue #1186). These are serialized in Cassandra's IndexInfo form
517            // (`ClusteringPrefix.serializer.serialize`), which prepends the
518            // `Kind.ordinal()` byte — NOT the values-only Data.db row form.
519            //
520            // The kind byte differs by unfiltered type:
521            // - A ROW clustering name is always kind CLUSTERING (0x04); a single
522            //   `int` clustering yields the Cassandra-exact 6 bytes `04 00 <int>`.
523            // - A range-tombstone MARKER name carries its actual BOUND kind ordinal
524            //   (INCL_START_BOUND=1 / EXCL_END_BOUND=0 / INCL_END_BOUND=6 /
525            //   EXCL_START_BOUND=7), computed identically to `write_range_bound`,
526            //   NOT 0x04 (roborev MEDIUM). The fallbacks produce an empty values
527            //   header of the correct kind.
528            let ck_bytes: Vec<u8> = match &item {
529                PartitionItem::Row(row) => {
530                    if let Some(ck) = row.clustering_key {
531                        serialize_clustering_prefix_for_index(ck, schema)
532                            .unwrap_or_else(|_| empty_clustering_prefix_for_index())
533                    } else {
534                        empty_clustering_prefix_for_index() // no clustering key
535                    }
536                }
537                PartitionItem::Marker { bound, is_open, .. } => {
538                    serialize_marker_bound_prefix_for_index(bound, *is_open, schema)
539                        .unwrap_or_else(|_| marker_bound_prefix_for_index(bound, *is_open))
540                }
541                PartitionItem::Boundary {
542                    kind, clustering, ..
543                } => serialize_boundary_prefix_for_index(*kind, clustering, schema)
544                    .unwrap_or_else(|_| boundary_prefix_for_index_fallback(*kind)),
545            };
546
547            if current_block_first_ck.is_none() {
548                current_block_first_ck = Some(ck_bytes.clone());
549                // OSS50 byte-comparable separator from the first unfiltered's
550                // clustering values (issue #910). Markers and no-CK rows yield
551                // None (no row-index separator). Encoding errors degrade to None
552                // — the partition then keeps a direct Data.db offset rather than
553                // emitting a separator the reader cannot reconstruct.
554                let ck_values: Option<Vec<Value>> = match &item {
555                    PartitionItem::Row(row) => row
556                        .clustering_key
557                        .filter(|ck| !ck.columns.is_empty())
558                        .map(|ck| ck.columns.iter().map(|(_, v)| v.clone()).collect()),
559                    // A marker / boundary yields no row-index separator: the
560                    // promoted-index row trie keys off concrete row clusterings,
561                    // not tombstone bound names.
562                    PartitionItem::Marker { .. } | PartitionItem::Boundary { .. } => None,
563                };
564                // Per-column clustering ORDER (ASC/DESC). For a DESC column the
565                // OSS50 separator bytes must be the REVERSED byte-comparable form
566                // (Cassandra `ReversedType`/`ByteSource.invert`: complement every
567                // byte) so that descending value order maps to ascending byte
568                // order — otherwise the strict-ascending separator check rejects
569                // the trie and the wide partition silently falls back to a direct
570                // Data.db offset (roborev MEDIUM, issue #910 follow-up).
571                let is_reversed: Vec<bool> = schema
572                    .clustering_keys
573                    .iter()
574                    .map(|c| c.order == crate::schema::ClusteringOrder::Desc)
575                    .collect();
576                current_block_oss50 = Some(ck_values.and_then(|vals| {
577                    crate::storage::sstable::bti::encode_clustering_bound_oss50_with_order(
578                        &vals,
579                        &is_reversed,
580                    )
581                    .ok()
582                }));
583            }
584            current_block_last_ck = Some(ck_bytes.clone());
585
586            // Write the item
587            prev_unfiltered_size = match item {
588                PartitionItem::Row(row) => {
589                    // One physical row. Issue #851 (review): count the cells the
590                    // writer ACTUALLY serialized (returned by the write path),
591                    // not `row.ops.len()` — `write_merged_cells` skips null-valued
592                    // writes, so a row whose only write is null contributes a live
593                    // row with zero columns, matching Data.db exactly.
594                    emit.rows += 1;
595                    let (bytes, cells) =
596                        self.write_merged_row_with_prev_size(&row, schema, prev_unfiltered_size)?;
597                    emit.columns += cells;
598                    bytes as u64
599                }
600                PartitionItem::Marker {
601                    bound,
602                    is_open,
603                    deletion_time,
604                    local_deletion_time,
605                } => self.write_range_bound(
606                    bound,
607                    is_open,
608                    deletion_time,
609                    local_deletion_time,
610                    schema,
611                    prev_unfiltered_size,
612                )? as u64,
613                PartitionItem::Boundary {
614                    kind,
615                    clustering,
616                    end_deletion_time,
617                    end_local_deletion_time,
618                    start_deletion_time,
619                    start_local_deletion_time,
620                } => self.write_range_boundary(
621                    kind,
622                    clustering,
623                    end_deletion_time,
624                    end_local_deletion_time,
625                    start_deletion_time,
626                    start_local_deletion_time,
627                    schema,
628                    prev_unfiltered_size,
629                )? as u64,
630            };
631
632            // Bytes written in the current block so far
633            let current_buf_offset = self.buffer.len() - partition_buf_start;
634            let block_bytes = (current_buf_offset - block_start_buf_offset) as u64;
635
636            if block_bytes >= COLUMN_INDEX_SIZE_BYTES {
637                // Close this block and start a new one
638                blocks.push(PromotedIndexBlock {
639                    first_name: current_block_first_ck
640                        .take()
641                        .unwrap_or_else(empty_clustering_prefix_for_index),
642                    last_name: current_block_last_ck
643                        .take()
644                        .unwrap_or_else(empty_clustering_prefix_for_index),
645                    offset: block_start_buf_offset as u64,
646                    width: block_bytes,
647                    oss50_separator: current_block_oss50.take().flatten(),
648                });
649                block_start_buf_offset = current_buf_offset;
650                // first/last reset for next block
651                current_block_first_ck = None;
652                current_block_last_ck = None;
653                current_block_oss50 = None;
654            }
655        }
656
657        // Write end-of-partition marker
658        self.buffer.push(END_OF_PARTITION);
659
660        // Close the final block (if any items were written)
661        if let (Some(first), Some(last)) = (current_block_first_ck, current_block_last_ck) {
662            let current_buf_offset = self.buffer.len() - partition_buf_start;
663            let block_bytes = (current_buf_offset - block_start_buf_offset) as u64;
664            if block_bytes > 0 {
665                blocks.push(PromotedIndexBlock {
666                    first_name: first,
667                    last_name: last,
668                    offset: block_start_buf_offset as u64,
669                    width: block_bytes,
670                    oss50_separator: current_block_oss50.take().flatten(),
671                });
672            }
673        }
674
675        self.flush_partition()?;
676
677        Ok((partition_offset, blocks, emit))
678    }

680    /// Write an empty static-row prelude.
681    ///
682    /// Required by Cassandra whenever the schema has any static column, even
683    /// when this particular partition writes no static cells.
684    ///
685    /// Binary form:
686    /// ```text
687    /// [0x80]              ← row_flags = ROW_HAS_EXTENDED_FLAGS only
688    /// [0x01]              ← extended_flags = EXTENDED_IS_STATIC
689    /// [row_size: VUInt]   ← size of (prev_size VInt + bitmap)
690    /// [prev_size: VUInt]
691    /// [bitmap: VUInt]     ← all-missing bitmap: (1 << N) - 1 for N static cols
692    ///                       (encoded via write_column_subset with empty present set)
693    /// ```
694    pub(super) fn write_empty_static_row(
695        &mut self,
696        prev_size: u64,
697        schema: &TableSchema,
698    ) -> Result<usize> {
699        let start_len = self.buffer.len();
700
701        // flags = only HAS_EXTENDED_FLAGS; no timestamp, no TTL, no deletion,
702        // no HAS_ALL_COLUMNS, no HAS_COMPLEX_DELETION.
703        let flags: u8 = ROW_HAS_EXTENDED_FLAGS;
704        self.buffer.push(flags);
705        self.buffer.push(EXTENDED_IS_STATIC);
706
707        // Build the row body: just prev_size VInt + column bitmap (all missing).
708        let mut body = Vec::new();
709
710        // Column bitmap: "all columns missing" for every static column.
711        // write_column_subset with an empty present_set.
712        let static_columns = self.static_columns(schema);
713        let empty_present: std::collections::HashSet<&str> = std::collections::HashSet::new();
714        self.write_column_subset(&mut body, &static_columns, &empty_present)?;
715
716        let prev_size_vint_len = unsigned_len(prev_size);
717        let row_body_size = prev_size_vint_len as u64 + body.len() as u64;
718
719        let mut row_size_buf = Vec::new();
720        encode_unsigned(row_body_size, &mut row_size_buf);
721        self.buffer.extend_from_slice(&row_size_buf);
722
723        encode_unsigned(prev_size, &mut self.buffer);
724        self.buffer.extend_from_slice(&body);
725
726        Ok(self.buffer.len() - start_len)
727    }

729    /// Write partition header
730    ///
731    /// Format (V5CompressedLegacy / Cassandra BigFormat):
732    /// ```text
733    /// [key_length: u16 BE]           ← Partition key length (2-byte unsigned short)
734    /// [key_bytes]                    ← Raw partition key bytes
735    /// [local_deletion_time: i32 BE]  ← i32::MAX for LIVE (DeletionTime.LIVE)
736    /// [deletion_timestamp: i64 BE]   ← i64::MIN for LIVE (DeletionTime.LIVE)
737    /// ```
738    ///
739    /// Note: Cassandra uses `ByteBufferUtil.writeWithShortLength()` for the key,
740    /// which is a 2-byte BE unsigned short. There is NO separate flags byte.
741    /// DeletionTime.LIVE uses sentinel values (Integer.MAX_VALUE, Long.MIN_VALUE).
742    pub(super) fn write_partition_header(
743        &mut self,
744        key: &DecoratedKey,
745        tombstone: Option<&PartitionTombstone>,
746    ) -> Result<()> {
747        // Partition key length (u16 BE, matching Cassandra's writeWithShortLength)
748        if key.key.len() > 65535 {
749            return Err(Error::InvalidInput(format!(
750                "Partition key too large: {} bytes (max 65535)",
751                key.key.len()
752            )));
753        }
754        self.buffer
755            .write_all(&(key.key.len() as u16).to_be_bytes())?;
756
757        // Partition key bytes
758        self.buffer.extend_from_slice(&key.key);
759
760        // Partition deletion info.
761        //
762        // Issue #1741: the on-disk `DeletionTime` layout is format-specific. `da`
763        // (BTI, oa `hasUIntDeletionTime`) uses `DeletionTime.Serializer`:
764        //   LIVE    = 1 byte `0x80` (IS_LIVE_DELETION);
765        //   DELETED = markedForDeleteAt (i64 BE) + localDeletionTime (u32 BE) = 12 bytes.
766        // The legacy na/`nb` layout (`legacySerializer`) is always 12 bytes:
767        //   localDeletionTime (i32 BE) + markedForDeleteAt (i64 BE), LIVE encoded as
768        //   (Integer.MAX_VALUE, Long.MIN_VALUE). Writing the `nb` layout into a `da`
769        //   file makes the reader's oa branch misread the LIVE sentinel as a
770        //   tombstone (its first byte `0x7F` != `0x80`), which the read-side
771        //   shadowing then treats as a partition delete — so `da` MUST use the oa
772        //   layout. Authority: DeletionTime.java (Serializer / legacySerializer),
773        //   BigFormat.java:409 (`hasUIntDeletionTime`).
774        match (self.oa_partition_deletion, tombstone) {
775            (true, Some(ts)) => {
776                // oa DELETED: markedForDeleteAt (i64) then localDeletionTime (u32).
777                self.buffer.write_all(&ts.deletion_time.to_be_bytes())?;
778                self.buffer
779                    .write_all(&(ts.local_deletion_time as u32).to_be_bytes())?;
780            }
781            (true, None) => {
782                // oa LIVE: single IS_LIVE_DELETION byte.
783                self.buffer.write_all(&[0x80])?;
784            }
785            (false, Some(ts)) => {
786                // nb DELETED: localDeletionTime (i32) then markedForDeleteAt (i64).
787                self.buffer
788                    .write_all(&ts.local_deletion_time.to_be_bytes())?;
789                self.buffer.write_all(&ts.deletion_time.to_be_bytes())?;
790            }
791            (false, None) => {
792                // nb LIVE: DeletionTime.LIVE sentinel (Integer.MAX_VALUE, Long.MIN_VALUE).
793                self.buffer.write_all(&i32::MAX.to_be_bytes())?;
794                self.buffer.write_all(&i64::MIN.to_be_bytes())?;
795            }
796        }
797
798        Ok(())
799    }
