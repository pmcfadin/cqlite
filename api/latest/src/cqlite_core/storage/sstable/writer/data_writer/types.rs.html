9/// A surviving cell operation in a merged row, tagged with the timestamp and
10/// row-level TTL of the mutation it came from.
11///
12/// Epic #899 (Phase B): for whole-column ops (`Write`/`WriteWithTtl`/`Delete`)
13/// `timestamp_micros` is the originating mutation's row timestamp. For the
14/// per-element complex ops (`WriteComplexElement`/`ComplexDeletion`) the
15/// element's OWN timestamp/ttl/ldt/cell_path live INSIDE the op itself; the
16/// `timestamp_micros` field still carries the originating mutation's row
17/// timestamp so the writer can decide `USE_ROW_TIMESTAMP` vs an explicit delta
18/// per element.
19pub(crate) struct MergedOp<'a> {
20    pub(crate) op: &'a crate::storage::write_engine::mutation::CellOperation,
21    pub(crate) timestamp_micros: i64,
22    /// Row-level TTL (`Mutation::ttl_seconds`) of the originating mutation.
23    /// Per-cell TTL lives inside `CellOperation::WriteWithTtl` itself.
24    pub(crate) row_ttl_seconds: Option<u32>,
25    /// Local deletion time (seconds since epoch) for a `Delete` cell tombstone,
26    /// honoring the originating mutation's explicit `local_deletion_time` when
27    /// present (Issue #764). Derived from the timestamp otherwise.
28    pub(crate) cell_local_deletion_time: i32,
29}
30
31/// Adapt a writer-path [`MergedOp`] into the shared
32/// [`ReconcileCell`](crate::storage::write_engine::reconcile_rules::ReconcileCell)
33/// view (issue #947) so the per-column cell tie-break in `merge_row_group` calls
34/// the one shared `Cells#reconcile` rule. A whole-column op is a tombstone iff it
35/// is a `CellOperation::Delete` (a `WriteWithTtl` is an EXPIRING — live — cell,
36/// not a tombstone), matching the merge path's `Value::Tombstone` recognition.
37impl crate::storage::write_engine::reconcile_rules::ReconcileCell for MergedOp<'_> {
38    fn timestamp(&self) -> i64 {
39        self.timestamp_micros
40    }
41    fn is_tombstone(&self) -> bool {
42        matches!(
43            self.op,
44            crate::storage::write_engine::mutation::CellOperation::Delete { .. }
45        )
46    }
47}

49/// One element to emit inside a per-element complex column (epic #899, Phase B).
50///
51/// Carries the element's OWN write metadata and its PRESERVED source cell path
52/// (never regenerated). `value == None` with `is_deleted` true is an
53/// element-level tombstone; `value == None` without `is_deleted` is an
54/// empty-value element (e.g. a SET member). The writer stamps each element with
55/// `USE_ROW_TIMESTAMP` only when `timestamp_micros` equals the row timestamp,
56/// otherwise it clears the flag and writes an explicit unsigned delta.
57#[derive(Debug, Clone)]
58pub(crate) struct ComplexElementWrite {
59    pub(crate) cell_path: Vec<u8>,
60    pub(crate) value: Option<Value>,
61    pub(crate) timestamp_micros: i64,
62    pub(crate) ttl_seconds: Option<u32>,
63    pub(crate) local_deletion_time: Option<i32>,
64    pub(crate) is_deleted: bool,
65}

67/// One complex column's reconciled contents while grouping per-element ops:
68/// `(optional complex deletion (markedForDeleteAt µs, localDeletionTime s),
69/// surviving elements)` (epic #899, Phase B).
70pub(crate) type ComplexColumnGroup = (Option<(i64, i32)>, Vec<ComplexElementWrite>);

72/// A surviving static-column operation, tagged with the timestamp and explicit
73/// local deletion time of the mutation it came from.
74///
75/// Issue #764: static-column tombstones must be stamped with their ORIGINATING
76/// mutation's `local_deletion_time` (and timestamp), not a single synthetic
77/// value taken from the newest static-contributing mutation. A surviving delete
78/// from an older mutation otherwise inherits the wrong LDT — corrupting the
79/// unsigned-VInt delta when stats were seeded from that older delete's explicit
80/// (lower) LDT.
81pub(crate) struct StaticMergedOp {
82    pub(crate) op: crate::storage::write_engine::mutation::CellOperation,
83    /// Timestamp (µs) of the originating mutation.
84    pub(crate) timestamp_micros: i64,
85    /// Local deletion time (s) for a `Delete` tombstone, honoring the
86    /// originating mutation's explicit `local_deletion_time` when set.
87    pub(crate) cell_local_deletion_time: i32,
88    /// Statement-level TTL (`Mutation::ttl_seconds`) of the originating mutation
89    /// (issue #1196 regression fix). Cassandra encodes a static `USING TTL`
90    /// write as an EXPIRING CELL (per-cell TTL), never as row-level liveness on
91    /// the static block. A plain `Write` carrying this TTL must therefore be
92    /// emitted via `write_cell_with_ttl`, not as a non-expiring cell; a per-cell
93    /// `WriteWithTtl` keeps its own TTL (which takes precedence, matching the
94    /// regular-row path where per-cell TTL is honored verbatim).
95    pub(crate) row_ttl_seconds: Option<u32>,
96}

98/// The exact rows and cells `DataWriter` emitted to Data.db for one partition.
99///
100/// Issue #851: Statistics' `totalRows` (`row_count`) and `totalColumnsSet`
101/// (`column_count`) MUST equal what is physically written. Rather than
102/// re-deriving the counts from the raw mutations in a parallel loop (which kept
103/// diverging from the emitter — rejected commit `5afce78c`), the emission code
104/// is the single source of truth: it tallies a row whenever it writes a row
105/// (static prelude or merged clustering row) and tallies cells from the same
106/// reconciled `ops` it serializes. The empty static-row prelude and range
107/// tombstone markers write no `Row`, so they contribute nothing — matching
108/// Cassandra `Row.isEmpty()` / `Row.columnCount()`.
109#[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
110pub struct PartitionEmitCounts {
111    /// Rows physically written to Data.db (static prelude + merged clustering
112    /// rows). Excludes the empty static prelude and range tombstone markers.
113    pub rows: u64,
114    /// Regular + static cells physically written. Primary-key (partition +
115    /// clustering) columns are encoded positionally and never counted; row
116    /// tombstones (`DeleteRow`) set no columns.
117    pub columns: u64,
118}

120/// One Data.db row assembled by merging every mutation of a partition that
121/// shares the same clustering key (Issues #716/#717: a partition must never
122/// contain two rows with equal clustering).
123pub(crate) struct RowWrite<'a> {
124    pub(crate) clustering_key: Option<&'a crate::storage::write_engine::mutation::ClusteringKey>,
125    /// Primary-key liveness timestamp. `None` for pure row tombstones —
126    /// Cassandra serializes those without HAS_TIMESTAMP.
127    pub(crate) liveness_ts: Option<i64>,
128    /// Row-level TTL from the liveness-providing mutation.
129    pub(crate) ttl_seconds: Option<u32>,
130    /// Row deletion as (marked_for_delete_at µs, local_deletion_time s).
131    pub(crate) row_deletion: Option<(i64, i32)>,
132    /// Surviving WHOLE-COLUMN cell operations (already reconciled, unsorted).
133    /// `Write`/`WriteWithTtl`/`Delete` — one per column (last-write-wins).
134    pub(crate) ops: Vec<MergedOp<'a>>,
135    /// Surviving PER-ELEMENT complex ops (epic #899, Phase B):
136    /// `WriteComplexElement` + `ComplexDeletion`. These are NOT deduped per
137    /// column (a column has many elements) and are written via
138    /// `write_complex_column_per_element`. Empty for every existing scenario —
139    /// the real pipeline (`merge_entry_to_mutation`) does not yet emit these
140    /// ops, so this stays empty until Phase C (keeping output byte-neutral).
141    pub(crate) complex_element_ops: Vec<MergedOp<'a>>,
142}
