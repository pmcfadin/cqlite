9/// A surviving cell operation in a merged row, tagged with the timestamp and
10/// row-level TTL of the mutation it came from.
11///
12/// Epic #899 (Phase B): for whole-column ops (`Write`/`WriteWithTtl`/`Delete`)
13/// `timestamp_micros` is the originating mutation's row timestamp. For the
14/// per-element complex ops (`WriteComplexElement`/`ComplexDeletion`) the
15/// element's OWN timestamp/ttl/ldt/cell_path live INSIDE the op itself; the
16/// `timestamp_micros` field still carries the originating mutation's row
17/// timestamp so the writer can decide `USE_ROW_TIMESTAMP` vs an explicit delta
18/// per element.
19pub(crate) struct MergedOp<'a> {
20    pub(crate) op: &'a crate::storage::write_engine::mutation::CellOperation,
21    pub(crate) timestamp_micros: i64,
22    /// Row-level TTL (`Mutation::ttl_seconds`) of the originating mutation.
23    /// Per-cell TTL lives inside `CellOperation::WriteWithTtl` itself.
24    pub(crate) row_ttl_seconds: Option<u32>,
25    /// Local deletion time (seconds since epoch) for a `Delete` cell tombstone,
26    /// honoring the originating mutation's explicit `local_deletion_time` when
27    /// present (Issue #764). Derived from the timestamp otherwise.
28    pub(crate) cell_local_deletion_time: i32,
29    /// The ONE wall-clock reading (GC-clock seconds) captured when this row group was merged
30    /// (#4530, roborev). An expiring cell WITHOUT an explicit LDT gets
31    /// `implicit_expiration(now_seconds, ttl)` - the SAME value is used to rank it in
32    /// `cell_wins` and to stamp it on disk (the row's `RowWrite::now_seconds` carries this
33    /// reading to the emission), so the reconcile winner is exactly the cell that is written.
34    pub(crate) now_seconds: i32,
35}
36
37/// The wall clock, in GC-clock seconds (`FBUtilities.nowInSeconds()`); `0` before the epoch.
38pub(crate) fn wall_clock_seconds() -> i32 {
39    std::time::SystemTime::now()
40        .duration_since(std::time::UNIX_EPOCH)
41        .map(|d| i32::try_from(d.as_secs()).unwrap_or(i32::MAX))
42        .unwrap_or(0)
43}
44
45/// Expiration instant of an expiring write that carries no explicit LDT: `now + ttl`, CAPPED at
46/// the maximum deletion time. Cassandra 5.0.8 applies the same cap by default
47/// (`ExpirationDateOverflowHandling.ExpirationDateOverflowPolicy.CAP`; the cap is
48/// `Cell.MAX_DELETION_TIME`, `MAX_DELETION_TIME_2038_LEGACY_CAP` = `i32::MAX` in 4.x-compatible
49/// mode), so ranking and emission saturate identically instead of wrapping.
50pub(crate) fn implicit_expiration(now_seconds: i32, ttl_seconds: u32) -> i32 {
51    now_seconds.saturating_add(i32::try_from(ttl_seconds).unwrap_or(i32::MAX))
52}

54/// Adapt a writer-path [`MergedOp`] into the shared
55/// [`ReconcileCell`](crate::storage::write_engine::reconcile_rules::ReconcileCell)
56/// view (issue #947) so the per-column cell tie-break in `merge_row_group` calls
57/// the one shared `Cells#reconcile` rule. A whole-column op is a tombstone iff it
58/// is a `CellOperation::Delete` (a `WriteWithTtl` is an EXPIRING — live — cell,
59/// not a tombstone), matching the merge path's `Value::Tombstone` recognition.
60impl crate::storage::write_engine::reconcile_rules::ReconcileCell for MergedOp<'_> {
61    fn timestamp(&self) -> i64 {
62        self.timestamp_micros
63    }
64    fn is_tombstone(&self) -> bool {
65        matches!(
66            self.op,
67            crate::storage::write_engine::mutation::CellOperation::Delete { .. }
68        )
69    }
70    fn expiring_ttl(&self) -> Option<u32> {
71        use crate::storage::write_engine::mutation::CellOperation as Op;
72        // TTL 0 is Cassandra's `NO_TTL` (`LivenessInfo.NO_TTL`): not an expiring cell.
73        let ttl = match self.op {
74            Op::WriteWithTtl { ttl_seconds, .. } => Some(*ttl_seconds),
75            Op::Write { .. } => self.row_ttl_seconds,
76            _ => None,
77        };
78        ttl.filter(|t| *t != 0)
79    }
80    fn local_deletion_time(&self) -> i32 {
81        use crate::storage::write_engine::mutation::CellOperation as Op;
82        match self.op {
83            Op::Delete { .. } => self.cell_local_deletion_time,
84            // The authoritative per-cell LDT when surfaced; otherwise `now + ttl` from the
85            // clock reading captured once for this row group - the value emission stamps.
86            Op::WriteWithTtl {
87                local_deletion_time: Some(l),
88                ..
89            } => *l,
90            _ => self
91                .expiring_ttl()
92                .map(|t| implicit_expiration(self.now_seconds, t))
93                .unwrap_or(0),
94        }
95    }
96    fn value_bytes(&self) -> Vec<u8> {
97        use crate::storage::write_engine::mutation::CellOperation as Op;
98        let mut out = Vec::new();
99        if let Op::Write { value, .. } | Op::WriteWithTtl { value, .. } = self.op {
100            let _ = super::encoding::serialize_value_into(value, &mut out);
101        }
102        out
103    }
104}

106/// One element to emit inside a per-element complex column (epic #899, Phase B).
107///
108/// Carries the element's OWN write metadata and its PRESERVED source cell path
109/// (never regenerated). `value == None` with `is_deleted` true is an
110/// element-level tombstone; `value == None` without `is_deleted` is an
111/// empty-value element (e.g. a SET member). The writer stamps each element with
112/// `USE_ROW_TIMESTAMP` only when `timestamp_micros` equals the row timestamp,
113/// otherwise it clears the flag and writes an explicit unsigned delta.
114#[derive(Debug, Clone)]
115pub(crate) struct ComplexElementWrite {
116    pub(crate) cell_path: Vec<u8>,
117    pub(crate) value: Option<Value>,
118    pub(crate) timestamp_micros: i64,
119    pub(crate) ttl_seconds: Option<u32>,
120    pub(crate) local_deletion_time: Option<i32>,
121    pub(crate) is_deleted: bool,
122}
123
124/// One complex column's reconciled contents while grouping per-element ops:
125/// `(optional complex deletion (markedForDeleteAt µs, localDeletionTime s),
126/// surviving elements)` (epic #899, Phase B).
127pub(crate) type ComplexColumnGroup = (Option<(i64, i32)>, Vec<ComplexElementWrite>);

129/// A surviving static-column operation, tagged with the timestamp and explicit
130/// local deletion time of the mutation it came from.
131///
132/// Issue #764: static-column tombstones must be stamped with their ORIGINATING
133/// mutation's `local_deletion_time` (and timestamp), not a single synthetic
134/// value taken from the newest static-contributing mutation. A surviving delete
135/// from an older mutation otherwise inherits the wrong LDT — corrupting the
136/// unsigned-VInt delta when stats were seeded from that older delete's explicit
137/// (lower) LDT.
138pub(crate) struct StaticMergedOp {
139    pub(crate) op: crate::storage::write_engine::mutation::CellOperation,
140    /// Timestamp (µs) of the originating mutation.
141    pub(crate) timestamp_micros: i64,
142    /// Local deletion time (s) for a `Delete` tombstone, honoring the
143    /// originating mutation's explicit `local_deletion_time` when set.
144    pub(crate) cell_local_deletion_time: i32,
145    /// Statement-level TTL (`Mutation::ttl_seconds`) of the originating mutation
146    /// (issue #1196 regression fix). Cassandra encodes a static `USING TTL`
147    /// write as an EXPIRING CELL (per-cell TTL), never as row-level liveness on
148    /// the static block. A plain `Write` carrying this TTL must therefore be
149    /// emitted via `write_cell_with_ttl`, not as a non-expiring cell; a per-cell
150    /// `WriteWithTtl` keeps its own TTL (which takes precedence, matching the
151    /// regular-row path where per-cell TTL is honored verbatim).
152    pub(crate) row_ttl_seconds: Option<u32>,
153}

155/// The exact rows and cells `DataWriter` emitted to Data.db for one partition.
156///
157/// Issue #851: Statistics' `totalRows` (`row_count`) and `totalColumnsSet`
158/// (`column_count`) MUST equal what is physically written. Rather than
159/// re-deriving the counts from the raw mutations in a parallel loop (which kept
160/// diverging from the emitter — rejected commit `5afce78c`), the emission code
161/// is the single source of truth: it tallies a row whenever it writes a row
162/// (static prelude or merged clustering row) and tallies cells from the same
163/// reconciled `ops` it serializes. The empty static-row prelude and range
164/// tombstone markers write no `Row`, so they contribute nothing — matching
165/// Cassandra `Row.isEmpty()` / `Row.columnCount()`.
166#[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
167pub struct PartitionEmitCounts {
168    /// Rows physically written to Data.db (static prelude + merged clustering
169    /// rows). Excludes the empty static prelude and range tombstone markers.
170    pub rows: u64,
171    /// Regular + static cells physically written. Primary-key (partition +
172    /// clustering) columns are encoded positionally and never counted; row
173    /// tombstones (`DeleteRow`) set no columns.
174    pub columns: u64,
175}

177/// One Data.db row assembled by merging every mutation of a partition that
178/// shares the same clustering key (Issues #716/#717: a partition must never
179/// contain two rows with equal clustering).
180pub(crate) struct RowWrite<'a> {
181    pub(crate) clustering_key: Option<&'a crate::storage::write_engine::mutation::ClusteringKey>,
182    /// Primary-key liveness timestamp. `None` for pure row tombstones —
183    /// Cassandra serializes those without HAS_TIMESTAMP.
184    pub(crate) liveness_ts: Option<i64>,
185    /// Row-level TTL from the liveness-providing mutation.
186    pub(crate) ttl_seconds: Option<u32>,
187    /// The liveness marker's authoritative `localExpirationTime` (#4581); `None` derives `now + ttl`.
188    pub(crate) liveness_ldt: Option<i32>,
189    /// Row deletion as (marked_for_delete_at µs, local_deletion_time s).
190    pub(crate) row_deletion: Option<(i64, i32)>,
191    /// Surviving WHOLE-COLUMN cell operations (already reconciled, unsorted).
192    /// `Write`/`WriteWithTtl`/`Delete` — one per column (last-write-wins).
193    pub(crate) ops: Vec<MergedOp<'a>>,
194    /// Surviving PER-ELEMENT complex ops (epic #899, Phase B):
195    /// `WriteComplexElement` + `ComplexDeletion`. These are NOT deduped per
196    /// column (a column has many elements) and are written via
197    /// `write_complex_column_per_element`. Empty for every existing scenario —
198    /// the real pipeline (`merge_entry_to_mutation`) does not yet emit these
199    /// ops, so this stays empty until Phase C (keeping output byte-neutral).
200    pub(crate) complex_element_ops: Vec<MergedOp<'a>>,
201    /// The clock reading `ops` were ranked with (see [`MergedOp::now_seconds`]); emission derives
202    /// every implicit expiration from THIS value. `0` on rows with no expiring cells.
203    pub(crate) now_seconds: i32,
204}

206impl super::DataWriter {
207    /// The compaction path's decision (`MarkerMode::Explicit`, #4581): `None` when the group is
208    /// flush/WAL (liveness is cell-derived), else `Some(marker)` where `marker` may be `None`
209    /// ("no marker": a row that never had one, or whose marker was purged or shadowed).
210    pub(crate) fn authoritative_marker(
211        group: &[&super::Mutation],
212        deletion_ts: Option<i64>,
213    ) -> Option<Option<crate::storage::write_engine::mutation::RowMarker>> {
214        group
215            .iter()
216            .any(|m| m.marker_mode.is_explicit())
217            .then(|| Self::explicit_row_marker(group, deletion_ts))
218    }
219
220    /// The group's authoritative liveness marker (compaction path, #4581): the newest explicit
221    /// [`RowMarker`] strictly newer than `deletion_ts`. A marker at or below the covering
222    /// deletion is shadowed (`DeletionTime.deletes`: `ts <= markedForDeleteAt`).
223    pub(crate) fn explicit_row_marker(
224        group: &[&super::Mutation],
225        deletion_ts: Option<i64>,
226    ) -> Option<crate::storage::write_engine::mutation::RowMarker> {
227        group
228            .iter()
229            .filter_map(|m| m.marker_mode.marker())
230            .filter(|mk| deletion_ts.is_none_or(|dts| mk.timestamp_micros > dts))
231            .max_by_key(|mk| mk.timestamp_micros)
232    }
233
234    /// Write a plain (non-expiring) scalar cell: `USE_ROW_TIMESTAMP` when it carries the row's
235    /// liveness timestamp, an explicit timestamp delta otherwise. Used for ordinary writes AND for
236    /// a `WriteWithTtl { ttl_seconds: 0 }`, because TTL 0 is `NO_TTL` (Cassandra 5.0.8
237    /// `Attributes.getTimeToLive`) and must never be serialized as an expiring cell (#4530).
238    pub(super) fn write_scalar_cell_no_ttl(
239        &self,
240        buf: &mut Vec<u8>,
241        row: &RowWrite<'_>,
242        mop: &MergedOp<'_>,
243        column: &str,
244        value: &crate::types::Value,
245    ) -> crate::Result<()> {
246        if row.liveness_ts == Some(mop.timestamp_micros) {
247            self.write_cell(buf, column, value, mop.timestamp_micros)
248        } else {
249            self.write_cell_explicit_ts(buf, column, value, mop.timestamp_micros)
250        }
251    }
