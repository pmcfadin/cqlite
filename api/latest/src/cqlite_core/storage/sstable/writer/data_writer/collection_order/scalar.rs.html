1//! Scalar leaf comparators for SET-element / MAP-key ordering (issue #1275).
2//!
3//! These mirror the per-type Cassandra `AbstractType.compare` for the SCALAR
4//! element/key types whose order differs from raw unsigned serialized bytes:
5//! signed integers, `Float.compare`/`Double.compare` (Java NaN-last total order),
6//! signed `varint`, scale-aware `decimal`, and `UUIDType`. The composite
7//! comparators (issue #1296) recurse back into the public dispatcher in `mod.rs`,
8//! which bottoms out here at the leaves — these are the SINGLE source of the
9//! signed/float/varint/decimal/uuid order and are never reimplemented elsewhere.

14/// Signed two's-complement big-integer comparison of two `varint` bodies, matching
15/// Cassandra `IntegerType.compare` (issue #1275). Empty bodies are treated as the
16/// most-significant byte being 0 (non-negative). The bodies may differ in length;
17/// sign is taken from the high bit of the most-significant byte.
18pub(super) fn compare_signed_varint(a: &[u8], b: &[u8]) -> Ordering {
19    let neg_a = a.first().is_some_and(|&b0| b0 & 0x80 != 0);
20    let neg_b = b.first().is_some_and(|&b0| b0 & 0x80 != 0);
21    match (neg_a, neg_b) {
22        (false, true) => return Ordering::Greater,
23        (true, false) => return Ordering::Less,
24        _ => {}
25    }
26    // Same sign: left-pad the shorter body with its sign-extension byte (0xFF for
27    // negative, 0x00 for non-negative), then compare lexicographically — for
28    // two's-complement of equal length this IS the signed order.
29    let fill_a: u8 = if neg_a { 0xFF } else { 0x00 };
30    let fill_b: u8 = if neg_b { 0xFF } else { 0x00 };
31    let len = a.len().max(b.len());
32    for i in 0..len {
33        let ea = a
34            .len()
35            .checked_sub(len - i)
36            .and_then(|idx| a.get(idx))
37            .copied()
38            .unwrap_or(fill_a);
39        let eb = b
40            .len()
41            .checked_sub(len - i)
42            .and_then(|idx| b.get(idx))
43            .copied()
44            .unwrap_or(fill_b);
45        match ea.cmp(&eb) {
46            Ordering::Equal => continue,
47            other => return other,
48        }
49    }
50    Ordering::Equal
51}

53/// Cassandra `UUIDType.compareCustom` for two 16-byte UUID values (issue #1275).
54///
55/// Matches `org.apache.cassandra.db.marshal.UUIDType.compareCustom`
56/// (`~/projects/cassandra/.../db/marshal/UUIDType.java`) exactly:
57///   1. Compare the 4-bit version nibble (top nibble of byte 6). If versions
58///      differ, order by `version1 - version2`.
59///   2. If both are version 1, compare the most-significant 64 bits AFTER
60///      `reorderTimestampBytes` (the time-UUID timestamp reshuffle, copied from
61///      `AbstractTimeUUIDType.reorderTimestampBytes`); the reordered value is
62///      `>= 0` so `Long.compare` of the reordered longs is correct.
63///      Otherwise compare the MSB 64 bits as an UNSIGNED long
64///      (`UnsignedLongs.compare`).
65///   3. Tie-break on the least-significant 64 bits as an UNSIGNED long.
66///
67/// WHY `UUIDType` and not `TimeUUIDType`: the writer's `Value::Uuid([u8; 16])`
68/// carries NO uuid-vs-timeuuid discriminator, and this shared comparator only
69/// receives `&Value` — the declared CQL element/key type (`uuid` vs `timeuuid`)
70/// is NOT threaded to this layer (no-heuristics forbids inferring it from the
71/// version nibble: a v1 UUID is legal in a `set<uuid>`). `UUIDType.compareCustom`
72/// is the universal comparator that correctly orders UUIDs of EVERY version
73/// (Cassandra "freely takes time UUIDs" in `UUIDType`), so it is the correct
74/// order for a declared `uuid` element for all versions. For a declared
75/// `timeuuid`, `AbstractTimeUUIDType.compareCustom` agrees on steps 1-2 and
76/// differs ONLY in the v1 LSB tie-break, where `TimeUUIDType` compares the low
77/// 64 bits with SIGNED-per-byte semantics rather than unsigned (the documented
78/// CASSANDRA-8730 discrepancy). That tie-break only changes the relative order of
79/// two UUIDs sharing an identical version-1 timestamp — vanishingly rare and not
80/// representable without the missing type tag — so we apply the `uuid` order
81/// uniformly and document the one-case divergence here rather than guess.
82pub(super) fn compare_uuid_cassandra(a: &[u8; 16], b: &[u8; 16]) -> Ordering {
83    let msb1 = read_be_u64(&a[0..8]);
84    let msb2 = read_be_u64(&b[0..8]);
85
86    let version1 = ((msb1 >> 12) & 0xf) as i32;
87    let version2 = ((msb2 >> 12) & 0xf) as i32;
88    if version1 != version2 {
89        return version1.cmp(&version2);
90    }
91
92    let msb_cmp = if version1 == 1 {
93        // Reordered timestamp longs are non-negative; signed Long.compare ==
94        // their natural numeric order.
95        reorder_timestamp_bytes(msb1).cmp(&reorder_timestamp_bytes(msb2))
96    } else {
97        // UnsignedLongs.compare(msb1, msb2)
98        msb1.cmp(&msb2)
99    };
100    if msb_cmp != Ordering::Equal {
101        return msb_cmp;
102    }
103
104    // UnsignedLongs.compare of the least-significant 64 bits.
105    read_be_u64(&a[8..16]).cmp(&read_be_u64(&b[8..16]))
106}

108/// Big-endian read of an 8-byte slice as a `u64` (mirrors Java `getLong`). The
109/// slice is always exactly 8 bytes at the call sites; a short slice pads with
110/// zeros to stay panic-free, but that never happens for a 16-byte UUID.
111fn read_be_u64(bytes: &[u8]) -> u64 {
112    let mut buf = [0u8; 8];
113    let n = bytes.len().min(8);
114    buf[..n].copy_from_slice(&bytes[..n]);
115    u64::from_be_bytes(buf)
116}

118/// `AbstractTimeUUIDType.reorderTimestampBytes`, copied verbatim from
119/// `~/projects/cassandra/.../db/marshal/AbstractTimeUUIDType.java`:
120/// reshuffles the version-1 timestamp halves so a plain long compare orders by
121/// time. All shifts are on `u64` to match Java `>>>` (unsigned) and `<<`.
122fn reorder_timestamp_bytes(input: u64) -> u64 {
123    (input << 48) | ((input << 16) & 0xFFFF_0000_0000) | (input >> 32)
124}

126/// Java `Float.compare` total order for two `f32`, matching Cassandra
127/// `FloatType.compare` (issue #1275). Unlike Rust's `f32::total_cmp`, Java treats
128/// EVERY NaN (any payload/sign) as greater than every non-NaN value (NaN sorts
129/// last) and does NOT distinguish NaN bit-patterns; it also orders `-0.0 < +0.0`.
130pub(super) fn compare_f32_java(x: f32, y: f32) -> Ordering {
131    match (x.is_nan(), y.is_nan()) {
132        (true, true) => Ordering::Equal,
133        (true, false) => Ordering::Greater,
134        (false, true) => Ordering::Less,
135        // Neither is NaN: total_cmp gives the numeric order with -0.0 < +0.0.
136        (false, false) => x.total_cmp(&y),
137    }
138}

140/// Java `Double.compare` total order for two `f64`, matching Cassandra
141/// `DoubleType.compare` (issue #1275). See [`compare_f32_java`] for the NaN /
142/// signed-zero rules.
143pub(super) fn compare_f64_java(x: f64, y: f64) -> Ordering {

145        (true, true) => Ordering::Equal,
146        (true, false) => Ordering::Greater,
147        (false, true) => Ordering::Less,

152/// Scale-aware numeric comparison of two `decimal` values, matching Cassandra
153/// `DecimalType.compare` (`BigDecimal.compareTo`, arbitrary precision) (issue
154/// #1275). The unscaled part is a two's-complement varint; the value is
155/// `unscaled * 10^-scale`.
156///
157/// IMPORTANT (DoS-safe): we must NEVER materialize `10^(max_scale - own_scale)`,
158/// because a valid `i32` scale difference (e.g. `0` vs `2_147_483_647`) would
159/// build an astronomically large `BigInt` and hang/OOM the writer. Instead we
160/// mirror `BigDecimal.compareTo`'s bounded strategy:
161///
162///   1. Sign first: negative < zero < positive. Differing signs decide with no
163///      big-int multiply at all.
164///   2. Same-sign non-zero: compare by ADJUSTED EXPONENT
165///      `adj = (num_unscaled_digits - 1) - scale`, the base-10 exponent of the
166///      most-significant digit (`value ≈ d.dddd × 10^adj`). A larger adjusted
167///      exponent means a larger magnitude; apply the (shared) sign. Decided
168///      WITHOUT any `10^p`.
169///   3. Only when the adjusted exponents are EQUAL (same decimal magnitude band)
170///      do we align and compare exactly. There the rescale multiplier exponent
171///      is the scale delta, which — because the adjusted exponents match — is
172///      bounded by the difference in digit counts (`|digits_a - digits_b|`), a
173///      small number. That `10^delta` is safe to build.
174///
175/// Uses `num_bigint::BigInt` (already a cqlite-core dependency) so unscaled
176/// magnitudes wider than 128 bits compare exactly, with no saturation.
177pub(super) fn compare_decimal(
178    scale_a: i32,
179    unscaled_a: &[u8],
180    scale_b: i32,
181    unscaled_b: &[u8],
182) -> Ordering {
183    let big_a = varint_to_bigint(unscaled_a);
184    let big_b = varint_to_bigint(unscaled_b);
185
186    // 1. Sign comparison: negative < zero < positive. Differing signs decide
187    //    immediately with no rescale.
188    let sign_a = big_a.sign();
189    let sign_b = big_b.sign();
190    if sign_a != sign_b {
191        return sign_a.cmp(&sign_b);
192    }
193    // Equal signs that are both Zero → both values are exactly 0 (scale is
194    // irrelevant for a zero unscaled part), so the values are equal.
195    if sign_a == num_bigint::Sign::NoSign {
196        return Ordering::Equal;
197    }
198
199    // 2. Same-sign non-zero: compare by adjusted exponent (magnitude band).
200    //    digits = number of base-10 digits of |unscaled| (>= 1 here).
201    let digits_a = decimal_digit_count(&big_a);
202    let digits_b = decimal_digit_count(&big_b);
203    // adjusted exponent = (digits - 1) - scale, as i64 to avoid i32 overflow
204    // for extreme scales near i32::MIN/MAX.
205    let adj_a = (digits_a as i64 - 1) - scale_a as i64;
206    let adj_b = (digits_b as i64 - 1) - scale_b as i64;
207    let positive = sign_a == num_bigint::Sign::Plus;
208    if adj_a != adj_b {
209        let by_magnitude = adj_a.cmp(&adj_b);
210        // For negatives the larger magnitude is the SMALLER value.
211        return if positive {
212            by_magnitude
213        } else {
214            by_magnitude.reverse()
215        };
216    }
217
218    // 3. Adjusted exponents equal: same magnitude band, so an exact compare
219    //    requires aligning scales. The scale delta is bounded by the digit-count
220    //    difference (`adj_a == adj_b` ⇒ `scale_a - scale_b == digits_a -
221    //    digits_b`), which is small — safe to build `10^delta`.
222    let delta = scale_a as i64 - scale_b as i64; // == digits_a - digits_b
223    let (mut lhs, mut rhs) = (big_a, big_b);
224    match delta.cmp(&0) {
225        // scale_a > scale_b: a is finer-grained; scale b UP by multiplying b.
226        Ordering::Greater => rhs *= pow10(delta as u32),
227        // scale_a < scale_b: scale a UP by multiplying a.
228        Ordering::Less => lhs *= pow10((-delta) as u32),
229        Ordering::Equal => {}
230    }
231    lhs.cmp(&rhs)
232}

234/// Number of base-10 digits in the magnitude of `n` (with `n != 0`), matching the
235/// "precision" of `BigDecimal`. Derived from the decimal string of the magnitude
236/// so it is exact; the BigInt here is bounded by the unscaled body length (an
237/// actual serialized value), never by an unbounded power of ten.
238fn decimal_digit_count(n: &BigInt) -> u64 {
239    // magnitude() yields the BigUint; its base-10 string length is the digit
240    // count. Cheap because the unscaled body is a real (short) serialized value.
241    n.magnitude().to_str_radix(10).len() as u64
242}

244/// Decode a two's-complement big-endian `varint` body into an arbitrary-precision
245/// signed big integer (matches the `from_signed_bytes_be` decode used elsewhere in
246/// the crate). An empty body is `0`.
247fn varint_to_bigint(bytes: &[u8]) -> BigInt {
248    if bytes.is_empty() {
249        return BigInt::ZERO;
250    }
251    BigInt::from_signed_bytes_be(bytes)
252}
253
254/// Arbitrary-precision `10^exp` for decimal scale alignment.
255fn pow10(exp: u32) -> BigInt {
256    BigInt::from(10).pow(exp)
257}

532    /// Build a 16-byte UUID with the given version nibble (high nibble of byte 6),
533    /// `msb_lo48` packed into the timestamp/version region, and `lsb` as the low
534    /// 64 bits. `msb_hi32` is the time-low (bytes 0-3); `msb_mid16` is time-mid
535    /// (bytes 4-5). Layout follows the standard big-endian UUID byte order so the
536    /// comparator's `getLong(0)`/`getLong(8)` reads match Cassandra.
537    fn make_uuid(time_low: u32, time_mid: u16, version: u8, time_hi: u16, lsb: u64) -> [u8; 16] {
538        let mut u = [0u8; 16];
539        u[0..4].copy_from_slice(&time_low.to_be_bytes());
540        u[4..6].copy_from_slice(&time_mid.to_be_bytes());
541        // bytes 6-7: top nibble = version, remaining 12 bits = time_hi (0..=0xFFF).
542        let ver_and_hi: u16 = ((version as u16 & 0xf) << 12) | (time_hi & 0x0fff);
543        u[6..8].copy_from_slice(&ver_and_hi.to_be_bytes());
544        u[8..16].copy_from_slice(&lsb.to_be_bytes());
545        u
546    }

548    /// UUIDType step 1: version nibble decides first. A version-1 UUID whose raw
549    /// bytes are LARGER must still sort BEFORE a version-4 UUID, because
550    /// `version1 - version2` (1 - 4 < 0) wins regardless of the rest. Under the
551    /// old raw-serialized-byte fallback the all-0xFF v1 value would sort AFTER the
552    /// small v4 value — the REVERSE — so this fails before the fix, passes after.
553    #[test]
554    fn uuid_version_decides_first() {
555        // v1 with maximal bytes everywhere it can control.
556        let v1 = make_uuid(0xFFFF_FFFF, 0xFFFF, 1, 0x0FFF, u64::MAX);
557        // v4 with minimal bytes.
558        let v4 = make_uuid(0x0000_0000, 0x0000, 4, 0x0000, 0);
559        // Raw-byte fallback (the OLD behavior) would order v4 < v1 here:
560        assert_eq!(v4.as_slice().cmp(v1.as_slice()), Ordering::Less);
561        // Cassandra UUIDType orders by version first: v1 (1) < v4 (4), so
562        // compare(v1, v4) is Less (and compare(v4, v1) is Greater).
563        assert_eq!(
564            compare_collection_elements(&Value::Uuid(v1), &Value::Uuid(v4)),
565            Ordering::Less
566        );
567        assert_eq!(
568            compare_collection_elements(&Value::Uuid(v4), &Value::Uuid(v1)),
569            Ordering::Greater
570        );
571        let mut v = vec![Value::Uuid(v4), Value::Uuid(v1)];
572        v.sort_by(compare_collection_elements);
573        assert_eq!(v, vec![Value::Uuid(v1), Value::Uuid(v4)]);
574    }

576    /// UUIDType step 2 (version 1): two v1 UUIDs order by TIMESTAMP via
577    /// `reorderTimestampBytes`, NOT by raw bytes. `earlier` has a smaller
578    /// time_hi/time_mid but a LARGER time_low; the reorder puts time_hi/time_mid
579    /// in the most-significant position, so `earlier` (smaller hi) sorts first
580    /// even though its raw leading bytes (time_low) are larger. Grounded in
581    /// `UUIDType.compareCustom` + `reorderTimestampBytes`.
582    #[test]
583    fn uuid_v1_orders_by_timestamp() {
584        // earlier: time_hi=0x000, time_mid=0x0000, time_low=0xFFFF_FFFF
585        let earlier = make_uuid(0xFFFF_FFFF, 0x0000, 1, 0x000, 0);
586        // later: time_hi=0x001, everything else minimal
587        let later = make_uuid(0x0000_0000, 0x0000, 1, 0x001, 0);
588        // Raw bytes would call `earlier` (leading 0xFF..) GREATER than `later`:
589        assert_eq!(earlier.as_slice().cmp(later.as_slice()), Ordering::Greater);
590        // But the timestamp reorder makes the high-12-bits dominate: earlier < later.
591        assert_eq!(
592            compare_collection_elements(&Value::Uuid(earlier), &Value::Uuid(later)),
593            Ordering::Less
594        );
595        let mut v = vec![Value::Uuid(later), Value::Uuid(earlier)];
596        v.sort_by(compare_collection_elements);
597        assert_eq!(v, vec![Value::Uuid(earlier), Value::Uuid(later)]);
598    }

600    /// UUIDType step 2 (non-v1) + step 3: two version-4 UUIDs compare MSB then LSB
601    /// as UNSIGNED longs. Equal MSB, differing LSB → LSB unsigned order decides.
602    /// Also a v4 whose MSB has the high bit set must sort ABOVE one without it
603    /// (unsigned), which signed-long compare would get wrong.
604    #[test]
605    fn uuid_v4_orders_unsigned_msb_then_lsb() {
606        // Same MSB region (version 4), LSB differs: 1 < 2.
607        let a = make_uuid(0x0000_0001, 0x0000, 4, 0x000, 1);
608        let b = make_uuid(0x0000_0001, 0x0000, 4, 0x000, 2);
609        assert_eq!(
610            compare_collection_elements(&Value::Uuid(a), &Value::Uuid(b)),
611            Ordering::Less
612        );
613
614        // MSB high bit set (time_low top bit) must be treated as UNSIGNED-larger.
615        let low_msb = make_uuid(0x0000_0001, 0x0000, 4, 0x000, 0);
616        let high_msb = make_uuid(0x8000_0000, 0x0000, 4, 0x000, 0);
617        assert_eq!(
618            compare_collection_elements(&Value::Uuid(low_msb), &Value::Uuid(high_msb)),
619            Ordering::Less
620        );
621
622        let mut v = vec![Value::Uuid(b), Value::Uuid(high_msb), Value::Uuid(a)];
623        v.sort_by(compare_collection_elements);
624        // a (msb small, lsb 1) < b (msb small, lsb 2) < high_msb (msb big)
625        assert_eq!(
626            v,
627            vec![Value::Uuid(a), Value::Uuid(b), Value::Uuid(high_msb)]
628        );
629    }
