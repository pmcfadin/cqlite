1//! Static-row emission: static row write path, static row body, static column bitmap and static cells.
2//!
3//! Part of the `data_writer` responsibility split (issue #1118): this module
4//! holds one `impl DataWriter` block. `use super::*` pulls the shared writer
5//! types, serialization/schema helpers, flag constants, and crate imports
6//! re-exported from `data_writer/mod.rs`. No emitted bytes change.

10impl DataWriter {
11    /// Write a static row for the current partition
12    ///
13    /// Static rows contain STATIC column values at partition level.
14    /// They use extended flags and have NO clustering prefix.
15    ///
16    /// # Arguments
17    /// * `mutation` - Mutation containing static column values
18    /// * `schema` - Table schema for column metadata
19    ///
20    /// # Binary Format
21    /// ```text
22    /// [row_flags: u8]        ← 0x80 | other_flags (always HAS_EXTENDED_FLAGS)
23    /// [extended_flags: u8]   ← 0x01 (EXTENDED_IS_STATIC)
24    /// [row_size: VInt]       ← Size of body after this
25    /// [prev_size: VInt]      ← 0 or previous row size
26    /// [timestamp: VInt]      ← If HAS_TIMESTAMP (delta)
27    /// [ttl: VInt]            ← If HAS_TTL (delta)
28    /// [deletion: 2 VInts]    ← If HAS_DELETION
29    /// [column_bitmap]        ← If NOT HAS_ALL_COLUMNS
30    /// [cell_data...]         ← Static column cells only
31    /// ```
32    pub fn write_static_row(&mut self, mutation: &Mutation, schema: &TableSchema) -> Result<()> {
33        // Legacy/test/public entry point: derive per-op metadata from the single
34        // mutation (each op inherits the mutation's effective LDT).
35        let static_ops: Vec<StaticMergedOp> = mutation
36            .operations
37            .iter()
38            .map(|op| StaticMergedOp {
39                // #921 finding 2: a `Delete` cell tombstone keeps its own surfaced
40                // LDT; every other op falls back to the mutation's effective LDT.
41                cell_local_deletion_time: op_cell_local_deletion_time(op, mutation),
42                // #1018: a static `Write`/`WriteWithTtl`/`Delete` cell carries its
43                // OWN per-cell timestamp when the compaction merge→mutation path
44                // recorded one in `Mutation::cell_write_timestamps`. Resolve it the
45                // SAME way `collect_static_operations` (encoding.rs) does so a
46                // caller passing a compacted static mutation through this public
47                // entry point does not rewrite older static cells (live OR cell
48                // tombstones) to the row max. There is no partition tombstone
49                // shadow_floor in the single-mutation entry point, so the per-cell
50                // shadow drop does not apply here; the no-override path resolves to
51                // exactly `mutation.timestamp_micros`, leaving the single-writetime
52                // behavior byte-identical.
53                timestamp_micros: op_cell_write_timestamp(op, mutation),
54                op: op.clone(),
55                // #1196: carry statement-level TTL so a static `USING TTL` Write
56                // is emitted as an expiring cell, not a non-expiring one.
57                row_ttl_seconds: mutation.ttl_seconds,
58            })
59            .collect();
60        self.write_static_row_with_prev_size(
61            &static_ops,
62            mutation.timestamp_micros,
63            mutation.ttl_seconds,
64            schema,
65            0,
66        )?;
67        Ok(())
68    }

72    /// Write a static row from the merged static operations of a partition.
73    ///
74    /// Issue #764: each `StaticMergedOp` carries its own originating timestamp
75    /// and local deletion time, so a surviving static-column delete from an
76    /// older mutation keeps its own LDT instead of inheriting a single
77    /// synthetic mutation-level value (which corrupted the unsigned-VInt delta
78    /// when stats were seeded from that older delete's explicit lower LDT).
79    ///
80    /// Returns `(bytes_written, cells_written)` (Issue #851, review). The cell
81    /// count is sourced from the physical static-cell write path so Statistics'
82    /// column count matches Data.db (0 for a static row tombstone).
83    pub(super) fn write_static_row_with_prev_size(
84        &mut self,
85        static_ops: &[StaticMergedOp],
86        liveness_ts: i64,
87        ttl_seconds: Option<u32>,
88        schema: &TableSchema,
89        prev_size: u64,
90    ) -> Result<(usize, u64)> {
91        let start_len = self.buffer.len();
92
93        // Build row header flags - always includes HAS_EXTENDED_FLAGS for static rows
94        let mut flags = ROW_HAS_EXTENDED_FLAGS;
95
96        // Check if this is a row tombstone (only reachable via the public
97        // single-mutation entry point; `collect_static_operations` never emits
98        // a DeleteRow into the merged set).
99        let is_row_tombstone = static_ops.iter().any(|mop| {
100            matches!(
101                mop.op,
102                crate::storage::write_engine::mutation::CellOperation::DeleteRow
103            )
104        });
105
106        if is_row_tombstone {
107            flags |= ROW_HAS_DELETION;
108        }
109
110        // Issue #1196: a static row carries NO row-level primary-key liveness.
111        // Cassandra's static block is the pseudo-row keyed on the empty
112        // clustering; PK/row liveness lives on regular (clustering) rows, never
113        // on the static row. Every committed Cassandra 5.0.2 golden confirms
114        // this — no `static_block` ever has a `liveness_info`, so the static-row
115        // flags byte never sets ROW_HAS_TIMESTAMP (e.g. a static-only UPDATE
116        // emits `0xa0`, not `0xa4`). The writetime instead rides on each static
117        // CELL (see `write_static_cells`, which writes explicit per-cell
118        // timestamps below). We therefore never set ROW_HAS_TIMESTAMP here and
119        // never write a row-level timestamp delta in the body.
120
121        // TTL: a static row has no row-level liveness timestamp, so a row-level
122        // TTL/expiring marker would be meaningless without it (Cassandra never
123        // sets ROW_HAS_TTL on the static block). Per-cell TTL rides on the static
124        // cell via `write_cell_with_ttl` (which is self-describing: explicit
125        // timestamp + LDT + TTL deltas, no USE_ROW_* borrowing). We therefore
126        // never set ROW_HAS_TTL here; `ttl_seconds` is threaded through to
127        // `build_static_row_body` only so the (now-unreached) row-TTL body branch
128        // stays consistent.
129
130        // Check if all static columns are present
131        if !is_row_tombstone {
132            let all_writes = static_ops.iter().all(|mop| {
133                matches!(
134                    mop.op,
135                    crate::storage::write_engine::mutation::CellOperation::Write { .. }
136                        | crate::storage::write_engine::mutation::CellOperation::WriteWithTtl { .. }
137                )
138            });
139            let has_nulls = static_ops.iter().any(|mop| match &mop.op {
140                crate::storage::write_engine::mutation::CellOperation::Write { value, .. }
141                | crate::storage::write_engine::mutation::CellOperation::WriteWithTtl {
142                    value,
143                    ..
144                } => {
145                    matches!(value, Value::Null)
146                }
147                _ => false,
148            });
149            // Count static columns only for static row
150            let static_column_count = schema.columns.iter().filter(|c| c.is_static).count();
151
152            if all_writes && !has_nulls && static_ops.len() == static_column_count {
153                flags |= ROW_HAS_ALL_COLUMNS;
154            }
155        }
156
157        // Write row flags
158        self.buffer.push(flags);
159
160        // Write extended flags - always EXTENDED_IS_STATIC for static rows
161        self.buffer.push(EXTENDED_IS_STATIC);
162
163        // NO clustering prefix for static rows (key difference from write_row)
164
165        // Build row body
166        let (row_body, cells_written) =
167            self.build_static_row_body(static_ops, liveness_ts, ttl_seconds, schema, flags)?;
168
169        let prev_size_vint_len = unsigned_len(prev_size);
170
171        // Write row_size (VInt) — includes prev_unfiltered_size VInt + rest of body
172        let row_body_size = prev_size_vint_len as u64 + row_body.len() as u64;
173        let mut row_size_buf = Vec::new();
174        encode_unsigned(row_body_size, &mut row_size_buf);
175        self.buffer.extend_from_slice(&row_size_buf);
176
177        // Write prev_unfiltered_size (VInt, inside the row body)
178        encode_unsigned(prev_size, &mut self.buffer);
179
180        // Write rest of row body
181        self.buffer.extend_from_slice(&row_body);
182
183        Ok((self.buffer.len() - start_len, cells_written))
184    }

186    /// Build static row body (everything after row_size VInt)
187    ///
188    /// Similar to build_row_body but only processes static columns.
189    ///
190    /// Returns the body bytes and the number of static cells (columns)
191    /// physically written (Issue #851, review). A static row tombstone writes no
192    /// cells (count 0); otherwise the count is sourced from `write_static_cells`.
193    pub(super) fn build_static_row_body(
194        &self,
195        static_ops: &[StaticMergedOp],
196        liveness_ts: i64,
197        ttl_seconds: Option<u32>,
198        schema: &TableSchema,
199        flags: u8,
200    ) -> Result<(Vec<u8>, u64)> {
201        let mut body = Vec::new();
202
203        // Issue #2038 Scope B: capture "now" ONCE for this static row write and
204        // share it with every expiring cell derived below (mirrors
205        // `build_merged_row_body`; see `DataWriter::capture_now_seconds`).
206        let now_seconds = self.capture_now_seconds()?;
207
208        // Write timestamp delta (if HAS_TIMESTAMP)
209        //
210        // Fix #644 (S6): Cassandra writes UNSIGNED VInt for all temporal deltas.
211        // SerializationHeader.java:167: out.writeUnsignedVInt(timestamp - stats.minTimestamp)
212        if (flags & ROW_HAS_TIMESTAMP) != 0 {
213            let timestamp_delta = (liveness_ts - self.stats.min_timestamp) as u64;
214            encode_unsigned(timestamp_delta, &mut body);
215        }
216
217        // Write TTL delta (if HAS_TTL)
218        //
219        // Fix #644 (S6): Both TTL and LDT deltas are UNSIGNED VInt.
220        // SerializationHeader.java:177: out.writeUnsignedVInt32(ttl - stats.minTTL)
221        // SerializationHeader.java:172: out.writeUnsignedVInt32(ldt - stats.minLocalDeletionTime)
222        if (flags & ROW_HAS_TTL) != 0 {
223            if let Some(ttl) = ttl_seconds {
224                let ttl_delta = ttl as i64 - self.stats.min_ttl as i64;
225                if ttl_delta < 0 {
226                    return Err(Error::InvalidInput(format!(
227                        "TTL {} is less than min_ttl {}",
228                        ttl, self.stats.min_ttl
229                    )));
230                }
231                encode_unsigned(ttl_delta as u64, &mut body);
232
233                let local_deletion_time = self.expiring_local_deletion_time(now_seconds, ttl);
234                let ldt_delta =
235                    (local_deletion_time as i64) - (self.stats.min_local_deletion_time as i64);
236                if ldt_delta < 0 {
237                    return Err(Error::InvalidInput(format!(
238                        "Local deletion time {} is less than min_local_deletion_time {}",
239                        local_deletion_time, self.stats.min_local_deletion_time
240                    )));
241                }
242                encode_unsigned(ldt_delta as u64, &mut body);
243            }
244        }
245
246        // Write deletion (if HAS_DELETION)
247        if (flags & ROW_HAS_DELETION) != 0 {
248            // Row tombstone: Cassandra canonical order (markedForDeleteAt first, then localDeletionTime)
249            // Per SerializationHeader.writeDeletionTime(): writeTimestamp() then writeLocalDeletionTime()
250            // Fix #644 (S6): both are UNSIGNED VInt.
251            //
252            // The DeleteRow op carries the deletion timestamp + explicit LDT
253            // (Issue #764). Reachable only via the single-mutation entry point.
254            let delete_op = static_ops.iter().find(|mop| {
255                matches!(
256                    mop.op,
257                    crate::storage::write_engine::mutation::CellOperation::DeleteRow
258                )
259            });
260            let (deletion_ts, local_deletion_time) = delete_op
261                .map(|mop| (mop.timestamp_micros, mop.cell_local_deletion_time))
262                .unwrap_or((liveness_ts, (liveness_ts / 1_000_000) as i32));
263            let ts_delta = (deletion_ts - self.stats.min_timestamp) as u64;
264            encode_unsigned(ts_delta, &mut body);
265
266            // Issue #873: reject a row-tombstone LDT that is genuinely below the
267            // baseline, in normal (non-negative i32) time space — a silent
268            // `wrapping_sub` here would zero-extend the underflow into a huge u32,
269            // emit a multi-byte VInt, and corrupt the row body / row-size. A
270            // far-future LDT (negative as i32, value in [2^31, 2^32)) is a
271            // legitimate value, not corruption, so the wrapping arithmetic is
272            // intended there (matches the complex-deletion guard).
273            if local_deletion_time >= 0
274                && self.stats.min_local_deletion_time >= 0
275                && local_deletion_time < self.stats.min_local_deletion_time
276            {
277                return Err(Error::InvalidInput(format!(
278                    "Row tombstone: local deletion time {} is less than min_local_deletion_time {}",
279                    local_deletion_time, self.stats.min_local_deletion_time
280                )));
281            }
282            let ldt_delta =
283                local_deletion_time.wrapping_sub(self.stats.min_local_deletion_time) as u32;
284            encode_unsigned(ldt_delta as u64, &mut body);
285
286            // Issue #717: the columns subset is NOT optional for tombstone rows.
287            // Cassandra's UnfilteredSerializer always reads it after the deletion
288            // times whenever HAS_ALL_COLUMNS is unset; omitting it makes the
289            // reader consume the next row's bytes as a subset bitmask
290            // ("Invalid Columns subset bytes; too many bits set").
291            if (flags & ROW_HAS_ALL_COLUMNS) == 0 {
292                let static_columns = self.static_columns(schema);
293                let empty_present: std::collections::HashSet<&str> =
294                    std::collections::HashSet::new();
295                self.write_column_subset(&mut body, &static_columns, &empty_present)?;
296            }
297
298            // No cells written for row tombstones
299            return Ok((body, 0));
300        }
301
302        // Write column bitmap (if NOT HAS_ALL_COLUMNS)
303        // For static rows, bitmap only covers static columns
304        if (flags & ROW_HAS_ALL_COLUMNS) == 0 {
305            self.write_static_column_bitmap(&mut body, static_ops, schema)?;
306        }
307
308        // Write cell data for static columns only
309        let cells_written =
310            self.write_static_cells(&mut body, static_ops, liveness_ts, schema, now_seconds)?;
311
312        Ok((body, cells_written))
313    }

315    /// Write column bitmap for static columns only.
316    ///
317    /// Same Cassandra `Columns.Serializer.serializeSubset()` format as
318    /// `write_column_bitmap()` but scoped to static columns.
319    pub(super) fn write_static_column_bitmap(
320        &self,
321        buf: &mut Vec<u8>,
322        static_ops: &[StaticMergedOp],
323        schema: &TableSchema,
324    ) -> Result<()> {
325        // Collect names of columns that are present (non-NULL writes + deletes)
326        let present_columns: std::collections::HashSet<&str> = static_ops
327            .iter()
328            .filter_map(|mop| match &mop.op {
329                crate::storage::write_engine::mutation::CellOperation::Write { column, value }
330                | crate::storage::write_engine::mutation::CellOperation::WriteWithTtl {
331                    column,
332                    value,
333                    ..
334                } if !matches!(value, Value::Null) => Some(column.as_str()),
335                crate::storage::write_engine::mutation::CellOperation::Delete {
336                    column, ..
337                } => Some(column.as_str()),
338                _ => None,
339            })
340            .collect();
341
342        let static_columns = self.static_columns(schema);
343        self.write_column_subset(buf, &static_columns, &present_columns)
344    }

346    /// Write cells for static columns only.
347    ///
348    /// Issue #764: deletes use their ORIGINATING op's timestamp and local
349    /// deletion time (carried in `StaticMergedOp`), not a single synthetic
350    /// mutation-level value.
351    ///
352    /// Issue #851 (review): returns the number of static cells (columns)
353    /// physically serialized — sourced from this loop, the only place that
354    /// decides whether a static cell is emitted (null writes skipped; deletes
355    /// and non-null writes written) — so Statistics cannot drift from Data.db.
356    pub(super) fn write_static_cells(
357        &self,
358        buf: &mut Vec<u8>,
359        static_ops: &[StaticMergedOp],
360        liveness_ts: i64,
361        schema: &TableSchema,
362        now_seconds: i32,
363    ) -> Result<u64> {
364        // Get set of static column names for validation
365        let static_column_names: std::collections::HashSet<_> = schema
366            .columns
367            .iter()
368            .filter(|c| c.is_static)
369            .map(|c| &c.name)
370            .collect();
371
372        let mut cells_written: u64 = 0;
373        for mop in self.sorted_static_ops(static_ops, schema) {
374            match &mop.op {
375                crate::storage::write_engine::mutation::CellOperation::Write { column, value } => {
376                    // Only write if it's a static column
377                    if static_column_names.contains(column) && !matches!(value, Value::Null) {
378                        cells_written += 1;
379                        // roborev #1020 Finding 1: a frozen-UDT (or UDT-bearing
380                        // frozen collection/tuple) static cell is canonicalized
381                        // against the column's declared marshal before serialization
382                        // (declared order / `-1` padding / unknown-field reject). A
383                        // non-UDT static column is returned unchanged.
384                        let canon = canonicalize_static_value(schema, column, value)?;
385                        let value = canon.as_ref();
386                        // Issue #1196: a static row carries NO row-level liveness
387                        // (HAS_TIMESTAMP is never set on the static block — see
388                        // write_static_row_with_prev_size). There is therefore no
389                        // row timestamp to borrow: every static cell must carry its
390                        // OWN explicit timestamp delta (cell flags WITHOUT
391                        // CELL_USE_ROW_TIMESTAMP), matching Cassandra 5.0.2 (a
392                        // static-only UPDATE writes the static cell with flags 0x00
393                        // + an explicit timestamp delta, not 0x08/USE_ROW_TIMESTAMP).
394                        let _ = liveness_ts; // static cells never use row timestamp
395                                             // Issue #1196 (regression fix): a static `USING TTL` write
396                                             // arrives as a plain `Write` with the statement TTL in
397                                             // `row_ttl_seconds`. Cassandra encodes that TTL on the CELL
398                                             // (an expiring cell), never as row-level liveness on the
399                                             // static block. Without this routing the TTL would be
400                                             // dropped and the cell serialized as non-expiring (lives
401                                             // forever) — a data-liveness regression. The no-TTL path
402                                             // stays byte-identical (explicit-ts, flags 0x00).
403                        match mop.row_ttl_seconds {
404                            // Row-level `USING TTL` static write (no per-cell LDT
405                            // source): derive `now_seconds + ttl` (historical
406                            // behavior).
407                            Some(ttl) => self.write_cell_with_ttl(
408                                buf,
409                                column,
410                                value,
411                                mop.timestamp_micros,
412                                ttl,
413                                None,
414                                now_seconds,
415                            )?,
416                            None => self.write_cell_explicit_ts(
417                                buf,
418                                column,
419                                value,
420                                mop.timestamp_micros,
421                            )?,
422                        }
423                    }
424                }
425                crate::storage::write_engine::mutation::CellOperation::WriteWithTtl {
426                    column,
427                    value,
428                    ttl_seconds,
429                    local_deletion_time,
430                } => {
431                    // Only write if it's a static column
432                    if static_column_names.contains(column) && !matches!(value, Value::Null) {
433                        cells_written += 1;
434                        // roborev #1020 Finding 1: schema-aware frozen-UDT value.
435                        let canon = canonicalize_static_value(schema, column, value)?;
436                        // Issue #1538: honor the authoritative per-cell LDT verbatim.
437                        self.write_cell_with_ttl(
438                            buf,
439                            column,
440                            canon.as_ref(),
441                            mop.timestamp_micros,
442                            *ttl_seconds,
443                            *local_deletion_time,
444                            now_seconds,
445                        )?;
446                    }
447                }
448                crate::storage::write_engine::mutation::CellOperation::Delete {
449                    column, ..
450                } => {
451                    // Only process if it's a static column
452                    if static_column_names.contains(column) {
453                        cells_written += 1;
454                        // Issue #764: honor the originating op's explicit LDT.
455                        self.write_tombstone_cell(
456                            buf,
457                            column,
458                            mop.timestamp_micros,
459                            mop.cell_local_deletion_time,
460                        )?;
461                    }
462                }
463                crate::storage::write_engine::mutation::CellOperation::DeleteRow => {
464                    // Row deletion handled at row level with HAS_DELETION flag
465                }
466                // Per-element complex ops (epic #899) are never collected into the
467                // static-op set (collect_static_operations skips them); STATIC
468                // complex columns are out of scope for the Phase B capability.
469                crate::storage::write_engine::mutation::CellOperation::WriteComplexElement {
470                    ..
471                }
472                | crate::storage::write_engine::mutation::CellOperation::ComplexDeletion {
473                    ..
474                } => {}
475            }
476        }
477
478        Ok(cells_written)
479    }

481    /// Sort merged static ops into Cassandra static-column serialization order
482    /// (simple columns before complex, then by name).
483    pub(super) fn sorted_static_ops<'a, 'b>(
484        &self,
485        ops: &'b [StaticMergedOp],
486        schema: &'a TableSchema,
487    ) -> Vec<&'b StaticMergedOp> {
488        let columns = self.static_columns(schema);
489        let column_order: std::collections::HashMap<&str, usize> = columns
490            .iter()
491            .enumerate()
492            .map(|(idx, column)| (column.name.as_str(), idx))
493            .collect();
494
495        let mut sorted: Vec<&'b StaticMergedOp> = ops.iter().collect();
496        sorted.sort_by_key(|mop| match &mop.op {
497            crate::storage::write_engine::mutation::CellOperation::Write { column, .. }
498            | crate::storage::write_engine::mutation::CellOperation::WriteWithTtl {
499                column, ..
500            }
501            | crate::storage::write_engine::mutation::CellOperation::Delete { column, .. }
502            | crate::storage::write_engine::mutation::CellOperation::WriteComplexElement {
503                column,
504                ..
505            }
506            | crate::storage::write_engine::mutation::CellOperation::ComplexDeletion {
507                column,
508                ..
509            } => column_order
510                .get(column.as_str())
511                .copied()
512                .unwrap_or(usize::MAX - 1),
513            crate::storage::write_engine::mutation::CellOperation::DeleteRow => usize::MAX,
514        });
515        sorted
516    }
