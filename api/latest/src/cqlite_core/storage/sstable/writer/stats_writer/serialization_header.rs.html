1//! SERIALIZATION_HEADER component builder (MetadataType ordinal 3).
2//!
3//! Serialises the table's EncodingStats baselines and schema (key type,
4//! clustering types, static/regular columns) into the Statistics.db
5//! SERIALIZATION_HEADER component.
6
7use super::marshal::cql_type_to_marshal_type;
8use super::metadata::StatisticsMetadata;
9use super::{StatisticsWriter, DELETION_TIME_EPOCH, TIMESTAMP_EPOCH, TTL_EPOCH};
10use crate::error::Result;
11use crate::parser::vint::encode_vuint;
12use crate::schema::TableSchema;
13use std::io::Write;

15impl StatisticsWriter {
16    /// Build SERIALIZATION_HEADER component (MetadataType ordinal 3)
17    ///
18    /// Format (SerializationHeader.java Serializer, lines 594-603):
19    /// - EncodingStats: 3 unsigned VInts (minTimestamp, minLocalDeletionTime, minTTL deltas from epochs)
20    /// - keyType: VInt length + UTF-8 type string
21    /// - clusteringTypes: unsigned VInt count + list of types
22    /// - staticColumns: unsigned VInt count + map of (column name, type)
23    /// - regularColumns: unsigned VInt count + map of (column name, type)
24    ///
25    /// When `schema` is Some, populates keyType, clustering types, and column
26    /// names/types from the actual table schema. When None, falls back to a
27    /// minimal stub (BytesType, zero columns).
28    ///
29    /// # Column-set encoding (>64 columns)
30    ///
31    /// The static/regular column sets are encoded exactly as Cassandra's
32    /// `SerializationHeader.Serializer.writeColumnsWithTypes`
33    /// (cassandra-5.0.0 `SerializationHeader.java` lines 489-497): an unsigned-VInt
34    /// column count followed by `count` `(VInt-length name, VInt-length marshal type)`
35    /// pairs. This path has **no** 64-column limit and never uses a bitmap.
36    ///
37    /// The 64-bit bitmap encoding lives in `Columns.serializer.serializeSubset`
38    /// (`Columns.java` lines 503-531) and is only used to serialise a per-row column
39    /// subset against a pre-shared superset (Data.db rows / inter-node messaging),
40    /// where `supersetCount < 64` selects the bitmap and `>= 64` switches to a VInt
41    /// delta list. It is never used for the SSTable SERIALIZATION_HEADER, so wide
42    /// tables (>64 columns) round-trip losslessly here.
43    pub(super) fn build_serialization_header_component(
44        &self,
45        schema: Option<&TableSchema>,
46        metadata: &StatisticsMetadata,
47    ) -> Result<Vec<u8>> {
48        let mut buffer = Vec::new();
49
50        // EncodingStats: 3 unsigned VInts representing deltas from epochs.
51        // These baselines MUST match the values used by DataWriter for delta encoding.
52        // Cassandra: EncodingStats.Serializer.serialize() writes:
53        //   writeUnsignedVInt(minTimestamp - TIMESTAMP_EPOCH)
54        //   writeUnsignedVInt(minLocalDeletionTime - DELETION_TIME_EPOCH)
55        //   writeUnsignedVInt(minTTL - TTL_EPOCH)
56
57        // minTimestamp delta from epoch
58        let min_ts = if metadata.min_timestamp == i64::MAX {
59            // No data recorded: use epoch as baseline
60            TIMESTAMP_EPOCH as u64
61        } else {
62            metadata.min_timestamp as u64
63        };
64        let min_ts_delta = min_ts.wrapping_sub(TIMESTAMP_EPOCH as u64);
65        buffer.write_all(&encode_vuint(min_ts_delta))?;
66
67        // minLocalDeletionTime delta from epoch.
68        //
69        // Cassandra: `EncodingStats.Serializer.serialize` writes the local-deletion
70        // baseline with `writeUnsignedVInt32(minLocalDeletionTime - DELETION_TIME_EPOCH)`
71        // (both operands are Java `int`s), and the reader recovers it with
72        // `readUnsignedVInt32()`, which runs `VIntCoding.checkedCast` and REJECTS any
73        // decoded value that does not round-trip through a SIGNED 32-bit `int`
74        // (`(int)value != value`). The on-disk form therefore carries the SIGN-EXTENDED
75        // delta: for a small LDT like 2, `2 - DELETION_TIME_EPOCH` is the negative int
76        // `-1442879998`, written as the sign-extended 64-bit VInt `0xFFFFFFFFAA…` which
77        // `readUnsignedVInt32` accepts and folds back to `2`. Truncating to a bare `u32`
78        // (e.g. `2852087298`) is OUT OF RANGE for `checkedCast` and makes Cassandra's
79        // `sstabledump` reject the SSTable (verified live against `cassandra:5.0`).
80        //
81        // Casting the i32 delta to `u64` sign-extends in Rust exactly as Java's
82        // `writeUnsignedVInt32` requires, and it also handles a far-future LDT stored as
83        // a negative i32 bit pattern identically (the bit pattern IS the signed int the
84        // reader expects). This mirrors the DataWriter per-row deletion deltas.
85        let min_ldt = if metadata.min_local_deletion_time == i32::MAX {
86            // No deletions: use Integer.MAX_VALUE as baseline (DeletionTime.LIVE)
87            i32::MAX
88        } else {
89            metadata.min_local_deletion_time
90        };
91        let min_del_delta = (min_ldt.wrapping_sub(DELETION_TIME_EPOCH) as i64) as u64;
92        buffer.write_all(&encode_vuint(min_del_delta))?;
93
94        // minTTL delta from TTL_EPOCH (TTL_EPOCH=0)
95        let min_ttl = if metadata.min_ttl == i32::MAX {
96            // No TTL: use 0 as baseline
97            0u64
98        } else {
99            metadata.min_ttl as u64
100        };
101        let min_ttl_delta = min_ttl.wrapping_sub(TTL_EPOCH as u64);
102        buffer.write_all(&encode_vuint(min_ttl_delta))?;
103
104        match schema {
105            Some(s) => {
106                // keyType: single PK → simple type, composite PK → CompositeType(...)
107                let key_marshal = if s.partition_keys.len() > 1 {
108                    let inner: Vec<String> = s
109                        .partition_keys
110                        .iter()
111                        .map(|pk| cql_type_to_marshal_type(&pk.data_type))
112                        .collect();
113                    format!(
114                        "org.apache.cassandra.db.marshal.CompositeType({})",
115                        inner.join(",")
116                    )
117                } else if !s.partition_keys.is_empty() {
118                    cql_type_to_marshal_type(&s.partition_keys[0].data_type)
119                } else {
120                    "org.apache.cassandra.db.marshal.BytesType".to_string()
121                };
122                buffer.write_all(&encode_vuint(key_marshal.len() as u64))?;
123                buffer.write_all(key_marshal.as_bytes())?;
124
125                // clusteringTypes: VUInt count + for each CK: VUInt-length-prefixed marshal type
126                buffer.write_all(&encode_vuint(s.clustering_keys.len() as u64))?;
127                for ck in &s.clustering_keys {
128                    let ck_marshal = cql_type_to_marshal_type(&ck.data_type);
129                    buffer.write_all(&encode_vuint(ck_marshal.len() as u64))?;
130                    buffer.write_all(ck_marshal.as_bytes())?;
131                }
132
133                // Collect partition key and clustering key names for filtering
134                let pk_names: std::collections::HashSet<&str> =
135                    s.partition_keys.iter().map(|k| k.name.as_str()).collect();
136                let ck_names: std::collections::HashSet<&str> =
137                    s.clustering_keys.iter().map(|k| k.name.as_str()).collect();
138
139                // staticColumns: filter for is_static && not PK/CK, sorted alphabetically
140                let mut static_cols: Vec<_> = s
141                    .columns
142                    .iter()
143                    .filter(|c| {
144                        c.is_static
145                            && !pk_names.contains(c.name.as_str())
146                            && !ck_names.contains(c.name.as_str())
147                    })
148                    .collect();
149                static_cols.sort_by(|a, b| a.name.cmp(&b.name));
150                buffer.write_all(&encode_vuint(static_cols.len() as u64))?;
151                for col in &static_cols {
152                    // Column name: VUInt length + UTF-8 bytes
153                    buffer.write_all(&encode_vuint(col.name.len() as u64))?;
154                    buffer.write_all(col.name.as_bytes())?;
155                    // Column type: VUInt length + marshal type bytes
156                    let col_marshal = cql_type_to_marshal_type(&col.data_type);
157                    buffer.write_all(&encode_vuint(col_marshal.len() as u64))?;
158                    buffer.write_all(col_marshal.as_bytes())?;
159                }
160
161                // regularColumns: filter for !is_static && not PK/CK, sorted alphabetically
162                // Cassandra's SerializationHeader stores columns in natural order (alphabetical)
163                let mut regular_cols: Vec<_> = s
164                    .columns
165                    .iter()
166                    .filter(|c| {
167                        !c.is_static
168                            && !pk_names.contains(c.name.as_str())
169                            && !ck_names.contains(c.name.as_str())
170                    })
171                    .collect();
172                regular_cols.sort_by(|a, b| a.name.cmp(&b.name));
173                buffer.write_all(&encode_vuint(regular_cols.len() as u64))?;
174                for col in &regular_cols {
175                    buffer.write_all(&encode_vuint(col.name.len() as u64))?;
176                    buffer.write_all(col.name.as_bytes())?;
177                    let col_marshal = cql_type_to_marshal_type(&col.data_type);
178                    buffer.write_all(&encode_vuint(col_marshal.len() as u64))?;
179                    buffer.write_all(col_marshal.as_bytes())?;
180                }
181            }
182            None => {
183                // Minimal stub: BytesType key, no clustering, no columns
184                let key_type = b"org.apache.cassandra.db.marshal.BytesType";
185                buffer.write_all(&encode_vuint(key_type.len() as u64))?;
186                buffer.write_all(key_type)?;
187
188                // clusteringTypes: 0
189                buffer.write_all(&encode_vuint(0))?;
190                // staticColumns: 0
191                buffer.write_all(&encode_vuint(0))?;
192                // regularColumns: 0
193                buffer.write_all(&encode_vuint(0))?;
194            }
195        }
196
197        Ok(buffer)
198    }
199}

352    /// Parse a Cassandra SSTable SERIALIZATION_HEADER column-set the way Cassandra's
353    /// `SerializationHeader.Serializer.readColumnsWithType` does (cassandra-5.0.0
354    /// `SerializationHeader.java` lines 510-520):
355    ///
356    /// ```text
357    /// unsigned-vint  count
358    /// repeat count times:
359    ///   vint-length-prefixed UTF-8  column name
360    ///   vint-length-prefixed UTF-8  marshal type
361    /// ```
362    ///
363    /// Returns the parsed `(name, marshal_type)` pairs and the slice remaining after
364    /// the column set, so chained sets (static then regular) can be parsed.
365    fn parse_columns_with_types(input: &[u8]) -> (Vec<(String, String)>, &[u8]) {
366        let (mut rest, count) = parse_vuint(input).expect("column count vint");
367        let mut cols = Vec::with_capacity(count as usize);
368        for _ in 0..count {
369            let (after_name_len, name_len) = parse_vuint(rest).expect("name length vint");
370            let name =
371                String::from_utf8(after_name_len[..name_len as usize].to_vec()).expect("name utf8");
372            let after_name = &after_name_len[name_len as usize..];
373
374            let (after_type_len, type_len) = parse_vuint(after_name).expect("type length vint");
375            let marshal =
376                String::from_utf8(after_type_len[..type_len as usize].to_vec()).expect("type utf8");
377            rest = &after_type_len[type_len as usize..];
378
379            cols.push((name, marshal));
380        }
381        (cols, rest)
382    }

421    /// Regression test for issue #763: a table with more than 64 regular columns
422    /// must produce a SERIALIZATION_HEADER that encodes every column.
423    ///
424    /// Cassandra's on-disk header uses `writeColumnsWithTypes` (SerializationHeader.java
425    /// lines 489-497): an unsigned-VInt count followed by `count` (name, type) pairs.
426    /// There is NO 64-bit bitmap on this path — the bitmap (`Columns.serializeSubset`,
427    /// Columns.java lines 503-531) is only used for per-row column subsets against a
428    /// pre-shared superset, never for the SSTable header. So a 70-column table is a
429    /// fully supported, lossless encoding.
430    #[test]
431    fn test_serialization_header_70_columns_roundtrip() {
432        let schema = wide_schema(70);
433        let writer = StatisticsWriter::new(PathBuf::from("test.db"));
434        let meta = StatisticsMetadata::new();
435        let bytes = writer
436            .build_serialization_header_component(Some(&schema), &meta)
437            .expect("build header for 70-column schema");
438
439        // Skip the 3 EncodingStats VInts, the key type, and the clustering list to
440        // reach the static/regular column sets.
441        let (rest, _min_ts) = parse_vuint(&bytes).expect("encoding stats minTimestamp");
442        let (rest, _min_ldt) = parse_vuint(rest).expect("encoding stats minLocalDeletionTime");
443        let (rest, _min_ttl) = parse_vuint(rest).expect("encoding stats minTTL");
444
445        // keyType: vint-length-prefixed UTF-8
446        let (rest, key_len) = parse_vuint(rest).expect("key type length");
447        let key_type = std::str::from_utf8(&rest[..key_len as usize]).expect("key type utf8");
448        assert_eq!(key_type, "org.apache.cassandra.db.marshal.UUIDType");
449        let rest = &rest[key_len as usize..];
450
451        // clusteringTypes: vint count (0 here)
452        let (rest, ck_count) = parse_vuint(rest).expect("clustering count");
453        assert_eq!(ck_count, 0, "no clustering columns");
454
455        // staticColumns then regularColumns
456        let (statics, rest) = parse_columns_with_types(rest);
457        assert_eq!(statics.len(), 0, "no static columns");
458
459        let (regulars, rest) = parse_columns_with_types(rest);
460        assert!(rest.is_empty(), "header fully consumed, no trailing bytes");
461
462        // All 70 regular columns must be present (the PK `id` is excluded).
463        assert_eq!(
464            regulars.len(),
465            70,
466            "all 70 regular columns must be encoded, got {}",
467            regulars.len()
468        );
469
470        // Columns are emitted in alphabetical order; verify a sample round-trips.
471        assert_eq!(regulars[0].0, "c000");
472        assert_eq!(regulars[0].1, "org.apache.cassandra.db.marshal.Int32Type");
473        assert_eq!(regulars[69].0, "c069");
474
475        // Every column name and type must be intact (lossless).
476        let mut names: Vec<String> = regulars.iter().map(|(n, _)| n.clone()).collect();
477        names.sort();
478        for (i, name) in names.iter().enumerate().take(70) {
479            assert_eq!(*name, format!("c{i:03}"));
480        }
481    }

522    /// Regression: the EncodingStats `minLocalDeletionTime` delta must be written
523    /// as the SIGN-EXTENDED `int` delta `minLocalDeletionTime - DELETION_TIME_EPOCH`,
524    /// because Cassandra recovers it with `readUnsignedVInt32()` →
525    /// `VIntCoding.checkedCast`, which REJECTS any decoded value that does not
526    /// round-trip through a signed 32-bit `int` (`(int)value != value`). A bare
527    /// `u32` truncation of a small LDT (e.g. delta `2852087298` for LDT `2`) is out
528    /// of range for `checkedCast` and makes `cassandra:5.0` `sstabledump` reject the
529    /// SSTable (verified live; see tests/issue_911_bti_partition_deletion_stats.rs).
530    ///
531    /// A far-future LDT in `[2^31, 2^32)` is stored as a negative i32 bit pattern,
532    /// which IS the signed int the reader expects, so the same sign-extending path
533    /// handles it. This pins both: the small-LDT delta is a sign-extended 64-bit
534    /// VInt, and the decoded value round-trips through a signed i32.
535    #[test]
536    fn test_serialization_header_ldt_baseline_sign_extends_for_checked_cast() {
537        // Helper: decode the minLocalDeletionTime EncodingStats delta for a given
538        // baseline and assert it round-trips through Cassandra's checkedCast (i.e.
539        // the decoded u64, reinterpreted as i64, fits a signed i32).
540        let decode_delta = |min_ldt: i32| -> u64 {
541            let mut meta = StatisticsMetadata::new();
542            meta.min_local_deletion_time = min_ldt;
543            let writer = StatisticsWriter::new(PathBuf::from("test.db"));
544            let bytes = writer
545                .build_serialization_header_component(None, &meta)
546                .expect("build header");
547            let (rest, _min_ts_delta) = parse_vuint(&bytes).expect("minTimestamp delta");
548            let (_rest, min_ldt_delta) = parse_vuint(rest).expect("minLocalDeletionTime delta");
549            min_ldt_delta
550        };
551
552        // checkedCast accepts `value` iff `(int)value == value` — i.e. the decoded
553        // u64, reinterpreted as i64, equals its own truncation to i32.
554        let passes_checked_cast = |delta: u64| -> bool {
555            let v = delta as i64;
556            (v as i32) as i64 == v
557        };
558
559        // Small LDT (e.g. 2): the regression case. The delta is the negative int
560        // `2 - DELETION_TIME_EPOCH`, sign-extended.
561        let small = decode_delta(2);
562        let expected_small = (2i32.wrapping_sub(DELETION_TIME_EPOCH) as i64) as u64;
563        assert_eq!(small, expected_small);
564        assert!(
565            passes_checked_cast(small),
566            "small-LDT delta must round-trip through a signed i32 (checkedCast), got {small:#x}"
567        );
568
569        // Far-future LDT in [2^31, 2^32): a negative i32 bit pattern. Same path.
570        let far_future = ((1u32 << 31) + 5) as i32;
571        assert!(far_future < 0, "sanity: far-future LDT is a negative i32");
572        let far = decode_delta(far_future);
573        let expected_far = (far_future.wrapping_sub(DELETION_TIME_EPOCH) as i64) as u64;
574        assert_eq!(far, expected_far);
575        assert!(
576            passes_checked_cast(far),
577            "far-future LDT delta must round-trip through a signed i32 (checkedCast), got {far:#x}"
578        );
579    }
