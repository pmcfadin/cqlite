1//! Statistics metadata collected during memtable flush.
2//!
3//! Holds the per-SSTable aggregates (timestamp/TTL/deletion-time ranges, counts,
4//! key range, tombstone drop-time histogram) consumed by the Statistics.db
5//! component builders.
6
7use super::estimated_histogram::EstimatedHistogram;
8use std::collections::BTreeMap;
9
10/// Maximum number of bins in the tombstone drop-time histogram.
11///
12/// Mirrors Cassandra's `StreamingTombstoneHistogramBuilder.MAX_BIN_SIZE = 100`.
13pub(super) const TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE: i32 = 100;
14
15/// Streaming tombstone drop-time histogram.
16///
17/// Mirrors Cassandra's `StreamingTombstoneHistogramBuilder` with a fixed maximum
18/// of 100 bins. Tombstone local-deletion-times are accumulated; when the bin count
19/// exceeds the maximum the two nearest bins are merged, keeping the bin count at or
20/// below `MAX_BIN_SIZE`.
21///
22/// Serialisation uses the **legacy** (nb/mc) format:
23/// ```text
24/// i32 BE  maxBinSize  (100 when non-empty, 0 when empty)
25/// i32 BE  size        (actual number of bins)
26/// for each bin:
27///   f64 BE  point  (local-deletion-time as a double)
28///   i64 BE  value  (count of tombstones in this bin)
29/// ```
30#[derive(Debug, Clone, Default)]
31pub struct TombstoneHistogram {
32    /// Bins keyed by `local_deletion_time as i64` (for stable ordering).
33    /// Each value is the count of tombstones that map to this bin.
34    bins: BTreeMap<i64, i64>,
35}
36
37impl TombstoneHistogram {
38    /// Create an empty histogram.
39    pub fn new() -> Self {
40        Self::default()
41    }
42
43    /// Record a tombstone with the given local-deletion-time (Unix seconds).
44    ///
45    /// If adding the new point would exceed `MAX_BIN_SIZE` the two closest
46    /// existing bins are merged first, exactly as Cassandra's builder does.
47    pub fn update(&mut self, local_deletion_time: i32) {
48        let key = local_deletion_time as i64;
49        *self.bins.entry(key).or_insert(0) += 1;
50
51        // Merge bins while over capacity
52        while self.bins.len() as i32 > TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE {
53            self.merge_closest_bins();
54        }
55    }
56
57    /// Return `true` if no tombstone deletion times have been recorded.
58    pub fn is_empty(&self) -> bool {
59        self.bins.is_empty()
60    }
61
62    /// Return the number of populated bins — NOT the number of observations
63    /// folded (issue #4246 roborev round-4 HIGH finding: a caller asserting
64    /// "folded exactly once/twice" must use [`Self::total_observations`]
65    /// instead — two `update()` calls for the SAME `local_deletion_time`
66    /// land in the SAME bin, so `size()` stays `1` either way).
67    pub fn size(&self) -> i32 {
68        self.bins.len() as i32
69    }
70
71    /// Total observation count across every bin (`Σ` of each bin's count) —
72    /// the exact quantity `update()` increments once per call, regardless of
73    /// how many DISTINCT `local_deletion_time` values have been observed.
74    /// `pub(crate)` and `#[cfg(test)]` (issue #4246 roborev round-4 finding)
75    /// so a stats-fold regression test in a DIFFERENT module (same crate,
76    /// `writer::stats_fold::tests`) can assert a marker was folded exactly
77    /// once, not merely that it landed in one bin — `size()` cannot
78    /// distinguish "folded once" from "folded twice at the identical LDT".
79    /// `#[cfg(test)]` because it has no production caller: gating it lets a
80    /// crate-wide `--cfg test` build (any `cargo test`) see it while the
81    /// normal (non-test) lib build — which `cargo clippy`'s dead-code pass
82    /// checks independently of any test target, even under `--all-targets`
83    /// — compiles it out entirely rather than flagging it dead.
84    #[cfg(test)]
85    pub(crate) fn total_observations(&self) -> i64 {
86        self.bins.values().sum()
87    }
88
89    /// Merge the two bins whose keys are closest together.
90    ///
91    /// The merged bin's key is the weighted average of the two points;
92    /// its value is the sum of both counts.  This matches Cassandra's
93    /// `mergeNearestBins` approach.
94    fn merge_closest_bins(&mut self) {
95        if self.bins.len() < 2 {
96            return;
97        }
98
99        // Find the pair of adjacent keys with the smallest gap.
100        let keys: Vec<i64> = self.bins.keys().copied().collect();
101        let mut min_gap = i64::MAX;
102        let mut merge_idx = 0usize;
103        for i in 0..keys.len() - 1 {
104            let gap = keys[i + 1] - keys[i];
105            if gap < min_gap {
106                min_gap = gap;
107                merge_idx = i;
108            }
109        }
110
111        let k1 = keys[merge_idx];
112        let k2 = keys[merge_idx + 1];
113        let v1 = self.bins.remove(&k1).unwrap_or(0);
114        let v2 = self.bins.remove(&k2).unwrap_or(0);
115        let total = v1 + v2;
116        // Weighted average key (same formula as Cassandra)
117        let merged_key = if total == 0 {
118            (k1 + k2) / 2
119        } else {
120            (k1 * v1 + k2 * v2) / total
121        };
122        *self.bins.entry(merged_key).or_insert(0) += total;
123    }
124
125    /// Iterate over bins in ascending key order as `(point_as_f64, count)` pairs.
126    pub(super) fn entries(&self) -> impl Iterator<Item = (f64, i64)> + '_ {
127        self.bins.iter().map(|(&k, &v)| (k as f64, v))
128    }
129}

131/// Statistics metadata collected during memtable flush
132///
133/// This structure holds all the metadata needed to write Statistics.db.
134/// Values are collected as rows are written to Data.db.
135#[derive(Debug, Clone)]
136pub struct StatisticsMetadata {
137    /// Minimum timestamp in the SSTable (microseconds since epoch)
138    pub min_timestamp: i64,
139    /// Maximum timestamp in the SSTable (microseconds since epoch)
140    pub max_timestamp: i64,
141    /// Minimum local deletion time (seconds since epoch, for tombstones)
142    pub min_local_deletion_time: i32,
143    /// Maximum local deletion time (seconds since epoch)
144    pub max_local_deletion_time: i32,
145    /// Minimum TTL value (seconds, 0 if no TTL)
146    pub min_ttl: i32,
147    /// Maximum TTL value (seconds, 0 if no TTL)
148    pub max_ttl: i32,
149    /// Total number of partitions in the SSTable
150    pub partition_count: u64,
151    /// Total number of rows (live + tombstones)
152    pub row_count: u64,
153    /// Total number of columns across all rows
154    pub column_count: u64,
155    /// Total size of all rows in bytes
156    pub total_rows_size: u64,
157    /// Histogram of tombstone local-deletion-times for `estimatedTombstoneDropTime`.
158    ///
159    /// Populated by `update_local_deletion_time`; serialised into Statistics.db
160    /// so Cassandra can compute `estimatedDroppableTombstoneRatio` and schedule
161    /// tombstone compaction.  Uses the same streaming builder algorithm as
162    /// Cassandra's `StreamingTombstoneHistogramBuilder` (max 100 bins).
163    pub tombstone_histogram: TombstoneHistogram,
164    /// First (lowest-token) decorated partition-key bytes written to this
165    /// SSTable, or `None` if no partition has been written. Serialised as the
166    /// `da`-format `StatsMetadata.firstKey` (BtiFormat `hasKeyRange`); tracked in
167    /// token order by `SSTableWriter::write_partition`.
168    pub first_key: Option<Vec<u8>>,
169    /// Last (highest-token) decorated partition-key bytes written to this
170    /// SSTable. Serialised as the `da`-format `StatsMetadata.lastKey`.
171    pub last_key: Option<Vec<u8>>,
172    /// Whether any partition written to this SSTable carries a partition-level
173    /// deletion (partition tombstone).
174    ///
175    /// Serialised as the `da`-format `StatsMetadata.hasPartitionLevelDeletions`
176    /// boolean (`hasPartitionLevelDeletionsPresenceMarker`); written as a single
177    /// `writeBoolean` byte (`0x01` true, `0x00` false) — matching
178    /// cassandra-5.0.0 `StatsMetadata.StatsMetadataSerializer.serialize`
179    /// (line 495). Set by `SSTableWriter::write_partition` whenever a mutation
180    /// contributes a `partition_tombstone`. Ignored by the legacy BIG (nb/oa)
181    /// STATS body, which never serialises this field.
182    pub has_partition_level_deletions: bool,
183
184    /// `repairedAt` repair timestamp (`0` = unrepaired). Serialised verbatim
185    /// into the STATS component `repairedAt` field. Preserved through compaction
186    /// from the (compatible) input SSTables (issue #1021); a fresh memtable flush
187    /// leaves it `0`.
188    pub repaired_at: i64,
189
190    /// `pendingRepair` incremental-repair session UUID (`None` = no pending
191    /// repair). Serialised as the STATS `pendingRepair` nullable field (presence
192    /// byte then 16-byte UUID). Preserved through compaction from compatible
193    /// inputs (issue #1021); a fresh flush leaves it `None`.
194    pub pending_repair: Option<[u8; 16]>,
195
196    /// `isTransient` flag (transiently-replicated data). Serialised as the STATS
197    /// `isTransient` boolean. Preserved through compaction from compatible inputs
198    /// (issue #1021); a fresh flush leaves it `false`.
199    pub is_transient: bool,
200
201    /// `estimatedPartitionSize` EstimatedHistogram — one observation per
202    /// partition of its serialized Data.db size in bytes. Serialised as the FIRST
203    /// STATS field; the authoritative partition-count decode (`read_table_counts`,
204    /// issue #944) sums its bucket counts, so `Σ counts == partition_count`
205    /// (issue #1327). Populated by [`Self::record_partition`].
206    pub estimated_partition_size: EstimatedHistogram,
207
208    /// `estimatedCellPerPartitionCount` EstimatedHistogram — one observation per
209    /// partition of its cell count. Serialised as the SECOND STATS field.
210    /// Seeded with its OWN authoritative shape (118 offsets → 119 buckets), which
211    /// is DISTINCT from the partition-size histogram's 155/156 shape (issue #1327
212    /// finding 1; cassandra-5.0.0
213    /// `MetadataCollector.defaultCellPerPartitionCountHistogram` → `EstimatedHistogram(118)`).
214    /// Populated by [`Self::record_partition`].
215    pub estimated_cell_count: EstimatedHistogram,
216
217    /// An EXPLICIT `SerializationHeader.EncodingStats` delta-encoding baseline,
218    /// for the case where it is NOT the same triple as this SSTable's own
219    /// content minima (issue #4197).
220    ///
221    /// `None` — the overwhelmingly common case, and every write path's — means
222    /// "the baseline IS `min_timestamp`/`min_local_deletion_time`/`min_ttl`",
223    /// which is correct for a flush: Cassandra derives a flushed SSTable's
224    /// `EncodingStats` from the same memtable content the `StatsMetadata`
225    /// minima are collected over, so one triple serves both.
226    ///
227    /// `Some(..)` separates them, because in Cassandra they are TWO DIFFERENT
228    /// VALUES produced by two different mechanisms:
229    ///
230    /// * `EncodingStats` (SERIALIZATION_HEADER) is the baseline every row's
231    ///   timestamp/TTL/LDT VInt was delta-encoded against. At compaction it is
232    ///   MERGED FORWARD from the input SSTables' own headers
233    ///   (`SerializationHeader.make(metadata, sstables)` → `EncodingStats.merge`,
234    ///   `cassandra-5.0.8`), so it can sit strictly BELOW anything present in
235    ///   the output's own rows.
236    /// * `StatsMetadata.minTimestamp`/`minLocalDeletionTime`/`minTTL` (STATS)
237    ///   are folded from the cells and tombstones actually WRITTEN
238    ///   (`MetadataCollector`).
239    ///
240    /// `cqlite rebuild` is the first caller that must hold both at once: it
241    /// RECOVERS the baseline verbatim from the original `Statistics.db` (the
242    /// only authoritative record of what the unchanged `Data.db` was encoded
243    /// against) while RE-FOLDING the STATS minima from the rows it decodes. If
244    /// one field carried both, a compaction-produced input's regenerated
245    /// `Statistics.db` would claim the inherited baseline as its STATS minimum
246    /// — a value no row in the file has.
247    ///
248    /// Read ONLY by `build_serialization_header_component`, via
249    /// [`Self::effective_encoding_stats_baseline`]. The STATS body never reads
250    /// it.
251    pub encoding_stats_baseline: Option<EncodingStatsBaseline>,
252}
253
254/// The whole-SSTable `SerializationHeader.EncodingStats` triple — the
255/// delta-encoding baseline every row's timestamp/TTL/local-deletion-time VInt
256/// in `Data.db` is measured against (issue #4197).
257///
258/// Distinct from `StatsMetadata`'s minima of the same names; see
259/// [`StatisticsMetadata::encoding_stats_baseline`] for why Cassandra has two.
260/// `i64::MAX`/`i32::MAX` are the "nothing recorded" sentinels the header
261/// serializer normalises to Cassandra's own epoch/`LIVE` baselines, exactly as
262/// it does for the content minima.
263#[derive(Debug, Clone, Copy, PartialEq, Eq)]
264pub struct EncodingStatsBaseline {
265    /// `EncodingStats.minTimestamp` (microseconds).
266    pub min_timestamp: i64,
267    /// `EncodingStats.minLocalDeletionTime` (seconds).
268    pub min_local_deletion_time: i32,
269    /// `EncodingStats.minTTL` (seconds).
270    pub min_ttl: i32,
271}
272
273impl Default for StatisticsMetadata {
274    fn default() -> Self {
275        Self {
276            min_timestamp: i64::MAX,
277            max_timestamp: i64::MIN,
278            min_local_deletion_time: i32::MAX,
279            max_local_deletion_time: i32::MIN,
280            min_ttl: i32::MAX,
281            max_ttl: 0,
282            partition_count: 0,
283            row_count: 0,
284            column_count: 0,
285            total_rows_size: 0,
286            tombstone_histogram: TombstoneHistogram::new(),
287            first_key: None,
288            last_key: None,
289            has_partition_level_deletions: false,
290            repaired_at: 0,
291            pending_repair: None,
292            is_transient: false,
293            // Two DISTINCT Cassandra-canonical shapes (issue #1327 finding 1):
294            // partition-size seeds EH(155) (156 buckets), cell-count seeds
295            // EH(118) (119 buckets). cassandra-5.0.0 MetadataCollector
296            // defaultPartitionSizeHistogram / defaultCellPerPartitionCountHistogram.
297            estimated_partition_size: EstimatedHistogram::partition_size(),
298            estimated_cell_count: EstimatedHistogram::cell_per_partition_count(),
299            // `None` = "the EncodingStats baseline IS the content minima
300            // above", which is what every write path wants (see the field's
301            // doc comment).
302            encoding_stats_baseline: None,
303        }
304    }
305}
306
307impl StatisticsMetadata {
308    /// Create a new empty statistics metadata
309    pub fn new() -> Self {
310        Self::default()
311    }
312
313    /// The `SerializationHeader.EncodingStats` baseline to serialize: the
314    /// explicit [`Self::encoding_stats_baseline`] when a caller set one, else
315    /// this SSTable's own content minima (the write-path default — see that
316    /// field's doc comment for why the two can legitimately differ).
317    ///
318    /// Returns the RAW (pre-[`Self::finalize`]) values, sentinels included:
319    /// `build_serialization_header_component` does its own
320    /// `i64::MAX`/`i32::MAX` → epoch/`LIVE` normalisation, and the EncodingStats
321    /// baseline must match what `DataWriter` delta-encoded against, which is
322    /// also the raw value.
323    pub fn effective_encoding_stats_baseline(&self) -> EncodingStatsBaseline {
324        self.encoding_stats_baseline
325            .unwrap_or(EncodingStatsBaseline {
326                min_timestamp: self.min_timestamp,
327                min_local_deletion_time: self.min_local_deletion_time,
328                min_ttl: self.min_ttl,
329            })
330    }
331
332    /// Update timestamp range with a new timestamp value.
333    ///
334    /// LIVE deletion markers must never enter the min/max aggregates (issue
335    /// #851, Cassandra `d5bc7fb5`). Cassandra encodes `DeletionTime.LIVE` with
336    /// `markedForDeleteAt = Long.MIN_VALUE`, and a `NO_DELETION` / absent
337    /// liveness timestamp surfaces as `Long.MAX_VALUE`. Folding either sentinel
338    /// into `minTimestamp` / `maxTimestamp` would poison the stats, so they are
339    /// skipped here at the single aggregation chokepoint.
340    pub fn update_timestamp(&mut self, timestamp: i64) {
341        if Self::is_live_timestamp(timestamp) {
342            return;
343        }
344        self.min_timestamp = self.min_timestamp.min(timestamp);
345        self.max_timestamp = self.max_timestamp.max(timestamp);
346    }
347
348    /// Update local deletion time range (for tombstones) and the drop-time histogram.
349    ///
350    /// Call this for every tombstone local-deletion-time encountered while writing:
351    /// cell tombstones, row deletions, range tombstones, and partition tombstones.
352    ///
353    /// A LIVE marker (`DeletionTime.LIVE.localDeletionTime == Integer.MAX_VALUE`,
354    /// the same value as `Cell.NO_DELETION_TIME`) is not a tombstone: it must not
355    /// pull `minLocalDeletionTime` down to the sentinel nor inflate the tombstone
356    /// drop-time histogram (issue #851, Cassandra `d5bc7fb5`).
357    pub fn update_local_deletion_time(&mut self, deletion_time: i32) {
358        if Self::is_live_local_deletion_time(deletion_time) {
359            return;
360        }
361        self.min_local_deletion_time = self.min_local_deletion_time.min(deletion_time);
362        self.max_local_deletion_time = self.max_local_deletion_time.max(deletion_time);
363        self.tombstone_histogram.update(deletion_time);
364    }
365
366    /// Fold a LIVE (non-TTL) cell's `NO_DELETION_TIME` sentinel into the
367    /// `maxLocalDeletionTime` aggregate ONLY.
368    ///
369    /// Cassandra stamps every live, non-expiring cell with
370    /// `Cell.NO_DELETION_TIME` (`Integer.MAX_VALUE`) as its `localDeletionTime`,
371    /// and `MetadataCollector.update(Cell)` folds that value into
372    /// `maxLocalDeletionTime`. So any SSTable that contains at least one live
373    /// non-TTL cell finalizes `maxLocalDeletionTime == Integer.MAX_VALUE`,
374    /// EVEN when the same SSTable also carries older tombstones with a smaller
375    /// real LDT (a mixed SSTable — e.g. after compaction). Without this fold a
376    /// mixed SSTable would keep `maxLocalDeletionTime` at the largest TOMBSTONE
377    /// LDT and diverge from Cassandra (issue #1728).
378    ///
379    /// Unlike [`Self::update_local_deletion_time`], the live sentinel must NOT
380    /// pull `minLocalDeletionTime` down nor enter the tombstone drop-time
381    /// histogram (those describe real tombstones only; issue #851). This method
382    /// therefore raises the max alone. A pure-live SSTable ends at
383    /// `max_local_deletion_time == i32::MAX`, which serialises to the same
384    /// `NO_DELETION_TIME` sentinel as the `== 0` "no deletions" case, so the
385    /// pure-live output byte-parity is unchanged.
386    pub fn note_live_local_deletion_time(&mut self) {
387        // `i32::MAX` is the largest possible local-deletion-time, so folding it
388        // into the running max is simply an assignment (a real tombstone LDT is
389        // always strictly smaller). Written directly to avoid the no-op
390        // `.max(i32::MAX)` clippy::unnecessary_min_or_max lint.
391        self.max_local_deletion_time = i32::MAX;
392    }
393
394    /// True when `timestamp` is a LIVE / `NO_DELETION` marker rather than a real
395    /// deletion timestamp. CQLite writes `DeletionTime.LIVE` as `Long.MIN_VALUE`
396    /// (see `data_writer.rs`); Cassandra's `NO_DELETION` / `NO_TIMESTAMP` sentinel
397    /// is `Long.MAX_VALUE`. Both mean "no deletion" and must be excluded from
398    /// timestamp aggregation.
399    fn is_live_timestamp(timestamp: i64) -> bool {
400        timestamp == i64::MIN || timestamp == i64::MAX
401    }
402
403    /// True when `deletion_time` is a LIVE marker (`Integer.MAX_VALUE`) rather
404    /// than a real tombstone local-deletion-time.
405    fn is_live_local_deletion_time(deletion_time: i32) -> bool {
406        deletion_time == i32::MAX
407    }
408
409    /// Update TTL range
410    pub fn update_ttl(&mut self, ttl: i32) {
411        if ttl > 0 {
412            self.min_ttl = self.min_ttl.min(ttl);
413            self.max_ttl = self.max_ttl.max(ttl);
414        }
415    }
416
417    /// Increment partition count
418    pub fn increment_partition_count(&mut self) {
419        self.partition_count += 1;
420    }
421
422    /// Record one partition's serialized size and cell count into the
423    /// `estimatedPartitionSize` / `estimatedCellPerPartitionCount` histograms.
424    ///
425    /// Called once per partition written to Data.db with the exact serialized
426    /// partition byte length (`serialized_size`) and the number of cells emitted
427    /// (`cell_count`). Because it adds exactly one observation per partition, the
428    /// `estimatedPartitionSize` histogram's Σ bucket counts equals the SSTable's
429    /// partition count — the value the read-side authoritative decode
430    /// (`read_table_counts`, issue #944) reports. This mirrors Cassandra's
431    /// `MetadataCollector.addPartitionSizeInBytes` /
432    /// `addCellPerPartitionCount` (issue #1327).
433    pub fn record_partition(&mut self, serialized_size: u64, cell_count: u64) {
434        self.estimated_partition_size.add(serialized_size);
435        self.estimated_cell_count.add(cell_count);
436    }
437
438    /// Record a partition key in the SSTable key range.
439    ///
440    /// Partitions are written in ascending token order (enforced by
441    /// `SSTableWriter::write_partition`), so the FIRST key seen is the lowest and
442    /// the LAST is the highest. Used to populate the `da`-format
443    /// `StatsMetadata.firstKey`/`lastKey` (`hasKeyRange`). The clone is bounded by
444    /// the partition-key size (typically tens of bytes), not the partition data.
445    pub fn update_key_range(&mut self, key: &[u8]) {
446        if self.first_key.is_none() {
447            self.first_key = Some(key.to_vec());
448        }
449        self.last_key = Some(key.to_vec());
450    }
451
452    /// Record that a partition-level deletion (partition tombstone) was written.
453    ///
454    /// Once set, this stays `true` for the lifetime of the SSTable and drives the
455    /// `da`-format `StatsMetadata.hasPartitionLevelDeletions` marker. Called by
456    /// `SSTableWriter::write_partition` for any partition carrying a
457    /// `partition_tombstone`.
458    pub fn mark_partition_level_deletion(&mut self) {
459        self.has_partition_level_deletions = true;
460    }
461
462    /// Set the persisted repair state (`repairedAt`, `pendingRepair`,
463    /// `isTransient`) carried into the output STATS component.
464    ///
465    /// Used by the compaction merge path to preserve the repair metadata of
466    /// compatible inputs through to the merged output (issue #1021). A fresh
467    /// memtable flush never calls this, so the default unrepaired state
468    /// (`repaired_at = 0`, `pending_repair = None`, `is_transient = false`) is
469    /// retained.
470    pub fn set_repair_state(
471        &mut self,
472        repaired_at: i64,
473        pending_repair: Option<[u8; 16]>,
474        is_transient: bool,
475    ) {
476        self.repaired_at = repaired_at;
477        self.pending_repair = pending_repair;
478        self.is_transient = is_transient;
479    }
480
481    /// Increment row count
482    pub fn increment_row_count(&mut self) {
483        self.row_count += 1;
484    }
485
486    /// Add to column count
487    pub fn add_column_count(&mut self, count: u64) {
488        self.column_count += count;
489    }
490
491    /// Add to total rows size
492    pub fn add_rows_size(&mut self, size: u64) {
493        self.total_rows_size += size;
494    }
495
496    /// Finalize metadata before writing (normalize sentinel values)
497    pub fn finalize(&mut self) {
498        // If no timestamps were recorded, set to 0
499        if self.min_timestamp == i64::MAX {
500            self.min_timestamp = 0;
501        }
502        if self.max_timestamp == i64::MIN {
503            self.max_timestamp = 0;
504        }
505
506        // If no deletion times were recorded, set to 0
507        if self.min_local_deletion_time == i32::MAX {
508            self.min_local_deletion_time = 0;
509        }
510        if self.max_local_deletion_time == i32::MIN {
511            self.max_local_deletion_time = 0;
512        }
513
514        // If no TTLs were recorded, set min_ttl to 0
515        if self.min_ttl == i32::MAX {
516            self.min_ttl = 0;
517        }
518    }
519}
