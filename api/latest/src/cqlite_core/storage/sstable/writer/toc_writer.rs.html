1//! TOC.txt writer - writes table of contents
2//!
3//! Generates the TOC.txt component listing all SSTable files.
4//! Acts as a publication barrier - SSTable is not visible until TOC.txt exists.
5//!
6//! Critical requirements:
7//! - MUST be written LAST (after all other components)
8//! - Lists all component files
9//! - Plain text format, one component per line
10//! - Self-referential: TOC.txt must list itself
11

19/// Entry representing a component to be written to TOC.txt
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub struct ComponentEntry {
22    /// Component type (e.g., Data, Index, etc.)
23    pub component: SSTableComponent,
24}
25
26impl ComponentEntry {
27    /// Create a new component entry
28    pub fn new(component: SSTableComponent) -> Self {
29        Self { component }
30    }
31
32    /// Get the filename for this component
33    pub fn filename(&self) -> &'static str {
34        self.component.file_extension()
35    }
36}

38/// TOC.txt component writer
39///
40/// Writes the table of contents listing all SSTable component files.
41/// This file acts as the publication barrier - an SSTable is not considered
42/// complete and readable until TOC.txt exists.
43///
44/// # Format
45///
46/// Plain text file with one component filename per line:
47/// ```text
48/// CompressionInfo.db
49/// Data.db
50/// Digest.crc32
51/// Filter.db
52/// Index.db
53/// Statistics.db
54/// Summary.db
55/// TOC.txt
56/// ```
57///
58/// # Critical Requirements
59///
60/// 1. TOC.txt MUST be written LAST after all other components
61/// 2. TOC.txt MUST include itself in the listing
62/// 3. Component order is Cassandra 5.0.8's (sorted file names, see `write`); readers
63///    treat the file as a set
64///
65/// # Example
66///
67/// ```rust,ignore
68/// let mut components = vec![
69///     ComponentEntry::new(SSTableComponent::Data),
70///     ComponentEntry::new(SSTableComponent::Index),
71///     ComponentEntry::new(SSTableComponent::Statistics),
72///     ComponentEntry::new(SSTableComponent::Filter),
73///     ComponentEntry::new(SSTableComponent::Summary),
74///     ComponentEntry::new(SSTableComponent::CompressionInfo),
75///     ComponentEntry::new(SSTableComponent::Digest),
76/// ];
77///
78/// let writer = TocWriter::new(toc_path);
79/// writer.write(&components)?;
80/// ```
81#[derive(Debug)]
82pub struct TocWriter {
83    /// Path to the TOC.txt file to write
84    path: PathBuf,
85}
86
87impl TocWriter {
88    /// Create a new TOC.txt writer
89    ///
90    /// # Arguments
91    ///
92    /// * `path` - Full path to the TOC.txt file (e.g., "nb-1-big-TOC.txt")
93    pub fn new(path: PathBuf) -> Self {
94        Self { path }
95    }
96
97    /// Write TOC.txt file with the given component list
98    ///
99    /// Automatically adds TOC.txt itself to the listing if not already present.
100    /// Components are written in Cassandra 5.0.8's order: distinct file names in
101    /// case-sensitive `String` natural order (see the ORDER RULE in the body).
102    ///
103    /// # Arguments
104    ///
105    /// * `components` - List of components to include in TOC.txt
106    ///
107    /// # Returns
108    ///
109    /// Ok(()) on success, Error on I/O failure
110    ///
111    /// # Errors
112    ///
113    /// Returns Error::Storage if:
114    /// - Unable to create the TOC.txt file
115    /// - Unable to write component names
116    /// - Unable to flush/sync the file
117    pub fn write(&self, components: &[ComponentEntry]) -> Result<()> {
118        // Ensure TOC itself is in the list (add if missing)
119        let mut final_components: Vec<ComponentEntry> = components.to_vec();
120
121        if !final_components
122            .iter()
123            .any(|e| e.component == SSTableComponent::TOC)
124        {
125            final_components.push(ComponentEntry::new(SSTableComponent::TOC));
126        }
127
128        // ORDER RULE (#4486), pinned at `cassandra-5.0.8` (CASSANDRA-20494):
129        //   TOCComponent.updateTOC builds `new TreeSet<>(Collections2.transform(
130        //   components, Component::name))` and writes `new ArrayList<>(set)` with
131        //   FileUtils.write(..., CREATE, TRUNCATE_EXISTING, SYNC).
132        // i.e. the TOC lines are the DISTINCT component FILE-NAME strings (`Data.db`,
133        // `Digest.crc32`, `TOC.txt`, ...: the on-disk suffix after `<version>-<fmt>-`,
134        // no other normalisation) in `String.compareTo` natural order: UTF-16 code
135        // unit order, CASE-SENSITIVE (so `CRC.db` < `CompressionInfo.db` < `Data.db`),
136        // not "alphabetical, case-folded". For these ASCII names that equals Rust's
137        // `str` byte order, which is what `sort`/`dedup` on the file-name use here.
138        // Each line ends in `\n`. (5.0.2 wrote an arbitrary insertion order; every
139        // CQLite reader treats the TOC as a set, pinned by #4482 stage 1.)
140        final_components.sort_by(|a, b| a.filename().cmp(b.filename()));
141        final_components.dedup_by(|a, b| a.filename() == b.filename());
142
143        // Create file with buffered writing
144        let file = File::create(&self.path).map_err(|e| {
145            Error::storage(format!(
146                "Failed to create TOC.txt file {:?}: {}",
147                self.path, e
148            ))
149        })?;
150
151        let mut writer = BufWriter::new(file);
152
153        // Write each component filename on its own line
154        for entry in &final_components {
155            writeln!(writer, "{}", entry.filename()).map_err(|e| {
156                Error::storage(format!(
157                    "Failed to write component {} to TOC.txt: {}",
158                    entry.filename(),
159                    e
160                ))
161            })?;
162        }
163
164        // Flush buffer and sync to disk (durability requirement)
165        writer
166            .flush()
167            .map_err(|e| Error::storage(format!("Failed to flush TOC.txt: {}", e)))?;
168
169        let file = writer
170            .into_inner()
171            .map_err(|e| Error::storage(format!("Failed to extract file from buffer: {}", e)))?;
172
173        file.sync_all()
174            .map_err(|e| Error::storage(format!("Failed to sync TOC.txt to disk: {}", e)))?;
175
176        Ok(())
177    }
178}

186    #[test]
187    fn test_toc_writer_basic() {
188        let temp_dir = TempDir::new().unwrap();
189        let toc_path = temp_dir.path().join("nb-1-big-TOC.txt");
190
191        let components = vec![
192            ComponentEntry::new(SSTableComponent::Data),
193            ComponentEntry::new(SSTableComponent::Index),
194            ComponentEntry::new(SSTableComponent::Statistics),
195        ];
196
197        let writer = TocWriter::new(toc_path.clone());
198        writer.write(&components).unwrap();
199
200        // Verify file exists
201        assert!(toc_path.exists());
202
203        // Read and verify contents
204        let contents = fs::read_to_string(&toc_path).unwrap();
205        let lines: Vec<&str> = contents.lines().collect();
206
207        // Should include the 3 components plus TOC.txt itself
208        assert_eq!(lines.len(), 4);
209
210        // Verify TOC.txt is included
211        assert!(lines.contains(&"TOC.txt"));
212        assert!(lines.contains(&"Data.db"));
213        assert!(lines.contains(&"Index.db"));
214        assert!(lines.contains(&"Statistics.db"));
215    }

256    #[test]
257    fn test_toc_writer_order_is_case_sensitive_and_deduplicated() {
258        let temp_dir = TempDir::new().unwrap();
259        let toc_path = temp_dir.path().join("da-1-bti-TOC.txt");
260        // CRC.db sorts BEFORE CompressionInfo.db ('R' < 'o'), Rows.db after Partitions.db.
261        let components: Vec<_> = [
262            SSTableComponent::Rows,
263            SSTableComponent::Data,
264            SSTableComponent::Data,
265            SSTableComponent::CompressionInfo,
266            SSTableComponent::Partitions,
267            SSTableComponent::Crc,
268        ]
269        .into_iter()
270        .map(ComponentEntry::new)
271        .collect();
272        TocWriter::new(toc_path.clone()).write(&components).unwrap();
273        let contents = fs::read_to_string(&toc_path).unwrap();
274        assert_eq!(
275            contents,
276            "CRC.db\nCompressionInfo.db\nData.db\nPartitions.db\nRows.db\nTOC.txt\n"
277        );
278    }
