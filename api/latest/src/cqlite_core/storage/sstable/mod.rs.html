126/// The AUTHORITATIVE partition-key shape of a table, derived schema-less from the
127/// SSTable Statistics.db SerializationHeader (issue #1750).
128///
129/// Carries ONLY facts Cassandra actually serialises: how many partition-key
130/// components and clustering keys the table has, and the REAL names of the
131/// non-key (regular + static) columns. The partition-/clustering-key column NAMES
132/// are deliberately absent — Cassandra never serialises them, so any pk-name a
133/// schema-less parser reports is a synthesised placeholder that must not drive
134/// routing decisions (no-heuristics mandate #28).
135#[derive(Debug, Clone, PartialEq, Eq)]
136pub struct PartitionKeyShape {
137    /// Number of partition-key components (a single-component pk is `1`).
138    pub partition_key_count: usize,
139    /// Number of clustering-key columns.
140    pub clustering_key_count: usize,
141    /// The real names of the non-key (regular + static) columns.
142    pub non_key_column_names: std::collections::HashSet<String>,
143    /// For a single-component pk ONLY: the sole partition-key column's authoritative
144    /// CQL type (from the SerializationHeader `keyType`) and its header-carried name.
145    ///
146    /// The CQL type is authoritative — Cassandra serialises partition-key TYPES even
147    /// though it never serialises their NAMES (issue #1750, roborev 3784). It drives
148    /// the TYPED single-component key encoding so a schema-less `WHERE int_pk = 42`
149    /// builds the 4-byte `int` key Cassandra wrote, not an 8-byte `BigInt` key. The
150    /// `name` is the header's synthesised placeholder (`id` for uuid/timeuuid, else
151    /// `partition_key`) — NOT a trusted identity, only the label under which
152    /// `build_row_from_scan` reconstructs the pk column so the seeked row can be
153    /// re-validated against the predicate. `None` for a composite pk (the typed
154    /// single-component fast path does not apply) or when no reader exposed a
155    /// single-component pk type.
156    pub single_pk_component: Option<PartitionKeyComponent>,
157}

170/// Resolve the AUTHORITATIVE [`PartitionKeyShape`] by UNIONing the non-key column
171/// names across EVERY reader's SerializationHeader while requiring CONSISTENT key
172/// metadata (issue #1750, roborev 3786).
173///
174/// This is the pure core of [`SSTableManager::partition_key_shape`], factored out so
175/// the multi-reader union + consistency logic is unit-testable without SSTable
176/// binaries. `headers` is one `&[ColumnInfo]` per resolved reader for the table (an
177/// empty or non-header reader contributes nothing).
178///
179/// A multi-generation / schema-evolved table can add a REGULAR column in a LATER
180/// generation absent from an earlier generation's header. Using only the first
181/// header would omit that column from the non-key name set, so a
182/// `WHERE <later-gen-regular-col> = <val>` could be MISCLASSIFIED as a partition-key
183/// seek. To prevent that, the non-key names are UNIONed across all headers, and the
184/// partition-key count, clustering-key count, and single-component pk type+name must
185/// agree across every header that has partition-key columns.
186///
187/// The classifier proves a column IS the partition key by its ABSENCE from the unioned
188/// non-key set — which is sound ONLY if the union is COMPLETE, i.e. every reader's
189/// columns were visible. So this is fail-safe on incomplete metadata (roborev 3788):
190///
191/// Returns `None` when
192/// - there are no readers (fail-safe: caller full-scans), OR
193/// - ANY resolved reader lacks a usable partition-key shape — an empty/missing
194///   SerializationHeader, or a populated header exposing no partition-key column
195///   (we cannot see that reader's regular columns, so the union is incomplete and a
196///   column could be misclassified as the pk by absence), OR
197/// - the headers' key metadata is INCONSISTENT (pk-count / clustering-count / single-pk
198///   type+name disagreement — should never happen for one table, but if it did we
199///   cannot trust the shape).
200///
201/// `Some(shape)` is returned only when EVERY reader contributed a usable, consistent
202/// pk shape. All facts come from SerializationHeaders — no heuristics (#28).
203fn partition_key_shape_from_headers<'a>(
204    headers: impl IntoIterator<Item = &'a [crate::parser::header::ColumnInfo]>,
205) -> Option<PartitionKeyShape> {
206    // Key metadata the first header-bearing reader established; every subsequent
207    // header MUST match it or the whole shape is declined (inconsistent → None).
208    struct AgreedKeyShape {
209        partition_key_count: usize,
210        clustering_key_count: usize,
211        single_pk_component: Option<PartitionKeyComponent>,
212    }
213    let mut agreed: Option<AgreedKeyShape> = None;
214    let mut non_key_column_names = std::collections::HashSet::new();
215
216    for columns in headers {
217        // COMPLETE-UNION fail-safe (roborev 3788): the classifier proves a column is
218        // the partition key by its ABSENCE from `non_key_column_names`. That proof is
219        // only sound if we can see EVERY reader's columns. A reader with no usable
220        // SerializationHeader (empty columns) would silently drop its regular columns
221        // from the union, so a regular column present ONLY in that headerless reader
222        // could be misclassified as the pk. Decline the whole shape → the caller
223        // full-scans (always correct). No heuristics (#28).
224        if columns.is_empty() {
225            return None;
226        }
227        let mut partition_key_count = 0usize;
228        let mut clustering_key_count = 0usize;
229        // The sole partition-key component's authoritative CQL type + header-
230        // synthesised name (issue #1750, roborev 3784). Recorded for the FIRST pk
231        // column seen in THIS header; only surfaced when the pk is single-component.
232        // Cassandra serialises pk TYPES (authoritative) but not NAMES (`name` is a
233        // placeholder used only for pk reconstruction).
234        let mut first_pk_component: Option<PartitionKeyComponent> = None;
235        for col in columns {
236            match (col.is_primary_key, col.is_clustering) {
237                (true, false) => {
238                    if first_pk_component.is_none() {
239                        first_pk_component = Some(PartitionKeyComponent {
240                            name: col.name.clone(),
241                            cql_type: col.column_type.clone(),
242                        });
243                    }
244                    partition_key_count += 1;
245                }
246                (true, true) => clustering_key_count += 1,
247                (false, _) => {
248                    // Union non-key names across ALL readers (roborev 3786): a
249                    // regular column present only in a later generation must be
250                    // recognised, else it could be misread as the partition key.
251                    non_key_column_names.insert(col.name.clone());
252                }
253            }
254        }
255        // A SerializationHeader always names at least one partition-key component; a
256        // header that populated columns yet exposes NO partition-key column is not a
257        // usable pk shape. Under the complete-union fail-safe (roborev 3788) we cannot
258        // trust the union when any reader's pk shape is unknown, so decline → the
259        // caller full-scans rather than record a bogus all-zero shape.
260        if partition_key_count == 0 {
261            return None;
262        }
263        // The typed single-component pk type is authoritative ONLY for a single-
264        // component key; drop it for a composite pk (that fast path does not apply
265        // and a composite key isn't one raw value).
266        let single_pk_component = (partition_key_count == 1)
267            .then_some(first_pk_component)
268            .flatten();
269
270        match &agreed {
271            None => {
272                agreed = Some(AgreedKeyShape {
273                    partition_key_count,
274                    clustering_key_count,
275                    single_pk_component,
276                });
277            }
278            Some(prev) => {
279                // Require CONSISTENT key metadata across readers; any disagreement
280                // means we cannot trust the shape → decline (#28).
281                if prev.partition_key_count != partition_key_count
282                    || prev.clustering_key_count != clustering_key_count
283                    || prev.single_pk_component != single_pk_component
284                {
285                    return None;
286                }
287            }
288        }
289    }
290
291    agreed.map(|shape| PartitionKeyShape {
292        partition_key_count: shape.partition_key_count,
293        clustering_key_count: shape.clustering_key_count,
294        non_key_column_names,
295        single_pk_component: shape.single_pk_component,
296    })
297}

332/// Extract table name from SSTable directory path.
333///
334/// SSTable files are stored in directories named `<table_name>-<uuid>`.
335/// For example: `simple_table-6aa08200a25111f0a3fef1a551383fb9/na-1-big-Data.db`
336///
337/// This function extracts the table name portion by:
338/// 1. Getting the parent directory name
339/// 2. Splitting on '-' and removing the UUID suffix
340///
341/// Removes the UUID suffix from directory names like:
342/// - `simple_table-6aa08200a25111f0a3fef1a551383fb9` → `simple_table`
343/// - `my-test-table-UUID` → `my-test-table`
344///
345/// Returns `None` if the path doesn't contain a valid directory component.
346///
347/// Note: Table names can contain hyphens, so we need to be careful to only remove the UUID.
348/// UUIDs in Cassandra directory names are 32 hex chars without hyphens (e.g., 6aa08200a25111f0a3fef1a551383fb9).
349///
350/// Snapshot-aware (issue #2384): routes through the authoritative
351/// [`snapshot_path`] parser so a snapshot read
352/// (`.../{table}-{uuid}/snapshots/{tag}/...-Data.db`) keys the `SSTableManager`
353/// under the REAL table name, never the snapshot tag.
354pub(crate) fn extract_table_name(sstable_path: &Path) -> Option<String> {
355    snapshot_path::extract_table_name(sstable_path)
356}

358/// Extract the fully-qualified table key (`"keyspace.table"`) from an SSTable path.
359///
360/// Cassandra on-disk layout is: `<data_dir>/<keyspace>/<table>-<uuid>/<sstable_files>`
361///
362/// This function walks up two directory levels from the SSTable file to identify both the
363/// table directory (`parent`) and keyspace directory (`grandparent`), producing a
364/// `"keyspace.table"` key that uniquely identifies a table across keyspaces.
365///
366/// When datasets-v3 added `test_oa.simple_table` alongside the existing
367/// `test_basic.simple_table`, using the unqualified name `"simple_table"` as the
368/// `table_readers` key caused both tables' SSTables to be registered under the same
369/// entry, returning combined rows for any query.  This function is the authoritative
370/// source of table identity for `SSTableManager` (Issue #680).
371///
372/// # Returns
373///
374/// `Some((keyspace, table_name))` when both directory levels can be extracted;
375/// `None` if the path does not contain enough components (e.g., a flat test directory).
376///
377/// # Examples
378///
379/// ```text
380/// ".../sstables/test_basic/simple_table-6aa08200a25111f0a3fef1a551383fb9/nb-1-big-Data.db"
381///   → Some(("test_basic", "simple_table"))
382///
383/// ".../sstables/test_oa/simple_table-4b7cd05064e711f1bd3ac7dbf655c673/oa-2-big-Data.db"
384///   → Some(("test_oa", "simple_table"))
385///
386/// "nb-1-big-Data.db"   (flat, no parent dirs)
387///   → None
388/// ```
389///
390/// Snapshot-aware (issue #2384): both the keyspace and the table name are derived
391/// through the authoritative [`snapshot_path`] parser, so a snapshot read resolves
392/// the REAL `{keyspace}.{table}` for `SSTableManager` keying rather than
393/// `snapshots`/`{tag}`.
394pub fn extract_keyspace_and_table_name(sstable_path: &Path) -> Option<(String, String)> {
395    let table_name = snapshot_path::extract_table_name(sstable_path)?;
396    let keyspace = snapshot_path::extract_keyspace(sstable_path)?;
397    Some((keyspace, table_name))
398}

414/// Deterministic test gate that pauses a scan mid-flight so a test can observe
415/// whether the `table_readers` read guard is held across the scan's I/O (issue
416/// #1591). Armed per-manager (never process-globally) so parallel tests never
417/// interfere: only scans on the manager whose gate is armed ever wait. No
418/// wall-clock sleeps — the scan signals `reached` when it arrives and blocks on
419/// `release` until the test lets it proceed.
420#[cfg(test)]
421pub(crate) mod scan_gate {
422    use std::sync::Arc;
423    use tokio::sync::Notify;
424
425    /// Two one-shot rendezvous points shared between the paused scan and the test.
426    #[derive(Debug, Default)]
427    pub(crate) struct Gate {
428        /// Scan → test: "I have arrived at the gate."
429        pub reached: Notify,
430        /// Test → scan: "you may proceed."
431        pub release: Notify,
432    }
433
434    /// Called from inside `scan`'s per-reader loop. When a gate is armed on this
435    /// manager, signal arrival and block until the test releases it; otherwise a
436    /// no-op. Placed at the per-reader I/O so it sits INSIDE the read-guarded
437    /// region before the fix and OUTSIDE it after (the guard is dropped at the
438    /// snapshot boundary) — the exact seam the test asserts on.
439    pub(crate) async fn wait_if_armed(gate: &Option<Arc<Gate>>) {
440        if let Some(gate) = gate {
441            gate.reached.notify_one();
442            gate.release.notified().await;
443        }
444    }
445}

519/// same `config.memory.block_cache.enabled` read-cache toggle B2 established
520/// (issue #1568).
521///
522/// When enabled (the default) this returns a shared `Arc` clone of the PROCESS-GLOBAL
523/// [`GlobalKeyOffsetCache`](crate::storage::cache::GlobalKeyOffsetCache) — ONE

525/// footprint is bounded by a single fixed global cap

594    /// Recursively find all *-Data.db files up to `max_depth` levels deep
595    fn find_data_files<'a>(
596        platform: &'a Platform,
597        dir: &'a Path,
598        max_depth: usize,
599    ) -> std::pin::Pin<Box<dyn std::future::Future<Output = Result<Vec<PathBuf>>> + Send + 'a>>
600    {
601        let dir = dir.to_path_buf();
602        Box::pin(async move {
603            let mut results = Vec::new();
604
605            let mut dir_entries = match platform.fs().read_dir(&dir).await {
606                Ok(entries) => entries,
607                Err(_) => return Ok(results),
608            };
609
610            while let Some(entry) = dir_entries.next_entry().await? {
611                let path = entry.path();
612                if let Some(filename) = path.file_name().and_then(|n| n.to_str()) {
613                    // Skip macOS AppleDouble sidecars via is_apple_double_sidecar().
614                    // See Issue #481.
615                    if filename.ends_with("-Data.db") && !is_apple_double_sidecar(filename) {
616                        results.push(path);
617                    } else if max_depth > 0 {
618                        // Check if it's a directory and recurse
619                        if entry
620                            .file_type()
621                            .await
622                            .map(|ft| ft.is_dir())
623                            .unwrap_or(false)
624                        {
625                            let sub_results =
626                                Self::find_data_files(platform, &path, max_depth - 1).await?;
627                            results.extend(sub_results);
628                        }
629                    }
630                }
631            }
632
633            Ok(results)
634        })
635    }

661    /// Scan a range of keys from all SSTables for a table.
662    ///
663    /// # Arguments
664    /// * `table_id` - The table to scan
665    /// * `start_key` - Optional start key for range scan
666    /// * `end_key` - Optional end key for range scan
667    /// * `limit` - Optional limit on number of results
668    /// * `schema` - Optional table schema for schema-aware parsing. When provided,
669    ///   enables accurate type detection and avoids heuristic-based parsing.
670    ///   Strongly recommended for Cassandra 5.0+ formats.
671    ///
672    /// Cross-generation reconciliation (last-write-wins + tombstone shadowing) is
673    /// applied via the authoritative k-way merger when more than one SSTable
674    /// generation backs the table and `write-support` + a schema are available;
675    /// otherwise rows from each reader are concatenated. That concat fallback is
676    /// the documented multi-generation limitation (Issue #883) and is now
677    /// IDENTICAL across every feature build: the `tombstones` build takes exactly
678    /// this path too (it no longer runs its own partition-keyed merge). So no
679    /// build regresses relative to the default — a `tombstones`-without-
680    /// `write-support` multi-generation read behaves the same as the default
681    /// `not(tombstones)`-without-`write-support` build, and the prior `tombstones`
682    /// "merge" it replaces was the row-collapsing bug, not real reconciliation.
683    ///
684    /// Issue #1085: this is the SINGLE `scan` implementation for every feature
685    /// build. The former `#[cfg(feature = "tombstones")]` variant grouped per-row
686    /// results into a `HashMap` keyed on `RowKey` (which carries only the
687    /// partition-key bytes, no clustering) and ran `TombstoneMerger`, so it
688    /// collapsed all clustering rows of a partition into one — a full `SELECT *`
689    /// over a clustered table returned ~one row per partition. Concatenating
690    /// per-reader rows here (and reconciling only ACROSS generations) is correct
691    /// for clustered tables in every build.
692    ///
693    /// Lookup order (Issue #680):
694    ///   1. Exact match on the full `table_id` string (e.g. `"test_basic.simple_table"`)
695    ///   2. Unqualified table name (e.g. `"simple_table"`) — for backward compatibility
696    ///      with flat/non-Cassandra directory layouts that have no keyspace parent.
697    pub async fn scan(
698        &self,
699        table_id: &TableId,
700        start_key: Option<&RowKey>,
701        end_key: Option<&RowKey>,
702        limit: Option<usize>,
703        schema: Option<&crate::schema::TableSchema>,
704    ) -> Result<Vec<(RowKey, ScanRow)>> {
705        // ONE meter per materializing read OPERATION (issue #1701, roborev F1). This
706        // is the DOMINANT public read surface — the CLI SELECT path and most of the
707        // query executor call it — so leaving it unmetered left the four headline read
708        // metrics flat for almost every real read. Started HERE, at the PUBLIC entry
709        // (F4), so reader-lock + resolution latency is inside the reported duration,
710        // and its `Drop` reports every early return and `?` propagation with zero rows.
711        // FORMAT-AGNOSTIC: this scan spans a table's generations, which need not share
712        // an on-disk format, so no single `sstable.format` label would be honest.
713        self.scan_with_meter(
714            table_id,
715            start_key,
716            end_key,
717            limit,
718            schema,
719            ReadOpMeter::start(None),
720        )
721        .await
722    }

724    /// [`scan`](Self::scan) with the read operation's meter supplied by the CALLER
725    /// (issue #1701, roborev round 9).
726    ///
727    /// ONE meter per LOGICAL read, owned by the OUTERMOST read API. A targeted
728    /// single-partition read served by scanning and filtering — the `tombstones`
729    /// build's [`scan_partition`](Self::scan_partition), in
730    /// `manager_tombstones_partition_scan.rs` — must report the rows it DELIVERED,
731    /// not the whole table's, so it passes [`ReadOpMeter::inert`] here and meters its
732    /// own post-filter result instead. Public `scan` passes a started meter, so every
733    /// direct caller is metered exactly as before.
734    pub(in crate::storage::sstable) async fn scan_with_meter(
735        &self,
736        table_id: &TableId,
737        start_key: Option<&RowKey>,
738        end_key: Option<&RowKey>,
739        limit: Option<usize>,
740        schema: Option<&crate::schema::TableSchema>,
741        mut meter: ReadOpMeter,
742    ) -> Result<Vec<(RowKey, ScanRow)>> {
743        tracing::debug!("SSTableManager::scan - Scanning table_id='{}'", table_id);
744
745        // Issue #1591: snapshot the reader list and DROP the read guard before any
746        // I/O. Holding it across the whole scan let one queued writer FIFO-park
747        // every later point read behind the slowest in-flight scan.
748        let (reader_list, _fully_qualified_match) = self.resolve_reader_snapshot(table_id).await;
749
750        if reader_list.is_empty() {
751            tracing::debug!(
752                "SSTableManager::scan - No readers found for table '{}'",
753                table_id
754            );
755            return Ok(Vec::new());
756        }
757
758        tracing::debug!(
759            "SSTableManager::scan - Found {} readers for table '{}'",
760            reader_list.len(),
761            table_id
762        );
763
764        // Issue #883: when a table directory holds more than one SSTable
765        // generation, plain concatenation of each reader's live rows is wrong —
766        // it duplicates rows that exist in several generations and resurrects
767        // rows deleted in a later generation (each reader suppresses only its
768        // OWN tombstones). Reconcile across generations with the same
769        // last-write-wins + tombstone-shadowing rule compaction uses, reusing
770        // the authoritative k-way merger (write-support only; requires schema).
771        #[cfg(feature = "write-support")]
772        if reader_list.len() > 1 {
773            if let Some(schema) = schema {
774                match generation_merge::merge_generations_for_read(
775                    &reader_list,
776                    schema,
777                    start_key,
778                    end_key,
779                    limit,
780                    None,
781                )
782                .await
783                {
784                    Ok(merged) => {
785                        tracing::debug!(
786                            "SSTableManager::scan - cross-generation merge produced {} rows",
787                            merged.len()
788                        );
789                        meter.record_keys(merged.iter().map(|(k, ..)| k));
790                        return Ok(merged);
791                    }
792                    Err(e) => {
793                        // Never fail a read because the merge path hit an
794                        // unsupported format; fall back to concatenation.
795                        tracing::warn!(
796                            "SSTableManager::scan - cross-generation merge failed for '{}' ({}); \
797                             falling back to per-reader concatenation",
798                            table_id,
799                            e
800                        );
801                    }
802                }
803            }
804        }
805
806        // Each reader returns rows already in Cassandra token order; k-way
807        // merge the per-reader streams in token order (issue #1580) instead of
808        // concatenating + re-sorting by RAW key bytes (wrong global order +
809        // O(n log n) on the async worker). The merge is O(n log k) and
810        // early-exits once `limit` is satisfied.
811        let mut per_reader = Vec::with_capacity(reader_list.len());
812        for reader in &reader_list {
813            #[cfg(test)]
814            {
815                // Issue #1591: deterministic pause during the scan's I/O. On the
816                // fixed path the `table_readers` read guard is already dropped
817                // here (we operate on a cloned snapshot), so a test can prove no
818                // guard is held across the scan.
819                let gate = self.scan_gate.lock().ok().and_then(|g| g.clone());
820                scan_gate::wait_if_armed(&gate).await;
821            }
822            let results = reader
823                .scan(table_id, start_key, end_key, None, schema)
824                .await?;
825            per_reader.push(results);
826        }
827
828        let all_results = scan_merge::kway_merge_token_order(per_reader, limit);
829
830        tracing::debug!(
831            "SSTableManager::scan - Returning {} final results (token-ordered k-way merge)",
832            all_results.len()
833        );
834
835        meter.record_keys(all_results.iter().map(|(k, ..)| k));
836        Ok(all_results)
837    }

839    /// Partition-targeted scan: return only the rows for a single partition key,
840    /// touching only the SSTables whose bloom filter / BTI trie admit the key.
841    ///
842    /// This is the storage-layer fast path for a fully-constrained `WHERE pk = ?`
843    /// (Issue #949). Rather than scanning every SSTable for the table and filtering
844    /// in memory, it prunes the reader set with
845    /// [`might_contain_partition`](reader::SSTableReader::might_contain_partition)
846    /// — an O(1) bloom check for BIG format, an O(log n) trie walk for BTI — and
847    /// only parses the surviving candidates. On a table backed by thousands of
848    /// SSTables, a single-partition read drops from "open and scan all of them" to
849    /// "scan only the handful that can hold the key".
850    ///
851    /// Output matches filtering the full [`scan`](Self::scan) result down to
852    /// `partition_key`: the same per-reader parse and the same cross-generation
853    /// reconciliation run, just over the pruned candidate set. Concretely, with
854    /// more than one candidate generation this drives the authoritative k-way
855    /// merge (write-support, schema present); the single-candidate and concat
856    /// fallbacks behave exactly as the corresponding `scan` paths do — including
857    /// sharing `scan`'s known multi-generation concat limitation (Issue #883) when
858    /// the merge is unavailable. The caller still applies its own predicate
859    /// evaluation, so any over-inclusion (e.g. a BTI prefix-collision candidate) is
860    /// filtered out downstream.
861    ///
862    /// Gated on `not(tombstones)` because the bloom/BTI prune fast path it relies
863    /// on ([`scan_partition_clustering`](reader::SSTableReader::scan_partition_clustering))
864    /// is itself `not(tombstones)`-only. Under `tombstones` the executor falls back
865    /// to a full [`scan`](Self::scan) + predicate filter (since #1085, `scan` is the
866    /// same correct implementation in both builds, so the fallback is correct — just
867    /// without the single-partition prune).
868    ///
869    /// `partition_key` is the raw on-disk partition-key bytes produced by
870    /// [`encode_partition_key_columns`](crate::storage::partition_key_codec::encode_partition_key_columns),
871    /// which match the bytes the bloom filter, Index.db/BTI trie, and scan RowKeys
872    /// are keyed on.
873    ///
874    /// Within-SSTable seek (Issue #953): for the SINGLE-candidate case (the common
875    /// point-lookup path) this seeks directly to the partition's `Data.db` offset
876    /// — resolved via the BTI Partitions.db trie or the BIG `Index.db` — and
877    /// decodes ONLY that partition via
878    /// [`scan_single_partition_clustering`](reader::SSTableReader::scan_single_partition_clustering),
879    /// instead of full-parsing the candidate and retaining one partition. The
880    /// decode reuses the scan path's `parse_block_emit`, so its output is
881    /// byte-for-byte identical to `scan(...).retain(matches_key)`; when the offset
882    /// cannot be resolved authoritatively (no `Index.db` hit, or an unsupported
883    /// format) it falls back to the full scan + retain for that candidate. The
884    /// MULTI-candidate path is unchanged: it still reconciles via the k-way merge
885    /// (or the per-candidate concat fallback), so cross-generation LWW / tombstone
886    /// shadowing (#883) is preserved.
887    ///
888    /// Returns `(rows, engaged)`. On this build `engaged` is always `true`: the
889    /// underlying [`scan_partition_clustering`](Self::scan_partition_clustering)
890    /// prunes the SSTable set via `might_contain_partition` before decoding, so a
891    /// caller may honestly report a partition-targeted access path. The
892    /// `tombstones`-build counterpart returns `false` because it has no prune
893    /// (Epic #951, honest access paths).
894    #[cfg(not(feature = "tombstones"))]
895    pub async fn scan_partition(
896        &self,
897        table_id: &TableId,
898        partition_key: &[u8],
899        schema: Option<&crate::schema::TableSchema>,
900    ) -> Result<(Vec<(RowKey, ScanRow)>, bool)> {
901        // The clustering-aware path always prunes via the bloom/BTI candidate
902        // filter, so the partition-targeted access path is genuinely engaged
903        // regardless of whether the within-partition clustering seek narrowed.
904        let (rows, _clustering_engaged) = self
905            .scan_partition_clustering(table_id, partition_key, None, schema)
906            .await?;
907        Ok((rows, true))
908    }

910    /// Metadata-carrying partition-targeted scan (Issue #962, Epic #951).
911    ///
912    /// The WRITETIME/TTL-projection sibling of [`scan_partition`](Self::scan_partition):
913    /// it returns only the rows for the single partition identified by the raw
914    /// `partition_key` bytes, WITH per-cell write metadata
915    /// ([`CellWriteMetadata`] — write timestamp / TTL), while still PRUNING the
916    /// SSTable set down to the candidates whose bloom filter / BTI trie admit the
917    /// key. A `SELECT WRITETIME(col), TTL(col) ... WHERE pk = ?` therefore opens
918    /// only the handful of SSTables that can hold the partition, never all N — the
919    /// SSTable-level prune is the must-have that distinguishes this from the
920    /// full-table [`scan_with_cell_metadata`](Self::scan_with_cell_metadata).
921    ///
922    /// Output is identical to filtering `scan_with_cell_metadata(table, ..)` down to
923    /// `partition_key`: the same per-reader metadata decode and the same
924    /// cross-generation reconciliation run, just over the pruned candidate set, so
925    /// the caller's post-scan predicate evaluation is a pure correctness backstop
926    /// (it removes any bloom/BTI false-positive over-inclusion).
927    ///
928    /// Reconciliation mirrors `scan_partition`:
929    /// - More than one candidate generation (write-support + schema): drive the
930    ///   authoritative k-way merge via
931    ///   `generation_merge::merge_generations_for_read_with_metadata` TARGETED to
932    ///   just this partition. This preserves
933    ///   per-cell cross-generation LWW / tombstone shadowing for WRITETIME/TTL
934    ///   (Issue #885) on the targeted path.
935    /// - Otherwise: decode each candidate via the reader's metadata path and retain
936    ///   this partition's rows, concatenating across candidates.
937    ///
938    /// Within-SSTable decode currently full-decodes each surviving candidate's
939    /// metadata and retains the partition; the SSTable-level prune (avoiding the
940    /// full TABLE/SSTable scan) is the property #962 requires. A within-partition
941    /// metadata seek (bounding the decode to the partition's `Data.db` offset, as
942    /// `scan_single_partition_clustering` does for the plain path) is a documented
943    /// follow-up.
944    ///
945    /// Gated on `not(tombstones)` to match the `scan_partition` variant it parallels.
946    ///
947    /// Returns `(rows, engaged)`. On this build `engaged` is always `true`: the
948    /// candidate set is pruned via `might_contain_partition` before any decode, so
949    /// the partition-targeted metadata access path is genuinely engaged. The
950    /// `tombstones`-build counterpart returns `false` (no prune; full metadata
951    /// scan + retain) so the caller reports an honest fallback (Epic #951).
952    /// Prune a reader snapshot to the candidate generations that admit
953    /// `partition_key`, hoisting the BTI key hash+encoding to ONCE per read instead
954    /// of once per candidate (issue #1575 / C4).
955    ///
956    /// When ANY candidate is a BTI ("da") reader, the Murmur3-based byte-comparable
957    /// trie key is computed a SINGLE time and reused for every candidate's trie
958    /// prune (`might_contain_partition_encoded`); on `main` each candidate re-encoded
959    /// the same key, so an N-generation fan-out paid N identical Murmur3 hashes
960    /// (`KEY_HASH_CALLS == N`) where C4 pays 1. A BIG (`nb`) candidate has no BTI
961    /// encoding to hoist — its raw-key bloom check runs unchanged — so a non-BTI or
962    /// mixed candidate set stays correct. The pruning decision (and therefore the
963    /// resulting rows) is byte-identical to the per-candidate path.
964    /// Returns `(admitted, pruned)`: `admitted` are the SAME byte-for-byte
965    /// filtered candidates the original single-`Vec` `prune_candidates` returned;
966    /// `pruned` are every reader EXCLUDED by the presence-oracle check — each is,
967    /// by construction, a definitive-negative exclusion (the filter's ONLY
968    /// criterion), so `pruned` is exactly the set
969    /// [`verify_pruned_candidates`](Self::verify_pruned_candidates) needs to
970    /// authoritatively re-check (issue #2163's core silent-miss case: a false
971    /// negative HERE, at the multi-generation candidate prune, would otherwise
972    /// drop that SSTable from the read with no verification ever reached — the
973    /// per-reader `get_with_resolution` verify hook only covers a single reader's
974    /// OWN point read, never a candidate eliminated at this layer).
975    #[cfg(not(feature = "tombstones"))]
976    fn prune_candidates(
977        readers: &[Arc<reader::SSTableReader>],
978        partition_key: &[u8],
979    ) -> (
980        Vec<Arc<reader::SSTableReader>>,
981        Vec<Arc<reader::SSTableReader>>,
982    ) {
983        use crate::storage::sstable::bti::encode_partition_key_for_bti_trie;
984        // Encode once iff a BTI reader is present (BIG readers ignore `encoded`).
985        let encoded = readers
986            .iter()
987            .any(|r| r.is_bti())
988            .then(|| encode_partition_key_for_bti_trie(partition_key));
989        readers.iter().cloned().partition(|r| {
990            // BTI candidates prune via `might_contain_partition_encoded` (fed the
991            // hoisted C4 encoding), the SAME single emit site the raw-key BIG
992            // bloom prune below uses (issue #2163): the `Partitions.db` trie is
993            // the authoritative presence oracle, so a `false` (trie miss) is a
994            // definitive absent (drop) — recorded once as
995            // `cqlite.read.sstables_pruned{format=bti}` — and an `Err` (corrupt
996            // trie) is conservatively kept (not pruned, not recorded). This is
997            // congruent with the façade's resolution — a BTI trie result is
998            // authoritative. BIG candidates DELIBERATELY stay on the bloom-based
999            // `might_contain_partition`: a BIG `Index.db` miss is NOT a definitive
1000            // absent (#1572 truncated-index invariant), so pruning a BIG reader on
1001            // the façade's index probe would drop a candidate that actually holds
1002            // the partition. Only the bloom filter never yields a false negative.
1003            // Sharing ONE emit site (`might_contain_partition[_encoded]`) for both
1004            // formats means a caller that also `locate`s the same key afterward
1005            // (the actual read) never double-emits: `locate`/`locate_encoded`
1006            // themselves never call `emit_sstable_pruned`.
1007            match (r.is_bti(), &encoded) {
1008                (true, Some(enc)) => r.might_contain_partition_encoded(partition_key, enc),
1009                // Non-BTI reader (or, defensively, no encoding): bloom prune.
1010                _ => r.might_contain_partition(partition_key),
1011            }
1012        })
1013    }
1014
1015    /// Verify presence-oracle negatives for SSTables EXCLUDED by
1016    /// [`prune_candidates`](Self::prune_candidates) — issue #2163's core
1017    /// silent-miss case, roborev r6. A bloom/BTI-trie false negative during a
1018    /// MULTI-generation candidate prune would otherwise drop that SSTable from
1019    /// the read entirely, unverified: the per-reader `get_with_resolution` opt-in
1020    /// verify hook (`data_access/mod.rs`) never runs for a candidate eliminated
1021    /// one layer up, at THIS prune site — a candidate excluded here never reaches
1022    /// `get_with_resolution` at all for this read.
1023    ///
1024    /// STRICTLY inside the opt-in switch (default-off = zero extra work,
1025    /// unchanged behaviour): when
1026    /// [`presence_verification::enabled`](reader::presence_verification::enabled)
1027    /// is `false` (the default), this returns immediately without touching
1028    /// `pruned` at all. When enabled, each pruned reader is verified via
1029    /// [`SSTableReader::verify_presence_oracle_negative`] (an authoritative
1030    /// confirmation scan): a genuine contradiction increments
1031    /// `cqlite.read.bloom.false_negatives` EXACTLY ONCE per contradicted
1032    /// negative (the verify method's own single per-call emit — this loop calls
1033    /// it once per pruned reader, never more). Any `Err` fails OPEN (the
1034    /// caller's admitted `candidates` list is never touched) but is surfaced
1035    /// LOUDLY via the SAME loud-failure contract `get_with_resolution` uses
1036    /// (roborev r4): an error-level log with context, and a record through the
1037    /// EXISTING error-rate signal (`cqlite.errors.total{subsystem=reader}`,
1038    /// issue #1038) — never a new metric.
1039    #[cfg(not(feature = "tombstones"))]
1040    async fn verify_pruned_candidates(
1041        pruned: &[Arc<reader::SSTableReader>],
1042        table_id: &TableId,
1043        partition_key: &[u8],
1044    ) {
1045        if !reader::presence_verification::enabled() {
1046            return;
1047        }
1048        for r in pruned {
1049            if let Err(e) = r
1050                .verify_presence_oracle_negative(table_id, partition_key)
1051                .await
1052            {
1053                tracing::error!(
1054                    error = %e,
1055                    sstable_format = r.sstable_format_label(),
1056                    "opt-in presence-oracle false-negative verification scan FAILED for a \
1057                     prune_candidates-excluded SSTable — the read itself is unaffected \
1058                     (fail-open), but this soundness check could not run for this SSTable and \
1059                     needs investigation"
1060                );
1061                crate::observability::record_error(&e, "reader");
1062            }
1063        }
1064    }

1068        &self,
1069        table_id: &TableId,
1070        partition_key: &[u8],
1071        schema: Option<&crate::schema::TableSchema>,
1072    ) -> Result<(
1073        Vec<(RowKey, ScanRow, HashMap<String, CellWriteMetadata>)>,
1074        bool,
1075    )> {
1076        // ONE meter per materializing read OPERATION (issue #1701, roborev F1),
1077        // started at ENTRY (F4) so reader-lock + resolution latency is inside the
1078        // duration and `Drop` reports every early return with zero rows.
1079        // Format-agnostic: the operation spans a table's generations.
1080        let mut meter = ReadOpMeter::start(None);
1081
1082        // Issue #1591: snapshot the reader list and DROP the read guard before any
1083        // I/O (bloom/BTI prune, per-candidate decode, cross-generation merge).
1084        let (reader_list, _fully_qualified_match) = self.resolve_reader_snapshot(table_id).await;
1085        if reader_list.is_empty() {
1086            return Ok((Vec::new(), true));
1087        }
1088
1089        // Prune: keep only SSTables whose bloom filter / BTI trie admit the key.
1090        // This is the property #962 requires — only candidates are opened, never N.
1091        // C4 (#1575): the BTI key hash+encoding is hoisted to once per read here.
1092        let (candidates, pruned) = Self::prune_candidates(&reader_list, partition_key);
1093
1094        // Issue #2163 (roborev r6): opt-in verification of every EXCLUDED
1095        // candidate — a strict no-op unless the switch is on. Runs BEFORE the
1096        // `candidates.is_empty()` early-return below, since the core silent-miss
1097        // case is exactly "every admitted candidate came up empty because a
1098        // false negative wrongly pruned the ONE generation holding the key".
1099        Self::verify_pruned_candidates(&pruned, table_id, partition_key).await;
1100
1101        tracing::debug!(
1102            "SSTableManager::scan_partition_with_cell_metadata - {}/{} SSTables admit partition \
1103             key (len={}) for '{}'",
1104            candidates.len(),
1105            reader_list.len(),
1106            partition_key.len(),
1107            table_id
1108        );
1109
1110        if candidates.is_empty() {
1111            return Ok((Vec::new(), true));
1112        }
1113
1114        let matches_key = |entry: &(RowKey, ScanRow, HashMap<String, CellWriteMetadata>)| {
1115            entry.0.as_bytes() == partition_key
1116        };
1117
1118        // Multiple candidate generations may hold the same partition; reconcile
1119        // with the same authoritative metadata-aware k-way merge the full metadata
1120        // scan uses (write-support only, schema present), TARGETED to just this
1121        // partition (issue #1579): the merge keeps only `partition_key`'s rows and
1122        // stops as soon as it finds them, byte-identical to the former
1123        // full-merge-then-`retain(matches_key)` but without materializing every
1124        // other partition. This preserves per-cell cross-generation WRITETIME/TTL.
1125        #[cfg(feature = "write-support")]
1126        if candidates.len() > 1 {
1127            if let Some(schema) = schema {
1128                let target = RowKey::new(partition_key.to_vec());

1130                    &candidates,
1131                    schema,
1132                    None,
1133                    None,
1134                    None,
1135                    Some(&target),
1136                )
1137                .await
1138                {
1139                    Ok(merged) => {
1140                        // Work-counter gate (Issue #958): the merge parsed every
1141                        // surviving candidate; `merged` (already the target partition
1142                        // only) is exactly the partitions this lookup returns.
1143                        work_counters::add_sstables_scanned(candidates.len() as u64);
1144                        work_counters::add_partitions_parsed(merged.len() as u64);
1145                        meter.record_keys(merged.iter().map(|(k, ..)| k));
1146                        return Ok((merged, true));
1147                    }
1148                    Err(e) => {
1149                        tracing::warn!(
1150                            "SSTableManager::scan_partition_with_cell_metadata - cross-generation \
1151                             metadata merge failed for '{}' ({}); falling back to per-reader \
1152                             concatenation",
1153                            table_id,
1154                            e
1155                        );
1156                    }
1157                }
1158            }
1159        }
1160
1161        // Single candidate (common case) or the multi-candidate concat fallback:
1162        // decode each candidate's metadata and retain this partition's rows.
1163        let mut all_results = Vec::new();
1164        for reader in &candidates {
1165            // Work-counter gate (Issue #958): one real Data.db touch per surviving
1166            // candidate. Counted here (not at prune time) so the counter reflects
1167            // SSTables actually opened/scanned.
1168            work_counters::add_sstables_scanned(1);
1169
1170            let mut results = reader
1171                .scan_with_cell_metadata(table_id, None, None, None, schema)
1172                .await?;
1173            results.retain(matches_key);
1174            all_results.append(&mut results);
1175        }
1176        // #1917: rows here already share `partition_key` (prior retain) so this is a no-op today; kept for future multi-partition safety + parity with the IN fan-out / streaming scan's token order (raw-byte = #1580 wrong-order).
1177        if candidates.len() > 1 {
1178            all_results.sort_by(|a, b| cmp_partition_keys_by_token(a.0.as_bytes(), b.0.as_bytes()));
1179        }
1180        work_counters::add_partitions_parsed(all_results.len() as u64);
1181        meter.record_keys(all_results.iter().map(|(k, ..)| k));
1182        Ok((all_results, true))
1183    }

1185    /// Clustering-slice-aware partition-targeted scan (Issue #954, Epic #951).
1186    ///
1187    /// Identical to [`scan_partition`](Self::scan_partition) but, when `clustering`
1188    /// is `Some(slice)` AND exactly one candidate SSTable admits the key AND that
1189    /// candidate's single-partition seek can use its authoritative row index, the
1190    /// within-partition decode is bounded to the row-index block(s) covering the
1191    /// requested clustering range — so a `WHERE pk = ? AND ck </>/= ?` slice over a
1192    /// wide partition decodes O(matched rows + index block), not the whole
1193    /// partition.
1194    ///
1195    /// Returns `(rows, clustering_seek_engaged)`. `clustering_seek_engaged` is
1196    /// `true` only when the within-partition clustering narrowing actually bounded
1197    /// the decode (so the caller may report
1198    /// [`AccessPath::ClusteringSlice`](crate::query::access_path::AccessPath::ClusteringSlice));
1199    /// it is `false` for the multi-candidate / merge / full-decode fallbacks,
1200    /// which still return correct rows for the honest `PartitionLookup` path. The
1201    /// rows are ALWAYS the full partition (or its clustering-narrowed superset):
1202    /// the caller's post-scan `evaluate_leaf` applies the exact clustering bound,
1203    /// so output is byte-identical regardless of whether the seek engaged.
1204    #[cfg(not(feature = "tombstones"))]
1205    pub async fn scan_partition_clustering(
1206        &self,

1209        clustering: Option<&reader::ClusteringSlice>,
1210        schema: Option<&crate::schema::TableSchema>,
1211    ) -> Result<(Vec<(RowKey, ScanRow)>, bool)> {
1212        // Issue #1591: snapshot the reader list + the authoritative
1213        // `fully_qualified_match` signal and DROP the read guard before any I/O.
1214        //
1215        // `fully_qualified_match`: did resolution match the FULLY-QUALIFIED
1216        // `keyspace.table` key exactly, or fall back to the bare table name? An
1217        // unqualified query is treated as an exact match (no keyspace to mismatch).
1218        // This authoritative signal gates the seek's table-consistency guard: only
1219        // an exact FQ match may relax to a name-only check across a header-keyspace
1220        // divergence; a fully-qualified query resolved via the bare-name fallback
1221        // keeps strict keyspace matching so it never returns another keyspace's
1222        // same-named rows (#1284 review).
1223        // ONE meter per materializing read OPERATION (issue #1701, roborev F1),
1224        // started at ENTRY (F4) so reader-lock + resolution latency is inside the
1225        // duration and `Drop` reports every early return with zero rows.
1226        // Format-agnostic: the operation spans a table's generations.
1227        let mut meter = ReadOpMeter::start(None);
1228
1229        let (reader_list, fully_qualified_match) = self.resolve_reader_snapshot(table_id).await;
1230        if reader_list.is_empty() {
1231            return Ok((Vec::new(), false));
1232        }
1233
1234        // Prune: keep only SSTables whose bloom filter / BTI trie admit the key.
1235        // C4 (#1575): the BTI key hash+encoding is hoisted to once per read here.
1236        let (candidates, pruned) = Self::prune_candidates(&reader_list, partition_key);
1237
1238        // Issue #2163 (roborev r6): opt-in verification of every EXCLUDED
1239        // candidate — a strict no-op unless the switch is on. See
1240        // `verify_pruned_candidates` for the fail-open / loud-failure contract.
1241        Self::verify_pruned_candidates(&pruned, table_id, partition_key).await;
1242
1243        tracing::debug!(
1244            "SSTableManager::scan_partition - {}/{} SSTables admit partition key (len={}) for '{}'",
1245            candidates.len(),
1246            reader_list.len(),
1247            partition_key.len(),
1248            table_id
1249        );
1250
1251        if candidates.is_empty() {
1252            return Ok((Vec::new(), false));
1253        }
1254
1255        let matches_key = |entry: &(RowKey, ScanRow)| entry.0.as_bytes() == partition_key;
1256
1257        // Multiple candidate generations may hold the same partition; reconcile
1258        // with the same authoritative k-way merge the full scan uses (write-support
1259        // only), TARGETED to just this partition. Issue #2096: SEEK each candidate
1260        // directly to the target partition's `Data.db` offset (BTI trie / Index.db)
1261        // and reconcile through the partition-seeking merger (issues #2207/#2346)
1262        // instead of the full-scan `KWayMerger::new`, which decoded every partition
1263        // with token <= the target before reaching it (O(partitions-below-target)).
1264        // The seeking merge reconciles through the SAME `KWayMerger`
1265        // (`from_row_iterators`), so its output is byte-identical to the former
1266        // full-merge-then-`retain(matches_key)`, only over O(target) work.
1267        #[cfg(all(feature = "write-support", not(feature = "tombstones")))]
1268        if candidates.len() > 1 {
1269            if let Some(schema) = schema {
1270                let target = RowKey::new(partition_key.to_vec());
1271                match generation_merge::seek_merge_generations_for_read(
1272                    &candidates,
1273                    schema,
1274                    &target,
1275                )
1276                .await
1277                {
1278                    Ok(merged) => {
1279                        // Work-counter gate (Issue #958): the k-way merge parsed
1280                        // every surviving candidate, and `merged` (already the target
1281                        // partition only) is exactly the partitions this lookup returns.
1282                        work_counters::add_sstables_scanned(candidates.len() as u64);
1283                        work_counters::add_partitions_parsed(merged.len() as u64);
1284                        // The cross-generation merge decodes full partitions; the
1285                        // clustering seek does not engage here (#954). Correct rows
1286                        // via the post-scan backstop; honest non-engaged signal.
1287                        meter.record_keys(merged.iter().map(|(k, ..)| k));
1288                        return Ok((merged, false));
1289                    }
1290                    Err(e) => {
1291                        tracing::warn!(
1292                            "SSTableManager::scan_partition - cross-generation merge failed for \
1293                             '{}' ({}); falling back to per-reader concatenation",
1294                            table_id,
1295                            e
1296                        );
1297                    }
1298                }
1299            }
1300        }
1301
1302        // Single candidate (the common case): SEEK directly to the partition's
1303        // Data.db offset and decode ONLY that partition (Issue #953), instead of a
1304        // full parse-then-retain. The seek resolves the offset via the BTI trie /
1305        // Index.db and runs the same per-partition decode the scan path uses, so
1306        // its output is byte-for-byte identical to `scan(...).retain(matches_key)`.
1307        // If the seek is not applicable for this reader (no authoritative offset,
1308        // or an unsupported format), it returns `Ok(None)` and we FALL BACK to the
1309        // full scan + retain for that candidate (Constraint #4: correctness over
1310        // optimization). The multi-candidate concat fallback below is unchanged —
1311        // only the single-candidate path gets the seek.
1312        let mut all_results = Vec::new();
1313        let mut clustering_engaged = false;
1314        for reader in &candidates {
1315            // Work-counter gate (Issue #958): one real Data.db touch per surviving
1316            // candidate. Counted here (not at prune time) so the counter reflects
1317            // SSTables actually opened/scanned, the cost a regression would balloon.
1318            work_counters::add_sstables_scanned(1);
1319
1320            let mut results = if candidates.len() == 1 {
1321                // Issue #954: thread the clustering slice into the seek so it can
1322                // narrow the within-partition decode via the authoritative row
1323                // index. `engaged` records whether the clustering narrowing
1324                // actually bounded the decode (vs a full-partition decode).
1325                match reader
1326                    .scan_single_partition_clustering(
1327                        table_id,
1328                        partition_key,
1329                        clustering,
1330                        fully_qualified_match,
1331                        schema,
1332                    )
1333                    .await
1334                {
1335                    // Seek resolved authoritatively: use its rows directly. They
1336                    // already match exactly this partition's key, so no retain.
1337                    Ok(Some((rows, engaged))) => {
1338                        clustering_engaged = engaged;
1339                        rows
1340                    }
1341                    // Seek not applicable (Constraint #4): full scan + retain.
1342                    Ok(None) => {
1343                        let mut r = reader.scan(table_id, None, None, None, schema).await?;
1344                        r.retain(matches_key);
1345                        r
1346                    }
1347                    Err(e) => return Err(e),
1348                }
1349            } else {
1350                // Multi-candidate concat fallback (merge unavailable): preserve the
1351                // existing full-scan + retain behaviour per candidate (Constraint #2).
1352                let mut r = reader.scan(table_id, None, None, None, schema).await?;
1353                r.retain(matches_key);
1354                r
1355            };
1356            all_results.append(&mut results);
1357        }
1358        // #1917: rows here already share `partition_key` (prior retain) so this is a no-op today; kept for future multi-partition safety + parity with the IN fan-out / streaming scan's token order (raw-byte = #1580 wrong-order).
1359        if candidates.len() > 1 {
1360            all_results.sort_by(|a, b| cmp_partition_keys_by_token(a.0.as_bytes(), b.0.as_bytes()));
1361        }
1362        work_counters::add_partitions_parsed(all_results.len() as u64);
1363        meter.record_keys(all_results.iter().map(|(k, ..)| k));
1364        Ok((all_results, clustering_engaged))
1365    }

1452    /// the metadata-path twin of [`scan_with_meter`](Self::scan_with_meter), for the
1453    /// same reason: the `tombstones` build's targeted
1454    /// [`scan_partition_with_cell_metadata`](Self::scan_partition_with_cell_metadata)
1455    /// filters this result down to one partition and must report what it DELIVERED.
1456    pub(in crate::storage::sstable) async fn scan_with_cell_metadata_with_meter(
1457        &self,
1458        table_id: &TableId,
1459        start_key: Option<&RowKey>,
1460        end_key: Option<&RowKey>,
1461        limit: Option<usize>,
1462        schema: Option<&crate::schema::TableSchema>,
1463        mut meter: ReadOpMeter,

1466        // I/O (per-reader metadata decode and cross-generation merge).
1467        let (reader_list, _fully_qualified_match) = self.resolve_reader_snapshot(table_id).await;
1468        if reader_list.is_empty() {
1469            return Ok(Vec::new());
1470        }
1471
1472        // Issue #885: the metadata path (WRITETIME/TTL projection) must
1473        // reconcile across SSTable generations exactly like the plain `scan`
1474        // path (#883) — otherwise a multi-generation directory returns
1475        // duplicate rows and resurrects rows/cells deleted in a later
1476        // generation. Drive the same authoritative k-way merger, then surface
1477        // the WINNING cell's per-cell write timestamp / TTL (write-support
1478        // only; requires schema). Single-generation reads skip this entirely.
1479        #[cfg(feature = "write-support")]
1480        if reader_list.len() > 1 {
1481            if let Some(schema) = schema {
1482                match generation_merge::merge_generations_for_read_with_metadata(
1483                    &reader_list,
1484                    schema,
1485                    start_key,
1486                    end_key,
1487                    limit,
1488                    None,
1489                )
1490                .await
1491                {
1492                    Ok(merged) => {
1493                        meter.record_keys(merged.iter().map(|(k, ..)| k));
1494                        return Ok(merged);
1495                    }
1496                    Err(e) => {
1497                        // Never fail a read because the merge path hit an
1498                        // unsupported format; fall back to concatenation.
1499                        tracing::warn!(
1500                            "SSTableManager::scan_with_cell_metadata - cross-generation merge \
1501                             failed for '{}' ({}); falling back to per-reader concatenation",
1502                            table_id,
1503                            e
1504                        );
1505                    }
1506                }
1507            }
1508        }
1509
1510        // K-way merge the per-reader token-ordered streams (issue #1580),
1511        // mirroring `scan`: the metadata payload rides alongside the row.
1512        // Merging in token order (not raw key bytes) matches Cassandra's
1513        // cross-SSTable order and avoids the full O(n log n) re-sort.
1514        let mut per_reader = Vec::with_capacity(reader_list.len());
1515        for reader in &reader_list {
1516            let results = reader
1517                .scan_with_cell_metadata(table_id, start_key, end_key, None, schema)
1518                .await?;

1520                results

1537    /// Snapshot the resolved reader set for `table_id` and DROP the
1538    /// `table_readers` read guard before returning (issue #1591).
1539    ///
1540    /// Returns the cloned `Arc<SSTableReader>` handles serving the table plus the
1541    /// authoritative `fully_qualified_match` signal (see
1542    /// [`fully_qualified_match`](Self::fully_qualified_match)), acquiring the read
1543    /// guard only long enough to clone the small `Vec` of `Arc`s.
1544    ///
1545    /// # Why (tail-latency fix)
1546    ///
1547    /// `tokio::sync::RwLock` is FIFO-fair: a single queued writer (reader reload,
1548    /// schema set, generation removal) parks EVERY later-arriving reader behind
1549    /// the longest in-flight read guard. Holding the guard across a whole
1550    /// multi-reader scan therefore made one slow scan plus one admin write stall
1551    /// every subsequent point read — bimodal tail latency. Cloning the `Arc` list
1552    /// and releasing the guard immediately means scans and point reads run without
1553    /// holding the lock across their I/O, so a queued writer can never park them.
1554    ///
1555    /// # Semantics (unchanged)
1556    ///
1557    /// A scan operating on this snapshot may miss a reader added to the map AFTER
1558    /// the snapshot was taken. That is the SAME semantics as holding the guard:
1559    /// the guard would have blocked the writer until the scan finished, so the
1560    /// scan never observed the new reader either. Readers removed from the map
1561    /// mid-scan stay alive for the snapshot holder — that is precisely what the
1562    /// `Arc` clone guarantees.
1563    pub(crate) async fn resolve_reader_snapshot(
1564        &self,
1565        table_id: &TableId,
1566    ) -> (Vec<Arc<reader::SSTableReader>>, bool) {
1567        let table_readers = self.table_readers.read().await;
1568        let table_name = table_id.name();
1569        let fully_qualified_match = Self::fully_qualified_match(&table_readers, table_name);
1570        let readers = Self::resolve_reader_list(&table_readers, table_name)
1571            .cloned()
1572            .unwrap_or_default();
1573        // Guard dropped here, before any reader I/O.
1574        (readers, fully_qualified_match)
1575    }

1578    /// SSTable readers' Statistics.db SerializationHeader (issue #1750).
1579    ///
1580    /// This is the metadata a schema-less reader DOES have without a CQL schema:
1581    /// the SerializationHeader (surfaced on each reader's `header().columns`)
1582    /// records, per column, the authoritative `is_primary_key` / `is_clustering`
1583    /// flags and the REAL names of the regular + static (non-key) columns. It does
1584    /// NOT record the partition-/clustering-key column NAMES — Cassandra never
1585    /// serialises those (they live in `system_schema`, absent schema-less), so the
1586    /// parser synthesises a placeholder pk name (`build_partition_key_columns`) that
1587    /// MUST NOT be trusted as an identity.
1588    ///
1589    /// Returns [`PartitionKeyShape`] carrying ONLY authoritative facts: the count of
1590    /// partition-key components, the count of clustering keys, and the set of real
1591    /// non-key column names. The caller confirms a predicate column is the sole
1592    /// partition key BY ELIMINATION (single pk component, zero clustering keys, and
1593    /// the column absent from the non-key name set) — never by the synthesised name.
1594    ///
1595    /// The shape is resolved across ALL readers serving the table, not just the
1596    /// first (issue #1750, roborev 3786). A multi-generation / schema-evolved table
1597    /// can add a REGULAR column in a LATER generation that is absent from an earlier

1601    ///   * the non-key column names are UNIONed across every reader's header, and
1602    ///   * the partition-key count, clustering-key count, and the single-component
1603    ///     pk type+name must be CONSISTENT across all readers with a header.
1604    ///
1605    /// Returns `None` when no reader for the table exposes a SerializationHeader OR
1606    /// when the readers' key metadata is INCONSISTENT (a pk-count / clustering-count

1609    /// caller then keeps the honest full-scan path (all from SerializationHeaders,
1610    /// no heuristics — #28).

1614    }

1675    ///   branch is LAZY since #1579 (streams via

1681        _table_id: &TableId,

1720        // materializes regardless of generation count.

1724    /// Streaming scan (issue #790): merge per-SSTable streams lazily into a
1725    /// bounded output channel, in key (token) order, without materializing the
1726    /// whole result.
1727    ///
1728    /// Each reader yields entries already in token order; a k-way merge over the
1729    /// per-reader heads produces globally ordered output while holding only one
1730    /// pending entry per SSTable. Live heap is bounded by `buffer_size` plus the
1731    /// number of SSTables, independent of total row count — the streaming analog
1732    /// of the materializing [`scan`](Self::scan) (concat + stable sort by key).
1733    ///
1734    /// # Multi-generation correctness (Issue #957)
1735    ///
1736    /// The lazy per-reader k-way merge above is only the streaming analog of
1737    /// `scan`'s **concat + sort** path, which is correct for a single generation.
1738    /// When a table directory holds more than one SSTable generation, the same
1739    /// `(partition, clustering)` row can live in several generations and a
1740    /// row/cell tombstone in a newer generation suppresses only its own
1741    /// generation's copy — so a pure key-ordered merge would emit overwritten
1742    /// rows twice and resurrect rows deleted in a later generation. `scan` avoids
1743    /// this by routing the multi-generation case through the authoritative
1744    /// `KWayMerger` (the same LWW + tombstone-shadowing k-way merge compaction
1745    /// uses); this streaming path must reconcile identically or `execute()` and
1746    /// `execute_streaming()` diverge.
1747    ///
1748    /// # Streaming the multi-generation merge (Issue #1579 / D3)
1749    ///
1750    /// The multi-generation case runs
1751    /// [`generation_merge::stream_generations_for_read`], which drives that same
1752    /// `KWayMerger` on a blocking task and feeds each stepped partition's live
1753    /// rows STRAIGHT into the bounded channel via `blocking_send` (backpressure
1754    /// preserved) — instead of collecting the ENTIRE reconciled table, sorting it,
1755    /// and only then dribbling it (the pre-D3 behaviour). Live heap is therefore
1756    /// O(one partition + channel), and time-to-first-row is O(first partition),
1757    /// not O(full merge). The merger already emits partitions in `(token, key)`
1758    /// order, byte-identical to `scan`'s `sort_by_token_order`, so the emitted
1759    /// order is unchanged (issue #1579 ordering guardrail). A merger CONSTRUCTION
1760    /// failure now PROPAGATES (issue #3154) — the only two it can report,
1761    /// `Error::Schema` and a thread-spawn `Error::Storage`, used to be answered with
1762    /// the concat, so this diverges from `scan` there (follow-up #3170) — as does a
1763    /// merge producer that DIED, because the concat is not reconciling and answering a
1764    /// panic with it returns full-length WRONG data (issue #3124). The
1765    /// single-generation / no-schema / no-`write-support` cases keep the lazy
1766    /// streaming merge, which already matches `scan`'s concat path exactly and
1767    /// preserves LIMIT/backpressure.
1768    #[cfg(not(feature = "tombstones"))]
1769    pub async fn scan_stream(
1770        &self,
1771        table_id: &TableId,
1772        start_key: Option<&RowKey>,
1773        end_key: Option<&RowKey>,
1774        schema: Option<&crate::schema::TableSchema>,
1775        buffer_size: usize,
1776    ) -> Result<reader::RowScanStream> {
1777        let readers = self.resolve_table_readers(table_id).await;
1778
1779        // Issue #957: keep the materializing `scan` and this streaming path in lockstep
1780        // ON THE SUCCESS PATH. Reuse the EXACT guard `scan` uses for cross-generation
1781        // reconciliation (`reader_list.len() > 1 && schema present`, write-support only)
1782        // and the same merger, then forward the reconciled rows through the streaming
1783        // channel. Without this, a partition spread across generations duplicates
1784        // overwritten rows and resurrects deleted ones in the stream while `scan`
1785        // returns the merged, deduplicated, tombstone-honouring result. They are NO
1786        // LONGER in lockstep on the ERROR path (issue #3154): a schema whose
1787        // `dropped_columns` fails `validate_dropped_columns` (`schema/mod.rs:611`, from
1788        // `write_engine/merge/mod.rs:1728`) makes THIS path `Err` while `scan` still
1789        // concatenates and returns rows — deliberate; follow-up #3170 owns `scan`.
1790        #[cfg(feature = "write-support")]
1791        if readers.len() > 1 {
1792            // No schema: cross-generation LWW/tombstone reconciliation needs the
1793            // schema to drive the KWayMerger, so this deliberately falls through to
1794            // the lazy per-reader token-merge below — matching `scan`'s identical
1795            // no-schema fallback (both accept the documented Issue #883 concat
1796            // limitation rather than reconciling without a schema).
1797            if let Some(schema) = schema {
1798                match generation_merge::stream_generations_for_read(
1799                    &readers,
1800                    schema,
1801                    start_key,
1802                    end_key,
1803                    buffer_size,
1804                )
1805                .await
1806                {
1807                    Ok(rx) => {
1808                        tracing::debug!(
1809                            "SSTableManager::scan_stream - cross-generation merge streaming \
1810                             (O(window), not materialized)"
1811                        );
1812                        return Ok(rx);
1813                    }
1814                    // Gated on the TYPE, never the message (issues #3124/#3154, roborev):
1815                    // `fallback_eligible()` is true for the merger-INELIGIBLE input ALONE;
1816                    // every other setup failure propagates, because answering one with the
1817                    // concat returned a FULL-LENGTH UNRECONCILED result set under `Ok`.
1818                    // This arm is DEFENSIVE and DEAD IN PRODUCTION: `KWayMerger::new`
1819                    // cannot report an unsupported format/version — each input's
1820                    // `SSTableReader::open`, with every format/version gate, runs in the
1821                    // producer thread it spawns (`merge/producer_iter.rs:385-388`) and
1822                    // surfaces mid-stream at `step()`; its reachable errors are
1823                    // `Error::Schema` + spawn `Error::Storage`, which THIS PR narrowed to
1824                    // propagate. Enumeration + why it is kept: `merge_stream_setup`'s doc.
1825                    Err(setup) if setup.fallback_eligible() => {
1826                        tracing::warn!(
1827                            "SSTableManager::scan_stream - cross-generation RECONCILING merge \
1828                             could not be constructed for '{}' ({}); falling back to the lazy \
1829                             per-reader NON-reconciling token-order CONCATENATION: rows are NOT \
1830                             last-write-wins reconciled and may contain duplicated overwritten \
1831                             rows and rows resurrected from under a tombstone (issue #883)",
1832                            table_id,
1833                            setup.into_error()
1834                        );
1835                    }
1836                    Err(setup) => return Err(setup.into_counted_error()),
1837                }
1838            }
1839        }
1840
1841        Ok(scan_stream_fanout::spawn_fanout_merge(
1842            readers,
1843            table_id.clone(),
1844            start_key.cloned(),
1845            end_key.cloned(),
1846            schema.cloned(),
1847            buffer_size,
1848        ))
1849    }

1901    /// Streaming scan under the `tombstones` feature.
1902    ///
1903    /// Streaming the cross-generation tombstone merge is not yet implemented, so
1904    /// this falls back to the materializing [`scan`](Self::scan) and forwards the
1905    /// result through a bounded channel. The public API stays uniform across
1906    /// feature configs; the O(rows) memory win of issue #790 applies only to the
1907    /// default (non-`tombstones`) build.
1908    #[cfg(feature = "tombstones")]
1909    pub async fn scan_stream(
1910        &self,
1911        table_id: &TableId,
1912        start_key: Option<&RowKey>,
1913        end_key: Option<&RowKey>,
1914        schema: Option<&crate::schema::TableSchema>,
1915        buffer_size: usize,
1916    ) -> Result<reader::RowScanStream> {
1917        let results = self
1918            .scan(table_id, start_key, end_key, None, schema)
1919            .await?;
1920        let (tx, rx) = tokio::sync::mpsc::channel(buffer_size.max(1));
1921        // Issue #3124: the handle is RETAINED so a forwarder that dies mid-drain is
1922        // an error at the consumer rather than a clean end of stream (this build's
1923        // rows are already materialized, but the boundary must not be the one place
1924        // where a dead producer still reads as a finished scan).
1925        //
1926        // UNMETERED on purpose (issue #1701 F1): the `scan` above IS the metered read
1927        // operation on this build, so metering this forwarder too would count every
1928        // row of a `tombstones` streaming scan TWICE.
1929        let task = tokio::spawn(async move {
1930            for entry in results {
1931                if tx.send(Ok(entry)).await.is_err() {
1932                    break; // consumer dropped
1933                }
1934            }
1935        });
1936        Ok(reader::RowScanStream::new_over_counted_source(rx, task))
1937    }

2571        let (in_tx, in_rx) = tokio::sync::mpsc::channel::<Result<(RowKey, ScanRow)>>(16);

2578            .send(Err(crate::Error::Corruption("boom".to_string())))
2579            .await
2580            .unwrap();

2583        // Issue #3124: the re-chunker's source is a `RowScanStream`, so this stand-in
2584        // pairs the channel with a producer task that FINISHES cleanly — the source's
