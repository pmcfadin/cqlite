1//! Schema-Aware SSTable Reader
2//!
3//! This module provides a schema-aware wrapper around the SSTable reader that uses
4//! the SchemaParser for all value parsing operations instead of type guessing.
5//! It requires a complete schema definition and uses proper comparators for
6//! partition and clustering keys.

127    /// Create a new schema-aware reader with custom configuration
128    pub async fn new_with_config(
129        path: &Path,
130        schema: TableSchema,
131        schema_registry: Arc<SchemaRegistry>,
132        config: &Config,
133        platform: Arc<Platform>,
134        reader_config: SchemaAwareReaderConfig,
135    ) -> Result<Self> {
136        // Validate schema before proceeding
137        if reader_config.validate_schema_completeness {
138            Self::validate_schema_completeness(&schema)?;
139        }
140
141        // Create parsing context
142        let context = Self::create_parsing_context(&schema, &schema_registry)?;
143
144        // Validate context completeness if strict validation is enabled
145        if reader_config.strict_schema_validation && !context.is_complete() {
146            return Err(Error::Schema(format!(
147                "Incomplete parsing context for table {}.{}: missing schema or comparators",
148                schema.keyspace, schema.table
149            )));
150        }
151
152        // Create schema parser
153        let schema_parser = SchemaParser::new(context.clone())?;
154
155        // Open underlying SSTable reader
156        let reader = SSTableReader::open(path, config, platform.clone()).await?;
157
158        // Detect SSTable format from filename
159        let format = Self::detect_format(&reader)?;
160
161        // Get Cassandra version from header
162        let version = reader.cassandra_version();
163
164        Ok(Self {
165            file_path: path.to_path_buf(),
166            reader,
167            schema_parser,
168            context,
169            format,
170            version,
171            platform,
172        })
173    }

175    /// Validate that the schema contains all required fields for schema-aware parsing
176    pub fn validate_schema_completeness(schema: &TableSchema) -> Result<()> {
177        // Must have at least one partition key
178        if schema.partition_keys.is_empty() {
179            return Err(Error::Schema(format!(
180                "Schema for table {}.{} must have at least one partition key",
181                schema.keyspace, schema.table
182            )));
183        }
184
185        // All partition keys must have valid types
186        for (idx, key) in schema.partition_keys.iter().enumerate() {
187            CqlType::parse(&key.data_type).map_err(|e| {
188                Error::Schema(format!(
189                    "Invalid partition key type '{}' at position {} in {}.{}: {}",
190                    key.data_type, idx, schema.keyspace, schema.table, e
191                ))
192            })?;
193        }
194
195        // All clustering keys must have valid types
196        for (idx, key) in schema.clustering_keys.iter().enumerate() {
197            CqlType::parse(&key.data_type).map_err(|e| {
198                Error::Schema(format!(
199                    "Invalid clustering key type '{}' at position {} in {}.{}: {}",
200                    key.data_type, idx, schema.keyspace, schema.table, e
201                ))
202            })?;
203        }
204
205        // All columns must have valid types
206        for column in &schema.columns {
207            CqlType::parse(&column.data_type).map_err(|e| {
208                Error::Schema(format!(
209                    "Invalid column type '{}' for column '{}' in {}.{}: {}",
210                    column.data_type, column.name, schema.keyspace, schema.table, e
211                ))
212            })?;
213        }
214
215        // Validate positions are contiguous for partition keys
216        let mut positions: Vec<usize> = schema.partition_keys.iter().map(|k| k.position).collect();
217        positions.sort();
218        for (expected, &actual) in positions.iter().enumerate() {
219            if expected != actual {
220                return Err(Error::Schema(format!(
221                    "Non-contiguous partition key positions in {}.{}: expected {}, found {}",
222                    schema.keyspace, schema.table, expected, actual
223                )));
224            }
225        }
226
227        // Validate positions are contiguous for clustering keys (if any)
228        if !schema.clustering_keys.is_empty() {
229            let mut positions: Vec<usize> =
230                schema.clustering_keys.iter().map(|k| k.position).collect();
231            positions.sort();
232            for (expected, &actual) in positions.iter().enumerate() {
233                if expected != actual {
234                    return Err(Error::Schema(format!(
235                        "Non-contiguous clustering key positions in {}.{}: expected {}, found {}",
236                        schema.keyspace, schema.table, expected, actual
237                    )));
238                }
239            }
240        }
241
242        Ok(())
243    }

246    pub fn create_parsing_context(
247        schema: &TableSchema,
248        _registry: &SchemaRegistry,
249    ) -> Result<ParsingContext> {
250        // Get partition key comparators
251        let partition_comparators = schema.get_partition_key_comparators()?;
252
253        // Get clustering key comparators
254        let clustering_comparators = schema.get_clustering_key_comparators()?;
255
256        // Get all column comparators
257        let column_comparators = schema.get_all_comparators()?;
258
259        Ok(ParsingContext {
260            schema: schema.clone(),
261            partition_comparators,
262            clustering_comparators,
263            column_comparators,
264        })
265    }

276    /// Get a value by partition and clustering keys with schema-driven parsing
277    pub async fn get(
278        &self,
279        partition_key: &[Value],
280        clustering_key: Option<&[Value]>,
281    ) -> Result<Option<HashMap<String, Value>>> {
282        // Validate partition key against schema
283        self.validate_partition_key(partition_key)?;
284
285        // Validate clustering key if provided
286        if let Some(ck) = clustering_key {
287            self.validate_clustering_key(ck)?;
288        }
289
290        // Create RowKey from partition and clustering keys
291        let row_key = self.create_row_key(partition_key, clustering_key)?;
292
293        // Get raw value from underlying reader
294        let table_id = self.get_table_id();
295        if let Some(raw_value) = self.reader.get(&table_id, &row_key).await? {
296            // Parse the raw value using schema
297            let parsed_row = self.parse_row_value(&raw_value)?;
298            Ok(Some(parsed_row))
299        } else {
300            Ok(None)
301        }
302    }

304    /// Scan a range of rows with schema-driven parsing
305    pub async fn scan(
306        &self,
307        start_partition: Option<&[Value]>,
308        end_partition: Option<&[Value]>,
309        start_clustering: Option<&[Value]>,
310        end_clustering: Option<&[Value]>,
311        limit: Option<usize>,
312    ) -> Result<Vec<(Vec<Value>, Vec<Value>, HashMap<String, Value>)>> {
313        // Validate start partition key if provided
314        if let Some(pk) = start_partition {
315            self.validate_partition_key(pk)?;
316        }
317
318        // Validate end partition key if provided
319        if let Some(pk) = end_partition {
320            self.validate_partition_key(pk)?;
321        }
322
323        // Validate start clustering key if provided
324        if let Some(ck) = start_clustering {
325            self.validate_clustering_key(ck)?;
326        }
327
328        // Validate end clustering key if provided
329        if let Some(ck) = end_clustering {
330            self.validate_clustering_key(ck)?;
331        }
332
333        // Create start and end row keys
334        let start_key = if let Some(pk) = start_partition {
335            Some(self.create_row_key(pk, start_clustering)?)
336        } else {
337            None
338        };
339
340        let end_key = if let Some(pk) = end_partition {
341            Some(self.create_row_key(pk, end_clustering)?)
342        } else {
343            None
344        };
345
346        // Scan using underlying reader, passing schema for accurate parsing
347        let table_id = self.get_table_id();
348        let raw_results = self
349            .reader
350            .scan(
351                &table_id,
352                start_key.as_ref(),
353                end_key.as_ref(),
354                limit,
355                Some(&self.context.schema),
356            )
357            .await?;
358
359        // Parse results using schema
360        let mut parsed_results = Vec::new();
361        for (row_key, raw_value) in raw_results {
362            let (partition_values, clustering_values) = self.parse_row_key(&row_key)?;
363            let column_values = self.parse_row_value(&raw_value)?;
364            parsed_results.push((partition_values, clustering_values, column_values));
365        }
366
367        Ok(parsed_results)
368    }

370    /// Parse a raw row key into partition and clustering key values
371    pub fn parse_row_key(&self, row_key: &RowKey) -> Result<(Vec<Value>, Vec<Value>)> {
372        let key_bytes = row_key.as_bytes();
373
374        // Parse partition key portion
375        let partition_values = self.schema_parser.parse_partition_key(key_bytes)?;
376
377        // Calculate partition key length to find clustering key start
378        let partition_length = self.calculate_partition_key_length(&partition_values)?;
379
380        // Parse clustering key portion (if present)
381        let clustering_values = if key_bytes.len() > partition_length {
382            self.schema_parser
383                .parse_clustering_keys(&key_bytes[partition_length..])?
384        } else {
385            Vec::new()
386        };
387
388        Ok((partition_values, clustering_values))
389    }

391    /// Parse a raw row value into column values
392    pub fn parse_row_value(&self, raw_value: &Value) -> Result<HashMap<String, Value>> {
393        let value_bytes = raw_value
394            .as_bytes()
395            .ok_or_else(|| Error::Schema("Row value is not binary data".to_string()))?;
396
397        let mut column_values = HashMap::new();
398        let mut offset = 0;
399
400        // Parse each column according to schema
401        for column in &self.context.schema.columns {
402            // Skip key columns as they're parsed separately
403            if self.context.schema.is_partition_key(&column.name)
404                || self.context.schema.is_clustering_key(&column.name)
405            {
406                continue;
407            }
408
409            if offset >= value_bytes.len() {
410                break; // No more data
411            }
412
413            // Parse column value using schema parser with exact consumed-byte tracking
414            let (column_value, consumed) = self
415                .schema_parser
416                .parse_column_value(&column.name, &value_bytes[offset..])?;
417
418            offset += consumed;
419            column_values.insert(column.name.clone(), column_value);
420        }
421
422        Ok(column_values)
423    }

475    fn validate_partition_key(&self, key: &[Value]) -> Result<()> {
476        if key.len() != self.context.partition_comparators.len() {
477            return Err(Error::Schema(format!(
478                "Partition key length mismatch: expected {}, got {}",
479                self.context.partition_comparators.len(),
480                key.len()
481            )));
482        }
483        Ok(())
484    }
485
486    fn validate_clustering_key(&self, key: &[Value]) -> Result<()> {
487        if key.len() > self.context.clustering_comparators.len() {
488            return Err(Error::Schema(format!(
489                "Clustering key too long: expected max {}, got {}",
490                self.context.clustering_comparators.len(),
491                key.len()
492            )));
493        }
494        Ok(())
495    }

497    fn create_row_key(
498        &self,
499        partition_key: &[Value],
500        clustering_key: Option<&[Value]>,
501    ) -> Result<RowKey> {
502        // Serialize partition key
503        let mut key_bytes = Vec::new();
504        for (value, comparator) in partition_key
505            .iter()
506            .zip(&self.context.partition_comparators)
507        {
508            let serialized = self.serialize_value_with_comparator(value, comparator)?;
509            key_bytes.extend_from_slice(&serialized);
510        }
511
512        // Serialize clustering key if provided
513        if let Some(ck) = clustering_key {
514            for (value, comparator) in ck.iter().zip(&self.context.clustering_comparators) {
515                let serialized = self.serialize_value_with_comparator(value, comparator)?;
516                key_bytes.extend_from_slice(&serialized);
517            }
518        }
519
520        Ok(RowKey::new(key_bytes))
521    }
522
523    fn serialize_value_with_comparator(
524        &self,
525        value: &Value,
526        _comparator: &ComparatorType,
527    ) -> Result<Vec<u8>> {
528        // This would use the proper Cassandra serialization format for each type
529        // For now, return a simplified implementation
530        Ok(value.as_bytes().unwrap_or(&[]).to_vec())
531    }
532
533    fn calculate_partition_key_length(&self, partition_values: &[Value]) -> Result<usize> {
534        // Calculate the byte length of the serialized partition key
535        let mut total_length = 0;
536        for (value, comparator) in partition_values
537            .iter()
538            .zip(&self.context.partition_comparators)
539        {
540            let serialized = self.serialize_value_with_comparator(value, comparator)?;
541            total_length += serialized.len();
542        }
543        Ok(total_length)
544    }

555/// Error handling specific to schema-aware reading
556#[derive(Debug, thiserror::Error)]
557pub enum SchemaAwareReaderError {
558    #[error("Schema validation failed: {0}")]
559    SchemaValidation(String),
560
561    #[error("Parsing context incomplete: {0}")]
562    IncompleteContext(String),
563
564    #[error("Key validation failed: {0}")]
565    KeyValidation(String),
566
567    #[error("Value parsing failed for column '{column}': {reason}")]
568    ValueParsing { column: String, reason: String },
569
570    #[error("Format-specific error: {0}")]
571    FormatSpecific(String),
572
573    #[error("Incompatible Cassandra version: expected {expected:?}, found {found:?}")]
574    VersionMismatch {
575        expected: CassandraVersion,
576        found: CassandraVersion,
577    },
578}
579
580impl From<SchemaAwareReaderError> for Error {
581    fn from(err: SchemaAwareReaderError) -> Self {
582        Error::Schema(err.to_string())
583    }
