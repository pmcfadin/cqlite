15/// Check 8 (FULL): partition key/row ordering + partition-level
16/// `localDeletionTime` validity (issue #1282).
17///
18/// Two corruption classes Cassandra's `sstableverify` rejects that the earlier
19/// checks did not classify:
20///
21/// * **Out-of-order key/row** ([`VerifyErrorClass::OutOfOrderKeyOrRow`]).
22///   Cassandra stores partitions in ascending **Murmur3 token** order (ties
23///   broken by the raw key bytes). We recompute each partition's token with the
24///   authoritative [`cassandra_murmur3_token`] (Murmur3Partitioner, matching the
25///   rest of CQLite's BTI read path, issue #755) and flag the first
26///   `(token, key)` pair that is not strictly greater than its predecessor.
27///
28/// * **Invalid partition-level local-deletion-time**
29///   ([`VerifyErrorClass::InvalidLocalDeletionTime`]). `localDeletionTime` is
30///   seconds since the Unix epoch; the only special non-negative value is the
31///   live sentinel `i32::MAX`. On the legacy signed (`nb`) `DeletionTime` form a
32///   NEGATIVE partition-level value is unambiguously corrupt (Cassandra's
33///   `DeletionTime`/`Verifier` rejects it). The unsigned `oa`/`da` form
34///   legitimately represents far-future times in `[2^31, 2^32)` as a negative
35///   `i32`, so we ONLY flag a negative value when the on-disk format is the
36///   signed legacy form — the format, not a heuristic, decides.
37///
38/// Both facts come from the SAME authoritative partition-header decode the scan
39/// already performs (see [`SSTableReader::partition_verify_scan`]); this is not a
40/// second guessing pass. Environmental errors (reader open) are surfaced through
41/// the existing scan-error classifier rather than aborting verification.
42pub(super) async fn check_key_order_and_ldt(
43    data_path: &Path,
44    config: &Config,
45    platform: Arc<Platform>,
46    findings: &mut Vec<VerifyFinding>,
47) {
48    let reader = match SSTableReader::open(data_path, config, platform).await {
49        Ok(r) => r,
50        Err(_) => {
51            // A reader-open failure here is already surfaced by the Check 7 scan
52            // (it opens the same reader first); do not double-report it.
53            return;
54        }
55    };
56    let signed_ldt = !reader.has_uint_deletion_time();
57    let partitions = match reader.partition_verify_scan().await {
58        Ok(p) => p,
59        Err(_) => {
60            // A parse failure is Check 7's territory (RowScanFailed / decode);
61            // avoid a duplicate, differently-classed finding for the same cause.
62            return;
63        }
64    };
65
66    findings.extend(classify_order_and_ldt(&partitions, signed_ldt));
67
68    // Row-order half of OutOfOrderKeyOrRow (issue #1282 roborev follow-up):
69    // Cassandra's Verifier also rejects out-of-order CLUSTERING rows within a
70    // partition. Decode each partition's clustering rows in on-disk order and flag
71    // a non-increasing clustering step using the authoritative schema comparator
72    // (which respects reversed/DESC clustering order). A table with no clustering
73    // columns yields an empty scan and produces no findings.
74    if let Some(schema) = reader.effective_schema() {
75        // A decode failure is Check 7's territory; do not double-report. Only a
76        // successful scan feeds the clustering-order classifier.
77        if !schema.clustering_keys.is_empty() {
78            if let Ok(partition_rows) = reader.partition_clustering_verify_scan().await {
79                findings.extend(classify_clustering_row_order(&partition_rows, &schema));
80            }
81        }
82    }
83}

85/// Compare two clustering-key tuples in the authoritative schema clustering
86/// order (issue #1282 roborev follow-up).
87///
88/// Each column is compared with its non-gated [`ComparatorType`] (derived from the
89/// schema clustering type) and the result reversed for a DESC column, mirroring
90/// Cassandra's reversed-type ordering. An absent trailing component (a shorter
91/// tuple) is treated as NULL, which sorts first regardless of ASC/DESC — matching
92/// `ClusteringKey::compare`. NO heuristics: the format-derived comparator and the
93/// schema's ASC/DESC flag decide.
94fn compare_clustering_tuples(
95    a: &[crate::types::Value],
96    b: &[crate::types::Value],
97    schema: &crate::schema::TableSchema,
98) -> Result<std::cmp::Ordering> {
99    use crate::types::Value;
100    use std::cmp::Ordering;
101
102    let comparators = schema.get_clustering_key_comparators()?;
103    for (i, ck) in schema.clustering_keys.iter().enumerate() {
104        let av = a.get(i).unwrap_or(&Value::Null);
105        let bv = b.get(i).unwrap_or(&Value::Null);
106        // NULL/absent component sorts first regardless of ASC/DESC (no reversal).
107        let ord = match (av, bv) {
108            (Value::Null, Value::Null) => Ordering::Equal,
109            (Value::Null, _) => Ordering::Less,
110            (_, Value::Null) => Ordering::Greater,
111            (_, _) => {
112                let cmp = comparators
113                    .get(i)
114                    .ok_or_else(|| {
115                        Error::Schema(format!(
116                            "missing clustering comparator for column {}",
117                            ck.name
118                        ))
119                    })?
120                    .compare(av, bv)?;
121                if ck.order == crate::schema::ClusteringOrder::Desc {
122                    cmp.reverse()
123                } else {
124                    cmp
125                }
126            }
127        };
128        if ord != Ordering::Equal {
129            return Ok(ord);
130        }
131    }
132    Ok(Ordering::Equal)
133}

135/// Pure classifier for the ROW half of Check 8 (issue #1282 roborev follow-up):
136/// given each partition's clustering-key tuples in on-disk order and the
137/// authoritative schema, flag the first partition whose clustering rows are not in
138/// strictly ascending schema order as [`VerifyErrorClass::OutOfOrderKeyOrRow`].
139///
140/// The comparison applies each clustering column's ASC/DESC order via
141/// [`compare_clustering_tuples`] — NO heuristics. A non-increasing step (a row
142/// equal to or before its predecessor) is corruption Cassandra's `Verifier`
143/// rejects.
144///
145/// Kept side-effect-free so the public verify path and the unit tests drive the
146/// EXACT same classification (wiring evidence: `check_key_order_and_ldt` calls
147/// this, and `verify_sstable` calls that in FULL mode).
148pub(super) fn classify_clustering_row_order(
149    partition_rows: &[(usize, Vec<Vec<crate::types::Value>>)],
150    schema: &crate::schema::TableSchema,
151) -> Vec<VerifyFinding> {
152    use std::cmp::Ordering;
153
154    let mut findings = Vec::new();
155    for (part_idx, rows) in partition_rows {
156        for pair in rows.windows(2) {
157            let (prev, cur) = (&pair[0], &pair[1]);
158            // A comparator error (schema/type mismatch) is not an ordering fault;
159            // Check 7 owns decode/type failures, so skip rather than misclassify.
160            let ord = match compare_clustering_tuples(cur, prev, schema) {
161                Ok(o) => o,
162                Err(_) => continue,
163            };
164            // On disk a later clustering row MUST be strictly greater than its
165            // predecessor; Equal or Less is out-of-order corruption.
166            if ord != Ordering::Greater {
167                findings.push(VerifyFinding::new(
168                    VerifyErrorClass::OutOfOrderKeyOrRow,
169                    "Data.db",
170                    format!(
171                        "partition {} has an out-of-order clustering row: {:?} is not strictly after the previous row {:?} in schema clustering order",
172                        part_idx, cur, prev,
173                    ),
174                ));
175                break;
176            }
177        }
178    }
179    findings
180}

182/// Pure classifier for Check 8 (issue #1282): given the on-disk-ordered
183/// `(raw_partition_key, partition_local_deletion_time)` list from
184/// [`SSTableReader::partition_verify_scan`] and whether the on-disk
185/// `DeletionTime` is the legacy SIGNED form, return any order / LDT findings.
186///
187/// Kept side-effect-free so both the public verify path and the unit tests drive
188/// the EXACT same classification (wiring evidence: `check_key_order_and_ldt`
189/// calls this, and `verify_sstable` calls that in FULL mode).
190pub(super) fn classify_order_and_ldt(
191    partitions: &[(Vec<u8>, Option<i32>)],
192    signed_ldt: bool,
193) -> Vec<VerifyFinding> {
194    use crate::util::cassandra_murmur3::cassandra_murmur3_token;
195
196    let mut findings = Vec::new();
197    let hex = |b: &[u8]| b.iter().map(|x| format!("{:02x}", x)).collect::<String>();
198
199    // ---- Out-of-order partition keys (Murmur3 token order) -----------------
200    let mut prev: Option<(i64, Vec<u8>)> = None;
201    for (idx, (key, _ldt)) in partitions.iter().enumerate() {
202        let token = cassandra_murmur3_token(key);
203        if let Some((prev_token, prev_key)) = prev.as_ref() {
204            // Cassandra orders by (token, key bytes). A later partition MUST be
205            // strictly greater; equal or lesser is out-of-order corruption.
206            let ordered = (*prev_token, prev_key.as_slice()) < (token, key.as_slice());
207            if !ordered {
208                findings.push(VerifyFinding::new(
209                    VerifyErrorClass::OutOfOrderKeyOrRow,
210                    "Data.db",
211                    format!(
212                        "partition {} (key {}, token {}) is not strictly after the previous partition (key {}, token {}) — partitions are stored out of Murmur3 token order",
213                        idx,
214                        hex(key),
215                        token,
216                        hex(prev_key),
217                        prev_token,
218                    ),
219                ));
220                break;
221            }
222        }
223        prev = Some((token, key.clone()));
224    }
225
226    // ---- Negative (invalid) partition-level localDeletionTime (nb) ---------
227    if signed_ldt {
228        for (key, ldt) in partitions {
229            if let Some(ldt) = ldt {
230                // A deleted partition's localDeletionTime is epoch-seconds; it
231                // cannot be negative. (The live sentinel i32::MAX is positive and
232                // is already resolved to `None` by the header parser.)
233                if *ldt < 0 {
234                    findings.push(VerifyFinding::new(
235                        VerifyErrorClass::InvalidLocalDeletionTime,
236                        "Data.db",
237                        format!(
238                            "partition (key {}) has a negative localDeletionTime {} (0x{:08x}) on the signed (nb) DeletionTime form — a valid deletion time is >= 0 seconds since epoch",
239                            hex(key),
240                            ldt,
241                            *ldt as u32,
242                        ),
243                    ));
244                    break;
245                }
246            }
247        }
248    }
249
250    findings
