1//! Lazy Summary-guided `Index.db` open (issue #2412, design §A).
2//!
3//! Split out of `index_reader/mod.rs` (campsite #1116) — the eager constructors,
4//! struct definition, and sync accessors stay in `mod.rs` unchanged; this file adds
5//! the NEW lazy-open primitives `IndexReader::open_lazy` /
6//! `IndexReader::ensure_materialized` plus their internal `MaterializedIndex` cell
7//! type, used only by `SSTableReader`'s BIG-open composition
8//! (`reader::component_loading::load_index_reader`).

19/// Cap on the bounded validity-probe prefix [`IndexReader::open_lazy`] reads
20/// (issue #2302 open-time-detection preservation through the lazy-open change).
21/// Large enough to cover the WORST-CASE single `Index.db` entry's fixed FRAMING
22/// (`key_len` u16 + up to a 65535-byte key + two small vints — [`parse_big_index_entry_framing`]),
23/// so a legitimately huge partition KEY can never false-trigger the probe; bounded
24/// so the probe's cost stays O(1) at open time, never O(partition count). This
25/// intentionally does NOT need to cover the promoted-index PAYLOAD — that is
26/// UNBOUNDED in principle (see [`parse_big_index_entry_framing`]'s doc) and is
27/// validated structurally instead (declared length vs. the file's actual size),
28/// never by requiring the bytes to be physically present in this probe (roborev
29/// endgame finding, High — a probe requiring the full payload rejected HEALTHY
30/// wide-partition files).
31const VALIDITY_PROBE_PREFIX_CAP: usize = 70_000;
32
33/// The full parse result, materialized either eagerly (at construction, the
34/// unchanged legacy behavior every direct [`IndexReader::open`] caller relies on)
35/// or lazily (issue #2412, on the first [`IndexReader::ensure_materialized`] call).
36pub(super) struct MaterializedIndex {
37    pub(super) index_data: IndexData,
38    /// Whether the entry parse consumed the ENTIRE Index.db file (issue #2302).
39    /// The parser `break`s on the first unparseable entry and returns the parsed
40    /// PREFIX, so a mid-entry-truncated file opens with leftover bytes. `true` ⟺
41    /// no bytes remained — the authoritative signal the stream was not cut
42    /// mid-entry (WHOLE trailing entries dropped at an exact boundary are caught
43    /// separately at the enumeration site). Only the completeness-sensitive full
44    /// enumeration consults it; point-lookup callers tolerate a partial prefix.
45    pub(super) fully_parsed: bool,
46}

48impl IndexReader {
49    /// Path to the backing `Index.db` file (issue #2412 §B). The Summary-guided
50    /// point-lookup path (`reader::summary_point`) seeks bounded intervals from this
51    /// path WITHOUT materializing the whole map. Lives here (not the parent module)
52    /// as a lazy-open support accessor, keeping `index_reader/mod.rs` under the
53    /// campsite line (#1116).
54    pub(crate) fn index_path(&self) -> PathBuf {
55        // Owned snapshot of the interior-mutable path (issue #2356 roborev): a
56        // #2383 rebind can repoint it, so we cannot hand out a borrow into the
57        // `ArcSwap`. Cheap relative to the bounded-interval read that consumes it.
58        self.file_path.load().as_ref().clone()
59    }
60
61    /// Repoint this reader's DEFERRED `Index.db` open at `new_path` after a #2383
62    /// warm inode-rebind (issue #2356 roborev). The caller (`SSTableReader::rebind_path`)
63    /// has already proven the new generation is the SAME immutable on-disk generation by
64    /// authoritative `(device, inode, generation)` + size identity, and `new_path` is the
65    /// `Index.db` hardlink SIBLING of the rebound `Data.db`; snapshot components are
66    /// same-inode hardlinks, so the bytes are byte-identical (issue #28 no-heuristics).
67    /// Without this, a LAZY reader that has not yet materialized would `File::open` its
68    /// ORIGINAL (now torn-down) snapshot path on the next `ensure_materialized`/interval
69    /// read and ENOENT — the #2352 class the Data.db-only rebind reintroduced.
70    pub(crate) fn rebind_path(&self, new_path: &Path) {
71        self.file_path.store(Arc::new(new_path.to_path_buf()));
72    }
73
74    /// Whether the full `Index.db` parse has already run (issue #2412 §B). A lazily
75    /// opened reader ([`Self::open_lazy`]) reports `false` until the first
76    /// [`Self::ensure_materialized`]; an eagerly opened reader reports `true` from
77    /// construction. The Summary-guided point path uses this to prefer the resident
78    /// map once it is already in memory (e.g. after a full scan) and the bounded
79    /// interval read only while the map is still deferred.
80    pub(crate) fn is_materialized(&self) -> bool {
81        self.materialized.get().is_some()
82    }

84    /// Lazy Summary-guided open (issue #2412, design §A): checks the file EXISTS
85    /// and is STRUCTURALLY VALID via a bounded O(1) prefix probe, but performs NO
86    /// full `Index.db` parse. Used by `SSTableReader`'s BIG open composition when a
87    /// usable `Summary.db` is present, so open cost is bounded by `Summary.db`
88    /// size + the probe's fixed cap, not by `Index.db` partition count. The full
89    /// parse is deferred to the first [`Self::ensure_materialized`] call from an
90    /// internal consumer that genuinely needs the resident map (a full-enumeration
91    /// scan, or the point-lookup fallback before Stage 3's interval-bounded lookup
92    /// lands); a point-lookup-dominated workload may never pay it at all.
93    ///
94    /// ## Open-time present-but-unloadable detection (issue #2302 contract preserved)
95    ///
96    /// A present-but-EMPTY or present-but-structurally-broken `Index.db` (open/parse
97    /// fails at the very first entry) MUST be distinguishable from a genuinely
98    /// valid file AT OPEN TIME, not only when a later consumer happens to call
99    /// [`Self::ensure_materialized`] — issue #2302 killed exactly this class of
100    /// silent degradation (the pre-#2412 eager `open` always full-parsed at open,
101    /// so a corrupt file was detected immediately; a lazy open that ONLY checked
102    /// file existence would silently regress that guarantee, reporting a broken
103    /// file as a usable lazy reader until something eventually materialized it —
104    /// possibly never, for a point-lookup-only workload). This is NOT negotiable:
105    /// open-time detection is the safer contract (`component_loading::load_index_reader`
106    /// maps any non-`NotFound` `Err` here to the loud-WARN `PresentButUnloadable`
107    /// path, exactly as the eager path always has).
108    ///
109    /// The probe: read a BOUNDED prefix (≤ [`VALIDITY_PROBE_PREFIX_CAP`] bytes, or
110    /// the whole file when smaller), confirm the file is non-empty, and confirm the
111    /// FIRST entry's fixed FRAMING parses ([`parse_big_index_entry_framing`]) —
112    /// the SAME authoritative on-disk framing `ensure_materialized`'s full parse
113    /// and the eager `open` both use (no heuristics, issue #28). Deliberately
114    /// validates ONLY the framing, not the promoted-index payload bytes
115    /// themselves (roborev endgame finding, High): the payload is UNBOUNDED in
116    /// principle (~one `IndexInfo` per 64KiB of partition data), so a probe
117    /// requiring it to be PHYSICALLY PRESENT in a bounded read would reject a
118    /// perfectly healthy `Index.db` whose first partition is simply wide —
119    /// stricter than the eager-parse contract it exists to preserve. Instead the
120    /// declared `promoted_len` is checked STRUCTURALLY against the file's actual
121    /// length (`framing_len + promoted_len <= file_len`): a value that would push
122    /// the entry past EOF is a structural impossibility (corrupt), while any
123    /// value that fits — however large, however far past this probe's read
124    /// window — is accepted as a legitimately wide (not corrupt) partition.
125    ///
126    /// This is intentionally NOT a full structural guarantee (a corruption deep
127    /// inside the file, past the first entry, is still caught later by
128    /// `is_fully_parsed()`/Signal A at first materializing use, exactly as before
129    /// this change) — it is the SAME bounded confidence the original eager `open`
130    /// effectively offered before its first byte was even consumed, at O(1) cost
131    /// instead of O(partitions).
132    ///
133    /// Not counted on EITHER `index_parses_total` or `index_interval_parses_total`:
134    /// this probe is neither a full parse nor a Summary-guided per-lookup interval
135    /// read (design §F) — conflating it with either would muddy the field-round
136    /// parse-count dashboards those counters exist for. It is deliberately silent
137    /// on success (the common case) and loud only via the `PresentButUnloadable`
138    /// WARN path on failure, matching #2302's existing observability contract.
139    pub(crate) async fn open_lazy(path: &Path, platform: Arc<Platform>) -> Result<Self> {
140        if !platform.fs().exists(path).await? {
141            return Err(Error::not_found(format!(
142                "Index.db file not found: {}",
143                path.display()
144            )));
145        }
146
147        let mut file = File::open(path).await?;
148        let file_len = file.metadata().await?.len();
149        if file_len == 0 {
150            return Err(Error::corruption(format!(
151                "Index.db file is empty: {}",
152                path.display()
153            )));
154        }
155        let probe_len = file_len.min(VALIDITY_PROBE_PREFIX_CAP as u64) as usize;
156        let mut probe_buf = vec![0u8; probe_len];
157        file.read_exact(&mut probe_buf).await?;
158        let (framing_len, promoted_len) = match parse_big_index_entry_framing(&probe_buf) {
159            Ok((_, (framing_len, promoted_len))) => (framing_len, promoted_len),
160            Err(e) => {
161                return Err(Error::corruption(format!(
162                    "Index.db first entry's framing failed to parse (present-but-unloadable) \
163                     at {}: {:?}",
164                    path.display(),
165                    e
166                )));
167            }
168        };
169        // Structural bound (no-heuristics #28): the DECLARED promoted-index
170        // payload length must fit within the file's ACTUAL length — a value that
171        // would push the entry past EOF is a real structural impossibility,
172        // never accepted regardless of how the framing itself parsed. A payload
173        // that legitimately extends past this probe's bounded read window (the
174        // wide-partition field shape) is NOT an error here.
175        let implied_entry_end = (framing_len as u64).saturating_add(promoted_len);
176        if implied_entry_end > file_len {
177            return Err(Error::corruption(format!(
178                "Index.db first entry's declared promoted-index length ({promoted_len}) \
179                 extends past the file's actual length ({file_len} bytes) at {} — \
180                 structurally impossible, present-but-unloadable",
181                path.display()
182            )));
183        }
184
185        // `file_path`/`materialized` are private fields of `IndexReader`, defined in
186        // the parent `index_reader` module — visible here because `lazy` is a
187        // DESCENDANT module of it (Rust privacy: private items are visible to the
188        // defining module and all its descendants), so no field-access bridge
189        // methods are needed on the struct itself.
190        Ok(Self {
191            file_path: arc_swap::ArcSwap::from_pointee(path.to_path_buf()),
192            materialized: tokio::sync::OnceCell::new(),
193            platform,
194        })
195    }
196
197    /// Ensure the full `Index.db` parse has run, performing it exactly once
198    /// (memoized) on first call — the deferred cost [`Self::open_lazy`] skips at
199    /// open time (issue #2412). A no-op (immediate return) for a reader constructed
200    /// via the eager constructors, since their cell is already populated. Cancel-
201    /// aware: polls `cancel` the same way the eager parse does, so a cooperative
202    /// cancellation still aborts a large deferred parse promptly.
203    pub(crate) async fn ensure_materialized(
204        &self,
205        cancel: &crate::storage::scan_cancel::ScanCancel,
206    ) -> Result<()> {
207        self.materialized
208            .get_or_try_init(|| async {
209                cancel.check()?;
210                // Load the CURRENT (possibly #2383-rebound) path so a lazy reader
211                // whose original snapshot dir was torn down opens the live hardlink,
212                // not the dead path (issue #2356 roborev, #2352 class).
213                let path = self.file_path.load_full();
214                let mut file = File::open(path.as_ref()).await?;
215                let mut buffer = Vec::new();
216                file.read_to_end(&mut buffer).await?;
217                if buffer.is_empty() {
218                    return Err(Error::corruption(format!(
219                        "Index.db file is empty: {}",
220                        path.display()
221                    )));
222                }
223                let (remaining, index_data) =
224                    match parse_index_data_cancellable(&buffer, None, cancel) {
225                        Ok(pair) => pair,
226                        Err(e @ Error::Cancelled) => return Err(e),
227                        Err(e) => {
228                            return Err(Error::corruption(format!(
229                                "Failed to parse Index.db: {:?}",
230                                e
231                            )));
232                        }
233                    };
234                let fully_parsed = remaining.is_empty();
235                Ok(MaterializedIndex {
236                    index_data,
237                    fully_parsed,
238                })
239            })
240            .await?;
241        Ok(())
242    }
243}

480    /// Issue #2412 (coordinator-flagged regression, re-anchor of the #2383 fix-C
481    /// guard): a cancellation tripped DURING [`IndexReader::ensure_materialized`]'s
482    /// deferred full parse must abort promptly with `Error::Cancelled`, not run
483    /// the whole O(entries) parse to completion.
484    ///
485    /// `ensure_materialized` is the SURVIVING big-parse site for a Summary-usable
486    /// BIG reader (design §A/§C): open itself is now O(summary) and performs no
487    /// parse at all, but a consumer that still needs the resident map — a full
488    /// enumeration whose Summary-guided streaming walk `FellBack`, or a
489    /// compaction full-ring scan — calls `ensure_materialized`, which reuses the
490    /// SAME `parse_index_data_cancellable` entry loop the pre-#2412 eager `open`
491    /// always ran (unchanged: `CANCEL_POLL_INTERVAL`-bounded cooperative poll,
492    /// `index_reader/parse.rs`). This test proves that cancel-poll survived the
493    /// move to the deferred/lazy call site.
494    ///
495    /// Same calibrated-margin convention as
496    /// `cqlite-flight::warm::registry::spin_tests_2383::cancel_during_large_index_parse_aborts_promptly`
497    /// (issue #2383 roborev-1653 NIT 5 — no wall-clock races: the margin is a
498    /// small fraction of a JUST-measured baseline for this exact fixture on this
499    /// exact host, not a fixed sleep constant that flakes on a fast host).
500    #[tokio::test]
501    async fn ensure_materialized_cancel_mid_parse_aborts_promptly() {
502        use crate::storage::scan_cancel::ScanCancel;
503
504        // Large enough to span several `CANCEL_POLL_INTERVAL` (65536) windows, so
505        // the parse loop polls cancel multiple times before completing — proving
506        // "aborts DURING the parse", not merely at the coarse pre-check.
507        const N: usize = 400_000;
508        let mut bytes = Vec::new();
509        for i in 0..N {
510            let key = format!("key{i:010}");
511            bytes.extend_from_slice(&encode_entry(key.as_bytes(), i as u64));
512        }
513        let (dir, path) = temp_index_file("ensure-materialized-cancel", &bytes);
514
515        let config = crate::Config::default();
516        let platform = Arc::new(
517            crate::Platform::new(&config)
518                .await
519                .expect("platform must initialize"),
520        );
521
522        // Calibrate: fully materialize the SAME fixture once, uncancelled — also
523        // warms the OS page cache ahead of the timed run below (biasing it
524        // FASTER, never slower, than this baseline).
525        let calib_start = std::time::Instant::now();
526        let calib_reader = IndexReader::open_lazy(&path, platform.clone())
527            .await
528            .expect("calibration open_lazy");
529        calib_reader
530            .ensure_materialized(&ScanCancel::default())
531            .await
532            .expect("calibration materialize (uncancelled) completes");
533        let baseline = calib_start.elapsed();
534        // 1/20th of the measured baseline (same fraction as the flight-level
535        // sibling test): small enough to land inside the timed run, large enough
536        // to dwarf the coarse pre-parse check's same-thread, no-I/O comparison.
537        let margin = baseline / 20;
538
539        let reader = IndexReader::open_lazy(&path, platform)
540            .await
541            .expect("timed open_lazy");
542        let cancel = ScanCancel::new();
543        let canceller = {
544            let cancel = cancel.clone();
545            std::thread::spawn(move || {
546                std::thread::sleep(margin);
547                cancel.cancel();
548            })
549        };
550
551        let result = reader.ensure_materialized(&cancel).await;
552        canceller.join().expect("canceller");
553
554        assert!(
555            matches!(result, Err(Error::Cancelled)),
556            "a cancel tripped DURING ensure_materialized's deferred Index.db parse \
557             must abort promptly with Error::Cancelled (calibrated margin {margin:?} \
558             from baseline {baseline:?}), got {result:?} (issue #2412 re-anchor of \
559             #2383 fix C)"
560        );
561        // The aborted materialize must NOT have left a poisoned "materialized"
562        // state: entries stay empty/not-fully-parsed, matching `open_lazy`'s
563        // pre-materialize contract (never a half-parsed resident map).
564        assert!(
565            reader.get_partition_entries().is_empty(),
566            "a cancelled materialize must not expose a partial resident map"
567        );
568
569        let _ = std::fs::remove_dir_all(&dir);
570    }
