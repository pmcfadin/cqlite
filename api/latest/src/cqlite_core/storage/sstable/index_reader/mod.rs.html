34/// Index.db file header
35#[derive(Debug, Clone, Serialize, Deserialize)]
36pub struct IndexHeader {
37    /// Format version identifier
38    pub version: u32,
39    /// Number of index entries
40    pub entry_count: u32,
41    /// Size of the index data section
42    pub data_size: u64,
43    /// Checksum for validation
44    pub checksum: u32,
45}

47/// Partition index entry in Index.db
48#[derive(Debug, Clone)]
49pub struct PartitionIndexEntry {
50    /// Raw partition key bytes (length-prefixed in the on-disk BIG/NB Index.db format).
51    ///
52    /// NOTE (Issue #552): Despite the historical field name `key_digest`, this holds the
53    /// RAW partition key bytes, not an MD5 digest. The real Cassandra 5.0 NB Index.db entry
54    /// format is `[key_len: u16 BE][raw key bytes][data_offset: vint][promoted_len: vint]`.
55    /// There is no `0x0010` marker and no MD5 digest on disk. The field name is retained to
56    /// avoid churn in the zero-copy lookup table and downstream callers; it is used directly
57    /// as the partition key (e.g. for `RowKey`). The leading u16 is the key length
58    /// (e.g. 0x0010 for a 16-byte UUID, 0x0026 for a 38-byte composite key).
59    pub key_digest: Arc<[u8]>,
60    /// Raw partition key bytes (mirror of `key_digest`, kept for API compatibility).
61    /// Always `Some` now that all entries carry their raw key.
62    pub raw_key: Option<Arc<[u8]>>,
63    /// Offset in Data.db file
64    pub data_offset: u64,
65    /// Size of partition data
66    pub data_size: u32,
67    /// Promoted index entries for wide partitions (optional)
68    pub promoted_index: Option<PromotedIndexData>,
69}
70
71/// Promoted index for wide partitions (BIG "nb" format, Issue #993).
72///
73/// Holds the RAW promoted-index payload bytes captured from the Index.db entry
74/// (previously these bytes were decoded-away/discarded on the read path). The
75/// payload's `firstName`/`lastName` clustering prefixes are not self-delimiting
76/// without the table's clustering column types, so full structural decode is
77/// deferred to [`PromotedIndexData::decode`], which takes an authoritative,
78/// schema-driven clustering-prefix length callback (no heuristics — Issue #28).
79#[derive(Debug, Clone, Serialize, Deserialize)]
80pub struct PromotedIndexData {
81    /// Raw promoted-index payload (the bytes after the `promoted_index_size` VInt).
82    pub raw_payload: Vec<u8>,
83}
84
85impl PromotedIndexData {
86    /// Wrap a raw promoted-index payload.
87    pub fn from_raw(raw_payload: Vec<u8>) -> Self {
88        Self { raw_payload }
89    }
90
91    /// Number of bytes in the raw promoted-index payload.
92    pub fn len(&self) -> usize {
93        self.raw_payload.len()
94    }
95
96    /// True when the payload is empty.
97    pub fn is_empty(&self) -> bool {
98        self.raw_payload.is_empty()
99    }
100
101    /// Fully decode the promoted index using a schema-driven clustering-prefix
102    /// length callback. See [`crate::storage::sstable::promoted_index_reader`].
103    pub fn decode(
104        &self,
105        prefix_len: &super::promoted_index_reader::PrefixLen<'_>,
106    ) -> Result<super::promoted_index_reader::DecodedPromotedIndex> {
107        super::promoted_index_reader::decode_promoted_index(&self.raw_payload, prefix_len)
108    }
109
110    /// Read just the IndexInfo block count without decoding the (schema-dependent)
111    /// `firstName`/`lastName` prefixes. The leading `headerLength`, `DeletionTime`
112    /// and `count` fields are schema-free, so this is always recoverable.
113    ///
114    /// Returns 0 on a malformed/short payload (defensive — never panics).
115    pub fn block_count(&self) -> u32 {
116        super::promoted_index_reader::peek_block_count(&self.raw_payload).unwrap_or(0)
117    }
118}

120/// Complete Index.db data structure
121#[derive(Debug, Clone)]
122pub struct IndexData {
123    /// File header
124    pub header: IndexHeader,
125    /// All partition index entries
126    pub partition_entries: Vec<PartitionIndexEntry>,
127    /// Lookup table for efficient partition access - uses Arc<[u8]> as key type
128    ///
129    /// ## Zero-Copy Design (Issue #107, Problem 1)
130    ///
131    /// - Keys are `Arc<[u8]>` to enable reference counting without cloning digest bytes
132    /// - Lookups use `&[u8]` directly via Borrow trait (zero heap allocations)
133    /// - `Arc<[u8]>` implements `Borrow<[u8]>` enabling HashMap::get(&[u8]) without temporary Arc creation
134    pub key_lookup: HashMap<Arc<[u8]>, usize>,
135}

137/// High-level Index.db file reader
138#[allow(dead_code)]
139pub struct IndexReader {
140    /// Path to the Index.db file
141    file_path: PathBuf,
142    /// Parsed index data
143    index_data: IndexData,
144    /// Platform abstraction for file operations
145    platform: Arc<Platform>,
146    /// Whether the entry parse consumed the ENTIRE Index.db file (issue #2302).
147    /// The parser `break`s on the first unparseable entry and returns the parsed
148    /// PREFIX, so a mid-entry-truncated file opens with leftover bytes. `true` ⟺
149    /// no bytes remained — the authoritative signal the stream was not cut
150    /// mid-entry (WHOLE trailing entries dropped at an exact boundary are caught
151    /// separately at the enumeration site). Only the completeness-sensitive full
152    /// enumeration consults it; point-lookup callers tolerate a partial prefix.
153    fully_parsed: bool,
154}
155
156impl IndexReader {
157    /// Open and parse an Index.db file
158    pub async fn open(path: &Path, platform: Arc<Platform>) -> Result<Self> {
159        Self::open_with_summary(path, platform, None).await
160    }
161
162    /// Open and parse an Index.db file with Summary.db correlation for proper offset mapping
163    pub async fn open_with_summary(
164        path: &Path,
165        platform: Arc<Platform>,
166        summary_reader: Option<&SummaryReader>,
167    ) -> Result<Self> {
168        Self::open_with_summary_cancellable(
169            path,
170            platform,
171            summary_reader,
172            &crate::storage::scan_cancel::ScanCancel::default(),
173        )
174        .await
175    }
176
177    /// Cancel-aware variant of [`Self::open_with_summary`] (issue #2383 fix C):
178    /// the O(entries) partition-index parse polls `cancel` at a bounded interval
179    /// and returns [`Error::Cancelled`] promptly on a client-disconnect trip,
180    /// instead of pinning a worker to completion. Non-cancellable callers use
181    /// [`Self::open`]/[`Self::open_with_summary`] with a default never-cancel flag.
182    pub async fn open_with_summary_cancellable(
183        path: &Path,
184        platform: Arc<Platform>,
185        summary_reader: Option<&SummaryReader>,
186        cancel: &crate::storage::scan_cancel::ScanCancel,
187    ) -> Result<Self> {
188        if !platform.fs().exists(path).await? {
189            return Err(Error::not_found(format!(
190                "Index.db file not found: {}",
191                path.display()
192            )));
193        }
194
195        // Read the entire file
196        let mut file = File::open(path).await?;
197        let mut buffer = Vec::new();
198        file.read_to_end(&mut buffer).await?;
199
200        // Check for empty file
201        if buffer.is_empty() {
202            return Err(Error::corruption(format!(
203                "Index.db file is empty: {}",
204                path.display()
205            )));
206        }
207
208        // Parse the index data with optional Summary.db correlation.
209        //
210        // NOTE (issue #1572 / #2302): `parse_all_partition_keys_with_summary`
211        // `break`s on the first unparseable entry and returns the parsed prefix, so
212        // a truncated Index.db opens with a PARTIAL map — `open` intentionally still
213        // returns `Ok` (BIG point-lookup callers tolerate a partial prefix +
214        // whole-file miss fallback via `big_get_with_resolution`; failing here would
215        // regress them). Whether the whole file was consumed is RECORDED below as
216        // `fully_parsed` (exposed via `is_fully_parsed()`) so the completeness-
217        // sensitive FULL enumeration can reject a mid-entry-truncated index; the
218        // residual boundary-aligned-drop gap is closed at the enumeration site by
219        // the last-partition coverage check. No heuristics (issue #28).
220        let (remaining, index_data) =
221            match parse_index_data_cancellable(&buffer, summary_reader, cancel) {
222                Ok((remaining, data)) => (remaining, data),
223                // A mid-parse cancellation (issue #2383) is surfaced by VARIANT, never
224                // masked as a corruption error, so the warm path maps it to
225                // `WarmError::Cancelled` rather than a fail-closed retain.
226                Err(e @ Error::Cancelled) => return Err(e),
227                Err(e) => {
228                    return Err(Error::corruption(format!(
229                        "Failed to parse Index.db: {:?}",
230                        e
231                    )));
232                }
233            };
234
235        // A well-formed Index.db is a sequence of WHOLE entries running exactly to
236        // EOF (the `rest.is_empty()` parity asserts in this module's tests). Leftover
237        // bytes ⇒ the stream was cut mid-entry ⇒ the entry set is not provably
238        // complete. Structural, no heuristics (issue #28).
239        let fully_parsed = remaining.is_empty();
240
241        Ok(Self {
242            file_path: path.to_path_buf(),
243            index_data,
244            platform,
245            fully_parsed,
246        })
247    }
248
249    /// Get all partition entries
250    pub fn get_partition_entries(&self) -> &[PartitionIndexEntry] {
251        &self.index_data.partition_entries
252    }
253
254    /// Whether the entry parse consumed the ENTIRE Index.db file (issue #2302).
255    /// `false` ⟺ the file was truncated mid-entry, so the parsed entry set is a
256    /// partial prefix and MUST NOT be treated as a complete partition enumeration.
257    pub fn is_fully_parsed(&self) -> bool {
258        self.fully_parsed
259    }
260
261    /// Look up a partition by key digest
262    ///
263    /// ## Zero-Allocation Optimization (Issue #107)
264    ///
265    /// This method performs HashMap lookup without heap allocation by leveraging
266    /// the `Borrow` trait. Since `Arc<[u8]>` implements `Borrow<[u8]>`, we can
267    /// lookup using `&[u8]` directly without creating a temporary Arc.
268    ///
269    /// **Before:** `let key_arc: Arc<[u8]> = key_digest.into();` (heap allocation per query)
270    /// **After:** Direct `get(key_digest)` using Borrow trait (zero allocations)
271    pub fn lookup_partition(&self, key_digest: &[u8]) -> Option<&PartitionIndexEntry> {
272        self.index_data
273            .key_lookup
274            .get(key_digest)
275            .and_then(|&index| self.index_data.partition_entries.get(index))
276    }
277
278    /// Get statistics about the index
279    pub fn get_statistics(&self) -> IndexStatistics {
280        let mut promoted_count = 0;
281        let mut total_promoted_entries = 0;
282
283        for entry in &self.index_data.partition_entries {
284            if let Some(ref promoted) = entry.promoted_index {
285                promoted_count += 1;
286                total_promoted_entries += promoted.block_count() as usize;
287            }
288        }
289
290        IndexStatistics {
291            total_partitions: self.index_data.partition_entries.len(),
292            partitions_with_promoted_index: promoted_count,
293            total_promoted_entries,
294            file_size: self.file_path.metadata().map(|m| m.len()).unwrap_or(0),
295        }
296    }
297
298    /// Validate index integrity against Data.db offsets
299    pub async fn validate_integrity(&self) -> Result<Vec<String>> {
300        let mut issues = Vec::new();
301
302        // Check for overlapping offsets
303        let mut offsets: Vec<_> = self
304            .index_data
305            .partition_entries
306            .iter()
307            .map(|e| (e.data_offset, e.data_size))
308            .collect();
309
310        offsets.sort_by_key(|&(offset, _)| offset);
311
312        for i in 1..offsets.len() {
313            let (prev_offset, prev_size) = offsets[i - 1];
314            let (curr_offset, _) = offsets[i];
315
316            if prev_offset + prev_size as u64 > curr_offset {
317                issues.push(format!(
318                    "Overlapping partitions: offset {} + size {} overlaps with offset {}",
319                    prev_offset, prev_size, curr_offset
320                ));
321            }
322        }
323
324        Ok(issues)
325    }
326}

552    /// Issue #552: Validate the BIG-format parser against REAL Cassandra 5.0 Index.db files.
553    ///
554    /// `simple_table` has a single 16-byte UUID partition key (entries start 0x0010).
555    /// `multi_partition_table` has a 38-byte composite partition key (entries start 0x0026).
556    /// Both must read back ALL entries with monotonically increasing offsets.
557    #[tokio::test]
558    #[ignore = "Requires test data files (CQLITE_DATASETS_ROOT)"]
559    async fn test_real_index_db_big_format() {
560        let datasets_root = env::var("CQLITE_DATASETS_ROOT").unwrap_or_else(|_| {
561            "/Users/patrickmcfadin/local_projects/cqlite/test-data/datasets".to_string()
562        });
563
564        // --- Composite-key table (38-byte keys, entries start 0x0026) ---
565        let multi_dir = format!(
566            "{}/sstables/test_basic/multi_partition_table-6ac52100a25111f0a3fef1a551383fb9",
567            datasets_root
568        );
569        let multi_index = format!("{}/nb-1-big-Index.db", multi_dir);
570        let bytes = std::fs::read(&multi_index).expect("read multi_partition_table Index.db");
571        assert_eq!(
572            u16::from_be_bytes([bytes[0], bytes[1]]),
573            38,
574            "Composite key length should be 38 (0x0026)"
575        );
576        let (rest, entries) = parse_all_partition_keys(&bytes).expect("parse composite Index.db");
577        assert!(rest.is_empty(), "Should consume all Index.db bytes");
578        assert!(
579            entries.len() >= 2,
580            "multi_partition_table should have multiple partitions (got {})",
581            entries.len()
582        );
583        // First key is 38 bytes; first offset must be 0.
584        assert_eq!(
585            entries[0].key_digest.len(),
586            38,
587            "First key should be 38 bytes"
588        );
589        assert_eq!(
590            entries[0].data_offset, 0,
591            "First partition offset should be 0"
592        );
593        // Offsets are strictly increasing in token order.
594        for i in 1..entries.len() {
595            assert!(
596                entries[i].data_offset > entries[i - 1].data_offset,
597                "Offsets must increase: entry {} ({}) <= entry {} ({})",
598                i,
599                entries[i].data_offset,
600                i - 1,
601                entries[i - 1].data_offset
602            );

779        // Create index data with two partition entries (NB format with VInt offsets)

781        // VInt for 100: 0x64 (single byte, value < 128)
