8/// Feature gates for a BTI-format SSTable.
9///
10/// BtiFormat only has one version (`da`).  All modern feature gates are TRUE
11/// for `da` (BtiFormat.java:321-418).
12#[derive(Debug, Clone, PartialEq, Eq)]
13pub struct BtiVersionGates {
14    /// Raw version string (always `"da"` for BTI).
15    pub version: String,
16
17    /// All gates are TRUE for `da`.  Fields mirror the BIG gates for API parity.
18    pub has_commit_log_lower_bound: bool,
19    pub has_commit_log_intervals: bool,
20    pub has_max_compressed_length: bool,
21    pub has_pending_repair: bool,
22    pub has_is_transient: bool,
23    pub has_metadata_checksum: bool,
24    /// `hasOldBfFormat` is **FALSE** for BTI (BtiFormat.java:357-360).
25    pub has_old_bf_format: bool,
26    pub has_originating_host_id: bool,
27    /// `hasAccurateMinMax` — **TRUE** for BTI `da`.
28    ///
29    /// Source: BtiFormat.java:363-366
30    /// ```java
31    /// public boolean hasAccurateMinMax() { return true; }
32    /// ```
33    pub has_accurate_min_max: bool,
34    /// `hasLegacyMinMax` — **FALSE** for BTI `da`.
35    ///
36    /// Source: BtiFormat.java:368-371
37    /// ```java
38    /// public boolean hasLegacyMinMax() { return false; }
39    /// ```
40    pub has_legacy_min_max: bool,
41    pub has_improved_min_max: bool,
42    pub has_token_space_coverage: bool,
43    pub has_partition_level_deletion_presence_marker: bool,
44    pub has_key_range: bool,
45    pub has_uint_deletion_time: bool,
46}
47
48impl BtiVersionGates {
49    /// Compute BTI gates for the given version string.
50    ///
51    /// # Errors
52    ///
53    /// Returns `Err(Error::UnsupportedVersion)` if the version is not `"da"`
54    /// (the only supported BTI version; #1249 version floor).
55    pub fn from_version(version: &str) -> Result<Self> {
56        if version != "da" {
57            return Err(Error::UnsupportedVersion {
58                version: version.to_string(),
59                floor: "da".to_string(),
60            });
61        }
62        Ok(Self {
63            version: version.to_string(),
64            has_commit_log_lower_bound: true,
65            has_commit_log_intervals: true,
66            has_max_compressed_length: true,
67            has_pending_repair: true,
68            has_is_transient: true,
69            has_metadata_checksum: true,
70            has_old_bf_format: false, // Always false for BTI (BtiFormat.java:357-360)
71            has_originating_host_id: true,
72            // BtiFormat.java:363-366: `public boolean hasAccurateMinMax() { return true; }`
73            has_accurate_min_max: true,
74            // BtiFormat.java:368-371: `public boolean hasLegacyMinMax() { return false; }`
75            has_legacy_min_max: false,
76            has_improved_min_max: true,
77            has_token_space_coverage: true,
78            has_partition_level_deletion_presence_marker: true,
79            has_key_range: true,
80            has_uint_deletion_time: true,
81        })
82    }
83}

85#[cfg(test)]
86mod tests {
87    use super::*;

89    #[test]
90    fn test_bti_da_gates() {
91        let g = BtiVersionGates::from_version("da").unwrap();
92        assert!(g.has_commit_log_lower_bound);
93        assert!(g.has_commit_log_intervals);
94        assert!(g.has_max_compressed_length);
95        assert!(g.has_pending_repair);
96        assert!(g.has_is_transient);
97        assert!(g.has_metadata_checksum);
98        assert!(!g.has_old_bf_format, "da: !hasOldBfFormat");
99        assert!(g.has_originating_host_id);
100        // BtiFormat.java:363-366: hasAccurateMinMax() → true
101        assert!(
102            g.has_accurate_min_max,
103            "da: hasAccurateMinMax (BtiFormat.java:363)"
104        );
105        // BtiFormat.java:368-371: hasLegacyMinMax() → false
106        assert!(
107            !g.has_legacy_min_max,
108            "da: !hasLegacyMinMax (BtiFormat.java:368)"
109        );
110        assert!(g.has_improved_min_max);
111        assert!(g.has_token_space_coverage);
112        assert!(g.has_partition_level_deletion_presence_marker);
113        assert!(g.has_key_range);
114        assert!(g.has_uint_deletion_time);
115    }

117    /// R1: a non-`da` BTI version is rejected with the typed
118    /// `Error::UnsupportedVersion` (naming the `da` floor), not a generic
119    /// `InvalidFormat`.
120    #[test]
121    fn test_bti_rejects_non_da() {
122        for v in &["nb", "oa", "na", "ca"] {
123            let err = match BtiVersionGates::from_version(v) {
124                Ok(_) => panic!("{} is not da and must be rejected", v),
125                Err(e) => e,
126            };
127            match err {
128                Error::UnsupportedVersion { version, floor } => {
129                    assert_eq!(version, *v, "error must name the offending version");
130                    assert_eq!(floor, "da", "error must name the da floor");
131                }
132                other => panic!("{}: expected UnsupportedVersion, got {:?}", v, other),
133            }
134        }
135        // The single supported BTI version still constructs gates.
136        assert!(BtiVersionGates::from_version("da").is_ok());
137    }
