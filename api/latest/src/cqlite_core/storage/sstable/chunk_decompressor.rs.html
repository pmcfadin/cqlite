1//! Bulletproof chunk-based decompression for SSTable Data.db files
2//!
3//! This module implements the proper decompression strategy for Cassandra SSTable files
4//! using CompressionInfo.db metadata to decompress chunks on-demand.

16/// Number of decompressed chunks the on-demand cache retains (16 * 16KB ≈ 256KB).
17///
18/// Kept as a `const NonZeroUsize` so [`LruCache::new`] takes it without any runtime
19/// `unwrap()`/`expect()` (library code must be panic-free). The `match` folds the
20/// `Option` at compile time; the `NonZeroUsize::MIN` arm is unreachable for a literal
21/// `16` and exists only to keep the `const` total.
22const CHUNK_CACHE_CAPACITY: NonZeroUsize = match NonZeroUsize::new(16) {
23    Some(n) => n,
24    None => NonZeroUsize::MIN,
25};
26
27/// Chunk-based decompressor for SSTable Data.db files
28pub struct ChunkDecompressor {
29    /// Compression metadata from CompressionInfo.db
30    compression_info: CompressionInfo,
31    /// Recency-tracked cache of decompressed chunks keyed by authoritative chunk
32    /// index (issue #1569, Epic B/B3).
33    ///
34    /// Values are `Arc<[u8]>`: a cache hit is an `Arc::clone` (refcount bump), never
35    /// a chunk-sized memcpy of the cached bytes. Eviction is real LRU — `LruCache`
36    /// tracks recency on every `get`/`put` and evicts the genuinely least-recently-
37    /// used entry at capacity — replacing the previous `FxHashMap` that cloned on
38    /// hit and evicted an arbitrary `keys().next()` entry. This cache is exercised
39    /// single-threaded (the decompressor is owned `&mut` by one `BulletproofReader`),
40    /// so no `Mutex`/sharding is needed here; contrast B1's shared
41    /// `DecompressedChunkCache`.
42    chunk_cache: LruCache<usize, Arc<[u8]>>,
43    /// Count of actual chunk decompressions performed (miss path only). A cache hit
44    /// never increments it, so a test can prove a repeated read of the same chunk
45    /// decompressed exactly once (issue #1569).
46    decompress_calls: u64,
47    /// Data file path for error reporting
48    data_file_path: Option<String>,
49    /// Cached Data.db length, captured on the first chunk read (issue #1586).
50    ///
51    /// The file is an immutable SSTable, so its size is derived exactly once
52    /// with a single seek probe and then reused for every subsequent chunk's
53    /// `compressed_chunk_size` bounds math — instead of re-seeking to `End(0)`
54    /// (and back) on every chunk read.
55    cached_data_file_size: Option<u64>,
56    /// The reader's logical position after the most recent chunk read within the
57    /// current `read_data` / `decompress_chunk_by_index` call (issue #1586).
58    ///
59    /// Sequential chunk reads leave the reader positioned exactly at the next
60    /// chunk's offset, so the explicit `seek(Start(offset))` can be skipped when
61    /// it would be a no-op. Reset to `None` at each public entry point so it is
62    /// only ever trusted within a single call against a single reader.
63    stream_pos: Option<u64>,
64}

108    /// Read data from compressed SSTable at specified offset and length
109    pub fn read_data<R: Read + Seek>(
110        &mut self,
111        reader: &mut R,
112        offset: u64,
113        length: usize,
114    ) -> Result<Vec<u8>> {
115        let mut result = Vec::with_capacity(length);
116        let mut remaining = length;
117        let mut current_offset = offset;
118
119        // The reader's logical position is unknown at the start of a call (the
120        // caller may have passed a fresh reader), so the first chunk read always
121        // seeks; subsequent sequential reads within this call skip the redundant
122        // seek (issue #1586).
123        self.stream_pos = None;
124
125        while remaining > 0 {
126            // Determine which chunk contains this offset
127            let chunk_index = self.compression_info.chunk_for_offset(current_offset);
128            let offset_in_chunk = self.compression_info.offset_within_chunk(current_offset);
129
130            // Get the decompressed chunk
131            let chunk_data = self.get_decompressed_chunk(reader, chunk_index)?;
132
133            // Extract the requested data from this chunk
134            let chunk_start = offset_in_chunk as usize;
135            let chunk_end = std::cmp::min(chunk_start + remaining, chunk_data.len());
136
137            if chunk_start >= chunk_data.len() {
138                return Err(Error::InvalidFormat(format!(
139                    "Offset {} beyond chunk {} size {}",
140                    chunk_start,
141                    chunk_index,
142                    chunk_data.len()
143                )));
144            }
145
146            let chunk_slice = &chunk_data[chunk_start..chunk_end];
147            result.extend_from_slice(chunk_slice);
148
149            let bytes_read = chunk_slice.len();
150            remaining -= bytes_read;
151            current_offset += bytes_read as u64;
152        }
153
154        Ok(result)
155    }

157    /// Get a decompressed chunk, using the cache if available.
158    ///
159    /// On a hit this bumps the chunk's LRU recency and returns an `Arc::clone`
160    /// (a refcount bump, never a chunk-sized memcpy). On a miss it decompresses
161    /// once, converts the buffer to `Arc<[u8]>` exactly once, and inserts it — the
162    /// `LruCache` evicts the genuinely least-recently-used entry at capacity
163    /// (issue #1569).
164    fn get_decompressed_chunk<R: Read + Seek>(
165        &mut self,
166        reader: &mut R,
167        chunk_index: usize,
168    ) -> Result<Arc<[u8]>> {
169        // Cache hit: refcount bump + recency update, no copy of the chunk bytes.
170        if let Some(cached_chunk) = self.chunk_cache.get(&chunk_index) {
171            return Ok(Arc::clone(cached_chunk));
172        }
173
174        // Miss: decompress, then convert `Vec<u8>` -> `Arc<[u8]>` exactly once.
175        let chunk_data = self.decompress_chunk(reader, chunk_index)?;
176        let arc: Arc<[u8]> = Arc::from(chunk_data.into_boxed_slice());
177
178        // Real LRU eviction: `put` inserts as most-recently-used and evicts the
179        // least-recently-used entry when at capacity.
180        self.chunk_cache.put(chunk_index, Arc::clone(&arc));
181        Ok(arc)
182    }
183
184    /// Exact uncompressed size Cassandra wrote for chunk `chunk_index`, derived from
185    /// `data_length` and `chunk_length`.
186    ///
187    /// Cassandra lays chunks out so that random access works via
188    /// `position / chunk_length`: every chunk that holds data up to the final byte is
189    /// exactly `chunk_length`, the chunk containing the final byte is the partial
190    /// remainder, and any chunk whose start is at/after `data_length` is empty (some
191    /// flush/close paths append a degenerate trailing chunk — e.g. `system/local`:
192    /// data_length 708, chunk_length 16384, chunk 0 = 708 bytes, chunk 1 = 0). This is
193    /// the precise per-chunk invariant: stricter than `<= chunk_length` (it still
194    /// rejects a short chunk that has addressable data after it — corruption that would
195    /// also break Cassandra's own position→chunk mapping), but not the over-strict
196    /// `== chunk_length` that rejected valid partial/trailing chunks.
197    fn expected_decompressed_len(&self, chunk_index: usize) -> u64 {
198        let chunk_length = self.compression_info.chunk_length as u64;
199        let start = (chunk_index as u64).saturating_mul(chunk_length);
200        self.compression_info
201            .data_length
202            .saturating_sub(start)
203            .min(chunk_length)
204    }

206    /// Decompress a specific chunk from the compressed data file
207    fn decompress_chunk<R: Read + Seek>(
208        &mut self,
209        reader: &mut R,
210        chunk_index: usize,
211    ) -> Result<Vec<u8>> {
212        // Count every actual decompression (miss path only). A cache hit returns
213        // before reaching here, so a repeated read of a resident chunk leaves this
214        // unchanged — the wiring evidence for issue #1569.
215        self.decompress_calls = self.decompress_calls.saturating_add(1);
216
217        // Get compressed chunk offset
218        let compressed_offset = self
219            .compression_info
220            .compressed_chunk_offset(chunk_index)
221            .ok_or_else(|| Error::InvalidFormat(format!("No offset for chunk {}", chunk_index)))?;
222
223        // Determine record size (compressed payload + 4-byte inline CRC) using the
224        // Data.db length. The SSTable is immutable, so its size is derived once
225        // with a single seek probe and cached — never re-probed per chunk (#1586).
226        let file_size = match self.cached_data_file_size {
227            Some(size) => size,
228            None => {
229                let current_pos = reader.stream_position().map_err(Error::Io)?;
230                let size = reader.seek(SeekFrom::End(0)).map_err(Error::Io)?;
231                reader
232                    .seek(SeekFrom::Start(current_pos))
233                    .map_err(Error::Io)?;
234                self.cached_data_file_size = Some(size);
235                // The probe restored the reader to `current_pos`.
236                self.stream_pos = Some(current_pos);
237                size
238            }
239        };
240
241        // compressed_chunk_size returns the full record delta including the 4-byte inline CRC.
242        // CompressedSequentialWriter.java:203: chunkOffset += compressedLength + 4
243        let record_size = self
244            .compression_info
245            .compressed_chunk_size(chunk_index, file_size)
246            .ok_or_else(|| {
247                Error::InvalidFormat(format!("Cannot determine size for chunk {}", chunk_index))
248            })?;
249
250        // Bug #639 fix: subtract the 4-byte inline CRC from the delta.
251        // The old code passed all (delta) bytes to the decompressor, which included the
252        // trailing CRC and caused decompression failures on well-formed chunks.
253        if record_size < 4 {
254            return Err(Error::InvalidFormat(format!(
255                "Chunk {} record size {} is too small (minimum 4 bytes for inline CRC)",
256                chunk_index, record_size
257            )));
258        }
259        let compressed_len = (record_size - 4) as usize;
260
261        // Seek to compressed chunk offset and read compressed payload only.
262        // Skip the seek when the previous sequential chunk read already left the
263        // reader positioned exactly here (issue #1586 step 3).
264        if self.stream_pos != Some(compressed_offset) {
265            reader
266                .seek(SeekFrom::Start(compressed_offset))
267                .map_err(Error::Io)?;
268        }
269        // Position is unknown until the record's bytes are fully read.
270        self.stream_pos = None;
271
272        let mut compressed_data = vec![0u8; compressed_len];
273        reader.read_exact(&mut compressed_data).map_err(Error::Io)?;
274
275        // Read the 4-byte inline CRC32 (big-endian) and validate it over the compressed bytes.
276        // Authority: CompressedSequentialWriter.java:192 + read path lines 275-282.
277        let mut crc_bytes = [0u8; 4];
278        reader.read_exact(&mut crc_bytes).map_err(Error::Io)?;
279        // The record (payload + 4-byte CRC) has been consumed; the reader now sits
280        // at the next chunk's offset, so a subsequent sequential read can skip its
281        // seek (issue #1586).
282        self.stream_pos = Some(compressed_offset.saturating_add(record_size));
283        let stored_crc = u32::from_be_bytes(crc_bytes);
284        let computed_crc = crc32fast::hash(&compressed_data);
285        if stored_crc != computed_crc {
286            let file_info = match &self.data_file_path {
287                Some(path) => format!(" in file {}", path),
288                None => String::new(),
289            };
290            return Err(Error::InvalidFormat(format!(
291                "CRC32 mismatch for chunk {} at offset 0x{:x}{}: stored=0x{:08x}, computed=0x{:08x}, compressed_len={}",
292                chunk_index, compressed_offset, file_info, stored_crc, computed_crc, compressed_len
293            )));
294        }
295
296        tracing::debug!(
297            "Reading chunk {} at offset {} ({} bytes compressed, CRC OK)",
298            chunk_index,
299            compressed_offset,
300            compressed_len
301        );
302
303        // Incompressible-chunk fallback (Bug #639):
304        // When compressedLength >= maxCompressedLength, Cassandra stored the chunk uncompressed.
305        // CompressedSequentialWriter.java:160-177: if compressedLen >= maxCompressedLen, use raw buffer.
306        let max_compressed_length = self.compression_info.max_compressed_length as usize;
307        if compressed_len >= max_compressed_length {
308            tracing::debug!(
309                "Chunk {} is incompressible (compressed_len={} >= max_compressed_length={}), returning raw bytes",
310                chunk_index, compressed_len, max_compressed_length
311            );
312            // A raw chunk's stored bytes ARE the uncompressed bytes, so they must match
313            // the exact size Cassandra wrote for this chunk index — the same invariant
314            // the compressed path checks below (see `expected_decompressed_len`).
315            let expected_size = self.expected_decompressed_len(chunk_index);
316            if compressed_data.len() as u64 != expected_size {
317                return Err(Error::InvalidFormat(format!(
318                    "Raw (incompressible) chunk {} size {} != expected {} (data_length {}, chunk_length {}) — corrupt or misdecoded",
319                    chunk_index,
320                    compressed_data.len(),
321                    expected_size,
322                    self.compression_info.data_length,
323                    self.compression_info.chunk_length,
324                )));
325            }
326            return Ok(compressed_data);
327        }
328
329        // Decompress based on algorithm
330        let decompressed = match self.compression_info.algorithm.as_str() {
331            "LZ4Compressor" => self.decompress_lz4_chunk(&compressed_data, chunk_index),
332            "SnappyCompressor" => self.decompress_snappy_chunk(&compressed_data, chunk_index),
333            "DeflateCompressor" => self.decompress_deflate_chunk(&compressed_data, chunk_index),
334            "ZstdCompressor" => self.decompress_zstd_chunk(&compressed_data, chunk_index),
335            algorithm => Err(Error::UnsupportedFormat(format!(
336                "Unknown compression algorithm: {}",
337                algorithm
338            ))),
339        }?;
340
341        // Validate the decompressed chunk size against the exact size Cassandra wrote
342        // for this chunk index (see `expected_decompressed_len`). This handles full
343        // chunks, the partial final-data chunk, and degenerate empty trailing chunks,
344        // while still rejecting a short non-final chunk that has addressable data after
345        // it (corruption). The inline CRC above is the primary integrity guard.
346        let expected_size = self.expected_decompressed_len(chunk_index);
347        if decompressed.len() as u64 != expected_size {
348            return Err(Error::InvalidFormat(format!(
349                "Decompressed chunk {} size {} != expected {} (data_length {}, chunk_length {}) — corrupt or misdecoded",
350                chunk_index,
351                decompressed.len(),
352                expected_size,
353                self.compression_info.data_length,
354                self.compression_info.chunk_length,
355            )));
356        }
357
358        Ok(decompressed)
359    }

361    /// Decompress LZ4 chunk - Cassandra uses 4-byte little-endian length prefix
362    /// Upper bound on a single decompressed chunk: the configured uncompressed
363    /// chunk length (the final chunk is shorter, never larger). Used to bound the
364    /// streaming Deflate/Zstd decoders against decompression bombs. Falls back to
365    /// the 128MB global cap if chunk_length is unset/zero.
366    ///
367    /// Only the Snappy/Deflate/Zstd paths consume this bound, so the method is
368    /// gated to match its callers and stay dead-code-free under minimal builds.
369    #[cfg(any(feature = "snappy", feature = "deflate", feature = "zstd"))]
370    fn chunk_size_guard(&self) -> u64 {
371        match self.compression_info.chunk_length {
372            0 => 128 * 1024 * 1024,
373            n => n as u64,
374        }
375    }

377    #[cfg(feature = "lz4")]
378    fn decompress_lz4_chunk(&self, compressed_data: &[u8], chunk_index: usize) -> Result<Vec<u8>> {
379        let file_info = match &self.data_file_path {
380            Some(path) => format!(" in file {}", path),
381            None => String::new(),
382        };
383
384        if compressed_data.len() < 4 {
385            return Err(Error::InvalidFormat(format!(
386                "LZ4 compressed data too short for chunk {}{} (need at least 4 bytes for length prefix, got {})",
387                chunk_index, file_info, compressed_data.len()
388            )));
389        }
390
391        // CRITICAL: Cassandra's LZ4Compressor prepends a 4-byte little-endian length prefix
392        // See: org.apache.cassandra.io.compress.LZ4Compressor.decompress() lines 169-172
393        // This is NOT the lz4_flex size-prepended format (which uses varint encoding)
394        let decompressed_length = u32::from_le_bytes([
395            compressed_data[0],
396            compressed_data[1],
397            compressed_data[2],
398            compressed_data[3],
399        ]) as usize;
400
401        // Validate the LZ4 length prefix against the exact size Cassandra wrote for
402        // this chunk index (full / partial-final / empty-trailing — see
403        // `expected_decompressed_len`). Fails fast on a misdecoded prefix before
404        // attempting decompression.
405        let expected_size = self.expected_decompressed_len(chunk_index) as usize;
406        if decompressed_length != expected_size {
407            return Err(Error::InvalidFormat(format!(
408                "LZ4 length prefix mismatch for chunk {} at offset 0x{:x}: expected {}, got {} (first 4 bytes: {:02x} {:02x} {:02x} {:02x}){}",
409                chunk_index,
410                self.compression_info
411                    .chunk_offsets
412                    .get(chunk_index)
413                    .unwrap_or(&0),
414                expected_size,
415                decompressed_length,
416                compressed_data[0],
417                compressed_data[1],
418                compressed_data[2],
419                compressed_data[3],
420                file_info
421            )));
422        }
423
424        // Skip the 4-byte length prefix and decompress the actual LZ4 data
425        let lz4_data = &compressed_data[4..];
426
427        match lz4_flex::decompress(lz4_data, decompressed_length) {
428            Ok(decompressed) => {
429                if decompressed.len() != decompressed_length {
430                    return Err(Error::InvalidFormat(format!(
431                        "LZ4 decompression size mismatch for chunk {} at offset 0x{:x}: expected {}, got {}{}",
432                        chunk_index,
433                        self.compression_info
434                            .chunk_offsets
435                            .get(chunk_index)
436                            .unwrap_or(&0),
437                        decompressed_length,
438                        decompressed.len(),
439                        file_info
440                    )));
441                }
442                Ok(decompressed)
443            }
444            Err(e) => Err(Error::InvalidFormat(format!(
445                "LZ4 decompression failed for chunk {} at offset 0x{:x}: {}{}",
446                chunk_index,
447                self.compression_info
448                    .chunk_offsets
449                    .get(chunk_index)
450                    .unwrap_or(&0),
451                e,
452                file_info
453            ))),
454        }
455    }

466    /// Decompress Snappy chunk - strict mode for modern formats
467    fn decompress_snappy_chunk(
468        &self,
469        compressed_data: &[u8],
470        chunk_index: usize,
471    ) -> Result<Vec<u8>> {
472        #[cfg(feature = "snappy")]
473        {
474            use snap::raw::Decoder;
475
476            // Decompression-bomb guard (issue #1588): a raw Snappy block advertises
477            // its decompressed length as a leading varint, and `decompress_vec`
478            // pre-allocates that size BEFORE decoding. Reject an over-limit advertised
479            // length up front — a chunk decodes to at most `chunk_length` bytes — so a
480            // crafted block cannot force a huge allocation (mirrors the Deflate/Zstd
481            // chunk guards, which bound by `chunk_size_guard()`).
482            let max_chunk = self.chunk_size_guard();
483            let advertised = snap::raw::decompress_len(compressed_data).map_err(|e| {
484                Error::InvalidFormat(format!(
485                    "Snappy length decode failed for chunk {}: {}",
486                    chunk_index, e
487                ))
488            })?;
489            if advertised as u64 > max_chunk {
490                return Err(Error::InvalidFormat(format!(
491                    "Snappy chunk {} advertises {} bytes, beyond chunk_length {} (decompression-bomb guard)",
492                    chunk_index, advertised, max_chunk
493                )));
494            }
495
496            let mut decoder = Decoder::new();
497            match decoder.decompress_vec(compressed_data) {
498                Ok(decompressed) => Ok(decompressed),
499                Err(e) => Err(Error::InvalidFormat(format!(
500                    "Snappy decompression failed for chunk {} at offset 0x{:x}: {}. No fallback allowed for modern formats.",
501                    chunk_index,
502                    self.compression_info
503                        .chunk_offsets
504                        .get(chunk_index)
505                        .unwrap_or(&0),
506                    e
507                ))),
508            }
509        }
510
511        #[cfg(not(feature = "snappy"))]
512        {
513            let _ = (compressed_data, chunk_index); // Suppress unused warnings
514            Err(Error::UnsupportedFormat(
515                "Snappy support not compiled in".to_string(),
516            ))
517        }
518    }

520    /// Decompress Deflate chunk - strict mode for modern formats
521    fn decompress_deflate_chunk(
522        &self,
523        compressed_data: &[u8],
524        chunk_index: usize,
525    ) -> Result<Vec<u8>> {
526        #[cfg(feature = "deflate")]
527        {
528            // Cassandra's DeflateCompressor uses java.util.zip.Deflater/Inflater,
529            // which emit zlib-wrapped streams (2-byte header 0x78 0x9c + Adler-32
530            // trailer), NOT raw DEFLATE. Decode with ZlibDecoder to match. (#1082)
531            use flate2::read::ZlibDecoder;
532            use std::io::Read;
533
534            // Decompression-bomb guard: a chunk decompresses to at most
535            // `chunk_length` bytes (the final chunk is shorter). Cap the reader at
536            // chunk_length + 1 so a crafted zlib stream cannot expand into an
537            // unbounded Vec before we validate the size (roborev).
538            let max_chunk = self.chunk_size_guard();
539            let mut decoder = ZlibDecoder::new(compressed_data).take(max_chunk + 1);
540            let mut decompressed = Vec::new();
541
542            match decoder.read_to_end(&mut decompressed) {
543                Ok(_) if decompressed.len() as u64 > max_chunk => Err(Error::InvalidFormat(format!(
544                    "Deflate chunk {} expands beyond chunk_length {} (decompression-bomb guard)",
545                    chunk_index, max_chunk
546                ))),
547                Ok(_) => Ok(decompressed),
548                Err(e) => Err(Error::InvalidFormat(format!(
549                    "Deflate decompression failed for chunk {} at offset 0x{:x}: {}. No fallback allowed for modern formats.",
550                    chunk_index,
551                    self.compression_info
552                        .chunk_offsets
553                        .get(chunk_index)
554                        .unwrap_or(&0),
555                    e
556                ))),
557            }
558        }
559
560        #[cfg(not(feature = "deflate"))]
561        {
562            let _ = (compressed_data, chunk_index); // Suppress unused warnings
563            Err(Error::UnsupportedFormat(
564                "Deflate support not compiled in".to_string(),
565            ))
566        }
567    }

569    /// Decompress Zstd chunk - strict mode for modern formats
570    fn decompress_zstd_chunk(&self, compressed_data: &[u8], chunk_index: usize) -> Result<Vec<u8>> {
571        #[cfg(feature = "zstd")]
572        {
573            use std::io::Read;
574            use zstd::stream::read::Decoder as ZstdDecoder;
575
576            // AUTHORITATIVE dictionary detection (issue #1414): fail closed with a
577            // typed, feature-naming error BEFORE a plain decode when the metadata or
578            // frame header declares a dictionary — an unsupported FEATURE, not
579            // corruption/checksum, so it classifies distinctly (never a guess, #28).
580            if let Some(msg) =
581                zstd_dictionary_rejection(&self.compression_info, compressed_data, chunk_index)
582            {
583                return Err(Error::UnsupportedFormat(msg));
584            }
585
586            // Decompression-bomb guard: stream through a reader capped at
587            // chunk_length + 1 rather than `decode_all`, which would pre-allocate
588            // whatever content size the frame declares (roborev).
589            let max_chunk = self.chunk_size_guard();
590            let mut decoder = match ZstdDecoder::new(compressed_data) {
591                Ok(d) => d.take(max_chunk + 1),
592                Err(e) => {
593                    return Err(Error::InvalidFormat(format!(
594                        "Zstd decoder init failed for chunk {}: {}",
595                        chunk_index, e
596                    )))
597                }
598            };
599            let mut decompressed = Vec::new();
600            match decoder.read_to_end(&mut decompressed) {
601                Ok(_) if decompressed.len() as u64 > max_chunk => Err(Error::InvalidFormat(format!(
602                    "Zstd chunk {} expands beyond chunk_length {} (decompression-bomb guard)",
603                    chunk_index, max_chunk
604                ))),
605                Ok(_) => Ok(decompressed),
606                Err(e) => Err(Error::InvalidFormat(format!(
607                    "Zstd decompression failed for chunk {} at offset 0x{:x}: {}. No fallback allowed for modern formats.",
608                    chunk_index,
609                    self.compression_info
610                        .chunk_offsets
611                        .get(chunk_index)
612                        .unwrap_or(&0),
613                    e
614                ))),
615            }
616        }
617
618        #[cfg(not(feature = "zstd"))]
619        {
620            let _ = (compressed_data, chunk_index); // Suppress unused warnings
621            Err(Error::UnsupportedFormat(
622                "Zstd support not compiled in".to_string(),
623            ))
624        }
625    }

649    /// Read all data from the compressed file (for testing/debugging).
650    ///
651    /// Bounded by `data_length`, so it does NOT touch a degenerate empty trailing
652    /// chunk that holds no addressable data. To validate EVERY chunk record's inline
653    /// CRC (including such trailing chunks), iterate `chunk_count()` and call
654    /// `decompress_chunk_by_index`.
655    pub fn read_all_data<R: Read + Seek>(&mut self, reader: &mut R) -> Result<Vec<u8>> {
656        self.read_data(reader, 0, self.compression_info.data_length as usize)
657    }

660    pub fn chunk_count(&self) -> usize {
661        self.compression_info.chunk_offsets.len()
662    }

664    /// Decompress a single chunk by index, validating its inline CRC32 trailer.
665    ///
666    /// Exposed for strict per-chunk parity tests that must exercise every chunk record
667    /// — including degenerate trailing chunks that `read_all_data` (bounded by
668    /// `data_length`) never reads.
669    pub fn decompress_chunk_by_index<R: Read + Seek>(
670        &mut self,
671        reader: &mut R,
672        chunk_index: usize,
673    ) -> Result<Vec<u8>> {
674        // Single-chunk entry point: the reader position is unknown here too, so
675        // force the first (only) read to seek (issue #1586).
676        self.stream_pos = None;
677        // Cold per-chunk parity/verification path — hand callers an owned `Vec`
678        // (explicit copy, off the hot hit path) so the public signature is
679        // unchanged (issue #1569).
680        Ok(self.get_decompressed_chunk(reader, chunk_index)?.to_vec())
681    }

689/// Utility function to create a chunk decompressor from CompressionInfo.db file
690pub fn create_decompressor_from_file(
691    compression_info_path: &std::path::Path,
692) -> Result<ChunkDecompressor> {
693    let compression_data = std::fs::read(compression_info_path).map_err(Error::Io)?;
694
695    // Parse CompressionInfo using the deterministic Cassandra format.
696    // (Bug #638: old heuristic parse_alternative_format violated the no-heuristics mandate
697    // and has been removed.  The standard parse() is authoritative for all supported files.)
698    let compression_info = CompressionInfo::parse(&compression_data)?;
699
700    tracing::info!("Loaded compression info:");
701    tracing::info!("   Algorithm: {}", compression_info.algorithm);
702    tracing::info!("   Chunk Length: {} bytes", compression_info.chunk_length);
703    tracing::info!("   Data Length: {} bytes", compression_info.data_length);
704    tracing::info!("   Chunk Count: {}", compression_info.chunk_offsets.len());
705
706    ChunkDecompressor::new(compression_info, CassandraVersion::V5_0Release)
707}

754    /// A `Read + Seek` wrapper that counts every `seek()` call (which includes
755    /// `stream_position()`, since the default `Seek::stream_position` is
756    /// `seek(SeekFrom::Current(0))`). Used to prove the chunk read path does not
757    /// re-derive the immutable file size with a seek probe on every chunk
758    /// (issue #1586).
759    ///
760    /// Only the lz4-gated multichunk tests construct it, so it is gated to match and
761    /// stay dead-code-free under single-feature test builds (issue #1873).
762    #[cfg(feature = "lz4")]
763    struct SeekCountingReader {
764        inner: std::io::Cursor<Vec<u8>>,
765        seeks: std::rc::Rc<std::cell::Cell<usize>>,
766    }
767
768    #[cfg(feature = "lz4")]
769    impl std::io::Read for SeekCountingReader {
770        fn read(&mut self, buf: &mut [u8]) -> std::io::Result<usize> {
771            self.inner.read(buf)
772        }
773    }
774
775    #[cfg(feature = "lz4")]
776    impl Seek for SeekCountingReader {
777        fn seek(&mut self, pos: SeekFrom) -> std::io::Result<u64> {
778            self.seeks.set(self.seeks.get() + 1);
779            self.inner.seek(pos)
780        }
781    }

783    /// Build a synthetic `k`-chunk LZ4 Data.db body plus its `CompressionInfo`.
784    /// Each chunk record is `[LE u32 uncompressed_len][lz4 bytes][BE u32 CRC32]`
785    /// exactly as Cassandra's `CompressedSequentialWriter` lays them out.
786    #[cfg(feature = "lz4")]
787    fn build_multichunk_lz4(
788        k: usize,
789        chunk_len: usize,
790        last_len: usize,
791    ) -> (Vec<u8>, CompressionInfo) {
792        let mut data_db: Vec<u8> = Vec::new();
793        let mut offsets: Vec<u64> = Vec::new();
794        let mut data_length: u64 = 0;
795        for i in 0..k {
796            offsets.push(data_db.len() as u64);
797            let unclen = if i + 1 == k { last_len } else { chunk_len };
798            let uncompressed: Vec<u8> = (0..unclen).map(|j| ((i * 7 + j) % 251) as u8).collect();
799            data_length += unclen as u64;
800            let mut payload = (unclen as u32).to_le_bytes().to_vec();
801            payload.extend_from_slice(&lz4_flex::compress(&uncompressed));
802            let crc = crc32fast::hash(&payload);
803            data_db.extend_from_slice(&payload);
804            data_db.extend_from_slice(&crc.to_be_bytes());
805        }
806        let info = CompressionInfo {
807            algorithm: "LZ4Compressor".to_string(),
808            chunk_length: chunk_len as u32,
809            data_length,
810            chunk_offsets: offsets,
811            option_pairs: vec![],
812            max_compressed_length: i32::MAX as u32,
813        };
814        (data_db, info)
815    }

817    /// Issue #1586: reading every chunk of a K-chunk file must NOT re-probe the
818    /// immutable file size with a `seek(End)`/`seek(Start)` dance on each chunk.
819    /// A full `read_all_data` scan must issue `<= K + O(1)` seeks, not the ~4*K
820    /// seeks the per-chunk size probe produced. On unfixed code this asserts
821    /// ~4*K and FAILS; after threading the cached file size it passes.

913    /// Issue #1569 (B3), real LRU eviction: at capacity 2, access A, B, A, then
914    /// insert C. B (least recently used) must be evicted while A (re-accessed) and
915    /// C (just inserted) survive. RED on old code, whose `keys().next()` eviction
916    /// could drop A instead of B.
917    #[cfg(feature = "lz4")]
918    #[test]
919    fn eviction_is_real_lru_not_arbitrary() {
920        let (data_db, info) = build_multichunk_lz4(3, 200, 200);
921        let mut dec = ChunkDecompressor::new(info, CassandraVersion::V5_0Release).unwrap();
922        // Force capacity 2 for a deterministic A,B,A,C test (in-module test can set
923        // the private cache field directly).
924        dec.chunk_cache = LruCache::new(NonZeroUsize::new(2).expect("capacity 2"));
925        let mut reader = std::io::Cursor::new(data_db);
926
927        // A = chunk 0, B = chunk 1, C = chunk 2.
928        let _a = dec.get_decompressed_chunk(&mut reader, 0).expect("A cold");
929        let _b = dec.get_decompressed_chunk(&mut reader, 1).expect("B cold");
930        let _a2 = dec.get_decompressed_chunk(&mut reader, 0).expect("A hit"); // bump A recency
931        assert_eq!(
932            dec.decompress_call_count(),
933            2,
934            "A,B decompressed; A re-access is a hit"
935        );
936
937        // Insert C → over capacity → evict LRU, which is now B (A was just accessed).
938        let _c = dec.get_decompressed_chunk(&mut reader, 2).expect("C cold");
939        assert_eq!(dec.decompress_call_count(), 3, "C is a fresh decompress");
940
941        let calls = dec.decompress_call_count();
942        // A survives (recently used) → hit, no new decompress.
943        let _ = dec
944            .get_decompressed_chunk(&mut reader, 0)
945            .expect("A still resident");
946        assert_eq!(
947            dec.decompress_call_count(),
948            calls,
949            "A (recently used) must survive"
950        );
951        // C survives (just inserted) → hit.
952        let _ = dec
953            .get_decompressed_chunk(&mut reader, 2)
954            .expect("C still resident");
955        assert_eq!(
956            dec.decompress_call_count(),
957            calls,
958            "C (just inserted) must survive"
959        );
960        // B was evicted → miss → re-decompress.
961        let _ = dec
962            .get_decompressed_chunk(&mut reader, 1)
963            .expect("B re-read");
964        assert_eq!(
965            dec.decompress_call_count(),
966            calls + 1,
967            "B (least recently used) must have been evicted"
968        );
969    }
