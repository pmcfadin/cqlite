1//! Cross-SSTable scan ordering for [`SSTableManager`](super::SSTableManager)
2//! (issue #1580, Epic D4).
3//!
4//! # The ordering contract (the oracle)
5//!
6//! Cassandra's default `Murmur3Partitioner` orders partitions by their **token**
7//! — the Murmur3 hash of the partition-key bytes — with equal-token ties broken by
8//! raw key bytes. A full scan / `sstabledump` therefore emits partitions in
9//! ascending token order, which in general is **not** the lexicographic order of
10//! the raw partition-key bytes (e.g. UUID `0x22…` sorts *before* `0x11…` by token).
11//!
12//! Each per-reader scan already returns rows in this exact order (the reader's
13//! `sort_by_token_order`). The manager previously concatenated every reader's
14//! output and RE-SORTED the whole concatenation by raw `RowKey` bytes
15//! (`sort_by(|a, b| a.0.cmp(&b.0))`), which both produced the wrong global order
16//! and cost a full O(n log n) sort on the async worker.
17//!
18//! # What this module provides
19//!
20//! * [`kway_merge_token_order`] — a k-way merge over the per-reader token-ordered
21//!   streams that emits the merged stream in token order in **O(n log k)** key
22//!   comparisons (k = number of readers), with early-exit once a pushed-down
23//!   `limit` is satisfied. This replaces the concat + O(n log n) re-sort.
24//! * [`sort_by_token_order`] — a stable token-order sort for the few paths that
25//!   have already materialized a merged `Vec` (the multi-generation
26//!   `KWayMerger` paths); it corrects the old raw-byte sort while remaining a
27//!   no-op when the input is already token-ordered.
28//!
29//! # Instrumentation
30//!
31//! Key-comparison counting used by the O(n log k) no-full-sort regression test is a
32//! **per-thread** counter ([`SCAN_KEY_COMPARISONS`], compiled only under `test`/the
33//! `metrics` feature) exposed to tests via [`kway_merge_counting`], which resets and
34//! reads it within a single call. Because it is thread-local rather than a
35//! process-global atomic, concurrent tests on cargo's multithreaded runner can never
36//! race on it, and the production merge path pays nothing (the increment is not even
37//! compiled in). [`MANAGER_SCAN_FULL_SORTS`] counts materialized full-vector sorts.
38

50/// Number of materialized full-vector token-order sorts performed by
51/// [`sort_by_token_order`] since process start.
52///
53/// Only the `write-support` cross-generation merge paths materialize a merged
54/// `Vec` and call [`sort_by_token_order`], so this counter — and the function —
55/// are compiled only under that feature; the minimal build carries neither
56/// (they would otherwise be dead code and fail `-D warnings`).
57#[cfg(feature = "write-support")]
58pub(crate) static MANAGER_SCAN_FULL_SORTS: AtomicU64 = AtomicU64::new(0);

60// Per-thread key-comparison counter for `kway_merge_token_order`, incremented by
61// `Candidate::cmp`. Thread-local (not a process-global atomic) so concurrent
62// tests on cargo's multithreaded runner never race on it; compiled only under
63// `test` or the `metrics` feature, so production `Candidate::cmp` pays nothing.
64#[cfg(any(test, feature = "metrics"))]
65thread_local! {
66    static SCAN_KEY_COMPARISONS: std::cell::Cell<u64> = const { std::cell::Cell::new(0) };
67}

69/// Heap element pairing an ordering key ([`Candidate`]) with its opaque payload
70/// `T`. Ordering delegates entirely to the [`Candidate`] so no `T: Ord` bound is
71/// needed — the row value/metadata never participates in the merge order.
72struct HeapItem<T> {
73    cand: Candidate,
74    val: T,
75}
76
77impl<T> PartialEq for HeapItem<T> {
78    fn eq(&self, other: &Self) -> bool {
79        self.cand.cmp(&other.cand) == Ordering::Equal
80    }
81}
82impl<T> Eq for HeapItem<T> {}
83impl<T> PartialOrd for HeapItem<T> {
84    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
85        Some(self.cmp(other))
86    }
87}
88impl<T> Ord for HeapItem<T> {
89    fn cmp(&self, other: &Self) -> Ordering {
90        self.cand.cmp(&other.cand)
91    }
92}

94/// The ordering key for one reader stream's current head: its precomputed token
95/// (hashed once per row, not per comparison), partition key, and reader index.
96/// Ordered ascending `(token, key_bytes, reader_idx)`; `reader_idx` breaks ties so
97/// equal-`(token, key)` rows from different readers emit in reader order,
98/// reproducing the stable concat order the old raw-byte `sort_by` gave. Under
99/// `test`/`metrics`, every [`Candidate::cmp`] bumps the per-thread
100/// [`SCAN_KEY_COMPARISONS`] counter; production builds compile no counting at all.
101struct Candidate {
102    token: i64,
103    key: RowKey,
104    reader_idx: usize,
105}
106
107impl Candidate {
108    fn cmp(&self, other: &Self) -> Ordering {
109        #[cfg(any(test, feature = "metrics"))]
110        SCAN_KEY_COMPARISONS.with(|c| c.set(c.get().wrapping_add(1)));
111        self.token
112            .cmp(&other.token)
113            .then_with(|| self.key.cmp(&other.key))
114            .then_with(|| self.reader_idx.cmp(&other.reader_idx))
115    }
116}

118/// K-way merge over per-reader token-ordered streams.
119///
120/// `per_reader[i]` MUST already be sorted in ascending Cassandra token order (with
121/// equal-token ties by raw key bytes and clustering rows of one partition
122/// contiguous) — which every `SSTableReader` scan guarantees via
123/// `sort_by_token_order`. The output is the globally token-ordered concatenation.
124///
125/// Cost: **O(n log k)** key comparisons (k = number of non-empty streams), versus
126/// the O(n log n) full re-sort it replaces. When `limit` is `Some(n)`, the merge
127/// stops after emitting `n` rows (D1 pushed-down-limit early-exit).
128pub(crate) fn kway_merge_token_order<T>(
129    per_reader: Vec<Vec<(RowKey, T)>>,
130    limit: Option<usize>,
131) -> Vec<(RowKey, T)> {
132    // Fast paths avoid heap setup for the overwhelmingly common cases.
133    let non_empty = per_reader.iter().filter(|s| !s.is_empty()).count();
134    if non_empty <= 1 {
135        // Zero or one contributing stream: it is already in token order. Just
136        // take it (the reader emitted it sorted) and apply the limit.
137        let mut out: Vec<(RowKey, T)> = per_reader
138            .into_iter()
139            .find(|s| !s.is_empty())
140            .unwrap_or_default();
141        if let Some(lim) = limit {
142            out.truncate(lim);
143        }
144        return out;
145    }
146
147    let total: usize = per_reader.iter().map(|s| s.len()).sum();
148    let cap = limit.map_or(total, |l| l.min(total));
149    if cap == 0 {
150        // `limit == Some(0)`: emit nothing. Guard here because the in-loop
151        // `out.len() == cap` early-exit is only checked *after* the first push, so
152        // it would never fire for cap == 0 and would leak the entire result set.
153        return Vec::new();
154    }
155
156    // Convert each stream into a forward iterator so we can pull heads lazily.
157    let mut iters: Vec<std::vec::IntoIter<(RowKey, T)>> =
158        per_reader.into_iter().map(|s| s.into_iter()).collect();
159
160    // Min-heap over the current head of each stream. `Reverse` turns the max-heap
161    // into a min-heap; the payload rides inside `HeapItem` so no `T: Ord` bound.
162    let mut heap: BinaryHeap<std::cmp::Reverse<HeapItem<T>>> =
163        BinaryHeap::with_capacity(iters.len());
164    for (idx, it) in iters.iter_mut().enumerate() {
165        if let Some((key, val)) = it.next() {
166            let token = cassandra_murmur3_token(key.as_bytes());
167            heap.push(std::cmp::Reverse(HeapItem {
168                cand: Candidate {
169                    token,
170                    key,
171                    reader_idx: idx,
172                },
173                val,
174            }));
175        }
176    }
177
178    let mut out: Vec<(RowKey, T)> = Vec::with_capacity(cap);
179    while let Some(std::cmp::Reverse(HeapItem { cand, val })) = heap.pop() {
180        let idx = cand.reader_idx;
181        out.push((cand.key, val));
182        if out.len() == cap {
183            break; // pushed-down limit satisfied — stop early
184        }
185        if let Some((key, val)) = iters[idx].next() {
186            let token = cassandra_murmur3_token(key.as_bytes());
187            heap.push(std::cmp::Reverse(HeapItem {
188                cand: Candidate {
189                    token,
190                    key,
191                    reader_idx: idx,
192                },
193                val,
194            }));
195        }
196    }
197    out
198}

200/// Stable in-place token-order sort of an already-materialized result vector, then
201/// truncate to `limit`.
202///
203/// `key_of` extracts each element's partition `RowKey` (the first tuple field for
204/// both the plain `(RowKey, ScanRow)` and the metadata `(RowKey, ScanRow, Meta)`
205/// paths). Elements are ordered by ascending `(token, key_bytes)`; the sort is
206/// stable, so equal-`(token, key)` rows (a partition's clustering rows) keep their
207/// input order.
208///
209/// Used by the multi-generation `KWayMerger` read paths, which hand back a single
210/// merged `Vec` (already token-ordered by the merger): this replaces their prior
211/// raw-byte `sort_by`, so a stray raw-byte ordering can never leak out, while the
212/// sort is effectively a no-op when the input is already ordered. Computes each
213/// key's token once to avoid recomputation inside the comparator.
214///
215/// Compiled only under `write-support`: its sole call sites are the
216/// cross-generation merge paths, which are themselves `#[cfg(feature =
217/// "write-support")]`. Without the gate this is dead code in the minimal build
218/// and fails `-D warnings`.
219#[cfg(feature = "write-support")]
220pub(crate) fn sort_by_token_order<E>(
221    results: &mut Vec<E>,
222    limit: Option<usize>,
223    key_of: impl Fn(&E) -> &RowKey,
224) {
225    MANAGER_SCAN_FULL_SORTS.fetch_add(1, AtomicOrdering::Relaxed);
226    let mut tagged: Vec<(i64, E)> = results
227        .drain(..)
228        .map(|e| {
229            let t = cassandra_murmur3_token(key_of(&e).as_bytes());
230            (t, e)
231        })
232        .collect();
233    tagged.sort_by(|a, b| a.0.cmp(&b.0).then_with(|| key_of(&a.1).cmp(key_of(&b.1))));
234    results.extend(tagged.into_iter().map(|(_, e)| e));
235    if let Some(lim) = limit {
236        results.truncate(lim);
237    }
238}

244    /// Runs [`kway_merge_token_order`] and returns the number of key comparisons it
245    /// performed on the current thread alongside the merged output. Resets and reads
246    /// the per-thread [`SCAN_KEY_COMPARISONS`] counter within this single call, so
247    /// the count is a purely local value — concurrent tests on other threads cannot
248    /// perturb it, eliminating the cross-test data race of a process-global counter.
249    fn kway_merge_counting<T>(
250        per_reader: Vec<Vec<(RowKey, T)>>,
251        limit: Option<usize>,
252    ) -> (Vec<(RowKey, T)>, u64) {
253        SCAN_KEY_COMPARISONS.with(|c| c.set(0));
254        let out = kway_merge_token_order(per_reader, limit);
255        let count = SCAN_KEY_COMPARISONS.with(|c| c.get());
256        (out, count)
257    }
258
259    // UUID keys whose token order differs from raw-byte order (pinned in
260    // `util::cassandra_murmur3` tests): token(0x22) < token(0x11) < token(0x33).
261    fn uuid_key(b: u8) -> RowKey {
262        RowKey::new(vec![b; 16])
263    }
264
265    fn keys_in_order<T>(rows: &[(RowKey, T)]) -> Vec<u8> {
266        let mut out = Vec::new();
267        for (k, _) in rows {
268            let first = k.as_bytes()[0];
269            if out.last() != Some(&first) {
270                out.push(first);
271            }
272        }
273        out
274    }

276    #[test]
277    fn merge_emits_token_order_across_readers() {
278        // Two readers, each internally token-ordered. Reader A: 0x22, 0x33.
279        // Reader B: 0x11. Global token order must be 0x22, 0x11, 0x33.
280        let reader_a = vec![(uuid_key(0x22), 1u32), (uuid_key(0x33), 3u32)];
281        let reader_b = vec![(uuid_key(0x11), 2u32)];
282        let merged = kway_merge_token_order(vec![reader_a, reader_b], None);
283        assert_eq!(keys_in_order(&merged), vec![0x22, 0x11, 0x33]);
284    }

333    /// The no-full-sort assertion (issue #1580): the k-way merge over k readers
334    /// must use O(n log k) key comparisons, NOT the O(n log n) of a full re-sort.
335    /// With k = 4 and n = 4000, n·log2(k) = 8000 while n·log2(n) ≈ 47726 — the
336    /// bound below (≈ 4·n·ceil(log2 k)) sits well under n·log2(n).
337    #[test]
338    fn merge_comparison_count_is_n_log_k_not_n_log_n() {
339        let k = 4usize;
340        let per_stream = 1000usize;
341        let n = k * per_stream;
342
343        // k readers, each with `per_stream` distinct token-ordered partitions.
344        // Distinct 4-byte big-endian int keys, then per-reader token-sorted so the
345        // merge input honours its precondition.
346        let mut per_reader: Vec<Vec<(RowKey, u32)>> = Vec::with_capacity(k);
347        let mut next: i32 = 0;
348        for _ in 0..k {
349            let mut stream: Vec<(RowKey, u32)> = Vec::with_capacity(per_stream);
350            for _ in 0..per_stream {
351                let key = RowKey::new(next.to_be_bytes().to_vec());
352                stream.push((key, next as u32));
353                next += 1;
354            }
355            stream.sort_by_key(|(k, _)| cassandra_murmur3_token(k.as_bytes()));
356            per_reader.push(stream);
357        }
358
359        let (merged, cmps) = kway_merge_counting(per_reader, None);
360
361        assert_eq!(merged.len(), n, "merge must emit every row");
362
363        // Output is globally token-ordered.
364        for w in merged.windows(2) {
365            let ta = cassandra_murmur3_token(w[0].0.as_bytes());
366            let tb = cassandra_murmur3_token(w[1].0.as_bytes());
367            assert!(ta <= tb, "merge output must be ascending token order");
368        }
369
370        let log2_k = (usize::BITS - (k - 1).leading_zeros()) as u64; // ceil(log2 k)
371        let n_log_k_bound = 4 * (n as u64) * (log2_k + 1);
372        let n_log_n = (n as f64 * (n as f64).log2()) as u64;
373        assert!(
374            cmps <= n_log_k_bound,
375            "k-way merge did {cmps} comparisons; expected O(n log k) ≤ {n_log_k_bound} \
376             (n={n}, k={k}); a full O(n log n) re-sort would be ≈ {n_log_n}"
377        );
378        assert!(
379            cmps < n_log_n,
380            "comparison count {cmps} must be strictly below the O(n log n)≈{n_log_n} \
381             full-sort cost the k-way merge replaces"
382        );
383    }
