1//! The ≠1-generation streaming read path: the lazy per-generation fan-out k-way
2//! merge and the per-row → batch re-chunker (issues #1579/#1592; fail-closed in
3//! #3124).
4//!
5//! # What lives here
6//!
7//! * [`spawn_fanout_merge`] — the token-ordered k-way merge over ONE per-row
8//!   sub-scan per generation, used by `SSTableManager::scan_stream` whenever the
9//!   authoritative cross-generation `KWayMerger` is not applicable (no schema, no
10//!   `write-support`, or a merger-construction fallback). This is the query engine's
11//!   multi-generation full-scan producer.
12//! * [`rechunk_into_batches`] — the adapter that re-chunks that per-row stream into
13//!   the public batched surface (`SSTableManager::scan_stream_batched`).
14//!
15//! # Issue #3124: two discarded `JoinHandle`s and a consumer that trusted a closed
16//! channel
17//!
18//! Both boundaries collapsed a channel DISCONNECT into "the scan finished":
19//!
20//! * the merge task was `tokio::spawn`ed with its `JoinHandle` DROPPED, and
21//!   `scan_stream` returned the bare `mpsc::Receiver`, so a merge task that UNWOUND
22//!   dropped `out_tx` with no error and no terminator — every consumer (the query
23//!   engine's streaming SELECT, the LIMIT-pushdown re-scan, `scan_stream_batched`)
24//!   read that as a complete scan and returned FEWER ROWS WITH NO ERROR;
25//! * the re-chunker read `per_row.recv() == None` as end-of-scan, so a dead merge
26//!   task (or a dead per-generation sub-scan whose death the merge could not see
27//!   either) produced a short batch stream that ENDED CLEANLY.
28//!
29//! Both now carry [`RowScanStream`]/`BatchedScanStream`, which JOIN their producer on
30//! channel close and surface a `JoinError` as an `Error::Internal` — the same
31//! mechanism issue #3106 established for the single-source batched scan, generalised
32//! over the item type rather than re-invented (the machinery lives in
33//! `storage/sstable/reader/data_access/joined_scan_stream.rs`).
34//!
35//! Extracted from `sstable/mod.rs` (campsite rule, epic #1116): that file is far over
36//! the ~800-line source threshold, so the #3124 wiring lands here instead of growing
37//! it.
38

52/// Spawn the lazy per-generation token-ordered k-way merge and return its per-row
53/// stream (issue #1579; fail-closed in #3124).
54///
55/// Emits `(token, key)`-ordered entries — byte-identical to `scan`'s
56/// `sort_by_token_order` — with ties broken by reader index, holding at most one
57/// primed head per generation resident. The returned [`RowScanStream`] owns the merge
58/// task's `JoinHandle`, so a merge task that dies is reported to the caller as an
59/// error instead of a clean end of stream (issue #3124, site 1).
60#[cfg(not(feature = "tombstones"))]
61pub(super) fn spawn_fanout_merge(
62    readers: Vec<Arc<reader::SSTableReader>>,
63    table_id: TableId,
64    start_key: Option<RowKey>,
65    end_key: Option<RowKey>,
66    schema: Option<crate::schema::TableSchema>,
67    buffer_size: usize,
68) -> RowScanStream {
69    let (out_tx, out_rx) = tokio::sync::mpsc::channel(buffer_size.max(1));
70
71    // Issue #3124 (site 1): this task's ONE test-only fault checkpoint needs the
72    // reader identity INSIDE the task, so it is captured as an owned scope here (a
73    // zero-sized no-op in a production build). Any generation's path identifies the
74    // table directory a test scopes to; an empty reader set yields an empty scope,
75    // which matches nothing.
76    let fault_scope = FaultScope::capture(|| {
77        readers
78            .first()
79            .map(|reader| reader.file_path())
80            .unwrap_or_default()
81    });
82
83    // The fan-out k-way merge IS the top-level read operation (issue #1701), and its
84    // meter starts HERE — BEFORE the spawn (roborev round 7). Constructed after it, the
85    // merge task could begin, or finish, before timing began. FORMAT-AGNOSTIC: the
86    // reconciled rows come from possibly mixed BIG/BTI inputs, so no single format label
87    // would be honest at this grain (the rule `catalog::READ_ROWS` documents).
88    let meter = crate::observability::read_metrics::ReadOpMeter::start(None);
89
90    // The `JoinHandle` is RETAINED (issue #3124, site 1): it is what lets the
91    // returned stream tell "the merge finished" apart from "the merge DIED".
92    let task = tokio::spawn(async move {
93        fault_scope.checkpoint(ScanTaskSite::FanoutMerge);
94        // Admission control (issue #1594, F4): this fan-out merge is ONE
95        // top-level scan OPERATION that legitimately needs all N per-generation
96        // sub-scans live AT ONCE (it primes a head from every sub-scan before
97        // draining any). Acquire exactly ONE admission permit here, for the
98        // whole operation, and open each sub-scan `Exempt` (below) so the
99        // sub-scans do NOT each independently admit. If they did, a fan-out to
100        // `N > cap` generations would deadlock: `cap` sub-scans would win
101        // permits and park in backpressure while the rest blocked forever at
102        // `admit`, and this priming loop — waiting on the blocked sub-scans —
103        // would never drain the permit-holders. Held via this RAII guard for
104        // the whole merge; released on every exit. See `scan_admission` docs.
105        let _admission = reader::scan_stream_windowed::scan_admission::admit().await;
106
107        // Open one streaming scan per reader. Each is `Exempt` — the single
108        // permit above covers the whole fan-out operation (issue #1594). Each is a
109        // `RowScanStream`, so a sub-scan task that DIES surfaces here as
110        // `Some(Err(..))` and is forwarded below as a terminal item, instead of the
111        // `None` that used to read as "this generation is exhausted" (issue #3124,
112        // site 2).
113        let mut streams: Vec<RowScanStream> = readers
114            .into_iter()
115            .map(|reader| {
116                reader.scan_stream_admitted(
117                    table_id.clone(),
118                    start_key.clone(),
119                    end_key.clone(),
120                    schema.clone(),
121                    buffer_size,
122                    reader::scan_stream_windowed::scan_admission::ScanAdmission::Exempt,
123                    // Issue #1704: the merged stream below forwards and counts.
124                    reader::ScanErrorReporting::Nested,
125                )
126            })
127            .collect();
128
129        // Prime one head per stream. Each head carries its precomputed
130        // Cassandra Murmur3 token so the merge orders by (token, key) — the
131        // authoritative cross-SSTable order (issue #1580) — and never hashes a
132        // key more than once. Comparing by raw `RowKey` bytes here (as this
133        // path previously did) diverged from `scan`'s token order.
134        let token_of =
135            |key: &RowKey| crate::util::cassandra_murmur3::cassandra_murmur3_token(key.as_bytes());
136        let mut heads: Vec<Option<(i64, RowKey, ScanRow)>> = Vec::with_capacity(streams.len());
137        for stream in streams.iter_mut() {
138            match stream.recv().await {
139                Some(Ok((key, row))) => heads.push(Some((token_of(&key), key, row))),
140                Some(Err(e)) => {
141                    let _ = out_tx.send(Err(e)).await;
142                    return;
143                }
144                None => heads.push(None),
145            }
146        }
147
148        // K-way merge: repeatedly emit the head with the smallest
149        // (token, key), ties broken by reader index to match the stable
150        // token-order merge of `scan`.
151        loop {
152            let mut min_idx: Option<usize> = None;
153            for (i, head) in heads.iter().enumerate() {
154                if let Some((ref token, ref key, _)) = head {
155                    match min_idx {
156                        None => min_idx = Some(i),
157                        Some(m) => {
158                            if let Some((ref min_token, ref min_key, _)) = heads[m] {
159                                if (token, key) < (min_token, min_key) {
160                                    min_idx = Some(i);
161                                }
162                            }
163                        }
164                    }
165                }
166            }
167            let idx = match min_idx {
168                Some(idx) => idx,
169                None => break, // all streams exhausted
170            };
171
172            // Take the winning entry and advance only that stream.
173            let entry = match heads[idx].take() {
174                Some((_, key, row)) => (key, row),
175                None => break, // unreachable: min_idx points to a Some head
176            };
177            match streams[idx].recv().await {
178                Some(Ok((key, row))) => heads[idx] = Some((token_of(&key), key, row)),
179                Some(Err(e)) => {
180                    let _ = out_tx.send(Err(e)).await;
181                    return;
182                }
183                None => {} // stream exhausted; head stays None
184            }
185
186            if out_tx.send(Ok(entry)).await.is_err() {
187                return; // consumer dropped
188            }
189        }
190    });
191
192    // Each per-generation sub-scan is opened `Exempt` and therefore unmeasured, so the
193    // merged stream carries the operation's meter, started above.
194    RowScanStream::new_measured_rows(out_rx, task, meter)
195}

197/// Re-chunk a per-row streaming scan into `BATCH_EMIT_ROWS`-sized `Vec` batches over
198/// a bounded channel (issue #1592). Preserves order and content exactly (FIFO
199/// push/flush) and preserves backpressure: the batch channel is bounded, so a stalled
200/// consumer stops the drain of the per-row source, which stops the upstream scan. The
201/// trailing partial batch is flushed at end. A mid-stream error is forwarded as a
202/// terminal item.
203///
204/// Used for the zero/multi-generation and `tombstones` cases, where a
205/// straight-through single-reader hand-off is not applicable (the per-row source is a
206/// k-way merge / materialized reconciliation, not one reader).
207///
208/// Issue #3124 (site 3): the source is a [`RowScanStream`], not a bare
209/// `mpsc::Receiver`, so `recv() == None` here means "the per-row producer PROVABLY
210/// finished" rather than merely "its sender was dropped" — a dead producer arrives as
211/// `Some(Err(..))` and is forwarded as this stream's terminal error. The re-chunker's
212/// own task handle is likewise retained by the returned `BatchedScanStream`.
213pub(super) fn rechunk_into_batches(
214    mut per_row: RowScanStream,
215    buffer_size: usize,
216) -> reader::BatchedScanStream {
217    use reader::scan_stream_windowed::BATCH_EMIT_ROWS;
218    // Bound the batch channel so its resident-row budget stays comparable to
219    // the per-row surface's `buffer_size`, not `buffer_size * BATCH_EMIT_ROWS`.
220    let cap = buffer_size.div_ceil(BATCH_EMIT_ROWS).max(1);
221    let (tx, rx) = tokio::sync::mpsc::channel(cap);
222    let task = tokio::spawn(async move {
223        let mut batch: Vec<(RowKey, ScanRow)> = Vec::with_capacity(BATCH_EMIT_ROWS);
224        while let Some(item) = per_row.recv().await {
225            match item {
226                Ok(entry) => {
227                    batch.push(entry);
228                    if batch.len() >= BATCH_EMIT_ROWS {
229                        if tx.send(Ok(std::mem::take(&mut batch))).await.is_err() {
230                            return; // consumer dropped
231                        }
232                        batch.reserve(BATCH_EMIT_ROWS);
233                    }
234                }
235                Err(e) => {
236                    // Flush already-received Ok rows BEFORE surfacing the
237                    // error, to match the per-row `scan_stream` guarantee
238                    // that confirmed rows are delivered ahead of a terminal
239                    // error (issue #1143 / #1592). Dropping them here would
240                    // silently lose up to BATCH_EMIT_ROWS-1 rows.
241                    if !batch.is_empty() {
242                        let _ = tx.send(Ok(std::mem::take(&mut batch))).await;
243                    }
244                    let _ = tx.send(Err(e)).await;
245                    return;
246                }
247            }
248        }
249        if !batch.is_empty() {
250            let _ = tx.send(Ok(batch)).await;
251        }
252    });
253    reader::BatchedScanStream::new_over_counted_source(rx, task)
254}
