15/// Validation report for SSTable directory scanning
16#[derive(Debug, Clone, Serialize, Deserialize)]
17pub struct ValidationReport {
18    /// Total number of generations found
19    pub total_generations: usize,
20    /// Number of valid generations (all required components present)
21    pub valid_generations: usize,
22    /// List of validation errors found
23    pub validation_errors: Vec<String>,
24    /// List of TOC.txt inconsistencies
25    pub toc_inconsistencies: Vec<String>,
26    /// List of header inconsistencies across components
27    pub header_inconsistencies: Vec<String>,
28    /// List of corrupted or inaccessible files
29    pub corrupted_files: Vec<String>,
30    /// Detailed component analysis per generation
31    pub component_analysis: Vec<ComponentAnalysis>,
32}

34/// Detailed analysis of components for a single generation
35#[derive(Debug, Clone, Serialize, Deserialize)]
36pub struct ComponentAnalysis {
37    /// Generation number
38    pub generation: u32,
39    /// Format (big, da, etc.)
40    pub format: String,
41    /// Required components present
42    pub required_components_present: Vec<SSTableComponent>,
43    /// Required components missing
44    pub required_components_missing: Vec<SSTableComponent>,
45    /// Optional components present
46    pub optional_components_present: Vec<SSTableComponent>,
47    /// File size analysis
48    pub file_sizes: HashMap<SSTableComponent, u64>,
49    /// Accessibility status
50    pub accessibility_status: HashMap<SSTableComponent, bool>,
51}

53impl ValidationReport {
54    /// Check if the validation passed (no critical errors)
55    pub fn is_valid(&self) -> bool {
56        // Critical errors: validation errors, TOC inconsistencies, corrupted files
57        let has_critical_errors = !self.validation_errors.is_empty()
58            || !self.toc_inconsistencies.is_empty()
59            || !self.corrupted_files.is_empty();
60
61        // Header inconsistencies are only critical if they indicate actual corruption
62        let has_critical_header_issues = self
63            .header_inconsistencies
64            .iter()
65            .any(|inc| inc.contains("corrupted") && !inc.contains("failed to fill whole buffer"));
66
67        !has_critical_errors && !has_critical_header_issues
68    }
69
70    /// Get summary of validation results
71    pub fn summary(&self) -> String {
72        format!(
73            "Validation Summary: {}/{} generations valid, {} errors, {} TOC inconsistencies, {} header issues, {} corrupted files",
74            self.valid_generations,
75            self.total_generations,
76            self.validation_errors.len(),
77            self.toc_inconsistencies.len(),
78            self.header_inconsistencies.len(),
79            self.corrupted_files.len()
80        )
81    }
82
83    /// Get detailed validation report as formatted string
84    pub fn detailed_report(&self) -> String {
85        let mut report = String::new();
86        report.push_str("=== SSTable Directory Validation Report ===\n\n");
87        report.push_str(&format!("Total Generations: {}\n", self.total_generations));
88        report.push_str(&format!("Valid Generations: {}\n", self.valid_generations));
89        report.push_str(&format!(
90            "Success Rate: {:.1}%\n\n",
91            if self.total_generations > 0 {
92                (self.valid_generations as f64 / self.total_generations as f64) * 100.0
93            } else {
94                0.0
95            }
96        ));
97
98        if !self.validation_errors.is_empty() {
99            report.push_str(&format!(
100                "❌ Validation Errors ({}):\n",
101                self.validation_errors.len()
102            ));
103            for error in &self.validation_errors {
104                report.push_str(&format!("  • {}\n", error));
105            }
106            report.push('\n');
107        }
108
109        if !self.toc_inconsistencies.is_empty() {
110            report.push_str(&format!(
111                "📋 TOC Inconsistencies ({}):\n",
112                self.toc_inconsistencies.len()
113            ));
114            for inconsistency in &self.toc_inconsistencies {
115                report.push_str(&format!("  • {}\n", inconsistency));
116            }
117            report.push('\n');
118        }
119
120        if !self.header_inconsistencies.is_empty() {
121            report.push_str(&format!(
122                "🏷️ Header Inconsistencies ({}):\n",
123                self.header_inconsistencies.len()
124            ));
125            for inconsistency in &self.header_inconsistencies {
126                report.push_str(&format!("  • {}\n", inconsistency));
127            }
128            report.push('\n');
129        }
130
131        if !self.corrupted_files.is_empty() {
132            report.push_str(&format!(
133                "💥 Corrupted Files ({}):\n",
134                self.corrupted_files.len()
135            ));
136            for file in &self.corrupted_files {
137                report.push_str(&format!("  • {}\n", file));
138            }
139            report.push('\n');
140        }
141
142        if !self.component_analysis.is_empty() {
143            report.push_str("📊 Component Analysis by Generation:\n");
144            for analysis in &self.component_analysis {
145                report.push_str(&format!(
146                    "\n  Generation {} ({} format):\n",
147                    analysis.generation, analysis.format
148                ));
149                report.push_str(&format!(
150                    "    Required present: {:?}\n",
151                    analysis.required_components_present
152                ));
153                if !analysis.required_components_missing.is_empty() {
154                    report.push_str(&format!(
155                        "    Required missing: {:?}\n",
156                        analysis.required_components_missing
157                    ));
158                }
159                report.push_str(&format!(
160                    "    Optional present: {:?}\n",
161                    analysis.optional_components_present
162                ));
163                report.push_str(&format!(
164                    "    Total file size: {} bytes\n",
165                    analysis.file_sizes.values().sum::<u64>()
166                ));
167            }
168        }
169
170        report
171    }
172}

174/// Enhanced validation with detailed component analysis (M1: gated for post-M1)
175#[cfg(feature = "enhanced-index-validation")]
176pub fn validate_generation_components_enhanced(
177    generation: &SSTableGeneration,
178    analysis: &mut ComponentAnalysis,
179) -> Result<Vec<String>> {
180    let mut issues = Vec::new();
181
182    // Define required components based on format
183    let required_components = match generation.format.as_str() {
184        "big" => vec![
185            SSTableComponent::Data,
186            SSTableComponent::Statistics,
187            SSTableComponent::Index,
188            SSTableComponent::Summary,
189        ],
190        "da" => vec![
191            SSTableComponent::Data,
192            SSTableComponent::Statistics,
193            SSTableComponent::Partitions,
194            SSTableComponent::Rows,
195        ],
196        _ => vec![SSTableComponent::Data, SSTableComponent::Statistics], // Minimal requirements
197    };
198
199    // Check required components
200    for component in &required_components {
201        if generation.components.contains_key(component) {
202            analysis.required_components_present.push(component.clone());
203        } else {
204            analysis.required_components_missing.push(component.clone());
205            issues.push(format!(
206                "Missing required component: {:?} for generation {} (format: {})",
207                component, generation.generation, generation.format
208            ));
209        }
210    }
211
212    // Analyze all present components
213    for (component, path) in &generation.components {
214        // Check if component is optional
215        if !required_components.contains(component) {
216            analysis.optional_components_present.push(component.clone());
217        }
218
219        // File existence and accessibility
220        if !path.exists() {
221            issues.push(format!(
222                "Component file does not exist: {:?} at {:?} (generation {})",
223                component, path, generation.generation
224            ));
225            analysis
226                .accessibility_status
227                .insert(component.clone(), false);
228        } else {
229            match fs::metadata(path) {
230                Ok(metadata) => {
231                    let file_size = metadata.len();
232                    analysis.file_sizes.insert(component.clone(), file_size);
233
234                    // Zero-size file validation
235                    if file_size == 0 {
236                        if component.is_required() {
237                            issues.push(format!(
238                                "Required component file is empty: {:?} at {:?} (generation {})",
239                                component, path, generation.generation
240                            ));
241                        } else {
242                            // Log warning for empty optional files
243                            log::warn!("Optional component file is empty: {:?}", component);
244                        }
245                    }
246
247                    // File readability test
248                    match fs::File::open(path) {
249                        Ok(_) => {
250                            analysis
251                                .accessibility_status
252                                .insert(component.clone(), true);
253                        }
254                        Err(e) => {
255                            issues.push(format!(
256                                "Component file is not readable: {:?} at {:?} - {} (generation {})",
257                                component, path, e, generation.generation
258                            ));
259                            analysis
260                                .accessibility_status
261                                .insert(component.clone(), false);
262                        }
263                    }
264                }
265                Err(e) => {
266                    issues.push(format!(
267                        "Cannot access component file metadata: {:?} at {:?} - {} (generation {})",
268                        component, path, e, generation.generation
269                    ));
270                    analysis
271                        .accessibility_status
272                        .insert(component.clone(), false);
273                }
274            }
275        }
276    }
277
278    Ok(issues)
279}

282pub fn validate_generation_components(generation: &SSTableGeneration) -> Result<Vec<String>> {
283    #[cfg(feature = "enhanced-index-validation")]

305                issues.push(format!(
306                    "Missing required component {:?} in generation {}",
307                    component, generation.generation
308                ));
309            }
310        }
311
312        Ok(issues)
313    }

316/// Enhanced TOC validation with detailed component analysis (M1: gated for post-M1)
317#[cfg(feature = "enhanced-index-validation")]
318pub fn validate_toc_consistency_enhanced(generation: &SSTableGeneration) -> Result<Vec<String>> {
319    let mut inconsistencies = Vec::new();
320
321    if let Some(toc_path) = generation.components.get(&SSTableComponent::TOC) {
322        match parse_toc_file_detailed(toc_path) {
323            Ok((toc_components, unknown_components)) => {
324                // Validate TOC structure and completeness
325                if toc_components.is_empty() && unknown_components.is_empty() {
326                    inconsistencies
327                        .push("TOC.txt is empty or contains no valid components".to_string());
328                    return Ok(inconsistencies);
329                }
330
331                // Check for unknown/invalid components in TOC (like "NonExistent.db")
332                if !unknown_components.is_empty() {
333                    inconsistencies.push(format!(
334                        "TOC.txt lists unknown/invalid components: [{}]",
335                        unknown_components.join(", ")
336                    ));
337                }
338
339                // Check that all TOC components have corresponding files
340                let mut missing_files = Vec::new();
341                for toc_component in &toc_components {
342                    if !generation.components.contains_key(toc_component) {
343                        missing_files.push(format!("{:?}", toc_component));
344                    }
345                }
346
347                if !missing_files.is_empty() {
348                    inconsistencies.push(format!(
349                        "TOC.txt lists components without corresponding files: [{}]",
350                        missing_files.join(", ")
351                    ));
352                }
353
354                // Check that all files are listed in TOC (except TOC itself)
355                let mut unlisted_files = Vec::new();
356                for (file_component, path) in &generation.components {
357                    if *file_component != SSTableComponent::TOC
358                        && !toc_components.contains(file_component)
359                    {
360                        // Additional check: ensure the file actually exists before reporting as unlisted
361                        if path.exists() {
362                            unlisted_files.push(format!("{:?}", file_component));
363                        }
364                    }
365                }
366
367                if !unlisted_files.is_empty() {
368                    inconsistencies.push(format!(
369                        "Files exist but not listed in TOC.txt: [{}]",
370                        unlisted_files.join(", ")
371                    ));
372                }
373
374                // Validate expected components for format
375                let expected_components = match generation.format.as_str() {
376                    "big" => vec![
377                        SSTableComponent::Data,
378                        SSTableComponent::Statistics,
379                        SSTableComponent::Index,
380                        SSTableComponent::Summary,
381                        SSTableComponent::TOC,
382                    ],
383                    "da" => vec![
384                        SSTableComponent::Data,
385                        SSTableComponent::Statistics,
386                        SSTableComponent::Partitions,
387                        SSTableComponent::Rows,
388                        SSTableComponent::TOC,
389                    ],
390                    _ => vec![
391                        SSTableComponent::Data,
392                        SSTableComponent::Statistics,
393                        SSTableComponent::TOC,
394                    ],
395                };
396
397                let mut missing_expected = Vec::new();
398                for expected in &expected_components {
399                    if !toc_components.contains(expected) {
400                        missing_expected.push(format!("{:?}", expected));
401                    }
402                }
403
404                if !missing_expected.is_empty() {
405                    inconsistencies.push(format!(
406                        "TOC.txt missing expected components for {} format: [{}]",
407                        generation.format,
408                        missing_expected.join(", ")
409                    ));
410                }
411
412                // Check for duplicate entries in TOC
413                let mut seen_components = std::collections::HashSet::new();
414                let mut duplicates = Vec::new();
415                for component in &toc_components {
416                    if !seen_components.insert(component) {
417                        duplicates.push(format!("{:?}", component));
418                    }
419                }
420
421                if !duplicates.is_empty() {
422                    inconsistencies.push(format!(
423                        "TOC.txt contains duplicate entries: [{}]",
424                        duplicates.join(", ")
425                    ));
426                }
427            }
428            Err(e) => {
429                inconsistencies.push(format!("Failed to parse TOC.txt: {}", e));
430            }
431        }
432    } else {
433        inconsistencies.push(format!(
434            "No TOC.txt file found for generation {} (format: {})",
435            generation.generation, generation.format
436        ));
437    }
438
439    Ok(inconsistencies)
440}

467/// Validate file integrity by checking basic file properties
468pub(crate) fn validate_file_integrity(path: &Path) -> Result<bool> {
469    if !path.exists() {
470        return Ok(false);
471    }
472
473    let _metadata = fs::metadata(path)
474        .map_err(|e| Error::storage(format!("Cannot read metadata for {:?}: {}", path, e)))?;
475
476    // Check if file is readable
477    let _file = fs::File::open(path)
478        .map_err(|e| Error::storage(format!("Cannot open file for reading: {:?}: {}", path, e)))?;
479
480    // For now, consider file valid if it exists and is readable
481    // In a full implementation, this could include checksums, format validation, etc.
482    Ok(true)
483}

491/// Test all SSTable directories in the test environment
492pub fn test_all_directories<P: AsRef<Path>>(
493    base_path: P,
494) -> Result<Vec<(String, ValidationReport)>> {
495    let base_path = base_path.as_ref();
496    let mut results = Vec::new();
497
498    if !base_path.exists() {
499        return Err(Error::invalid_path(format!(
500            "Base test path does not exist: {:?}",
501            base_path
502        )));
503    }
504
505    let entries = fs::read_dir(base_path).map_err(|e| {
506        Error::storage(format!(
507            "Cannot read test directory: {:?}: {}",
508            base_path, e
509        ))
510    })?;
511
512    for entry in entries {
513        let entry = entry?;
514        let entry_path = entry.path();
515
516        if entry_path.is_dir() {
517            if let Some(dir_name) = entry_path.file_name().and_then(|n| n.to_str()) {
518                // Skip hidden directories and non-SSTable directories
519                if !dir_name.starts_with('.') && dir_name.contains('-') {
520                    match test_directory_validation(&entry_path) {
521                        Ok(report) => {
522                            results.push((dir_name.to_string(), report));
523                        }
524                        Err(e) => {
525                            log::error!("Failed to validate directory {}: {}", dir_name, e);
526                        }
527                    }
528                }
529            }
530        }
531    }
532
533    Ok(results)
