1//! Cross-generation read reconciliation for the SSTable manager (issues #883,
2//! #885, #957, #1579).
3//!
4//! When a table directory holds more than one SSTable generation, plain
5//! concatenation of each reader's live rows is wrong: the same
6//! `(partition, clustering)` row can appear in several generations, and a
7//! row/cell tombstone in a newer generation suppresses only its OWN generation's
8//! copy — so concatenation duplicates overwritten rows and resurrects rows deleted
9//! in a later generation. These helpers reconcile across generations with the
10//! authoritative [`KWayMerger`](crate::storage::write_engine::KWayMerger) — the
11//! same last-write-wins + tombstone-shadowing logic compaction uses — so the read
12//! path returns Cassandra's merged, deduplicated, tombstone-honouring result.
13//!
14//! Three drivers, ONE reconciliation kernel (the merger's `step()`):
15//!
16//! - [`merge_generations_for_read`] — materializing plain read (`scan`,
17//!   partition-targeted point reads). Collects the reconciled rows into a `Vec`
18//!   in Cassandra token order.
19//! - [`merge_generations_for_read_with_metadata`] — the `WRITETIME`/`TTL`
20//!   projection sibling; additionally surfaces the winning cell's per-cell write
21//!   metadata.
22//! - [`stream_generations_for_read`] — the STREAMING plain read (`scan_stream`,
23//!   issue #1579 / D3). Feeds each stepped partition straight into a bounded
24//!   channel instead of collecting the whole table, so live heap is O(one
25//!   partition + channel) and time-to-first-row is O(first partition), not
26//!   O(full merge).
27//!
28//! All three yield partitions in the merger's `DecoratedKey` order = `(token,
29//! key)`, which is byte-identical to [`scan_merge::sort_by_token_order`]. The
30//! materializing paths still apply that stable sort as a no-op guard; the
31//! streaming path relies on the merger's order directly (issue #1579).
32//!
33//! Extracted from `sstable/mod.rs` (issue #1116 campsite split): behaviour of the
34//! two materializing helpers is unchanged apart from the new `target_key`
35//! partition-targeting parameter (issue #1579, point-read path).
36//!
37//! The whole module is gated on `write-support` at its `mod` declaration in the
38//! parent (`sstable/mod.rs`).
39//!
40//! NOTE (#1116 file-size): already over the 800-line target on `origin/main`; the
41//! #2063 admission acquires add a few lines. Splitting is tracked under epic #1116.

66/// Post-merge read-time visibility for the multi-generation read path (issue #1849).
67///
68/// A `SELECT` over a table directory with more than one SSTable generation routes
69/// through the [`KWayMerger`], which performs cross-generation last-write-wins +
70/// tombstone RECONCILIATION but NOT read-time TTL expiry / partition-deletion
71/// visibility. Historically that let a multi-gen read return TTL-expired (or
72/// partition-shadowed) cells as live, even after #1741 fixed the single-gen path.
73///
74/// This runs the merger's already-reconciled output through the SAME single-gen
75/// [`PartitionShadow`] per-cell decision (`cell_shadowed_or_expired`) and row-level
76/// partition-shadow decision (`merged_row_shadowed_by_partition`), so there is ONE
77/// read-visibility implementation across single- and multi-generation reads. It adds
78/// READ visibility only — reconciliation and the compaction/write path are untouched
79/// (AC6). Range-tombstone shadowing of covered older cells is already applied by the
80/// merger (#933); read-time cell/row TTL expiry is the gap this closes.
81struct ReadShadow {
82    /// Read-time TTL clock (epoch seconds), captured ONCE per scan so a scan crossing
83    /// an expiration-second boundary decides every row with the same `now` — matching
84    /// the single-gen parser's per-scan clock capture (issue #1849 AC4). Honours the
85    /// `CQLITE_TTL_NOW_OVERRIDE_SECS` debug test seam via [`now_epoch_secs`].
86    now_secs: i64,
87    /// Names of the primary-key (partition + clustering) columns, surfaced by the
88    /// compaction reader as pseudo-cells. Excluded from the row-level DATA-cell
89    /// timestamp aggregate so only real data cells drive the partition-shadow row
90    /// decision — matching the single-gen fold, which ignores pk/ck pseudo-cells.
91    key_columns: HashSet<String>,
92}
93
94impl ReadShadow {
95    fn new(schema: &crate::schema::TableSchema, now_secs: i64) -> Self {
96        let mut key_columns =
97            HashSet::with_capacity(schema.partition_keys.len() + schema.clustering_keys.len());
98        for k in &schema.partition_keys {
99            key_columns.insert(k.name.clone());
100        }
101        for k in &schema.clustering_keys {
102            key_columns.insert(k.name.clone());
103        }
104        Self {
105            now_secs,
106            key_columns,
107        }
108    }
109
110    /// Filter one merged `RowData::Live` row's cells for read visibility given the
111    /// partition-tombstone `cover` (`markedForDeleteAt` µs, or `None`). Returns `None`
112    /// when the WHOLE row is hidden (partition-shadowed), else the surviving cells
113    /// with cell tombstones plus TTL-expired / partition-shadowed data cells dropped.
114    fn filter_live(&self, cover: Option<i64>, cells: Vec<CellData>) -> Option<Vec<CellData>> {
115        let now = self.now_secs;
116        let mut kept = Vec::with_capacity(cells.len());
117        // Fold over REAL data cells (kept AND dropped) so a partition tombstone that
118        // shadows every data cell is recognised at the row level — mirrors the
119        // single-gen `agg_max_cell_ts` fold in `row_data.rs`.
120        let mut max_data_ts: Option<i64> = None;
121        for cell in cells {
122            // A cell tombstone is never live data (a deleted column is absent).
123            if matches!(cell.value, Value::Tombstone(_)) {
124                continue;
125            }
126            // Primary-key (partition + clustering) pseudo-cells are STRUCTURAL: the
127            // compaction reader surfaces clustering columns as simple cells and both
128            // the single-gen read loop and the merger's `apply_partition_shadowing`
129            // leave them untouched (never subjected to the read-time shadow/expiry
130            // drop). Keep them verbatim and out of the `max_data_ts` fold, so a row
131            // kept alive by a newer DATA cell under a partition tombstone never loses
132            // its clustering-key value (a pseudo-cell whose own `ts <= cover` must NOT
133            // be stripped) — otherwise the multi-gen path would emit a malformed row
134            // diverging from the single-gen path (issue #1849; roborev multi-gen finding).
135            if self.key_columns.contains(&cell.column) {
136                kept.push(cell);
137                continue;
138            }
139            let eff_ts = Some(cell.timestamp);
140            let eff_exp = cell_expiry_secs(&cell);
141            let dropped = PartitionShadow::cell_shadowed_or_expired(cover, now, eff_ts, eff_exp);
142            if let Some(t) = eff_ts {
143                max_data_ts = Some(max_data_ts.map_or(t, |m| m.max(t)));
144            }
145            if !dropped {
146                kept.push(cell);
147            }
148        }
149        if merged_row_shadowed_by_partition(cover, max_data_ts) {
150            return None;
151        }
152        Some(kept)
153    }
154}

156/// The read-time expiry instant (epoch seconds) of a merged cell, or `None` when it
157/// is not an expiring cell. An expiring cell's `localDeletionTime` is its
158/// `localExpirationTime`; it is reinterpreted UNSIGNED so a post-2038 expiry stored
159/// as a negative `i32` bit pattern (oa/da `hasUIntDeletionTime`) is not wrapped
160/// negative and wrongly treated as long-expired — matching the single-gen unsigned
161/// LDT handling in `row_data.rs` (issue #1849 AC4).
162fn cell_expiry_secs(cell: &CellData) -> Option<i64> {
163    cell.ttl?;
164    cell.local_deletion_time.map(|s| (s as u32) as i64)
165}

167/// The partition-level `markedForDeleteAt` (µs) carried by the synthetic
168/// partition-tombstone carrier entry of a merged partition (issue #1072), or `None`
169/// when the partition has no tombstone. Used as the read-side partition-shadow cover
170/// across the partition's rows (issue #1849).
171fn partition_cover(rows: &[MergeEntry]) -> Option<i64> {
172    rows.iter()
173        .find_map(|e| e.partition_deletion.map(|(mfda, _ldt)| mfda))
174}

176/// Convert one reconciled partition's merge rows into the live scan rows the read
177/// path emits: drop cell tombstones, drop row tombstones, drop rows left empty.
178///
179/// The single emission kernel shared by the materializing plain merge and the
180/// streaming driver so tombstone filtering can never drift between them (issue
181/// #1334: emit the interned-name `ScanRow` carrier the read path consumes).
182fn partition_live_rows(
183    row_key: &RowKey,
184    rows: Vec<MergeEntry>,
185    shadow: &ReadShadow,
186) -> Vec<(RowKey, ScanRow)> {
187    // Issue #1849: partition-tombstone cover for this partition's rows, applied as a
188    // read-side shadow floor alongside per-cell read-time TTL expiry (post-merge).
189    let cover = partition_cover(&rows);
190    let mut out = Vec::new();
191    for entry in rows {
192        match entry.row_data {
193            RowData::Live { cells } => {
194                // Read-time visibility: drop cell tombstones + TTL-expired /
195                // partition-shadowed data cells; `None` hides the whole row.
196                let Some(surviving) = shadow.filter_live(cover, cells) else {
197                    continue;
198                };
199                let row_cells: RowCells = surviving
200                    .into_iter()
201                    .map(|c| (Arc::from(c.column.as_str()), c.value))
202                    .collect();
203                if !row_cells.is_empty() {
204                    out.push((row_key.clone(), ScanRow::Row(row_cells)));
205                }
206            }
207            // Row tombstone: the row is deleted across all generations — suppress.
208            RowData::Tombstone { .. } => {}
209        }
210    }
211    out
212}

214/// Reconcile multiple SSTable generations into the single authoritative live-row
215/// set, returning them materialized in Cassandra token order (issue #883).
216///
217/// This drives the same [`KWayMerger`](crate::storage::write_engine::KWayMerger)
218/// the compaction path uses, so reconciliation is byte-for-byte the
219/// last-write-wins + tombstone-shadowing logic (`merge_partition_rows`): per-cell
220/// LWW by write timestamp, row/cell tombstones shadow older cells, and
221/// fully-deleted rows are dropped. The merger manages its own reader
222/// threads/runtimes internally, so it runs on a blocking task.
223///
224/// `start_key`/`end_key` bound the merged output to the same inclusive
225/// `[start_key, end_key]` key range the per-reader
226/// [`scan`](reader::SSTableReader::scan) applies (skip `key < start`, skip
227/// `key > end`, using `RowKey`'s `Ord`) — Issue #957. The range filter runs
228/// before `limit`, matching the per-reader scan order (range then limit).
229///
230/// `target_key` restricts the output to a SINGLE partition (the point-read path,
231/// issue #1579): keep only the partition whose raw key bytes equal `target_key`
232/// and STOP as soon as it is found. Partition keys are unique in the merger's
233/// output, so once the target is seen nothing later can match — this avoids
234/// converting every other partition to `ScanRow` and decoding past the target,
235/// while remaining byte-identical to the caller's former
236/// `retain(|r| r.key == partition_key)`. `target_key` and the range bounds are
237/// not combined by any current caller (point reads pass `None`/`None` bounds).
238///
239/// With `None` target and `None`/`None` bounds the output is byte-for-byte the
240/// full reconciled set.
241pub(super) async fn merge_generations_for_read(
242    reader_list: &[Arc<reader::SSTableReader>],
243    schema: &crate::schema::TableSchema,
244    start_key: Option<&RowKey>,
245    end_key: Option<&RowKey>,
246    limit: Option<usize>,
247    target_key: Option<&RowKey>,
248) -> Result<Vec<(RowKey, ScanRow)>> {
249    // Issue #2063: one operation-level scan-admission permit; rationale + cancellation
250    // shape in `scan_admission.rs` `# Scope`. Moved into the closure below.
251    let admission = reader::scan_stream_windowed::scan_admission::admit().await;
252
253    // Own the bounds/target so the merge body can use them without borrowing
254    // across the await; cheap clone of the key bytes.
255    let start_key = start_key.cloned();
256    let end_key = end_key.cloned();
257    let target_key = target_key.map(|k| k.as_bytes().to_vec());
258
259    let paths = ordered_generation_paths(reader_list);
260    let schema = schema.clone();
261
262    let mut merged = tokio::task::spawn_blocking(move || -> Result<Vec<(RowKey, ScanRow)>> {
263        let _admission = admission; // #2063: hold across the detached blocking work.
264                                    // Issue #1849: capture the read-time TTL clock ONCE per scan.
265        let shadow = ReadShadow::new(&schema, now_epoch_secs());
266        let mut merger = KWayMerger::new(paths, &schema)?;
267        let mut out = Vec::new();
268        while let MergeStep::Partition { key, rows } = merger.step()? {
269            let row_key = RowKey::new(key.key.clone());
270
271            // Partition-targeted point read (#1579): keep ONLY the target and stop.
272            if let Some(ref target) = target_key {
273                if row_key.as_bytes() != target.as_slice() {
274                    continue;
275                }
276                out.extend(partition_live_rows(&row_key, rows, &shadow));
277                break;
278            }
279
280            // Full / range scan: inclusive [start, end] on `RowKey` order (#957).
281            if let Some(ref start) = start_key {
282                if &row_key < start {
283                    continue;
284                }
285            }
286            if let Some(ref end) = end_key {
287                if &row_key > end {
288                    continue;
289                }
290            }
291            out.extend(partition_live_rows(&row_key, rows, &shadow));
292        }
293        Ok(out)
294    })
295    .await
296    .map_err(|e| crate::Error::Storage(format!("cross-generation read merge task: {e}")))??;
297
298    // Match the plain-scan contract: the merger already emits partitions in
299    // Cassandra token order with clustering rows contiguous within a partition.
300    // Preserve that with a stable TOKEN-order sort (issue #1580), then apply
301    // LIMIT. The sort is a no-op ordering-wise when the input is already
302    // token-ordered; it only guards against a stray divergence.
303    scan_merge::sort_by_token_order(&mut merged, limit, |(k, _)| k);
304    Ok(merged)
305}

307/// Partition-SEEKING sibling of [`merge_generations_for_read`] for the
308/// multi-candidate `WHERE pk = ?` point read (issue #2096).
309///
310/// Where [`merge_generations_for_read`] with a `target_key` still builds the
311/// FULL-SCAN `KWayMerger::new` and sequentially DECODES every partition with token
312/// <= the target before breaking (O(partitions-below-target)), this reuses the
313/// Flight point path's partition-SEEKING merger
314/// ([`build_single_partition_merger_from_readers`](crate::storage::write_engine::build_single_partition_merger_from_readers),
315/// #2207/#2346): each candidate seeks straight to the target partition's `Data.db`
316/// offset (BTI trie / `Index.db`), or fail-safe filter-scans one SSTable when its
317/// index is unavailable, then reconciles through the SAME `KWayMerger`
318/// (`from_row_iterators`, run_index = position). The read-visibility kernel is
319/// IDENTICAL to the materializing helper — the same [`ReadShadow`] (#1849) captured
320/// ONCE, the same [`partition_live_rows`] emission + token-order guard, and
321/// candidates ordered NEWEST→OLDEST like [`ordered_generation_paths`] — so the
322/// output is byte-for-byte `merge_generations_for_read(.., Some(target))`, only
323/// over O(target) work. Blocking builder + `step()` run on `spawn_blocking`.
324#[cfg(all(feature = "write-support", not(feature = "tombstones")))]
325pub(super) async fn seek_merge_generations_for_read(
326    candidates: &[Arc<reader::SSTableReader>],
327    schema: &crate::schema::TableSchema,
328    target_key: &RowKey,
329) -> Result<Vec<(RowKey, ScanRow)>> {
330    use crate::storage::scan_cancel::ScanCancel;
331    use crate::storage::write_engine::merge::build_single_partition_merger_from_readers;
332
333    // Issue #2063: one operation-level scan-admission permit; ONLY call site is the
334    // top-level `scan_partition_clustering`, never nested. Rationale + cancellation
335    // shape in `scan_admission.rs` `# Scope`.
336    let admission = reader::scan_stream_windowed::scan_admission::admit().await;
337
338    // NEWEST→OLDEST like `ordered_generation_paths`, so the seeking merger's
339    // run_index (= position) equals the full-scan merger's LWW tie-break rank.
340    let mut ordered: Vec<Arc<reader::SSTableReader>> = candidates.to_vec();
341    ordered.sort_by(|a, b| b.generation.cmp(&a.generation));
342
343    let schema = schema.clone();
344    let target_bytes = target_key.as_bytes().to_vec();
345
346    let mut merged = tokio::task::spawn_blocking(move || -> Result<Vec<(RowKey, ScanRow)>> {
347        let _admission = admission; // #2063: hold across the detached blocking work.
348                                    // Issue #1849: same read-visibility kernel `merge_generations_for_read`
349                                    // applies (read-time TTL clock captured ONCE), REQUIRED for byte-identity.
350        let shadow = ReadShadow::new(&schema, now_epoch_secs());
351        let keys = [target_bytes.clone()];
352        let Some(mut merger) =
353            build_single_partition_merger_from_readers(ordered, &keys, &schema, ScanCancel::new())?
354        else {
355            return Ok(Vec::new()); // no candidate holds the target
356        };
357        let mut out = Vec::new();
358        while let MergeStep::Partition { key, rows } = merger.step()? {
359            let row_key = RowKey::new(key.key.clone());
360            // A fail-safe filter-scan run could surface a prefix-collision key; keep
361            // only the exact target and stop once seen (partition keys are unique).
362            if row_key.as_bytes() != target_bytes.as_slice() {
363                continue;
364            }
365            out.extend(partition_live_rows(&row_key, rows, &shadow));
366            break;
367        }
368        Ok(out)
369    })
370    .await
371    .map_err(|e| {
372        crate::Error::Storage(format!("seeking cross-generation read merge task: {e}"))
373    })??;
374
375    // Parity guard with the materializing helper: stable TOKEN-order sort (no-op for
376    // a single partition, kept for byte-parity, issue #1580).
377    scan_merge::sort_by_token_order(&mut merged, None, |(k, _)| k);
378    Ok(merged)
379}

381/// Metadata-aware sibling of [`merge_generations_for_read`] for the
382/// `WRITETIME(col)` / `TTL(col)` projection path (Issue #885).
383///
384/// Reconciles multiple SSTable generations with the same
385/// [`KWayMerger`](crate::storage::write_engine::KWayMerger) (per-cell LWW +
386/// row/cell tombstone shadowing) and additionally surfaces the **winning** cell's
387/// per-cell write metadata:
388///
389/// - `write_timestamp_micros` comes straight from the winning `CellData`
390///   (`reconcile_cluster` keeps each surviving cell's own timestamp), so it is the
391///   WRITETIME of the cell that actually won cross-generation LWW.
392/// - `expiration` (TTL) is recovered best-effort from the per-reader
393///   `scan_with_cell_metadata` outputs: for each surviving `(key, column)` we take
394///   the newest reader-surfaced metadata and attach its expiration only when its
395///   timestamp matches the merge winner. Absent/mismatched ⇒ `None`.
396///
397/// `start_key`/`end_key`/`target_key`/`limit` behave exactly as in
398/// [`merge_generations_for_read`], keeping this definitionally in lockstep with
399/// the plain helper. When `target_key` is `Some`, the best-effort TTL scan is also
400/// bounded to that single partition (issue #1579, point-read path).
401pub(super) async fn merge_generations_for_read_with_metadata(
402    reader_list: &[Arc<reader::SSTableReader>],
403    schema: &crate::schema::TableSchema,
404    start_key: Option<&RowKey>,
405    end_key: Option<&RowKey>,
406    limit: Option<usize>,
407    target_key: Option<&RowKey>,
408) -> Result<Vec<(RowKey, ScanRow, HashMap<String, CellWriteMetadata>)>> {
409    // Issue #2063: one operation-level scan-admission permit. Held as an OUTER future
410    // guard across the async per-reader `scan_with_cell_metadata` loop (cancellation
411    // there is clean — no detached blocking work exists yet, so early release is
412    // harmless), THEN MOVED into the `spawn_blocking` merge closure below so the permit
413    // is held until the detached blocking work TERMINATES. This matches the plain/seek
414    // helpers: no phase both runs detached blocking work AND has released the permit.
415    let admission = reader::scan_stream_windowed::scan_admission::admit().await;
416
417    // Own the bounds so the merge body can use them without borrowing across the
418    // await; cheap clone of the key bytes. Mirrors the plain helper.
419    let owned_start = start_key.cloned();
420    let owned_end = end_key.cloned();
421    let target_bytes = target_key.map(|k| k.as_bytes().to_vec());
422
423    // Best-effort TTL source: gather each reader's own per-cell metadata and keep,
424    // per (row-key bytes, column), the entry with the newest write timestamp. The
425    // merger surfaces accurate WRITETIME but no TTL, so this recovers expiration
426    // for the winning cell when the reader format carries it. For a point read the
427    // scan is bounded to the target partition (#1579); otherwise to the read range.
428    let table_id = CqlTableId::from(format!("{}.{}", schema.keyspace, schema.table).as_str());
429    let mut ttl_lookup: HashMap<(Vec<u8>, String), CellWriteMetadata> = HashMap::new();
430    for reader in reader_list {
431        let (ttl_start, ttl_end): (Option<&RowKey>, Option<&RowKey>) = match target_key {
432            Some(t) => (Some(t), Some(t)),
433            None => (owned_start.as_ref(), owned_end.as_ref()),
434        };
435        let per_reader = reader
436            .scan_with_cell_metadata(&table_id, ttl_start, ttl_end, None, Some(schema))
437            .await?;
438        for (row_key, _value, meta) in per_reader {
439            for (column, cell_meta) in meta {
440                ttl_lookup
441                    .entry((row_key.as_bytes().to_vec(), column))
442                    .and_modify(|existing| {
443                        if cell_meta.write_timestamp_micros > existing.write_timestamp_micros {
444                            *existing = cell_meta.clone();
445                        }
446                    })
447                    .or_insert(cell_meta);
448            }
449        }
450    }
451
452    let paths = ordered_generation_paths(reader_list);
453    let merge_schema = schema.clone();
454    let start_key = owned_start;
455    let end_key = owned_end;
456    let target_for_merge = target_bytes;
457
458    let merged_rows = tokio::task::spawn_blocking(move || -> Result<Vec<MergedMetaRow>> {
459        let _admission = admission; // #2063: hold across the detached blocking work.
460                                    // Issue #1849: capture the read-time TTL clock ONCE per scan.
461        let shadow = ReadShadow::new(&merge_schema, now_epoch_secs());
462        let mut merger = KWayMerger::new(paths, &merge_schema)?;
463        let mut out = Vec::new();
464        while let MergeStep::Partition { key, rows } = merger.step()? {
465            let row_key = RowKey::new(key.key.clone());
466
467            // Partition-targeted point read (#1579): keep ONLY the target and stop.
468            if let Some(ref target) = target_for_merge {
469                if row_key.as_bytes() != target.as_slice() {
470                    continue;
471                }
472                push_metadata_rows(&key.key, rows, &mut out, &shadow);
473                break;
474            }
475
476            // Full / range scan: inclusive [start, end] on `RowKey` order (#957).
477            if let Some(ref start) = start_key {
478                if &row_key < start {
479                    continue;
480                }
481            }
482            if let Some(ref end) = end_key {
483                if &row_key > end {
484                    continue;
485                }
486            }
487            push_metadata_rows(&key.key, rows, &mut out, &shadow);
488        }
489        Ok(out)
490    })
491    .await
492    .map_err(|e| crate::Error::Storage(format!("cross-generation metadata merge task: {e}")))??;
493
494    // Attach per-cell metadata: WRITETIME from the merge winner, TTL recovered from
495    // the reader lookup only when its timestamp matches the winner.
496    let mut results: Vec<(RowKey, ScanRow, HashMap<String, CellWriteMetadata>)> =
497        Vec::with_capacity(merged_rows.len());
498    for (key_bytes, value, timestamps) in merged_rows {
499        let mut meta_map: HashMap<String, CellWriteMetadata> =
500            HashMap::with_capacity(timestamps.len());
501        for (column, write_ts) in timestamps {
502            let expiration = ttl_lookup
503                .get(&(key_bytes.clone(), column.clone()))
504                .filter(|m| m.write_timestamp_micros == write_ts)
505                .and_then(|m| m.expiration.clone());
506            meta_map.insert(
507                column,
508                CellWriteMetadata {
509                    write_timestamp_micros: write_ts,
510                    expiration,
511                },
512            );
513        }
514        results.push((RowKey::new(key_bytes), value, meta_map));
515    }
516
517    // Stable TOKEN-order sort (issue #1580), then LIMIT — identical ordering to the
518    // plain merge path; the metadata payload rides in the tuple's tail.
519    scan_merge::sort_by_token_order(&mut results, limit, |(k, _, _)| k);
520    Ok(results)
521}

523/// Convert one reconciled partition's merge rows into metadata-carrying rows: the
524/// live cells plus each surviving cell's `(column, write_timestamp)`. Shared by the
525/// range and point-targeted branches of the metadata merge so the tombstone
526/// filtering + timestamp capture live once.
527fn push_metadata_rows(
528    key_bytes: &[u8],
529    rows: Vec<MergeEntry>,
530    out: &mut Vec<MergedMetaRow>,
531    shadow: &ReadShadow,
532) {
533    // Issue #1849: same read-side partition cover + per-cell TTL/shadow filter as the
534    // plain path, so the WRITETIME/TTL projection agrees with `SELECT *`.
535    let cover = partition_cover(&rows);
536    for entry in rows {
537        if let RowData::Live { cells } = entry.row_data {
538            // `filter_live` drops cell tombstones + TTL-expired / partition-shadowed
539            // data cells; `None` hides the whole row (partition-shadowed).
540            let Some(surviving) = shadow.filter_live(cover, cells) else {
541                continue;
542            };
543            let mut row_cells: RowCells = Vec::with_capacity(surviving.len());
544            let mut timestamps: Vec<(String, i64)> = Vec::with_capacity(surviving.len());
545            for c in surviving {
546                timestamps.push((c.column.clone(), c.timestamp));
547                row_cells.push((Arc::from(c.column.as_str()), c.value));
548            }
549            if !row_cells.is_empty() {
550                out.push((key_bytes.to_vec(), ScanRow::Row(row_cells), timestamps));
551            }
552        }
553        // Row tombstones suppress the row entirely (no emission).
554    }
555}

557/// STREAMING cross-generation reconciliation for `scan_stream` (issue #1579 / D3).
558///
559/// Runs the authoritative [`KWayMerger`](crate::storage::write_engine::KWayMerger)
560/// on a blocking task and feeds each stepped partition's live rows STRAIGHT into a
561/// bounded channel via `blocking_send` (backpressure preserved), instead of
562/// collecting the entire reconciled table, sorting it, and dribbling it. Live heap
563/// is therefore O(one partition + channel), and the first row is available after
564/// the first partition rather than after the whole merge.
565///
566/// Reconciliation is byte-identical to [`merge_generations_for_read`] (shared
567/// `partition_live_rows` kernel), and the emission order is the merger's
568/// `DecoratedKey` = `(token, key)` order — byte-identical to the collect+sort
569/// path's [`scan_merge::sort_by_token_order`] output (issue #1579 ordering
570/// guardrail).
571///
572/// Construction (`KWayMerger::new`, which opens the input files) happens on the
573/// blocking task; its success/failure is signalled back over a oneshot BEFORE any
574/// streaming, so the caller can FALL BACK to the lazy per-reader streaming merge on
575/// a construction error exactly as the materializing `scan` falls back to
576/// concatenation. A runtime `step()` error mid-stream is delivered as an `Err`
577/// item on the channel (the consumer sees it), matching the lazy path's read-error
578/// behaviour.
579///
580/// This is a deliberate, documented error-path asymmetry (issue #1579): the
581/// caller's fallback-to-concatenation only ever applies to the CONSTRUCTION
582/// failure above (nothing has been streamed yet, so falling back cannot mix
583/// reconciled and unreconciled rows). A `step()` failure after some partitions
584/// were already emitted downstream is NEVER retried/fallen-back — it ends the
585/// stream via the `Err` channel item instead. That is safer than it sounds: the
586/// materializing `merge_generations_for_read` has no equivalent mid-collection
587/// failure signal — a `step()` error there simply propagates the whole call as
588/// `Err` before any partial `Vec` is returned to the caller. The streaming
589/// driver's `Err` item preserves the same "no half-reconciled result surfaces
590/// silently" guarantee while still letting the caller observe exactly how many
591/// good rows it already received.
592#[cfg(not(feature = "tombstones"))]
593pub(super) async fn stream_generations_for_read(
594    reader_list: &[Arc<reader::SSTableReader>],
595    schema: &crate::schema::TableSchema,
596    start_key: Option<&RowKey>,
597    end_key: Option<&RowKey>,
598    buffer_size: usize,
599) -> Result<mpsc::Receiver<Result<(RowKey, ScanRow)>>> {
600    let start_key = start_key.cloned();
601    let end_key = end_key.cloned();
602    let paths = ordered_generation_paths(reader_list);
603    let schema = schema.clone();
604
605    let (ready_tx, ready_rx) = oneshot::channel::<Result<()>>();
606    let (out_tx, out_rx) = mpsc::channel::<Result<(RowKey, ScanRow)>>(buffer_size.max(1));
607
608    tokio::task::spawn_blocking(move || {
609        // Issue #1849: capture the read-time TTL clock ONCE per scan.
610        let shadow = ReadShadow::new(&schema, now_epoch_secs());
611        let mut merger = match KWayMerger::new(paths, &schema) {
612            Ok(m) => {
613                // Signal readiness; if the caller already dropped, stop.
614                if ready_tx.send(Ok(())).is_err() {
615                    return;
616                }
617                m
618            }
619            Err(e) => {
620                // Let the caller fall back to the lazy per-reader streaming merge.
621                let _ = ready_tx.send(Err(e));
622                return;
623            }
624        };
625
626        loop {
627            let step = match merger.step() {
628                Ok(s) => s,
629                Err(e) => {
630                    let _ = out_tx.blocking_send(Err(e));
631                    return;
632                }
633            };
634            let (key, rows) = match step {
635                MergeStep::Partition { key, rows } => (key, rows),
636                MergeStep::Complete => return,
637            };
638
639            let row_key = RowKey::new(key.key.clone());
640            if let Some(ref start) = start_key {
641                if &row_key < start {
642                    continue;
643                }
644            }
645            if let Some(ref end) = end_key {
646                if &row_key > end {
647                    continue;
648                }
649            }
650
651            let live = partition_live_rows(&row_key, rows, &shadow);
652            // Issue #1579: the streaming producer holds ONE partition's rows
653            // resident at a time — record the window (not the whole table) so the
654            // memory guard observes O(window).
655            stream_merge_probe::record_resident(live.len() as u64);
656            for entry in live {
657                if out_tx.blocking_send(Ok(entry)).is_err() {
658                    return; // consumer dropped
659                }
660            }
661        }
662    });
663
664    match ready_rx.await {
665        Ok(Ok(())) => Ok(out_rx),
666        Ok(Err(e)) => Err(e),
667        Err(_) => Err(crate::Error::Storage(
668            "cross-generation streaming merge task ended before signalling readiness".to_string(),
669        )),
670    }
671}

673/// The merger expects inputs ordered newest → oldest (run_index 0 = newest) for
674/// its stable tie-break; the reader `Vec` order is discovery-dependent, so sort
675/// explicitly by generation descending and collect the input `Data.db` paths.
676fn ordered_generation_paths(reader_list: &[Arc<reader::SSTableReader>]) -> Vec<PathBuf> {
677    let mut ordered: Vec<&Arc<reader::SSTableReader>> = reader_list.iter().collect();
678    ordered.sort_by(|a, b| b.generation.cmp(&a.generation));
679    ordered.iter().map(|r| r.file_path()).collect()
680}
