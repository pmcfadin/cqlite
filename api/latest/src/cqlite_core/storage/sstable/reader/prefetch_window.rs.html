1//! Direct-I/O read-ahead window sizing (issue #1596, Epic F / F6).
2//!
3//! The direct-I/O scan cursor ([`DirectCursor`](super::source::DirectCursor))
4//! refills an *aligned* read-ahead window with one `pread` per refill, then hands
5//! sub-ranges of that window to the chunk reader. Each on-disk compression chunk
6//! is `chunk_size` payload bytes + a trailing 4-byte CRC32, so the record the
7//! chunk reader consumes in one shot is `chunk_size + 4` bytes.
8//!
9//! If the configured window (`direct_io_prefetch_bytes`) is SMALLER than one
10//! chunk record, a single chunk read straddles multiple window refills — read
11//! amplification: `ceil(record / window)` `pread`s for what should be one. Worse,
12//! a window that is not a small multiple of the record makes the straddle count
13//! fluctuate as chunks drift across window boundaries.
14//!
15//! [`clamp_direct_prefetch_window`] removes that pathology at *source
16//! construction* (the audit F6 fix): it floors the window at `2 × (chunk_size +
17//! 4)` and keeps 4K alignment. With a window ≥ 2 records, any one chunk record
18//! spans at most two aligned windows, so per-chunk refills are bounded to
19//! `ceil(record / window) + 1 = 2` regardless of where the record falls. The
20//! default window (1 MiB) already exceeds this for the default 64 KiB chunk, so
21//! the clamp only bites when a caller sets `direct_io_prefetch_bytes` below a
22//! chunk — turning a silent amplification into the intended single-window read.
23
24/// I/O alignment (bytes) for direct reads — mirrors `source::DIRECT_IO_ALIGN` and
25/// `read_at::DIRECT_IO_ALIGN`. Kept here (non-`cfg`) so window sizing is available
26/// on every platform (the direct backend degrades to buffered off-Unix, but the
27/// window value is still computed at the shared call site).
28pub(crate) const DIRECT_IO_ALIGN: usize = 4096;
29
30/// Cassandra's default compression chunk length (`chunk_length_in_kb: 64`). Used
31/// as the chunk-size basis for the window floor at reader open, where the actual
32/// per-SSTable `CompressionInfo.chunk_length` is not yet parsed. This fully
33/// protects the common (default-chunk) case; a caller that both picks a larger
34/// custom chunk AND a deliberately tiny window retains that explicit choice.
35pub(crate) const DEFAULT_COMPRESSION_CHUNK_BYTES: usize = 64 * 1024;
36
37/// Clamp/round a requested direct-I/O prefetch window so a single compression
38/// chunk record never straddles more than two aligned refills.
39///
40/// Returns `max(prefetch_bytes, 2 × (chunk_size + 4))`, never below one alignment
41/// unit, rounded UP to [`DIRECT_IO_ALIGN`]. All arithmetic saturates (a
42/// near-`usize::MAX` `chunk_size`/`prefetch_bytes` yields the largest aligned
43/// value rather than overflowing), so the caller's downstream aligned allocation
44/// simply fails and falls back to buffered I/O instead of panicking.
45pub(crate) fn clamp_direct_prefetch_window(prefetch_bytes: usize, chunk_size: usize) -> usize {
46    // On-disk record = payload + trailing 4-byte CRC32.
47    let record = chunk_size.saturating_add(4);
48    // Two records so any one record spans at most two aligned windows.
49    let floor = record.saturating_mul(2);
50    let want = prefetch_bytes.max(floor).max(DIRECT_IO_ALIGN);
51    want.checked_next_multiple_of(DIRECT_IO_ALIGN)
52        .unwrap_or(usize::MAX & !(DIRECT_IO_ALIGN - 1))
53}

55#[cfg(test)]
56mod tests {
57    use super::*;
58
59    /// Model the direct cursor's per-chunk refill count: reading a record of
60    /// `record` bytes through an aligned `window`, worst-case starting offset,
61    /// costs `ceil(record / window) + 1` refills (the `+1` for a record that
62    /// straddles a window boundary). This mirrors `DirectCursor::ensure_buffered`
63    /// without needing `O_DIRECT`, so the amplification invariant is testable on
64    /// every platform.
65    fn worst_case_refills(window: usize, record: usize) -> usize {
66        record.div_ceil(window) + 1
67    }
68
69    /// RED-first invariant (issue #1596 TDD): a prefetch window SMALLER than one
70    /// chunk record amplifies reads (this is the "fails today" state the audit
71    /// calls out), and the F6 clamp restores the `≤ ceil(chunk/window)+1 = 2`
72    /// bound. Both halves are asserted against the same `worst_case_refills`
73    /// model so the clamp's effect is falsifiable, not asserted by fiat.
74    #[test]
75    fn clamp_bounds_per_chunk_refills_to_two() {
76        let chunk = DEFAULT_COMPRESSION_CHUNK_BYTES;
77        let record = chunk + 4;
78
79        // A deliberately tiny window (below one chunk) — the misconfiguration the
80        // audit targets. Worst case it straddles many refills (> 2): read amp.
81        let tiny = 8 * 1024;
82        assert!(
83            worst_case_refills(tiny, record) > 2,
84            "an unclamped sub-chunk window must amplify (>2 refills/chunk); \
85             got {} for window={} record={}",
86            worst_case_refills(tiny, record),
87            tiny,
88            record
89        );
90
91        // After the clamp the window holds ≥ 2 records, so any one record spans
92        // at most two aligned windows: refills bounded to exactly 2.
93        let clamped = clamp_direct_prefetch_window(tiny, chunk);
94        assert!(
95            clamped >= 2 * record,
96            "clamped window must hold >= 2 records: {clamped} < {}",
97            2 * record
98        );
99        assert_eq!(
100            worst_case_refills(clamped, record),
101            2,
102            "clamped window must bound per-chunk refills to ceil(chunk/window)+1 = 2"
103        );
104    }
105
106    #[test]
107    fn clamp_result_is_4k_aligned_and_at_least_floor() {
108        for &(pf, chunk) in &[
109            (0usize, 64 * 1024usize),
110            (1, 64 * 1024),
111            (8 * 1024, 64 * 1024),
112            (4095, 4096),
113            (1024 * 1024, 64 * 1024), // default window already above floor
114            (128 * 1024, 256 * 1024), // larger custom chunk
115        ] {
116            let w = clamp_direct_prefetch_window(pf, chunk);
117            assert_eq!(w % DIRECT_IO_ALIGN, 0, "window {w} not 4K-aligned");
118            assert!(
119                w >= 2 * (chunk + 4),
120                "window {w} below floor 2*(chunk+4)={} (chunk={chunk})",
121                2 * (chunk + 4)
122            );
123            assert!(w >= pf, "window {w} dropped below requested {pf}");
124            assert!(w >= DIRECT_IO_ALIGN, "window {w} below one alignment unit");
125        }
126    }
127
128    /// A window already larger than the floor is preserved (only rounded up to
129    /// alignment) — the clamp never SHRINKS a well-sized window.
130    #[test]
131    fn clamp_preserves_large_windows() {
132        let big = 4 * 1024 * 1024;
133        assert_eq!(
134            clamp_direct_prefetch_window(big, DEFAULT_COMPRESSION_CHUNK_BYTES),
135            big,
136            "an already-aligned window above the floor is returned unchanged"
137        );
138    }
139
140    /// Saturating arithmetic: a pathological `chunk_size`/`prefetch_bytes` near
141    /// `usize::MAX` yields the largest aligned value, never a panic/overflow.
142    #[test]
143    fn clamp_saturates_on_overflow() {
144        let w = clamp_direct_prefetch_window(usize::MAX, usize::MAX);
145        assert_eq!(w % DIRECT_IO_ALIGN, 0, "saturated window must stay aligned");
146        assert_eq!(w, usize::MAX & !(DIRECT_IO_ALIGN - 1));
147    }
