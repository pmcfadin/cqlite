1//! IO-half chunk decode for the windowed streaming scan (issue #1940, D2).
2//!
3//! Split out of `scan_stream_windowed.rs` to keep that source file under the
4//! campsite-rule size limit (epic #1116). `decode_scan_chunk` is the one place the
5//! windowed-scan IO half turns a freshly-read, CRC-verified compressed chunk into
6//! the refcounted decompressed `Bytes` substrate shipped on the chunk channel,
7//! while handing the compressed buffer back so the feed loop recycles it as the
8//! next read's scratch (≤1 alloc/chunk). Included via
9//! `#[path = "scan_stream_windowed_decode.rs"] impl`-carrying module in the parent.
10
11use super::SSTableReader;
12use crate::Result;
13
14impl SSTableReader {
15    /// This reader's configured decompression algorithm, or `None` for an
16    /// uncompressed SSTable — the plane maps `None` onto the bounded `"none"`
17    /// compression label (issue #1701), so no call site invents a label of its own.
18    fn scan_algorithm(
19        &self,
20    ) -> Option<&crate::storage::sstable::compression::CompressionAlgorithm> {
21        self.compression_reader.as_ref().map(|r| r.algorithm())
22    }
23
24    /// Decode ONE compressed chunk on the IO half (issue #1940, D2): CRC was
25    /// already verified inside the read path, so this does cache-lookup-or-decompress
26    /// and returns the refcounted decompressed `Bytes` substrate shipped on the
27    /// channel — while HANDING BACK the compressed buffer (`compressed`) so the feed
28    /// loop can RECYCLE it as the next read's scratch (no per-chunk compressed-buffer
29    /// allocation). The single decode plane (`chunk_source`) still owns the one
30    /// decompress call. Two cases consume the buffer instead of handing it back
31    /// (returning an empty scratch, so a fresh buffer is minted next read): an
32    /// incompressible-raw chunk (stored uncompressed by Cassandra, compressed len
33    /// ≥ `max_compressed_length`) is passed through zero-copy `Vec`→`Bytes`; and the
34    /// no-compressor case (raw/uncompressed NB scan with no `CompressionInfo` — the
35    /// read buffer already holds finished bytes) is MOVED into the B1 cache zero-copy
36    /// via `Bytes::from(Vec)`, never `to_vec()`-copied (issue #1940 BLOCKER-1;
37    /// uncompressed is CQLite's own write-surface format, a first-class path).
38    /// `chunk_index` is the ABSOLUTE chunk index (the feed reads from data-section
39    /// start in order), matching the prior parse-half keying.
40    pub(super) fn decode_scan_chunk(
41        &self,
42        chunk_index: usize,
43        max_compressed_length: usize,
44        compressed: Vec<u8>,
45    ) -> Result<(bytes::Bytes, Vec<u8>)> {
46        use crate::storage::sstable::compression::Compression;
47
48        // Incompressible-raw chunks (stored uncompressed by Cassandra) skip the
49        // cache and are passed through zero-copy; the buffer is consumed.
50        if compressed.len() >= max_compressed_length {
51            // cqlite.read.bytes (issue #1701 roborev B2): these ARE `Data.db` payload
52            // bytes this scan just read — Cassandra stored the chunk raw — so they are
53            // counted here, at the exit that BYPASSES the decode plane where the
54            // sibling compressed exit counts. Skipping them would understate real I/O.
55            super::super::chunk_source::count_raw_chunk(&compressed, self.scan_algorithm());
56            return Ok((bytes::Bytes::from(compressed), Vec::new()));
57        }
58        let key = crate::storage::cache::ChunkKey::new(
59            self.chunk_cache_id ^ super::super::data_access::NS_WINDOWED_CHUNK,
60            chunk_index as u64,
61        );
62        // Cache hit → refcount-bump clone (B1 contract), never a memcpy; the
63        // compressed buffer is recycled unused. Warm scans must take the hit rather
64        // than re-decompress and overwrite (issue #1598 roborev Medium).
65        if let Some(hit) = self.chunk_cache.get(&key) {
66            // cqlite.read.bytes (issue #1701, roborev round 5): a hit here still READ
67            // the chunk off disk. The windowed feed calls
68            // `read_compressed_chunk_sync` / `read_uncompressed_piece_sync`
69            // UNCONDITIONALLY and only then asks the cache (see the caller in
70            // `scan_stream_windowed_read.rs`; neither reader consults the cache), so
71            // this cache saves DECOMPRESSION, not I/O — the comment above says as much
72            // when it calls `compressed` "recycled unused". Returning uncounted made a
73            // warm scan report ZERO bytes for I/O it genuinely performed, which is the
74            // opposite of what an I/O-amplification metric is for.
75            //
76            // Counted as the DECOMPRESSED size, like every other site: `read.bytes` is
77            // documented post-decompression, so counting `compressed.len()` here would
78            // make the same chunk contribute two different amounts cold vs warm and
79            // break comparability of the series.
80            super::super::chunk_source::count_raw_chunk(&hit, self.scan_algorithm());
81            return Ok((hit, compressed));
82        }
83        // No compressor (raw/uncompressed NB scan, no CompressionInfo): the read
84        // buffer already holds the finished, uncompressed chunk bytes. MOVE it into
85        // the B1 cache as zero-copy `Bytes` (`Bytes::from(Vec)` reuses the read
86        // buffer's heap allocation) rather than `to_vec()`-copying it — this is
87        // CQLite's own uncompressed write-surface format, a first-class path (issue
88        // #1940 BLOCKER-1). Nothing to recycle on the raw path (the buffer became
89        // the cached substrate), so the returned scratch is empty; a fresh read
90        // buffer is minted for the next chunk. No decompress here, so the
91        // decode-thread probe does NOT fire (it pins where decompression runs).
92        if self.compression_reader.is_none() {
93            // cqlite.read.bytes (issue #1701 roborev B2): the UNCOMPRESSED scan's
94            // chunk bytes are `Data.db` payload read from disk. This exit also
95            // bypasses the plane, and uncompressed is a FIRST-CLASS path (CQLite's own
96            // write surface emits only uncompressed SSTables, the #1406 claim
97            // boundary), so leaving it uncounted made every uncompressed read
98            // invisible to the metric.
99            super::super::chunk_source::count_raw_chunk(&compressed, None);
100            return Ok((self.chunk_cache.insert(key, compressed), Vec::new()));
101        }
102        // Miss → decompress from the BORROWED slice (so we keep `compressed` to
103        // recycle) into the single decode plane, cache the resident `Bytes`.
104        let compression = self
105            .compression_reader
106            .as_ref()
107            .map(|cr| Compression::new(*cr.algorithm()))
108            .transpose()?;
109        // Runtime-placement guard (issue #1940): record the thread this decode
110        // actually decompresses on — placed at the REAL decompress site (past the
111        // cache-hit / incompressible-raw / no-compressor early exits, none of which
112        // decompress) so a guard test can prove decompression runs on a
113        // spawn_blocking thread, NOT an async worker (the D2 substrate moved
114        // decompression into the IO-half feed loop, which must stay off the reactor
115        // for EVERY backend). Compiled only under `scan-offload-probe`.
116        #[cfg(feature = "scan-offload-probe")]
117        super::probe::record_decode_thread();
118        let comp_info_dummy = crate::storage::sstable::compression_info::CompressionInfo {
119            algorithm: String::new(),
120            option_pairs: vec![],
121            chunk_length: 0,
122            max_compressed_length: max_compressed_length as u32,
123            data_length: 0,
124            chunk_offsets: vec![],
125        };
126        let chunk_source = super::super::chunk_source::ChunkSource::new(
127            // Correct-by-intent: this is a SCAN path, so hand it the unadvised
128            // scan plane (issue #2876). `decode_borrowed` performs no I/O today, so
129            // this is inert — but `ChunkSource::chunk()` CAN read, and a future
130            // fallback here (a decode-error re-read, or a chunk absent from the
131            // caller's buffer) would otherwise silently route full-scan I/O back
132            // through the MADV_RANDOM point mapping and re-create #2876 with no
133            // test failure. Free to get right now; a trap to leave wrong.
134            self.scan_positional_source.as_ref(),
135            &comp_info_dummy,
136            compression.as_ref(),
137            &self.chunk_cache,
138            0, // unused
139            0, // unused
140            super::super::data_access::NS_WINDOWED_CHUNK,
141            self.chunk_cache_id,
142        );
143        let decoded = chunk_source.decode_borrowed(key, &compressed)?;
144        Ok((decoded, compressed))
145    }
146}
