1//! Test-only probe (issues #1143, #1333) for the windowed streaming scan.
2//!
3//! Records (a) the [`std::thread::ThreadId`] on which the windowed scan's
4//! decompress+parse half actually ran, so a guard test can deterministically
5//! prove that work executed on a `spawn_blocking` thread and NOT on a tokio async
6//! worker (#1143); and (b) how many times the per-partition scratch buffer GREW
7//! its backing store, so a guard test can prove the scratch is reused across
8//! partitions rather than reallocated per partition (#1333).
9//!
10//! Included via `#[cfg(feature = "scan-offload-probe")] #[path =
11//! "scan_stream_windowed_probe.rs"] pub mod probe;` in the parent module and
12//! compiled ONLY under the non-default `scan-offload-probe` feature. In a
13//! normal/default/release build this module, its statics, and its call-sites do
14//! not exist at all — the probe never enters the crate's public surface and adds
15//! zero cost (issue #1143 finding 1). Kept in a sibling file so the parent stays
16//! under the campsite-rule size limit (epic #1116).
17
18use std::sync::atomic::{AtomicBool, AtomicUsize, Ordering};
19use std::sync::Mutex;
20use std::thread::ThreadId;
21
22static ARMED: AtomicBool = AtomicBool::new(false);
23static LAST_PARSE_THREAD: Mutex<Option<ThreadId>> = Mutex::new(None);
24/// Number of times the per-partition scratch buffer GREW its backing store
25/// during the armed scan (issue #1333). With the scratch hoisted out of the
26/// per-partition loop and `.clear()`-reused this is a small bounded count
27/// (the buffer grows only up to its high-water mark); if the buffer were
28/// reallocated per partition it would grow from empty every partition, so this
29/// count would scale with partition count.
30static SCRATCH_ALLOCS: AtomicUsize = AtomicUsize::new(0);
31
32/// Arm the probe and clear any previously recorded thread + scratch-growth
33/// count. Call from a test before driving a scan.
34pub fn arm() {
35    if let Ok(mut g) = LAST_PARSE_THREAD.lock() {
36        *g = None;
37    }
38    SCRATCH_ALLOCS.store(0, Ordering::SeqCst);
39    ARMED.store(true, Ordering::SeqCst);
40}
41
42/// Disarm the probe (restores the production no-op state).
43pub fn disarm() {
44    ARMED.store(false, Ordering::SeqCst);
45}
46
47/// Record the current thread as the parse thread, if armed. Called from the
48/// blocking parse half after a scan's parse work completes.
49pub(super) fn record_parse_thread() {
50    if ARMED.load(Ordering::Relaxed) {
51        if let Ok(mut g) = LAST_PARSE_THREAD.lock() {
52            *g = Some(std::thread::current().id());
53        }
54    }
55}
56
57/// The [`ThreadId`] recorded by the most recent parse, if any.
58pub fn recorded_parse_thread() -> Option<ThreadId> {
59    LAST_PARSE_THREAD.lock().ok().and_then(|g| *g)
60}
61
62/// Record that the per-partition scratch buffer's capacity changed from
63/// `before` to `after`. Counts one growth event whenever `after > before`
64/// (a Vec grows its backing store only by reallocating). Called once per
65/// partition from [`super::SSTableReader::drain_scan_window`]; a no-op unless
66/// armed. Issue #1333 scratch-reuse guard.
67pub(super) fn note_scratch_capacity(before: usize, after: usize) {
68    if ARMED.load(Ordering::Relaxed) && after > before {
69        SCRATCH_ALLOCS.fetch_add(1, Ordering::Relaxed);
70    }
71}
72
73/// The number of scratch-buffer growth events recorded during the armed scan.
74/// Bounded and independent of partition count when the scratch is reused
75/// (issue #1333); scales with partition count if it is reallocated per
76/// partition.
77pub fn recorded_scratch_allocs() -> usize {
78    SCRATCH_ALLOCS.load(Ordering::Relaxed)
79}
