1//! Positional (`pread`-style) read backends for the SSTable point-read path
2//! (issue #1573, Epic C / C2).
3//!
4//! # Why this exists
5//!
6//! The historical point-read path served every read through a single
7//! [`BlockSource`](super::source::BlockSource) behind an
8//! `Arc<Mutex<BlockSource>>`. That mutex exists ONLY to guard a mutable seek
9//! cursor, and it was held **across disk I/O**, so N concurrent point reads on
10//! one SSTable serialized one-at-a-time (the convoy). The BTI variant sidestepped
11//! the mutex by paying `File::open` **per lookup** instead — trading the convoy
12//! for per-op `open(2)` syscalls and an fd-exhaustion risk under load.
13//!
14//! Both pathologies share one root cause: a *stateful* file cursor. [`ReadAt`]
15//! removes the state. A positioned read carries its offset as a call parameter,
16//! not as shared mutable file state, so:
17//!
18//! - **No mutable position** — the trait has no seek cursor.
19//! - **No `&mut self`** — every method takes `&self`, so concurrent callers share
20//!   one value with no exclusive borrow and no mutex.
21//! - **No seek** — the offset is a parameter of the read.
22//!
23//! A positioned read on a shared file descriptor is independent of every other
24//! positioned read (POSIX `pread`/`FileExt::read_at` does not touch the fd's
25//! offset), so the lock is not merely moved off the point path — it is removed.
26//!
27//! # Backends
28//!
29//! | Backend | `read_at` | Notes |
30//! |---------|-----------|-------|
31//! | [`PlainFileReadAt`] | `FileExt::read_at` (Unix, lock-free) / `Mutex`-guarded `seek_read` (Windows) on ONE shared `File` | Unix `pread` is positional so reads are independent (no lock); Windows `seek_read` moves the shared cursor, so it is serialized under a `Mutex`. Positioned reads on a shared fd are independent — no lock. |
32//! | [`MmapReadAt`] | slice indexing into the resident map | Bounds-checked; zero syscalls per read. |
33//! | [`DirectReadAt`] (Unix) | aligned positioned read honoring the `O_DIRECT` 4K rule | A per-call aligned bounce buffer keeps `&self` (no shared cursor). |
34//!
35//! The trait is intentionally shaped so the windowed streaming scan can adopt it
36//! later (audit F3) without requiring it now; this change migrates only the
37//! point-read path (design.md Decision 4).
38

45/// A positional byte source: reads are addressed by an explicit `offset`, never
46/// by a shared mutable cursor.
47///
48/// Every method takes `&self`. Two concurrent `read_at` calls at different
49/// offsets are independent — neither observes nor disturbs the other's position,
50/// because there is no position. This is the whole point of the refactor
51/// (issue #1573): concurrent point reads on one SSTable no longer serialize on a
52/// cross-I/O mutex, and BTI lookups no longer `open(2)` per call.
53pub(crate) trait ReadAt: Send + Sync {
54    /// Read up to `buf.len()` bytes starting at absolute file `offset`, returning
55    /// the number of bytes read. A short read (fewer than `buf.len()`) is legal
56    /// at/near EOF, exactly like POSIX `pread`.
57    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize>;
58
59    /// Total length of the backing file in bytes.
60    fn len(&self) -> u64;
61
62    /// Read EXACTLY `buf.len()` bytes starting at `offset`, erroring with
63    /// [`std::io::ErrorKind::UnexpectedEof`] if the source is exhausted first.
64    ///
65    /// Default impl loops [`read_at`](Self::read_at) so every backend gets an
66    /// exact-read helper for free; backends may override if they can do better.
67    fn read_exact_at(&self, offset: u64, buf: &mut [u8]) -> Result<()> {
68        let mut filled = 0usize;
69        while filled < buf.len() {
70            let n = self.read_at(offset + filled as u64, &mut buf[filled..])?;
71            if n == 0 {
72                return Err(Error::Io(std::io::Error::new(
73                    std::io::ErrorKind::UnexpectedEof,
74                    format!(
75                        "read_exact_at: reached EOF after {filled} of {} bytes at offset 0x{offset:x}",
76                        buf.len()
77                    ),
78                )));
79            }
80            filled += n;
81        }
82        Ok(())
83    }
84
85    /// Like [`read_exact_at`](Self::read_exact_at) but may REUSE the caller-owned
86    /// [`DirectScratch`] aligned bounce buffer across calls, so the Direct-I/O
87    /// windowed scan does not allocate a fresh ~chunk-sized aligned buffer per
88    /// chunk read (issue #2319 — the regression #1940 introduced by routing the
89    /// Direct windowed scan through the generic per-call [`DirectReadAt::read_at`]
90    /// instead of the pre-#1940 per-scan `DirectCursor` that reused ONE window
91    /// buffer).
92    ///
93    /// The default impl IGNORES `scratch` and defers to
94    /// [`read_exact_at`](Self::read_exact_at): the mmap and plain-file backends
95    /// never allocate a bounce buffer at all, so buffer reuse is a no-op for them
96    /// and the window-as-`Bytes` substrate (#1940) is untouched. ONLY
97    /// [`DirectReadAt`] overrides it, threading the reusable aligned buffer through
98    /// its per-read alignment math.
99    fn read_exact_at_reusing(
100        &self,
101        offset: u64,
102        buf: &mut [u8],
103        _scratch: &mut DirectScratch,
104    ) -> Result<()> {
105        self.read_exact_at(offset, buf)
106    }
107}

109/// A caller-owned reusable aligned bounce buffer for the Direct-I/O windowed-scan
110/// read path (issue #2319).
111///
112/// The windowed feed creates ONE `DirectScratch` per scan and threads a `&mut` to
113/// it through every chunk read via [`ReadAt::read_exact_at_reusing`]. When the
114/// backend is [`DirectReadAt`], the read reuses this single aligned buffer instead
115/// of allocating one per chunk — restoring the pre-#1940 per-scan reused-window
116/// behaviour. For every other backend it carries no state and costs nothing (the
117/// aligned buffer is `Option`-wrapped and stays `None`). Since the feed owns it
118/// exclusively for the scan's lifetime, no cursor/mutex is needed and concurrent
119/// scans (each with their own `DirectScratch`) never share the buffer.
120pub(crate) struct DirectScratch {
121    /// The reused aligned buffer, grown on demand to the working-set high-water
122    /// mark. `None` until the first Direct read; always `None` on non-Direct
123    /// backends and non-Unix targets (no direct I/O there).
124    #[cfg(unix)]
125    bounce: Option<AlignedBuf>,
126}
127
128impl DirectScratch {
129    /// A fresh scratch with no buffer allocated yet.
130    pub(crate) fn new() -> Self {
131        Self {
132            #[cfg(unix)]
133            bounce: None,
134        }
135    }
136}

138/// Delegating impl so an `Arc<dyn ReadAt>` (or `Arc<ConcreteBackend>`) is itself a
139/// `ReadAt`. Lets the point path hold a single `Arc<dyn ReadAt>` and lets a test
140/// wrap the reader's existing shared source without unsizing gymnastics.
141impl<T: ReadAt + ?Sized> ReadAt for Arc<T> {
142    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize> {
143        (**self).read_at(offset, buf)
144    }
145    fn len(&self) -> u64 {
146        (**self).len()
147    }
148    fn read_exact_at(&self, offset: u64, buf: &mut [u8]) -> Result<()> {
149        (**self).read_exact_at(offset, buf)
150    }
151    // Forward the reusing read so a `DirectReadAt` behind an `Arc<dyn ReadAt>`
152    // (exactly `point_source`'s type) still reaches its override — otherwise the
153    // Direct windowed scan would silently fall back to the default per-chunk-alloc
154    // path (issue #2319).
155    fn read_exact_at_reusing(
156        &self,
157        offset: u64,
158        buf: &mut [u8],
159        scratch: &mut DirectScratch,
160    ) -> Result<()> {
161        (**self).read_exact_at_reusing(offset, buf, scratch)
162    }
163}

240/// Positional reads over a plain file handle.
241///
242/// Wraps ONE shared `std::fs::File`, opened once at reader open and shared for the
243/// reader's lifetime (design.md Decision 3).
244///
245/// On **Unix**, `FileExt::read_at` issues a `pread` that does not touch the fd's
246/// offset, so concurrent reads on this single handle are independent — no lock, no
247/// per-lookup `open(2)`.
248///
249/// On **Windows** there is no positional `pread`: `seek_read` MOVES the handle's
250/// file pointer, so two concurrent reads on one shared handle would race the cursor
251/// and return the wrong bytes (silent corruption). The Windows arm therefore guards
252/// the handle with a `std::sync::Mutex` and performs the seek+read under the lock,
253/// serializing Windows point reads for correctness while still opening the fd only
254/// once. The Unix arm keeps its lock-free `Arc<File>` (issue #1573 roborev finding).
255pub(crate) struct PlainFileReadAt {
256    /// Unix: shared handle, read lock-free via positional `pread` (no cursor).
257    #[cfg(unix)]
258    file: Arc<std::fs::File>,
259    /// Windows: `seek_read` mutates the shared cursor, so the handle is `Mutex`-
260    /// guarded and read+seek happen atomically under the lock (see struct docs).
261    #[cfg(windows)]
262    file: std::sync::Mutex<std::fs::File>,
263    len: u64,
264}
265
266impl PlainFileReadAt {
267    /// Wrap an already-open file handle whose length is `len`.
268    pub(crate) fn new(file: std::fs::File, len: u64) -> Self {
269        Self {
270            #[cfg(unix)]
271            file: Arc::new(file),
272            #[cfg(windows)]
273            file: std::sync::Mutex::new(file),
274            len,
275        }
276    }
277
278    /// Open `path` read-only for positioned reads, recording one `open(2)` in the
279    /// read-work counters (consumer C2). This is the single per-reader open the
280    /// BTI point path used to pay per lookup.
281    pub(crate) fn open(path: &std::path::Path, len: u64) -> Result<Self> {
282        crate::storage::sstable::read_work_counters::record_file_open();
283        let file = std::fs::File::open(path)?;
284        Ok(Self::new(file, len))
285    }
286}

288impl ReadAt for PlainFileReadAt {
289    #[cfg(unix)]
290    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize> {
291        use std::os::unix::fs::FileExt;
292        Ok(self.file.read_at(buf, offset)?)
293    }
294
295    #[cfg(windows)]
296    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize> {
297        // Windows has no positional `pread`; `seek_read` MOVES the handle's file
298        // pointer, so two concurrent `read_at` calls on the one shared handle would
299        // race the cursor and return the wrong bytes (silent corruption). We hold a
300        // `Mutex` across the seek+read so Windows point reads are serialized and
301        // correct, while still opening the fd only once (issue #1573 roborev
302        // finding). This intentionally trades Windows concurrency for correctness;
303        // the Unix arm above needs no lock because `FileExt::read_at` is positional
304        // and never touches the fd offset. Recover a poisoned lock in-place (a
305        // panicking reader cannot leave the handle in a torn state) rather than
306        // `unwrap`/`expect` in library code.
307        use std::os::windows::fs::FileExt;
308        let file = self.file.lock().unwrap_or_else(|e| e.into_inner());
309        Ok(file.seek_read(buf, offset)?)
310    }
311
312    fn len(&self) -> u64 {
313        self.len
314    }
315}

318///
319/// A slice read out of the resident map — zero syscalls, and trivially `&self`
320/// (the map is immutable). Shares the reader's existing `Arc<Mmap>` when the
321/// backend is mmap, so no extra mapping or fd is created.

323    mmap: Arc<Mmap>,
324}

326impl MmapReadAt {
327    pub(crate) fn new(mmap: Arc<Mmap>) -> Self {
328        Self { mmap }
329    }

332impl ReadAt for MmapReadAt {
333    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize> {
334        let data: &[u8] = &self.mmap;
335        let len = data.len() as u64;
336        // At/past EOF: zero bytes, matching `File`/pread semantics.
337        if offset >= len {
338            return Ok(0);
339        }
340        let start = offset as usize;
341        let avail = &data[start..];
342        let n = avail.len().min(buf.len());
343        buf[..n].copy_from_slice(&avail[..n]);
344        Ok(n)
345    }
346
347    fn len(&self) -> u64 {
348        self.mmap.len() as u64
349    }
350}

352/// I/O alignment (bytes) for `O_DIRECT`: the offset, length, and buffer must all
353/// be a multiple of the device logical block size. 4096 is a safe superset of the
354/// common 512/4096 sector sizes (mirrors `source::DIRECT_IO_ALIGN`).
355#[cfg(unix)]
356const DIRECT_IO_ALIGN: usize = 4096;
357
358/// Positional reads over a file opened with direct I/O (`O_DIRECT` / `F_NOCACHE`).
359///
360/// `O_DIRECT` mandates aligned transfers, so each `read_at` allocates its OWN
361/// aligned bounce buffer, `pread`s an aligned superset of the requested range,
362/// and copies out the requested sub-range. Because the bounce buffer is per-call
363/// (never shared), the backend keeps `&self` with no shared mutable cursor — two
364/// concurrent `read_at`s never touch the same buffer.
365#[cfg(unix)]
366pub(crate) struct DirectReadAt {
367    file: Arc<std::fs::File>,
368    len: u64,
369}

375    /// I/O so the caller can fall back to buffered.

377        let file = super::source::open_direct_file(path)?;
378        crate::storage::sstable::read_work_counters::record_file_open();
379        Ok(Self {
380            file: Arc::new(file),

402#[cfg(unix)]
403impl DirectReadAt {
404    /// Aligned positioned read shared by the per-call [`ReadAt::read_at`] and the
405    /// buffer-reusing [`ReadAt::read_exact_at_reusing`] paths (issue #2319).
406    ///
407    /// `bounce` is a slot for the aligned transfer buffer. When it already holds a
408    /// buffer big enough for this read's aligned span, that buffer is REUSED
409    /// (no allocation); otherwise a new one is allocated INTO the slot so a later
410    /// read reuses it. `read_at` passes a throwaway local slot (per-call alloc, as
411    /// before); the windowed scan passes its per-scan [`DirectScratch`] slot so one
412    /// buffer serves every chunk read.
413    fn read_at_into(
414        &self,
415        offset: u64,
416        buf: &mut [u8],
417        bounce: &mut Option<AlignedBuf>,
418    ) -> Result<usize> {
419        use std::os::unix::fs::FileExt;
420        if offset >= self.len || buf.is_empty() {
421            return Ok(0);
422        }
423        let align = DIRECT_IO_ALIGN as u64;
424        let aligned_off = (offset / align) * align;
425        let head = (offset - aligned_off) as usize;
426        // Round the aligned span up to cover [offset, offset+buf.len()).
427        let want_end = offset
428            .checked_add(buf.len() as u64)
429            .ok_or_else(|| Error::corruption("direct read_at: offset+len overflow"))?
430            .min(self.len);
431        let span = (want_end - aligned_off) as usize;
432        let aligned_len = span
433            .checked_next_multiple_of(DIRECT_IO_ALIGN)
434            .unwrap_or(usize::MAX & !(DIRECT_IO_ALIGN - 1));
435        // Reuse the slot's buffer when it is already large enough; otherwise
436        // (re)allocate and keep it for subsequent reads. `pread` fills only the
437        // aligned span we ask for, so a larger reused buffer is fine — we read into
438        // its leading `aligned_len` bytes.
439        //
440        // Grow GEOMETRICALLY (issue #2319 MEDIUM): a grow-to-exact `aligned_len`
441        // reallocated on EVERY chunk whose aligned span exceeded the previous one —
442        // for a windowed scan over progressively larger records that meant a realloc
443        // per chunk, defeating the reuse this issue restores. Doubling the previous
444        // capacity (bounded below by the required `aligned_len`) stabilizes the
445        // buffer after a few growths. Both operands are multiples of
446        // `DIRECT_IO_ALIGN` (`aligned_len` via `next_multiple_of`; a prior capacity
447        // was itself allocated as such a multiple), so `new_cap` stays a 4096
448        // multiple and the `O_DIRECT` alignment rule holds.
449        let need_alloc = bounce
450            .as_ref()
451            .map(|b| b.capacity() < aligned_len)
452            .unwrap_or(true);
453        if need_alloc {
454            let old_cap = bounce.as_ref().map(AlignedBuf::capacity).unwrap_or(0);
455            let new_cap = aligned_len.max(old_cap.saturating_mul(2));
456            *bounce = Some(AlignedBuf::new(new_cap, DIRECT_IO_ALIGN)?);
457        }
458        // SAFETY of unwrap avoidance: `bounce` is `Some` here (just set, or already
459        // large enough). Match to keep the no-`unwrap` rule in library code.
460        let slot = match bounce.as_mut() {
461            Some(b) => b,
462            None => return Err(Error::corruption("direct read_at: bounce buffer missing")),
463        };
464        let got = self
465            .file
466            .read_at(&mut slot.as_mut_slice()[..aligned_len], aligned_off)?;
467        if got <= head {
468            return Ok(0);
469        }
470        let usable = &slot.as_slice()[head..got];
471        let n = usable.len().min(buf.len());
472        buf[..n].copy_from_slice(&usable[..n]);
473        Ok(n)
474    }
475}

477#[cfg(unix)]
478impl ReadAt for DirectReadAt {
479    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize> {
480        // Per-call bounce buffer (no reuse) — the point-read path allocates one per
481        // call, exactly as before issue #2319.
482        let mut bounce: Option<AlignedBuf> = None;
483        self.read_at_into(offset, buf, &mut bounce)
484    }
485
486    fn len(&self) -> u64 {
487        self.len
488    }
489
490    fn read_exact_at_reusing(
491        &self,
492        offset: u64,
493        buf: &mut [u8],
494        scratch: &mut DirectScratch,
495    ) -> Result<()> {
496        // Reuse the caller's per-scan aligned bounce buffer across every chunk read
497        // (issue #2319). Loops `read_at_into` for exact semantics, threading the ONE
498        // `scratch.bounce` buffer so no per-chunk aligned allocation occurs.
499        let mut filled = 0usize;
500        while filled < buf.len() {
501            let n = self.read_at_into(
502                offset + filled as u64,
503                &mut buf[filled..],
504                &mut scratch.bounce,
505            )?;
506            if n == 0 {
507                return Err(Error::Io(std::io::Error::new(
508                    std::io::ErrorKind::UnexpectedEof,
509                    format!(
510                        "read_exact_at_reusing: reached EOF after {filled} of {} bytes at \
511                         offset 0x{offset:x}",
512                        buf.len()
513                    ),
514                )));
515            }
516            filled += n;
517        }
518        Ok(())
519    }
520}

522/// A heap allocation whose start address is aligned to a requested boundary, as
523/// required for `O_DIRECT` transfer buffers. Local to the point-read path so it
524/// stays independent of the scan-cursor `AlignedBuf` in `source.rs`.
525#[cfg(unix)]
526struct AlignedBuf {
527    ptr: std::ptr::NonNull<u8>,
528    layout: std::alloc::Layout,
529}
530
531// SAFETY: `AlignedBuf` uniquely owns its allocation for its whole lifetime; the
532// raw pointer is never aliased or shared, so moving it across threads is sound.
533#[cfg(unix)]
534unsafe impl Send for AlignedBuf {}
535
536/// Test-only counter of [`AlignedBuf`] allocations, so the issue #2319 reuse test
537/// can PROVE the Direct windowed-scan read path allocates ONE aligned bounce buffer
538/// across N chunk reads (reused) rather than one per read. Process-global; the
539/// reuse test serializes on it and resets before measuring.
540#[cfg(all(test, unix))]
541pub(crate) static ALIGNED_BUF_ALLOCS: std::sync::atomic::AtomicUsize =
542    std::sync::atomic::AtomicUsize::new(0);
543
544#[cfg(unix)]
545impl AlignedBuf {
546    fn new(size: usize, align: usize) -> Result<Self> {
547        #[cfg(test)]
548        ALIGNED_BUF_ALLOCS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
549        let size = size.max(align);
550        let layout = std::alloc::Layout::from_size_align(size, align)
551            .map_err(|e| Error::Io(std::io::Error::new(std::io::ErrorKind::InvalidInput, e)))?;
552        // SAFETY: `layout` has non-zero size (size >= align >= 1).
553        let raw = unsafe { std::alloc::alloc_zeroed(layout) };
554        let ptr = std::ptr::NonNull::new(raw).ok_or_else(|| {
555            Error::Io(std::io::Error::new(
556                std::io::ErrorKind::OutOfMemory,
557                "aligned alloc failed",
558            ))
559        })?;
560        Ok(Self { ptr, layout })
561    }
562
563    fn as_slice(&self) -> &[u8] {
564        // SAFETY: `ptr` is valid for `layout.size()` initialised bytes (zeroed on
565        // alloc, then overwritten by reads) and borrowed immutably here.
566        unsafe { std::slice::from_raw_parts(self.ptr.as_ptr(), self.layout.size()) }
567    }
568
569    fn as_mut_slice(&mut self) -> &mut [u8] {
570        // SAFETY: `ptr` is valid for `layout.size()` bytes and uniquely borrowed.
571        unsafe { std::slice::from_raw_parts_mut(self.ptr.as_ptr(), self.layout.size()) }
572    }
573
574    /// The number of aligned bytes this buffer can hold — used to decide whether a
575    /// reused buffer is big enough for the next read's aligned span (issue #2319).
576    fn capacity(&self) -> usize {
577        self.layout.size()
578    }
579}

583    fn drop(&mut self) {
584        // SAFETY: `ptr`/`layout` are exactly the values from the matching
585        // `alloc_zeroed`, freed exactly once here.
586        unsafe { std::alloc::dealloc(self.ptr.as_ptr(), self.layout) };
587    }

689    /// The `O_DIRECT` backend's per-call 4K alignment math (`aligned_off`/`head`/
690    /// `want_end`/`checked_next_multiple_of`), aligned bounce buffer, and
691    /// short-read/EOF handling must return EXACTLY the bytes a plain positioned
692    /// read would, at every awkward alignment: offset 0, an unaligned offset, an
693    /// offset spanning a 4K block boundary, and a partial final block at/near EOF.
694    /// `PlainFileReadAt` over the same file is the oracle. Guarded `#[cfg(unix)]`
695    /// and SKIPS GRACEFULLY when the temp filesystem refuses `O_DIRECT` (common on
696    /// tmpfs/overlayfs) — mirroring `source::direct_cursor_reads_and_seeks`.
697    #[cfg(unix)]
698    #[test]
699    // Shares the process-global `ALIGNED_BUF_ALLOCS` counter with the #2319 reuse
700    // test, so both run serially to keep that counter's measurements isolated.
701    #[serial_test::serial(aligned_buf_allocs)]
702    fn direct_read_at_matches_plain_oracle_at_boundaries() {
703        // File size deliberately NOT a 4096 multiple → forces a partial final
704        // block and non-trivial `aligned_len` rounding.
705        let len = DIRECT_IO_ALIGN * 2 + 1234;
706        let mut data = vec![0u8; len];
707        // Deterministic pseudo-random content (LCG) so mismatches are meaningful.
708        let mut x = 0x1234_5678u32;
709        for b in data.iter_mut() {
710            x = x.wrapping_mul(1_664_525).wrapping_add(1_013_904_223);
711            *b = (x >> 24) as u8;
712        }
713        let path = temp_file(&data, "direct");
714        let oracle = PlainFileReadAt::open(&path, data.len() as u64).unwrap();
715
716        // Open may fail on filesystems without O_DIRECT support → skip cleanly.
717        let direct = match DirectReadAt::open(&path, data.len() as u64) {
718            Ok(d) => d,
719            Err(e) => {
720                eprintln!("O_DIRECT open unsupported here ({e}); skipping DirectReadAt test");
721                std::fs::remove_file(&path).ok();
722                return;
723            }
724        };
725        // Some filesystems accept the O_DIRECT open but refuse aligned reads
726        // (EINVAL) → probe once and skip cleanly rather than fail.
727        let mut probe = [0u8; 16];
728        if let Err(e) = direct.read_at(0, &mut probe) {
729            eprintln!("O_DIRECT read unsupported here ({e}); skipping DirectReadAt test");
730            std::fs::remove_file(&path).ok();
731            return;
732        }
733
734        // (offset, length) pairs exercising each alignment corner.
735        let align = DIRECT_IO_ALIGN as u64;
736        let cases: &[(u64, usize)] = &[
737            (0, 100),                  // offset 0, within first block
738            (37, 200),                 // unaligned, within first block
739            (align - 10, 50),          // spans the first 4K boundary
740            (align, DIRECT_IO_ALIGN),  // aligned start, spans into next block
741            (align + 7, 5000),         // unaligned start crossing multiple blocks
742            ((len - 300) as u64, 300), // exact final bytes (partial final block)
743            ((len - 10) as u64, 5),    // near EOF within the partial final block
744        ];
745        for &(off, n) in cases {
746            let mut want = vec![0u8; n];
747            oracle.read_exact_at(off, &mut want).unwrap();
748            let mut got = vec![0u8; n];
749            direct.read_exact_at(off, &mut got).unwrap();
750            assert_eq!(got, want, "DirectReadAt mismatch at off={off} n={n}");
751        }
752
753        // Short read at EOF: request more than remains → exactly the tail, and
754        // byte-for-byte identical to the oracle's short read.
755        let tail_off = (len - 100) as u64;
756        let mut big = vec![0u8; 500];
757        let got_n = direct.read_at(tail_off, &mut big).unwrap();
758        let mut oracle_big = vec![0u8; 500];
759        let oracle_n = oracle.read_at(tail_off, &mut oracle_big).unwrap();
760        assert_eq!(got_n, oracle_n, "short-read length must match oracle");
761        assert_eq!(got_n, 100, "should read exactly the 100 remaining bytes");
762        assert_eq!(&big[..got_n], &oracle_big[..oracle_n]);
763
764        // Fully past EOF → 0 bytes (matches File/pread and MmapReadAt).
765        let mut none = [0u8; 32];
766        assert_eq!(direct.read_at(len as u64, &mut none).unwrap(), 0);
767        assert_eq!(direct.read_at(len as u64 + align, &mut none).unwrap(), 0);
768
769        // read_exact_at fully past EOF → UnexpectedEof, consistent with the
770        // other backends.
771        let err = direct.read_exact_at(len as u64, &mut none).unwrap_err();
772        match err {
773            Error::Io(e) => assert_eq!(e.kind(), std::io::ErrorKind::UnexpectedEof),
774            other => panic!("expected UnexpectedEof, got {other:?}"),
775        }
776
777        std::fs::remove_file(&path).ok();
778    }

780    /// Issue #2319: the Direct-I/O windowed-scan read path REUSES one aligned bounce
781    /// buffer across chunk reads instead of allocating a fresh ~chunk-sized aligned
782    /// buffer per read (the per-chunk-alloc regression #1940 introduced).
783    ///
784    /// This asserts the ALLOCATION BEHAVIOUR of the code path -- the pure user-space
785    /// alignment + bounce-buffer logic -- so it needs NO working `O_DIRECT` syscall
786    /// (frequently unavailable on macOS/tmpfs/CI; probing it live is what stalled a
787    /// prior implementer). `DirectReadAt::from_plain_fd_for_test` runs that exact
788    /// path over a regular fd; the bounce buffer, alignment math, and reuse decision
789    /// are identical whether or not the fd was opened with direct I/O.
790    ///
791    /// The three scenarios live in ONE `#[test]` function on purpose: they share the
792    /// process-global [`ALIGNED_BUF_ALLOCS`] counter (serialized on `aligned_buf_allocs`)
793    /// and folding them here keeps the crate's test-function COUNT identical to before
794    /// #2319, so the in-process test parallelism -- and thus the timing of unrelated
795    /// process-global work-counter tests -- is unperturbed.
796    ///
797    /// Scenario 1 (equal-sized chunks -- reuse vs per-call contrast):
798    /// - The OLD per-chunk path (`read_at`, still the point-read behaviour) allocates
799    ///   ONE aligned buffer PER read -> `N` allocs for `N` reads (> 1). This is the
800    ///   regression state the reuse path fixes; the assertion below fails on it.
801    /// - The FIXED windowed path (`read_exact_at_reusing` with a shared
802    ///   [`DirectScratch`]) allocates the buffer ONCE and reuses it -> exactly 1 alloc
803    ///   across all `N` reads, in steady state.
804    ///
805    /// Scenario 2 (increasing sizes -- geometric growth, issue #2319 MEDIUM): with the
806    /// reused scratch fed reads of STRICTLY INCREASING aligned span, the buffer grows
807    /// GEOMETRICALLY -- a small constant number of reallocations, NOT one per read. The
808    /// pre-fix grow-to-exact code reallocated on every size increase, preserving the
809    /// very per-chunk-alloc regression #2319 removes; scenario 1 cannot catch it
810    /// because its span never grows.
811    ///
812    /// Scenario 3 (Arc-dyn forward -- wiring evidence): production reaches the reuse
813    /// path through an `Arc<dyn ReadAt>` (`point_source`), whose blanket-impl forward
814    /// ([`ReadAt for Arc<T>`]) must delegate `read_exact_at_reusing` to the concrete
815    /// `DirectReadAt` override. If that forward hop were dropped the `Arc` would fall
816    /// back to the per-chunk-alloc default and production would regress while a
817    /// concrete-typed assertion stayed green -- so this exercises reuse THROUGH the dyn
818    /// `Arc` and asserts a single allocation.
819    ///
820    /// Serialized on the process-global [`ALIGNED_BUF_ALLOCS`] counter.
821    #[cfg(unix)]
822    #[test]
823    #[serial_test::serial(aligned_buf_allocs)]
824    fn direct_windowed_read_reuses_one_bounce_buffer_across_chunks() {
825        use std::sync::atomic::Ordering;
826
827        // Deterministic pseudo-random content (LCG) so mismatches are meaningful.
828        fn fill(data: &mut [u8], mut x: u32) {
829            for b in data.iter_mut() {
830                x = x.wrapping_mul(1_664_525).wrapping_add(1_013_904_223);
831                *b = (x >> 24) as u8;
832            }
833        }
834
835        // ---- Scenario 1: equal-sized chunks, reuse vs per-call contrast ----
836        // 8 aligned, equal-sized "chunks" so every read's aligned span is identical
837        // (`head == 0`, span == chunk) -> the reused buffer never needs to grow, so a
838        // reused buffer means EXACTLY one allocation across all reads.
839        let chunk = DIRECT_IO_ALIGN; // 4096
840        let n_chunks = 8usize;
841        let len = chunk * n_chunks;
842        let mut data = vec![0u8; len];
843        fill(&mut data, 0x9e37_79b9);
844        let path = temp_file(&data, "reuse");
845        let file = std::fs::File::open(&path).unwrap();
846        let direct = DirectReadAt::from_plain_fd_for_test(file, len as u64);
847
848        // FIXED windowed path: one shared scratch reused across every chunk read.
849        ALIGNED_BUF_ALLOCS.store(0, Ordering::Relaxed);
850        let mut scratch = DirectScratch::new();
851        for i in 0..n_chunks {
852            let off = (i * chunk) as u64;
853            let mut got = vec![0u8; chunk];
854            direct
855                .read_exact_at_reusing(off, &mut got, &mut scratch)
856                .unwrap();
857            assert_eq!(
858                got,
859                &data[i * chunk..(i + 1) * chunk],
860                "reused-buffer read returned wrong bytes for chunk {i}"
861            );
862        }
863        let reuse_allocs = ALIGNED_BUF_ALLOCS.load(Ordering::Relaxed);
864        assert_eq!(
865            reuse_allocs, 1,
866            "#2319: the Direct windowed-scan path must allocate ONE aligned bounce buffer \
867             and reuse it across all {n_chunks} chunk reads (got {reuse_allocs} allocs)"
868        );
869
870        // OLD per-chunk path: `read_at` allocates a fresh bounce buffer per call.
871        // This is the regression the reuse path removes; it allocates once per read,
872        // so `> 1` for `N > 1` -- the exact behaviour the assertion above rejects.
873        ALIGNED_BUF_ALLOCS.store(0, Ordering::Relaxed);
874        for i in 0..n_chunks {
875            let off = (i * chunk) as u64;
876            let mut got = vec![0u8; chunk];
877            direct.read_at(off, &mut got).unwrap();
878        }
879        let percall_allocs = ALIGNED_BUF_ALLOCS.load(Ordering::Relaxed);
880        assert_eq!(
881            percall_allocs, n_chunks,
882            "#2319 (regression witness): the per-call point-read path allocates one aligned \
883             buffer PER read ({n_chunks} reads -> {n_chunks} allocs); the windowed scan must \
884             NOT use it (got {percall_allocs})"
885        );
886        assert!(
887            reuse_allocs < percall_allocs,
888            "#2319: reuse ({reuse_allocs}) must allocate strictly fewer buffers than the \
889             per-chunk path ({percall_allocs})"
890        );
891        std::fs::remove_file(&path).ok();
892
893        // ---- Scenario 2: increasing sizes must grow GEOMETRICALLY (#2319 MEDIUM) ----
894        // `M` reads at offset 0 of length `(i+1)*4096` need caps 4096, 8192, ... 32768.
895        // - grow-to-exact: every cap strictly exceeds the last -> `M` allocs (== reads).
896        // - geometric (max(need, old*2)): 4096->8192->16384->32768 -> ~log2(M) allocs,
897        //   well under `M`. The bound below fails on the old code and passes on the fix.
898        let n_reads = 8usize;
899        let grow_len = chunk * n_reads; // largest read spans the whole file
900        let mut grow_data = vec![0u8; grow_len];
901        fill(&mut grow_data, 0x1357_9bdf);
902        let grow_path = temp_file(&grow_data, "reuse_grow");
903        let grow_file = std::fs::File::open(&grow_path).unwrap();
904        let grow_direct = DirectReadAt::from_plain_fd_for_test(grow_file, grow_len as u64);
905
906        ALIGNED_BUF_ALLOCS.store(0, Ordering::Relaxed);
907        let mut grow_scratch = DirectScratch::new();
908        for i in 0..n_reads {
909            let want_len = (i + 1) * chunk; // strictly increasing aligned span
910            let mut got = vec![0u8; want_len];
911            grow_direct
912                .read_exact_at_reusing(0, &mut got, &mut grow_scratch)
913                .unwrap();
914            assert_eq!(
915                got,
916                &grow_data[..want_len],
917                "increasing-size reused-buffer read returned wrong bytes at read {i}"
918            );
919        }
920        let grow_allocs = ALIGNED_BUF_ALLOCS.load(Ordering::Relaxed);
921        // A few geometric growths, NOT one per read. Bound comfortably above the
922        // ~log2(M)=4 growths yet strictly below `n_reads` so grow-to-exact (which
923        // reallocates every read -> n_reads allocs) fails here.
924        assert!(
925            grow_allocs <= 5,
926            "#2319: geometric growth must bound reallocations to a small constant across \
927             {n_reads} increasing-size reads (got {grow_allocs} allocs)"
928        );
929        assert!(
930            grow_allocs < n_reads,
931            "#2319: growth must allocate strictly fewer times than {n_reads} reads -- \
932             grow-to-exact reallocated on every size increase (got {grow_allocs})"
933        );
934        std::fs::remove_file(&grow_path).ok();
935
936        // ---- Scenario 3: reuse must survive the Arc<dyn ReadAt> forward hop ----
937        let arc_chunks = 6usize;
938        let arc_len = chunk * arc_chunks;
939        let mut arc_data = vec![0u8; arc_len];
940        fill(&mut arc_data, 0x2468_ace0);
941        let arc_path = temp_file(&arc_data, "reuse_arc");
942        let arc_file = std::fs::File::open(&arc_path).unwrap();
943        // Erase to the exact production type: an `Arc<dyn ReadAt>` (point_source).
944        let src: Arc<dyn ReadAt> = Arc::new(DirectReadAt::from_plain_fd_for_test(
945            arc_file,
946            arc_len as u64,
947        ));
948
949        ALIGNED_BUF_ALLOCS.store(0, Ordering::Relaxed);
950        let mut arc_scratch = DirectScratch::new();
951        for i in 0..arc_chunks {
952            let off = (i * chunk) as u64;
953            let mut got = vec![0u8; chunk];
954            // Goes Arc<dyn ReadAt> -> forward -> DirectReadAt::read_exact_at_reusing.
955            src.read_exact_at_reusing(off, &mut got, &mut arc_scratch)
956                .unwrap();
957            assert_eq!(
958                got,
959                &arc_data[i * chunk..(i + 1) * chunk],
960                "Arc-dyn reused-buffer read returned wrong bytes for chunk {i}"
961            );
962        }
963        let arc_allocs = ALIGNED_BUF_ALLOCS.load(Ordering::Relaxed);
964        assert_eq!(
965            arc_allocs, 1,
966            "#2319: the Arc<dyn ReadAt> forward must reach DirectReadAt's reuse override -- \
967             exactly ONE aligned buffer across {arc_chunks} equal-sized chunk reads (got \
968             {arc_allocs}); a dropped forward would fall back to the per-chunk-alloc default"
969        );
970        std::fs::remove_file(&arc_path).ok();
971    }
