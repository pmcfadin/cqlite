1//! Synchronous positional chunk reader for the windowed streaming scan's IO half
2//! (issue #1940 restructure).
3//!
4//! # Why this exists
5//!
6//! The windowed feed runs inside a single `spawn_blocking` task (issues #1593,
7//! #1940): both the raw chunk read AND its decompression must stay OFF the async
8//! worker pool. The former implementation drove the read with
9//! `futures::executor::block_on(read_next_block_parts(..))` — an async read over a
10//! `tokio::fs::File`. Even though it ran on a blocking thread, `tokio::fs`'s
11//! `poll_read` RE-DISPATCHES the actual `read(2)` to tokio's blocking pool, so each
12//! chunk read amplified into a second blocking-pool thread. rust-reviewer proved it
13//! bounded, but roborev flagged it as a latent hang under a small custom
14//! `max_blocking_threads`. Owner decision (issue #1940): REMOVE the hazard class —
15//! read SYNCHRONOUSLY on the `spawn_blocking` thread, with zero nested async and
16//! zero blocking-pool amplification.
17//!
18//! These helpers read via the reader's `point_source` — the positional
19//! [`ReadAt`](super::read_at::ReadAt) plane built once at open for EVERY backend
20//! (buffered = `pread` on a dedicated `std::fs::File`; mmap = a resident slice;
21//! `O_DIRECT` = an aligned positioned read). A positional read carries its offset
22//! as a parameter and touches no tokio reactor/timer, so it completes fully on the
23//! calling (blocking) thread — no `tokio::fs`, no `block_on`, no second blocking
24//! thread. `read_at.rs`'s module docs anticipated exactly this adoption ("shaped so
25//! the windowed streaming scan can adopt it later"). Decompression continues to run
26//! in `decode_scan_chunk` on the same blocking thread (the D2 substrate placement,
27//! unchanged).
28//!
29//! Byte-parity + integrity are preserved verbatim from the former cursor path:
30//! - **compressed NB** (`CompressionInfo` present): the same per-chunk record
31//!   geometry (`compressed_chunk_offset`/`compressed_chunk_size`, incl. the +4
32//!   trailing CRC), the same corrupt-offset bounds check (roborev #970), the same
33//!   degenerate empty-trailing-chunk EOF bound (issue #2225), and the same
34//!   CRC32-BEFORE-decompress order (the CRC is verified here; the payload is
35//!   returned only after it matches, so a corrupt chunk never reaches the
36//!   decompressor). The compressed read reuses the caller's per-loop scratch (≤1
37//!   alloc/chunk, D2) and records a scratch-regrowth as a copy-chain alloc so the
38//!   `issue_1940_scan_window_substrate_allocs` guard stays honest.
39//! - **uncompressed NB** (no `CompressionInfo`; CQLite's own write surface): the
40//!   data section is fed in bounded pieces, and every `CRC.db` chunk covering a
41//!   piece is verified BEFORE the piece is handed on — computing the CRC from the
42//!   resident piece bytes when a chunk is fully contained, else reading the
43//!   straddling prefix/suffix positionally (mirrors
44//!   `verify_uncompressed_section_in_buffer`). Each chunk is verified at most once
45//!   per reader lifetime via the shared `verified_uncompressed_chunks` memo.
46//!
47//! Kept in a sibling file (campsite rule, epic #1116); included via
48//! `#[path = "scan_stream_windowed_read.rs"] impl`-carrying module in the parent.

58/// Bytes handed to the parse half per uncompressed-NB read. Mirrors Cassandra's
59/// default 64 KiB compression chunk so the piece size is uniform with the
60/// compressed path; the sliding window stitches across pieces, so the boundary is
61/// invisible to the parse output. CRC verification is done on `CRC.db` chunk
62/// boundaries independently of this piece size.
63const UNCOMPRESSED_READ_PIECE_BYTES: usize = 64 * 1024;
64
65/// Wrap a positional-read failure with human-readable `context` WITHOUT
66/// discarding its [`ErrorKind`](std::io::ErrorKind) (issue #1940 BLOCKER-2).
67///
68/// The synchronous positional read path used to relabel every read failure as
69/// `std::io::Error::other(..)` (kind `Other`), which erased the source kind — so
70/// a genuine transient fault (`Interrupted`/`WouldBlock`/`TimedOut`) surfacing
71/// through the feed could no longer be recognised by [`is_transient_io`] and was
72/// silently NOT retried. Rebuilding the wrapper with the SAME kind keeps the
73/// classifier honest and lets callers up the stack still see the true io kind,
74/// mirroring `block_io::io_error_with_context`. Non-`Io` variants
75/// (corruption/format) are already typed and pass through unchanged.
76fn io_error_with_context(context: impl std::fmt::Display, source: Error) -> Error {
77    match source {
78        Error::Io(io) => {
79            let kind = io.kind();
80            Error::Io(std::io::Error::new(kind, format!("{context}: {io}")))
81        }
82        other => other,
83    }
84}

86impl SSTableReader {
87    /// Positional `read_exact_at` that applies the SAME one-shot transient-I/O
88    /// retry the former async feed path had (issue #1588 / #1940 BLOCKER-2), and
89    /// records the read thread at the REAL read site (issue #1940 guard
90    /// integrity).
91    ///
92    /// A transient kernel fault (`Interrupted` EINTR / `WouldBlock` EAGAIN /
93    /// `TimedOut`) is retried EXACTLY once by re-reading at the SAME captured
94    /// `offset` (a positional read carries its offset as a parameter, so the retry
95    /// needs no re-seek); every other error — deterministic corruption/format, or
96    /// a non-transient io kind — fails fast. The returned error is the backend's
97    /// `Error::Io`, so its `ErrorKind` is PRESERVED for the caller to annotate via
98    /// [`io_error_with_context`] (never collapsed to `Error::other`).
99    ///
100    /// The `record_io_read_thread` probe fires HERE, at the actual
101    /// `point_source.read_exact_at` call site (not at the top of the feed closure),
102    /// so the #1940 no-nesting guard records the thread that performs the real read
103    /// — making its `io_read_thread == decode_thread` equality a true detector of a
104    /// read dispatched off the feed thread. Compiled only under `scan-offload-probe`.
105    fn positional_read_exact_retry_once(&self, offset: u64, buf: &mut [u8]) -> Result<()> {
106        #[cfg(feature = "scan-offload-probe")]
107        super::probe::record_io_read_thread();
108        let mut attempt = 0u8;
109        loop {
110            match self.point_source.read_exact_at(offset, buf) {
111                Ok(()) => return Ok(()),
112                Err(e)
113                    if attempt == 0
114                        && crate::storage::sstable::reader::block_io::is_transient_io(&e) =>
115                {
116                    tracing::warn!(
117                        "transient I/O fault ({e}) on positional read at offset 0x{offset:x}; \
118                         re-reading once"
119                    );
120                    attempt = 1;
121                }
122                Err(e) => return Err(e),
123            }
124        }
125    }

127    /// Read compressed chunk `chunk_index` SYNCHRONOUSLY via `point_source`,
128    /// verifying its trailing CRC32 before returning the compressed payload (issue
129    /// #1940). Returns `Ok(None)` at EOF (past the last chunk, or the degenerate
130    /// empty trailing chunk, issue #2225). Mirrors `read_nb_format_chunk_data`'s
131    /// geometry + integrity checks exactly, but positional (no cursor mutex, no
132    /// `tokio::fs`) and reusing the caller's `scratch` (≤1 alloc/chunk, D2).
133    pub(super) fn read_compressed_chunk_sync(
134        &self,
135        chunk_index: usize,
136        scratch: &mut Vec<u8>,
137    ) -> Result<Option<Vec<u8>>> {
138        let comp_info = match self.compression_info.as_ref() {
139            Some(ci) => ci,
140            // Callers gate on `compression_info.is_some()`; a None here is a bug.
141            None => return Ok(None),
142        };
143        let file_size = self.point_source.len();
144
145        if chunk_index >= comp_info.chunk_offsets.len() {
146            return Ok(None); // EOF
147        }
148
149        // Degenerate empty trailing chunk (issue #2225): a chunk whose logical
150        // start is at/after `data_length` carries 0 real bytes; Cassandra's reader
151        // never touches it (every logical position < data_length maps earlier), and
152        // handing 0 bytes to a decompressor fails. Treat it (and anything beyond,
153        // chunk starts being monotonic) as EOF — identical to the cursor path.
154        let chunk_length = comp_info.chunk_length as u64;
155        if chunk_length > 0 {
156            let logical_start = (chunk_index as u64).saturating_mul(chunk_length);
157            if logical_start >= comp_info.data_length {
158                return Ok(None);
159            }
160        }
161
162        let chunk_offset = comp_info
163            .compressed_chunk_offset(chunk_index)
164            .ok_or_else(|| Error::InvalidFormat(format!("No offset for chunk {chunk_index}")))?;
165        let total_chunk_size = comp_info
166            .compressed_chunk_size(chunk_index, file_size)
167            .ok_or_else(|| {
168                Error::InvalidFormat(format!(
169                    "Cannot determine size for chunk {chunk_index} (file_size={file_size})"
170                ))
171            })?;
172        if total_chunk_size < 4 {
173            return Err(Error::InvalidFormat(format!(
174                "Chunk {chunk_index} size too small: {total_chunk_size} bytes (minimum 4 for CRC)"
175            )));
176        }
177        // Bounds-check against the real Data.db length BEFORE allocating — a corrupt
178        // CompressionInfo offset must surface as a recoverable error, never a
179        // multi-exabyte allocation (roborev #970). NB chunk offsets are absolute
180        // (header_offset = 0).
181        let chunk_end = chunk_offset.checked_add(total_chunk_size);
182        match chunk_end {
183            Some(end) if end <= file_size => {}
184            _ => {
185                return Err(Error::InvalidFormat(format!(
186                    "Chunk {chunk_index} at offset 0x{chunk_offset:x} with size {total_chunk_size} \
187                     exceeds Data.db length {file_size} — corrupt CompressionInfo.db chunk offset"
188                )));
189            }
190        }
191        let chunk_data_size = (total_chunk_size - 4) as usize;
192
193        // A5 counters (E4 SEEK_CALLS + E3 READ_CALLS): the positioned read stands in
194        // for the cursor path's per-chunk seek + read, so record one of each per
195        // chunk to keep the `reads == decompresses` substrate invariant intact.
196        rwc::record_seek();
197        rwc::record_read();
198
199        // ONE positioned read for payload + trailing CRC into the REUSED scratch
200        // (D2). Grow-on-demand to the CHECKED `total_chunk_size` for THIS chunk —
201        // never a giant upfront reserve to the metadata `max_compressed_length`
202        // (which Cassandra commonly stamps as `i32::MAX`, an OOM/DoS on ordinary
203        // files, issue #1940 BLOCKER-1). `try_reserve_exact` returns a typed error
204        // on allocation failure instead of panicking/aborting. After the first few
205        // chunks the scratch has reached the working-set high-water mark, so
206        // subsequent chunks hit neither the reserve nor the resize regrowth —
207        // preserving the ≤1-alloc/chunk STEADY STATE (the `record_chunk_path_alloc`
208        // regrowth counter proves it). `clear()`+`resize` reuses the backing store
209        // when large enough (mirrors `read_nb_format_chunk_data`).
210        let mut buf = std::mem::take(scratch);
211        buf.clear();
212        let cap_before = buf.capacity();
213        let need = total_chunk_size as usize;
214        if need > buf.capacity() {
215            if let Err(e) = buf.try_reserve_exact(need) {
216                // Return the scratch to the caller so a retry could reuse it.
217                *scratch = buf;
218                return Err(Error::memory(format!(
219                    "failed to reserve {need} bytes for chunk {chunk_index} \
220                     (data+CRC at offset 0x{chunk_offset:x}): {e}"
221                )));
222            }
223        }
224        buf.resize(need, 0u8);
225        if buf.capacity() > cap_before {
226            rwc::record_chunk_path_alloc();
227        }
228        if let Err(e) = self.positional_read_exact_retry_once(chunk_offset, &mut buf) {
229            // Return the scratch's capacity to the caller so the next read reuses it.
230            buf.clear();
231            *scratch = buf;
232            // Preserve the source io kind (issue #1940 BLOCKER-2) — never collapse
233            // a transient fault to `Error::other`.
234            return Err(io_error_with_context(
235                format!(
236                    "Failed to read chunk {chunk_index} data+CRC ({total_chunk_size} bytes at \
237                     offset 0x{chunk_offset:x})"
238                ),
239                e,
240            ));
241        }
242
243        // Split the trailing 4-byte big-endian CRC32 off the payload and verify it
244        // BEFORE the payload is trusted (guardrail #1411): a mismatch is a typed
245        // error and the payload never reaches the decompressor.
246        let expected_crc = u32::from_be_bytes([
247            buf[chunk_data_size],
248            buf[chunk_data_size + 1],
249            buf[chunk_data_size + 2],
250            buf[chunk_data_size + 3],
251        ]);
252        buf.truncate(chunk_data_size);
253        let computed_crc = crc32fast::hash(&buf);
254        if computed_crc != expected_crc {
255            return Err(Error::InvalidFormat(format!(
256                "CRC32 mismatch for chunk {chunk_index} at offset 0x{chunk_offset:x}: \
257                 expected=0x{expected_crc:08x}, computed=0x{computed_crc:08x}, \
258                 chunk_size={chunk_data_size}"
259            )));
260        }
261        Ok(Some(buf))
262    }

264    /// Read the next bounded piece of an UNCOMPRESSED-NB data section
265    /// SYNCHRONOUSLY via `point_source`, starting at byte `pos` (an absolute
266    /// Data.db offset). Returns `Ok(None)` at EOF, else `Ok(Some((piece, next_pos)))`.
267    /// Every `CRC.db` chunk covering the piece is verified (once per reader
268    /// lifetime) BEFORE the piece is returned — from the resident bytes when a
269    /// chunk is fully contained, else by reading the straddling remainder
270    /// positionally (issue #1396; mirrors `verify_uncompressed_section_in_buffer`).
271    pub(super) fn read_uncompressed_piece_sync(&self, pos: u64) -> Result<Option<(Vec<u8>, u64)>> {
272        let file_size = self.point_source.len();
273        let remaining = file_size.saturating_sub(pos);
274        if remaining == 0 {
275            return Ok(None); // EOF
276        }
277        let to_read = remaining.min(UNCOMPRESSED_READ_PIECE_BYTES as u64) as usize;
278        let mut buf = vec![0u8; to_read];
279        // Same one-shot transient-retry + kind-preserving wrap as the compressed
280        // path (issue #1940 BLOCKER-2): a transient fault is re-read once, and the
281        // source io kind is preserved (never collapsed to `Error::other`).
282        self.positional_read_exact_retry_once(pos, &mut buf)
283            .map_err(|e| {
284                io_error_with_context(
285                    format!("Failed to read uncompressed piece ({to_read} bytes at 0x{pos:x})"),
286                    e,
287                )
288            })?;
289        let end = pos + to_read as u64;
290
291        // CRC verification (no-op when this reader has no CRC.db). Compute each
292        // covering chunk's CRC from the resident piece when fully contained, else
293        // read the straddling prefix/suffix positionally. `verify_covering_chunks`
294        // is synchronous and memoizes each chunk so it is checked at most once.
295        if let Some(crc) = self.crc_reader.as_deref() {
296            let piece = &buf;
297            self.verify_covering_chunks(crc, pos, end, |lo, hi, chunk| {
298                if lo >= pos && hi <= end {
299                    let s = (lo - pos) as usize;
300                    let e = (hi - pos) as usize;
301                    Ok(crc32fast::hash(&piece[s..e]))
302                } else {
303                    let mut cbuf = vec![0u8; (hi - lo) as usize];
304                    self.point_source
305                        .read_exact_at(lo, &mut cbuf)
306                        .map_err(|e| {
307                            Error::corruption(format!(
308                                "failed to read uncompressed chunk {chunk} at Data.db offset \
309                             0x{lo:x} for CRC verification: {e}"
310                            ))
311                        })?;
312                    Ok(crc32fast::hash(&cbuf))
313                }
314            })?;
315        }
316        Ok(Some((buf, end)))
317    }

319    /// Synchronous compressed-NB feed body (issue #1940): read each compressed
320    /// chunk positionally (CRC verified in the read), decode it (D2), and ship the
321    /// decompressed `Bytes` on `raw_tx`. `Ok(())` on clean EOF or
322    /// consumer-ended-early; `Err` on a read/decode failure. Runs on the feed's
323    /// `spawn_blocking` thread — no `tokio::fs`, no `block_on`, no blocking-pool
324    /// amplification.
325    pub(super) fn feed_compressed_chunks(
326        reader: &Arc<Self>,
327        raw_tx: &mpsc::Sender<bytes::Bytes>,
328        max_compressed_length: usize,
329    ) -> Result<()> {
330        // Reused compressed-read scratch (issue #1940, D2): read fills it, decode
331        // borrows it, recycle for the next read — no per-chunk compressed alloc.
332        //
333        // Reserve ONCE to the ACTUAL working-set high-water mark — the largest
334        // real compressed-chunk RECORD size, computed from the authoritative
335        // `chunk_offsets` deltas (issue #1940 BLOCKER-1). This is NOT the metadata
336        // `max_compressed_length`: Cassandra COMMONLY stamps that field as
337        // `i32::MAX` (it equals `i32::MAX` whenever `minCompressRatio == 0`, the
338        // DEFAULT — CompressionParams.java:186-189), so a `max_compressed_length`-
339        // driven reserve would attempt a multi-GB allocation on opening an ORDINARY
340        // compressed scan (an OOM/DoS before any real chunk size is known). The
341        // offset-delta high-water mark is bounded by the real per-chunk record size
342        // (a small multiple of `chunk_length`, e.g. 16 KiB), never the file size or
343        // the i32::MAX bound. Reserving to it once means the per-chunk
344        // `clear()`+`resize` never regrows in steady state — preserving the
345        // ≤1-alloc/chunk property the `issue_1940_scan_window_substrate_allocs`
346        // guard measures (the read path's per-chunk `try_reserve_exact` remains a
347        // bounds-checked safety net that records a regrowth only if a chunk ever
348        // exceeds this mark). `try_reserve_exact` returns a typed error on
349        // allocation failure instead of aborting; a corrupt oversized delta surfaces
350        // as `Error::memory`, never a panic. Authoritative metadata, no heuristic.
351        let mut scratch: Vec<u8> = Vec::new();
352        if let Some(ci) = reader.compression_info.as_ref() {
353            let file_size = reader.point_source.len();
354            let max_record = (0..ci.chunk_offsets.len())
355                .filter_map(|i| ci.compressed_chunk_size(i, file_size))
356                .max()
357                .unwrap_or(0);
358            let want = usize::try_from(max_record).unwrap_or(usize::MAX);
359            if want > 0 {
360                scratch.try_reserve_exact(want).map_err(|e| {
361                    Error::memory(format!(
362                        "failed to reserve {want} bytes for the windowed compressed-read scratch \
363                         (largest actual chunk record): {e}"
364                    ))
365                })?;
366            }
367        }
368        let mut chunk_index = 0usize;
369        loop {
370            // SYNCHRONOUS positional read + CRC (issue #1940): CRC is verified inside
371            // the read, BEFORE the payload is trusted (guardrail #1411), exactly as
372            // the former cursor path did.
373            match reader.read_compressed_chunk_sync(chunk_index, &mut scratch)? {
374                Some(compressed) => {
375                    // Decompression stays HERE, on this blocking thread (D2), off the
376                    // async reactor for every backend.
377                    let (decoded, recycled) =
378                        reader.decode_scan_chunk(chunk_index, max_compressed_length, compressed)?;
379                    scratch = recycled;
380                    chunk_index += 1;
381                    if raw_tx.blocking_send(decoded).is_err() {
382                        return Ok(()); // consumer/parse ended early
383                    }
384                }
385                None => return Ok(()), // EOF
386            }
387        }
388    }

390    /// Synchronous uncompressed-NB feed body (issue #1940): read the data section
391    /// in bounded positional pieces (each `CRC.db`-verified before it is shipped),
392    /// decode (zero-copy `Vec`→`Bytes`, no decompressor), and ship on `raw_tx`.
393    /// `Ok(())` on clean EOF or consumer-ended-early; `Err` on a read/CRC failure.
394    pub(super) fn feed_uncompressed_pieces(
395        reader: &Arc<Self>,
396        raw_tx: &mpsc::Sender<bytes::Bytes>,
397    ) -> Result<()> {
398        let mut pos = reader.calculate_header_size() as u64;
399        let mut chunk_index = 0usize;
400        loop {
401            match reader.read_uncompressed_piece_sync(pos)? {
402                Some((piece, next_pos)) => {
403                    // No compressor: `decode_scan_chunk` moves the piece into the B1
404                    // cache zero-copy (`Bytes::from(Vec)`) — no per-piece copy — and
405                    // returns the refcounted substrate the window borrows.
406                    let (decoded, _recycled) =
407                        reader.decode_scan_chunk(chunk_index, usize::MAX, piece)?;
408                    pos = next_pos;
409                    chunk_index += 1;
410                    if raw_tx.blocking_send(decoded).is_err() {
411                        return Ok(()); // consumer/parse ended early
412                    }
413                }
414                None => return Ok(()), // EOF
415            }
416        }
417    }
