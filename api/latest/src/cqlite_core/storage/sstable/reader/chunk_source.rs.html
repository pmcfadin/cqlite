1//! Single chunk decode plane: read → CRC → decompress → B1 cache (issue #1598, Epic G).
2//!
3//! The point-read (BTI get, BIG point), windowed-scan, and BIG-reverse decode paths all
4//! resolve their `Compression::decompress` here. The `iterate_all_partitions` and
5//! `sequential_scan` decode site (`parse_block_entries` in `parsing/block_entries.rs`)
6//! also routes its block decompress through `ChunkSource::decompress_only` (issue #2165),
7//! so `parsing/` no longer calls `Compression::decompress` inline. `parsing/` retains the
8//! legacy `self.file` + `compression_reader` block-read/CRC model (not `ReadAt` +
9//! `CompressionInfo` + chunk-index); only the decompress step is consolidated here.
10//! Architecture test: `tests/chunk_decode_single_plane.rs`.
11

24/// Single chunk decode plane: positioned read → CRC → decompress → B1 cache.
25///
26/// Composes the C2 read+CRC primitive (`read_compressed_chunk_at`) + the best-of-breed
27/// decompress+cache tail (moved from the BTI block). Every query-path chunk read funnels
28/// through one of these entry points, each of which decompresses ONLY CRC-validated bytes:
29/// - `chunk(index)`: whole-chunk read (self read+CRC) for the BTI/windowed-scan paths
30/// - `decode_and_cache`: shared decompress+cache tail for sites that already have
31///   CRC-validated compressed bytes (windowed-scan blocking half; the compressed
32///   offset-read window of `read_compressed_offset_window`, issue #1773)
33/// - `decompress_only`: uncached decompress for the BIG reverse path and the
34///   compressed offset-read window (both already CRC-validated upstream)
35pub(crate) struct ChunkSource<'a> {
36    /// Positioned read source (C2)
37    source: &'a dyn ReadAt,
38    /// Chunk offsets/sizes/algorithm
39    comp_info: &'a CompressionInfo,
40    /// Decompressor (None => raw passthrough)
41    compression: Option<&'a Compression>,
42    /// B1 decompressed-chunk cache
43    cache: &'a DecompressedChunkCache,
44    /// Data.db file size
45    file_size: u64,
46    /// Header offset (always 0 for NB/BTI)
47    header_offset: u64,
48    /// Cache-key namespace salt (NS_*)
49    namespace: u64,
50    /// Stable reader identity
51    cache_id: u64,
52}
53
54/// The bounded [`crate::observability::catalog::attr::COMPRESSION`] label for a
55/// decode site's configured algorithm (issue #1701) — `"none"` when there is none,
56/// never an absent label. Free function taking the ALGORITHM so every decode site
57/// resolves the label identically whether it holds a `Compression`, a
58/// `CompressionReader`, or nothing.
59pub(crate) fn compression_label_of(algorithm: Option<&CompressionAlgorithm>) -> &'static str {
60    match algorithm {
61        Some(a) => read_metrics::compression_attr(a),
62        None => read_metrics::COMPRESSION_NONE,
63    }
64}

66/// Count a chunk the caller hands through RAW — Cassandra stored it uncompressed
67/// because its compressed length would have met `max_compressed_length`, or the
68/// SSTable has no compressor at all — into
69/// [`crate::observability::catalog::READ_BYTES`], exactly as a DECOMPRESSED chunk's
70/// bytes are counted (issue #1701, roborev F3).
71///
72/// # Why this lives in the plane rather than at each call site
73///
74/// SIX decode exits do that raw passthrough THEMSELVES (`len >=
75/// max_compressed_length`, or "no compressor, the buffer already holds finished
76/// bytes") and so never reach [`ChunkSource::decompress_only`] /
77/// [`ChunkSource::decode_and_cache`]: the windowed scan's IO half (two exits), the
78/// compressed offset-read window, the BIG promoted seek window, the stitch path, and
79/// the UNCOMPRESSED arm of the BIG promoted/reverse partition window
80/// (`read_uncompressed_verified_at`, found by roborev round 4 — the sixth exit this
81/// doc predicted, reached by every `ORDER BY ... DESC` read of an uncompressed
82/// SSTable, which is the only shape CQLite's own write surface emits).
83/// Every one was a silent hole in `read.bytes` — a read reporting rows and a duration
84/// while reporting fewer bytes than it actually read. Counting them independently at
85/// five sites would leave the NEXT such exit free to bypass the metric again, so the
86/// counting is ONE named plane function they all call: greppable from the plane, and
87/// the thing to reach for when a sixth raw exit appears.
88pub(crate) fn count_raw_chunk(bytes: &[u8], algorithm: Option<&CompressionAlgorithm>) {
89    read_metrics::record_decompressed_bytes(bytes.len(), compression_label_of(algorithm));
90}
91
92/// [`count_raw_chunk`] for a site that hands the buffer ON as its own value: counts,
93/// then returns it unchanged. Same single boundary — this exists only so a site whose
94/// raw exit is one expression (`Ok(whole)`) can route through it without restructuring.
95pub(crate) fn counted_raw_chunk(
96    bytes: Vec<u8>,
97    algorithm: Option<&CompressionAlgorithm>,
98) -> Vec<u8> {
99    count_raw_chunk(&bytes, algorithm);
100    bytes
101}

103/// Credit an UNCOMPRESSED block read to `cqlite.read.bytes` (issue #1701, roborev
104/// F1/F3), returning the read unchanged.
105///
106/// Gated on `compression_info.is_none()`: for an SSTable with NO
107/// `CompressionInfo.db` the block bytes ARE the finished `Data.db` payload and never
108/// reach a decompress step, so this is their only counting opportunity. A COMPRESSED
109/// read must NOT be counted here — those bytes are still compressed and the plane
110/// counts that read's payload post-decompression, so counting both would report one
111/// read twice under two different sizes.
112///
113/// # This is the ONLY counting of an uncompressed block — downstream branches must NOT
114///
115/// Every caller of `block_io::read_next_block` receives bytes that are ALREADY counted
116/// when the SSTable has no `CompressionInfo.db`, so a "no compressor, pass the buffer
117/// through" branch further down that call chain must leave them alone. `compression_info`
118/// and `compression_reader` are derived from the SAME parse (`reader/mod.rs`), so
119/// `compression_reader.is_none()` holds EXACTLY when this gate fired — which makes the
120/// naive `counted_raw_chunk(chunk, None)` in such a branch a straight DOUBLE count, at
121/// twice the true byte total. That is not a corner case: #1406 makes UNCOMPRESSED the
122/// only shape CQLite's own write surface emits, so it is the ordinary compaction input.
123/// Both compaction decode loops (`data_access/compaction.rs`) and the query stitch
124/// (`data_access/mod.rs`) therefore return that branch's buffer UNCOUNTED. The
125/// incompressible-raw branch beside them is the opposite case and DOES count: it is
126/// reached only when `max_compressed_length` is a real value, i.e. `compression_info` is
127/// `Some`, i.e. this gate did NOT fire.
128///
129/// Lives here rather than in `block_io` so every `read.bytes` increment in the crate
130/// stays inside this module (grep `record_decompressed_bytes`).
131pub(crate) fn count_uncompressed_block(
132    compression_info: &Option<std::sync::Arc<CompressionInfo>>,
133    read: Result<Option<Vec<u8>>>,
134) -> Result<Option<Vec<u8>>> {
135    if compression_info.is_none() {
136        if let Ok(Some(block)) = &read {
137            count_raw_chunk(block, None);
138        }
139    }
140    read
141}

143impl<'a> ChunkSource<'a> {
144    /// Construct a ChunkSource for positioned chunk reads.
145    #[allow(clippy::too_many_arguments)]
146    pub(crate) fn new(
147        source: &'a dyn ReadAt,
148        comp_info: &'a CompressionInfo,
149        compression: Option<&'a Compression>,
150        cache: &'a DecompressedChunkCache,
151        file_size: u64,
152        header_offset: u64,
153        namespace: u64,
154        cache_id: u64,
155    ) -> Self {
156        Self {
157            source,
158            comp_info,
159            compression,
160            cache,
161            file_size,
162            header_offset,
163            namespace,
164            cache_id,
165        }
166    }
167
168    /// The bounded [`catalog::attr::COMPRESSION`] label for this source's
169    /// decompressor (issue #1701). Derived from the `CompressionAlgorithm` ENUM,
170    /// never from the algorithm string parsed out of `CompressionInfo.db` — a
171    /// file-controlled string would be an unbounded metric dimension.
172    ///
173    /// No decompressor is `"none"`, NOT an absent label: an uncompressed SSTable is a
174    /// real, first-class read path (CQLite's own writer emits only uncompressed
175    /// SSTables — the #1406 claim boundary), so its bytes are attributed to a named
176    /// series rather than an anonymous one.
177    ///
178    /// [`catalog::attr::COMPRESSION`]: crate::observability::catalog::attr::COMPRESSION
179    fn compression_label(&self) -> &'static str {
180        compression_label_of(self.compression.map(|c| c.algorithm()))
181    }
182
183    /// Whole-chunk read: positioned read → CRC → decompress → B1 cache.
184    ///
185    /// Used by the BTI target-chunk path (self-reading) and can be called by the windowed
186    /// scan if it moves to positioned reads. Returns `Ok(None)` at EOF.
187    pub(crate) fn chunk(&self, index: usize) -> Result<Option<Bytes>> {
188        // Build cache key from absolute chunk index in this namespace
189        let key = ChunkKey::new(self.cache_id ^ self.namespace, index as u64);
190
191        // B1 cache hit: Arc clone, no read, no decompress
192        if let Some(hit) = self.cache.get(&key) {
193            return Ok(Some(hit));
194        }
195
196        // Cache miss: positioned read + CRC (C2 primitive)
197        let compressed = match read_compressed_chunk_at(
198            self.source,
199            self.comp_info,
200            index,
201            self.file_size,
202            self.header_offset,
203        )? {
204            Some(c) => c,
205            None => return Ok(None), // EOF
206        };
207
208        // Incompressible-raw passthrough or decompress, then cache
209        let incompressible = compressed.len() >= self.comp_info.max_compressed_length as usize;
210        Ok(Some(self.decode_and_cache(
211            key,
212            compressed,
213            incompressible,
214        )?))
215    }
216
217    /// Shared decompress+cache tail: decompress (or raw-passthrough) → insert → Arc.
218    ///
219    /// Called by `chunk()` after its own read+CRC, and directly by the windowed-scan
220    /// blocking half (which receives CRC-validated compressed bytes over a channel).
221    /// Together with `decompress_only`, this is the ONLY place on the query path where
222    /// `Compression::decompress` is called — the architecture test proves it.
223    pub(crate) fn decode_and_cache(
224        &self,
225        key: ChunkKey,
226        compressed: Vec<u8>,
227        incompressible: bool,
228    ) -> Result<Bytes> {
229        if incompressible {
230            // Stored uncompressed by Cassandra: pass the raw bytes through (no
231            // decompress counter), but COUNT them — they are Data.db payload this
232            // read materialised, exactly like a decompressed chunk's (issue #1701 F3).
233            count_raw_chunk(&compressed, self.compression.map(|c| c.algorithm()));
234            return Ok(self.cache.insert(key, compressed));
235        }
236        let decompressed = if let Some(compression) = self.compression {
237            // Decompress: the single query-path decompress call site.
238            //
239            // decompress PHASE (issue #1707): the timed region is the COMPRESSOR CALL
240            // ONLY — not the read that fetched the bytes (that is the io phase) and
241            // not the cache insert. Wrapping it HERE, inside the architecturally
242            // single decompress plane (`chunk_decode_single_plane.rs` proves there is
243            // only one), covers every entry point at once.
244            let d = crate::observability::read_phase::timed(ReadPhase::Decompress, || {
245                compression.decompress(&compressed)
246            })
247            .map_err(|e| {
248                Error::corruption(format!(
249                    "ChunkSource: failed to decompress chunk (key={:?}): {}",
250                    key, e
251                ))
252            })?;
253            DECOMPRESS_CALLS.fetch_add(1, Ordering::Relaxed);
254            // CHUNK_PATH_ALLOCS (consumer E3/#1940): the decompress OUTPUT buffer is
255            // a per-chunk copy-chain heap allocation. It is the ONE surviving
256            // allocation after the D2 substrate work — it flows zero-copy into the
257            // B1 cache as `Bytes` (no `Arc::from` re-copy) and is the refcounted
258            // substrate the window borrows. Recording it here (not the old
259            // compressed-read-buffer site, which D2 turned into a reused scratch)
260            // makes the ≤1-alloc/chunk bound measurable. No-op in release.
261            crate::storage::sstable::read_work_counters::record_chunk_path_alloc();
262            d
263        } else {
264            // No compression reader: treat raw bytes as decompressed
265            compressed
266        };
267
268        // cqlite.read.bytes (issue #1701): the DECOMPRESSED Data.db payload this
269        // chunk materialised, counted ONCE per chunk decode — the coarsest grain at
270        // which the decompressed size is known, and never per row. A chunk served
271        // from the B1 cache returns above without reaching here, which is correct:
272        // a cache hit read no Data.db bytes.
273        read_metrics::record_decompressed_bytes(decompressed.len(), self.compression_label());
274
275        // Insert into B1 cache (Vec→Arc conversion happens once here) and return
276        Ok(self.cache.insert(key, decompressed))
277    }

279    /// Decompress a CRC-validated COMPRESSIBLE chunk from a BORROWED slice, cache
280    /// it, and return the resident [`Bytes`] — leaving the compressed input buffer
281    /// owned by the caller so it can be RECYCLED as a per-loop scratch (issue #1940,
282    /// D2). This is the windowed-scan IO half's decode entry: unlike
283    /// [`decode_and_cache`](Self::decode_and_cache) (which takes the compressed `Vec`
284    /// by value and drops it), this borrows `&compressed`, so the caller keeps the
285    /// compressed buffer and reuses it for the next chunk read — the compressed-read
286    /// side then performs no per-chunk allocation, and the ONE surviving copy-chain
287    /// allocation is the decompress output (recorded here). Callers handle the
288    /// incompressible-raw and no-compressor cases directly (they move the buffer);
289    /// this method is only for the decompress path (`self.compression` is `Some`).
290    /// A `None` compressor is treated as a raw passthrough of a COPY (rare/never on
291    /// the windowed path, where compression is always resolved).
292    pub(crate) fn decode_borrowed(&self, key: ChunkKey, compressed: &[u8]) -> Result<Bytes> {
293        let decompressed = if let Some(compression) = self.compression {
294            // decompress PHASE (issue #1707): the windowed scan's decode entry — the
295            // compressor call ONLY, same region rule as the two siblings above. This
296            // is the arm a full `SELECT *` over a compressed SSTable takes.
297            let d = crate::observability::read_phase::timed(ReadPhase::Decompress, || {
298                compression.decompress(compressed)
299            })
300            .map_err(|e| {
301                Error::corruption(format!(
302                    "ChunkSource: failed to decompress chunk (key={:?}): {}",
303                    key, e
304                ))
305            })?;
306            DECOMPRESS_CALLS.fetch_add(1, Ordering::Relaxed);
307            // CHUNK_PATH_ALLOCS (consumer E3/#1940): the decompress OUTPUT buffer is
308            // the ONE surviving per-chunk copy-chain allocation after D2 — it flows
309            // zero-copy into the B1 cache as `Bytes` (no `Arc::from` re-copy) and is
310            // the refcounted substrate the window borrows. No-op in release.
311            crate::storage::sstable::read_work_counters::record_chunk_path_alloc();
312            d
313        } else {
314            compressed.to_vec()
315        };
316        // cqlite.read.bytes (issue #1701) — same per-chunk grain as
317        // `decode_and_cache`; this is the windowed scan's decode entry.
318        read_metrics::record_decompressed_bytes(decompressed.len(), self.compression_label());
319        Ok(self.cache.insert(key, decompressed))
320    }

322    /// Decompress-only helper for already-CRC-validated compressed buffers: decompress
323    /// without caching.
324    ///
325    /// Consumers (all uncached — no B1 cache insertion, no DECOMPRESS_CALLS counter;
326    /// their own counters, e.g. work_counters::add_chunk_decompressed, apply):
327    /// - the BIG reverse/seek window path (`decompress_partition_window` via
328    ///   `block_io::read_compressed_chunk_at`),
329    /// - the stitch path (`stitch_all_chunks`, `data_access/mod.rs`),
330    /// - the sequential-scan block decode (`parse_block_entries`, issue #2165).
331    ///
332    /// This exists to consolidate the `Compression::decompress` call site into this
333    /// module while changing zero runtime behavior on those paths.
334    pub(crate) fn decompress_only(
335        compression: Option<&Compression>,
336        compressed: Vec<u8>,
337    ) -> Result<Vec<u8>> {
338        if let Some(c) = compression {
339            let compressed_len = compressed.len();
340            // decompress PHASE (issue #1707): the uncached decompress sites, same
341            // compressor-call-only region as `decode_and_cache`.
342            let decompressed =
343                crate::observability::read_phase::timed(ReadPhase::Decompress, || {
344                    c.decompress(&compressed)
345                })
346                .map_err(|e| {
347                    Error::corruption(format!(
348                        "ChunkSource: decompress failed ({} compressed bytes): {}",
349                        compressed_len, e
350                    ))
351                })?;
352            // cqlite.read.bytes (issue #1701): the uncached decompress sites (the
353            // sequential-scan block decode, the stitch path, the BIG reverse/seek
354            // window) read Data.db bytes exactly like the cached ones, so they are
355            // counted at the same per-chunk grain.
356            read_metrics::record_decompressed_bytes(
357                decompressed.len(),
358                read_metrics::compression_attr(c.algorithm()),
359            );
360            Ok(decompressed)
361        } else {
362            // No compressor: these bytes ARE the `Data.db` payload, so they are read
363            // work and must be counted (issue #1701, roborev B2). Skipping this branch
364            // left every UNCACHED UNCOMPRESSED read — the shape CQLite's own writer
365            // produces (#1406) — invisible to `cqlite.read.bytes`.
366            read_metrics::record_decompressed_bytes(
367                compressed.len(),
368                read_metrics::COMPRESSION_NONE,
369            );
370            Ok(compressed)
371        }
372    }
