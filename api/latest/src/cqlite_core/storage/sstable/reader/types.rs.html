88/// Integrity status levels
89///
90/// Only two states are PRODUCED: the integrity check delegates to the authoritative
91/// verifier `verify::verify_sstable` (issue #1283), which reports either zero
92/// findings (`Healthy`) or one-or-more findings (`Corrupted`). The former `Degraded`
93/// state was driven by a `checksum_mismatches` counter that was never incremented —
94/// it was unreachable dead code. `Degraded` is retained as a deprecated variant for
95/// API compatibility but is NEVER returned by `perform_integrity_check`.
96#[derive(Debug, Clone, PartialEq, Eq)]
97pub enum IntegrityStatus {
98    /// File is healthy (the verifier reported no findings)
99    Healthy,
100    /// File has minor issues but is readable.
101    ///
102    /// Deprecated (issue #1283): the consolidated integrity check never produces
103    /// this status — it projects `verify::verify_sstable` findings onto exactly
104    /// `Healthy` (no findings) or `Corrupted` (≥1 finding). Retained for API
105    /// compatibility only.
106    #[deprecated(
107        since = "0.12.0",
108        note = "consolidated integrity check never produces Degraded; only Healthy|Corrupted (#1283)"
109    )]
110    Degraded,
111    /// File has corruption (the verifier reported at least one finding)
112    Corrupted,
113}

151/// Configuration for SSTable reader
152#[derive(Debug, Clone)]
153pub struct SSTableReaderConfig {
154    /// Size of the read buffer in bytes
155    pub read_buffer_size: usize,
156    /// Whether to memory-map SSTable files instead of using buffered file I/O.
157    ///
158    /// **Opt-in; defaults to `false`.** When enabled, files at or above
159    /// [`Self::mmap_min_size_bytes`] are mapped into the address space and
160    /// served from the OS page cache with no per-block read syscall. This
161    /// mirrors Cassandra's `disk_access_mode: mmap` and benefits repeated local
162    /// scans of the same files. Enable only for immutable local SSTables — see
163    /// [`crate::Config`]'s `storage.use_mmap` for the platform/filesystem
164    /// constraints (network FS and external mutation can `SIGBUS`).
165    pub use_mmap: bool,
166    /// Minimum file size (bytes) for memory mapping to kick in.
167    ///
168    /// Files smaller than this use buffered I/O even when [`Self::use_mmap`] is
169    /// set, since the per-file mapping overhead is not worthwhile for tiny
170    /// files and mapping a zero-length file is invalid.
171    pub mmap_min_size_bytes: usize,
172    /// Maximum number of blocks to cache
173    pub block_cache_size: usize,
174    /// Whether to validate checksums
175    pub validate_checksums: bool,
176    /// Whether to use bloom filters
177    pub use_bloom_filter: bool,
178    /// Prefetch size for sequential reads
179    pub prefetch_size: usize,
180}
181
182impl Default for SSTableReaderConfig {
183    fn default() -> Self {
184        Self {
185            read_buffer_size: 64 * 1024, // 64KB
186            use_mmap: false,             // Opt-in; buffered I/O is the portable, safe default
187            mmap_min_size_bytes: 4096,   // Skip mmap for files smaller than a page
188            block_cache_size: 1000,      // Cache 1000 blocks
189            validate_checksums: true,
190            use_bloom_filter: true,
191            prefetch_size: 128 * 1024, // 128KB
192        }
193    }
194}

224    ///
225    /// Interior-mutable (issue #2383 / #2356, "rebind-by-inode"): a warm reader
226    /// whose backing snapshot dir the Trino connector cleared can be REBOUND to a
227    /// fresh same-inode hardlink path WITHOUT re-opening/re-parsing the whole
228    /// Index.db — see [`Self::rebind_path`]. The already-open point/scan handles
229    /// stay valid (the inode outlives the cleared dir via the surviving
230    /// hardlinks); only [`Self::new_scan_cursor`]'s lazy `File::open`-by-path
231    /// needs the live path, so a lock-free `ArcSwap` swap is all a rebind costs.
232    pub(crate) file_path: arc_swap::ArcSwap<PathBuf>,
233    /// Backing byte source for point reads (buffered file I/O or memory map).
234    ///
235    /// Used only by positioned point-read helpers (`get_cached_data`,
236    /// integrity checks) that lock, seek, read, and unlock atomically. Full
237    /// scans no longer use this shared cursor — they mint their own
238    /// [`ScanCursor`](super::source::ScanCursor) via [`Self::scan_source`] so
239    /// they run in parallel (issue #815).
240    pub(crate) file: Arc<Mutex<BlockSource>>,
241    /// Template for minting fresh per-scan [`BlockSource`]s so concurrent scans
242    /// never share a mutable file position or chunk index (issue #815).
243    pub(crate) scan_source: ScanSource,
244    /// Positional (`pread`-style) source for the POINT-READ path (issue #1573,
245    /// Epic C / C2). Opened ONCE at reader open and shared for the reader's
246    /// lifetime. Because [`ReadAt`](super::read_at::ReadAt) takes `&self` (no seek,
247    /// no mutable position), concurrent point reads on one reader neither
248    /// serialize on a cursor mutex nor `open(2)` `Data.db` per lookup — the two
249    /// point-path pathologies C2 removes. Shares the reader's `Arc<Mmap>` when the
250    /// backend is mmap (no extra mapping / fd); otherwise it holds one dedicated
251    /// read-only fd. Scans still use [`scan_source`](Self::scan_source); this
252    /// source is intentionally shaped so scans could adopt it later (audit F3)
253    /// without requiring it now.

266    /// Reader configuration

268    /// The full [`Config`] this reader was opened with.
269    ///
270    /// Retained so `perform_integrity_check` can delegate to the authoritative
271    /// verifier `verify::verify_sstable` — the single source of truth for SSTable
272    /// integrity (issue #1283) — using the SAME configuration the reader itself

275    /// Platform abstraction
276    pub(crate) platform: Arc<Platform>,
277    /// Statistics
278    pub(crate) stats: SSTableReaderStats,
279    /// Tombstone merger for deletion handling
280    #[cfg(feature = "tombstones")]
281    pub(crate) tombstone_merger: TombstoneMerger,
282    /// SSTable generation number (for multi-generation merging)
283    pub generation: u64,
284    /// Actual header size calculated during parsing
285    pub(crate) actual_header_size: usize,
286    /// Index.db reader for partition lookup and promoted index handling
287    pub(crate) index_reader: Option<IndexReader>,
288    /// `true` iff a sibling `Index.db` file EXISTS on disk but failed to
289    /// open/parse (issue #2302, roborev job 1606), i.e. `index_reader` is `None`

291    /// full enumeration uses this to emit a LOUD warning (present-but-unusable

297    pub(crate) summary_reader: Option<SummaryReader>,
298    /// Cached Cassandra Murmur3 tokens of this SSTable's authoritative

301    /// for the reader's lifetime, so their tokens are computed ONCE at reader

304    /// `Some((first_token, last_token))` only when `Summary.db` was present with

334    ///
335    /// Polled at a bounded interval inside the compaction streaming read and its
336    /// sequential-scan fallback so a Flight `do_get` whose client disconnected
337    /// abandons the (otherwise uninterruptible, fully-materialising) walk
338    /// promptly instead of burning CPU until the coarse ~1–2 min backstop. The

341    pub(crate) scan_cancel: crate::storage::scan_cancel::ScanCancel,
342    /// CompressionInfo metadata for chunked decompression (if compressed)
343    pub compression_info: Option<Arc<CompressionInfo>>,
344    /// `CRC.db` per-chunk checksums for read-time integrity of **uncompressed**
345    /// BIG (`nb`) SSTables (issue #1396). `Some` when this is an uncompressed BIG
346    /// SSTable that ships a `CRC.db` (Cassandra writes one for every uncompressed
347    /// BIG table); `None` for compressed tables (they carry inline per-chunk CRCs
348    /// instead), BTI (`da`) tables (no `CRC.db`), or an uncompressed BIG table
349    /// whose `CRC.db` is absent (warn-and-proceed, owner-pinned decision D4).
350    /// When present, every uncompressed Data.db chunk returned by
351    /// `read_uncompressed_data_block` is verified against it, default-on.
352    pub(crate) crc_reader: Option<Arc<super::crc::CrcDb>>,
353    /// Memo of uncompressed Data.db chunk indices already CRC-verified on the
354    /// offset-read path (`read_value_at_offset`, used by the index-based scan and
355    /// point lookups). A partition read touches a sub-range of one or more chunks;
356    /// verifying a chunk requires the WHOLE chunk, so this set ensures each chunk
357    /// is read+checked at most once per reader lifetime — keeping the cost at the
358    /// budgeted "one CRC32 pass per chunk" (issue #1396) even when many small
359    /// partitions share a chunk. Only ever populated when [`Self::crc_reader`] is
360    /// `Some`. The contiguous scan path verifies inline in `block_io` and does not
361    /// use this memo.
362    pub(crate) verified_uncompressed_chunks: std::sync::Mutex<std::collections::HashSet<u64>>,

364    ///
365    /// Computed once in `SSTableReader::open` via `VersionGates::from_path` and

367    /// enhanced_statistics_parser, row_decoder row parsing) can read the
368    /// gate values without re-deriving them from the filename each time.
369    ///
370    /// Decision points that WILL be gated in VG3 are annotated with
371    /// `// VG3: use self.version_gates.has_XXX()` comments at each call site.
372    pub(crate) version_gates: Arc<VersionGates>,

377    /// `None` for BIG-format SSTables. The BTI point-lookup path
378    /// (`lookup_partition_via_bti_trie` / `bti_point_lookup`) walks these bytes in
379    /// place (a borrowed `&[u8]` view of this buffer — no per-lookup copy, issue
380    /// #1574 C3) to resolve the uncompressed Data.db offset for a partition key —
381    /// an O(log n) point lookup instead of the sequential scan used with no index.
382    pub(crate) bti_partitions_db: Option<Arc<Vec<u8>>>,

384    /// when this reader was opened on a BTI ("da") SSTable (issue #909, #910).
385    ///
386    /// `Some` for BTI SSTables (always emitted, possibly 0 bytes for a
387    /// narrow-only table), `None` for BIG-format SSTables. The BTI point-lookup
388    /// path uses this to resolve a WIDE partition: the `Partitions.db` trie
389    /// returns a positive `RowsOffset` pointing at the partition's
390    /// `TrieIndexEntry` inside `Rows.db`; [`resolve_rows_db_entry`] then recovers
391    /// the partition's uncompressed `Data.db` position (`data_position`), which is
392    /// the same offset domain a NARROW partition's direct `DataOffset` uses.
393    ///
394    /// [`resolve_rows_db_entry`]: crate::storage::sstable::bti::resolve_rows_db_entry

397    ///
398    /// Every reader opened for the same logical dataset via `SSTableManager`
399    /// shares ONE instance (cloned `Arc`), so a chunk decompressed by one read
400    /// path is reused by any other. Readers opened via the back-compat
401    /// [`SSTableReader::open`] get a fresh per-reader cache. Consulted by the
402    /// three wired read sites before reading+decompressing; a hit is an
403    /// `Arc::clone` (never a memcpy or a re-decompress).
404    pub(crate) chunk_cache: Arc<crate::storage::cache::DecompressedChunkCache>,

407    /// two different SSTables never collide in the shared cache. Authoritative
408    /// (derived from immutable reader identity), never from byte content.
409    ///
410    /// [`ChunkKey::sstable`]: crate::storage::cache::ChunkKey::sstable
411    pub(crate) chunk_cache_id: u64,

414    /// `Partitions.db` trie (issue #953 / #951).
415    ///
416    /// `None` until the first within-SSTable seek requests a successor offset;
417    /// computed once via [`SSTableReader::bti_partition_offsets`] (a full trie DFS
418    /// resolving WIDE-partition `RowsOffset`s through `Rows.db`) and cached so the
419    /// successor lookup is an O(log n) binary search per seek, not an O(n) DFS.
420    /// Only ever populated for BTI readers; BIG readers use the sorted `Index.db`
421    /// entries directly.
422    pub(crate) bti_partition_offsets: std::sync::OnceLock<Vec<u64>>,
423    /// Single-slot same-key memo of the most recent BTI partition resolution
424    /// (issue #1574, audit C3). A single-candidate `WHERE pk = ?` point read
425    /// descends the `Partitions.db` trie twice — once for the candidate prune
426    /// (`might_contain_partition`) and once for the seek — for the SAME key. The
427    /// resolved uncompressed `Data.db` offset (`Some(off)`) or authoritative
428    /// absence (`None`) is a pure function of the immutable trie + key, so the
429    /// prune stores it here and the seek reuses it without a second descent
430    /// (`TRIE_WALKS` stays 1 per point read). A stale slot (a different key, or a
431    /// concurrent read) simply misses and re-walks — never a wrong result. Best
432    /// effort: a poisoned lock is treated as a miss. This is NOT a cross-lookup
433    /// key/offset cache (that is Epic B/B4); it is bounded to one entry.
434    pub(crate) bti_lookup_memo: BtiLookupMemo,
435    /// Per-reader key→partition-offset cache (issue #1570, Epic B/B4) — the
436    /// Cassandra key-cache analogue. Maps the raw partition-key bytes to the
437    /// location the index/trie descent produces, so a repeated hot point read can
438    /// return the location without re-probing `Index.db` (BIG,
439    /// `lookup_partition_with_index`) or re-descending the `Partitions.db` trie
440    /// (BTI, `lookup_partition_via_bti_trie`, `TRIE_WALKS`). Per-reader (not shared):
441    /// a location is meaningful only within this SSTable's offset domain, so the
442    /// cache dies with the reader on remove/reload — the audit's immutable-SSTable
443    /// invalidation rule, trivially satisfied. Positive-only (absent keys are never
444    /// stored, so a hit can never be fabricated). Built honoring
445    /// `config.memory.block_cache.enabled` (the B2 read-cache toggle): a disabled
446    /// cache is a genuine no-op so the point-read path bypasses it.
447    pub(crate) key_offset_cache: Arc<crate::storage::cache::KeyOffsetCache>,
