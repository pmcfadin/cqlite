1//! Row DISPLAY + static-merge helpers for the `V5CompressedLegacy` row decoder
2//! (split out of `mod.rs` per the campsite rule, epic #1116).
3//!
4//! Three decisions live here, each shared by every decode site so the rule exists
5//! in exactly ONE place:
6//! * `row_has_non_key_cell` — is there any non-primary-key data cell (issue #932);
7//! * `merge_static_cells` — positional, clustering-row-wins static injection (#1642);
8//! * `build_display_row` — the row-tombstone display rule (#505/#932);
9//! * `extract_clustering_values` — clustering identity for range-tombstone coverage.
10

13/// Issue #932: does the decoded cell map hold any NON-primary-key data cell?
14///
15/// Primary-key (partition + clustering) columns are surfaced into the cell map
16/// as pseudo-cells (#229) so the read-back path can recover the clustering
17/// identity; they are NOT row data. A row carrying `HAS_DELETION` is a PURE row
18/// tombstone only when no such data cell survives — otherwise the row deletion
19/// COEXISTS with surviving (strictly-newer) cells and the row displays as live.
20pub(super) fn row_has_non_key_cell(cells: &[(Arc<str>, Value)], schema: &TableSchema) -> bool {
21    cells.iter().any(|(name, _)| {
22        let name: &str = name;
23        !schema.partition_keys.iter().any(|k| k.name == name)
24            && !schema.clustering_keys.iter().any(|c| c.name == name)
25    })
26}

28/// Issue #1642 (K3): append accumulated static-column cells onto a clustering
29/// row's positional cell vector. This is an unconditional `extend` (O(n_static))
30/// — NOT a per-cell membership scan — because a static column name can NEVER
31/// collide with a name already in the clustering row's cells, so there is no
32/// clustering-row-wins conflict to resolve.
33///
34/// Disjointness proof (this codebase). `static_cells` is the cell vector of an
35/// IS_STATIC row; its names are exactly `RowColumnResolution::columns_for(true)`
36/// — header columns with `is_static == true` AND `!is_primary_key` AND
37/// `!is_clustering`. A static row has no clustering prefix, so it receives zero
38/// clustering-key pseudo-cells (row_data.rs: `clustering_values` is empty when
39/// static). A clustering row's `cells` names are the clustering-key pseudo-cells
40/// (issue #229) PLUS `columns_for(false)` — header columns with `is_static ==
41/// false` AND `!is_primary_key` AND `!is_clustering`. Every column has exactly
42/// one `is_static` value, so the `is_static == want_static` filter makes
43/// `columns_for(true)` and `columns_for(false)` name-disjoint; and
44/// `columns_for(true)` excludes clustering-key columns, so no static cell shares
45/// a clustering-key pseudo-cell name. Hence the two name sets are disjoint and
46/// the former membership guard could never fire.
47///
48/// Appending AFTER the clustering row's own cells keeps the merged order
49/// deterministic-by-construction (never user-visible: the query result is a
50/// name-keyed map, issue #1334).
51pub(super) fn merge_static_cells(cells: &mut RowCells, static_cells: &RowCells) {
52    cells.extend(
53        static_cells
54            .iter()
55            .map(|(name, value)| (Arc::clone(name), value.clone())),
56    );
57}

59/// Issue #932/#1741: build the user-facing `ScanRow` display value for a parsed
60/// clustering row from its decoded `cells` and row header. Shared by every
61/// user-facing emit path so the row-tombstone display rule lives in ONE place:
62/// a `HAS_DELETION` row that carries NO surviving non-key cell displays as a pure
63/// `Tombstone` marker (suppressed downstream by `filter_tombstone`); a row that
64/// still carries surviving cells displays as a live `Row` (the deletion shadows
65/// only already-absent older cells). An empty cell set becomes a null marker.
66pub(super) fn build_display_row(
67    cells: RowCells,
68    row_header_opt: Option<&RowHeader>,
69    schema: &TableSchema,
70) -> ScanRow {
71    let row_tombstone = row_header_opt.filter(|h| h.is_row_tombstone());
72    let has_data_cell = row_has_non_key_cell(&cells, schema);
73    if row_tombstone.is_some() && !has_data_cell {
74        ScanRow::Marker(
75            row_tombstone
76                .map(|h| h.row_tombstone())
77                .unwrap_or(Value::Null),
78        )
79    } else if cells.is_empty() {
80        ScanRow::Marker(Value::Null)
81    } else {
82        // Issue #1642 (K3): the decoder already emits cells positionally, in
83        // serialization-header (schema) column order — determinism comes from
84        // CONSTRUCTION, not a per-row sort. The former per-row `HashMap`
85        // allocation and alphabetical `sort_by` are gone; the interned
86        // `Arc<str>` name handles (#1334) move straight into the carrier.
87        ScanRow::Row(cells)
88    }
89}

91impl RowHeader {
92    /// Whether the row's liveness marker survives THIS row's own deletion: Cassandra 5.0.8
93    /// `Row.Deletion.deletes(LivenessInfo)` -> `DeletionTime.deletes(long ts)` is
94    /// `ts <= markedForDeleteAt` (`db/DeletionTime.java`), and `BTreeRow.Merger`/`Builder` empty a
95    /// covered marker (`if (rowDeletion.deletes(livenessInfo)) livenessInfo = EMPTY`). So an
96    /// older OR equal-timestamp marker is deleted; only a strictly newer one survives. A header
97    /// without `HAS_DELETION` (or without a marker timestamp) is not covered.
98    fn liveness_survives_row_deletion(&self) -> bool {
99        match (self.timestamp, self.marked_for_delete_at) {
100            (Some(marker), Some(deleted_at)) if self.is_row_tombstone() => marker > deleted_at,
101            _ => true,
102        }
103    }
104
105    /// The liveness predicate of the user-facing read rule (issue #4524): a marker is present,
106    /// unexpired ([`Self::liveness_live_at`]) AND not covered by the row's own deletion.
107    pub(super) fn liveness_visible_at(&self, now_secs: i64) -> bool {
108        self.liveness_live_at(now_secs) && self.liveness_survives_row_deletion()
109    }
110
111    /// Whether the row's PRIMARY-KEY liveness marker is present AND still live at `now_secs`
112    /// (the single-generation twin of `RowLiveness::marker_live_at`): present iff `HAS_TIMESTAMP`;
113    /// live-forever when it carries no TTL expiry, otherwise live strictly before its expiry
114    /// (`Cell.isLive`: `now < localExpirationTime`). The ONE predicate behind both the row-hide
115    /// decision and the liveness-only display rule (issue #4524).
116    pub(super) fn liveness_live_at(&self, now_secs: i64) -> bool {
117        self.timestamp.is_some()
118            && self
119                .liveness_expires_at_seconds
120                .is_none_or(|s| s > now_secs)
121    }
122}

124/// Issue #1741: extract a clustering row's clustering-key values (in schema
125/// clustering order) from its decoded cell map. Only called when a range
126/// tombstone is currently OPEN in the partition, so the per-row clone is off the
127/// tombstone-free hot path. Returns fewer values than the clustering arity only
128/// for a malformed/partial row (missing clustering pseudo-cells).
129pub(super) fn extract_clustering_values(
130    cells: &[(Arc<str>, Value)],
131    schema: &TableSchema,
132) -> Vec<Value> {
133    schema
134        .clustering_keys
135        .iter()
136        .filter_map(|ck| {
137            cells
138                .iter()
139                .find(|(name, _)| name.as_ref() == ck.name.as_str())
140                .map(|(_, v)| v.clone())
141        })
142        .collect()
143}

145/// Whether a built display row is VISIBLE to a user-facing `SELECT` (issue #3095).
146///
147/// [`build_display_row`] returns `ScanRow::Marker` for a pure row tombstone / absent
148/// row, and EVERY user-facing consumer suppresses a marker downstream
149/// (`integrity::filter_tombstone`, and `build_row_from_scan`'s `into_cells`, issue
150/// #505) — so a marker is not a row a `SELECT` returns. This is the single predicate
151/// the static-content-on-an-empty-partition rule uses for Cassandra's
152/// `partition.hasNext()`, which is likewise evaluated over the already-FILTERED
153/// `RowIterator`: `UnfilteredRowIterators.filter(iter, nowInSec)` →
154/// `FilteredRows.filter` installs `db/transform/Filter`, whose
155/// `applyToRow(row) = row.purge(PURGE_ALL, nowInSec, enforceStrictLiveness)` drops a
156/// row that purges to `null` (cassandra-5.0.8).
157pub(super) fn row_is_visible(row: &ScanRow) -> bool {
158    matches!(row, ScanRow::Row(_) | ScanRow::RawRow(_))
159}

161/// Build the display row for a CLUSTERING row of a static-bearing partition, on a
162/// user-facing SELECT read (issue #3095).
163///
164/// The ORDER is load-bearing: the row-tombstone display decision is taken over the
165/// row's OWN cells, and static cells are injected only into a row that survives it.
166///
167/// Cassandra authority, VERBATIM from the pinned `cassandra-5.0.8` tag:
168///
169/// * `db/transform/Filter.java` applies the static row and the clustering rows as
170///   SEPARATE transformations — `protected Row applyToStatic(Row row)` vs
171///   `protected Row applyToRow(Row row) { return row.purge(DeletionPurger.PURGE_ALL,
172///   nowInSec, enforceStrictLiveness); }`. A clustering row is purged from ITSELF and
173///   dropped when `purge` returns `null`; the static row never participates.
174/// * `db/rows/BTreeRow.java`'s `purge` → `update(info, deletion, newTree)` returns
175///   `null` when `info.isEmpty() && deletion.isLive() && BTree.isEmpty(newTree)` —
176///   the row's OWN primary-key liveness, OWN deletion and OWN cell btree. A static
177///   cell can never be in a clustering row's btree (the static row is a separate
178///   `Row` reached out of band via `BaseRowIterator.staticRow()`).
179/// * `db/transform/FilteredRows.java` states the separation outright:
180///   `public boolean isEmpty() { return staticRow().isEmpty() && !hasNext(); }` —
181///   `hasNext()` counts clustering rows only, which is why
182///   `SelectStatement.processPartition()` can branch on `!partition.hasNext()` while
183///   still reading `partition.staticRow()`.
184///
185/// So a static cell can NEVER make a row-tombstoned clustering row live, and a
186/// partition whose clustering rows all purge away still returns its static content.
187/// Injecting first (which
188/// [`merge_static_cells`] + [`build_display_row`] do when called in that order)
189/// made `row_has_non_key_cell` true for a PURE row tombstone, so the row surfaced
190/// as a live `ScanRow::Row` carrying the static value — a phantom row on the
191/// single-generation arm that the k-way merge arm correctly suppresses
192/// (`entry_to_row` drops `RowData::Tombstone`). That divergence is what made the
193/// two arms disagree on a static partition holding only deleted rows.
194///
195/// PHYSICAL consumers (compaction, `verify`, delta-scan) keep the historical
196/// inject-then-decide order — their callers must see every on-disk unfiltered and
197/// their output is byte-pinned — so this is used only where `read_shadowing` is on.
198pub(super) fn build_display_row_read_path(
199    cells: RowCells,
200    static_cells: &RowCells,
201    row_header_opt: Option<&RowHeader>,
202    schema: &TableSchema,
203    now_secs: i64,
204) -> ScanRow {
205    // Issue #4524: the user-facing visibility rule is Cassandra's `BTreeRow.purge`: a row
206    // survives iff it has live PRIMARY-KEY LIVENESS (`HAS_TIMESTAMP` in the row header: an
207    // `INSERT`, including `INSERT (pk, ck)` with no regular cell and `INSERT ... v = null`) OR
208    // at least one live non-key cell. `build_display_row` knows only the cells, so on a row
209    // WITH a header it needs the liveness half of the rule applied here, read path only
210    // (physical consumers keep the byte-pinned historical shape):
211    //
212    //  * liveness, no live cell  -> an empty `Row` (v = null on every regular column), NOT a
213    //    `Marker(Null)`: the explicit-null / liveness-only row Cassandra returns;
214    //  * NO liveness, no live cell -> `Marker(Null)`: an `UPDATE`-created row whose every cell
215    //    was deleted; only its clustering pseudo-cells remain, which must not make it a row.
216    // TTL-aware (`liveness_live_at`): an EXPIRED marker is no liveness (roborev r1).
217    // Roborev job 307: the marker must also survive the row's OWN deletion (`ts > markedForDeleteAt`;
218    // equal timestamps: the deletion wins) or an older marker would resurrect a deleted row.
219    let has_liveness = row_header_opt.is_some_and(|h| h.liveness_visible_at(now_secs));
220    let live_cell = |cells: &RowCells| row_has_non_key_cell(cells, schema);
221    // A marker NEWER than the row deletion keeps the row: it displays as an empty live row even
222    // though `build_display_row` would call a deletion-without-data header a pure tombstone.
223    if has_liveness && row_header_opt.is_some_and(|h| h.is_row_tombstone()) && !live_cell(&cells) {
224        let mut kept = cells;
225        merge_static_cells(&mut kept, static_cells);
226        return ScanRow::Row(kept);
227    }
228    match build_display_row(cells, row_header_opt, schema) {
229        ScanRow::Row(kept) if row_header_opt.is_some() && !has_liveness && !live_cell(&kept) => {
230            ScanRow::Marker(Value::Null)
231        }
232        ScanRow::Row(mut kept) => {
233            merge_static_cells(&mut kept, static_cells);
234            ScanRow::Row(kept)
235        }
236        ScanRow::Marker(Value::Null) if has_liveness => {
237            let mut kept: RowCells = Vec::new();
238            merge_static_cells(&mut kept, static_cells);
239            ScanRow::Row(kept)
240        }
241        // A pure row tombstone (or an empty row without liveness): statics do not revive it.
242        other => other,
243    }
244}
