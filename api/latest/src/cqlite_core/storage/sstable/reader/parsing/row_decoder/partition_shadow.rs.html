1//! Read-side tombstone/TTL shadowing for a single partition (issue #1741).
2//!
3//! Single-generation / full-scan reads historically bypassed reconciliation and
4//! returned partition-deleted, range-tombstoned, or TTL-expired data as live. This
5//! helper restores Cassandra `SELECT` semantics on the READ path without routing
6//! through the write-support `ReconcileState`: it captures the partition-level
7//! deletion, tracks the currently-open range tombstone as the row stream is walked
8//! in clustering order, and decides per row whether a user-facing scan must hide it.
9//!
10//! It is **un-gated** — NOT behind the `write-support` feature — because read
11//! correctness must not depend on a write feature (AC2). It is shared by every
12//! user-facing emit path so there is ONE read-side shadowing implementation.
13//!
14//! Timestamp comparisons follow Cassandra `DeletionTime.deletes(ts) = ts <=
15//! markedForDeleteAt`: a deletion at `d` shadows the whole row iff every piece of
16//! the row's data is older than (or equal to) `d`. A row that still carries a cell
17//! strictly newer than the deletion survives (the deletion shadows only the
18//! already-absent older cells), matching the row-tombstone coexistence rule (#932).
19

22/// Authoritative per-clustering-column reversal flags for `schema`, in schema
23/// clustering-key order: `true` for a column declared `CLUSTERING ORDER BY
24/// (... DESC)`, `false` for ASC (issue #1741). Threaded into [`PartitionShadow`]
25/// so range-tombstone coverage compares clustering prefixes in physical storage
26/// order. Same `is_reversed` derivation the BTI reader uses
27/// (`data_access/bti.rs`). No-heuristics: schema clustering order only.
28pub(super) fn clustering_reversed_flags(schema: &crate::schema::TableSchema) -> Vec<bool> {
29    schema
30        .clustering_keys
31        .iter()
32        .map(|c| matches!(c.order, crate::schema::ClusteringOrder::Desc))
33        .collect()
34}

36/// Per-partition read-side shadowing state (issue #1741).
37///
38/// The type is `pub(crate)` so the multi-generation read path
39/// (`storage::sstable::generation_merge`) can reuse its per-cell decision
40/// [`PartitionShadow::cell_shadowed_or_expired`] POST-merge (issue #1849), keeping
41/// ONE read-visibility implementation. Its stateful methods (`open`,
42/// `feed_range_marker`, `row_hidden`, …) stay `pub(super)` — only the stateless
43/// per-cell decision is shared outward.
44pub(crate) struct PartitionShadow {
45    /// Partition-level `markedForDeleteAt` (µs), or `None` when the partition is live.
46    partition_deletion: Option<i64>,
47    /// The currently-open range tombstone: `(start_bound_values, start_inclusive,
48    /// deleted_at_µs)`. Cassandra writes non-overlapping range tombstones in
49    /// clustering order — a start marker opens a range, an end marker closes it, and
50    /// a boundary marker closes one and opens the next — so at most ONE range is open
51    /// at a time. Every row physically between the open start marker and its
52    /// (not-yet-seen) end marker falls inside the range, so a row is covered while
53    /// the range is open iff it is at/after the start bound.
54    open_range: Option<(Vec<Value>, bool, i64)>,
55    /// Current wall-clock, epoch seconds, for read-time TTL expiry.
56    now_secs: i64,
57    /// Per-clustering-column reversal flags in schema order: `true` for a column
58    /// declared `CLUSTERING ORDER BY (... DESC)`. Range-tombstone coverage must
59    /// compare a row's clustering prefix against the range bounds in PHYSICAL
60    /// storage order, which for a DESC column is the reverse of value order — so
61    /// this carries the authoritative schema clustering order into the coverage
62    /// FSM. A missing/short entry is treated as ASC (`false`), matching the
63    /// `is_reversed` idiom used by the sibling reader code (`data_access/bti.rs`).
64    /// No-heuristics: sourced only from `TableSchema::clustering_keys`.
65    clustering_reversed: Vec<bool>,
66}
67
68impl PartitionShadow {
69    /// Open shadowing for a partition given its decoded partition-level deletion
70    /// (`Some((markedForDeleteAt_µs, localDeletionTime_s))` when the partition
71    /// carries a tombstone, `None` when live), the current epoch-seconds clock, and
72    /// the per-clustering-column reversal flags (`true` = DESC) in schema order.
73    pub(super) fn open(
74        now_secs: i64,
75        partition_deletion: Option<(i64, i32)>,
76        clustering_reversed: Vec<bool>,
77    ) -> Self {
78        Self {
79            partition_deletion: partition_deletion.map(|(mfda, _ldt)| mfda),
80            open_range: None,
81            now_secs,
82            clustering_reversed,
83        }
84    }
85
86    /// Feed one decoded range-tombstone bound marker, running the same start/end/
87    /// boundary FSM the delta-scan path uses. `bound_kind` is the raw Cassandra
88    /// `ClusteringPrefix.Kind` ordinal; `deleted_at_secondary` is present only for
89    /// boundary markers (kind 2/5) and carries the new range's deletion time.
90    pub(super) fn feed_range_marker(
91        &mut self,
92        bound_values: Vec<Value>,
93        bound_kind: u8,
94        deleted_at_primary: i64,
95        deleted_at_secondary: Option<i64>,
96    ) -> Result<()> {
97        match bound_kind {
98            // Simple start bound: open a range (INCL_START=1, EXCL_START=7).
99            1 | 7 => {
100                self.open_range = Some((bound_values, bound_kind == 1, deleted_at_primary));
101            }
102            // Simple end bound: close the open range (EXCL_END=0, INCL_END=6).
103            0 | 6 => {
104                self.open_range = None;
105            }
106            // EXCL_END_INCL_START_BOUNDARY: close prev, open new range (inclusive start).
107            2 => {
108                let d = deleted_at_secondary.unwrap_or(deleted_at_primary);
109                self.open_range = Some((bound_values, true, d));
110            }
111            // INCL_END_EXCL_START_BOUNDARY: close prev, open new range (exclusive start).
112            5 => {
113                let d = deleted_at_secondary.unwrap_or(deleted_at_primary);
114                self.open_range = Some((bound_values, false, d));
115            }
116            unknown => {
117                return Err(Error::corruption(format!(
118                    "read-shadow: unknown range tombstone bound kind {unknown} — cannot represent \
119                     faithfully (no-heuristics mandate, issue #28)"
120                )));
121            }
122        }
123        Ok(())
124    }

126    /// `true` when the row's clustering values must be extracted for a coverage
127    /// check — i.e. a range tombstone is currently open. The tombstone-free common
128    /// case returns `false`, so the caller skips the per-row clustering clone.
129    pub(super) fn needs_clustering(&self) -> bool {
130        self.open_range.is_some()
131    }
132
133    /// `true` when a user-facing `SELECT` must HIDE this row: it is shadowed by the
134    /// partition tombstone or the open range tombstone (all of the row's data is
135    /// older than the deletion), or it is TTL-expired. `clustering` is the row's
136    /// clustering-key values (needed only for range coverage; pass an empty slice
137    /// when [`Self::needs_clustering`] is `false`).
138    pub(super) fn row_hidden(&self, header: &RowHeader, clustering: &[Value]) -> bool {
139        // Partition-level deletion shadows every row whose data predates it.
140        if let Some(d) = self.partition_deletion {
141            if header.shadowed_by_deletion_at(d) {
142                return true;
143            }
144        }
145        // Open range tombstone shadows covered rows whose data predates it.
146        if let Some((start, inclusive, d)) = &self.open_range {
147            if self.at_or_after_start(clustering, start, *inclusive)
148                && header.shadowed_by_deletion_at(*d)
149            {
150                return true;
151            }
152        }
153        // Read-time TTL expiry.
154        header.row_liveness_expired(self.now_secs)
155    }
156
157    /// Whether `row_ck` lies at/after the range's start bound in PHYSICAL storage
158    /// order. While a range is open its end has not yet been seen, so a covered row
159    /// need only satisfy the start bound. Comparison honours the per-column
160    /// clustering order: for a DESC column the physical order is the reverse of
161    /// value order, so we reverse that component's comparison — the same rule
162    /// [`crate::storage::write_engine::mutation::ClusteringKey::compare`] applies
163    /// (`ordering.reverse()` when the column is `ClusteringOrder::Desc`). Using the
164    /// threaded `clustering_reversed` flags keeps this schema-authoritative with no
165    /// per-row allocation (we compare the borrowed `Value` slices in place rather
166    /// than materialising `ClusteringKey`/`TableSchema` per row). A prefix start
167    /// bound (fewer components than the clustering arity) treats its missing
168    /// components as −∞, per Cassandra clustering-bound semantics.
169    fn at_or_after_start(&self, row_ck: &[Value], start: &[Value], inclusive: bool) -> bool {
170        // Unbounded (open) start covers everything.
171        if start.is_empty() {
172            return true;
173        }
174        for (i, (a, b)) in row_ck.iter().zip(start.iter()).enumerate() {
175            // Clustering-key values share a type per component, so `partial_cmp`
176            // yields a total order here; fall back to Equal defensively.
177            let mut ord = a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal);
178            // DESC column: physical order is the reverse of value order.
179            if self.clustering_reversed.get(i).copied().unwrap_or(false) {
180                ord = ord.reverse();
181            }
182            match ord {
183                std::cmp::Ordering::Less => return false,
184                std::cmp::Ordering::Greater => return true,
185                std::cmp::Ordering::Equal => {}
186            }
187        }
188        // All compared components equal.
189        match row_ck.len().cmp(&start.len()) {
190            // Row extends a shorter (prefix) start bound: its missing components are
191            // −∞, so the row is strictly after the start ⇒ covered.
192            std::cmp::Ordering::Greater => true,
193            // Row is a prefix of a longer start bound (fewer clustering components
194            // than the bound) — not a well-formed data row; treat as before start.
195            std::cmp::Ordering::Less => false,
196            // Exact boundary point: covered only when the start bound is inclusive.
197            std::cmp::Ordering::Equal => inclusive,
198        }
199    }

201    /// The effective covering deletion timestamp (µs) for a row with clustering prefix
202    /// `clustering`: the partition tombstone `markedForDeleteAt`, folded with the open
203    /// range tombstone's `deleted_at` when the row falls inside the range. `None` when
204    /// nothing covers the row. Shared by per-cell shadow filtering (issue #1741,
205    /// Finding 1) and the primary-key-liveness decision below.
206    pub(super) fn covering_deleted_at(&self, clustering: &[Value]) -> Option<i64> {
207        let mut cover = self.partition_deletion;
208        if let Some((start, inclusive, d)) = &self.open_range {
209            if self.at_or_after_start(clustering, start, *inclusive) {
210                cover = Some(cover.map_or(*d, |c| c.max(*d)));
211            }
212        }
213        cover
214    }

216    /// Per-cell shadow context for the row-data cell loop (issue #1741, Finding 1):
217    /// `(covering_deleted_at, now_secs)`. A data cell is dropped when its effective
218    /// write ts <= `covering_deleted_at`, or it is TTL-expired at `now_secs`.
219    pub(super) fn cell_context(&self, clustering: &[Value]) -> (Option<i64>, i64) {
220        (self.covering_deleted_at(clustering), self.now_secs)
221    }

223    /// Whether a data cell is dropped by read-side shadow/TTL filtering (issue #1741,
224    /// Finding 1): its effective write ts is shadowed by the covering deletion
225    /// (`eff_ts <= cover`, matching `DeletionTime.deletes`), OR it is TTL-expired at
226    /// `now` (`eff_exp <= now`). A cell with no authoritative write ts is NEVER
227    /// shadowed (no-heuristics: we hide only what authoritative metadata proves
228    /// stale). Associated function so both the row-data cell loop and its unit pins
229    /// share one decision.
230    pub(crate) fn cell_shadowed_or_expired(
231        cover: Option<i64>,
232        now: i64,
233        eff_ts: Option<i64>,
234        eff_exp: Option<i64>,
235    ) -> bool {
236        let shadowed = matches!((cover, eff_ts), (Some(d), Some(t)) if t <= d);
237        let expired = eff_exp.is_some_and(|e| e <= now);
238        shadowed || expired
239    }

241    /// Whether a decoded SIMPLE CELL must be DROPPED from a user-facing row
242    /// because it is a CELL TOMBSTONE — i.e. a deleted cell (issue #3094).
243    ///
244    /// A deleted cell is ABSENT from the row Cassandra reconciles: `Cell.isLive`
245    /// is false for a tombstone, so the `Row`'s `ColumnData` carries nothing for
246    /// that column and CQL renders it NULL. Dropping it here is what makes
247    /// `SELECT` return NULL instead of surfacing the raw
248    /// `Value::Tombstone(CellTombstone)` that `cell_value.rs` decodes. Before
249    /// this drop that tombstone reached the row carrier, and the Arrow encoder
250    /// correctly fail-closed (#1485) on it, erroring the WHOLE Flight `do_get`
251    /// stream (`column 'w': expected Text value, got Tombstone(..)`).
252    ///
253    /// `user_facing` is the SAME discriminator the #1741 per-cell shadow/TTL
254    /// filter uses: `true` exactly when a [`PartitionShadow`] was threaded into
255    /// the decode (a query read). The PHYSICAL consumers — compaction merge
256    /// input, sstabledump parity, delta scan — thread `None` and MUST keep
257    /// receiving the tombstone with its authoritative deletion timestamp (#505),
258    /// so their streams stay byte-unchanged.
259    ///
260    /// A tombstone is never folded into the row LIVENESS aggregate
261    /// (`has_live_forever_data_cell` / `max_data_cell_expires_at`): it is not live
262    /// data. Nor does it supply a TIMESTAMP to the row's shadow maximum — only the
263    /// PRESENCE fact `has_deleted_data_cell`; see [`Self::has_shadow_evidence`].
264    ///
265    /// This mirrors the multi-generation read path, whose merged-cell filter
266    /// (`generation_merge::ReadShadow::filter_live`) skips `Value::Tombstone` cells
267    /// and reports their PRESENCE onward exactly as this path does, and the
268    /// collection path, which already skipped tombstoned elements — so all three
269    /// read paths agree.
270    #[inline]
271    pub(crate) fn cell_tombstone_dropped(user_facing: bool, is_cell_tombstone: bool) -> bool {
272        user_facing && is_cell_tombstone
273    }
274
275    /// Whether a decoded simple-cell `value` is specifically a CELL TOMBSTONE —
276    /// the ONLY tombstone shape [`Self::cell_tombstone_dropped`] may drop.
277    ///
278    /// Matching the `Value::Tombstone` discriminant alone would silently accept a
279    /// row/range/partition tombstone value if one ever surfaced in the simple-cell
280    /// loop (none can today: `cell_value.rs` builds exactly
281    /// `TombstoneType::CellTombstone` there) and drop it as if it were a deleted
282    /// cell. Naming the variant keeps the decision authoritative rather than
283    /// inferred from a broader shape (no-heuristics, #28); an unexpected shape
284    /// instead flows on unchanged and fails closed downstream.
285    #[inline]
286    pub(crate) fn is_cell_tombstone(value: &Value) -> bool {
287        matches!(
288            value,
289            Value::Tombstone(t) if matches!(t.tombstone_type, TombstoneType::CellTombstone)
290        )
291    }

293    /// Whether the row carries enough authoritative evidence for
294    /// [`RowHeader::shadowed_by_deletion_at`] to hide it — i.e. whether its
295    /// `i64::MIN` "no authoritative timestamp" FAIL-SAFE is defeated (issue #3094).
296    /// `max_write_ts` is the row's LIVE-write maximum (`max(liveness marker, live data
297    /// cells)`, `i64::MIN` when neither exists); `has_deleted_data_cell` is the mere
298    /// PRESENCE of a decoded cell TOMBSTONE.
299    ///
300    /// ## The invariant
301    ///
302    /// Tombstone evidence may ONLY defeat the fail-safe; it must NEVER raise the row
303    /// maximum. That maximum is compared `max_ts <= markedForDeleteAt`, so raising it
304    /// can only move a row hidden → VISIBLE — the exact resurrection direction #3094
305    /// closes. A tombstone making a row visible is never correct: Cassandra purges a
306    /// surviving tombstone (`Filter.applyToRow` → `row.purge(…, PURGE_ALL, …)`) and
307    /// drops the emptied row (guide Ch.11), so it can only REMOVE data from a row.
308    ///
309    /// The counterexample that forced presence over an earlier "fallback max"
310    /// (`live.or(deleted_only)`): a LIVE LIVENESS MARKER at `T`, NO data cell, a cell
311    /// tombstone at `T + 10s`, under a partition deletion at `T + 5s`. With no
312    /// live-cell evidence the fallback fired, so `T + 10s` became the aggregate and
313    /// `max_write_timestamp` MAXed it with the liveness `T` → `T + 10s > T + 5s`: an
314    /// all-null phantom row out of a deleted partition where Cassandra returns 0 rows.
315    ///
316    /// ## Why presence suffices
317    ///
318    /// A row whose ONLY cells are tombstones (`UPDATE t SET w = null WHERE …`, no
319    /// INSERT liveness marker) has `max_write_ts == i64::MIN`, and the fail-safe exists
320    /// solely to avoid hiding an EMPTY/TRUNCATED parse. A decoded tombstone proves a
321    /// genuinely reduced row, so any covering deletion hides it (`i64::MIN <= cover`
322    /// always) — the direction Cassandra answers, since a row with no live data is
323    /// dropped whether or not the deletion predates the tombstone. Residual (#3121):
324    /// with NO covering deletion such a row is still emitted, all-null.
325    #[inline]
326    pub(crate) fn has_shadow_evidence(max_write_ts: i64, has_deleted_data_cell: bool) -> bool {
327        max_write_ts != i64::MIN || has_deleted_data_cell
328    }
329
330    /// Fold `value` into a running `Option` MAX accumulator: `None` contributes
331    /// nothing, so an absent authoritative timestamp never becomes a `0` (which
332    /// would read as "written at the epoch" — a heuristic, #28).
333    #[inline]
334    pub(crate) fn fold_max(acc: Option<i64>, value: Option<i64>) -> Option<i64> {
335        match (acc, value) {
336            (Some(a), Some(v)) => Some(a.max(v)),
337            (Some(a), None) => Some(a),
338            (None, v) => v,
339        }
340    }
341}

343/// Whether a partition-level deletion at `cover` (µs, `markedForDeleteAt`) shadows
344/// the WHOLE of a merged (post-`KWayMerger`) row (issues #1849, #3094).
345///
346/// This is the multi-generation read path's reuse of the single-gen row-level
347/// decision [`RowHeader::shadowed_by_deletion_at`]: it builds the same `RowHeader`
348/// the single-gen emit path folds and applies the identical `ts <=
349/// markedForDeleteAt` rule, from the three pieces of authoritative evidence the
350/// merged row carries:
351///
352/// - `marker_timestamp` — the merged row's primary-key liveness marker write ts
353///   (`MergeEntry::row_liveness.marker_timestamp`, #2374/#2789), `None` when the row
354///   has no marker. It is COMPARED against `cover` here, never trusted to be newer.
355///   The merger upstream DOES now judge the marker on this same field
356///   (`apply_partition_shadowing` → `RowLiveness::marker_survives_floor`, #3094),
357///   substituting `RowLiveness::default()` when its floor covers it — but that drop is
358///   not RELIED on here. The floor it applied is the merge's own MAX partition
359///   `markedForDeleteAt`, whereas `cover` is read independently from the merged
360///   partition-tombstone carrier, and this function is the read path's own
361///   authoritative application of `ts <= markedForDeleteAt`. So a marker at/below
362///   `cover` is handled as shadowed rather than assumed impossible.
363/// - `max_data_cell_timestamp` — max write ts across the merged LIVE data cells
364///   (never a tombstone's, #3094). `None` when the row surfaced no live data cell.
365/// - `has_deleted_data_cell` — mere PRESENCE of a merged cell TOMBSTONE, threaded
366///   from `ReadShadow::filter_live`'s tombstone skip. It only ever defeats the
367///   `i64::MIN` fail-safe (see [`PartitionShadow::has_shadow_evidence`]); it carries
368///   no timestamp, so it can never move a row hidden → visible.
369///
370/// A row with NO evidence at all (no marker, no live cell, no tombstone) is NEVER
371/// shadowed (fail-safe / no-heuristics: we hide only what authoritative metadata
372/// proves predates the deletion).
373///
374/// Read-time TTL expiry is applied PER CELL on the merged path (via
375/// [`PartitionShadow::cell_shadowed_or_expired`]); whole-row TTL hiding needs the
376/// marker's EXPIRY (`RowLiveness::expires_at_seconds`), which is not threaded here,
377/// so it is deliberately NOT decided here (issue #1849 scope note).
378///
379/// Gated on `write-support`: the sole caller is the cross-generation read path
380/// (`generation_merge`), whose `mod` declaration in `sstable/mod.rs` is itself
381/// `#[cfg(feature = "write-support")]`. Without this matching gate the function is
382/// orphaned (zero callers) under the `minimal-build` feature set and `-D dead_code`
383/// fails the build. `cell_shadowed_or_expired` needs no such gate — the single-gen
384/// read path (`row_data`/`complex_column`) calls it unconditionally.
385#[cfg(feature = "write-support")]
386pub(crate) fn merged_row_shadowed_by_partition(
387    cover: Option<i64>,
388    marker_timestamp: Option<i64>,
389    max_data_cell_timestamp: Option<i64>,
390    has_deleted_data_cell: bool,
391) -> bool {
392    let Some(deleted_at) = cover else {
393        return false;
394    };
395    let header = RowHeader {
396        // The SURVIVING liveness marker's write ts (#2374/#2789), not a hardcoded
397        // `None`: with #3094's presence rule in force, dropping it would hide a row
398        // whose marker is strictly NEWER than the deletion (Cassandra returns that
399        // row — the marker outlives the deletion and the tombstone is merely purged).
400        timestamp: marker_timestamp,
401        ttl: None,
402        liveness_expires_at_seconds: None,
403        local_deletion_time: None,
404        marked_for_delete_at: None,
405        header_size: 0,
406        row_size_vint_len: 0,
407        missing_columns_bitmap: None,
408        max_data_cell_timestamp,
409        max_data_cell_expires_at: None,
410        has_live_forever_data_cell: false,
411        // Threaded from the caller, NOT hardcoded (#3094 round 4): a cell tombstone
412        // strictly newer than `markedForDeleteAt` DOES survive the merge —
413        // `apply_partition_shadowing`'s `is_data` test is BY NAME — and
414        // `ReadShadow::filter_live` reports its PRESENCE from the same skip that drops
415        // it. Single- and multi-generation reads therefore agree on this shape.
416        has_deleted_data_cell,
417    };
418    header.shadowed_by_deletion_at(deleted_at)
419}
