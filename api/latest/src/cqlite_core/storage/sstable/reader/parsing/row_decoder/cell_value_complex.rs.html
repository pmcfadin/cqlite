1//! Complex value-decode ladder for the V5CompressedLegacy per-cell decoder.
2//!
3//! Campsite split of `cell_value.rs` (issue #1795): this file owns the
4//! frozen / tuple / non-frozen-collection / marshal-UDT / `varint` / unknown-scalar
5//! decode ladder ([`CellKind::Complex`]); the scalar arms live in
6//! `cell_value_scalar.rs` and the flag/conditional-field parsing in `cell_value.rs`.
7//!
8//! The blob/`varint` fall-throughs route their length prefix through
9//! [`super::V5CompressedLegacyParser::read_vint_length_prefixed_bytes`] for
10//! overflow-safe bounds checks (issue #1795).
11
12use super::marshal_element::MarshalCollectionElements;
13use super::*;
14
15impl V5CompressedLegacyParser {
16    /// Decode a live cell value for a [`CellKind::Complex`] column: frozen types,
17    /// tuples, non-frozen collections, marshal-form frozen UDTs, CQL `varint`, and
18    /// the unknown-scalar blob fall-through. `lowered` is the already-lowercased
19    /// declared type string. Advances `offset` past the consumed value bytes.
20    pub(super) fn decode_complex_cell_value(
21        &self,
22        data: &[u8],
23        offset: &mut usize,
24        lowered: &str,
25        column: &crate::schema::Column,
26        header_type: Option<&str>,
27        reader: &crate::storage::sstable::reader::types::SSTableReader,
28    ) -> Result<Value> {
29        let mut off = *offset;
30        let type_str: &str = lowered;
31        let value = if type_str.starts_with("frozen<") {
32            // Frozen types: unwrap inner type and route to appropriate parser
33            let inner_type = self.extract_frozen_inner_type(type_str)?;
34
35            tracing::debug!(
36                "V5CompressedLegacy: Parsing frozen type '{}' -> inner type '{}'",
37                type_str,
38                inner_type
39            );
40
41            // Issue #1340: extract the AUTHORITATIVE marshal element type(s)
42            // from the on-disk SerializationHeader marshal type (`header_type`).
43            // When an element is a `frozen<UDT>`, threading the marshal type lets
44            // it decode to a typed `Value::Frozen(Value::Udt)` registry-free
45            // (precedence: header marshal → registry → Blob, no byte-pattern
46            // inference — no-heuristics #28). Extracted once per frozen cell
47            // (before the element loop); the result borrows from `header_type`,
48            // so the per-element loop is allocation-free.
49            let marshal_elems = header_type.and_then(Self::extract_marshal_collection_elements);
50            // Shared for list & set (both are `Sequence`): the borrowed element
51            // marshal type, or `None` for a map / absent / mismatched marshal.
52            let sequence_marshal_elem = match &marshal_elems {
53                Some(MarshalCollectionElements::Sequence(m)) => Some(*m),
54                _ => None,
55            };
56
57            // Route to appropriate frozen collection parser
58            let (inner_value, new_offset) = if inner_type.starts_with("list<") {
59                let schema_elem = self.extract_collection_element_type(&inner_type, "list")?;
60                let element_type =
61                    Self::prefer_udt_marshal_element(sequence_marshal_elem, &schema_elem);
62                self.parse_frozen_list_value(data, off, element_type, column)?
63            } else if inner_type.starts_with("set<") {
64                let schema_elem = self.extract_collection_element_type(&inner_type, "set")?;
65                let element_type =
66                    Self::prefer_udt_marshal_element(sequence_marshal_elem, &schema_elem);
67                self.parse_frozen_set_value(data, off, element_type, column)?
68            } else if inner_type.starts_with("map<") {
69                let (schema_key, schema_val) = self.extract_map_types(&inner_type)?;
70                let (marshal_key, marshal_val) = match &marshal_elems {
71                    Some(MarshalCollectionElements::Map(k, v)) => (Some(*k), Some(*v)),
72                    _ => (None, None),
73                };
74                // Same shared rule as the MULTICELL map reader uses, so the two
75                // cannot form two opinions about a map key's decode type (issue
76                // #3612, roborev round 8 finding 1). A NO-OP here in both branches
77                // by construction: this side's marshal key never carries the outer
78                // `FrozenType` (Cassandra omits it inside a frozen collection) and
79                // its schema branch is untouched — wired anyway so the rule has ONE
80                // home.
81                let key_type = Self::map_key_type_for_decode(marshal_key, &schema_key);
82                let value_type = Self::prefer_udt_marshal_element(marshal_val, &schema_val);
83                self.parse_frozen_map_value(data, off, &key_type, value_type, column)?
84            } else if Self::is_udt_type(&column.data_type) {
85                // Frozen UDT - parse using UDT parser
86                // The column.data_type contains the full Cassandra type string including UserType
87                tracing::debug!(
88                    "V5CompressedLegacy: Parsing frozen UDT column '{}' type='{}'",
89                    column.name,
90                    column.data_type
91                );
92
93                // Parse UDT definition from the type string
94                let udt_def = Self::parse_udt_type_definition(&column.data_type)?;
95
96                // First read the VInt-prefixed blob length
97                let (remaining, blob_len_raw) = parse_vuint(&data[off..]).map_err(|e| {
98                    Error::corruption(format!(
99                        "Frozen UDT '{}': failed to parse blob length: {:?}",
100                        column.name, e
101                    ))
102                })?;
103                if blob_len_raw > MAX_CELL_VALUE_LENGTH {
104                    return Err(Error::corruption(format!(
105                        "Frozen UDT '{}': blob_len {} exceeds maximum {}",
106                        column.name, blob_len_raw, MAX_CELL_VALUE_LENGTH
107                    )));
108                }
109                let blob_len = blob_len_raw as usize;
110                let bytes_consumed = data[off..].len() - remaining.len();
111                off += bytes_consumed;
112
113                if blob_len > data.len().saturating_sub(off) {
114                    return Err(Error::corruption(format!(
115                        "Frozen UDT '{}': need {} bytes but only {} available",
116                        column.name,
117                        blob_len,
118                        data.len() - off
119                    )));
120                }
121
122                // Parse UDT value from the blob
123                let udt_data = &data[off..off + blob_len];
124                // The trailing `0` is the ROOT nesting depth, written here on purpose:
125                // this is a COLUMN-level decode, so it genuinely starts the count.
126                // `parse_udt_value` has no zero-depth overload precisely so that a
127                // caller INSIDE a decode cannot pick one by accident (issue #3631).
128                let (udt_value, n) = self.parse_udt_value(udt_data, 0, &udt_def, column, 0)?;
129                // #3811 (finding C): `parse_udt_value` REPORTS; this caller dropped it,
130                // so a frozen UDT blob with trailing bytes or a partial trailing field
131                // header was accepted where the collection and tuple paths refuse.
132                Self::require_fully_consumed(n, udt_data.len(), &column.name, "frozen UDT")?;
133                off += blob_len;
134
135                (udt_value, off)
136            } else if let Some(udt_def) = self
137                .udt_registry
138                .as_ref()
139                .and_then(|reg| reg.get_udt_qualified(&self.keyspace, &inner_type).cloned())
140            {
141                // frozen<short_udt_name>: look up the concrete UDT definition in the
142                // registry (Issue #502).  This handles type strings like
143                // `frozen<person>` where "person" is a registered UDT rather than a
144                // collection or a full marshal-format UserType string.
145                tracing::debug!(
146                    "V5CompressedLegacy: Resolving frozen UDT '{}' via registry for column '{}'",
147                    inner_type,
148                    column.name,
149                );
150
151                // Read VUInt-prefixed blob length (same framing as tuple and
152                // marshal-format UDT cells).
153                let (remaining, blob_len_raw) = parse_vuint(&data[off..]).map_err(|e| {
154                    Error::corruption(format!(
155                        "Frozen UDT '{}' (column '{}'): failed to parse blob length: {:?}",
156                        inner_type, column.name, e
157                    ))
158                })?;
159                if blob_len_raw > MAX_CELL_VALUE_LENGTH {
160                    return Err(Error::corruption(format!(
161                        "Frozen UDT '{}' (column '{}'): blob_len {} exceeds maximum {}",
162                        inner_type, column.name, blob_len_raw, MAX_CELL_VALUE_LENGTH
163                    )));
164                }
165                let blob_len = blob_len_raw as usize;
166                let len_bytes_consumed = data[off..].len() - remaining.len();
167                off += len_bytes_consumed;
168
169                if blob_len > data.len().saturating_sub(off) {
170                    return Err(Error::corruption(format!(
171                        "Frozen UDT '{}' (column '{}'): need {} bytes but only {} available",
172                        inner_type,
173                        column.name,
174                        blob_len,
175                        data.len() - off
176                    )));
177                }
178
179                let udt_data = &data[off..off + blob_len];
180                // The trailing `0` is the ROOT nesting depth, written here on purpose:
181                // this is a COLUMN-level decode, so it genuinely starts the count.
182                // `parse_udt_value` has no zero-depth overload precisely so that a
183                // caller INSIDE a decode cannot pick one by accident (issue #3631).
184                let (udt_value, n) = self.parse_udt_value(udt_data, 0, &udt_def, column, 0)?;
185                // #3811 (finding C): `parse_udt_value` REPORTS; this caller dropped it,
186                // so a frozen UDT blob with trailing bytes or a partial trailing field
187                // header was accepted where the collection and tuple paths refuse.
188                Self::require_fully_consumed(n, udt_data.len(), &column.name, "frozen UDT")?;
189                off += blob_len;
190
191                (udt_value, off)
192            } else if let Some(ht) =
193                header_type.filter(|ht| Self::marshal_is_top_level_frozen_udt(ht))
194            {
195                // Issue #1080: NO UdtRegistry is wired and the supplied schema
196                // short form `frozen<person_type>` carries no field defs, but
197                // the AUTHORITATIVE on-disk SerializationHeader marshal type for
198                // this column is the full
199                // `FrozenType(UserType(ks,hexname,field:Type,...))`. Decode the
200                // UDT STRUCTURALLY from that header type (no guessing, issue #28)
201                // rather than dropping the column (which also broke the Err→break
202                // loop, silently losing all trailing columns).
203                self.decode_frozen_udt_from_header_type(data, off, ht, column)?
204            } else {
205                // Detect bare identifiers that look like unregistered UDT names.
206                // A bare identifier has no '<' (not a container or tuple) and does not
207                // match any known CQL primitive type.  If we reach this branch with
208                // such an identifier it means the UDT was not in the registry — return
209                // an actionable schema error rather than silently producing a Blob.
210                //
211                // Legitimate fall-through types handled below:
212                //   • tuple<...>  (contains '<')
213                //   • known primitives: int, text, uuid, boolean, blob, float, double,
214                //     decimal, varint, bigint, counter, timestamp, date, time, duration,
215                //     inet, smallint, tinyint, varchar, ascii, timeuuid
216                const KNOWN_PRIMITIVES: &[&str] = &[
217                    "int",
218                    "bigint",
219                    "counter",
220                    "smallint",
221                    "tinyint",
222                    "text",
223                    "varchar",
224                    "ascii",
225                    "uuid",
226                    "timeuuid",
227                    "boolean",
228                    "blob",
229                    "float",
230                    "double",
231                    "decimal",
232                    "varint",
233                    "timestamp",
234                    "date",
235                    "time",
236                    "duration",
237                    "inet",
238                ];
239                let is_container = inner_type.contains('<');
240                let is_primitive = KNOWN_PRIMITIVES.contains(&inner_type.as_str());
241                if !is_container && !is_primitive {
242                    // Bare identifier that is neither a container nor a primitive —
243                    // this is an unregistered UDT name.
244                    return Err(Error::schema(format!(
245                        "frozen<{inner}>: UDT '{inner}' not found in registry for keyspace '{}'; \
246                         register it before reading",
247                        self.keyspace,
248                        inner = inner_type,
249                    )));
250                }
251                // Non-collection / primitive frozen type — recurse normally.
252                // The recursive call now returns 4 elements; we only need value + offset.
253                let mut inner_column = column.clone();
254                inner_column.data_type = inner_type.clone();
255                let (inner_val, _inner_ts, _inner_exp, inner_off) = self
256                    .parse_cell_value_schema_order(
257                        data,
258                        off,
259                        &inner_column,
260                        None,
261                        // Frozen-inner recursion: resolve the tag locally from
262                        // `inner_column.data_type` (bounded, off the per-cell scan
263                        // hot path).
264                        None,
265                        reader,
266                    )?;
267                (inner_val, inner_off)
268            };
269
270            off = new_offset;
271
272            // Wrap in Frozen
273            Value::Frozen(Box::new(inner_value))
274        } else if type_str.starts_with("tuple<") {
275            // Tuple types: parse fixed number of elements
276            self.parse_tuple_value(data, &mut off, type_str, column)?
277        }
278        // Non-frozen collections: list, set, map
279        // TODO(Issue #162, Task 3): Multi-cell collection parsing
280        //
281        // Collections in V5CompressedLegacy are stored as MULTIPLE CELLS with path identifiers,
282        // NOT as single blob values. The current single-cell parser cannot handle this.
283        //
284        // Format (from sstabledump analysis):
285        //   {"name": "scores", "deletion_info": {...}},  // Collection tombstone
286        //   {"name": "scores", "path": ["uuid1"], "value": 23},  // Element 1
287        //   {"name": "scores", "path": ["uuid2"], "value": 99},  // Element 2
288        //
289        // Required implementation:
290        //   1. Parse cell path (clustering key bytes) for each collection element
291        //   2. Detect collection tombstone cell (has deletion_info, no path/value)
292        //   3. Read N element cells (each with path + value)
293        //   4. Aggregate elements into Value::List/Set/Map based on column type
294        //   5. Handle different path encodings:
295        //      - list<T>: path is UUID bytes (timeuuid for ordering)
296        //      - set<T>: path is serialized element value (key), value is empty
297        //      - map<K,V>: path is serialized key, value is serialized value
298        //
299        // This is a fundamental architectural change requiring cell-level parsing
300        // before column-level aggregation. For now, return stub to unblock downstream work.
301        else if type_str.starts_with("list<")
302            || type_str.starts_with("set<")
303            || type_str.starts_with("map<")
304        {
305            warn!(
306                "V5CompressedLegacy: Non-frozen collection '{}' type '{}' requires multi-cell parsing (not yet implemented). \
307                 Collections are stored as multiple cells with path identifiers, requiring cell-level aggregation. \
308                 Returning empty collection as placeholder. See Issue #162 Task 3 for implementation plan.",
309                column.name, column.data_type
310            );
311
312            // Return empty collection based on type
313            if type_str.starts_with("list<") {
314                Value::List(Vec::new())
315            } else if type_str.starts_with("set<") {
316                Value::Set(Vec::new())
317            } else {
318                Value::Map(Vec::new())
319            }
320        }
321        // Issue #1080 / roborev job 1363: marshal-form frozen UDT. When the
322        // schema is DERIVED FROM the on-disk header (rather than supplied as a
323        // CQL short form), `column.data_type` is the authoritative marshal
324        // string `org.apache.cassandra.db.marshal.FrozenType(...UserType...)`,
325        // which does NOT start with CQL `frozen<` and so misses the arm above.
326        // Decode it structurally from that marshal type (same authoritative
327        // path as the supplied-schema header fallback) instead of blobbing it.
328        // `marshal_is_top_level_frozen_udt` accepts ONLY a top-level
329        // `FrozenType(UserType(...))` (NOT a frozen collection that contains a
330        // UDT, e.g. `FrozenType(ListType(UserType(...)))` — roborev 1365), and a
331        // non-frozen top-level UDT is routed to the complex branch by
332        // `is_complex_column`, so reaching here means a single-cell frozen UDT
333        // → wrap in `Value::Frozen` (consistent with the CQL `frozen<` arm).
334        else if Self::marshal_is_top_level_frozen_udt(&column.data_type) {
335            let (udt_value, new_offset) =
336                self.decode_frozen_udt_from_header_type(data, off, &column.data_type, column)?;
337            off = new_offset;
338            Value::Frozen(Box::new(udt_value))
339        }
340        // Cassandra 5.0 `vector<element, n>` (issue #4114). Sited BEFORE the blob
341        // fall-through, which is exactly what used to swallow it: a fixed-width
342        // vector value carries NO length prefix, so the fall-through read the first
343        // float's leading byte as a vint length and — depending on that ONE BYTE OF
344        // USER DATA — either errored blaming the data or returned a WRONG VALUE at
345        // exit 0 (`.drive-issue-4114/silent-misdecode-measurement.md`).
346        //
347        // `type_str` is the lowercased DECLARED type. Only the CQL spelling is
348        // matched here because every header marshal type reaching a column's
349        // `data_type` has already been normalized by the ONE conversion point,
350        // `enhanced_statistics_parser::marshal_type::convert_marshal_type_to_cql`
351        // (which now emits `vector<…, n>`); a marshal-spelled vector inside a UDT
352        // field instead travels the `CqlType` route (`type_string.rs` ->
353        // `typed_value.rs`).
354        else if type_str.starts_with("vector<") {
355            // Three-valued probe: a malformed vector type is an `Err` naming the type
356            // (never a fall-through to the blob arm below), and the `NotAVector`
357            // reading is unreachable under this `starts_with("vector<")` guard yet is
358            // still refused explicitly rather than defaulted (roborev job 109).
359            let args = crate::schema::vector_type::cql_vector_kind(type_str)
360                .into_args(type_str)?
361                .ok_or_else(|| {
362                    Error::schema(format!(
363                        "Cell '{}': type '{}' begins 'vector<' but is not a vector type",
364                        column.name, type_str
365                    ))
366                })?;
367            // AC4: the dimension comes from the type, and an element type CQLite
368            // does not implement is refused BY NAME rather than decoded as
369            // something else (#28).
370            let element = crate::schema::CqlType::parse(args.element)?;
371            crate::schema::vector_type::vector_value::require_float_element(
372                &element,
373                args.dimension,
374            )?;
375            crate::schema::vector_type::vector_value::decode_float_vector_at(
376                data,
377                &mut off,
378                &column.name,
379                args.dimension,
380            )?
381        }
382        // CQL `varint`: [VInt len][big-endian two's-complement bytes]. Decode to
383        // Value::Varint, matching the block path (issue #1885). Without this arm the
384        // value fell through to the blob default and was mis-typed as Value::Blob.
385        else if type_str == "varint" {
386            let bytes = Self::read_vint_length_prefixed_bytes(data, &mut off, column, "varint")?;
387            // Issue #1644: adapt the #1885 varint arm to the Bytes-backed enum
388            // via the `Value::varint` constructor (preserves #1885's owned-copy
389            // semantics; this arm was not part of #1644's zero-copy borrow scope).
390            Value::varint(bytes.to_vec())
391        }
392        // Default: treat as VInt-length-prefixed blob (unknown scalar type).
393        else {
394            let bytes = Self::read_vint_length_prefixed_bytes(data, &mut off, column, "blob")?;
395            // Issue #1644 (K5 stage 2): the complex blob default shares the scalar
396            // blob arm's zero-copy borrow (mirrors the pre-split shared closure).
397            Value::Blob(crate::storage::sstable::reader::value_borrow::borrow_active(bytes))
398        };
399
400        *offset = off;
401        Ok(value)
402    }
