1//! Issue #1741 / #1853: the read-time TTL "now" clock.
2//!
3//! Extracted from `mod.rs` (campsite rule / file-size ratchet, epic #1116) so
4//! the parser module doesn't grow past the source-file threshold.
5
6/// Issue #1853 roborev round 3: ALL of the override-parsing/fallback logic as
7/// a PURE function — no env access, no globals. This makes the override path
8/// unit-testable via plain argument passing instead of process-global
9/// `std::env::set_var`/`remove_var`, which is unsound to share with any other
10/// test in the same `cqlite-core` lib-test binary that constructs a
11/// `V5CompressedLegacyParser` in parallel (e.g. `block_entries.rs`,
12/// `frozen.rs`) — a pure function makes that race unrepresentable rather than
13/// merely guarded (roborev findings 1435-Low-1 and 1436-Medium both addressed
14/// structurally, not procedurally).
15///
16/// `raw_override`, when `Some` and parseable as an `i64` (epoch seconds), is
17/// honored verbatim (bypassing the wall clock entirely) — mirrors
18/// `now_epoch_secs()`'s prior "valid override always wins" behavior. `None`
19/// or an unparseable string falls back to `wall_clock_secs`.
20fn now_from(raw_override: Option<&str>, wall_clock_secs: i64) -> i64 {
21    raw_override
22        .and_then(|raw| raw.trim().parse::<i64>().ok())
23        .unwrap_or(wall_clock_secs)
24}
25
26/// Current wall clock as epoch seconds, saturating to `0` (nothing appears
27/// expired) if the clock is somehow before the Unix epoch.
28fn wall_clock_now_secs() -> i64 {
29    std::time::SystemTime::now()
30        .duration_since(std::time::UNIX_EPOCH)
31        .map(|d| d.as_secs() as i64)
32        .unwrap_or(0)
33}

35/// Issue #1853: test-only seam to pin the read-time TTL "now" clock
36/// deterministically. When `CQLITE_TTL_NOW_OVERRIDE_SECS` holds a valid `i64`
37/// (epoch seconds), it overrides the wall clock for read-time TTL expiry so
38/// parity tests can read a long-expired fixture "as of" its capture time. An
39/// unset or malformed value is IGNORED (falls back to the wall clock) and
40/// never panics — this is a library-code path.
41///
42/// Gated `#[cfg(debug_assertions)]` (roborev #1853 finding 1): this is a raw
43/// env-var read with no feature plumbing, so it must never be reachable from a
44/// real user's release build — a `--release` process that happens to inherit
45/// this variable from its environment must not have its TTL-expiry decisions
46/// silently overridden. `debug_assertions` is on for every `cargo test`/`cargo
47/// build` (debug) invocation, including the agent gate, so the seam and its
48/// coverage are always exercised in CI; it compiles out entirely (dead code,
49/// zero cost, zero env read) in `--release`. A Cargo feature was considered
50/// and rejected: a `required-features` integration test silently *skips* when
51/// the feature is off, which recreates the exact dead-coverage failure #1853
52/// exists to fix. Reusing the `experimental` feature was also rejected — it
53/// would muddy that flag's no-heuristics meaning.
54///
55/// Reading this env var is only ever done here, from the separate-process
56/// `issue_694_writetime_ttl_parity.rs` integration test binary (whose own
57/// `EnvVarGuard` mutation therefore races nothing in-process) — never from a
58/// `cqlite-core` unit test, which instead exercises the override/parse/
59/// fallback logic directly via the pure `now_from()` above (roborev round 3).
60#[cfg(debug_assertions)]
61const TTL_NOW_OVERRIDE_ENV: &str = "CQLITE_TTL_NOW_OVERRIDE_SECS";
62
63/// Issue #1741: current wall-clock as epoch seconds, for read-time TTL expiry.
64/// Falls back to `0` (nothing appears expired) if the clock is before the epoch.
65///
66/// Issue #1853: in debug builds, honors the `CQLITE_TTL_NOW_OVERRIDE_SECS` test
67/// seam first; an invalid/absent override leaves the wall-clock behavior
68/// unchanged. In release builds this is a straight `SystemTime::now()` call —
69/// the override seam (including the env read itself) is compiled out entirely.
70///
71/// Issue #2789: `pub` (but `#[doc(hidden)]`, an internal read-path seam) so the
72/// Flight producer (`cqlite-flight`) can thread the SAME authoritative
73/// reconciliation `now` into its k-way merger's TTL expiry as the core read
74/// path uses, rather than re-implementing the wall-clock/override logic and
75/// diverging on parity. Re-exported as
76/// [`crate::storage::write_engine::read_time_now_secs`].
77#[doc(hidden)]
78pub fn now_epoch_secs() -> i64 {
79    #[cfg(debug_assertions)]
80    let raw_override = std::env::var(TTL_NOW_OVERRIDE_ENV).ok();
81    #[cfg(not(debug_assertions))]
82    let raw_override: Option<String> = None;
83
84    now_from(raw_override.as_deref(), wall_clock_now_secs())
85}

87impl super::V5CompressedLegacyParser {
88    /// Issue #3058: pin this parser's read-time TTL "now" (epoch seconds) to a
89    /// clock the CALLER already sampled, instead of the ambient one
90    /// [`now_epoch_secs`] samples at construction.
91    ///
92    /// The Flight `do_get` producer captures ONE reconciliation `now` per request
93    /// (`read_time_now_secs`, this module's `now_epoch_secs`) and threads it into
94    /// its k-way merger via `with_now_secs`. Its single-source fast path must use
95    /// the SAME instant for `PartitionShadow`'s TTL expiry, or the two arms could
96    /// decide a cell straddling an expiration second differently and a PINNED
97    /// `now` (the query-semantics oracles) would not be honored on the fast arm.
98    /// So the fast path threads the request's `now` down to here rather than
99    /// letting the parser re-read the wall clock.
100    ///
101    /// A caller that has no request-scoped clock keeps the constructor default.
102    /// Only consulted when `read_shadowing` is `true`.
103    ///
104    /// Lives in `now_clock` (a CHILD of the module defining the parser, so the
105    /// private field is in scope) to keep `row_decoder/mod.rs` from growing past
106    /// the file-size ratchet (epic #1116).
107    pub fn with_now_secs(mut self, now_secs: i64) -> Self {
108        self.now_secs = now_secs;
109        self
110    }
111
112    /// The read-time TTL "now" (epoch seconds) this parser will evaluate expiry
113    /// against. Exposed so a test can PROVE a caller-pinned clock actually
114    /// reached the parser (issue #3058) instead of assuming it.
115    pub fn now_secs(&self) -> i64 {
116        self.now_secs
117    }
118
119    /// Whether SELECT-semantic read shadowing is enabled on this parser (issue
120    /// #1741). Exposed alongside [`Self::now_secs`] so the single-generation
121    /// query path can ASSERT its decode posture rather than assume it (#3058).
122    pub fn read_shadowing(&self) -> bool {
123        self.read_shadowing
124    }
125}

127#[cfg(test)]
128mod tests {
129    use super::*;
130
131    // Issue #1853 roborev round 3: these tests call the PURE `now_from()`
132    // directly with an explicit override argument — no env::set_var/remove_var,
133    // no guard type, no #[serial]. There is no process-global mutation here for
134    // any parallel test (in this binary or any other) to race.
135
136    /// Issue #3058: a caller-pinned `now` must REPLACE the ambient sample the
137    /// constructor took, and must not disturb the read-shadowing posture — the
138    /// two properties the single-generation query path depends on.
139    #[test]
140    fn caller_pinned_now_replaces_the_ambient_sample() {
141        let parser = crate::storage::sstable::reader::V5CompressedLegacyParser::new(
142            "ks".to_string(),
143            "tbl".to_string(),
144            0,
145            0,
146            None,
147        )
148        .with_read_shadowing(true);
149        let ambient = parser.now_secs();
150        let pinned = parser.with_now_secs(1_782_950_400);
151        assert_eq!(
152            pinned.now_secs(),
153            1_782_950_400,
154            "the caller's instant, not the ambient sample ({ambient})"
155        );
156        assert!(
157            pinned.read_shadowing(),
158            "pinning the clock must not disturb the read-shadowing posture"
159        );
160    }
161
162    #[test]
163    fn override_pins_now() {
164        let got = now_from(Some("1759716000"), wall_clock_now_secs());
165        assert_eq!(got, 1_759_716_000, "override must be honored verbatim");
166    }
167
168    #[test]
169    fn invalid_override_falls_back_to_wall_clock() {
170        let wall = wall_clock_now_secs();
171        let got = now_from(Some("not-a-number"), wall);
172        assert_eq!(
173            got, wall,
174            "an invalid override must fall back to the wall clock"
175        );
176    }
177
178    #[test]
179    fn absent_override_falls_back_to_wall_clock() {
180        let wall = wall_clock_now_secs();
181        let got = now_from(None, wall);
182        assert_eq!(
183            got, wall,
184            "no override set must fall back to the wall clock"
185        );
186    }
