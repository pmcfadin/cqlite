4    /// Build a [`CompactionRow`] from a parsed row's pieces (epic #899).
5    ///
6    /// `cells` is the collapsed column→value map (simple columns plus the
7    /// collapsed `Value` for each complex column); `cell_meta` carries per-simple
8    /// -cell write timestamps / TTLs; `complex` carries the per-element capture
9    /// for the complex columns. The complex columns are split out of `cells` (the
10    /// collapsed complex `Value` is dropped in favour of the per-element cells).
11    ///
12    /// A row tombstone produces [`CompactionRowData::Tombstone`]; an empty row
13    /// (no cells, no tombstone) produces an empty `Live`.
14    fn build_compaction_row_data(
15        &self,
16        cells: HashMap<Arc<str>, Value>,
17        cell_meta: Option<HashMap<String, CellWriteMetadata>>,
18        complex: CompactionComplexColumns,
19        row_header_opt: &Option<RowHeader>,
20        row_ts: i64,
21        schema: &TableSchema,
22    ) -> crate::storage::sstable::reader::compaction_row::CompactionRowData {
23        use crate::storage::sstable::reader::compaction_row::{
24            CompactionRowData, ComplexColumn, SimpleCell,
25        };
26
27        // Issue #932: a row with `HAS_DELETION` may ALSO carry surviving cells
28        // (cells written strictly after the row deletion). The row deletion is
29        // captured here either as the coexisting `row_deletion` on a `Live` row
30        // (when data cells survive) or as a pure `Tombstone` (when only the
31        // deletion remains). The decision is made AFTER building the cell sets so
32        // we can tell whether any NON-clustering data cell survived.
33        let row_deletion: Option<(i64, i32)> = row_header_opt
34            .as_ref()
35            .filter(|h| h.is_row_tombstone())
36            .map(|h| {
37                (
38                    h.row_tombstone_deletion_time(),
39                    // localDeletionTime in SECONDS (GC-grace clock). Preserve the
40                    // far-future [2^31, 2^32) encoding via wrapping `as u32 as i32`.
41                    h.local_deletion_time.unwrap_or(0),
42                )
43            });
44
45        // Build complex columns (sorted by name for deterministic output, mirroring
46        // the collapsed-value path's column ordering).
47        let mut complex_cols: Vec<ComplexColumn> = complex
48            .into_iter()
49            .map(
50                |(column, (complex_deletion, elements, collapsed_value))| ComplexColumn {
51                    column,
52                    complex_deletion,
53                    elements,
54                    collapsed_value,
55                },
56            )
57            .collect();
58        complex_cols.sort_by(|a, b| a.column.cmp(&b.column));
59
60        // Simple cells are every collapsed cell whose column is NOT a complex
61        // column. Per-cell timestamp / ttl / local-deletion-time come from
62        // `cell_meta` when present, else inherit the row timestamp.
63        let complex_names: std::collections::HashSet<&str> =
64            complex_cols.iter().map(|c| c.column.as_str()).collect();
65
66        let mut simple_cells: Vec<SimpleCell> = cells
67            .into_iter()
68            .filter(|(name, _)| !complex_names.contains(name.as_ref()))
69            .map(|(column, value)| {
70                let (timestamp, ttl, local_deletion_time) =
71                    match cell_meta.as_ref().and_then(|m| m.get(column.as_ref())) {
72                        Some(meta) => {
73                            let ttl = meta.expiration.as_ref().map(|e| e.ttl_seconds as u32);
74                            let ldt = meta
75                                .expiration
76                                .as_ref()
77                                .map(|e| e.expires_at_seconds as u32 as i32);
78                            (meta.write_timestamp_micros, ttl, ldt)
79                        }
80                        None => (row_ts, None, None),
81                    };
82                SimpleCell {
83                    column: column.to_string(),
84                    value,
85                    timestamp,
86                    ttl,
87                    local_deletion_time,
88                }
89            })
90            .collect();
91        simple_cells.sort_by(|a, b| a.column.cmp(&b.column));
92
93        // Issue #932: a row deletion either COEXISTS with surviving data cells
94        // (kept as `Live { row_deletion: Some(..) }`) or — when no NON-primary-key
95        // cell and no complex element survives — is a pure row tombstone (kept as
96        // `Tombstone`, preserving the #912 clustering-prefix capture). The earlier
97        // code always took the `Tombstone` branch, DROPPING surviving cells and
98        // letting older cells of other columns resurrect in a partial compaction.
99        if let Some((deletion_time, local_deletion_time)) = row_deletion {
100            let primary_key: std::collections::HashSet<&str> = schema
101                .partition_keys
102                .iter()
103                .map(|k| k.name.as_str())
104                .chain(schema.clustering_keys.iter().map(|c| c.name.as_str()))
105                .collect();
106            let has_simple_data = simple_cells
107                .iter()
108                .any(|c| !primary_key.contains(c.column.as_str()));
109            let has_complex_data = complex_cols
110                .iter()
111                .any(|c| !c.elements.is_empty() || c.complex_deletion.is_some());
112
113            if !has_simple_data && !has_complex_data {
114                // Pure row tombstone: rebuild the clustering prefix in schema
115                // order from the surfaced clustering pseudo-cells (#912). A
116                // missing clustering column falls back to the `None` bucket.
117                let mut clustering: Vec<(String, Value)> =
118                    Vec::with_capacity(schema.clustering_keys.len());
119                for ck in &schema.clustering_keys {
120                    match simple_cells.iter().find(|c| c.column == ck.name) {
121                        Some(c) => clustering.push((ck.name.clone(), c.value.clone())),
122                        None => {
123                            clustering.clear();
124                            break;
125                        }
126                    }
127                }
128                return CompactionRowData::Tombstone {
129                    deletion_time,
130                    local_deletion_time,
131                    clustering,
132                };
133            }
134        }
135
136        CompactionRowData::Live {
137            simple: simple_cells,
138            complex: complex_cols,
139            row_deletion,
140        }
141    }

143    /// Parse all partitions in a decompressed block into per-element
144    /// [`CompactionRow`]s (epic #899, compaction-only). Thin Vec wrapper over
145    /// [`Self::parse_block_for_compaction_emit`].
146    pub fn parse_block_for_compaction(
147        &self,
148        data: &[u8],
149        schema: Option<&TableSchema>,
150        reader: &crate::storage::sstable::reader::types::SSTableReader,
151    ) -> Result<Vec<crate::storage::sstable::reader::compaction_row::CompactionRow>> {
152        let mut results = Vec::new();
153        self.parse_block_for_compaction_emit(data, schema, reader, |row| {
154            results.push(row);
155            Ok(std::ops::ControlFlow::Continue(()))
156        })?;
157        Ok(results)
158    }

160    /// Streaming per-element compaction variant of
161    /// [`Self::parse_block_with_timestamps_emit`]: emits a [`CompactionRow`]
162    /// (per-element complex cells + real complex deletion) per row (epic #899).
163    pub fn parse_block_for_compaction_emit<F>(
164        &self,
165        data: &[u8],
166        schema: Option<&TableSchema>,
167        reader: &crate::storage::sstable::reader::types::SSTableReader,
168        mut emit: F,
169    ) -> Result<()>
170    where
171        F: FnMut(
172            crate::storage::sstable::reader::compaction_row::CompactionRow,
173        ) -> Result<std::ops::ControlFlow<()>>,
174    {
175        if data.is_empty() {
176            return Ok(());
177        }
178
179        let schema = schema.ok_or_else(|| {
180            Error::schema(format!(
181                "V5CompressedLegacy (compaction) format requires schema for {}.{}",
182                self.keyspace, self.table_name
183            ))
184        })?;
185
186        let mut offset = 0;
187        let mut skipped_partitions = 0;
188
189        let broke = std::cell::Cell::new(false);
190        let mut tracking_emit = |row| -> Result<std::ops::ControlFlow<()>> {
191            let flow = emit(row)?;
192            if matches!(flow, std::ops::ControlFlow::Break(())) {
193                broke.set(true);
194            }
195            Ok(flow)
196        };
197
198        while offset < data.len() {
199            match self.parse_one_partition_for_compaction(
200                &data[offset..],
201                Some(schema),
202                reader,
203                true,
204                &mut tracking_emit,
205            )? {
206                ParseStep::Emitted(consumed) => {
207                    if consumed == 0 {
208                        skipped_partitions += 1;
209                        offset += 1;
210                    } else {
211                        offset += consumed;
212                    }
213                }
214                ParseStep::NeedMore | ParseStep::Done => break,
215            }
216            if broke.get() {
217                break;
218            }
219        }
220
221        if skipped_partitions > 0 {
222            log::warn!(
223                "V5CompressedLegacy (compaction): skipped {} malformed partitions",
224                skipped_partitions
225            );
226        }
227
228        Ok(())
229    }

231    /// Like [`Self::parse_block_for_compaction_emit`] but also reports, for every
232    /// emitted [`CompactionRow`], the byte offset within `data` at which that
233    /// row's partition begins.
234    ///
235    /// `data` is the WHOLE decompressed Data.db data section (header stripped),
236    /// so the reported offset is the partition's absolute decompressed-Data.db
237    /// position — i.e. the value a BTI `Partitions.db` leaf encodes as
238    /// `BtiPartitionLocation::DataOffset`. The verifier uses this to resolve a
239    /// `DataOffset` payload back to its raw partition key by IDENTITY
240    /// (issue #1103), closing the same-count wrong-payload corruption gap.
241    pub fn parse_block_for_compaction_emit_with_offset<F>(
242        &self,
243        data: &[u8],
244        schema: Option<&TableSchema>,
245        reader: &crate::storage::sstable::reader::types::SSTableReader,
246        mut emit: F,
247    ) -> Result<()>
248    where
249        F: FnMut(
250            usize,
251            crate::storage::sstable::reader::compaction_row::CompactionRow,
252        ) -> Result<std::ops::ControlFlow<()>>,
253    {
254        if data.is_empty() {
255            return Ok(());
256        }
257
258        let schema = schema.ok_or_else(|| {
259            Error::schema(format!(
260                "V5CompressedLegacy (compaction) format requires schema for {}.{}",
261                self.keyspace, self.table_name
262            ))
263        })?;
264
265        let mut offset = 0;
266        let mut skipped_partitions = 0;
267
268        let broke = std::cell::Cell::new(false);
269
270        while offset < data.len() {
271            // Capture the partition-start offset BEFORE parsing this partition so
272            // every row emitted for it is tagged with the same authoritative
273            // decompressed-Data.db position.
274            let partition_start = offset;
275            let mut tagging_emit = |row| -> Result<std::ops::ControlFlow<()>> {
276                let flow = emit(partition_start, row)?;
277                if matches!(flow, std::ops::ControlFlow::Break(())) {
278                    broke.set(true);
279                }
280                Ok(flow)
281            };
282
283            match self.parse_one_partition_for_compaction(
284                &data[offset..],
285                Some(schema),
286                reader,
287                true,
288                &mut tagging_emit,
289            )? {
290                ParseStep::Emitted(consumed) => {
291                    if consumed == 0 {
292                        skipped_partitions += 1;
293                        offset += 1;
294                    } else {
295                        offset += consumed;
296                    }
297                }
298                ParseStep::NeedMore | ParseStep::Done => break,
299            }
300            if broke.get() {
301                break;
302            }
303        }
304
305        if skipped_partitions > 0 {
306            log::warn!(
307                "V5CompressedLegacy (compaction): skipped {} malformed partitions",
308                skipped_partitions
309            );
310        }
311
312        Ok(())
313    }

315    /// Per-element compaction counterpart of
316    /// [`Self::parse_one_partition_with_timestamps`] (epic #899). Identical
317    /// sliding-window / `ParseStep` / buffering semantics, but emits a
318    /// [`CompactionRow`] carrying per-element complex cells and the real complex
319    /// deletion instead of the collapsed `(RowKey, ScanRow, ts)` tuple.
320    pub fn parse_one_partition_for_compaction<F>(
321        &self,
322        data: &[u8],
323        schema: Option<&TableSchema>,
324        reader: &crate::storage::sstable::reader::types::SSTableReader,
325        at_final_chunk: bool,
326        emit: &mut F,
327    ) -> Result<ParseStep>
328    where
329        F: FnMut(
330            crate::storage::sstable::reader::compaction_row::CompactionRow,
331        ) -> Result<std::ops::ControlFlow<()>>,
332    {
333        use crate::storage::sstable::reader::compaction_row::CompactionRow;
334
335        if data.is_empty() {
336            return Ok(ParseStep::Done);
337        }
338
339        let schema = schema.ok_or_else(|| {
340            Error::schema(format!(
341                "V5CompressedLegacy (compaction) format requires schema for {}.{}",
342                self.keyspace, self.table_name
343            ))
344        })?;
345
346        // Issue #1046: per-PARTITION resolution build (this driver is re-entered once
347        // per partition by the sliding-window compaction caller). Allocations scale
348        // with partition count, not row count.
349        let resolution = RowColumnResolution::build(schema, reader);
350
351        // #1741 (roborev HIGH): size the header need-more decision correctly for
352        // the oa/da DeletionTime form. A DELETED oa/da partition carries the full
353        // 12-byte DeletionTime, not the 1-byte LIVE sentinel; a header split across
354        // a NON-FINAL chunk with only the first deletion byte present must return
355        // `NeedMore`, NOT be mis-parsed and skipped as `Emitted(1)` — on the
356        // compaction path that would DROP the partition tombstone and let older /
357        // deleted data survive or resurface. The nb path is unchanged (fixed
358        // 12-byte signed form). No heuristics: keyed off the authoritative
359        // `hasUIntDeletionTime` gate and the canonical DeletionTime sentinel.
360        match self.partition_header_readiness(data) {
361            PartitionHeaderReadiness::Malformed => return Ok(ParseStep::Emitted(1)),
362            PartitionHeaderReadiness::Incomplete => {
363                return Ok(if at_final_chunk {
364                    ParseStep::Done
365                } else {
366                    ParseStep::NeedMore
367                });
368            }
369            PartitionHeaderReadiness::Ready => {}
370        }
371
372        let (partition_key, mut offset, partition_deletion) = match self
373            .parse_partition_header_full(data, 0)
374        {
375            Ok(v) => v,
376            // Defense-in-depth: `Ready` guarantees the DeletionTime is fully
377            // present, so this cannot be a truncation. On a non-final chunk
378            // only re-request bytes if the header is still incomplete;
379            // otherwise skip a byte to resynchronise (NeedMore on a complete
380            // buffer would loop forever). Under `Ready` this stays a byte skip.
381            Err(_) => {
382                if !at_final_chunk
383                    && self.partition_header_readiness(data) == PartitionHeaderReadiness::Incomplete
384                {
385                    return Ok(ParseStep::NeedMore);
386                }
387                return Ok(ParseStep::Emitted(1));
388            }
389        };
390
391        // Issue #1074: on the COMPACTION path the static row is emitted as its own
392        // partition-level `CompactionRow` (Cassandra's `Row.staticRow`), NOT folded
393        // into the clustering rows. The user-facing read paths
394        // (`parse_block_emit_*` / `parse_one_partition_with_timestamps`) still fold
395        // statics into each row so a `SELECT` surfaces the static columns per row;
396        // compaction must preserve the partition/clustering-row separation instead.
397        let mut pending: Vec<CompactionRow> = Vec::new();
398
399        // Issue #1072: a partition-level tombstone in this SSTable must shadow OLDER
400        // live rows in OTHER SSTables during a cross-generation compaction merge.
401        // Surface it as a synthetic partition-deletion `CompactionRow` (mirroring the
402        // `RangeMarker` carrier) so the merge can apply the partition floor and
403        // re-emit the surviving tombstone. Pushed exactly once per partition — even
404        // when the partition has zero rows (tombstone-only case) — because this
405        // header parse runs once per partition emit.
406        if let Some((mfda, ldt)) = partition_deletion {
407            use crate::storage::sstable::reader::compaction_row::CompactionRowData;
408            pending.push(CompactionRow {
409                key: partition_key.clone(),
410                row_timestamp: mfda,
411                row_data: CompactionRowData::PartitionDelete {
412                    deletion_time: mfda,
413                    local_deletion_time: ldt,
414                },
415            });
416        }
417
418        // In-flight range tombstone start bound (issue #933). A range tombstone is
419        // a pair of adjacent bound markers (start then end), or a sequence of
420        // boundary markers that close one range and open the next. We buffer the
421        // open start here and emit a complete `CompactionRowData::RangeMarker` when
422        // the matching end/boundary arrives. Local to this call so it is re-derived
423        // from the partition start on every sliding-window re-parse (the `pending`
424        // rows are only flushed on a clean `Emitted`).
425        //
426        // Tuple: (start bound, deletion_time µs, local_deletion_time s).
427        let mut pending_range_start: Option<(
428            crate::storage::sstable::reader::compaction_row::CompactionBound,
429            i64,
430            i32,
431        )> = None;
432
433        // Build a `CompactionBound` from decoded clustering prefix values. The
434        // writer emits an OPEN bound (Bottom/Top) as an inclusive bound with zero
435        // clustering values, so an empty prefix decodes to Bottom (start) / Top
436        // (end). A non-empty prefix pairs each value with its schema clustering
437        // column name (positionally; bounds may be PREFIX-length).
438        let make_bound = |values: Vec<Value>, inclusive: bool, is_start: bool| {
439            use crate::storage::sstable::reader::compaction_row::CompactionBound;
440            if values.is_empty() {
441                return if is_start {
442                    CompactionBound::Bottom
443                } else {
444                    CompactionBound::Top
445                };
446            }
447            let pairs: Vec<(String, Value)> = values
448                .into_iter()
449                .enumerate()
450                .map(|(i, v)| {
451                    let name = schema
452                        .clustering_keys
453                        .get(i)
454                        .map(|c| c.name.clone())
455                        .unwrap_or_else(|| format!("ck{i}"));
456                    (name, v)
457                })
458                .collect();
459            if inclusive {
460                CompactionBound::Inclusive(pairs)
461            } else {
462                CompactionBound::Exclusive(pairs)
463            }
464        };
465
466        macro_rules! flush_and_emitted {
467            ($consumed:expr, $pending:expr, $emit:expr) => {{
468                for row in $pending.drain(..) {
469                    match $emit(row)? {
470                        std::ops::ControlFlow::Continue(()) => {}
471                        std::ops::ControlFlow::Break(()) => break,
472                    }
473                }
474                Ok(ParseStep::Emitted($consumed))
475            }};
476        }
477
478        loop {
479            if offset < data.len() && Self::is_end_of_partition(data[offset]) {
480                offset += 1;
481                return flush_and_emitted!(offset, pending, emit);
482            }
483
484            if offset >= data.len() {
485                if at_final_chunk {
486                    return flush_and_emitted!(offset, pending, emit);
487                }
488                return Ok(ParseStep::NeedMore);
489            }
490
491            if Self::is_range_tombstone_marker(data[offset]) {
492                // Issue #933: parse the range-tombstone bound marker and pair it
493                // into a complete `RangeMarker` so the merge can shadow covered
494                // cells AND re-emit the surviving marker (previously SKIPPED, which
495                // dropped the tombstone entirely).
496                let (bound_values, bound_kind, (mfda_p, ldt_p), secondary, next_offset) =
497                    match self.parse_range_tombstone_marker_with_ldt(data, offset, schema) {
498                        Ok(v) => v,
499                        Err(_) => {
500                            // Truncated marker body at a chunk boundary (or corrupt
501                            // at the final chunk): request more / flush, exactly as
502                            // the prior skip path did.
503                            if at_final_chunk {
504                                return flush_and_emitted!(offset, pending, emit);
505                            }
506                            return Ok(ParseStep::NeedMore);
507                        }
508                    };
509                offset = next_offset;
510
511                use crate::storage::sstable::reader::compaction_row::{
512                    CompactionBound, CompactionRowData,
513                };
514
515                // Emit a complete range tombstone given a (start, deletion) and an
516                // end bound + the end's deletion time/ldt.
517                let mut emit_range =
518                    |start: CompactionBound, end: CompactionBound, dt: i64, ldt: i32| {
519                        pending.push(CompactionRow {
520                            key: partition_key.clone(),
521                            row_timestamp: dt,
522                            row_data: CompactionRowData::RangeMarker {
523                                start,
524                                end,
525                                deletion_time: dt,
526                                local_deletion_time: ldt,
527                            },
528                        });
529                    };
530
531                // ClusteringPrefix.Kind ordinals:
532                //   0 EXCL_END, 1 INCL_START, 2 EXCL_END_INCL_START_BOUNDARY,
533                //   5 INCL_END_EXCL_START_BOUNDARY, 6 INCL_END, 7 EXCL_START.
534                match bound_kind {
535                    1 | 7 => {
536                        // Simple start bound: buffer until the matching end.
537                        let start = make_bound(bound_values, bound_kind == 1, true);
538                        pending_range_start = Some((start, mfda_p, ldt_p));
539                    }
540                    0 | 6 => {
541                        // Simple end bound: close the buffered start.
542                        let end = make_bound(bound_values, bound_kind == 6, false);
543                        let start = pending_range_start
544                            .take()
545                            .map(|(s, _, _)| s)
546                            .unwrap_or(CompactionBound::Bottom);
547                        emit_range(start, end, mfda_p, ldt_p);
548                    }
549                    2 | 5 => {
550                        // Boundary marker: primary closes the previous range, the
551                        // secondary opens the next. kind 2 = EXCL_END + INCL_START,
552                        // kind 5 = INCL_END + EXCL_START.
553                        let end_inclusive = bound_kind == 5;
554                        if let Some((start, _, _)) = pending_range_start.take() {
555                            let end = make_bound(bound_values.clone(), end_inclusive, false);
556                            emit_range(start, end, mfda_p, ldt_p);
557                        }
558                        let (new_mfda, new_ldt) = secondary.unwrap_or((mfda_p, ldt_p));
559                        let new_start = make_bound(bound_values, bound_kind == 2, true);
560                        pending_range_start = Some((new_start, new_mfda, new_ldt));
561                    }
562                    _ => {
563                        // Unknown bound kind: skip it rather than mis-parse the
564                        // partition (the offset already advanced past the marker).
565                    }
566                }
567                continue;
568            }
569
570            // Compaction mode: capture per-column complex elements and request
571            // per-cell metadata so simple cells carry per-cell timestamps/TTLs.
572            let mut complex_capture: CompactionComplexColumns = HashMap::new();
573            match self.parse_row_data_with_offset_impl(
574                data,
575                offset,
576                Some(schema),
577                reader,
578                true,
579                Some(&mut complex_capture),
580                &resolution,
581                // Compaction is a physical consumer: no read-side shadowing.
582                None,
583            ) {
584                Ok((
585                    cells,
586                    cell_meta,
587                    row_header_opt,
588                    next_offset,
589                    // Issue #1074: the on-disk static flag no longer changes the
590                    // emit path — static and clustering rows each become their own
591                    // `CompactionRow` (see below) and the writer decides static
592                    // placement from the schema, not from on-disk folding.
593                    _is_static,
594                    _complex_meta,
595                )) => {
596                    offset = next_offset;
597
598                    let row_tombstone = row_header_opt.as_ref().filter(|h| h.is_row_tombstone());
599                    // Issue #932: a HAS_DELETION row may ALSO carry a liveness
600                    // timestamp (surviving cells written strictly after the
601                    // deletion). Prefer the liveness timestamp as the row
602                    // timestamp so those cells inherit it and are NOT shadowed by
603                    // the older row deletion during reconcile; fall back to
604                    // markedForDeleteAt only for a PURE row tombstone (which has no
605                    // HAS_TIMESTAMP).
606                    let row_ts = row_header_opt
607                        .as_ref()
608                        .and_then(|h| h.timestamp)
609                        .or_else(|| row_tombstone.map(|h| h.row_tombstone_deletion_time()))
610                        .unwrap_or(0);
611
612                    // Issue #1074: emit BOTH static and clustering rows as their own
613                    // `CompactionRow`. A static row carries no clustering prefix, so
614                    // it decodes to the merge's `None` bucket and reconciles
615                    // independently of the clustering rows — a clustering-row delete
616                    // can no longer shadow the static cell, a static-only partition is
617                    // no longer dropped, and the static cell keeps its OWN write
618                    // timestamp (instead of inheriting a clustering row's). The merge
619                    // re-emits it as a `clustering_key: None` mutation and the writer
620                    // rebuilds the partition static prelude via
621                    // `collect_static_operations` (static-ness decided by the schema,
622                    // not by on-disk folding).
623                    let row_data = self.build_compaction_row_data(
624                        cells,
625                        cell_meta,
626                        complex_capture,
627                        &row_header_opt,
628                        row_ts,
629                        schema,
630                    );
631
632                    pending.push(CompactionRow {
633                        key: partition_key.clone(),
634                        row_timestamp: row_ts,
635                        row_data,
636                    });
637
638                    if offset >= data.len() {
639                        if at_final_chunk {
640                            return flush_and_emitted!(offset, pending, emit);
641                        }
642                        return Ok(ParseStep::NeedMore);
643                    }
644                    if self.peek_is_partition_header(data, offset) {
645                        return flush_and_emitted!(offset, pending, emit);
646                    }
647                }
648                Err(_) => {
649                    if at_final_chunk {
650                        return flush_and_emitted!(offset, pending, emit);
651                    }
652                    return Ok(ParseStep::NeedMore);
653                }
654            }
655        }
656    }
657}
