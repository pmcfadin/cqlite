16/// Upper bound on collection capacity pre-allocated from a declared element/entry
17/// count. A corrupt or adversarial huge count (e.g. `2^30`) must not let us
18/// pre-allocate gigabytes up front; we reserve at most this many slots and grow
19/// on demand as real elements are decoded (issue #1632). Guard-only: the decoded
20/// value is unchanged, only the initial allocation is bounded.
21const REASONABLE_COLLECTION_CAPACITY: usize = 4096;

93/// Parse list value with recursive element parsing via closure
94///
95/// This allows testing the list parsing logic independently of SSTableReader.
96pub(crate) fn parse_list_value_with<F>(
97    data: &[u8],
98    element_comparator: &ComparatorType,
99    parse_element: F,
100) -> Result<Value>
101where
102    F: Fn(&[u8], &ComparatorType) -> Result<Value>,
103{
104    use crate::parser::vint::parse_vint_length;
105
106    let mut offset = 0;
107
108    // Parse element count
109    let (remaining, element_count) = parse_vint_length(&data[offset..])
110        .map_err(|_| Error::corruption("Failed to parse list element count"))?;
111    offset = data.len() - remaining.len();
112
113    // Clamp pre-allocation: a corrupt huge count must not pre-allocate GBs (#1632).
114    let mut elements = Vec::with_capacity(element_count.min(REASONABLE_COLLECTION_CAPACITY));
115
116    // Decode EXACTLY `element_count` elements. A valid collection cell holds
117    // exactly `count` fully-encoded elements, so a buffer that runs dry before
118    // `count` elements are decoded is corrupt/truncated and must Err — silently
119    // returning a short partial list would accept a truncated cell (#1632).
120    for _ in 0..element_count {
121        if offset >= data.len() {
122            return Err(Error::corruption(
123                "List declared more elements than present in buffer (truncated)",
124            ));
125        }
126
127        // Parse element length
128        let (remaining, element_len) = parse_vint_length(&data[offset..])
129            .map_err(|_| Error::corruption("Failed to parse list element length"))?;
130        offset = data.len() - remaining.len();
131
132        if element_len > remaining.len() {
133            return Err(Error::corruption(
134                "List element length exceeds available data",
135            ));
136        }
137
138        // Parse element value using provided closure
139        let element_data = &remaining[..element_len];
140        let element_value = parse_element(element_data, element_comparator)?;
141        elements.push(element_value);
142        offset += element_len;
143    }
144
145    Ok(Value::List(elements))
146}

148/// Parse set value with recursive element parsing via closure
149pub(crate) fn parse_set_value_with<F>(
150    data: &[u8],
151    element_comparator: &ComparatorType,
152    parse_element: F,
153) -> Result<Value>
154where
155    F: Fn(&[u8], &ComparatorType) -> Result<Value>,
156{
157    // Sets are parsed similarly to lists
158    let list_value = parse_list_value_with(data, element_comparator, parse_element)?;
159    if let Value::List(elements) = list_value {
160        Ok(Value::Set(elements))
161    } else {
162        Err(Error::corruption("Failed to parse set value"))
163    }
164}

166/// Parse map value with recursive key/value parsing via closure
167pub(crate) fn parse_map_value_with<F>(
168    data: &[u8],
169    key_comparator: &ComparatorType,
170    value_comparator: &ComparatorType,
171    parse_element: F,
172) -> Result<Value>
173where
174    F: Fn(&[u8], &ComparatorType) -> Result<Value>,
175{
176    use crate::parser::vint::parse_vint_length;
177
178    let mut offset = 0;
179
180    // Parse entry count
181    let (remaining, entry_count) = parse_vint_length(&data[offset..])
182        .map_err(|_| Error::corruption("Failed to parse map entry count"))?;
183    offset = data.len() - remaining.len();
184
185    // Clamp pre-allocation: a corrupt huge count must not pre-allocate GBs (#1632).
186    let mut entries = Vec::with_capacity(entry_count.min(REASONABLE_COLLECTION_CAPACITY));
187
188    // Decode EXACTLY `entry_count` key/value pairs. A valid map cell holds
189    // exactly `count` fully-encoded entries, so a buffer that runs dry before
190    // `count` entries are decoded is corrupt/truncated and must Err — silently
191    // returning a short partial map would accept a truncated cell (#1632).
192    for _ in 0..entry_count {
193        if offset >= data.len() {
194            return Err(Error::corruption(
195                "Map declared more entries than present in buffer (truncated)",
196            ));
197        }
198
199        // Parse key length and data
200        let (remaining, key_len) = parse_vint_length(&data[offset..])
201            .map_err(|_| Error::corruption("Failed to parse map key length"))?;
202        offset = data.len() - remaining.len();
203
204        if key_len > remaining.len() {
205            return Err(Error::corruption("Map key length exceeds available data"));
206        }
207
208        let key_data = &remaining[..key_len];
209        let key_value = parse_element(key_data, key_comparator)?;
210        offset += key_len;
211
212        // Parse value length and data
213        let (remaining, value_len) = parse_vint_length(&data[offset..])
214            .map_err(|_| Error::corruption("Failed to parse map value length"))?;
215        offset = data.len() - remaining.len();
216
217        if value_len > remaining.len() {
218            return Err(Error::corruption("Map value length exceeds available data"));
219        }
220
221        let val_data = &remaining[..value_len];
222        let val_value = parse_element(val_data, value_comparator)?;
223        entries.push((key_value, val_value));
224        offset += value_len;
225    }
226
227    Ok(Value::Map(entries))
228}

230/// Parse tuple value with recursive field parsing via closure
231///
232/// Cassandra tuple field lengths are encoded as 4-byte big-endian signed int32
233/// (TupleType.java uses `accessor.putInt`), with -1 meaning null.
234/// This is NOT VInt encoding.
235pub(crate) fn parse_tuple_value_with<F>(
236    data: &[u8],
237    field_comparators: &[ComparatorType],
238    parse_element: F,
239) -> Result<Value>
240where
241    F: Fn(&[u8], &ComparatorType) -> Result<Value>,
242{
243    let mut offset = 0;
244    let mut fields = Vec::new();
245
246    // Parse each field
247    for (i, field_comparator) in field_comparators.iter().enumerate() {
248        if offset >= data.len() {
249            break;
250        }
251
252        // Parse field length as 4-byte big-endian signed int32 (Cassandra specification)
253        // -1 = null field, 0 = empty field, >0 = byte count
254        if offset + 4 > data.len() {
255            return Err(Error::corruption(format!(
256                "Tuple field {} length prefix truncated",
257                i
258            )));
259        }
260        let field_len_i32 =
261            i32::from_be_bytes(data[offset..offset + 4].try_into().map_err(|_| {
262                Error::corruption(format!("Tuple field {} length bytes invalid", i))
263            })?);
264        offset += 4;
265
266        if field_len_i32 == -1 {
267            // Null field
268            fields.push(Value::Null);
269            continue;
270        }
271        if field_len_i32 < 0 {
272            return Err(Error::corruption(format!(
273                "Tuple field {} has invalid negative length {}",
274                i, field_len_i32
275            )));
276        }
277        let field_len = field_len_i32 as usize;
278
279        if offset + field_len > data.len() {
280            return Err(Error::corruption(format!(
281                "Tuple field {} length {} exceeds available data",
282                i, field_len
283            )));
284        }
285
286        // Parse field value using provided closure
287        let field_data = &data[offset..offset + field_len];
288        let field_value = parse_element(field_data, field_comparator)?;
289        fields.push(field_value);
290        offset += field_len;
291    }
292
293    Ok(Value::Tuple(fields))
294}

296/// Parse UDT value with recursive field parsing via closure
297///
298/// Cassandra UDT field lengths are encoded as 4-byte big-endian signed int32
299/// (TupleType.java / UserType.java use `accessor.putInt`), with -1 meaning null.
300/// This is NOT VInt encoding.
301pub(crate) fn parse_udt_value_with<F>(
302    data: &[u8],
303    field_comparators: &[(String, ComparatorType)],
304    parse_element: F,
305) -> Result<UdtValue>
306where
307    F: Fn(&[u8], &ComparatorType) -> Result<Value>,
308{
309    let mut offset = 0;
310    let mut fields = Vec::new();
311
312    // Parse each field
313    for (field_name, field_comparator) in field_comparators.iter() {
314        if offset >= data.len() {
315            break;
316        }
317
318        // Parse field length as 4-byte big-endian signed int32 (Cassandra specification)
319        // -1 = null field, 0 = empty field, >0 = byte count
320        if offset + 4 > data.len() {
321            return Err(Error::corruption(format!(
322                "UDT field {} length prefix truncated",
323                field_name
324            )));
325        }
326        let field_len_i32 =
327            i32::from_be_bytes(data[offset..offset + 4].try_into().map_err(|_| {
328                Error::corruption(format!("UDT field {} length bytes invalid", field_name))
329            })?);
330        offset += 4;
331
332        if field_len_i32 == -1 {
333            // Null field
334            fields.push(UdtField {
335                name: field_name.clone(),
336                value: None,
337            });
338            continue;
339        }
340        if field_len_i32 < 0 {
341            return Err(Error::corruption(format!(
342                "UDT field {} has invalid negative length {}",
343                field_name, field_len_i32
344            )));
345        }
346        let field_len = field_len_i32 as usize;
347
348        if offset + field_len > data.len() {
349            return Err(Error::corruption(format!(
350                "UDT field {} length {} exceeds available data",
351                field_name, field_len
352            )));
353        }
354
355        // Parse field value using provided closure
356        let field_data = &data[offset..offset + field_len];
357        let field_value = parse_element(field_data, field_comparator)?;
358
359        fields.push(UdtField {
360            name: field_name.clone(),
361            value: Some(field_value),
362        });
363        offset += field_len;
364    }
365
366    Ok(UdtValue {
367        keyspace: "unknown".to_string(),
368        type_name: "unknown".to_string(),
369        fields,
370    })
371}

378    /// Parse column value using schema-driven approach (no heuristics)
379    pub(in crate::storage::sstable::reader) fn parse_column_value_enhanced(
380        &self,
381        value_data: &[u8],
382        table_id: &TableId,
383        key: &RowKey,
384        schema: Option<&crate::schema::TableSchema>,
385    ) -> Result<Value> {
386        if value_data.is_empty() {
387            return Ok(Value::Null);
388        }
389
390        // Use schema information to determine exact type - NO GUESSING
391        if let Some(schema) = self.get_table_schema(schema) {
392            // Extract column name from key context if possible
393            if let Some(column_name) = self.extract_column_name_from_context(table_id, key) {
394                // Find column in schema
395                if let Some(column) = schema.columns.iter().find(|c| c.name == column_name) {
396                    // Parse using exact type from schema
397                    return self.parse_value_with_schema_type(value_data, &column.data_type);
398                }
399            }
400        }
401
402        // Modern formats should never use blob fallback without schema
403        match self.header.cassandra_version {
404            crate::parser::header::CassandraVersion::V5_0NewBig
405            | crate::parser::header::CassandraVersion::V5_0Bti => Err(Error::Schema(format!(
406                "Blob fallback not allowed for value parsing in modern format {:?}. \
407                     Use SchemaAwareReader with complete schema information.",
408                self.header.cassandra_version
409            ))),
410            _ => {
411                // Legacy formats can use blob fallback as last resort
412                #[cfg(feature = "legacy-heuristics")]
413                {
414                    Ok(Value::Blob(value_data.to_vec()))
415                }
416                #[cfg(not(feature = "legacy-heuristics"))]
417                {
418                    Err(Error::Schema(
419                        "Blob fallback requires legacy-heuristics feature for legacy compatibility.".to_string()
420                    ))
421                }
422            }
423        }
424    }

426    /// Parse value using exact schema type information
427    pub(in crate::storage::sstable::reader) fn parse_value_with_schema_type(
428        &self,
429        value_data: &[u8],
430        data_type: &str,
431    ) -> Result<Value> {
432        // Convert data type string directly to ComparatorType for decoding
433        let comparator = ComparatorType::from_data_type(data_type)?;
434
435        // Structural types keep this reader's recursion + modern-format guards;
436        // every scalar (incl. schema-derived Custom) routes through the ONE scalar
437        // decode body in `comparator_value_parsing` (issue #1636 / J2), so a scalar
438        // type fix lands in exactly one place.
439        // Top-level column value enters at depth 0; nested elements/fields/frozen
440        // inner types accumulate depth and are capped by MAX_VALUE_NESTING_DEPTH
441        // (issue #1632) so a corrupt/adversarial deeply-nested type errors rather
442        // than overflowing the stack.
443        match &comparator {
444            ComparatorType::List(element_comparator) => {
445                self.parse_list_value(value_data, element_comparator, 0)
446            }
447            ComparatorType::Set(element_comparator) => {
448                self.parse_set_value(value_data, element_comparator, 0)
449            }
450            ComparatorType::Map(key_comparator, value_comparator) => {
451                self.parse_map_value(value_data, key_comparator, value_comparator, 0)
452            }
453            ComparatorType::Tuple(field_comparators) => {
454                self.parse_tuple_value(value_data, field_comparators, 0)
455            }
456            ComparatorType::Udt {
457                field_comparators, ..
458            } => self.parse_udt_value(value_data, field_comparators, 0),
459            ComparatorType::Frozen(inner_comparator) => {
460                // The outer `frozen<...>` layer costs one nesting level, so enter
461                // the inner comparator at depth 1 — symmetric with the block path
462                // (`parse_value_with_comparator_at_depth`'s Frozen arm recurses at
463                // depth+1). Entering at depth 0 would silently allow one extra
464                // nested level past MAX_VALUE_NESTING_DEPTH (#1632, guard-only).
465                let inner_value =
466                    self.parse_value_with_comparator_at_depth(value_data, inner_comparator, 1)?;
467                Ok(Value::Frozen(Box::new(inner_value)))
468            }
469            _ => decode_scalar_comparator(value_data, &comparator),
470        }
471    }

473    /// Parse value directly using ComparatorType (helper for nested collection elements).
474    ///
475    /// Provides complete recursive type parsing for collection elements, including
476    /// UDTs, tuples, nested collections, and frozen types. Entry callers pass
477    /// `depth = 0`; every recursive descent into a nested element/field/frozen inner type
478    /// increments `depth`; exceeding [`MAX_VALUE_NESTING_DEPTH`] returns `Err`
479    /// instead of recursing to a stack overflow (issue #1632). Guard-only:
480    /// successful decoding within the depth budget is byte-identical to before.
481    pub(in crate::storage::sstable::reader) fn parse_value_with_comparator_at_depth(
482        &self,
483        value_data: &[u8],
484        comparator: &ComparatorType,
485        depth: usize,
486    ) -> Result<Value> {
487        if depth > MAX_VALUE_NESTING_DEPTH {
488            return Err(Error::corruption(format!(
489                "Value decode recursion depth {} exceeds maximum {}",
490                depth, MAX_VALUE_NESTING_DEPTH
491            )));
492        }
493        match comparator {
494            ComparatorType::List(element_comparator) => {
495                self.parse_list_value(value_data, element_comparator, depth)
496            }
497            ComparatorType::Set(element_comparator) => {
498                self.parse_set_value(value_data, element_comparator, depth)
499            }
500            ComparatorType::Map(key_comparator, value_comparator) => {
501                self.parse_map_value(value_data, key_comparator, value_comparator, depth)
502            }
503            ComparatorType::Tuple(field_comparators) => {
504                self.parse_tuple_value(value_data, field_comparators, depth)
505            }
506            ComparatorType::Udt {
507                keyspace,
508                type_name,
509                field_comparators,
510            } => {
511                // Parse UDT fields inline with full type info (Issue #238 fix)
512                // This avoids the V5 format check in parse_udt_value() which incorrectly
513                // returns an error even when we have complete schema information.
514                // Field lengths are 4-byte big-endian signed int32 per Cassandra specification.
515                let mut offset = 0;
516                let mut fields = Vec::new();
517
518                for (field_name, field_comparator) in field_comparators.iter() {
519                    if offset >= value_data.len() {
520                        break;
521                    }
522                    // Parse field length as 4-byte big-endian signed int32 (Cassandra spec)
523                    if offset + 4 > value_data.len() {
524                        return Err(Error::corruption(format!(
525                            "UDT field {} length prefix truncated",
526                            field_name
527                        )));
528                    }
529                    let field_len_i32 = i32::from_be_bytes(
530                        value_data[offset..offset + 4].try_into().map_err(|_| {
531                            Error::corruption(format!("UDT field {} length invalid", field_name))
532                        })?,
533                    );
534                    offset += 4;
535
536                    if field_len_i32 == -1 {
537                        fields.push(UdtField {
538                            name: field_name.clone(),
539                            value: None,
540                        });
541                        continue;
542                    }
543                    if field_len_i32 < 0 {
544                        return Err(Error::corruption(format!(
545                            "UDT field {} has invalid negative length {}",
546                            field_name, field_len_i32
547                        )));
548                    }
549                    let field_len = field_len_i32 as usize;
550
551                    if offset + field_len > value_data.len() {
552                        return Err(Error::corruption(format!(
553                            "UDT field {} length {} exceeds available data",
554                            field_name, field_len
555                        )));
556                    }
557
558                    let field_data = &value_data[offset..offset + field_len];
559                    let field_value = self.parse_value_with_comparator_at_depth(
560                        field_data,
561                        field_comparator,
562                        depth + 1,
563                    )?;
564
565                    fields.push(UdtField {
566                        name: field_name.clone(),
567                        value: Some(field_value),
568                    });
569                    offset += field_len;
570                }
571
572                Ok(Value::Udt(UdtValue {
573                    keyspace: keyspace.clone().unwrap_or_else(|| "unknown".to_string()),
574                    type_name: type_name.clone(),
575                    fields,
576                }))
577            }
578            ComparatorType::Frozen(inner_comparator) => {
579                let inner_value = self.parse_value_with_comparator_at_depth(
580                    value_data,
581                    inner_comparator,
582                    depth + 1,
583                )?;
584                Ok(Value::Frozen(Box::new(inner_value)))
585            }
586            // Every scalar (incl. schema-derived Custom) routes through the ONE
587            // scalar decode body in `comparator_value_parsing` (issue #1636 / J2).
588            _ => decode_scalar_comparator(value_data, comparator),
589        }
590    }

644    /// Parse UDT value using field comparators
645    ///
646    /// Cassandra UDT field lengths are 4-byte big-endian signed int32 (not VInt).
647    pub(in crate::storage::sstable::reader) fn parse_udt_value(
648        &self,
649        value_data: &[u8],
650        field_comparators: &[(String, ComparatorType)],
651        depth: usize,
652    ) -> Result<Value> {
653        let mut offset = 0;
654        let mut fields = Vec::new();
655
656        // Parse each field
657        for (field_name, field_comparator) in field_comparators.iter() {
658            if offset >= value_data.len() {
659                break;
660            }
661
662            // Parse field length as 4-byte big-endian signed int32 (Cassandra specification)
663            if offset + 4 > value_data.len() {
664                return Err(Error::corruption(format!(
665                    "UDT field {} length prefix truncated",
666                    field_name
667                )));
668            }
669            let field_len_i32 =
670                i32::from_be_bytes(value_data[offset..offset + 4].try_into().map_err(|_| {
671                    Error::corruption(format!("UDT field {} length invalid", field_name))
672                })?);
673            offset += 4;
674
675            if field_len_i32 == -1 {
676                fields.push(UdtField {
677                    name: field_name.clone(),
678                    value: None,
679                });
680                continue;
681            }
682            if field_len_i32 < 0 {
683                return Err(Error::corruption(format!(
684                    "UDT field {} has invalid negative length {}",
685                    field_name, field_len_i32
686                )));
687            }
688            let field_len = field_len_i32 as usize;
689
690            if offset + field_len > value_data.len() {
691                return Err(Error::corruption(format!(
692                    "UDT field {} length {} exceeds available data",
693                    field_name, field_len
694                )));
695            }
696
697            // Parse field value using field comparator
698            let field_data = &value_data[offset..offset + field_len];
699            let field_value =
700                self.parse_value_with_comparator_at_depth(field_data, field_comparator, depth + 1)?;
701            offset += field_len;
702
703            fields.push(UdtField {
704                name: field_name.clone(),
705                value: Some(field_value),
706            });
707        }
708
709        // Modern formats should never use generic UDT fabrication without schema
710        match self.header.cassandra_version {
711            crate::parser::header::CassandraVersion::V5_0NewBig
712            | crate::parser::header::CassandraVersion::V5_0Bti => Err(Error::Schema(format!(
713                "Generic UDT fabrication not allowed for modern format {:?}. \
714                     Use SchemaAwareReader with complete UDT schema information.",
715                self.header.cassandra_version
716            ))),
717            _ => {
718                // Legacy formats can use generic UDT fabrication as last resort
719                #[cfg(feature = "legacy-heuristics")]
720                {
721                    Ok(Value::Udt(UdtValue {
722                        keyspace: "unknown".to_string(),
723                        type_name: "unknown".to_string(),
724                        fields,
725                    }))
726                }
727                #[cfg(not(feature = "legacy-heuristics"))]
728                {
729                    Err(Error::Schema(
730                        "Generic UDT fabrication requires legacy-heuristics feature for legacy compatibility.".to_string()
731                    ))
732                }
733            }
734        }
735    }
