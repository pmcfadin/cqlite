1//! **The ONE package rule** for a Cassandra marshal class name (issue #3631,
2//! roborev job 76).
3//!
4//! # The defect this module exists for
5//!
6//! `native_marshal_to_cql_type` used to take the text after the LAST `.` of a
7//! marshal name and match it against Cassandra's native types — so it ignored the
8//! PACKAGE entirely, and a third-party `com.acme.Int32Type` was decoded as CQL
9//! `int`. That is a no-heuristics violation (#28): CQLite knew nothing about that
10//! class and used its NAME RESEMBLANCE to pick a byte layout.
11//!
12//! # Authority — the pinned tag, never CQLite's own tables (#3041)
13//!
14//! `cassandra-5.0.8:src/java/org/apache/cassandra/db/marshal/TypeParser.java:450`
15//! (and the identical line 466) is the single place Cassandra turns a name inside
16//! a marshal type string into a class:
17//!
18//! ```java
19//! String className = compareWith.contains(".") ? compareWith
20//!                  : "org.apache.cassandra.db.marshal." + compareWith;
21//! ```
22//!
23//! So a marshal name has **exactly two** legal spellings — the bare simple name
24//! (resolved in the marshal package) and a fully-qualified class name — and every
25//! `AbstractType` CQLite maps lives in that ONE package: at the pinned tag every
26//! `.java` under `src/java/org/apache/cassandra/db/marshal/` declares
27//! `package org.apache.cassandra.db.marshal;`. There is no second legitimate
28//! package for a native marshal type. (`org.apache.cassandra.cql3.functions.types`
29//! also holds a `TupleType` and a `UserType`, but those are DRIVER types, not
30//! `AbstractType`s — `getAbstractType` would load one and then fail to find its
31//! `instance` field — so they are not a marshal spelling of anything.)
32//!
33//! A name qualified with any OTHER package is therefore a third-party
34//! `AbstractType` whose byte layout CQLite does not know.
35

38impl V5CompressedLegacyParser {
39    /// Cassandra's marshal package, trailing `.` included — the ONE spelling of
40    /// this string on the read path.
41    pub(in crate::storage::sstable::reader::parsing::row_decoder) const MARSHAL_PACKAGE:
42        &'static str = "org.apache.cassandra.db.marshal.";
43
44    /// The class-name HEAD of a marshal type string: everything before the first
45    /// `(`, trimmed. `MapType(A,B)` -> `MapType`; `Int32Type` -> `Int32Type`.
46    ///
47    /// The head, and never the whole string, is what the package rule applies to:
48    /// the last `.` of `com.acme.VectorType(org.apache.cassandra.db.marshal.Int32Type)`
49    /// is inside the ARGUMENTS, so a rule applied to the whole string would read
50    /// that foreign class as living in the marshal package.
51    pub(in crate::storage::sstable::reader::parsing::row_decoder) fn marshal_head(
52        type_str: &str,
53    ) -> &str {
54        let head = type_str.trim();
55        match head.find('(') {
56            Some(open) => head[..open].trim(),
57            None => head,
58        }
59    }
60
61    /// Split a marshal class-name HEAD into `(package, simple)`, where `package`
62    /// KEEPS its trailing `.` and is `""` for an unqualified head.
63    ///
64    /// One parse of the head, so the rule below and the diagnostics that report a
65    /// rejection cannot disagree about where the package ends.
66    pub(in crate::storage::sstable::reader::parsing::row_decoder) fn split_marshal_head(
67        head: &str,
68    ) -> (&str, &str) {
69        let head = head.trim();
70        match head.rfind('.') {
71            // `..=dot` keeps the separating `.`, so `package` is the whole package
72            // INCLUDING its terminator — which is what makes the equality below
73            // reject `…db.marshalX.`.
74            Some(dot) => (&head[..=dot], &head[dot + 1..]),
75            None => ("", head),
76        }
77    }
78
79    /// **THE package rule.** The SIMPLE class name Cassandra's own `TypeParser`
80    /// would resolve `head` to, or `None` when `head` is qualified OUTSIDE
81    /// `org.apache.cassandra.db.marshal` (see this module's header for the pinned
82    /// authority).
83    ///
84    /// # Why the package is compared for EQUALITY
85    /// `starts_with`, `ends_with` and `contains` each admit a package that merely
86    /// RESEMBLES the marshal one, and all three shapes are real:
87    /// `org.apache.cassandra.db.marshalX.Int32Type` (prefix),
88    /// `notorg.apache.cassandra.db.marshal.Int32Type` (suffix) and
89    /// `my.org.apache.cassandra.db.marshal.UserType(…)` (substring — the shape the
90    /// `UserType(` marker locator admitted). Equality on the package, `.`
91    /// terminator included, is the only test that rejects all of them.
92    ///
93    /// The PACKAGE compare is ASCII-case-insensitive, preserving the tolerance the
94    /// reader already applied to the `UserType(` marker; a case variant of the
95    /// marshal package can never be a DIFFERENT package, so the tolerance cannot
96    /// admit a foreign type. The SIMPLE NAME stays case-SENSITIVE, because it is
97    /// the whole identity of the class and `classForName` is case-sensitive.
98    pub(in crate::storage::sstable::reader::parsing::row_decoder) fn marshal_simple_name(
99        head: &str,
100    ) -> Option<&str> {
101        let (package, simple) = Self::split_marshal_head(head);
102        (package.is_empty() || package.eq_ignore_ascii_case(Self::MARSHAL_PACKAGE))
103            .then_some(simple)
104    }
105
106    /// The simple name of a **QUALIFIED** marshal class name — `Some` only for the
107    /// canonical `org.apache.cassandra.db.marshal.X` spelling.
108    ///
109    /// Structural forms and the `UserType(` marker use this NARROWER gate rather
110    /// than [`Self::marshal_simple_name`], which is exactly what the reader has
111    /// always required of them: roborev jobs 1359/1361 deliberately put top-level
112    /// and nested `UserType` parsing on the same QUALIFIED marker, so admitting a
113    /// bare `ListType(`/`UserType(` at one site only would recreate the
114    /// partial-bare-support inconsistency they closed (nested UDT fields blobbed
115    /// while the top level parsed, or the reverse). It is defined THROUGH the
116    /// package rule, so "what the marshal package is" still has one definition,
117    /// and being strictly narrower it can never admit a foreign package.
118    fn qualified_marshal_simple_name(head: &str) -> Option<&str> {
119        let head = head.trim();
120        if !head.contains('.') {
121            return None;
122        }
123        Self::marshal_simple_name(head)
124    }
125
126    /// The argument text of the marshal parameterised form `<pkg>CLASS(…)` — the
127    /// slice starting just after the `(` — or `None` when `type_str` is not that
128    /// form under the package rule.
129    ///
130    /// The ONE dispatcher for every structural arm (`FrozenType`, `UserType`,
131    /// `ListType`, `SetType`, `MapType`, `TupleType`, `ReversedType`), which each
132    /// used to carry its own copy of the fully-qualified literal.
133    pub(in crate::storage::sstable::reader::parsing::row_decoder) fn marshal_parameterised_inner<
134        'a,
135    >(
136        type_str: &'a str,
137        class: &str,
138    ) -> Option<&'a str> {
139        let trimmed = type_str.trim();
140        let open = trimmed.find('(')?;
141        if Self::qualified_marshal_simple_name(&trimmed[..open])? != class {
142            return None;
143        }
144        Some(&trimmed[open + 1..])
145    }
146
147    /// A character that may appear in a Java class-name REFERENCE: an identifier
148    /// character or the package separator. Used to find where the class name
149    /// preceding a `(` begins.
150    fn is_marshal_name_char(c: char) -> bool {
151        c.is_ascii_alphanumeric() || c == '_' || c == '$' || c == '.'
152    }
153
154    /// The byte index just after the `(` of the outermost marshal `UserType(`
155    /// marker in `type_str`, or `None`.
156    ///
157    /// The marker is searched for (rather than matched at the start) because a
158    /// top-level UDT legitimately arrives wrapped — `FrozenType(UserType(…))`.
159    /// A plain substring `find` of the qualified marker is what admitted a package
160    /// SUFFIX (`my.org.apache.cassandra.db.marshal.UserType(…)`), so each
161    /// candidate's class-name HEAD is extracted and put through the package rule.
162    fn find_marshal_user_type_inner_start(type_str: &str) -> Option<usize> {
163        const NAME: &str = "usertype";
164        // `to_ascii_lowercase`, NOT `to_lowercase`: it is length- AND
165        // boundary-preserving, so an index found in it is a valid index into
166        // `type_str`. `to_lowercase` can change a string's length (`İ` -> `i̇`),
167        // which would make the offsets below slice the wrong bytes or panic.
168        let lower = type_str.to_ascii_lowercase();
169        let mut from = 0usize;
170        while let Some(rel) = lower[from..].find(NAME) {
171            let name_end = from + rel + NAME.len();
172            // Guarantees progress whichever branch below is taken.
173            from = name_end;
174            if !lower[name_end..].starts_with('(') {
175                continue;
176            }
177            // The class-name head is the identifier/package run ending at the `(`.
178            // Stepping back by `char_indices().rev()` keeps every index on a char
179            // boundary, which a `rfind(pred).map(|i| i + 1)` would not for a
180            // multi-byte delimiter.
181            let head_start = type_str[..name_end]
182                .char_indices()
183                .rev()
184                .find(|(_, c)| !Self::is_marshal_name_char(*c))
185                .map_or(0, |(i, c)| i + c.len_utf8());
186            if Self::qualified_marshal_simple_name(&type_str[head_start..name_end])
187                .is_some_and(|simple| simple.eq_ignore_ascii_case(NAME))
188            {
189                return Some(name_end + 1);
190            }
191        }
192        None
193    }

195    /// The comma-separated ARGUMENTS of the outermost marshal `UserType(...)` in
196    /// `type_str`: `[keyspace, hex(name), hex(field):type, …]`.
197    ///
198    /// ONE locator + paren walk for both `UserType(` consumers — the `CqlType`
199    /// parser (`parse_udt_type_definition_with_depth`) and the raw-marshal one
200    /// (`udt_field_marshal_types`) — which each carried an independent copy of
201    /// this code, and so each carried the same package-suffix hole.
202    pub(in crate::storage::sstable::reader::parsing::row_decoder) fn marshal_user_type_args(
203        type_str: &str,
204    ) -> Result<Vec<String>> {
205        let inner_start = Self::find_marshal_user_type_inner_start(type_str)
206            .ok_or_else(|| Error::schema(format!("Not a UserType: {}", type_str)))?;
207
208        // Find the `)` matching the marker's `(`.
209        let mut paren_depth = 1usize;
210        let mut end_idx = None;
211        for (rel, c) in type_str[inner_start..].char_indices() {
212            match c {
213                '(' => paren_depth += 1,
214                ')' => {
215                    paren_depth -= 1;
216                    if paren_depth == 0 {
217                        end_idx = Some(inner_start + rel);
218                        break;
219                    }
220                }
221                _ => {}
222            }
223        }
224        let Some(end_idx) = end_idx else {
225            return Err(Error::schema(format!(
226                "Unbalanced parentheses in UserType: {}",
227                type_str
228            )));
229        };
230
231        let parts = Self::split_type_args(&type_str[inner_start..end_idx])?;
232        if parts.len() < 2 {
233            return Err(Error::schema(format!(
234                "UserType requires at least keyspace and name: {}",
235                &type_str[inner_start..end_idx]
236            )));
237        }
238        Ok(parts)
239    }

241    /// The refusal for a marshal name qualified OUTSIDE Cassandra's marshal
242    /// package: an accurate, self-describing error rather than a native decode, a
243    /// silent [`Value::Blob`] or the misattributed "nested user-defined type"
244    /// message (issue #3631 / #28, roborev job 76).
245    pub(in crate::storage::sstable::reader::parsing::row_decoder) fn foreign_marshal_package_error(
246        type_str: &str,
247    ) -> Error {
248        let (package, simple) = Self::split_marshal_head(Self::marshal_head(type_str));
249        Error::unsupported_format(format!(
250            "cannot decode declared type '{type_str}': its marshal class is qualified \
251             outside Cassandra's marshal package (package '{package}', expected \
252             '{expected}'), so it is a third-party AbstractType whose byte layout CQLite \
253             does not know. Matching its simple name '{simple}' against Cassandra's \
254             native types would be a guess about that class's layout, and returning the \
255             raw bytes as a blob would silently discard the declared type \
256             (issue #3631 / #28)",
257            expected = Self::MARSHAL_PACKAGE,
258        ))
259    }
