1//! Per-column decode dispatch tag (Epic J, issue #1635).
2//!
3//! Type dispatch is constant per column but was resolved per cell: every non-key
4//! cell called `column.data_type.to_lowercase()` and walked a ~30-arm string ladder
5//! (`cell_value.rs`), plus a second `to_lowercase()` in `is_complex_column`. A
6//! 1M-row × 10-col scan performed ~20M transient type-string normalizations
7//! producing nothing but a branch target.
8//!
9//! `CellKind` is the precomputed dispatch tag. It is computed ONCE per column at
10//! `RowColumnResolution::build` time (issue #1046 hoist) and stored on
11//! `ColumnToParse`; the per-cell decode then `match`es on it (a jump table) with no
12//! per-cell allocation. Its scalar variants map 1:1 onto the existing scalar decode
13//! arms in [`super::V5CompressedLegacyParser::parse_cell_value_schema_order`]; the
14//! [`CellKind::Complex`] variant carries the already-lowercased type string for the
15//! frozen / tuple / non-frozen-collection / marshal-UDT / unknown-scalar slow paths
16//! (the "thin adapter" Epic J2 later collapses — J1 makes *dispatch* per-column, not
17//! the decode bodies).
18//!
19//! No-heuristics (issue #28): the tag is derived ONLY from the authoritative column
20//! type string (supplied schema type, or the on-disk SerializationHeader marshal
21//! type for a dropped column) — never from value byte patterns.
22
23use std::sync::Arc;
24
25/// A column's precomputed value-decode dispatch, resolved once per block.
26///
27/// Scalar variants correspond exactly to the scalar arms of
28/// `parse_cell_value_schema_order`; [`CellKind::Complex`] carries the lowercased
29/// declared type for the frozen/tuple/collection/marshal-UDT/default decode ladder.
30#[derive(Debug, Clone, PartialEq, Eq)]
31pub(super) enum CellKind {
32    Boolean,
33    /// CQL `int` (i32).
34    Int,
35    /// CQL `text` / `varchar` / `ascii`.
36    Text,
37    /// CQL `uuid` / `timeuuid`.
38    Uuid,
39    Decimal,
40    /// CQL `bigint` (i64).
41    BigInt,
42    Counter,
43    /// CQL `double` (f64).
44    Double,
45    Timestamp,
46    Date,
47    Duration,
48    /// CQL `float` (f32).
49    Float,
50    /// CQL `smallint` / `short`.
51    SmallInt,
52    /// CQL `tinyint` / `byte`.
53    TinyInt,
54    Time,
55    Inet,
56    /// The literal CQL `blob` type (decodes to `Blob`, empty-value → `Blob([])`).
57    Blob,
58    /// Frozen / tuple / non-frozen-collection / marshal-UDT / unknown-scalar types:
59    /// the already-lowercased declared type string, decoded by the retained string
60    /// ladder. Empty-value → `Null` (matching the pre-J1 `_ => Null` empty arm).
61    Complex(Arc<str>),
62}
63
64impl CellKind {
65    /// Resolve the decode dispatch for a declared type string. Lowercases the type
66    /// ONCE (the same normalization the per-cell path did, now paid once per column
67    /// at bind time) and maps it to the matching scalar arm, or [`CellKind::Complex`]
68    /// (carrying the lowercased string) for everything the scalar ladder does not
69    /// name directly. Pure; never inspects value bytes (no-heuristics, issue #28).
70    #[must_use]
71    pub(super) fn from_type(data_type: &str) -> CellKind {
72        let lowered = data_type.to_lowercase();
73        match lowered.as_str() {
74            "boolean" => CellKind::Boolean,
75            "int" => CellKind::Int,
76            "text" | "varchar" | "ascii" => CellKind::Text,
77            "uuid" | "timeuuid" => CellKind::Uuid,
78            "decimal" => CellKind::Decimal,
79            "bigint" => CellKind::BigInt,
80            "counter" => CellKind::Counter,
81            "double" => CellKind::Double,
82            "timestamp" => CellKind::Timestamp,
83            "date" => CellKind::Date,
84            "duration" => CellKind::Duration,
85            "float" => CellKind::Float,
86            "smallint" | "short" => CellKind::SmallInt,
87            "tinyint" | "byte" => CellKind::TinyInt,
88            "time" => CellKind::Time,
89            "inet" => CellKind::Inet,
90            "blob" => CellKind::Blob,
91            // frozen<…>, tuple<…>, list/set/map<…>, marshal forms, varint, and any
92            // unrecognized type: decoded by the retained string ladder. Store the
93            // lowercased string so the ladder needs no per-cell `to_lowercase`.
94            _ => CellKind::Complex(Arc::from(lowered.as_str())),
95        }
96    }
97}

99#[cfg(test)]
100mod tests {
101    use super::*;
102
103    #[test]
104    fn scalar_types_map_to_scalar_kinds() {
105        assert_eq!(CellKind::from_type("boolean"), CellKind::Boolean);
106        assert_eq!(CellKind::from_type("int"), CellKind::Int);
107        assert_eq!(CellKind::from_type("bigint"), CellKind::BigInt);
108        assert_eq!(CellKind::from_type("counter"), CellKind::Counter);
109        assert_eq!(CellKind::from_type("float"), CellKind::Float);
110        assert_eq!(CellKind::from_type("double"), CellKind::Double);
111        assert_eq!(CellKind::from_type("decimal"), CellKind::Decimal);
112        assert_eq!(CellKind::from_type("timestamp"), CellKind::Timestamp);
113        assert_eq!(CellKind::from_type("date"), CellKind::Date);
114        assert_eq!(CellKind::from_type("time"), CellKind::Time);
115        assert_eq!(CellKind::from_type("duration"), CellKind::Duration);
116        assert_eq!(CellKind::from_type("inet"), CellKind::Inet);
117        assert_eq!(CellKind::from_type("blob"), CellKind::Blob);
118    }
119
120    #[test]
121    fn text_family_and_uuid_family_collapse() {
122        assert_eq!(CellKind::from_type("text"), CellKind::Text);
123        assert_eq!(CellKind::from_type("varchar"), CellKind::Text);
124        assert_eq!(CellKind::from_type("ascii"), CellKind::Text);
125        assert_eq!(CellKind::from_type("uuid"), CellKind::Uuid);
126        assert_eq!(CellKind::from_type("timeuuid"), CellKind::Uuid);
127        assert_eq!(CellKind::from_type("smallint"), CellKind::SmallInt);
128        assert_eq!(CellKind::from_type("short"), CellKind::SmallInt);
129        assert_eq!(CellKind::from_type("tinyint"), CellKind::TinyInt);
130        assert_eq!(CellKind::from_type("byte"), CellKind::TinyInt);
131    }
132
133    #[test]
134    fn declared_type_is_normalized_once_case_insensitively() {
135        // The supplied schema may carry uppercase CQL type names; dispatch must be
136        // case-insensitive exactly like the removed per-cell `to_lowercase`.
137        assert_eq!(CellKind::from_type("TEXT"), CellKind::Text);
138        assert_eq!(CellKind::from_type("Int"), CellKind::Int);
139        assert_eq!(CellKind::from_type("BigInt"), CellKind::BigInt);
140    }
141
142    #[test]
143    fn complex_and_unknown_types_carry_lowercased_string() {
144        assert_eq!(
145            CellKind::from_type("frozen<list<int>>"),
146            CellKind::Complex(Arc::from("frozen<list<int>>"))
147        );
148        assert_eq!(
149            CellKind::from_type("Tuple<Int, Text>"),
150            CellKind::Complex(Arc::from("tuple<int, text>"))
151        );
152        // Non-frozen collections reach here only via the recursion/test paths; the
153        // hot loop routes them to `parse_complex_column` before dispatch.
154        assert_eq!(
155            CellKind::from_type("list<int>"),
156            CellKind::Complex(Arc::from("list<int>"))
157        );
158        // `varint` has no dedicated scalar arm — pre-J1 it fell to the default blob
159        // decode (live) / Null (empty); it must land in `Complex`, not `Blob`.
160        assert_eq!(
161            CellKind::from_type("varint"),
162            CellKind::Complex(Arc::from("varint"))
163        );
164    }
