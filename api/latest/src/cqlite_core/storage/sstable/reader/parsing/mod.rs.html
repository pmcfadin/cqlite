41/// Extract keyspace and table name from SSTable directory path
42///
43/// SSTable paths follow Cassandra convention:
44/// `/path/to/sstables/{keyspace}/{table_name}-{uuid}/nb-1-big-Data.db`
45///
46/// # Arguments
47/// * `path` - Path to the SSTable Data.db file
48///
49/// # Returns
50/// * `Ok((keyspace, table_name))` - Extracted names
51/// * `Err(Error::Schema)` - If path doesn't match expected format
52///
53/// # Examples
54/// ```ignore
55/// let path = Path::new("/data/test_basic/simple_table-abc/nb-1-big-Data.db");
56/// let (keyspace, table) = extract_keyspace_table_from_path(path)?;
57/// assert_eq!(keyspace, "test_basic");
58/// assert_eq!(table, "simple_table");
59/// ```
60fn extract_keyspace_table_from_path(path: &Path) -> Result<(String, String)> {
61    // Get parent directory containing table_name-uuid
62    let table_dir = path
63        .parent()
64        .ok_or_else(|| Error::schema("SSTable path has no parent directory"))?;
65
66    // Extract table directory name
67    let table_dir_name = table_dir
68        .file_name()
69        .and_then(|n| n.to_str())
70        .ok_or_else(|| Error::schema("Invalid table directory name"))?;
71
72    // Split on last hyphen to handle table names containing hyphens
73    // Format: "table_name-uuid" or "user-profiles-abc123"
74    let table_name = table_dir_name
75        .rsplit_once('-')
76        .ok_or_else(|| {
77            Error::schema(format!(
78                "Table directory '{}' does not match 'tablename-uuid' format",
79                table_dir_name
80            ))
81        })?
82        .0
83        .to_string();
84
85    // Get keyspace directory (parent of table directory)
86    let keyspace_dir = table_dir
87        .parent()
88        .ok_or_else(|| Error::schema("Table directory has no parent (keyspace) directory"))?;
89
90    // Extract keyspace name
91    let keyspace = keyspace_dir
92        .file_name()
93        .and_then(|n| n.to_str())
94        .ok_or_else(|| Error::schema("Invalid keyspace directory name"))?
95        .to_string();
96
97    Ok((keyspace, table_name))
98}

100impl SSTableReader {
101    /// Get table schema using four-tier lookup strategy
102    ///
103    /// This method implements a fallback chain for resolving table schemas:
104    /// 0. **Provided Schema**: Use schema passed from query executor (highest priority)
105    /// 1. **SSTable Header**: Check `self.schema` (extracted during SSTable opening from V5.0+ headers)
106    /// 2. **Schema Registry**: Look up schema from external registry (loaded via --schema flag)
107    /// 3. **Header Construction**: Build basic schema from header column metadata (fallback)
108    pub(in crate::storage::sstable::reader) fn get_table_schema(
109        &self,
110        provided_schema: Option<&TableSchema>,
111    ) -> Option<TableSchema> {
112        // Strategy 0: Use provided schema from query executor (highest priority)
113        if let Some(schema) = provided_schema {
114            debug!(
115                "get_table_schema: Using provided schema for {}.{}",
116                schema.keyspace, schema.table
117            );
118            return Some(schema.clone());
119        }
120
121        // Strategy 1: Use schema extracted from SSTable header (if available)
122        if let Some(schema) = self.schema.as_deref() {
123            debug!(
124                "get_table_schema: Using schema from SSTable header for {}.{}",
125                self.header.keyspace, self.header.table_name
126            );
127            return Some(schema.clone());
128        }
129
130        // Strategy 2: Look up schema from schema registry (if available)
131        #[cfg(feature = "state_machine")]
132        {
133            if let Some(registry_rwlock) = self.schema_registry.as_ref() {
134                // We need to call async methods from a sync context.
135                // Use futures::executor::block_on() which is safe here since this is
136                // called from parsing contexts that are already in async contexts.
137                if tokio::runtime::Handle::try_current().is_ok() {
138                    // We're in a tokio context, use block_on
139                    use futures::executor::block_on;
140
141                    let registry = block_on(registry_rwlock.read());
142
143                    // Extract keyspace/table from SSTable path (authoritative source)
144                    // Directory structure: {keyspace}/{table_name}-{uuid}/Data.db
145                    let (keyspace, table_name) = match extract_keyspace_table_from_path(
146                        &self.file_path,
147                    ) {
148                        Ok(names) => names,
149                        Err(e) => {
150                            debug!(
151                                "get_table_schema: Failed to extract names from path {}: {}. Falling back to header names.",
152                                self.file_path.display(), e
153                            );
154                            // Fallback to header names if path parsing fails
155                            (self.header.keyspace.clone(), self.header.table_name.clone())
156                        }
157                    };
158
159                    match block_on(registry.get_schema(&keyspace, &table_name)) {
160                        Ok(schema) => {
161                            debug!(
162                                "get_table_schema: Using schema from registry for {}.{}",
163                                keyspace, table_name
164                            );
165                            return Some(schema);
166                        }
167                        Err(e) => {
168                            debug!(
169                                "get_table_schema: Schema not found in registry for {}.{}: {}",
170                                keyspace, table_name, e
171                            );
172                        }
173                    }
174                } else {
175                    debug!("get_table_schema: Not in tokio context, skipping registry lookup");
176                }
177            }
178        }
179
180        // For non-state_machine builds, schema_registry is Arc<SchemaRegistry> (not async)
181        #[cfg(not(feature = "state_machine"))]
182        {
183            if let Some(registry) = self.schema_registry.as_ref() {
184                // Extract keyspace/table from SSTable path (authoritative source)
185                // Directory structure: {keyspace}/{table_name}-{uuid}/Data.db
186                let (keyspace, table_name) = match extract_keyspace_table_from_path(&self.file_path)
187                {
188                    Ok(names) => names,
189                    Err(e) => {
190                        debug!(
191                            "get_table_schema: Failed to extract names from path {}: {}. Falling back to header names.",
192                            self.file_path.display(), e
193                        );
194                        // Fallback to header names if path parsing fails
195                        (self.header.keyspace.clone(), self.header.table_name.clone())
196                    }
197                };
198
199                // Non-state_machine SchemaRegistry doesn't have async get_schema method
200                // This path is currently not implemented for non-async registries
201                debug!("get_table_schema: Schema registry lookup not available in non-state_machine builds");
202                let _ = (registry, &keyspace, &table_name); // Avoid unused variable warnings
203            }
204        }
205
206        // Strategy 3: Construct basic schema from header columns (existing logic)
207        debug!(
208            "get_table_schema: Falling back to header column construction for {}.{}",
209            self.header.keyspace, self.header.table_name
210        );
211
212        // Try to construct a basic schema from header information
213        if self.header.columns.is_empty() {
214            return None;
215        }
216
217        let mut columns = Vec::new();
218        let mut partition_keys = Vec::new();
219        let mut clustering_keys = Vec::new();
220
221        // Convert header columns to schema columns
222        for col_info in self.header.columns.iter() {
223            let column = Column {
224                name: col_info.name.clone(),
225                data_type: col_info.column_type.clone(), // Use column_type field
226                nullable: true,
227                default: None,
228                is_static: false, // TODO: Header doesn't track static columns yet
229            };
230
231            // Check if this is a key column based on primary key and clustering status
232            if col_info.is_primary_key && !col_info.is_clustering {
233                // This is a partition key
234                partition_keys.push(KeyColumn {
235                    name: col_info.name.clone(),
236                    data_type: col_info.column_type.clone(),
237                    position: partition_keys.len(),
238                });
239            } else if col_info.is_clustering {
240                clustering_keys.push(ClusteringColumn {
241                    name: col_info.name.clone(),
242                    data_type: col_info.column_type.clone(),
243                    position: clustering_keys.len(),
244                    order: crate::schema::ClusteringOrder::Asc,
245                });
246            }
247
248            columns.push(column);
249        }
250
251        Some(TableSchema {
252            keyspace: self.header.keyspace.clone(),
253            table: self.header.table_name.clone(),
254            partition_keys,
255            clustering_keys,
256            columns,
257            comments: HashMap::new(),
258        })
259    }

261    /// Parse partition data from raw bytes using schema-driven approach
262    ///
263    /// This method uses the row cell state machine to parse partition data into
264    /// individual row entries. It handles the SSTable row format including:
265    /// - Row headers and metadata
266    /// - Clustering key parsing
267    /// - Cell value extraction
268    /// - Tombstone handling
269    pub(in crate::storage::sstable::reader) fn parse_partition_data(
270        &self,
271        data: &[u8],
272        schema: Option<&crate::schema::TableSchema>,
273    ) -> Result<Option<Vec<(RowKey, Value)>>> {
274        if data.is_empty() {
275            return Ok(Some(Vec::new()));
276        }
277
278        // Create schema-aware state machine if schema is available
279        let mut state_machine = if let Some(schema) = schema {
280            // Get partition key comparators from schema
281            match schema.get_partition_key_comparators() {
282                Ok(comparators) if !comparators.is_empty() => {
283                    debug!("parse_partition_data: Creating schema-aware state machine with {} partition key comparators", comparators.len());
284                    super::super::row_cell_state_machine::RowCellStateMachine::with_schema_and_version(
285                        schema.clone(),
286                        comparators[0].clone(),
287                        self.header.cassandra_version
288                    )
289                }
290                Ok(_empty) => {
291                    warn!(
292                        "parse_partition_data: Schema for {}.{} has {} partition keys but comparator parsing returned empty - falling back to schemaless parsing",
293                        schema.keyspace, schema.table, schema.partition_keys.len()
294                    );
295                    super::super::row_cell_state_machine::RowCellStateMachine::new()
296                }
297                Err(e) => {
298                    warn!(
299                        "parse_partition_data: Failed to get partition key comparators for {}.{}: {} - falling back to schemaless parsing",
300                        schema.keyspace, schema.table, e
301                    );
302                    super::super::row_cell_state_machine::RowCellStateMachine::new()
303                }
304            }
305        } else {
306            debug!("parse_partition_data: No schema provided, using basic state machine");
307            super::super::row_cell_state_machine::RowCellStateMachine::new()
308        };
309        let mut results = Vec::new();
310
311        // Parse the partition data using the row cell state machine
312        match state_machine.parse_partition_data(data) {
313            Ok(parsed_rows) => {
314                for parsed_row in parsed_rows {
315                    // Extract row key and value from parsed row
316                    let row_key = self.extract_row_key_from_parsed_row(&parsed_row)?;
317
318                    // Use schema-aware extraction if schema is available
319                    let value = if let Some(s) = schema {
320                        // Use schema-aware extraction for proper typing
321                        self.extract_value_from_parsed_row_with_schema(&parsed_row, s)?
322                    } else {
323                        // Fallback to basic extraction when no schema
324                        self.extract_value_from_parsed_row_fallback(&parsed_row)?
325                    };
326
327                    results.push((row_key, value));
328                }
329                Ok(Some(results))
330            }
331            Err(e) => {
332                let context = if let Some(s) = schema {
333                    format!("table {}.{} with schema", s.keyspace, s.table)
334                } else {
335                    "unknown table without schema".to_string()
336                };
337
338                error!(
339                    "parse_partition_data: Failed to parse partition data for {} (format {:?}): {}",
340                    context, self.header.cassandra_version, e
341                );
342
343                // For Issue #35 compliance, we must not return synthetic data
344                // If parsing fails, we return None to indicate parsing failure
345                // This ensures zero-tolerance validation and forces proper implementation
346                Err(Error::corruption(format!(
347                    "Partition data parsing failed for {} (format {:?}): {}. \
348                     Ensure schema is provided for Cassandra 5.0+ formats (Issue #35 compliance).",
349                    context, self.header.cassandra_version, e
350                )))
351            }
352        }
353    }

355    /// Extract row key from parsed row data
356    pub(in crate::storage::sstable::reader) fn extract_row_key_from_parsed_row(
357        &self,
358        parsed_row: &ParsedRow,
359    ) -> Result<RowKey> {
360        // Extract the clustering key components from the parsed row
361        // Combine partition key + clustering key to form the full row key
362
363        // Use the row's clustering key if available
364        if let Some(clustering_key) = &parsed_row.clustering_key {
365            Ok(RowKey::from(clustering_key.clone()))
366        } else if !parsed_row.clustering_rows.is_empty() {
367            // Extract from first clustering row
368            let first_clustering_row = &parsed_row.clustering_rows[0];
369            let clustering_key_str = String::from_utf8_lossy(&first_clustering_row.clustering_key);
370            Ok(RowKey::from(clustering_key_str.to_string()))
371        } else {
372            // Fallback: create synthetic key based on partition key
373            let partition_key_str = String::from_utf8_lossy(&parsed_row.partition_key.key_bytes);
374            Ok(RowKey::from(format!("partition_{}", partition_key_str)))
375        }
376    }

378    /// Extract value from parsed row data (fallback for schemaless parsing)
379    ///
380    /// This method is used when no schema is available. It returns the first
381    /// non-null cell value, which may be a blob.
382    pub(in crate::storage::sstable::reader) fn extract_value_from_parsed_row_fallback(
383        &self,
384        parsed_row: &ParsedRow,
385    ) -> Result<Value> {
386        // Extract the primary value from the row's cells
387        // For tables with multiple columns, this might be a UDT or JSON representation
388
389        // First, try to get value from cells (the new flattened structure)
390        if !parsed_row.cells.is_empty() {
391            // Return the first non-null cell value
392            for cell in &parsed_row.cells {
393                if let Some(ref value) = cell.value {
394                    return Ok(value.clone());
395                }
396            }
397        }
398
399        // Fallback: try to extract from clustering rows
400        if !parsed_row.clustering_rows.is_empty() {
401            let first_row = &parsed_row.clustering_rows[0];
402            if !first_row.columns.is_empty() {
403                // Return the first column value
404                if let Some((_, value)) = first_row.columns.iter().next() {
405                    return Ok(value.clone());
406                }
407            }
408        }
409
410        // Fallback: try static row data
411        if let Some(ref static_row) = parsed_row.static_row {
412            if !static_row.columns.is_empty() {
413                if let Some((_, value)) = static_row.columns.iter().next() {
414                    return Ok(value.clone());
415                }
416            }
417        }
418
419        // Final fallback: return metadata about the row
420        let cell_count = parsed_row.cells.len();
421        let cluster_count = parsed_row.clustering_rows.len();
422        Ok(Value::Text(format!(
423            "row_with_{}_cells_{}_clusters",
424            cell_count, cluster_count
425        )))
426    }

428    /// Extract typed value from parsed row using schema information
429    ///
430    /// This method builds a complete row map with all columns properly typed
431    /// according to the schema, instead of returning just the first cell as a blob.
432    pub(in crate::storage::sstable::reader) fn extract_value_from_parsed_row_with_schema(
433        &self,
434        parsed_row: &ParsedRow,
435        schema: &crate::schema::TableSchema,
436    ) -> Result<Value> {
437        use std::collections::HashMap;
438
439        let mut columns: HashMap<String, Value> = HashMap::new();
440
441        debug!(
442            "extract_value_with_schema: Processing row with {} cells",
443            parsed_row.cells.len()
444        );
445        debug!(
446            "extract_value_with_schema: Schema has {} partition keys, {} clustering keys, {} columns",
447            schema.partition_keys.len(),
448            schema.clustering_keys.len(),
449            schema.columns.len()
450        );
451
452        // Process partition key components
453        for (idx, component) in parsed_row.partition_key.components.iter().enumerate() {
454            if let Some(pk_col) = schema.partition_keys.get(idx) {
455                debug!(
456                    "extract_value_with_schema: Processing partition key column: {}",
457                    pk_col.name
458                );
459
460                // Parse the component bytes with schema type
461                let typed_value =
462                    self.parse_value_with_schema_type(component, &pk_col.data_type)
463                        .unwrap_or_else(|e| {
464                            warn!(
465                                "extract_value_with_schema: Failed to parse partition key {}: {}, using blob fallback",
466                                pk_col.name, e
467                            );
468                            Value::Blob(component.clone())
469                        });
470
471                columns.insert(pk_col.name.clone(), typed_value);
472            }
473        }
474
475        // Process clustering key if present
476        if let Some(ref clustering_key) = parsed_row.clustering_key {
477            // The clustering_key is a String representation, we need to use the raw bytes
478            // from partition_key components if they represent clustering
479            // For now, we'll check if clustering_rows have data
480            if !parsed_row.clustering_rows.is_empty() {
481                // Extract clustering key from first clustering row
482                let first_clustering_row = &parsed_row.clustering_rows[0];
483                // clustering_key is Vec<u8>, parse it as composite if needed
484                let ck_bytes = &first_clustering_row.clustering_key;
485
486                // For single clustering key
487                if schema.clustering_keys.len() == 1 {
488                    let ck_col = &schema.clustering_keys[0];
489                    debug!(
490                        "extract_value_with_schema: Processing clustering key column: {}",
491                        ck_col.name
492                    );
493
494                    let typed_value =
495                        self.parse_value_with_schema_type(ck_bytes, &ck_col.data_type)
496                            .unwrap_or_else(|e| {
497                                warn!(
498                                    "extract_value_with_schema: Failed to parse clustering key {}: {}, using blob fallback",
499                                    ck_col.name, e
500                                );
501                                Value::Blob(ck_bytes.clone())
502                            });
503
504                    columns.insert(ck_col.name.clone(), typed_value);
505                } else if schema.clustering_keys.len() > 1 {
506                    // For composite clustering keys, we need to parse the composite structure
507                    // This is a TODO for now - use blob fallback
508                    warn!(
509                        "extract_value_with_schema: Composite clustering keys not yet implemented for {}.{} ({} keys) - using string representation fallback",
510                        schema.keyspace, schema.table, schema.clustering_keys.len()
511                    );
512                    for ck_col in &schema.clustering_keys {
513                        columns.insert(ck_col.name.clone(), Value::Text(clustering_key.clone()));
514                    }
515                }
516            } else {
517                // No clustering rows, just use the string representation
518                for ck_col in &schema.clustering_keys {
519                    columns.insert(ck_col.name.clone(), Value::Text(clustering_key.clone()));
520                }
521            }
522        }
523
524        // Process regular columns from cells
525        for cell in &parsed_row.cells {
526            if let Some(col) = schema.columns.iter().find(|c| c.name == cell.column_name) {
527                debug!(
528                    "extract_value_with_schema: Processing regular column: {}",
529                    cell.column_name
530                );
531
532                // Get value from cell - it's already parsed, but might be a blob
533                if let Some(ref cell_value) = cell.value {
534                    match cell_value {
535                        Value::Blob(bytes) if !bytes.is_empty() => {
536                            // Try to parse blob with schema type for better typing
537                            let typed_value =
538                                self.parse_value_with_schema_type(bytes, &col.data_type)
539                                    .unwrap_or_else(|e| {
540                                        debug!(
541                                            "extract_value_with_schema: Failed to parse column {}: {}, keeping blob",
542                                            cell.column_name, e
543                                        );
544                                        Value::Blob(bytes.clone())
545                                    });
546                            columns.insert(cell.column_name.clone(), typed_value);
547                        }
548                        _ => {
549                            // Use the already-typed value from cell
550                            columns.insert(cell.column_name.clone(), cell_value.clone());
551                        }
552                    }
553                } else {
554                    // Null value
555                    columns.insert(cell.column_name.clone(), Value::Null);
556                }
557            }
558        }
559
560        // Also process clustering row columns if present
561        for clustering_row in &parsed_row.clustering_rows {
562            for (col_name, col_value) in &clustering_row.columns {
563                if let Some(col) = schema.columns.iter().find(|c| c.name == *col_name) {
564                    debug!(
565                        "extract_value_with_schema: Processing clustering row column: {}",
566                        col_name
567                    );
568
569                    match col_value {
570                        Value::Blob(bytes) if !bytes.is_empty() => {
571                            // Try to parse blob with schema type
572                            let typed_value =
573                                self.parse_value_with_schema_type(bytes, &col.data_type)
574                                    .unwrap_or_else(|e| {
575                                        debug!(
576                                            "extract_value_with_schema: Failed to parse clustering row column {}: {}, keeping blob",
577                                            col_name, e
578                                        );
579                                        Value::Blob(bytes.clone())
580                                    });
581                            columns.insert(col_name.clone(), typed_value);
582                        }
583                        _ => {
584                            // Use the already-typed value
585                            columns.insert(col_name.clone(), col_value.clone());
586                        }
587                    }
588                }
589            }
590        }
591
592        // Process static row columns if present
593        if let Some(ref static_row) = parsed_row.static_row {
594            for (col_name, col_value) in &static_row.columns {
595                if let Some(col) = schema.columns.iter().find(|c| c.name == *col_name) {
596                    debug!(
597                        "extract_value_with_schema: Processing static row column: {}",
598                        col_name
599                    );
600
601                    match col_value {
602                        Value::Blob(bytes) if !bytes.is_empty() => {
603                            // Try to parse blob with schema type
604                            let typed_value =
605                                self.parse_value_with_schema_type(bytes, &col.data_type)
606                                    .unwrap_or_else(|e| {
607                                        debug!(
608                                            "extract_value_with_schema: Failed to parse static row column {}: {}, keeping blob",
609                                            col_name, e
610                                        );
611                                        Value::Blob(bytes.clone())
612                                    });
613                            columns.insert(col_name.clone(), typed_value);
614                        }
615                        _ => {
616                            // Use the already-typed value
617                            columns.insert(col_name.clone(), col_value.clone());
618                        }
619                    }
620                }
621            }
622        }
623
624        // Validate we got at least some columns
625        if columns.is_empty() {
626            return Err(Error::Schema(format!(
627                "No columns matched schema - parsed {} cells but none matched {} schema columns",
628                parsed_row.cells.len(),
629                schema.columns.len()
630            )));
631        }
632
633        debug!(
634            "extract_value_with_schema: Extracted {} columns into row map",
635            columns.len()
636        );
637
638        // Convert columns HashMap to UDT fields
639        let fields = columns
640            .into_iter()
641            .map(|(name, value)| crate::types::UdtField {
642                name,
643                value: Some(value),
644            })
645            .collect();
646
647        // Return as UDT value representing the row
648        // NOTE: We use Value::Udt to represent table rows because Value::Row doesn't
649        // exist in the current type system. The query executor treats UDT values with
650        // type_name matching the table name as row representations. This is semantically
651        // imperfect but structurally sound. Consider adding Value::Row in M3 type system refactor.
652        Ok(Value::Udt(crate::types::UdtValue {
653            keyspace: schema.keyspace.clone(),
654            type_name: schema.table.clone(), // Table name as UDT type name
655            fields,
656        }))
657    }
658
659    /// Parse partition data at a specific offset in the Data.db file
660    ///
661    /// This method reads and parses partition data from a specific offset,
662    /// returning the parsed row key-value pairs.
663    pub(in crate::storage::sstable::reader) async fn parse_partition_at_offset(
664        &self,
665        offset: u64,
666        size: u32,
667    ) -> Result<Option<Vec<(RowKey, Value)>>> {
668        // Read data from file at specified offset
669        use tokio::io::{AsyncReadExt, AsyncSeekExt};
670
671        let mut file = self.file.lock().await;
672        file.seek(std::io::SeekFrom::Start(offset)).await?;
673
674        let mut buffer = vec![0u8; size as usize];
675        file.read_exact(&mut buffer).await?;
676        drop(file); // Release lock early
677
678        // Decompress if needed
679        let data = if let Some(compression_reader) = &self.compression_reader {
680            let compression =
681                super::super::compression::Compression::new(*compression_reader.algorithm())?;
682            match compression.decompress(&buffer) {
683                Ok(decompressed) => decompressed,
684                Err(e) => {
685                    log::warn!("Decompression failed at offset {}: {}", offset, e);
686                    buffer // Fallback to raw data
687                }
688            }
689        } else {
690            buffer
691        };
692
693        // Parse the partition data with schema using four-tier lookup
694        let table_schema = self.get_table_schema(None);
695        self.parse_partition_data(&data, table_schema.as_ref())
696    }
697}
