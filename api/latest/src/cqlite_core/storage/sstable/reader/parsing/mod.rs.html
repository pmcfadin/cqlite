64/// Extract keyspace and table name from SSTable directory path
65///
66/// SSTable paths follow Cassandra convention:
67/// `/path/to/sstables/{keyspace}/{table_name}-{uuid}/nb-1-big-Data.db`
68///
69/// # Arguments
70/// * `path` - Path to the SSTable Data.db file
71///
72/// # Returns
73/// * `Ok((keyspace, table_name))` - Extracted names
74/// * `Err(Error::Schema)` - If path doesn't match expected format
75///
76/// # Examples
77/// ```ignore
78/// let path = Path::new("/data/test_basic/simple_table-abc/nb-1-big-Data.db");
79/// let (keyspace, table) = extract_keyspace_table_from_path(path)?;
80/// assert_eq!(keyspace, "test_basic");
81/// assert_eq!(table, "simple_table");
82/// ```
83pub(in crate::storage::sstable::reader) fn extract_keyspace_table_from_path(
84    path: &Path,
85) -> Result<(String, String)> {
86    // Get parent directory containing table_name-uuid
87    let table_dir = path
88        .parent()
89        .ok_or_else(|| Error::schema("SSTable path has no parent directory"))?;
90
91    // Extract table directory name
92    let table_dir_name = table_dir
93        .file_name()
94        .and_then(|n| n.to_str())
95        .ok_or_else(|| Error::schema("Invalid table directory name"))?;
96
97    // Split on last hyphen to handle table names containing hyphens
98    // Format: "table_name-uuid" or "user-profiles-abc123"
99    let table_name = table_dir_name
100        .rsplit_once('-')
101        .ok_or_else(|| {
102            Error::schema(format!(
103                "Table directory '{}' does not match 'tablename-uuid' format",
104                table_dir_name
105            ))
106        })?
107        .0
108        .to_string();
109
110    // Get keyspace directory (parent of table directory)
111    let keyspace_dir = table_dir
112        .parent()
113        .ok_or_else(|| Error::schema("Table directory has no parent (keyspace) directory"))?;
114
115    // Extract keyspace name
116    let keyspace = keyspace_dir
117        .file_name()
118        .and_then(|n| n.to_str())
119        .ok_or_else(|| Error::schema("Invalid keyspace directory name"))?
120        .to_string();
121
122    Ok((keyspace, table_name))
123}

125impl SSTableReader {
126    /// Get table schema using four-tier lookup strategy
127    ///
128    /// This method implements a fallback chain for resolving table schemas:
129    /// 0. **Provided Schema**: Use schema passed from query executor (highest priority)
130    /// 1. **SSTable Header**: Check `self.schema` (extracted during SSTable opening from V5.0+ headers)
131    /// 2. **Schema Registry**: Look up schema from external registry (loaded via --schema flag)
132    /// 3. **Header Construction**: Build basic schema from header column metadata (fallback)
133    pub(in crate::storage::sstable::reader) fn get_table_schema(
134        &self,
135        provided_schema: Option<&TableSchema>,
136    ) -> Option<TableSchema> {
137        // Strategy 0: Use provided schema from query executor (highest priority)
138        if let Some(schema) = provided_schema {
139            debug!(
140                "get_table_schema: Using provided schema for {}.{}",
141                schema.keyspace, schema.table
142            );
143            return Some(schema.clone());
144        }
145
146        // Strategy 1: Use schema extracted from SSTable header (if available)
147        if let Some(schema) = self.schema.as_deref() {
148            debug!(
149                "get_table_schema: Using schema from SSTable header for {}.{}",
150                self.header.keyspace, self.header.table_name
151            );
152            return Some(schema.clone());
153        }
154
155        // Strategy 2: Use the schema pre-resolved from the registry (issue #1692).
156        //
157        // This tier is reached only when the header schema (Strategy 1) is absent.
158        // The registry is async (an `Arc<RwLock<SchemaRegistry>>` whose `get_schema`
159        // itself awaits), so it CANNOT be consulted safely from this sync fn: the
160        // old code `block_on`-ed the async lock on a tokio worker thread, which —
161        // with a small runtime and a pending registry WRITE guard — could park the
162        // workers and stall the whole runtime (AG3). Instead the registry schema is
163        // resolved ONCE, properly awaited, at the async wiring point
164        // (`SSTableReader::resolve_registry_schema`, called from
165        // `open_reader_with_schema`) and cached in `self.registry_schema`. Here we
166        // just read that plain field — no async lock, no `block_on`, no worker
167        // parking. A registry miss leaves it `None` and we fall through to header
168        // construction below, exactly as the old `block_on` path did on a miss.
169        #[cfg(feature = "state_machine")]
170        {
171            if let Some(schema) = self.registry_schema.as_deref() {
172                debug!(
173                    "get_table_schema: Using pre-resolved registry schema for {}.{}",
174                    schema.keyspace, schema.table
175                );
176                return Some(schema.clone());
177            }
178        }
179
180        // For non-state_machine builds, schema_registry is Arc<SchemaRegistry> (not async)
181        #[cfg(not(feature = "state_machine"))]
182        {
183            if let Some(registry) = self.schema_registry.as_ref() {
184                // Extract keyspace/table from SSTable path (authoritative source)
185                // Directory structure: {keyspace}/{table_name}-{uuid}/Data.db
186                let file_path = self.file_path();
187                let (keyspace, table_name) = match extract_keyspace_table_from_path(&file_path) {
188                    Ok(names) => names,
189                    Err(e) => {
190                        debug!(
191                            "get_table_schema: Failed to extract names from path {}: {}. Falling back to header names.",
192                            file_path.display(), e
193                        );
194                        // Fallback to header names if path parsing fails
195                        (self.header.keyspace.clone(), self.header.table_name.clone())
196                    }
197                };
198
199                // Non-state_machine SchemaRegistry doesn't have async get_schema method
200                // This path is currently not implemented for non-async registries
201                debug!("get_table_schema: Schema registry lookup not available in non-state_machine builds");
202                let _ = (registry, &keyspace, &table_name); // Avoid unused variable warnings
203            }
204        }
205
206        // Strategy 3: Construct basic schema from header columns (existing logic)
207        debug!(
208            "get_table_schema: Falling back to header column construction for {}.{}",
209            self.header.keyspace, self.header.table_name
210        );
211
212        // Try to construct a basic schema from header information
213        if self.header.columns.is_empty() {
214            return None;
215        }
216
217        let mut columns = Vec::new();
218        let mut partition_keys = Vec::new();
219        let mut clustering_keys = Vec::new();
220
221        // Convert header columns to schema columns
222        for col_info in self.header.columns.iter() {
223            let column = Column {
224                name: col_info.name.clone(),
225                data_type: col_info.column_type.clone(), // Use column_type field
226                nullable: true,
227                default: None,
228                // Static-column classification comes from the Statistics.db
229                // SerializationHeader (authoritative metadata) carried on
230                // ColumnInfo.is_static. Issue #758 / Epic #756.
231                is_static: col_info.is_static,
232            };
233
234            // Check if this is a key column based on primary key and clustering status
235            if col_info.is_primary_key && !col_info.is_clustering {
236                // This is a partition key
237                partition_keys.push(KeyColumn {
238                    name: col_info.name.clone(),
239                    data_type: col_info.column_type.clone(),
240                    position: partition_keys.len(),
241                });
242            } else if col_info.is_clustering {
243                clustering_keys.push(ClusteringColumn {
244                    name: col_info.name.clone(),
245                    data_type: col_info.column_type.clone(),
246                    position: clustering_keys.len(),
247                    order: crate::schema::ClusteringOrder::Asc,
248                });
249            }
250
251            columns.push(column);
252        }
253
254        Some(TableSchema {
255            keyspace: self.header.keyspace.clone(),
256            table: self.header.table_name.clone(),
257            partition_keys,
258            clustering_keys,
259            columns,
260            comments: HashMap::new(),
261            dropped_columns: HashMap::new(),
262        })
263    }
264}
