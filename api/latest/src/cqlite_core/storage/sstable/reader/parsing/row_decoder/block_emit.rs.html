10impl V5CompressedLegacyParser {
11    /// Parse decompressed block into (TableId, RowKey, ScanRow) entries
12    ///
13    /// # Arguments
14    /// * `data` - Decompressed block bytes
15    /// * `schema` - Optional table schema for type-aware parsing
16    /// * `reader` - Reference to SSTableReader for value parsing
17    ///
18    /// # Returns
19    /// * `Ok(Vec<(TableId, RowKey, ScanRow)>)` - Parsed entries
20    /// * `Err(Error)` - Parse error with context
21    pub fn parse_block(
22        &self,
23        data: &[u8],
24        extent: BufferExtent,
25        schema: Option<&TableSchema>,
26        reader: &crate::storage::sstable::reader::types::SSTableReader,
27    ) -> Result<Vec<(TableId, RowKey, ScanRow)>> {
28        let mut results = Vec::new();
29        self.parse_block_emit(data, extent, schema, reader, |entry| {
30            results.push(entry);
31            Ok(std::ops::ControlFlow::Continue(()))
32        })?;
33        Ok(results)
34    }

36    /// Parse a block and return both row values and per-cell write metadata.
37    ///
38    /// Identical to [`parse_block`] but the returned vector carries a fourth element:
39    /// the per-cell `CellWriteMetadata` map (column name → metadata). Used by the
40    /// executor when `ProjectionFlags::include_cell_metadata` is set (i.e. when the
41    /// query contains `WRITETIME(col)` or `TTL(col)` expressions).
42    pub fn parse_block_with_cell_metadata(
43        &self,
44        data: &[u8],
45        extent: BufferExtent,
46        schema: Option<&TableSchema>,
47        reader: &crate::storage::sstable::reader::types::SSTableReader,
48    ) -> Result<ParsedBlockWithMeta> {
49        let mut results = Vec::new();
50        self.parse_block_emit_with_metadata(data, extent, schema, reader, |entry| {
51            results.push(entry);
52            Ok(std::ops::ControlFlow::Continue(()))
53        })?;
54        Ok(results)
55    }

57    /// Internal streaming variant of `parse_block_with_cell_metadata`.
58    fn parse_block_emit_with_metadata<F>(
59        &self,
60        data: &[u8],
61        extent: BufferExtent,
62        schema: Option<&TableSchema>,
63        reader: &crate::storage::sstable::reader::types::SSTableReader,
64        mut emit: F,
65    ) -> Result<()>
66    where
67        F: FnMut(
68            (TableId, RowKey, ScanRow, HashMap<String, CellWriteMetadata>),
69        ) -> Result<std::ops::ControlFlow<()>>,
70    {
71        if data.is_empty() {
72            return Ok(());
73        }
74
75        let schema = schema.ok_or_else(|| {
76            Error::schema(format!(
77                "V5CompressedLegacy format requires schema for {}.{} (cells lack column names in binary data)",
78                self.keyspace, self.table_name
79            ))
80        })?;
81
82        // Issue #1046: build the header→schema column resolution ONCE per block,
83        // reused across every partition/row below — zero per-row schema-lookup alloc.
84        let resolution = RowColumnResolution::build(schema, reader);
85
86        let mut offset = 0;
87        let mut partition_index = 0;
88        // Issue #1741 (F2): read-time TTL clock — captured ONCE per parser (== once per
89        // read/scan operation) in `V5CompressedLegacyParser::new`, reused for every
90        // block/partition here so a scan crossing an expiration-second boundary decides
91        // all rows with the SAME `now`.
92        let now_secs = self.now_secs;
93        // Issue #1741: authoritative per-clustering-column reversal flags (DESC),
94        // built ONCE per block so range-tombstone coverage compares clustering
95        // prefixes in physical storage order. Bounded by clustering arity.
96        let clustering_reversed = clustering_reversed_flags(schema);
97
98        while offset < data.len() {
99            // Parse partition header: returns (RowKey, next_data_offset, deletion)
100            let (partition_key, next_data_offset, partition_deletion) =
101                match self.parse_partition_header_full(data, offset) {
102                    Ok(ph) => ph,
103                    // Issue #3928: on a PROVEN-COMPLETE buffer no further bytes
104                    // can arrive to finish this header, so the bare `break` was a
105                    // silent truncation of the whole REMAINDER of the walk — and
106                    // when the FIRST partition's header is the undecodable one,
107                    // of everything. Measured on a real `da` fixture whose first
108                    // partition's `DeletionTime` discriminator was flipped to a
109                    // value Cassandra's own reader throws on: this route
110                    // (`bti_scan_with_metadata_cancellable` → here) answered `Ok`
111                    // with **0 of 468 rows**, which is the 0-rows-when-present
112                    // failure. Mid-window the break STAYS: a header can
113                    // legitimately straddle a chunk-covering window's tail.
114                    Err(e) => {
115                        if extent.is_complete() {
116                            return Err(e);
117                        }
118                        break;
119                    }
120                };
121
122            let table_id = TableId::new(format!("{}.{}", self.keyspace, self.table_name));
123            offset = next_data_offset;
124            partition_index += 1;
125
126            // Issue #1741: per-partition read-side shadowing (WRITETIME/TTL projection
127            // path), active ONLY for user-facing query reads (`self.read_shadowing`).
128            // Un-gated; hides partition/range-tombstone-shadowed and TTL-expired rows
129            // to match a Cassandra SELECT. `None` for physical consumers.
130            let mut shadow = self.read_shadowing.then(|| {
131                PartitionShadow::open(now_secs, partition_deletion, clustering_reversed.clone())
132            });
133
134            // Issue #1642 (K3): static cells accumulate as a positional `RowCells`
135            // vector, matching the decoder's positional emit. Metadata stays keyed.
136            let mut static_cells: RowCells = Vec::new();
137            let mut static_cell_meta: HashMap<String, CellWriteMetadata> = HashMap::new();
138            let mut row_count = 0;
139            // Issue #3095: Cassandra's `partition.hasNext()` (clustering rows only)
140            // plus proof this call saw the partition's END — see the
141            // static-only-partition emission after the loop.
142            let mut emitted_clustering_row = false;
143            let mut partition_complete = false;
144
145            loop {
146                if offset < data.len() && Self::is_end_of_partition(data[offset]) {
147                    offset += 1;
148                    partition_complete = true;
149                    break;
150                }
151
152                if offset < data.len() && Self::is_range_tombstone_marker(data[offset]) {
153                    // Issue #1741: when shadowing is active, decode + feed the
154                    // range-tombstone FSM so covered rows are shadowed; otherwise
155                    // (physical read) only advance past the marker.
156                    if let Some(sh) = shadow.as_mut() {
157                        match self.parse_range_tombstone_marker_full(data, offset, schema) {
158                            Ok((bv, bk, dp, ds, next_offset)) => {
159                                // Issue #3721: the marker is FRAMED (`next_offset`
160                                // is bound, so the body continues) and the FSM
161                                // refuses only an unrepresentable bound kind — a
162                                // `break` here reported `Ok` while dropping the
163                                // tombstone and every later row of the partition.
164                                if let Err(e) = sh.feed_range_marker(bv, bk, dp, ds) {
165                                    return Err(range_marker_error::range_marker_refused(
166                                        e,
167                                        &partition_index,
168                                        offset,
169                                        next_offset,
170                                    ));
171                                }
172                                offset = next_offset;
173                                continue;
174                            }
175                            // Issue #3721 (roborev job 78): a marker that
176                            // cannot be PARSED is corruption here, never a
177                            // framing terminator — this block is fully buffered,
178                            // so no refill can complete it, and `break` reported
179                            // `Ok` with the tombstone AND every later row of the
180                            // partition gone. See `range_marker_error`'s docs.
181                            Err(cause) => {
182                                return Err(
183                                    range_marker_error::unparseable_marker_in_buffered_block(
184                                        cause,
185                                        &partition_index,
186                                        offset,
187                                    ),
188                                )
189                            }
190                        }
191                    }
192                    match self.skip_range_tombstone_marker(data, offset, schema) {
193                        Ok(next_offset) => {
194                            offset = next_offset;
195                            continue;
196                        }
197                        // Same decision on the PHYSICAL path (no shadowing).
198                        Err(cause) => {
199                            return Err(range_marker_error::unparseable_marker_in_buffered_block(
200                                cause,
201                                &partition_index,
202                                offset,
203                            ))
204                        }
205                    }
206                }
207
208                match self.parse_row_data_with_offset(
209                    data,
210                    offset,
211                    Some(schema),
212                    reader,
213                    true,
214                    &resolution,
215                    shadow.as_ref(),
216                ) {
217                    Ok((
218                        mut cells,
219                        row_cell_meta_opt,
220                        row_header_opt,
221                        next_offset,
222                        is_static,
223                        _complex_meta,
224                    )) => {
225                        let mut row_cell_meta = row_cell_meta_opt.unwrap_or_default();
226                        offset = next_offset;
227                        row_count += 1;
228
229                        if is_static {
230                            // Issue #1741 (Finding 1): drop a static row's cells/metadata
231                            // when the static row is itself shadowed by the partition
232                            // tombstone or expired by its own TTL, so a surviving
233                            // clustering row does not resurface the stale static value.
234                            // Static rows have no clustering key (empty clustering); no-op
235                            // when shadowing is off.
236                            let static_hidden = shadow.as_ref().is_some_and(|sh| {
237                                row_header_opt
238                                    .as_ref()
239                                    .is_some_and(|h| sh.row_hidden(h, &[]))
240                            });
241                            if static_hidden {
242                                static_cells = Vec::new();
243                                static_cell_meta = HashMap::new();
244                            } else {
245                                static_cells = cells;
246                                static_cell_meta = row_cell_meta;
247                            }
248                        } else {
249                            // Merge static metadata into the clustering row
250                            // (clustering-row-wins; positional, issue #1642). The CELL
251                            // merge happens below, AFTER the row-tombstone decision on
252                            // a read path (issue #3095).
253                            for (k, v) in &static_cell_meta {
254                                row_cell_meta.entry(k.clone()).or_insert_with(|| v.clone());
255                            }
256
257                            // Issue #1741: hide partition/range-tombstone-shadowed and
258                            // TTL-expired rows (matching a Cassandra SELECT). Active
259                            // only for query reads (shadow is `Some`). A row reduced to
260                            // only its primary key by per-cell filtering is hidden here
261                            // too: the dropped cells still fold into the row aggregate,
262                            // so `row_hidden` sees the whole row as shadowed/expired.
263                            let hidden = shadow.as_ref().is_some_and(|sh| {
264                                row_header_opt.as_ref().is_some_and(|h| {
265                                    let clustering = if sh.needs_clustering() {
266                                        extract_clustering_values(&cells, schema)
267                                    } else {
268                                        Vec::new()
269                                    };
270                                    sh.row_hidden(h, &clustering)
271                                })
272                            });
273
274                            // Issue #505/#932 row-tombstone display rule. Issue #3095:
275                            // on a user-facing SELECT read the decision is taken over
276                            // the row's OWN cells FIRST so a static value cannot revive
277                            // a row-tombstoned row; physical consumers keep the
278                            // historical order. See `build_display_row_read_path`.
279                            let row_value = if shadow.is_some() {
280                                build_display_row_read_path(
281                                    cells,
282                                    &static_cells,
283                                    row_header_opt.as_ref(),
284                                    schema,
285                                )
286                            } else {
287                                merge_static_cells(&mut cells, &static_cells);
288                                build_display_row(cells, row_header_opt.as_ref(), schema)
289                            };
290
291                            if !hidden {
292                                // Issue #3095: only a VISIBLE row counts (see
293                                // `row_is_visible`) — a suppressed `ScanRow::Marker`
294                                // must not hide a static-only partition's row.
295                                emitted_clustering_row |= row_is_visible(&row_value);
296                                match emit((
297                                    table_id.clone(),
298                                    partition_key.clone(),
299                                    row_value,
300                                    row_cell_meta,
301                                ))? {
302                                    std::ops::ControlFlow::Continue(()) => {}
303                                    std::ops::ControlFlow::Break(()) => return Ok(()),
304                                }
305                            }
306                        }
307
308                        if offset >= data.len() {
309                            break;
310                        }
311
312                        if self.peek_is_partition_header(data, offset) {
313                            tracing::debug!(
314                                "V5CompressedLegacy: Partition {} detected at offset {} after {} rows",
315                                partition_index + 1, offset, row_count
316                            );
317                            partition_complete = true;
318                            break;
319                        }
320                    }
321                    Err(e) => {
322                        // Issue #3721: `Err` here is normally the end-of-partition
323                        // signal; a per-column decode failure is NOT, and only
324                        // `column_decode_error` decides which is which.
325                        // Issue #3782 composes with it: the tolerance decision belongs
326                        // to the CALLER's declared extent, never to this parse. When the
327                        // buffer is COMPLETE no further bytes can arrive, so the
328                        // discrimination above is authoritative and a column decode
329                        // failure is data loss. When it is INCOMPLETE the same failure is
330                        // the ordinary straddling-row case a refill fixes, so the
331                        // tolerant break stays — measured on #3782: over 42 well-formed
332                        // corpus tables the tolerant path fires 614 times, every one of
333                        // them with an incomplete extent and none with a complete one.
334                        // #3782 FIRST: a proven-complete buffer makes ANY failure data
335                        // loss. Then #3721's discrimination for the incomplete case, so a
336                        // column decode failure still reaches the caller that owns the
337                        // tolerance decision rather than being swallowed here.
338                        if extent.is_complete() {
339                            return Err(e);
340                        }
341                        column_decode_error::end_of_partition_or_bail(
342                            e,
343                            partition_index,
344                            row_count,
345                            offset,
346                        )?;
347                        break;
348                    }
349                }
350            }
351
352            // Issue #3095: Cassandra's static-content-on-an-empty-partition rule
353            // (`SelectStatement.processPartition()`, cassandra-5.0.8 L1099-1120) on
354            // the WRITETIME/TTL-projection decode, so a `SELECT … WRITETIME(s)`
355            // returns the same result SHAPE as a plain `SELECT *`. Same guards and
356            // rationale as the primary site in `block_emit_windowed.rs`: user-facing
357            // SELECT reads only (`read_shadowing`), a CONFIRMED-complete partition,
358            // no clustering row emitted, and a live static row.
359            //
360            // Carries the SAME stated residual as that site: `partition_complete` is
361            // established PER BLOCK, so a static-only partition whose
362            // `END_OF_PARTITION` byte lands in the next decompressed block yields 0
363            // rows instead of 1 — fail-closed, and absent on the sliding-window path,
364            // which has an explicit `at_final_chunk` contract.
365            if self.read_shadowing
366                && partition_complete
367                && !emitted_clustering_row
368                && !static_cells.is_empty()
369            {
370                let row_value = build_display_row(static_cells, None, schema);
371                match emit((
372                    table_id.clone(),
373                    partition_key.clone(),
374                    row_value,
375                    static_cell_meta,
376                ))? {
377                    std::ops::ControlFlow::Continue(()) => {}
378                    std::ops::ControlFlow::Break(()) => return Ok(()),
379                }
380            }
381        }
382
383        Ok(())
384    }

386    /// Streaming variant of [`parse_block`]: invokes `emit` for each parsed
387    /// `(TableId, RowKey, ScanRow)` entry instead of collecting them into a `Vec`,
388    /// so callers can forward rows into a bounded channel without materializing
389    /// the whole block at once (issue #790). Returning `ControlFlow::Break` from
390    /// `emit` stops parsing early — used when the streaming consumer is dropped.
391    pub fn parse_block_emit<F>(
392        &self,
393        data: &[u8],
394        extent: BufferExtent,
395        schema: Option<&TableSchema>,
396        reader: &crate::storage::sstable::reader::types::SSTableReader,
397        emit: F,
398    ) -> Result<()>
399    where
400        F: FnMut((TableId, RowKey, ScanRow)) -> Result<std::ops::ControlFlow<()>>,
401    {
402        // Whole-block decode: no within-partition row-body window narrowing.
403        self.parse_block_emit_windowed(data, extent, schema, reader, None, emit)
404    }
