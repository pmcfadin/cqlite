3impl V5CompressedLegacyParser {
4    /// Map a PRIMITIVE Cassandra marshal type (e.g.
5    /// `org.apache.cassandra.db.marshal.Int32Type`) to the canonical CQL short
6    /// form (`"int"`) understood by [`parse_value_from_raw_bytes`]'s match
7    /// (issue #1081). Returns `None` for any non-primitive marshal form
8    /// (UserType / collection / tuple / reversed / frozen / custom), so the
9    /// caller leaves those to the dedicated arms. The suffix set is a *superset*
10    /// of the authoritative marshal→`CqlType` mapping in
11    /// [`parse_cassandra_type_with_depth`] (no heuristics — issue #28): in
12    /// addition to the scalars that mapping enumerates, this also normalizes a
13    /// few marshal forms that `parse_cassandra_type_with_depth` routes to
14    /// `Custom` (`VarcharType`, `CounterColumnType`, `LexicalUUIDType`,
15    /// `ShortType`, `ByteType`). Those extra mappings are required so we can
16    /// decode the corresponding scalar UDT field values — e.g. `ShortType`/
17    /// `ByteType` are needed to read `smallint`/`tinyint` UDT fields, which
18    /// otherwise fall through to the blob default.
19    fn primitive_marshal_to_cql_short(marshal_type: &str) -> Option<&'static str> {
20        // Composite marshal forms carry a `(` after the type name; primitives do
21        // not. Reject anything parameterised so we never misread a collection /
22        // UDT as a scalar.
23        if marshal_type.contains('(') {
24            return None;
25        }
26        let s = marshal_type;
27        let short = if s.ends_with("UTF8Type") || s.ends_with("VarcharType") {
28            "text"
29        } else if s.ends_with("AsciiType") {
30            "ascii"
31        } else if s.ends_with("Int32Type") {
32            "int"
33        } else if s.ends_with("LongType") || s.ends_with("CounterColumnType") {
34            "bigint"
35        } else if s.ends_with("FloatType") {
36            "float"
37        } else if s.ends_with("DoubleType") {
38            "double"
39        } else if s.ends_with("BooleanType") {
40            "boolean"
41        } else if s.ends_with("TimeUUIDType") {
42            "timeuuid"
43        } else if s.ends_with("UUIDType") || s.ends_with("LexicalUUIDType") {
44            "uuid"
45        } else if s.ends_with("SimpleDateType") {
46            // CQL `date` (`SimpleDateType`) is a 4-byte unsigned days-since-epoch
47            // value. This is distinct from the legacy `DateType` handled below.
48            "date"
49        } else if s.ends_with("DateType") {
50            // Legacy Cassandra `DateType` is an 8-byte millis-since-epoch value —
51            // the same wire format as `TimestampType`. Mapping it to `date` would
52            // wrongly decode only the first 4 bytes, so route it to `timestamp`.
53            // NOTE: this `ends_with` arm must follow the `SimpleDateType` arm above
54            // because `SimpleDateType` also ends with `DateType`.
55            "timestamp"
56        } else if s.ends_with("TimestampType") {
57            "timestamp"
58        } else if s.ends_with("TimeType") {
59            "time"
60        } else if s.ends_with("DecimalType") {
61            "decimal"
62        } else if s.ends_with("IntegerType") {
63            "varint"
64        } else if s.ends_with("DurationType") {
65            "duration"
66        } else if s.ends_with("ShortType") {
67            "smallint"
68        } else if s.ends_with("ByteType") {
69            "tinyint"
70        } else if s.ends_with("InetAddressType") {
71            "inet"
72        } else if s.ends_with("BytesType") {
73            "blob"
74        } else {
75            return None;
76        };
77        Some(short)
78    }

80    /// Parse a value from a complete, bounded byte slice.
81    ///
82    /// This is used when the outer Cassandra collection format already provides
83    /// explicit `[i32 BE len][raw bytes]` boundaries and we have extracted exactly
84    /// the bytes that constitute the value. The entire `data` slice IS the value.
85    ///
86    /// - Variable-width types (text, blob, varint, decimal, inet): consume the full slice
87    /// - Fixed-width types (int, bigint, uuid, etc.): read from offset 0
88    /// - Nested collections: use the bounded sub-format `[i32 BE count][i32 BE len][bytes]...`
89    pub(super) fn parse_value_from_raw_bytes(
90        &self,
91        data: &[u8],
92        type_str: &str,
93        column_name: &str,
94        depth: usize,
95    ) -> Result<Value> {
96        if depth > MAX_TYPE_NESTING_DEPTH {
97            return Err(Error::corruption(format!(
98                "Frozen element '{}': recursion depth {} exceeds maximum {}",
99                column_name, depth, MAX_TYPE_NESTING_DEPTH
100            )));
101        }
102        // Issue #1081: scalar marshal forms (e.g.
103        // `org.apache.cassandra.db.marshal.Int32Type` / `BooleanType`) reach this
104        // function for multicell-UDT field values, which resolve their field
105        // types from the authoritative on-disk `UserType(...)` marshal string.
106        // The match below only enumerates short forms plus a handful of text
107        // marshal aliases, so a bare scalar marshal type would otherwise fall
108        // through to the blob default. Normalize a primitive marshal type to its
109        // canonical CQL short form (via the existing authoritative marshal→CqlType
110        // mapping, no heuristics) and re-dispatch. Composite/UDT marshal forms
111        // (UserType/ListType/MapType/SetType/etc.) are left untouched here — they
112        // are handled by the dedicated arms below — so this only rewrites scalars.
113        if type_str.contains("org.apache.cassandra.db.marshal.") {
114            if let Some(short) = Self::primitive_marshal_to_cql_short(type_str) {
115                return self.parse_value_from_raw_bytes(data, short, column_name, depth);
116            }
117        }
118
119        // Preserve the ORIGINAL-CASE type string. Below, the `match` scrutinee is
120        // `type_str.to_lowercase()` and each `type_str if ...` arm binding SHADOWS
121        // the function parameter with the lowercased string. The collection/tuple/
122        // frozen extraction helpers slice their element/inner types out of the
123        // string they are handed, so if we passed the lowercased binding the nested
124        // element marshal type would come back lowercased (e.g. `...int32type`) and
125        // would NOT re-normalize via the CASE-SENSITIVE `primitive_marshal_to_cql_short`
126        // suffix match, wrongly falling through to blob. The marshal-form arms below
127        // therefore extract from `raw_type_str` (original case) so nested element
128        // marshal types keep their case. The CQL-short-form arms are unaffected
129        // because their inner types are already canonical lowercase.
130        let raw_type_str = type_str;
131        let normalized_type = type_str.to_lowercase();
132        match normalized_type.as_str() {
133            "text"
134            | "varchar"
135            | "ascii"
136            | "org.apache.cassandra.db.marshal.utf8type"
137            | "org.apache.cassandra.db.marshal.asciitype"
138            | "org.apache.cassandra.db.marshal.varchartype" => {
139                let text = String::from_utf8(data.to_vec()).map_err(|e| {
140                    Error::corruption(format!(
141                        "Frozen element '{}': invalid UTF-8 in text value: {}",
142                        column_name, e
143                    ))
144                })?;
145                Ok(Value::Text(text))
146            }
147            "blob" | "bytes" => Ok(Value::Blob(data.to_vec())),
148            "int" => {
149                if data.len() < 4 {
150                    return Err(Error::corruption(format!(
151                        "Frozen element '{}': need 4 bytes for int, got {}",
152                        column_name,
153                        data.len()
154                    )));
155                }
156                Ok(Value::Integer(i32::from_be_bytes([
157                    data[0], data[1], data[2], data[3],
158                ])))
159            }
160            "bigint" | "counter" => {
161                if data.len() < 8 {
162                    return Err(Error::corruption(format!(
163                        "Frozen element '{}': need 8 bytes for bigint, got {}",
164                        column_name,
165                        data.len()
166                    )));
167                }
168                Ok(Value::BigInt(i64::from_be_bytes([
169                    data[0], data[1], data[2], data[3], data[4], data[5], data[6], data[7],
170                ])))
171            }
172            "boolean" => {
173                if data.is_empty() {
174                    return Err(Error::corruption(format!(
175                        "Frozen element '{}': need 1 byte for boolean",
176                        column_name
177                    )));
178                }
179                Ok(Value::Boolean(data[0] != 0))
180            }
181            "uuid" | "timeuuid" => {
182                if data.len() < 16 {
183                    return Err(Error::corruption(format!(
184                        "Frozen element '{}': need 16 bytes for UUID, got {}",
185                        column_name,
186                        data.len()
187                    )));
188                }
189                let uuid: [u8; 16] = data[..16]
190                    .try_into()
191                    .map_err(|_| Error::corruption("UUID byte conversion failed"))?;
192                Ok(Value::Uuid(uuid))
193            }
194            "float" => {
195                if data.len() < 4 {
196                    return Err(Error::corruption(format!(
197                        "Frozen element '{}': need 4 bytes for float, got {}",
198                        column_name,
199                        data.len()
200                    )));
201                }
202                let f = f32::from_be_bytes([data[0], data[1], data[2], data[3]]);
203                Ok(Value::Float(f as f64))
204            }
205            "double" => {
206                if data.len() < 8 {
207                    return Err(Error::corruption(format!(
208                        "Frozen element '{}': need 8 bytes for double, got {}",
209                        column_name,
210                        data.len()
211                    )));
212                }
213                Ok(Value::Float(f64::from_be_bytes([
214                    data[0], data[1], data[2], data[3], data[4], data[5], data[6], data[7],
215                ])))
216            }
217            "smallint" | "short" => {
218                if data.len() < 2 {
219                    return Err(Error::corruption(format!(
220                        "Frozen element '{}': need 2 bytes for smallint, got {}",
221                        column_name,
222                        data.len()
223                    )));
224                }
225                Ok(Value::SmallInt(i16::from_be_bytes([data[0], data[1]])))
226            }
227            "tinyint" | "byte" => {
228                if data.is_empty() {
229                    return Err(Error::corruption(format!(
230                        "Frozen element '{}': need 1 byte for tinyint",
231                        column_name
232                    )));
233                }
234                Ok(Value::TinyInt(data[0] as i8))
235            }
236            "timestamp" => {
237                if data.len() < 8 {
238                    return Err(Error::corruption(format!(
239                        "Frozen element '{}': need 8 bytes for timestamp, got {}",
240                        column_name,
241                        data.len()
242                    )));
243                }
244                Ok(Value::Timestamp(i64::from_be_bytes([
245                    data[0], data[1], data[2], data[3], data[4], data[5], data[6], data[7],
246                ])))
247            }
248            "date" => {
249                if data.len() < 4 {
250                    return Err(Error::corruption(format!(
251                        "Frozen element '{}': need 4 bytes for date, got {}",
252                        column_name,
253                        data.len()
254                    )));
255                }
256                let stored = u32::from_be_bytes([data[0], data[1], data[2], data[3]]);
257                let days_since_epoch = stored.wrapping_add(i32::MIN as u32) as i32;
258                Ok(Value::Date(days_since_epoch))
259            }
260            "time" => {
261                if data.len() < 8 {
262                    return Err(Error::corruption(format!(
263                        "Frozen element '{}': need 8 bytes for time, got {}",
264                        column_name,
265                        data.len()
266                    )));
267                }
268                Ok(Value::Time(i64::from_be_bytes([
269                    data[0], data[1], data[2], data[3], data[4], data[5], data[6], data[7],
270                ])))
271            }
272            "duration" => {
273                // Issue #1081: in this function the entire `data` slice IS the value
274                // (the element/cell length prefix already bounded it) — there is NO
275                // outer `[VInt len]` prefix. Decode three consecutive SIGNED VInts
276                // directly over `data`: months, days, nanos (Cassandra
277                // DurationSerializer). Contrast `parse_raw_type_value`'s duration arm,
278                // which reads an outer `[VInt len]` first because its framing differs.
279                let (remaining, months) = parse_vint(data).map_err(|e| {
280                    Error::corruption(format!(
281                        "Frozen element '{}': failed to parse duration months: {:?}",
282                        column_name, e
283                    ))
284                })?;
285                let pos = data.len() - remaining.len();
286
287                let (remaining, days) = parse_vint(&data[pos..]).map_err(|e| {
288                    Error::corruption(format!(
289                        "Frozen element '{}': failed to parse duration days: {:?}",
290                        column_name, e
291                    ))
292                })?;
293                let pos = data.len() - remaining.len();
294
295                let (_remaining, nanos) = parse_vint(&data[pos..]).map_err(|e| {
296                    Error::corruption(format!(
297                        "Frozen element '{}': failed to parse duration nanos: {:?}",
298                        column_name, e
299                    ))
300                })?;
301
302                // months/days are i32 in Cassandra's DurationType. Reject
303                // (rather than silently truncate via `as i32`) any encoded value
304                // outside the i32 range so a corrupt encoding errors instead of
305                // wrapping (issue #1632, item b).
306                let months = i32::try_from(months).map_err(|_| {
307                    Error::corruption(format!(
308                        "Frozen element '{}': duration months out of i32 range",
309                        column_name
310                    ))
311                })?;
312                let days = i32::try_from(days).map_err(|_| {
313                    Error::corruption(format!(
314                        "Frozen element '{}': duration days out of i32 range",
315                        column_name
316                    ))
317                })?;
318                Ok(Value::Duration {
319                    months,
320                    days,
321                    nanos,
322                })
323            }
324            "varint" => Ok(Value::Varint(data.to_vec())),
325            "decimal" => {
326                if data.len() < 4 {
327                    return Err(Error::corruption(format!(
328                        "Frozen element '{}': decimal too short ({} bytes)",
329                        column_name,
330                        data.len()
331                    )));
332                }
333                let scale = i32::from_be_bytes([data[0], data[1], data[2], data[3]]);
334                let unscaled = data[4..].to_vec();
335                Ok(Value::Decimal { scale, unscaled })
336            }
337            "inet" => Ok(Value::Inet(data.to_vec())),
338            // Nested list/set/map inside a bounded element (e.g. map<text, list<int>>).
339            //
340            // Issue #1081: the guards accept BOTH the CQL short form (`list<...>`)
341            // and the authoritative Cassandra marshal form
342            // (`org.apache.cassandra.db.marshal.ListType(...)`). Multicell-UDT field
343            // values resolve their field types from the on-disk `UserType(...)`
344            // marshal string, so a collection-typed UDT field arrives here in marshal
345            // form and would otherwise fall through to the blob default. The
346            // extraction helpers (`extract_collection_element_type` / `extract_map_types`)
347            // already accept marshal forms; we extract from `raw_type_str`
348            // (original case) so the returned nested element marshal type keeps its
349            // case and re-normalizes correctly on recursion (see note above).
350            type_str
351                if type_str.starts_with("list<")
352                    || type_str.starts_with("org.apache.cassandra.db.marshal.listtype(") =>
353            {
354                let element_type = self.extract_collection_element_type(raw_type_str, "list")?;
355                let (val, _) = self.parse_frozen_list_value_raw(
356                    data,
357                    0,
358                    &element_type,
359                    column_name,
360                    depth + 1,
361                )?;
362                Ok(val)
363            }
364            type_str
365                if type_str.starts_with("set<")
366                    || type_str.starts_with("org.apache.cassandra.db.marshal.settype(") =>
367            {
368                let element_type = self.extract_collection_element_type(raw_type_str, "set")?;
369                let (val, _) = self.parse_frozen_set_value_raw(
370                    data,
371                    0,
372                    &element_type,
373                    column_name,
374                    depth + 1,
375                )?;
376                Ok(val)
377            }
378            type_str
379                if type_str.starts_with("map<")
380                    || type_str.starts_with("org.apache.cassandra.db.marshal.maptype(") =>
381            {
382                let (key_type, value_type) = self.extract_map_types(raw_type_str)?;
383                let (val, _) = self.parse_frozen_map_value_raw(
384                    data,
385                    0,
386                    &key_type,
387                    &value_type,
388                    column_name,
389                    depth + 1,
390                )?;
391                Ok(val)
392            }
393            // Nested tuple inside a frozen collection element.
394            // The caller (read_frozen_element) has already extracted the raw element bytes
395            // into `data`, so there is no outer VUInt length here — just the sequence of
396            // [i32 BE len][bytes] fields as written by serialize_value for Value::Tuple.
397            // Issue #1081: also accept the marshal form `TupleType(...)`, extracting
398            // element types from the original-case `raw_type_str`.
399            type_str
400                if type_str.starts_with("tuple<")
401                    || type_str.starts_with("org.apache.cassandra.db.marshal.tupletype(") =>
402            {
403                let element_types = self.extract_tuple_element_types(raw_type_str)?;
404                if element_types.is_empty() {
405                    return Err(Error::schema(format!(
406                        "Nested tuple element '{}': empty tuple type",
407                        column_name
408                    )));
409                }
410                let mut off = 0usize;
411                let blob_end = data.len();
412                let elements = self.parse_tuple_elements_raw(
413                    data,
414                    &mut off,
415                    blob_end,
416                    &element_types,
417                    column_name,
418                    depth + 1,
419                )?;
420                Ok(Value::Tuple(elements))
421            }
422            // Issue #1081: accept BOTH the CQL short form (`frozen<...>`) and the
423            // authoritative Cassandra marshal form
424            // (`org.apache.cassandra.db.marshal.FrozenType(...)`). Collection/UDT
425            // fields inside a multicell UDT must be frozen, and their field types
426            // resolve from the on-disk `UserType(...)` marshal string where a frozen
427            // field is spelled `FrozenType(...)` — e.g. `frozen<list<int>>` arrives
428            // as `FrozenType(ListType(Int32Type))` and `frozen<some_udt>` as
429            // `FrozenType(UserType(...))`. Without this arm those bypass the frozen
430            // handling and fall through to the blob default. `extract_frozen_inner_type`
431            // accepts both forms; we extract from `raw_type_str` (original case) so the
432            // inner marshal type keeps its case and re-routes to the marshal
433            // collection/UDT/scalar arms above on recursion.
434            type_str
435                if type_str.starts_with("frozen<")
436                    || type_str.starts_with("org.apache.cassandra.db.marshal.frozentype(") =>
437            {
438                let inner_type = self.extract_frozen_inner_type(raw_type_str)?;
439                let inner =
440                    self.parse_value_from_raw_bytes(data, &inner_type, column_name, depth + 1)?;
441                Ok(Value::Frozen(Box::new(inner)))
442            }
443            // UDT (User-Defined Type): delegate to parse_raw_type_value which has the full
444            // UDT parsing logic including field count validation and nested type resolution.
445            // The raw bytes representation is identical between the two function conventions.
446            other if Self::is_udt_type(other) => {
447                let (val, _offset) =
448                    self.parse_raw_type_value(data, 0, type_str, column_name, depth)?;
449                Ok(val)
450            }
451            other => {
452                // Check if it's a short UDT name in the registry (e.g., "address_type").
453                // This handles the case where parse_value_from_raw_bytes is called recursively
454                // from the frozen<> arm with the stripped inner type (e.g., frozen<address_type>
455                // → "address_type"). Since parse_raw_type_value already has a registry-lookup
456                // fallback that correctly handles bare UDT names, we delegate there.
457                // The byte-level encoding is identical: UDT fields use 4-byte i32 length prefixes
458                // with no overall cell-level length prefix, so parse_raw_type_value offset=0 is
459                // correct for already-extracted cell value bytes.
460                // See Issue #481 regression fix.
461                if let Some(ref registry) = self.udt_registry {
462                    if registry.get_udt(&self.keyspace, other).is_some() {
463                        log::debug!(
464                            "parse_value_from_raw_bytes: type '{}' for '{}' resolved as UDT via registry, delegating to parse_raw_type_value",
465                            other,
466                            column_name,
467                        );
468                        let (val, _offset) =
469                            self.parse_raw_type_value(data, 0, type_str, column_name, depth)?;
470                        return Ok(val);
471                    }
472                }
473                // Truly unknown type: fall back to blob.
474                log::debug!(
475                    "parse_value_from_raw_bytes: unknown type '{}' for '{}', treating as blob ({} bytes)",
476                    other,
477                    column_name,
478                    data.len()
479                );
480                Ok(Value::Blob(data.to_vec()))
481            }
482        }
483    }

493    /// Issue #1081: `primitive_marshal_to_cql_short` must normalize every
494    /// PRIMITIVE Cassandra marshal type (fully-qualified) to its canonical CQL
495    /// short form, and must reject any parameterised/composite marshal form
496    /// (anything containing `(` — UDT / collection / reversed) so the
497    /// no-heuristics `(`-rejection guard never misreads a composite as a scalar.
498    #[test]
499    fn primitive_marshal_to_cql_short_maps_scalars_and_rejects_composites() {
500        const P: &str = "org.apache.cassandra.db.marshal.";
501
502        // (marshal type name, expected canonical CQL short form)
503        let cases: &[(&str, &str)] = &[
504            ("UTF8Type", "text"),
505            ("AsciiType", "ascii"),
506            ("Int32Type", "int"),
507            ("LongType", "bigint"),
508            ("FloatType", "float"),
509            ("DoubleType", "double"),
510            ("BooleanType", "boolean"),
511            ("UUIDType", "uuid"),
512            ("TimeUUIDType", "timeuuid"),
513            ("TimestampType", "timestamp"),
514            ("SimpleDateType", "date"),
515            // Legacy `DateType` is an 8-byte millis-since-epoch value (same wire
516            // format as `TimestampType`), NOT the 4-byte CQL `date`
517            // (`SimpleDateType`). It must normalize to `timestamp`.
518            ("DateType", "timestamp"),
519            ("TimeType", "time"),
520            ("DecimalType", "decimal"),
521            ("IntegerType", "varint"),
522            ("DurationType", "duration"),
523            ("ShortType", "smallint"),
524            ("ByteType", "tinyint"),
525            ("InetAddressType", "inet"),
526            ("BytesType", "blob"),
527        ];
528
529        for (marshal, expected) in cases {
530            let full = format!("{}{}", P, marshal);
531            assert_eq!(
532                V5CompressedLegacyParser::primitive_marshal_to_cql_short(&full),
533                Some(*expected),
534                "primitive marshal {} should map to {}",
535                full,
536                expected
537            );
538        }
539
540        // Parameterised / composite marshal forms must be rejected (return None)
541        // by the `(`-guard, leaving them to the dedicated composite arms.
542        let composites = [
543            "org.apache.cassandra.db.marshal.UserType(...)",
544            "org.apache.cassandra.db.marshal.ListType(org.apache.cassandra.db.marshal.UTF8Type)",
545            "org.apache.cassandra.db.marshal.ReversedType(org.apache.cassandra.db.marshal.Int32Type)",
546        ];
547        for composite in composites {
548            assert_eq!(
549                V5CompressedLegacyParser::primitive_marshal_to_cql_short(composite),
550                None,
551                "composite marshal {} must be rejected by the `(`-guard",
552                composite
553            );
554        }
555    }

557    #[test]
558    fn test_parse_value_from_raw_bytes_primitives() {
559        let parser =
560            V5CompressedLegacyParser::new("test".to_string(), "table".to_string(), 0, 0, None);
561
562        // int
563        let data = 42i32.to_be_bytes();
564        let val = parser
565            .parse_value_from_raw_bytes(&data, "int", "col", 0)
566            .unwrap();
567        assert_eq!(val, Value::Integer(42));

676    /// Issue #1081: a multicell-UDT field declared `duration` resolves its
677    /// field type from the authoritative on-disk `UserType(...)` marshal string
678    /// as `org.apache.cassandra.db.marshal.DurationType`. That bare scalar
679    /// marshal form must normalize to `"duration"` and decode the three
680    /// consecutive SIGNED VInts (months, days, nanos) that constitute the value
681    /// — NOT fall through to `Value::Blob`. In `parse_value_from_raw_bytes` the
682    /// entire slice IS the value (no outer `[VInt len]` prefix).
683    #[test]
684    fn test_parse_value_from_raw_bytes_duration_marshal() {
685        let parser =
686            V5CompressedLegacyParser::new("test".to_string(), "table".to_string(), 0, 0, None);
687
688        // Build three SIGNED VInts (months=1, days=2, nanos=3) using the same
689        // ZigZag-over-unsigned-VInt scheme `parse_vint` decodes. ZigZag:
690        // 1 -> 2, 2 -> 4, 3 -> 6.
691        let zigzag = |v: i64| ((v << 1) ^ (v >> 63)) as u64;
692        let mut data = Vec::new();
693        encode_unsigned(zigzag(1), &mut data); // months
694        encode_unsigned(zigzag(2), &mut data); // days
695        encode_unsigned(zigzag(3), &mut data); // nanos
696
697        // Round-trip via the signed VInt parser to confirm the encoding before
698        // exercising the decode arm under test.
699        let (rem, m) = parse_vint(&data).unwrap();
700        let consumed = data.len() - rem.len();
701        let (rem2, d) = parse_vint(&data[consumed..]).unwrap();
702        let consumed2 = data.len() - rem2.len();
703        let (_rem3, n) = parse_vint(&data[consumed2..]).unwrap();
704        assert_eq!(
705            (m, d, n),
706            (1, 2, 3),
707            "hand-encoded duration vints round-trip"
708        );
709
710        // The bare scalar marshal form must normalize and decode to Duration,
711        // NOT fall through to Blob.
712        let val = parser
713            .parse_value_from_raw_bytes(
714                &data,
715                "org.apache.cassandra.db.marshal.DurationType",
716                "col",
717                0,
718            )
719            .unwrap();
720        assert_eq!(
721            val,
722            Value::Duration {
723                months: 1,
724                days: 2,
725                nanos: 3
726            }
727        );
728
729        // The canonical CQL short form must decode identically.
730        let val_short = parser
731            .parse_value_from_raw_bytes(&data, "duration", "col", 0)
732            .unwrap();
733        assert_eq!(val_short, val);
734    }
