3/// `DeletionTime.Serializer` LIVE sentinel for oa/da (`hasUIntDeletionTime`)
4/// partitions: a single byte `0x80` (`IS_LIVE_DELETION = 0b1000_0000`) encodes a
5/// live partition; any other leading byte introduces the full 12-byte deleted
6/// form (8-byte `markedForDeleteAt` + 4-byte unsigned `localDeletionTime`).
7/// Authority: `DeletionTime.java:208` (`Serializer.serialize`).
8pub(super) const OA_IS_LIVE_DELETION: u8 = 0x80;
9
10/// Whether the leading partition header in a sliding-window chunk buffer can be
11/// safely parsed yet. Computed WITHOUT consuming bytes so the two sliding
12/// parsers ([`V5CompressedLegacyParser::parse_one_partition_with_timestamps`] and
13/// [`V5CompressedLegacyParser::parse_one_partition_for_compaction`]) agree on the
14/// need-more decision for both the nb (fixed 12-byte) and oa/da (1-byte LIVE /
15/// 12-byte DELETED) `DeletionTime` forms.
16#[derive(Debug, Clone, Copy, PartialEq, Eq)]
17pub(super) enum PartitionHeaderReadiness {
18    /// Enough bytes are present to parse the full header, INCLUDING the complete
19    /// partition `DeletionTime` for its live/deleted form. Parsing cannot fail
20    /// due to truncation.
21    Ready,
22    /// The header — or, for an oa/da deleted partition, its 12-byte
23    /// `DeletionTime` — is split across the chunk boundary. More bytes must be
24    /// appended before it can be parsed (non-final chunk → `NeedMore`).
25    Incomplete,
26    /// The header shape is invalid (zero or over-long key length): the caller
27    /// should skip one byte to resynchronise.
28    Malformed,
29}

31/// Result of the non-allocating partition-BOUNDARY peek (issue #1641, K2).
32///
33/// The post-row emit loop asks "does the next thing begin a new partition
34/// header?" after every row. On `main` that ran the FULL allocating
35/// [`V5CompressedLegacyParser::parse_partition_header_full`] as a trial (throwaway
36/// key `to_vec` + eager `format!` error strings + a `PARTITION_HEADER_TRY_PARSES`
37/// increment) purely to learn a boolean. This enum is returned by
38/// [`V5CompressedLegacyParser::peek_partition_boundary`], which reaches the same
39/// verdict by READING bytes — no allocation, no error strings, no gauge.
40#[derive(Debug, Clone, Copy, PartialEq, Eq)]
41pub(super) enum BoundaryPeek {
42    /// The bytes at the offset are structurally a partition header — i.e. the
43    /// full parser would return `Ok` here. The caller runs the real
44    /// (allocating) parse exactly once, at this confirmed start.
45    Header,
46    /// Definitely NOT a partition header at this offset: the leading byte is an
47    /// END_OF_PARTITION / range-tombstone marker, the key length is zero, or the
48    /// header is structurally invalid (e.g. an illegal oa/da IS_LIVE byte) so the
49    /// full parser would return `Err` on the complete bytes.
50    NotHeader,
51    /// The header — or, for an oa/da deleted partition, its full `DeletionTime` —
52    /// is not entirely present yet (truncated / split across a chunk boundary).
53    /// The caller should request more bytes rather than decide.
54    NeedMoreBytes,
55}

57/// Byte-offset layout of a structurally-valid partition header, produced WITHOUT
58/// allocating the key (issue #1641, K2). Shared by the allocating full parser and
59/// the non-allocating boundary peek so their structural rules cannot drift.
60struct PartitionHeaderLayout {
61    /// Range of the partition key bytes within the input buffer (not copied).
62    key_range: std::ops::Range<usize>,
63    /// Offset immediately after the header (start of the first row / marker).
64    next_offset: usize,
65    /// Partition-level deletion `(markedForDeleteAt µs, localDeletionTime s)`, or
66    /// `None` for a live partition. Same contract as
67    /// [`V5CompressedLegacyParser::parse_partition_header_full`].
68    partition_deletion: Option<(i64, i32)>,
69}

72    /// Decide whether the leading partition header in `data` is fully present.
73    ///
74    /// Issue #1741 (roborev HIGH): the oa/da (`hasUIntDeletionTime`) partition
75    /// `DeletionTime` is 1 byte only when LIVE (the [`OA_IS_LIVE_DELETION`]
76    /// sentinel); a DELETED partition carries the full 12-byte form. Sizing the
77    /// header minimum at a flat 1 byte let a deleted partition header that was
78    /// split across a NON-FINAL chunk (partition key + only the first deletion
79    /// byte present) pass the guard, so `parse_partition_header_full` failed
80    /// mid-buffer and the sliding parser returned `Emitted(1)` instead of
81    /// `NeedMore` — desyncing the scan and, on the compaction path, dropping the
82    /// partition tombstone (data-resurrection risk).
83    ///
84    /// This peeks the deletion-time discriminator to size the header correctly
85    /// and, when a `Ready` verdict is returned, guarantees the subsequent
86    /// `parse_partition_header_full` has every byte it needs — so a truncated
87    /// deletion-time can never be mis-parsed as a complete partition. No
88    /// heuristics: the branch keys off the authoritative `hasUIntDeletionTime`
89    /// version-gate flag and the canonical `DeletionTime` sentinel.
90    pub(super) fn partition_header_readiness(&self, data: &[u8]) -> PartitionHeaderReadiness {
91        // Cassandra partition key size limits (see the block loop guards).
92        const CASSANDRA_MAX_KEY_SIZE: usize = 65536; // 64KB per Cassandra spec
93        const FORMAT_MAX_KEY_SIZE: usize = 255; // u8 length field limit
94
95        // flags(1) + key_len(1) must both be present before anything else.
96        if data.len() < 2 {
97            return PartitionHeaderReadiness::Incomplete;
98        }
99        let key_len = data[1] as usize;
100        if key_len == 0 || key_len > FORMAT_MAX_KEY_SIZE.min(CASSANDRA_MAX_KEY_SIZE) {
101            return PartitionHeaderReadiness::Malformed;
102        }
103
104        // Offset of the DeletionTime, immediately after flags + key_len + key.
105        let deletion_offset = 2 + key_len;
106        let deletion_time_min = if self.has_uint_deletion_time() {
107            // oa/da: peek the DeletionTime discriminator to size the header.
108            match data.get(deletion_offset) {
109                // Discriminator byte itself is not present yet — split header.
110                None => return PartitionHeaderReadiness::Incomplete,
111                // LIVE sentinel (0x80): exactly 1 byte.
112                Some(&b) if (b & OA_IS_LIVE_DELETION) != 0 => 1,
113                // Any other leading byte introduces the full 12-byte deleted form.
114                Some(_) => 12,
115            }
116        } else {
117            // nb: fixed 12-byte signed DeletionTime (4-byte LDT + 8-byte MFDA).
118            12
119        };
120
121        if deletion_offset + deletion_time_min > data.len() {
122            PartitionHeaderReadiness::Incomplete
123        } else {
124            PartitionHeaderReadiness::Ready
125        }
126    }

176    /// Check if row flags indicate end of partition (Issue #229 fix)
177    ///
178    /// Per Cassandra's UnfilteredSerializer.java, END_OF_PARTITION is written as exactly 0x01
179    /// by the `writeEndOfPartition()` method. The Cassandra source uses a bitmask check
180    /// `(flags & END_OF_PARTITION) != 0`, but in practice the marker is always 0x01.
181    ///
182    /// We use an EXACT match to 0x01 to avoid false positives with row data that
183    /// incidentally has bit 0 set (e.g., 0xb7 which would wrongly match a bitmask check).
184    ///
185    /// When END_OF_PARTITION (0x01) is detected, nothing follows the flags byte.
186    /// The partition is complete and parsing should move to the next partition.
187    #[inline]
188    pub(super) fn is_end_of_partition(flags: u8) -> bool {
189        flags == END_OF_PARTITION // Exact match, not bitmask
190    }

192    /// Check if row flags indicate a range tombstone marker (not a data row)
193    ///
194    /// Per Cassandra's UnfilteredSerializer.java, IS_MARKER (0x02) indicates a range
195    /// tombstone boundary. The marker flag can be combined with other metadata flags
196    /// (e.g., 0x52 = IS_MARKER | deletion metadata, 0x7a, 0x36, etc.).
197    ///
198    /// Issue #258 fix: Use bitwise AND to detect markers with additional flags.
199    /// Previously used exact match (flags == 0x02) which missed markers like 0x52.
200    #[inline]
201    pub(super) fn is_range_tombstone_marker(flags: u8) -> bool {
202        // Check if IS_MARKER bit is set, but END_OF_PARTITION bit is NOT set
203        // IS_MARKER = 0x02, END_OF_PARTITION = 0x01
204        // If END_OF_PARTITION bit is set (even with other bits), it's end of partition, not a marker
205        (flags & IS_MARKER) != 0 && (flags & END_OF_PARTITION) == 0
206    }

208    /// Skip a range tombstone marker body (Issue #229 fix, VG6 fix)
209    ///
210    /// Range tombstone markers for SSTable format have this on-disk layout:
211    ///   [flags: u8]                        ← IS_MARKER (0x02) bit set
212    ///   [extended_flags: u8]               ← only if ROW_HAS_EXTENDED_FLAGS set
213    ///   [bound_kind: u8]                   ← ordinal of ClusteringBoundOrBoundary.Kind
214    ///   [cluster_count: u16 big-endian]    ← number of clustering values (bound.size())
215    ///   [cluster_header: VUInt]            ← 2 bits per value (0=present, 1=empty, 2=null)
216    ///   [cluster_values: ...]              ← type-specific bytes for non-null/non-empty values
217    ///   [marker_body_size: VUInt]          ← size of the body that follows (including prev_size)
218    ///   [prev_unfiltered_size: VUInt]      ← size of the previous unfiltered item
219    ///   [marked_for_delete_at: VUInt]      ← timestamp delta from min_timestamp (µs)
220    ///   [local_deletion_time: VUInt32]     ← seconds delta from min_local_deletion_time
221    ///   [marked_for_delete_at2: VUInt]     ← ONLY for boundaries (kind 2 or 5)
222    ///   [local_deletion_time2: VUInt32]    ← ONLY for boundaries (kind 2 or 5)
223    ///
224    /// Authority:
225    ///   UnfilteredSerializer.java:282-303  (serialize(RangeTombstoneMarker, ...))
226    ///   ClusteringBoundOrBoundary.Serializer.serialize (lines 103-107):
227    ///     out.writeByte(bound.kind().ordinal())   ← kind byte
228    ///     out.writeShort(bound.size())            ← u16 cluster count
229    ///     ClusteringPrefix.serializer.serializeValuesWithoutSize(...)
230    ///   SerializationHeader.writeDeletionTime (lines 180-183):
231    ///     writeTimestamp → writeUnsignedVInt      ← VUInt, NOT ZigZag
232    ///     writeLocalDeletionTime → writeUnsignedVInt32 ← VUInt, NOT ZigZag
233    ///
234    /// VG6 fix: The previous implementation had three bugs:
235    ///   1. After kind byte: did not read the u16 cluster_count before the VUInt header.
236    ///      The 2-byte short was being consumed as part of the clustering values, causing
237    ///      all deletion-time bytes to be misaligned.
238    ///   2. After clustering values: did not skip marker_body_size + prev_unfiltered_size
239    ///      VUInts that precede the deletion times in SSTable format.
240    ///   3. Used parse_vint (ZigZag) instead of parse_vuint (unsigned) for deletion times.
241    ///
242    /// Implementation strategy: use marker_body_size to skip the entire body
243    /// (prev_size + deletion times) without manually decoding individual fields.
244    pub(super) fn skip_range_tombstone_marker(
245        &self,
246        data: &[u8],
247        offset: usize,
248        schema: &TableSchema,
249    ) -> Result<usize> {
250        let mut pos = offset;
251
252        if pos >= data.len() {
253            return Err(Error::corruption(
254                "V5CompressedLegacy: Unexpected end at range tombstone marker",
255            ));
256        }
257
258        let marker_flags = data[pos];
259        pos += 1; // Skip flags byte
260
261        tracing::debug!(
262            "V5CompressedLegacy: Skipping range tombstone marker with flags=0x{:02x} at offset {}",
263            marker_flags,
264            offset
265        );
266
267        // Extended flags if present (unlikely for markers, but handle it)
268        if (marker_flags & ROW_HAS_EXTENDED_FLAGS) != 0 {
269            if pos >= data.len() {
270                return Err(Error::corruption(
271                    "V5CompressedLegacy: Unexpected end reading marker extended flags",
272                ));
273            }
274            pos += 1;
275        }
276
277        // Read bound kind byte.
278        // Authority: ClusteringBoundOrBoundary.Serializer.serialize (line 104):
279        //   out.writeByte(bound.kind().ordinal())
280        //
281        // Kind ordinals (ClusteringPrefix.java:67-81):
282        //   0 = EXCL_END_BOUND (simple, 1 deletion time)
283        //   1 = INCL_START_BOUND (simple, 1 deletion time)
284        //   2 = EXCL_END_INCL_START_BOUNDARY (boundary, 2 deletion times)
285        //   5 = INCL_END_EXCL_START_BOUNDARY (boundary, 2 deletion times)
286        //   6 = INCL_END_BOUND (simple, 1 deletion time)
287        //   7 = EXCL_START_BOUND (simple, 1 deletion time)
288        if pos >= data.len() {
289            return Err(Error::corruption(
290                "V5CompressedLegacy: Unexpected end reading range tombstone bound kind",
291            ));
292        }
293        let bound_kind = data[pos];
294        pos += 1;
295
296        tracing::debug!(
297            "V5CompressedLegacy: Range tombstone bound_kind={}",
298            bound_kind,
299        );
300
301        // Read cluster count (u16 big-endian).
302        // Authority: ClusteringBoundOrBoundary.Serializer.serialize (line 105):
303        //   out.writeShort(bound.size())
304        //
305        // This is the number of clustering values in the bound. It is NOT the same as
306        // schema.clustering_keys.len() — for regular rows, no count is written; for markers,
307        // a u16 is always present. Failing to read this u16 causes all subsequent bytes to
308        // be misaligned (the two count bytes get consumed as the VUInt header + first value
309        // byte, producing garbage alignment).
310        if pos + 2 > data.len() {
311            return Err(Error::corruption(
312                "V5CompressedLegacy: Unexpected end reading range tombstone cluster count (u16)",
313            ));
314        }
315        let cluster_count = u16::from_be_bytes([data[pos], data[pos + 1]]) as usize;
316        pos += 2;
317
318        tracing::debug!(
319            "V5CompressedLegacy: Range tombstone cluster_count={}",
320            cluster_count,
321        );
322
323        // Read the clustering VUInt header + clustering values.
324        // Authority: ClusteringPrefix.Serializer.serializeValuesWithoutSize (lines 455-477):
325        //   Writes VUInt header (2 bits per value, 0=present/1=empty/2=null), then value bytes.
326        //
327        // Use a truncated schema when cluster_count < schema.clustering_keys.len() to avoid
328        // reading past the bound's bytes into the marker body (prefix bound case).
329        if cluster_count > 0 {
330            let prefix_schema_owned = Self::clustering_prefix_schema(schema, cluster_count);
331            let effective_schema = prefix_schema_owned.as_ref().unwrap_or(schema);
332            let (_, new_pos) = self.parse_clustering_prefix(data, pos, effective_schema)?;
333            pos = new_pos;
334        }
335
336        // Read marker_body_size and skip the body.
337        // Authority: UnfilteredSerializer.java:291 (for SSTable format):
338        //   out.writeUnsignedVInt(serializedMarkerBodySize(marker, header, previousUnfilteredSize, version))
339        //   out.writeUnsignedVInt(previousUnfilteredSize)
340        //   ... deletion time(s) ...
341        //
342        // serializedMarkerBodySize() returns the size of (prev_size + deletion_times).
343        // So after reading marker_body_size, we can skip exactly that many bytes to reach
344        // the next unfiltered item, without needing to decode individual deletion time VUInts.
345        //
346        // This is exactly the same pattern as regular row_size: after reading row_size,
347        // skip row_size bytes to reach the next row/marker.
348        let (remaining, marker_body_size) = parse_vuint(&data[pos..]).map_err(|e| {
349            Error::corruption(format!(
350                "V5CompressedLegacy: Failed to parse marker_body_size at offset {}: {:?}",
351                pos, e
352            ))
353        })?;
354        let body_size_vint_len = data[pos..].len() - remaining.len();
355        pos += body_size_vint_len;
356
357        // Skip marker_body_size bytes (prev_size + deletion time(s))
358        let body_end = pos + marker_body_size as usize;
359        if body_end > data.len() {
360            return Err(Error::corruption(format!(
361                "V5CompressedLegacy: marker_body_size={} at pos={} exceeds data length {}",
362                marker_body_size,
363                pos,
364                data.len()
365            )));
366        }
367        pos = body_end;
368
369        tracing::debug!(
370            "V5CompressedLegacy: Skipped range tombstone marker, advanced from {} to {}",
371            offset,
372            pos
373        );
374
375        Ok(pos)
376    }

378    /// Parse row metadata AFTER flags and clustering prefix (Issue #213 fix)
379    ///
380    /// # Corrected Format (from Cassandra UnfilteredSerializer.java)
381    /// ```text
382    /// [row_flags: u8]           ← Parsed by parse_row_flags()
383    /// [extended_flags: u8]      ← Parsed by parse_row_flags()
384    /// [clustering_prefix]       ← Parsed by parse_clustering_prefix()
385    /// [row_size: VInt]          ← This function starts here
386    /// [prev_size: VInt]
387    /// [timestamp: VInt if 0x04 set] ← Delta from min_timestamp
388    /// [ttl: VInt if 0x08 set] ← Delta from min_ttl
389    /// [deletion: 2 VInts if 0x10 set]
390    /// [column_bitmap: VUInt bitmask of missing columns if NOT 0x20]
391    /// ```
392    ///
393    /// Returns RowHeader with decoded metadata, calculated header_size, and row_size.
394    pub(super) fn parse_row_metadata(
395        &self,
396        data: &[u8],
397        offset: usize,
398        row_flags: u8,
399        _extended_flags: Option<u8>,
400    ) -> Result<(RowHeader, u64)> {
401        let mut pos = offset;
402
403        // V5CompressedLegacy format: row_size and prev_size come AFTER clustering
404        // (which has already been parsed before this function is called)
405
406        // Read row size (VInt) - CRITICAL for partition boundary detection!
407        debug!(
408            "V5CompressedLegacy: Parsing row_size VInt at pos={}, hex={:02x?}",
409            pos,
410            &data[pos..std::cmp::min(pos + 5, data.len())]
411        );
412        let (remaining, row_size) = parse_vuint(&data[pos..]).map_err(|e| {
413            Error::corruption(format!(
414                "V5CompressedLegacy: Failed to parse row size at offset {}: {:?}",
415                pos, e
416            ))
417        })?;
418        let row_size_vint_len = data[pos..].len() - remaining.len();
419        debug!(
420            "V5CompressedLegacy: row_size={}, consumed {} bytes, pos before={}, pos after={}",
421            row_size,
422            row_size_vint_len,
423            pos,
424            pos + row_size_vint_len
425        );
426        pos += row_size_vint_len;
427
428        // Read prev size (VInt)
429        debug!(
430            "V5CompressedLegacy: Parsing prev_size VInt at pos={}, hex={:02x?}",
431            pos,
432            &data[pos..std::cmp::min(pos + 5, data.len())]
433        );
434        let (remaining, _prev_size) = parse_vuint(&data[pos..]).map_err(|e| {
435            Error::corruption(format!(
436                "V5CompressedLegacy: Failed to parse prev size at offset {}: {:?}",
437                pos, e
438            ))
439        })?;
440        let bytes_consumed = data[pos..].len() - remaining.len();
441        debug!(
442            "V5CompressedLegacy: prev_size={}, consumed {} bytes, pos before={}, pos after={}",
443            _prev_size,
444            bytes_consumed,
445            pos,
446            pos + bytes_consumed
447        );
448        pos += bytes_consumed;
449
450        // Read timestamp if HAS_TIMESTAMP flag is set.
451        //
452        // Fix #629 (C2): Cassandra writes an UNSIGNED VInt delta here
453        // (SerializationHeader.java:165: out.writeUnsignedVInt(timestamp - stats.minTimestamp)).
454        // The old code used parse_vint (ZigZag), causing ~50% undercount of timestamp deltas.
455        let timestamp = if (row_flags & ROW_HAS_TIMESTAMP) != 0 {
456            let (remaining, delta) = parse_vuint(&data[pos..]).map_err(|e| {
457                Error::corruption(format!(
458                    "V5CompressedLegacy: Failed to parse timestamp delta at offset {}: {:?}",
459                    pos, e
460                ))
461            })?;
462            let bytes_consumed = data[pos..].len() - remaining.len();
463            pos += bytes_consumed;
464
465            // Apply delta decoding: absolute_timestamp = min_timestamp + delta
466            let absolute_timestamp = self.min_timestamp.wrapping_add(delta as i64);
467            debug!(
468                "V5CompressedLegacy: Row timestamp: delta={}, min={}, absolute={}",
469                delta, self.min_timestamp, absolute_timestamp
470            );
471            Some(absolute_timestamp)
472        } else {
473            None
474        };
475
476        // Read TTL and liveness local expiration time if HAS_TTL flag is set.
477        //
478        // Fix #630 (C3): Cassandra writes TWO VInt32 fields when HAS_TTL is set
479        // (UnfilteredSerializer.java:225-228):
480        //   1. pk_liveness.ttl()               → header.writeTTL(ttl, out)        [VInt32]
481        //   2. pk_liveness.localExpirationTime()→ header.writeLocalDeletionTime(ldt, out) [VInt32]
482        // The old code read only ONE VInt (TTL), leaving the LDT byte(s) unread and
483        // misaligning all subsequent fields in HAS_TTL rows.
484        let (ttl, ttl_liveness_ldt) = if (row_flags & ROW_HAS_TTL) != 0 {
485            let (remaining, ttl_delta) = parse_vuint(&data[pos..]).map_err(|e| {
486                Error::corruption(format!(
487                    "V5CompressedLegacy: Failed to parse TTL delta at offset {}: {:?}",
488                    pos, e
489                ))
490            })?;
491            let bytes_consumed = data[pos..].len() - remaining.len();
492            pos += bytes_consumed;
493
494            // Apply delta decoding: absolute_ttl = min_ttl + delta
495            let absolute_ttl = if let Some(min_ttl) = self.min_ttl {
496                min_ttl.wrapping_add(ttl_delta as i64) as i32
497            } else {
498                ttl_delta as i32
499            };
500
501            // Read liveness local expiration time (second mandatory field after TTL).
502            let (remaining, ldt_delta) = parse_vuint(&data[pos..]).map_err(|e| {
503                Error::corruption(format!(
504                    "V5CompressedLegacy: Failed to parse liveness LDT delta at offset {}: {:?}",
505                    pos, e
506                ))
507            })?;
508            let ldt_bytes_consumed = data[pos..].len() - remaining.len();
509            pos += ldt_bytes_consumed;
510
511            // Liveness localExpirationTime (SECONDS). Same VG3 unsigned-deletion-time
512            // reinterpretation as the row/complex-cell deletion LDT reader below
513            // (BigFormat.java:409, hasUIntDeletionTime): on oa/da a post-2038 expiry
514            // occupies [2^31, 2^32) and must be read as UNSIGNED (year-2106-safe) so
515            // it stays a large POSITIVE second count. Storing it as `i64` here (vs the
516            // old `as i32`) prevents the wrap-to-negative that would make the read-time
517            // TTL filter treat a still-live row as long-expired and hide it (#1741 F1).
518            // nb (signed) values are capped at ~2038 so sign-extension is a no-op.
519            let raw_liveness_ldt = self.min_local_deletion_time.wrapping_add(ldt_delta as i64);
520            let absolute_ldt: i64 = if self.has_uint_deletion_time() {
521                (raw_liveness_ldt as u32) as i64
522            } else {
523                raw_liveness_ldt as i32 as i64
524            };
525
526            debug!(
527                "V5CompressedLegacy: Row TTL: ttl_delta={}, min={:?}, ttl={}, ldt_delta={}, ldt={}",
528                ttl_delta, self.min_ttl, absolute_ttl, ldt_delta, absolute_ldt
529            );
530            (Some(absolute_ttl), Some(absolute_ldt))
531        } else {
532            (None, None)
533        };
534
535        // Read deletion if HAS_DELETION flag is set.
536        //
537        // Cassandra canonical DeletionTime.Serializer order (matches the CQLite writer,
538        // data_writer.rs write_*_row HAS_DELETION block and write_complex_deletion):
539        //   1. markedForDeleteAt: UNSIGNED VInt delta, base min_timestamp, MICROSECONDS
540        //      -> the authoritative reconciliation timestamp (LWW shadowing).
541        //   2. localDeletionTime: UNSIGNED VInt delta, base min_local_deletion_time, SECONDS
542        //      -> the GC-grace clock, NOT a reconciliation timestamp.
543        //
544        // Fix #629 (C2): Both deltas are UNSIGNED per Cassandra SerializationHeader.java.
545        // The old code used parse_vint (ZigZag) for markedForDeleteAt, causing ~50% undercount.
546        //
547        // (The complex-cell deletion reader, parse_complex_column, already uses this
548        // markedForDeleteAt-first order; this aligns the row-level header with it.)
549        let (marked_for_delete_at, local_deletion_time) = if (row_flags & ROW_HAS_DELETION) != 0 {
550            // First VInt: markedForDeleteAt delta (unsigned).
551            let (remaining, mfda_delta) = parse_vuint(&data[pos..]).map_err(|e| {
552                Error::corruption(format!(
553                    "V5CompressedLegacy: Failed to parse markedForDeleteAt delta at offset {}: {:?}",
554                    pos, e
555                ))
556            })?;
557            let bytes_consumed = data[pos..].len() - remaining.len();
558            pos += bytes_consumed;
559
560            // Second VInt: localDeletionTime delta (unsigned VInt32).
561            //
562            // For both nb and oa the on-disk format is the same: an unsigned VInt32
563            // encoding `(int)(localDeletionTime - stats.minLocalDeletionTime)`.
564            // See: SerializationHeader.java — `writeLocalDeletionTime` /
565            //      `readLocalDeletionTime` (same for all BIG versions).
566            //
567            // VG3 gate: hasUIntDeletionTime (BigFormat.java:409, oa+)
568            // The interpretation of the *result* differs:
569            //   nb: `min_local_deletion_time + delta` cast to i32 (values capped at ~year 2038)
570            //   oa: `min_local_deletion_time + delta` treated as u32 to support ~year 2106
571            //
572            // When the sum overflows an i32 (> 2^31-1 seconds) the value is negative
573            // in a signed context; with hasUIntDeletionTime we reinterpret it as an
574            // unsigned u32 (CassandraUInt.toLong, CassandraUInt.java).  For current
575            // test fixtures all deletion times are well within i32 range so both
576            // interpretations produce identical bit patterns; the gate is a no-op
577            // in practice but is wired correctly for future large TTL values.
578            let (remaining, ldt_delta) = parse_vuint(&data[pos..]).map_err(|e| {
579                Error::corruption(format!(
580                    "V5CompressedLegacy: Failed to parse localDeletionTime delta at offset {}: {:?}",
581                    pos, e
582                ))
583            })?;
584            let bytes_consumed = data[pos..].len() - remaining.len();
585            pos += bytes_consumed;
586
587            // markedForDeleteAt: absolute = min_timestamp + delta (microseconds).
588            let absolute_marked_for_delete_at = self.min_timestamp.wrapping_add(mfda_delta as i64);
589            // localDeletionTime: absolute = min_local_deletion_time + delta (seconds).
590            //
591            // VG3 gate: hasUIntDeletionTime (BigFormat.java:409, oa/da)
592            //   nb: store as i32 (may overflow for dates > ~year 2038)
593            //   oa: reinterpret the 32-bit bit-pattern as unsigned (supports ~year 2106)
594            //
595            // Source: SerializationHeader.java readLocalDeletionTime + UnfilteredSerializer.java:671-676
596            // "if (complexDeletion.localDeletionTime() < 0) {
597            //    complexDeletion = DeletionTime.build(..., Cell.deletionTimeUnsignedIntegerToLong((int) ...));
598            //  }" — this reinterpretation fires when hasUIntDeletionTime && bit31 set.
599            let has_uint_ldt = match self.version_gates.as_ref() {
600                crate::storage::sstable::version_gate::VersionGates::Big(g) => {
601                    g.has_uint_deletion_time
602                }
603                crate::storage::sstable::version_gate::VersionGates::Bti(g) => {
604                    g.has_uint_deletion_time
605                }
606            };
607            let raw_ldt = self.min_local_deletion_time.wrapping_add(ldt_delta as i64);
608            let absolute_local_deletion_time = if has_uint_ldt {
609                // Reinterpret the low 32 bits as an unsigned integer (year-2106-safe).
610                // CassandraUInt.toLong(int) = Integer.toUnsignedLong(int), so negative
611                // i32 values get promoted to the [2^31, 2^32) long range.
612                (raw_ldt as u32) as i32
613            } else {
614                raw_ldt as i32
615            };
616            debug!(
617                "V5CompressedLegacy: Row deletion: markedForDeleteAt(delta={}, min_ts={}, abs={} us), localDeletionTime(delta={}, min_ldt={}, abs={} s)",
618                mfda_delta,
619                self.min_timestamp,
620                absolute_marked_for_delete_at,
621                ldt_delta,
622                self.min_local_deletion_time,
623                absolute_local_deletion_time
624            );
625            (
626                Some(absolute_marked_for_delete_at),
627                Some(absolute_local_deletion_time),
628            )
629        } else {
630            (None, None)
631        };
632
633        // Parse column bitmap if HAS_ALL_COLUMNS is NOT set
634        let missing_columns_bitmap = if (row_flags & ROW_HAS_ALL_COLUMNS) == 0 {
635            // Cassandra Columns.Serializer.serializeSubset() format:
636            // Single unsigned VInt encoding a bitmask of MISSING columns
637            // (bit=1 means column is missing, bit=0 means present)
638            let (remaining, bitmap) = parse_vuint(&data[pos..]).map_err(|e| {
639                Error::corruption(format!(
640                    "V5CompressedLegacy: Failed to parse column bitmap at offset {}: {:?}",
641                    offset + pos,
642                    e
643                ))
644            })?;
645            let bytes_consumed = data[pos..].len() - remaining.len();
646            pos += bytes_consumed;
647
648            debug!(
649                "V5CompressedLegacy: Parsed column bitmap: missing_bitmap=0x{:X} ({} bytes)",
650                bitmap, bytes_consumed
651            );
652            Some(bitmap)
653        } else {
654            None
655        };
656
657        let header_size = pos - offset;
658        debug!(
659            "V5CompressedLegacy: Row header parsing complete: offset_start={}, pos_end={}, header_size={} bytes, row_size={} bytes (total row including cells), timestamp={:?}, ttl={:?}, deletion={:?}",
660            offset, pos, header_size, row_size, timestamp, ttl, local_deletion_time
661        );
662
663        // Note: ttl_liveness_ldt (from HAS_TTL) is the pk_liveness local expiration time.
664        // It is distinct from local_deletion_time (from HAS_DELETION, row tombstone GC clock).
665        // We store it in RowHeader so the delta-scan path can populate CellMeta.expires_at
666        // for TTL-bearing INSERT rows (Issue #702).  is_row_tombstone() still checks
667        // local_deletion_time (HAS_DELETION only), so TTL rows are NOT misclassified.
668
669        Ok((
670            RowHeader {
671                timestamp,
672                ttl,
673                liveness_expires_at_seconds: ttl_liveness_ldt,
674                local_deletion_time,
675                marked_for_delete_at,
676                header_size,
677                row_size_vint_len,
678                missing_columns_bitmap,
679                // Issue #1741: populated after the cell loop in
680                // `parse_row_data_with_offset_impl`; defaults describe a row with no
681                // decoded data cells.
682                max_data_cell_timestamp: None,
683                max_data_cell_expires_at: None,
684                has_live_forever_data_cell: false,
685            },
686            row_size,
687        ))
688    }

705    /// Like [`parse_partition_header`] but also returns the partition-level deletion
706    /// `(markedForDeleteAt µs, localDeletionTime s)`, if the partition is deleted.
707    ///
708    /// Returns `(RowKey, next_offset, Option<(markedForDeleteAt_micros, localDeletionTime_secs)>)`.
709    ///
710    /// `None` means the partition is live (no partition tombstone).
711    /// `Some((mfda, ldt))` means the partition carries a tombstone; `mfda` is the
712    /// authoritative reconciliation timestamp in microseconds since the Unix epoch and
713    /// `ldt` is the GC-grace clock in seconds (carried as the wrapping `as u32 as i32`
714    /// for far-future local-deletion-times, matching the row/range tombstone contract).
715    ///
716    /// Authority: DeletionTime.java (getSerializer / legacySerializer / Serializer),
717    /// BigFormat.java:409 (`hasUIntDeletionTime`).
718    #[allow(clippy::type_complexity)]
719    pub fn parse_partition_header_full(
720        &self,
721        data: &[u8],
722        offset: usize,
723    ) -> Result<(RowKey, usize, Option<(i64, i32)>)> {
724        // Issue #1618 (H5): count every speculative partition-header parse — the
725        // real allocating parse, which runs once per partition at a confirmed
726        // start. The per-row BOUNDARY peek (issue #1641, K2) does NOT come through
727        // here: it uses the non-allocating `peek_partition_boundary`, which shares
728        // the structural walk below via `scan_partition_header` but records no
729        // gauge and copies no key.
730        crate::storage::sstable::read_work_counters::record_partition_header_try_parse();
731
732        let layout = self.scan_partition_header(data, offset)?;
733        // The single legitimate key allocation: once per real header parse.
734        let key_bytes = data[layout.key_range].to_vec();
735        Ok((
736            RowKey::new(key_bytes),
737            layout.next_offset,
738            layout.partition_deletion,
739        ))
740    }

742    /// Structural walk of a partition header WITHOUT allocating the key or
743    /// recording the `PARTITION_HEADER_TRY_PARSES` gauge (issue #1641, K2).
744    ///
745    /// This is the single structural authority: [`parse_partition_header_full`]
746    /// wraps it (gauge + key `to_vec`), and [`peek_partition_boundary`] uses it as
747    /// the boundary detector. Every validation and every `Err` message is
748    /// identical to the former inline body of `parse_partition_header_full`, so a
749    /// peek can never accept a header the full parser rejects (no drift).
750    ///
751    /// [`parse_partition_header_full`]: Self::parse_partition_header_full
752    /// [`peek_partition_boundary`]: Self::peek_partition_boundary
753    fn scan_partition_header(
754        &self,
755        data: &[u8],
756        mut offset: usize,
757    ) -> Result<PartitionHeaderLayout> {
758        let start_offset = offset;
759
760        if offset >= data.len() {
761            return Err(Error::corruption(format!(
762                "V5CompressedLegacy: Partition header offset {} out of bounds (data len: {})",
763                offset,
764                data.len()
765            )));
766        }
767
768        // Byte 0: Flags (ignore for now - may indicate static rows, deletions, etc.)
769        let _flags = data[offset];
770        offset += 1;
771
772        // Byte 1: Partition key length (u8, NOT VInt)
773        if offset >= data.len() {
774            return Err(Error::corruption(
775                "V5CompressedLegacy: Unexpected end at partition key length",
776            ));
777        }
778        let key_len = data[offset] as usize;
779        offset += 1;
780
781        // Issue #258 FIX: Partition key length must be non-zero
782        // A key_len of 0 indicates this is NOT a valid partition header (likely row data).
783        // This validation is critical for peek_is_partition_header() to correctly
784        // distinguish partition headers from row data in the row loop.
785        if key_len == 0 {
786            return Err(Error::corruption(format!(
787                "V5CompressedLegacy: Invalid partition key length 0 at offset {} (not a partition header)",
788                start_offset
789            )));
790        }
791
792        debug!(
793            "V5CompressedLegacy: Partition key length = {} bytes",
794            key_len
795        );
796
797        // Next key_len bytes: Partition key data (raw bytes, no component structure)
798        if offset + key_len > data.len() {
799            return Err(Error::corruption(format!(
800                "V5CompressedLegacy: Partition key extends beyond data (offset: {}, key_len: {}, data_len: {})",
801                offset, key_len, data.len()
802            )));
803        }
804        // Record the key's byte RANGE — no copy here. The allocation happens
805        // once, in `parse_partition_header_full`, at a confirmed header start
806        // (issue #1641, K2); the boundary peek needs no key at all.
807        let key_range = offset..offset + key_len;
808        offset += key_len;
809
810        // Partition-level DeletionTime deserialization.
811        //
812        // VG3 gate: hasUIntDeletionTime (BigFormat.java:409, oa+)
813        //
814        // oa format uses a compact DeletionTime.Serializer
815        // (DeletionTime.java, Serializer inner class):
816        //   LIVE:    1 byte = 0x80 (IS_LIVE_DELETION = 0b10000000)
817        //   DELETED: 8 bytes markedForDeleteAt (long) +
818        //            4 bytes localDeletionTimeUnsignedInteger (int) = 12 bytes total
819        //
820        // nb format uses DeletionTime.legacySerializer:
821        //   Always:  4 bytes localDeletionTime (int) +
822        //            8 bytes markedForDeleteAt (long) = 12 bytes total
823        //
824        // Authority: DeletionTime.java:191-219 (getSerializer / Serializer.serialize)
825        let partition_deletion: Option<(i64, i32)>;
826        if self.has_uint_deletion_time() {
827            // oa / da format
828            if offset >= data.len() {
829                return Err(Error::corruption(
830                    "V5CompressedLegacy: Unexpected end at oa partition deletion time byte",
831                ));
832            }
833            let del_flags = data[offset];
834            if (del_flags & OA_IS_LIVE_DELETION) != 0 {
835                // LIVE partition: exactly 1 byte — no tombstone.
836                if del_flags != OA_IS_LIVE_DELETION {
837                    return Err(Error::corruption(format!(
838                        "V5CompressedLegacy: Invalid IS_LIVE_DELETION byte 0x{:02x} at offset {} \
839                         (only 0x80 is valid for oa-format LIVE partitions, per DeletionTime.java:227-229)",
840                        del_flags, offset
841                    )));
842                }
843                offset += 1;
844                partition_deletion = None;
845            } else {
846                // DELETED partition (oa): 8 bytes markedForDeleteAt (big-endian i64)
847                //                       + 4 bytes localDeletionTime (big-endian u32)
848                if offset + 12 > data.len() {
849                    return Err(Error::corruption(
850                        "V5CompressedLegacy: Unexpected end at oa partition deletion time (deleted)",
851                    ));
852                }
853                let mfda = i64::from_be_bytes(
854                    data[offset..offset + 8]
855                        .try_into()
856                        .map_err(|_| Error::corruption("V5CompressedLegacy: oa mfda slice"))?,
857                );
858                // localDeletionTime is the next 4 bytes (big-endian u32). Keep the
859                // wrapping `as u32 as i32` representation so far-future LDTs in
860                // [2^31, 2^32) round-trip exactly like row/range tombstones.
861                let ldt = u32::from_be_bytes(
862                    data[offset + 8..offset + 12]
863                        .try_into()
864                        .map_err(|_| Error::corruption("V5CompressedLegacy: oa ldt slice"))?,
865                ) as i32;
866                offset += 12; // markedForDeleteAt(8) + localDeletionTime(4)
867                partition_deletion = Some((mfda, ldt));
868            }
869        } else {
870            // nb format: 4 bytes localDeletionTime (big-endian i32)
871            //          + 8 bytes markedForDeleteAt (big-endian i64)
872            if offset + 12 > data.len() {
873                return Err(Error::corruption(
874                    "V5CompressedLegacy: Unexpected end at nb partition deletion time",
875                ));
876            }
877            // localDeletionTime is the first 4 bytes.
878            let local_deletion_time = i32::from_be_bytes(
879                data[offset..offset + 4]
880                    .try_into()
881                    .map_err(|_| Error::corruption("V5CompressedLegacy: nb ldt slice"))?,
882            );
883            offset += 4;
884            // markedForDeleteAt is next 8 bytes (big-endian i64).
885            let mfda = i64::from_be_bytes(
886                data[offset..offset + 8]
887                    .try_into()
888                    .map_err(|_| Error::corruption("V5CompressedLegacy: nb mfda slice"))?,
889            );
890            offset += 8;
891            // A live (not deleted) partition in nb format has
892            // localDeletionTime = 0x7fffffff (i32::MAX, DeletionTime.LIVE sentinel).
893            // Any other value indicates a real partition tombstone.
894            const NB_LIVE_LOCAL_DELETION_TIME: i32 = i32::MAX;
895            if local_deletion_time == NB_LIVE_LOCAL_DELETION_TIME {
896                partition_deletion = None;
897            } else {
898                partition_deletion = Some((mfda, local_deletion_time));
899            }
900        }
901
902        debug!(
903            "V5CompressedLegacy: Parsed partition header at offset {}, consumed {} bytes, \
904             partition_deletion={:?}",
905            start_offset,
906            offset - start_offset,
907            partition_deletion
908        );
909
910        Ok(PartitionHeaderLayout {
911            key_range,
912            next_offset: offset,
913            partition_deletion,
914        })
915    }

917    /// Non-allocating partition-BOUNDARY peek (issue #1641, K2).
918    ///
919    /// Answers "do the bytes at `offset` begin a new partition header?" for the
920    /// post-row emit loop while skipping the success-path key `to_vec` and the
921    /// `PARTITION_HEADER_TRY_PARSES` counter that the full parse incurs. The
922    /// fast-reject paths (marker pre-check + readiness gate) allocate nothing at
923    /// all; only the strict scan on a `Ready` buffer may still build a discarded
924    /// `format!` error string on a structural mismatch (inside
925    /// `scan_partition_header`). It reaches the same verdict the old allocating
926    /// `peek_is_partition_header` did — proved by the `#[cfg(test)]` proptest
927    /// (`Header` ⟺ `!marker && parse.is_ok()`).
928    ///
929    /// Algorithm (fast-reject paths allocate nothing; no gauge on any path):
930    /// 1. Marker pre-check (issue #229): an END_OF_PARTITION (`0x01`) or
931    ///    range-tombstone (IS_MARKER) leading byte is never a partition header;
932    ///    an offset past the buffer end is `NeedMoreBytes`.
933    /// 2. Completeness gate via the shared #1741 [`partition_header_readiness`]
934    ///    classifier: `Incomplete` → `NeedMoreBytes`, `Malformed` → `NotHeader`.
935    /// 3. Under `Ready` every header byte is present, so a [`scan_partition_header`]
936    ///    failure is a genuine STRUCTURAL rejection (e.g. an illegal oa/da IS_LIVE
937    ///    byte), never truncation: `Ok` → `Header`, `Err` → `NotHeader`.
938    ///
939    /// [`partition_header_readiness`]: Self::partition_header_readiness
940    /// [`scan_partition_header`]: Self::scan_partition_header
941    pub(super) fn peek_partition_boundary(&self, data: &[u8], offset: usize) -> BoundaryPeek {
942        // Step 1: marker / end-of-buffer pre-check.
943        match data.get(offset) {
944            None => return BoundaryPeek::NeedMoreBytes,
945            Some(&flags) => {
946                if Self::is_end_of_partition(flags) || Self::is_range_tombstone_marker(flags) {
947                    return BoundaryPeek::NotHeader;
948                }
949            }
950        }
951
952        // Step 2 + 3: completeness gate, then the strict shared structural scan.
953        // `offset < data.len()` holds (step 1 returned on `None`), so the subslice
954        // is valid and zero-cost.
955        match self.partition_header_readiness(&data[offset..]) {
956            PartitionHeaderReadiness::Incomplete => BoundaryPeek::NeedMoreBytes,
957            PartitionHeaderReadiness::Malformed => BoundaryPeek::NotHeader,
958            PartitionHeaderReadiness::Ready => match self.scan_partition_header(data, offset) {
959                Ok(_) => BoundaryPeek::Header,
960                Err(_) => BoundaryPeek::NotHeader,
961            },
962        }
963    }

966    /// inclusivity flags, and deletion timestamp(s).

990    /// | 2 | `EXCL_END_INCL_START_BOUNDARY` | boundary: end of prev range (exclusive) + start of new range (inclusive) |
991    /// | 5 | `INCL_END_EXCL_START_BOUNDARY` | boundary: end of prev range (inclusive) + start of new range (exclusive) |

1007        // fields the delta-scan caller does not need — the on-disk marker grammar
1008        // is decoded in exactly one place (issue #933 cleanup).
1009        let (bound_values, bound_kind, (mfda_primary, _ldt_primary), secondary, next_offset) =
1010            self.parse_range_tombstone_marker_with_ldt(data, offset, schema)?;
1011        Ok((
1012            bound_values,
1013            bound_kind,
1014            mfda_primary,
1015            secondary.map(|(mfda, _ldt)| mfda),
1016            next_offset,
1017        ))
1018    }

1020    /// Return a schema view truncated to `n` clustering keys.
1021    ///
1022    /// Range-tombstone bound markers may be **prefix bounds**: the Cassandra serializer
1023    /// writes only `cluster_count` clustering values when a DELETE specifies fewer
1024    /// clustering components than the full key arity (e.g. `DELETE WHERE pk=? AND ck1=?`
1025    /// on a table with `(ck1, ck2)` only pins the first component).
1026    ///
1027    /// Returns `None` (callers use the original schema) when `n >= schema.clustering_keys.len()`,
1028    /// avoiding a clone in the common non-prefix case.  Returns `Some(truncated)` when a
1029    /// shorter view is needed.
1030    fn clustering_prefix_schema(schema: &TableSchema, n: usize) -> Option<TableSchema> {
1031        if n >= schema.clustering_keys.len() {
1032            None
1033        } else {
1034            Some(TableSchema {

1046    /// Decode one `(markedForDeleteAt, localDeletionTime)` pair, returning BOTH
1047    /// absolute values (issue #933 compaction range-tombstone surfacing).
1048    ///
1049    /// Decodes the `(markedForDeleteAt delta, localDeletionTime delta)` VInt pair
1050    /// and returns BOTH absolute values — the absolute
1051    /// `localDeletionTime` (GC-grace clock, seconds) so the compaction merger can
1052    /// retain/purge a surviving range marker by gc_grace AND the writer can
1053    /// re-emit the marker's LDT verbatim. The LDT is decoded with the same
1054    /// `hasUIntDeletionTime` reinterpretation as the row-level deletion reader
1055    /// (`min_local_deletion_time + delta`, then `as u32 as i32` for far-future
1056    /// values) so a year-2106 LDT round-trips bit-for-bit (#853/#889).
1057    ///
1058    /// Advances `*pos` past both fields.
1059    fn parse_deletion_time_pair_with_ldt(
1060        &self,
1061        data: &[u8],
1062        pos: &mut usize,
1063    ) -> Result<(i64, i32)> {
1064        // markedForDeleteAt delta (unsigned VInt, µs since epoch delta).
1065        let (remaining, mfda_delta) = parse_vuint(&data[*pos..]).map_err(|e| {
1066            Error::corruption(format!(
1067                "V5CompressedLegacy: Failed to parse markedForDeleteAt in marker body at {}: {:?}",
1068                *pos, e
1069            ))
1070        })?;
1071        *pos += data[*pos..].len() - remaining.len();
1072        let absolute_mfda = self.min_timestamp.wrapping_add(mfda_delta as i64);
1073
1074        // localDeletionTime delta (unsigned VInt, seconds delta).
1075        let (remaining2, ldt_delta) = parse_vuint(&data[*pos..]).map_err(|e| {
1076            Error::corruption(format!(
1077                "V5CompressedLegacy: Failed to parse localDeletionTime in marker body at {}: {:?}",
1078                *pos, e
1079            ))
1080        })?;
1081        *pos += data[*pos..].len() - remaining2.len();
1082
1083        let has_uint_ldt = match self.version_gates.as_ref() {
1084            crate::storage::sstable::version_gate::VersionGates::Big(g) => g.has_uint_deletion_time,
1085            crate::storage::sstable::version_gate::VersionGates::Bti(g) => g.has_uint_deletion_time,
1086        };
1087        let raw_ldt = self.min_local_deletion_time.wrapping_add(ldt_delta as i64);
1088        let absolute_ldt = if has_uint_ldt {
1089            (raw_ldt as u32) as i32
1090        } else {
1091            raw_ldt as i32
1092        };
1093
1094        Ok((absolute_mfda, absolute_ldt))
1095    }

1097    /// Parse a range-tombstone bound marker in full, returning the decoded bound
1098    /// values, kind, AND the primary/secondary `(markedForDeleteAt, localDeletionTime)`
1099    /// pairs (issue #933).
1100    ///
1101    /// This is the compaction counterpart of [`Self::parse_range_tombstone_marker_full`]:
1102    /// it additionally surfaces the `localDeletionTime` of each deletion so the
1103    /// compaction path can re-emit / gc_grace-purge the marker faithfully.
1104    ///
1105    /// Returns `(bound_values, bound_kind, (mfda_primary, ldt_primary),
1106    /// Option<(mfda_secondary, ldt_secondary)>, next_offset)`.
1107    #[allow(clippy::type_complexity)]
1108    pub(super) fn parse_range_tombstone_marker_with_ldt(
1109        &self,
1110        data: &[u8],
1111        offset: usize,
1112        schema: &TableSchema,
1113    ) -> Result<(Vec<Value>, u8, (i64, i32), Option<(i64, i32)>, usize)> {
1114        let mut pos = offset;
1115
1116        if pos >= data.len() {
1117            return Err(Error::corruption(
1118                "V5CompressedLegacy: Unexpected end at range tombstone marker (compaction)",
1119            ));
1120        }
1121
1122        let marker_flags = data[pos];
1123        pos += 1;
1124
1125        if (marker_flags & ROW_HAS_EXTENDED_FLAGS) != 0 {
1126            if pos >= data.len() {
1127                return Err(Error::corruption(
1128                    "V5CompressedLegacy: Unexpected end reading marker extended flags (compaction)",
1129                ));
1130            }
1131            pos += 1;
1132        }
1133
1134        // Bound kind byte.
1135        if pos >= data.len() {
1136            return Err(Error::corruption(
1137                "V5CompressedLegacy: Unexpected end reading range tombstone bound kind (compaction)",
1138            ));
1139        }
1140        let bound_kind = data[pos];
1141        pos += 1;
1142
1143        // Cluster count (u16 big-endian).
1144        if pos + 2 > data.len() {
1145            return Err(Error::corruption(
1146                "V5CompressedLegacy: Unexpected end reading range tombstone cluster count (compaction)",
1147            ));
1148        }
1149        let cluster_count = u16::from_be_bytes([data[pos], data[pos + 1]]) as usize;
1150        pos += 2;
1151
1152        // Clustering values (may be a PREFIX shorter than the full arity).
1153        let bound_values = if cluster_count > 0 {
1154            let prefix_schema_owned = Self::clustering_prefix_schema(schema, cluster_count);
1155            let effective_schema = prefix_schema_owned.as_ref().unwrap_or(schema);
1156            let (values, new_pos) = self.parse_clustering_prefix(data, pos, effective_schema)?;
1157            pos = new_pos;
1158            values
1159        } else {
1160            Vec::new()
1161        };
1162
1163        // marker_body_size VUInt — size of (prev_size VUInt + deletion_time(s)).
1164        let (remaining, marker_body_size) = parse_vuint(&data[pos..]).map_err(|e| {
1165            Error::corruption(format!(
1166                "V5CompressedLegacy: Failed to parse marker_body_size (compaction) at offset {}: {:?}",
1167                pos, e
1168            ))
1169        })?;
1170        pos += data[pos..].len() - remaining.len();
1171
1172        let body_end = pos + marker_body_size as usize;
1173        if body_end > data.len() {
1174            return Err(Error::corruption(format!(
1175                "V5CompressedLegacy: marker_body_size={} at pos={} exceeds data length {} (compaction)",
1176                marker_body_size,
1177                pos,
1178                data.len()
1179            )));
1180        }
1181
1182        // prev_unfiltered_size VUInt — skip.
1183        let (remaining2, _prev_size) = parse_vuint(&data[pos..]).map_err(|e| {
1184            Error::corruption(format!(
1185                "V5CompressedLegacy: Failed to parse prev_size in marker body (compaction) at {}: {:?}",
1186                pos, e
1187            ))
1188        })?;
1189        pos += data[pos..].len() - remaining2.len();
1190
1191        let primary = self.parse_deletion_time_pair_with_ldt(data, &mut pos)?;
1192        let secondary = if bound_kind == 2 || bound_kind == 5 {
1193            Some(self.parse_deletion_time_pair_with_ldt(data, &mut pos)?)
1194        } else {
1195            None
1196        };
1197
1198        pos = body_end;
1199
1200        Ok((bound_values, bound_kind, primary, secondary, pos))
1201    }

1203    /// Parse clustering prefix section (between row header and cells)
1204    ///
1205    /// The clustering prefix encodes clustering key values using a compact VInt header
1206    /// with 2 bits per clustering column to indicate value state.
1207    ///
1208    /// # Format
1209    /// ```text
1210    /// [prefix_header: VInt] ← 2 bits per clustering column
1211    ///   - 00 = null
1212    ///   - 01 = empty
1213    ///   - 10/11 = has value
1214    /// [value_1: bytes if present]
1215    /// [value_2: bytes if present]
1216    /// [... more values ...]
1217    /// ```
1218    ///
1219    /// Returns: (clustering_values, new_offset)
1220    pub(super) fn parse_clustering_prefix(
1221        &self,
1222        data: &[u8],
1223        mut offset: usize,
1224        schema: &TableSchema,
1225    ) -> Result<(Vec<Value>, usize)> {
1226        // If no clustering keys, skip this section
1227        if schema.clustering_keys.is_empty() {
1228            tracing::debug!(
1229                "V5CompressedLegacy: No clustering keys in schema, skipping clustering prefix"
1230            );
1231            return Ok((Vec::new(), offset));
1232        }
1233
1234        tracing::debug!(
1235            "V5CompressedLegacy: Parsing clustering prefix at offset {} for {} clustering keys",
1236            offset,
1237            schema.clustering_keys.len()
1238        );
1239
1240        // Read header VInt (2 bits per clustering column)
1241        let (remaining, header_vint) = parse_vuint(&data[offset..]).map_err(|e| {
1242            Error::corruption(format!(
1243                "V5CompressedLegacy: Failed to parse clustering prefix header VInt at offset {}: {:?}",
1244                offset, e
1245            ))
1246        })?;
1247        let bytes_consumed = data[offset..].len() - remaining.len();
1248        offset += bytes_consumed;
1249
1250        tracing::debug!(
1251            "V5CompressedLegacy: Clustering prefix header = 0x{:x}, consumed {} bytes",
1252            header_vint,
1253            bytes_consumed
1254        );
1255
1256        // Decode each clustering value based on 2-bit state
1257        let mut clustering_values = Vec::new();
1258        for (i, col) in schema.clustering_keys.iter().enumerate() {
1259            let state = (header_vint >> (i * 2)) & 0x03;
1260            tracing::debug!(
1261                "V5CompressedLegacy: Clustering key {} '{}' state = {} (from bits {}..{})",
1262                i,
1263                col.name,
1264                state,
1265                i * 2,
1266                i * 2 + 1
1267            );
1268
1269            // Issue #229 FIX: Correct state interpretation per Cassandra's ClusteringPrefix.Kind
1270            //
1271            // Per Cassandra 5.0 UnfilteredSerializer.java and ClusteringPrefix.Kind:
1272            // - 0 (PRESENT): Value is present, type-specific bytes follow
1273            // - 1 (EMPTY): Empty value (zero-length, no bytes follow)
1274            // - 2 (NULL): NULL value (no bytes follow)
1275            // - 3: Reserved
1276            //
1277            // Previous code had 0=NULL, 2/3=PRESENT which was inverted!
1278            match state {
1279                0 => {
1280                    // PRESENT - parse value based on type
1281                    let (value, new_off) = self.parse_clustering_value(data, offset, col)?;
1282                    tracing::debug!(
1283                        "V5CompressedLegacy:   -> PRESENT: {:?} (consumed {} bytes)",
1284                        value,
1285                        new_off - offset
1286                    );
1287                    clustering_values.push(value);
1288                    offset = new_off;
1289                }
1290                1 => {
1291                    // EMPTY - zero-length value
1292                    //
1293                    // Per Cassandra's ClusteringPrefix, EMPTY means zero-length byte array.
1294                    // For variable-width types, this is valid. For fixed-width types (int,
1295                    // bigint, UUID), EMPTY should not normally occur.
1296                    let col_type = col.data_type.to_lowercase();
1297                    let empty_value = match col_type.as_str() {
1298                        "text" | "varchar" | "ascii" => Value::Text(String::new()),
1299                        "blob" => Value::Blob(vec![]),
1300                        _ => {
1301                            // Fixed-width types shouldn't have EMPTY state in normal data
1302                            tracing::warn!(
1303                                "V5CompressedLegacy: EMPTY state for clustering key '{}' (type {}), treating as NULL",
1304                                col.name, col.data_type
1305                            );
1306                            Value::Null
1307                        }
1308                    };
1309                    clustering_values.push(empty_value);
1310                    tracing::debug!("V5CompressedLegacy:   -> EMPTY");
1311                }
1312                2 => {
1313                    // NULL
1314                    clustering_values.push(Value::Null);
1315                    tracing::debug!("V5CompressedLegacy:   -> NULL");
1316                }
1317                3 => {
1318                    // Reserved - treat as NULL for safety
1319                    tracing::warn!("V5CompressedLegacy: Clustering key {} has reserved state 3, treating as NULL", col.name);
1320                    clustering_values.push(Value::Null);
1321                }
1322                _ => unreachable!(),
1323            }
1324        }
1325
1326        tracing::debug!(
1327            "V5CompressedLegacy: Parsed {} clustering values, new offset = {}",
1328            clustering_values.len(),
1329            offset
1330        );
1331
1332        Ok((clustering_values, offset))
1333    }

1335    /// Parse individual clustering value (type-specific)
1336    ///
1337    /// Clustering values are encoded based on their CQL type. This handles the most
1338    /// common clustering key types: timestamp, text, int, uuid.
1339    ///
1340    /// Returns: (value, new_offset)
1341    fn parse_clustering_value(
1342        &self,
1343        data: &[u8],
1344        offset: usize,
1345        col: &crate::schema::ClusteringColumn,
1346    ) -> Result<(Value, usize)> {
1347        let normalized = col.data_type.to_lowercase();
1348        tracing::debug!(
1349            "V5CompressedLegacy: Parsing clustering value '{}' type '{}' at offset {}",
1350            col.name,
1351            normalized,
1352            offset
1353        );
1354
1355        match normalized.as_str() {
1356            "timestamp" | "reversedtype(timestamptype)" => {
1357                // Fixed 8-byte timestamp (big-endian i64)
1358                if offset + 8 > data.len() {
1359                    return Err(Error::corruption(format!(
1360                        "V5CompressedLegacy: Clustering '{}': need 8 bytes for timestamp, only {} available",
1361                        col.name,
1362                        data.len() - offset
1363                    )));
1364                }
1365                let ts = i64::from_be_bytes([
1366                    data[offset],
1367                    data[offset + 1],
1368                    data[offset + 2],
1369                    data[offset + 3],
1370                    data[offset + 4],
1371                    data[offset + 5],
1372                    data[offset + 6],
1373                    data[offset + 7],
1374                ]);
1375                Ok((Value::Timestamp(ts), offset + 8))
1376            }
1377
1378            "text" | "utf8type" | "varchar" => {
1379                // VInt length + UTF-8 bytes
1380                let (remaining, len) = parse_vuint(&data[offset..]).map_err(|e| {
1381                    Error::corruption(format!(
1382                        "V5CompressedLegacy: Clustering '{}': failed to parse text length: {:?}",
1383                        col.name, e
1384                    ))
1385                })?;
1386                let bytes_consumed = data[offset..].len() - remaining.len();
1387                let len_offset = offset + bytes_consumed;
1388
1389                if len_offset + len as usize > data.len() {
1390                    return Err(Error::corruption(format!(
1391                        "V5CompressedLegacy: Clustering '{}': need {} bytes for text, only {} available",
1392                        col.name,
1393                        len,
1394                        data.len() - len_offset
1395                    )));
1396                }
1397
1398                let text = String::from_utf8(data[len_offset..len_offset + len as usize].to_vec())
1399                    .map_err(|e| {
1400                        Error::corruption(format!(
1401                            "V5CompressedLegacy: Clustering '{}': invalid UTF-8: {:?}",
1402                            col.name, e
1403                        ))
1404                    })?;
1405                Ok((Value::Text(text), len_offset + len as usize))
1406            }
1407
1408            "int" => {
1409                // Issue #258 fix: Fixed 4-byte int (big-endian i32) - NO length prefix
1410                // Per Cassandra format, fixed-width clustering types have no VInt length prefix
1411                if offset + 4 > data.len() {
1412                    return Err(Error::corruption(format!(
1413                        "V5CompressedLegacy: Clustering '{}': need 4 bytes for int, only {} available",
1414                        col.name,
1415                        data.len() - offset
1416                    )));
1417                }
1418
1419                let val = i32::from_be_bytes([
1420                    data[offset],
1421                    data[offset + 1],
1422                    data[offset + 2],
1423                    data[offset + 3],
1424                ]);
1425                Ok((Value::Integer(val), offset + 4))
1426            }
1427
1428            "uuid" | "timeuuid" => {
1429                // Issue #258 fix: Fixed 16-byte UUID - NO length prefix
1430                // Per Cassandra format, fixed-width clustering types have no VInt length prefix
1431                if offset + 16 > data.len() {
1432                    return Err(Error::corruption(format!(
1433                        "V5CompressedLegacy: Clustering '{}': need 16 bytes for UUID, only {} available",
1434                        col.name,
1435                        data.len() - offset
1436                    )));
1437                }
1438
1439                let uuid_bytes: [u8; 16] = data[offset..offset + 16]
1440                    .try_into()
1441                    .map_err(|_| Error::corruption("UUID byte conversion failed"))?;
1442
1443                Ok((Value::Uuid(uuid_bytes), offset + 16))
1444            }
1445
1446            "bigint" | "counter" => {
1447                // Issue #258 fix: Fixed 8-byte bigint (big-endian i64) - NO length prefix
1448                // Per Cassandra format, fixed-width clustering types have no VInt length prefix
1449                if offset + 8 > data.len() {
1450                    return Err(Error::corruption(format!(
1451                        "V5CompressedLegacy: Clustering '{}': need 8 bytes for bigint, only {} available",
1452                        col.name,
1453                        data.len() - offset
1454                    )));
1455                }
1456
1457                let val = i64::from_be_bytes([
1458                    data[offset],
1459                    data[offset + 1],
1460                    data[offset + 2],
1461                    data[offset + 3],
1462                    data[offset + 4],
1463                    data[offset + 5],
1464                    data[offset + 6],
1465                    data[offset + 7],
1466                ]);
1467                Ok((Value::BigInt(val), offset + 8))
1468            }
1469
1470            _ => {
1471                // For other types, read VInt length + skip that many bytes
1472                // Return as blob for now
1473                warn!(
1474                    "V5CompressedLegacy: Clustering '{}' has unsupported type '{}', treating as blob",
1475                    col.name, col.data_type
1476                );
1477                let (remaining, len) = parse_vuint(&data[offset..]).map_err(|e| {
1478                    Error::corruption(format!(
1479                        "V5CompressedLegacy: Clustering '{}': failed to parse blob length: {:?}",
1480                        col.name, e
1481                    ))
1482                })?;
1483                let bytes_consumed = data[offset..].len() - remaining.len();
1484                let len_offset = offset + bytes_consumed;
1485
1486                if len_offset + len as usize > data.len() {
1487                    return Err(Error::corruption(format!(
1488                        "V5CompressedLegacy: Clustering '{}': need {} bytes, only {} available",
1489                        col.name,
1490                        len,
1491                        data.len() - len_offset
1492                    )));
1493                }
1494
1495                Ok((
1496                    Value::Blob(data[len_offset..len_offset + len as usize].to_vec()),
1497                    len_offset + len as usize,
1498                ))
1499            }
1500        }
1501    }
