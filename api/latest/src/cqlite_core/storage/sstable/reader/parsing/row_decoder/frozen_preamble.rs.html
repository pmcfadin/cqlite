1//! The frozen-collection PREAMBLE: the VUInt outer blob length and the i32 BE
2//! element count that precede every frozen collection body.
3//!
4//! Split out of `frozen.rs` under the campsite rule (epic #1116) alongside the
5//! issue #3848 fix. The `impl` block is unchanged, so both helpers keep their
6//! path and visibility; `read_frozen_preamble` is `pub(super)` so the #3848
7//! regression module can call it directly with no reader/dataset dependency.
8
9use super::*;
10
11impl V5CompressedLegacyParser {
12    /// Read i32 BE element/entry count from a frozen collection blob.
13    ///
14    /// `bound` is the exclusive upper byte index for the collection data (either
15    /// `data.len()` for raw variants or `blob_end` for cell-level variants).
16    pub(super) fn read_frozen_count(
17        data: &[u8],
18        offset: &mut usize,
19        bound: usize,
20        collection_kind: &str,
21        column_name: &str,
22    ) -> Result<usize> {
23        if *offset + 4 > bound {
24            return Err(Error::corruption(format!(
25                "Frozen {} '{}': not enough bytes for element count",
26                collection_kind, column_name
27            )));
28        }
29        let count = i32::from_be_bytes([
30            data[*offset],
31            data[*offset + 1],
32            data[*offset + 2],
33            data[*offset + 3],
34        ]);
35        *offset += 4;
36
37        if count < 0 {
38            return Err(Error::corruption(format!(
39                "Frozen {} '{}': negative element count {}",
40                collection_kind, column_name, count
41            )));
42        }
43        let count = count as usize;
44        if count > MAX_FROZEN_COLLECTION_SIZE as usize {
45            return Err(Error::corruption(format!(
46                "Frozen {} '{}': element count {} exceeds maximum {}",
47                collection_kind, column_name, count, MAX_FROZEN_COLLECTION_SIZE
48            )));
49        }
50        Ok(count)
51    }
52
53    /// Read the frozen collection preamble: VUInt blob_len + i32 BE element count.
54    ///
55    /// Returns `(count, blob_end)` with `offset` advanced past the preamble.
56    pub(super) fn read_frozen_preamble(
57        data: &[u8],
58        offset: &mut usize,
59        collection_kind: &str,
60        column_name: &str,
61    ) -> Result<(usize, usize)> {
62        let (remaining, blob_len_raw) = parse_vuint(&data[*offset..]).map_err(|e| {
63            Error::corruption(format!(
64                "Frozen {} '{}': failed to parse blob length: {:?}",
65                collection_kind, column_name, e
66            ))
67        })?;
68        // Issue #3848: cap BEFORE the `as usize` cast. `parse_vuint` yields up to
69        // `u64::MAX` from a 9-byte encoding, so an adversarial `Data.db` length
70        // prefix must be rejected here rather than reaching the bounds add below
71        // (which, unchecked, panics in an overflow-checked build and wraps in
72        // release). Same guard as `parse_tuple_value` on the identical framing.
73        if blob_len_raw > MAX_CELL_VALUE_LENGTH {
74            return Err(Error::corruption(format!(
75                "Frozen {} '{}': blob_len {} exceeds maximum {}",
76                collection_kind, column_name, blob_len_raw, MAX_CELL_VALUE_LENGTH
77            )));
78        }
79        let blob_len = blob_len_raw as usize;
80        let bytes_consumed = data[*offset..].len() - remaining.len();
81        *offset += bytes_consumed;
82
83        // Issue #3848, second axis: the saturating form cannot overflow for ANY
84        // `blob_len`, so it holds even if the cap above is ever removed.
85        if blob_len > data.len().saturating_sub(*offset) {
86            return Err(Error::corruption(format!(
87                "Frozen {} '{}': blob_len {} exceeds available data {}",
88                collection_kind,
89                column_name,
90                blob_len,
91                data.len().saturating_sub(*offset)
92            )));
93        }
94
95        let blob_end = *offset + blob_len;
96        let count = Self::read_frozen_count(data, offset, blob_end, collection_kind, column_name)?;
97        Ok((count, blob_end))
98    }
99}
