53impl V5CompressedLegacyParser {
54    /// Map a PRIMITIVE Cassandra marshal type (e.g.
55    /// `org.apache.cassandra.db.marshal.Int32Type`) to the canonical CQL short
56    /// form (`"int"`) understood by [`parse_value_from_raw_bytes`]'s match
57    /// (issue #1081). Returns `None` for any non-primitive marshal form
58    /// (UserType / collection / tuple / reversed / frozen / custom), so the
59    /// caller leaves those to the dedicated arms.
60    ///
61    /// # ONE marshal-name table (issue #3631, roborev job 68)
62    /// This USED to be a second, independently-maintained suffix table, documented
63    /// as a deliberate *superset* of `parse_cassandra_type_with_depth`'s — and that
64    /// drift was the defect: the names this one had and that one lacked
65    /// (`VarcharType`, `CounterColumnType`, `LexicalUUIDType`, `ShortType`,
66    /// `ByteType`, and `TupleType`) are exactly the marshal-form UDT fields the
67    /// `CqlType` decoder then mis-routed to the nested-UDT decoder. So the table is
68    /// gone: this is now a projection of
69    /// [`Self::native_marshal_to_cql_type`] (the single marshal-name authority,
70    /// keyed on `cassandra-5.0.8`'s `CQL3Type.Native`) through
71    /// [`Self::cql_scalar_short_form`], and a name can no longer be known to one
72    /// side and unknown to the other.
73    ///
74    /// Two consequences of inheriting that table's rules, both intended:
75    /// `CounterColumnType` now normalizes to `"counter"` rather than `"bigint"`
76    /// (identical decode — the value arm is `"bigint" | "counter"` and the
77    /// cell-path width table gives both `[0, 8]`), and matching is EXACT on the
78    /// simple class name, so a third-party `com.acme.MyBytesType` is no longer
79    /// normalized to `blob` by a suffix match (it falls to the caller's
80    /// unknown-type path, as it always should have).
81    pub(super) fn primitive_marshal_to_cql_short(marshal_type: &str) -> Option<&'static str> {
82        Self::native_marshal_to_cql_type(marshal_type)
83            .as_ref()
84            .and_then(Self::cql_scalar_short_form)
85    }

87    /// Parse a value from a complete, bounded byte slice.
88    ///
89    /// This is used when the outer Cassandra collection format already provides
90    /// explicit `[i32 BE len][raw bytes]` boundaries and we have extracted exactly
91    /// the bytes that constitute the value. The entire `data` slice IS the value.
92    ///
93    /// **Issue #3811: that sentence is now ENFORCED.** This is a thin wrapper over
94    /// [`Self::parse_value_from_raw_bytes_reporting`], which threads a REAL
95    /// consumption count out of every arm; the wrapper then requires the decode to
96    /// have consumed every byte of `data`
97    /// ([`Self::require_fully_consumed`]). The rule is
98    /// `cassandra-5.0.8` `TupleType.split`: a genuinely SHORT encoding leaves
99    /// `position == length` and is legal, while trailing bytes (rule 4) and a
100    /// partial 1-3 byte component-length prefix (rule 2) both leave it short and
101    /// are `MarshalException`s.
102    ///
103    /// **Inheritance mechanism (AC2).** Every existing bounded call site keeps this
104    /// name and silently GAINS the check. The reporting twin is `pub(super)` inside
105    /// `raw_value::reporting`, so it is not nameable from any other `row_decoder`
106    /// child at all — a bounded caller elsewhere cannot opt out even deliberately
107    /// without first widening that visibility, which is the reviewable act. (An
108    /// earlier revision of this paragraph described reaching for the longer name as
109    /// something a caller could just do; it could not, and saying so invited the
110    /// next author to "fix" the privacy error by creating the opt-out.)
111    pub(super) fn parse_value_from_raw_bytes(
112        &self,
113        data: &[u8],
114        type_str: &str,
115        column_name: &str,
116        depth: usize,
117    ) -> Result<Value> {
118        let (value, consumed) =
119            self.parse_value_from_raw_bytes_reporting(data, type_str, column_name, depth)?;
120        Self::require_fully_consumed(consumed, data.len(), column_name, type_str)?;
121        Ok(value)
122    }
123}

132    /// Issue #1081: `primitive_marshal_to_cql_short` must normalize every
133    /// PRIMITIVE Cassandra marshal type (fully-qualified) to its canonical CQL
134    /// short form, and must reject any parameterised/composite marshal form

136    /// no-heuristics `(`-rejection guard never misreads a composite as a scalar.

138    fn primitive_marshal_to_cql_short_maps_scalars_and_rejects_composites() {
139        const P: &str = "org.apache.cassandra.db.marshal.";
140
141        // (marshal type name, expected canonical CQL short form)
142        let cases: &[(&str, &str)] = &[
143            ("UTF8Type", "text"),
144            ("AsciiType", "ascii"),
145            ("Int32Type", "int"),
146            ("LongType", "bigint"),
147            ("FloatType", "float"),
148            ("DoubleType", "double"),
149            ("BooleanType", "boolean"),
150            ("UUIDType", "uuid"),
151            ("TimeUUIDType", "timeuuid"),
152            ("TimestampType", "timestamp"),
153            ("SimpleDateType", "date"),
154            // Legacy `DateType` is an 8-byte millis-since-epoch value (same wire
155            // format as `TimestampType`), NOT the 4-byte CQL `date`
156            // (`SimpleDateType`). It must normalize to `timestamp`.
157            ("DateType", "timestamp"),
158            ("TimeType", "time"),
159            ("DecimalType", "decimal"),
160            ("IntegerType", "varint"),
161            ("DurationType", "duration"),
162            ("ShortType", "smallint"),
163            ("ByteType", "tinyint"),
164            ("InetAddressType", "inet"),
165            ("BytesType", "blob"),
166        ];
167
168        for (marshal, expected) in cases {
169            let full = format!("{}{}", P, marshal);
170            assert_eq!(
171                V5CompressedLegacyParser::primitive_marshal_to_cql_short(&full),
172                Some(*expected),
173                "primitive marshal {} should map to {}",
174                full,
175                expected
176            );
177        }
178
179        // Parameterised / composite marshal forms must be rejected (return None)
180        // by the `(`-guard, leaving them to the dedicated composite arms.
181        let composites = [
182            "org.apache.cassandra.db.marshal.UserType(...)",
183            "org.apache.cassandra.db.marshal.ListType(org.apache.cassandra.db.marshal.UTF8Type)",
184            "org.apache.cassandra.db.marshal.ReversedType(org.apache.cassandra.db.marshal.Int32Type)",
185        ];
186        for composite in composites {
187            assert_eq!(
188                V5CompressedLegacyParser::primitive_marshal_to_cql_short(composite),
189                None,
190                "composite marshal {} must be rejected by the `(`-guard",
191                composite
192            );
193        }

312    /// Issue #1081: a multicell-UDT field declared `duration` resolves its
313    /// field type from the authoritative on-disk `UserType(...)` marshal string
314    /// as `org.apache.cassandra.db.marshal.DurationType`. That bare scalar
315    /// marshal form must normalize to `"duration"` and decode the three
316    /// consecutive SIGNED VInts (months, days, nanos) that constitute the value
317    /// — NOT fall through to `Value::Blob`. In `parse_value_from_raw_bytes` the
318    /// entire slice IS the value (no outer `[VInt len]` prefix).
319    #[test]
320    fn test_parse_value_from_raw_bytes_duration_marshal() {
321        let parser =
322            V5CompressedLegacyParser::new("test".to_string(), "table".to_string(), 0, 0, None);
323
324        // Build three SIGNED VInts (months=1, days=2, nanos=3) using the same
325        // ZigZag-over-unsigned-VInt scheme `parse_vint` decodes. ZigZag:
326        // 1 -> 2, 2 -> 4, 3 -> 6.
327        let zigzag = |v: i64| ((v << 1) ^ (v >> 63)) as u64;
328        let mut data = Vec::new();
329        encode_unsigned(zigzag(1), &mut data); // months
330        encode_unsigned(zigzag(2), &mut data); // days
331        encode_unsigned(zigzag(3), &mut data); // nanos
332
333        // Round-trip via the signed VInt parser to confirm the encoding before
334        // exercising the decode arm under test.
335        let (rem, m) = parse_vint(&data).unwrap();
336        let consumed = data.len() - rem.len();
337        let (rem2, d) = parse_vint(&data[consumed..]).unwrap();
338        let consumed2 = data.len() - rem2.len();
339        let (_rem3, n) = parse_vint(&data[consumed2..]).unwrap();
340        assert_eq!(
341            (m, d, n),
342            (1, 2, 3),
343            "hand-encoded duration vints round-trip"
344        );
345
346        // The bare scalar marshal form must normalize and decode to Duration,
347        // NOT fall through to Blob.
348        let val = parser
349            .parse_value_from_raw_bytes(
350                &data,
351                "org.apache.cassandra.db.marshal.DurationType",
352                "col",
353                0,
354            )
355            .unwrap();
356        assert_eq!(
357            val,
358            Value::Duration {
359                months: 1,
360                days: 2,
361                nanos: 3
362            }
363        );
364
365        // The canonical CQL short form must decode identically.
366        let val_short = parser
367            .parse_value_from_raw_bytes(&data, "duration", "col", 0)
368            .unwrap();
369        assert_eq!(val_short, val);
370    }

372    /// Issue #1632 (item b): the raw-value frozen `duration` arm must REJECT a
373    /// months/days VInt outside the i32 range instead of silently wrapping via
374    /// `as i32`. On pre-fix code `months as i32` wraps `i32::MAX + 1` to a
375    /// negative value and returns Ok; the guard turns it into an error.
376    #[test]
377    fn test_parse_value_from_raw_bytes_duration_months_out_of_i32_range_errors() {
378        let parser =
379            V5CompressedLegacyParser::new("test".to_string(), "table".to_string(), 0, 0, None);
380        let zigzag = |v: i64| ((v << 1) ^ (v >> 63)) as u64;
381
382        // months = i32::MAX + 1 (overflows i32), days = 0, nanos = 0.
383        let mut over = Vec::new();
384        encode_unsigned(zigzag(i32::MAX as i64 + 1), &mut over);
385        encode_unsigned(zigzag(0), &mut over);
386        encode_unsigned(zigzag(0), &mut over);
387        assert!(
388            parser
389                .parse_value_from_raw_bytes(&over, "duration", "col", 0)
390                .is_err(),
391            "months > i32::MAX must error, not wrap via `as i32`"
392        );
393
394        // days = i32::MIN - 1 (underflows i32), months = 0, nanos = 0.
395        let mut under = Vec::new();
396        encode_unsigned(zigzag(0), &mut under);
397        encode_unsigned(zigzag(i32::MIN as i64 - 1), &mut under);
398        encode_unsigned(zigzag(0), &mut under);
399        assert!(
400            parser
401                .parse_value_from_raw_bytes(&under, "duration", "col", 0)

559    /// Issue #1081 (FINDING — last type-tree gap): a multicell-UDT field declared
560    /// `frozen<list<text>>` resolves from the on-disk `UserType(...)` marshal string
561    /// as `FrozenType(ListType(UTF8Type))`. Before the fix the frozen arm in

569            V5CompressedLegacyParser::new("test".to_string(), "table".to_string(), 0, 0, None);

571        let data = build_frozen_list_text(&["gamma", "delta"]);
572        let val = parser
573            .parse_value_from_raw_bytes(
574                &data,
575                "org.apache.cassandra.db.marshal.FrozenType(org.apache.cassandra.db.marshal.ListType(org.apache.cassandra.db.marshal.UTF8Type))",
