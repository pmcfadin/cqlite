5impl V5CompressedLegacyParser {
6    /// Parse a single cell value WITHOUT column name (schema-order format)
7    ///
8    /// Cell format in V5CompressedLegacy follows Cassandra 5.0 cell serialization:
9    /// - First byte: Cell flags (bitset, valid range: 0x00-0x1F)
10    ///   - 0x01 = IS_DELETED_MASK (tombstone)
11    ///   - 0x02 = IS_EXPIRING_MASK (has TTL)
12    ///   - 0x04 = HAS_EMPTY_VALUE_MASK (no value bytes)
13    ///   - 0x08 = USE_ROW_TIMESTAMP_MASK (use row timestamp)
14    ///   - 0x10 = USE_ROW_TTL_MASK (use row TTL)
15    /// - Conditional timestamp/TTL/deletion fields (based on flags)
16    /// - Value data (if HAS_EMPTY_VALUE not set)
17    ///
18    /// See CASSANDRA_5_CELL_DESERIALIZATION_FORMAT.md for complete specification.
19    ///
20    /// Returns: `(value, cell_own_timestamp, expiration, new_offset)` where:
21    /// - `cell_own_timestamp`: the cell's own decoded timestamp in µs, or `None`
22    ///   when the cell inherits the row-level timestamp (`USE_ROW_TIMESTAMP` flag).
23    /// - `expiration`: TTL / localDeletionTime pair when the cell is expiring, or
24    ///   `None` when the cell has no TTL.
25    pub(super) fn parse_cell_value_schema_order(
26        &self,
27        data: &[u8],
28        mut offset: usize,
29        column: &crate::schema::Column,
30        // AUTHORITATIVE on-disk SerializationHeader marshal type for this column
31        // (issue #1080). Used as a fallback to resolve a `frozen<udt>` whose
32        // supplied schema short form carries no field defs and no UdtRegistry is
33        // wired: the header type is the full `FrozenType(UserType(ks,name,fields))`
34        // marshal string, decoded structurally rather than guessed. `None` on the
35        // header-empty synthetic path and on internal recursive calls.
36        header_type: Option<&str>,
37        // J1 (issue #1635): the precomputed per-column dispatch tag (from
38        // `ColumnToParse.kind`, resolved ONCE per block). `Some` on the hot per-cell
39        // scan loop → dispatch on it with NO per-cell `to_lowercase`. `None` on the
40        // rare recursive frozen-inner / in-crate test callers → resolve it locally
41        // from `column.data_type` (bounded, off the per-cell scan hot path).
42        kind: Option<&CellKind>,
43        _reader: &crate::storage::sstable::reader::types::SSTableReader,
44    ) -> Result<(Value, Option<i64>, Option<CellExpiration>, usize)> {
45        // Cell flag constants (from Cassandra 5.0 Cell.Serializer)
46        const CELL_IS_DELETED: u8 = 0x01;
47        const CELL_IS_EXPIRING: u8 = 0x02;
48        const CELL_HAS_EMPTY_VALUE: u8 = 0x04;
49        const CELL_USE_ROW_TIMESTAMP: u8 = 0x08;
50        const CELL_USE_ROW_TTL: u8 = 0x10;
51
52        // Read cell flags byte
53        if offset >= data.len() {
54            return Err(Error::corruption(format!(
55                "Cell '{}': unexpected end at flags byte",
56                column.name
57            )));
58        }
59        let flags = data[offset];
60
61        // CRITICAL FIX (Issue #191): Validate flags are in valid range (0x00-0x1F)
62        // Bits 0x20, 0x40, 0x80 are row-level flags and should NEVER appear in cell flags.
63        // If we see these bits, the offset is misaligned (reading row data at cell position).
64        if flags > 0x1F {
65            return Err(Error::corruption(format!(
66                "Cell '{}': invalid cell flags 0x{:02x} at offset {} (bits 0x20/0x40/0x80 indicate offset misalignment)",
67                column.name, flags, offset
68            )));
69        }
70
71        offset += 1;
72
73        // Decode flags
74        let is_deleted = (flags & CELL_IS_DELETED) != 0;
75        let is_expiring = (flags & CELL_IS_EXPIRING) != 0;
76        let has_empty_value = (flags & CELL_HAS_EMPTY_VALUE) != 0;
77        let use_row_timestamp = (flags & CELL_USE_ROW_TIMESTAMP) != 0;
78        let use_row_ttl = (flags & CELL_USE_ROW_TTL) != 0;
79
80        tracing::debug!(
81            "V5CompressedLegacy: Cell '{}' flags=0x{:02x} (deleted={}, expiring={}, empty={}, use_row_ts={}, use_row_ttl={})",
82            column.name, flags, is_deleted, is_expiring, has_empty_value, use_row_timestamp, use_row_ttl
83        );
84
85        // === PHASE 2: Parse conditional fields between flags and value ===
86        // Based on Cassandra 5.0 Cell.Serializer format specification
87
88        // Step 1: Read timestamp (if not using row timestamp)
89        // Issue #505: capture the actual cell timestamp so deleted cells can carry it
90        // in a Value::Tombstone.
91        //
92        // Fix #629 (C2): Cell timestamp delta is UNSIGNED VInt per Cassandra
93        // SerializationHeader.java:165: out.writeUnsignedVInt(timestamp - stats.minTimestamp).
94        let mut cell_timestamp: Option<i64> = None;
95        if !use_row_timestamp {
96            let (remaining, timestamp_delta) = parse_vuint(&data[offset..]).map_err(|e| {
97                Error::corruption(format!(
98                    "Cell '{}': failed to parse timestamp delta as VUInt at offset {}: {:?}",
99                    column.name, offset, e
100                ))
101            })?;
102            let bytes_consumed = data[offset..].len() - remaining.len();
103            offset += bytes_consumed;
104            let absolute_ts = self.min_timestamp.wrapping_add(timestamp_delta as i64);
105            tracing::debug!(
106                "V5CompressedLegacy: Cell '{}' timestamp_delta={} (min_timestamp={}) absolute={}",
107                column.name,
108                timestamp_delta,
109                self.min_timestamp,
110                absolute_ts,
111            );
112            cell_timestamp = Some(absolute_ts);
113        }
114
115        // Step 2: Read localDeletionTime (if deleted or expiring, and not using row TTL)
116        // Captured as absolute epoch-seconds for CellExpiration.expires_at_seconds.
117        let mut cell_local_deletion_time: Option<i64> = None;
118        if !use_row_ttl && (is_deleted || is_expiring) {
119            let (remaining, deletion_delta) = parse_vuint(&data[offset..]).map_err(|e| {
120                Error::corruption(format!(
121                    "Cell '{}': failed to parse localDeletionTime delta as VUInt at offset {}: {:?}",
122                    column.name, offset, e
123                ))
124            })?;
125            let bytes_consumed = data[offset..].len() - remaining.len();
126            offset += bytes_consumed;
127            let abs_ldt = self
128                .min_local_deletion_time
129                .wrapping_add(deletion_delta as i64);
130            tracing::debug!(
131                "V5CompressedLegacy: Cell '{}' deletion_delta={} (min_local_deletion_time={}) abs_ldt={}",
132                column.name,
133                deletion_delta,
134                self.min_local_deletion_time,
135                abs_ldt
136            );
137            cell_local_deletion_time = Some(abs_ldt);
138        }
139
140        // Step 3: Read TTL (if expiring and not using row TTL)
141        // Captured as absolute TTL seconds for CellExpiration.ttl_seconds.
142        let mut cell_ttl_seconds: Option<i32> = None;
143        if !use_row_ttl && is_expiring {
144            let (remaining, ttl_delta) = parse_vuint(&data[offset..]).map_err(|e| {
145                Error::corruption(format!(
146                    "Cell '{}': failed to parse TTL delta as VUInt at offset {}: {:?}",
147                    column.name, offset, e
148                ))
149            })?;
150            let bytes_consumed = data[offset..].len() - remaining.len();
151            offset += bytes_consumed;
152            // Absolute TTL = min_ttl + delta (seconds).  Clamp to i32 range for the
153            // CellExpiration.ttl_seconds field (Cassandra caps TTL at ~630M seconds).
154            let abs_ttl = self.min_ttl.unwrap_or(0).wrapping_add(ttl_delta as i64);
155            tracing::debug!(
156                "V5CompressedLegacy: Cell '{}' ttl_delta={} (min_ttl={:?}) abs_ttl={}",
157                column.name,
158                ttl_delta,
159                self.min_ttl,
160                abs_ttl
161            );
162            cell_ttl_seconds = Some(abs_ttl.min(i32::MAX as i64) as i32);
163        }
164
165        // Build per-cell expiration metadata (used when the flag is set).
166        // Available at both return sites below — the tombstone path also uses
167        // cell_timestamp so we compute expiration here before the tombstone check.
168        let cell_expiration: Option<CellExpiration> =
169            match (is_expiring, cell_local_deletion_time, cell_ttl_seconds) {
170                (true, Some(expires_at), Some(ttl_secs)) => Some(CellExpiration {
171                    ttl_seconds: ttl_secs,
172                    expires_at_seconds: expires_at,
173                }),
174                // use_row_ttl path: expiration info comes from the row header (caller handles it).
175                _ => None,
176            };
177
178        // Step 4: Cell path for complex columns (multi-cell collections/UDTs)
179        // For now, skip this - we'll add in a future iteration when we handle complex columns.
180        // Simple columns (int, text, boolean, uuid, etc.) don't have cell paths.
181
182        // === End of Phase 2 conditional field parsing ===
183
184        // CRITICAL: Inverted logic for HAS_EMPTY_VALUE_MASK
185        // Flag NOT set (0x04 absent) = cell HAS value → read value bytes
186        // Flag SET (0x04 present) = cell has NO value → return empty/null immediately
187        let has_value = !has_empty_value;
188
189        // Handle deleted cells (tombstones)
190        // According to Cassandra 5.0 Cell.Serializer, deleted cells:
191        // 1. Have IS_DELETED flag set
192        // 2. May have deletion metadata (timestamp, localDeletionTime)
193        // 3. Do NOT have value data (even if HAS_EMPTY_VALUE not set)
194        //
195        // Issue #505: emit Value::Tombstone(CellTombstone) so that the compaction
196        // merger can apply cell-level shadowing semantics.  The actual deletion
197        // timestamp is carried in the tombstone for timestamp-based LWW ordering.
198        if is_deleted {
199            let deletion_time = cell_timestamp.unwrap_or(0);
200            tracing::debug!(
201                "V5CompressedLegacy: Cell '{}' is tombstone (deleted), returning Tombstone(deletion_time={})",
202                column.name, deletion_time
203            );
204            return Ok((
205                Value::Tombstone(Box::new(TombstoneInfo {
206                    deletion_time,
207                    tombstone_type: TombstoneType::CellTombstone,
208                    // On-disk `localDeletionTime` (GC clock, seconds) for the cell
209                    // tombstone; `0` when not surfaced (#873).
210                    local_deletion_time: cell_local_deletion_time.unwrap_or(0),
211                    ttl: None,
212                    range_start: None,
213                    range_end: None,
214                })),
215                cell_timestamp,
216                cell_expiration,
217                offset,
218            ));
219        }
220
221        // J1 (issue #1635): dispatch on the PRECOMPUTED per-column tag. The hot
222        // per-cell scan loop passes `Some(&ctp.kind)` (resolved once per block in
223        // `RowColumnResolution::build`), so the per-cell `column.data_type
224        // .to_lowercase()` + string-ladder walk is gone — no per-cell normalization,
225        // no per-cell allocation. `None` (recursive frozen-inner / in-crate tests)
226        // resolves the tag locally from the declared type; that path is bounded and
227        // off the per-cell scan hot path, so it neither regresses throughput nor the
228        // `TYPE_NORMALIZE_CALLS` gauge (which measures the per-cell decode path).
229        let resolved_kind: CellKind;
230        let kind: &CellKind = match kind {
231            Some(k) => k,
232            None => {
233                resolved_kind = CellKind::from_type(&column.data_type);
234                &resolved_kind
235            }
236        };
237
238        // Handle empty cells (no value bytes to read)
239        if !has_value {
240            tracing::debug!(
241                "V5CompressedLegacy: Cell '{}' has HAS_EMPTY_VALUE flag, returning empty value",
242                column.name
243            );
244            // Issue #1077: the empty (zero-length) value MUST decode to the empty
245            // value of the column's DECLARED type — never blindly `Text("")`.
246            // An empty `blob` is `Blob([])` (sstabledump renders `"0x"`), an empty
247            // text/ascii/varchar is `Text("")`. Mirrors the clustering-key EMPTY
248            // handling above; fixed-width types should not normally carry an empty
249            // value, so treat that as NULL with a warning. Dispatched on the
250            // precomputed tag (Text ⇔ text/varchar/ascii, Blob ⇔ literal blob;
251            // every other declared type → NULL), byte-identical to the pre-J1
252            // `match normalized_type.as_str()` empty arm.
253            let empty_value = match kind {
254                CellKind::Text => Value::Text(String::new()),
255                CellKind::Blob => Value::Blob(Vec::new()),
256                _ => {
257                    tracing::warn!(
258                        "V5CompressedLegacy: EMPTY value for cell '{}' (type {}), treating as NULL",
259                        column.name,
260                        column.data_type
261                    );
262                    Value::Null
263                }
264            };
265            return Ok((empty_value, cell_timestamp, cell_expiration, offset));
266        }
267
268        // VInt-length-prefixed blob decode. Shared by the literal `blob` type
269        // (`CellKind::Blob`) and the `CellKind::Complex` default fall-through
270        // (unknown/`varint`), which decoded identically pre-J1 (both hit the single
271        // `_ =>` blob arm). Extracted to one closure so the two dispatch sites do not
272        // duplicate the decode. Advances the shared `offset` cursor.
273        let decode_vint_blob = |offset: &mut usize| -> Result<Value> {
274            if *offset >= data.len() {
275                return Err(Error::corruption(format!(
276                    "Cell '{}': unexpected end at blob length (type: {})",
277                    column.name, column.data_type
278                )));
279            }
280            // Parse blob length as unsigned VInt (can be > 255 bytes).
281            let (remaining, blob_len) = parse_vuint(&data[*offset..]).map_err(|e| {
282                Error::corruption(format!(
283                    "Cell '{}': failed to parse blob length as VInt: {:?}",
284                    column.name, e
285                ))
286            })?;
287            let blob_len = blob_len as usize;
288            let bytes_consumed = data[*offset..].len() - remaining.len();
289            *offset += bytes_consumed;
290
291            if *offset + blob_len > data.len() {
292                return Err(Error::corruption(format!(
293                    "Cell '{}': need {} bytes for blob, only {} available (type: {})",
294                    column.name,
295                    blob_len,
296                    data.len() - *offset,
297                    column.data_type
298                )));
299            }
300
301            let blob_bytes = data[*offset..*offset + blob_len].to_vec();
302            *offset += blob_len;
303            Ok(Value::Blob(blob_bytes))
304        };
305
306        // At this point, we have a live cell with value data.
307        // Dispatch the decode on the precomputed `CellKind` (jump table). The scalar
308        // arms map 1:1 onto the pre-J1 string-match arms; the frozen/tuple/collection/
309        // marshal-UDT/default ladder is retained verbatim inside `CellKind::Complex`
310        // (the thin adapter Epic J2 collapses), using the tag's already-lowercased
311        // declared type as `type_str` — exactly the pre-J1 `normalized_type`.
312        let value = match kind {
313            CellKind::Blob => decode_vint_blob(&mut offset)?,
314            CellKind::Boolean => {
315                // Boolean: [0x08][u8 value]
316                if offset >= data.len() {
317                    return Err(Error::corruption(format!(
318                        "Cell '{}': unexpected end at boolean value",
319                        column.name
320                    )));
321                }
322                let bool_byte = data[offset];
323                offset += 1;
324                Value::Boolean(bool_byte != 0)
325            }
326
327            CellKind::Int => {
328                // Integer (i32): fixed-width 4 bytes (no length prefix in Cassandra 5.0)
329                if offset + 4 > data.len() {
330                    return Err(Error::corruption(format!(
331                        "Cell '{}': need 4 bytes for int, only {} available",
332                        column.name,
333                        data.len() - offset
334                    )));
335                }
336                let int_val = i32::from_be_bytes([
337                    data[offset],
338                    data[offset + 1],
339                    data[offset + 2],
340                    data[offset + 3],
341                ]);
342                offset += 4;
343                Value::Integer(int_val)
344            }
345
346            CellKind::Text => {
347                // Text: [VInt len][text bytes]
348                // V5CompressedLegacy uses VInt length encoding for variable-length types
349                let (remaining, text_len) = parse_vuint(&data[offset..]).map_err(|e| {
350                    Error::corruption(format!(
351                        "Cell '{}': failed to parse text length as VInt: {:?}",
352                        column.name, e
353                    ))
354                })?;
355                let text_len = text_len as usize;
356                let bytes_consumed = data[offset..].len() - remaining.len();
357                offset += bytes_consumed;
358
359                if offset + text_len > data.len() {
360                    return Err(Error::corruption(format!(
361                        "Cell '{}': need {} bytes for text, only {} available",
362                        column.name,
363                        text_len,
364                        data.len() - offset
365                    )));
366                }
367
368                let text_bytes = &data[offset..offset + text_len];
369                let text = String::from_utf8(text_bytes.to_vec()).map_err(|e| {
370                    Error::corruption(format!(
371                        "Cell '{}': invalid UTF-8 in text value: {}",
372                        column.name, e
373                    ))
374                })?;
375
376                offset += text_len;
377                Value::Text(text)
378            }
379
380            CellKind::Uuid => {
381                // UUID/TimeUUID: fixed-width 16 bytes (no length prefix in Cassandra 5.0 writer)
382                if offset + 16 > data.len() {
383                    return Err(Error::corruption(format!(
384                        "Cell '{}': need 16 bytes for UUID, only {} available",
385                        column.name,
386                        data.len() - offset
387                    )));
388                }
389
390                let uuid_bytes: [u8; 16] = data[offset..offset + 16]
391                    .try_into()
392                    .map_err(|_| Error::corruption("UUID byte conversion failed"))?;
393
394                offset += 16;
395                Value::Uuid(uuid_bytes)
396            }
397
398            CellKind::Decimal => {
399                // Decimal: [VInt total_len][i32 scale][unscaled bytes]
400                if offset >= data.len() {
401                    return Err(Error::corruption(format!(
402                        "Cell '{}': unexpected end at decimal length",
403                        column.name
404                    )));
405                }
406
407                let (remaining, total_len) = parse_vuint(&data[offset..]).map_err(|e| {
408                    Error::corruption(format!(
409                        "Cell '{}': failed to parse decimal length as VInt: {:?}",
410                        column.name, e
411                    ))
412                })?;
413                let total_len = total_len as usize;
414                let bytes_consumed = data[offset..].len() - remaining.len();
415                offset += bytes_consumed;
416
417                if offset + total_len > data.len() {
418                    return Err(Error::corruption(format!(
419                        "Cell '{}': need {} bytes for decimal, only {} available",
420                        column.name,
421                        total_len,
422                        data.len() - offset
423                    )));
424                }
425
426                // First 4 bytes: scale (i32 BE)
427                if total_len < 4 {
428                    return Err(Error::corruption(format!(
429                        "Cell '{}': decimal length {} too small for scale",
430                        column.name, total_len
431                    )));
432                }
433                let scale = i32::from_be_bytes([
434                    data[offset],
435                    data[offset + 1],
436                    data[offset + 2],
437                    data[offset + 3],
438                ]);
439
440                // Remaining bytes: unscaled value
441                let unscaled = data[offset + 4..offset + total_len].to_vec();
442                offset += total_len;
443
444                Value::Decimal { scale, unscaled }
445            }
446
447            CellKind::BigInt => {
448                // BigInt: fixed-width 8 bytes (no length prefix in Cassandra 5.0)
449                if offset + 8 > data.len() {
450                    return Err(Error::corruption(format!(
451                        "Cell '{}': need 8 bytes for bigint, only {} available",
452                        column.name,
453                        data.len() - offset
454                    )));
455                }
456                let val = i64::from_be_bytes([
457                    data[offset],
458                    data[offset + 1],
459                    data[offset + 2],
460                    data[offset + 3],
461                    data[offset + 4],
462                    data[offset + 5],
463                    data[offset + 6],
464                    data[offset + 7],
465                ]);
466                offset += 8;
467                Value::BigInt(val)
468            }
469
470            CellKind::Counter => {
471                // Counter cells can arrive in two formats:
472                //
473                // 1. Real Cassandra CounterContext: [VInt length][header_size:i16][indices][shards]
474                //    The counter value is the sum of all shard counts.
475                //
476                // 2. CQLite writer format (raw i64): [VInt(8)][8 bytes big-endian i64]
477                //    The writer serialises Value::Counter as a plain 8-byte integer with a
478                //    length prefix of 8, identical to how BigInt is written.
479                //
480                // We try CounterContext first and fall back to the raw-i64 interpretation
481                // when the length prefix equals exactly 8 (the size of a raw i64).
482
483                // Read the VInt length prefix.
484                let (remaining, context_len) = parse_vuint(&data[offset..]).map_err(|e| {
485                    Error::corruption(format!(
486                        "Cell '{}': failed to parse counter context length as VInt: {:?}",
487                        column.name, e
488                    ))
489                })?;
490                let context_len = context_len as usize;
491                let len_bytes_consumed = data[offset..].len() - remaining.len();
492                offset += len_bytes_consumed;
493
494                tracing::debug!(
495                    "V5CompressedLegacy: Counter '{}' context_len={} (len prefix: {} bytes)",
496                    column.name,
497                    context_len,
498                    len_bytes_consumed
499                );
500
501                if offset + context_len > data.len() {
502                    return Err(Error::corruption(format!(
503                        "Cell '{}': need {} bytes for counter context, only {} available",
504                        column.name,
505                        context_len,
506                        data.len() - offset
507                    )));
508                }
509
510                // Try the full CounterContext parse first.
511                match Self::parse_counter_context(data, offset, &column.name) {
512                    Ok((total, consumed)) if consumed == context_len => {
513                        // Successfully parsed a proper CounterContext.
514                        offset += consumed;
515                        tracing::debug!(
516                            "V5CompressedLegacy: Counter '{}' value={} (CounterContext), total consumed {} bytes",
517                            column.name,
518                            total,
519                            len_bytes_consumed + context_len
520                        );
521                        Value::Counter(total)
522                    }
523                    _ if context_len == 8 => {
524                        // A real Cassandra CounterContext is at minimum 36 bytes
525                        // (2 header + 2 indices + 32 body for 1 shard), so
526                        // context_len == 8 can only be produced by the CQLite writer
527                        // which serialises Counter as a raw big-endian i64.
528                        // This intentionally swallows any parse_counter_context error
529                        // for 8-byte payloads, which is safe since a valid
530                        // CounterContext can never be 8 bytes.
531                        //
532                        // Bounds already verified by the context_len check above.
533                        let val = i64::from_be_bytes([
534                            data[offset],
535                            data[offset + 1],
536                            data[offset + 2],
537                            data[offset + 3],
538                            data[offset + 4],
539                            data[offset + 5],
540                            data[offset + 6],
541                            data[offset + 7],
542                        ]);
543                        offset += 8;
544                        tracing::debug!(
545                            "V5CompressedLegacy: Counter '{}' value={} (raw i64 fallback), total consumed {} bytes",
546                            column.name,
547                            val,
548                            len_bytes_consumed + 8
549                        );
550                        Value::Counter(val)
551                    }
552                    Err(e) => return Err(e),
553                    Ok((_, consumed)) => {
554                        return Err(Error::corruption(format!(
555                            "Counter '{}': VInt length ({}) doesn't match parsed context size ({})",
556                            column.name, context_len, consumed
557                        )));
558                    }
559                }
560            }
561
562            CellKind::Double => {
563                // Double: 8 bytes, f64 big-endian (NO length prefix)
564                if offset + 8 > data.len() {
565                    return Err(Error::corruption(format!(
566                        "Cell '{}': need 8 bytes for double, only {} available",
567                        column.name,
568                        data.len() - offset
569                    )));
570                }
571                let val = f64::from_be_bytes([
572                    data[offset],
573                    data[offset + 1],
574                    data[offset + 2],
575                    data[offset + 3],
576                    data[offset + 4],
577                    data[offset + 5],
578                    data[offset + 6],
579                    data[offset + 7],
580                ]);
581                offset += 8;
582                Value::Float(val)
583            }
584
585            CellKind::Timestamp => {
586                // Timestamp: 8 bytes, i64 milliseconds big-endian (NO length prefix, per Cassandra spec)
587                if offset + 8 > data.len() {
588                    return Err(Error::corruption(format!(
589                        "Cell '{}': need 8 bytes for timestamp, only {} available",
590                        column.name,
591                        data.len() - offset
592                    )));
593                }
594                let millis = i64::from_be_bytes([
595                    data[offset],
596                    data[offset + 1],
597                    data[offset + 2],
598                    data[offset + 3],
599                    data[offset + 4],
600                    data[offset + 5],
601                    data[offset + 6],
602                    data[offset + 7],
603                ]);
604                offset += 8;
605                Value::Timestamp(millis)
606            }
607
608            CellKind::Date => {
609                // Date: [VInt len=4][i32 BE days]
610                if offset >= data.len() {
611                    return Err(Error::corruption(format!(
612                        "Cell '{}': unexpected end at date length",
613                        column.name
614                    )));
615                }
616
617                let (remaining, date_len) = parse_vuint(&data[offset..]).map_err(|e| {
618                    Error::corruption(format!(
619                        "Cell '{}': failed to parse date length as VInt: {:?}",
620                        column.name, e
621                    ))
622                })?;
623                let date_len = date_len as usize;
624                let bytes_consumed = data[offset..].len() - remaining.len();
625                offset += bytes_consumed;
626
627                if date_len != 4 {
628                    return Err(Error::corruption(format!(
629                        "Cell '{}': expected date length 4, got {}",
630                        column.name, date_len
631                    )));
632                }
633
634                if offset + 4 > data.len() {
635                    return Err(Error::corruption(format!(
636                        "Cell '{}': need 4 bytes for date, only {} available",
637                        column.name,
638                        data.len() - offset
639                    )));
640                }
641
642                let stored = u32::from_be_bytes([
643                    data[offset],
644                    data[offset + 1],
645                    data[offset + 2],
646                    data[offset + 3],
647                ]);
648                offset += 4;
649                // Cassandra DATE: 4-byte unsigned int with Integer.MIN_VALUE offset
650                let days_since_epoch = stored.wrapping_add(i32::MIN as u32) as i32;
651                Value::Date(days_since_epoch)
652            }
653
654            CellKind::Duration => {
655                // Duration: [VInt len][months VInt][days VInt][nanos VInt]
656                // Format: Variable-length encoding with 3 VInt components
657                if offset >= data.len() {
658                    return Err(Error::corruption(format!(
659                        "Cell '{}': unexpected end at duration length",
660                        column.name
661                    )));
662                }
663
664                let (remaining, duration_len) = parse_vuint(&data[offset..]).map_err(|e| {
665                    Error::corruption(format!(
666                        "Cell '{}': failed to parse duration length as VInt: {:?}",
667                        column.name, e
668                    ))
669                })?;
670                let duration_len = duration_len as usize;
671                let bytes_consumed = data[offset..].len() - remaining.len();
672                offset += bytes_consumed;
673
674                if offset + duration_len > data.len() {
675                    return Err(Error::corruption(format!(
676                        "Cell '{}': need {} bytes for duration, only {} available",
677                        column.name,
678                        duration_len,
679                        data.len() - offset
680                    )));
681                }
682
683                // Parse three VInt components from the duration_len bytes
684                let duration_bytes = &data[offset..offset + duration_len];
685
686                // Parse months (signed VInt)
687                let (remaining, months) = parse_vint(duration_bytes).map_err(|e| {
688                    Error::corruption(format!(
689                        "Cell '{}': failed to parse duration months: {:?}",
690                        column.name, e
691                    ))
692                })?;
693                let pos = duration_bytes.len() - remaining.len();
694
695                // Parse days (signed VInt)
696                let (remaining, days) = parse_vint(&duration_bytes[pos..]).map_err(|e| {
697                    Error::corruption(format!(
698                        "Cell '{}': failed to parse duration days: {:?}",
699                        column.name, e
700                    ))
701                })?;
702                let pos = duration_bytes.len() - remaining.len();
703
704                // Parse nanoseconds (signed VInt)
705                let (remaining, nanos) = parse_vint(&duration_bytes[pos..]).map_err(|e| {
706                    Error::corruption(format!(
707                        "Cell '{}': failed to parse duration nanos: {:?}",
708                        column.name, e
709                    ))
710                })?;
711
712                // Verify we consumed all duration bytes
713                if !remaining.is_empty() {
714                    warn!(
715                        "V5CompressedLegacy: Duration '{}' has {} extra bytes after parsing",
716                        column.name,
717                        remaining.len()
718                    );
719                }
720
721                // months/days are i32 in Cassandra's DurationType. Reject
722                // (rather than silently truncate via `as i32`) any encoded value
723                // outside the i32 range so a corrupt encoding errors instead of
724                // wrapping (issue #1632, item b).
725                let months = i32::try_from(months).map_err(|_| {
726                    Error::corruption(format!(
727                        "Cell '{}': duration months out of i32 range",
728                        column.name
729                    ))
730                })?;
731                let days = i32::try_from(days).map_err(|_| {
732                    Error::corruption(format!(
733                        "Cell '{}': duration days out of i32 range",
734                        column.name
735                    ))
736                })?;
737
738                offset += duration_len;
739                Value::Duration {
740                    months,
741                    days,
742                    nanos,
743                }
744            }
745
746            CellKind::Float => {
747                // Float: 4 bytes, f32 big-endian (NO length prefix, fixed size)
748                if offset + 4 > data.len() {
749                    return Err(Error::corruption(format!(
750                        "Cell '{}': need 4 bytes for float, only {} available",
751                        column.name,
752                        data.len() - offset
753                    )));
754                }
755
756                let val = f32::from_be_bytes([
757                    data[offset],
758                    data[offset + 1],
759                    data[offset + 2],
760                    data[offset + 3],
761                ]);
762                offset += 4;
763                Value::Float(val as f64) // Convert f32 to f64 for storage
764            }
765
766            CellKind::SmallInt => {
767                // SmallInt: [VInt len=2][i16 BE]
768                if offset >= data.len() {
769                    return Err(Error::corruption(format!(
770                        "Cell '{}': unexpected end at smallint length",
771                        column.name
772                    )));
773                }
774
775                let (remaining, len) = parse_vuint(&data[offset..]).map_err(|e| {
776                    Error::corruption(format!(
777                        "Cell '{}': failed to parse smallint length as VInt: {:?}",
778                        column.name, e
779                    ))
780                })?;
781                let len = len as usize;
782                let bytes_consumed = data[offset..].len() - remaining.len();
783                offset += bytes_consumed;
784
785                if len != 2 {
786                    return Err(Error::corruption(format!(
787                        "Cell '{}': expected smallint length 2, got {}",
788                        column.name, len
789                    )));
790                }
791
792                if offset + 2 > data.len() {
793                    return Err(Error::corruption(format!(
794                        "Cell '{}': need 2 bytes for smallint, only {} available",
795                        column.name,
796                        data.len() - offset
797                    )));
798                }
799
800                let val = i16::from_be_bytes([data[offset], data[offset + 1]]);
801                offset += 2;
802                Value::SmallInt(val)
803            }
804
805            CellKind::TinyInt => {
806                // TinyInt: [VInt len=1][i8]
807                if offset >= data.len() {
808                    return Err(Error::corruption(format!(
809                        "Cell '{}': unexpected end at tinyint length",
810                        column.name
811                    )));
812                }
813
814                let (remaining, len) = parse_vuint(&data[offset..]).map_err(|e| {
815                    Error::corruption(format!(
816                        "Cell '{}': failed to parse tinyint length as VInt: {:?}",
817                        column.name, e
818                    ))
819                })?;
820                let len = len as usize;
821                let bytes_consumed = data[offset..].len() - remaining.len();
822                offset += bytes_consumed;
823
824                if len != 1 {
825                    return Err(Error::corruption(format!(
826                        "Cell '{}': expected tinyint length 1, got {}",
827                        column.name, len
828                    )));
829                }
830
831                if offset >= data.len() {
832                    return Err(Error::corruption(format!(
833                        "Cell '{}': need 1 byte for tinyint, only {} available",
834                        column.name,
835                        data.len() - offset
836                    )));
837                }
838
839                let val = data[offset] as i8;
840                offset += 1;
841                Value::TinyInt(val)
842            }
843
844            CellKind::Time => {
845                // Time: [VInt len=8][i64 BE nanoseconds since midnight]
846                if offset >= data.len() {
847                    return Err(Error::corruption(format!(
848                        "Cell '{}': unexpected end at time length",
849                        column.name
850                    )));
851                }
852                let (remaining, time_len) = parse_vuint(&data[offset..]).map_err(|e| {
853                    Error::corruption(format!(
854                        "Cell '{}': failed to parse time length as VInt: {:?}",
855                        column.name, e
856                    ))
857                })?;
858                let time_len = time_len as usize;
859                let bytes_consumed = data[offset..].len() - remaining.len();
860                offset += bytes_consumed;
861                if time_len != 8 {
862                    return Err(Error::corruption(format!(
863                        "Cell '{}': expected time length 8, got {}",
864                        column.name, time_len
865                    )));
866                }
867                if offset + 8 > data.len() {
868                    return Err(Error::corruption(format!(
869                        "Cell '{}': need 8 bytes for time value, only {} available",
870                        column.name,
871                        data.len() - offset
872                    )));
873                }
874                let nanos = i64::from_be_bytes([
875                    data[offset],
876                    data[offset + 1],
877                    data[offset + 2],
878                    data[offset + 3],
879                    data[offset + 4],
880                    data[offset + 5],
881                    data[offset + 6],
882                    data[offset + 7],
883                ]);
884                offset += 8;
885                Value::Time(nanos)
886            }
887
888            CellKind::Inet => {
889                // Inet: [VInt len][address bytes] (len is 4 for IPv4, 16 for IPv6)
890                if offset >= data.len() {
891                    return Err(Error::corruption(format!(
892                        "Cell '{}': unexpected end at inet length",
893                        column.name
894                    )));
895                }
896
897                let (remaining, len) = parse_vuint(&data[offset..]).map_err(|e| {
898                    Error::corruption(format!(
899                        "Cell '{}': failed to parse inet length as VInt: {:?}",
900                        column.name, e
901                    ))
902                })?;
903                let len = len as usize;
904                let bytes_consumed = data[offset..].len() - remaining.len();
905                offset += bytes_consumed;
906
907                if len != 4 && len != 16 {
908                    return Err(Error::corruption(format!(
909                        "Cell '{}': invalid inet length {}, expected 4 or 16",
910                        column.name, len
911                    )));
912                }
913
914                if offset + len > data.len() {
915                    return Err(Error::corruption(format!(
916                        "Cell '{}': need {} bytes for inet, only {} available",
917                        column.name,
918                        len,
919                        data.len() - offset
920                    )));
921                }
922
923                let bytes = data[offset..offset + len].to_vec();
924                offset += len;
925                Value::Inet(bytes)
926            }
927
928            // Complex types: frozen, tuple, non-frozen collection, marshal-UDT, and
929            // any unnamed/default type (e.g. `varint`). J1 (issue #1635): the DISPATCH
930            // is per-column via this single `CellKind::Complex` arm; the retained
931            // prefix ladder (Epic J2 collapses it) uses the tag's already-lowercased
932            // declared type as `type_str` — exactly the pre-J1 `normalized_type`, so
933            // there is NO per-cell `to_lowercase`. The ladder ordering is unchanged.
934            CellKind::Complex(lowered) => {
935                let type_str: &str = lowered;
936                if type_str.starts_with("frozen<") {
937                    // Frozen types: unwrap inner type and route to appropriate parser
938                    let inner_type = self.extract_frozen_inner_type(type_str)?;
939
940                    tracing::debug!(
941                        "V5CompressedLegacy: Parsing frozen type '{}' -> inner type '{}'",
942                        type_str,
943                        inner_type
944                    );
945
946                    // Issue #1340: extract the AUTHORITATIVE marshal element type(s)
947                    // from the on-disk SerializationHeader marshal type (`header_type`).
948                    // When an element is a `frozen<UDT>`, threading the marshal type lets
949                    // it decode to a typed `Value::Frozen(Value::Udt)` registry-free
950                    // (precedence: header marshal → registry → Blob, no byte-pattern
951                    // inference — no-heuristics #28). Extracted once per frozen cell
952                    // (before the element loop); the result borrows from `header_type`,
953                    // so the per-element loop is allocation-free.
954                    let marshal_elems =
955                        header_type.and_then(Self::extract_marshal_collection_elements);
956                    // Shared for list & set (both are `Sequence`): the borrowed element
957                    // marshal type, or `None` for a map / absent / mismatched marshal.
958                    let sequence_marshal_elem = match &marshal_elems {
959                        Some(MarshalCollectionElements::Sequence(m)) => Some(*m),
960                        _ => None,
961                    };
962
963                    // Route to appropriate frozen collection parser
964                    let (inner_value, new_offset) = if inner_type.starts_with("list<") {
965                        let schema_elem =
966                            self.extract_collection_element_type(&inner_type, "list")?;
967                        let element_type =
968                            Self::prefer_udt_marshal_element(sequence_marshal_elem, &schema_elem);
969                        self.parse_frozen_list_value(data, offset, element_type, column, _reader)?
970                    } else if inner_type.starts_with("set<") {
971                        let schema_elem =
972                            self.extract_collection_element_type(&inner_type, "set")?;
973                        let element_type =
974                            Self::prefer_udt_marshal_element(sequence_marshal_elem, &schema_elem);
975                        self.parse_frozen_set_value(data, offset, element_type, column, _reader)?
976                    } else if inner_type.starts_with("map<") {
977                        let (schema_key, schema_val) = self.extract_map_types(&inner_type)?;
978                        let (marshal_key, marshal_val) = match &marshal_elems {
979                            Some(MarshalCollectionElements::Map(k, v)) => (Some(*k), Some(*v)),
980                            _ => (None, None),
981                        };
982                        let key_type = Self::prefer_udt_marshal_element(marshal_key, &schema_key);
983                        let value_type = Self::prefer_udt_marshal_element(marshal_val, &schema_val);
984                        self.parse_frozen_map_value(
985                            data, offset, key_type, value_type, column, _reader,
986                        )?
987                    } else if Self::is_udt_type(&column.data_type) {
988                        // Frozen UDT - parse using UDT parser
989                        // The column.data_type contains the full Cassandra type string including UserType
990                        tracing::debug!(
991                            "V5CompressedLegacy: Parsing frozen UDT column '{}' type='{}'",
992                            column.name,
993                            column.data_type
994                        );
995
996                        // Parse UDT definition from the type string
997                        let udt_def = Self::parse_udt_type_definition(&column.data_type)?;
998
999                        // First read the VInt-prefixed blob length
1000                        let (remaining, blob_len_raw) =
1001                            parse_vuint(&data[offset..]).map_err(|e| {
1002                                Error::corruption(format!(
1003                                    "Frozen UDT '{}': failed to parse blob length: {:?}",
1004                                    column.name, e
1005                                ))
1006                            })?;
1007                        if blob_len_raw > MAX_CELL_VALUE_LENGTH {
1008                            return Err(Error::corruption(format!(
1009                                "Frozen UDT '{}': blob_len {} exceeds maximum {}",
1010                                column.name, blob_len_raw, MAX_CELL_VALUE_LENGTH
1011                            )));
1012                        }
1013                        let blob_len = blob_len_raw as usize;
1014                        let bytes_consumed = data[offset..].len() - remaining.len();
1015                        offset += bytes_consumed;
1016
1017                        if offset + blob_len > data.len() {
1018                            return Err(Error::corruption(format!(
1019                                "Frozen UDT '{}': need {} bytes but only {} available",
1020                                column.name,
1021                                blob_len,
1022                                data.len() - offset
1023                            )));
1024                        }
1025
1026                        // Parse UDT value from the blob
1027                        let udt_data = &data[offset..offset + blob_len];
1028                        let (udt_value, _) = self.parse_udt_value(udt_data, 0, &udt_def, column)?;
1029                        offset += blob_len;
1030
1031                        (udt_value, offset)
1032                    } else if let Some(udt_def) = self
1033                        .udt_registry
1034                        .as_ref()
1035                        .and_then(|reg| reg.get_udt(&self.keyspace, &inner_type).cloned())
1036                    {
1037                        // frozen<short_udt_name>: look up the concrete UDT definition in the
1038                        // registry (Issue #502).  This handles type strings like
1039                        // `frozen<person>` where "person" is a registered UDT rather than a
1040                        // collection or a full marshal-format UserType string.
1041                        tracing::debug!(
1042                        "V5CompressedLegacy: Resolving frozen UDT '{}' via registry for column '{}'",
1043                        inner_type,
1044                        column.name,
1045                    );
1046
1047                        // Read VUInt-prefixed blob length (same framing as tuple and
1048                        // marshal-format UDT cells).
1049                        let (remaining, blob_len_raw) =
1050                            parse_vuint(&data[offset..]).map_err(|e| {
1051                                Error::corruption(format!(
1052                            "Frozen UDT '{}' (column '{}'): failed to parse blob length: {:?}",
1053                            inner_type, column.name, e
1054                        ))
1055                            })?;
1056                        if blob_len_raw > MAX_CELL_VALUE_LENGTH {
1057                            return Err(Error::corruption(format!(
1058                                "Frozen UDT '{}' (column '{}'): blob_len {} exceeds maximum {}",
1059                                inner_type, column.name, blob_len_raw, MAX_CELL_VALUE_LENGTH
1060                            )));
1061                        }
1062                        let blob_len = blob_len_raw as usize;
1063                        let len_bytes_consumed = data[offset..].len() - remaining.len();
1064                        offset += len_bytes_consumed;
1065
1066                        if offset + blob_len > data.len() {
1067                            return Err(Error::corruption(format!(
1068                            "Frozen UDT '{}' (column '{}'): need {} bytes but only {} available",
1069                            inner_type,
1070                            column.name,
1071                            blob_len,
1072                            data.len() - offset
1073                        )));
1074                        }
1075
1076                        let udt_data = &data[offset..offset + blob_len];
1077                        let (udt_value, _) = self.parse_udt_value(udt_data, 0, &udt_def, column)?;
1078                        offset += blob_len;
1079
1080                        (udt_value, offset)
1081                    } else if let Some(ht) =
1082                        header_type.filter(|ht| Self::marshal_is_top_level_frozen_udt(ht))
1083                    {
1084                        // Issue #1080: NO UdtRegistry is wired and the supplied schema
1085                        // short form `frozen<person_type>` carries no field defs, but
1086                        // the AUTHORITATIVE on-disk SerializationHeader marshal type for
1087                        // this column is the full
1088                        // `FrozenType(UserType(ks,hexname,field:Type,...))`. Decode the
1089                        // UDT STRUCTURALLY from that header type (no guessing, issue #28)
1090                        // rather than dropping the column (which also broke the Err→break
1091                        // loop, silently losing all trailing columns).
1092                        self.decode_frozen_udt_from_header_type(data, offset, ht, column)?
1093                    } else {
1094                        // Detect bare identifiers that look like unregistered UDT names.
1095                        // A bare identifier has no '<' (not a container or tuple) and does not
1096                        // match any known CQL primitive type.  If we reach this branch with
1097                        // such an identifier it means the UDT was not in the registry — return
1098                        // an actionable schema error rather than silently producing a Blob.
1099                        //
1100                        // Legitimate fall-through types handled below:
1101                        //   • tuple<...>  (contains '<')
1102                        //   • known primitives: int, text, uuid, boolean, blob, float, double,
1103                        //     decimal, varint, bigint, counter, timestamp, date, time, duration,
1104                        //     inet, smallint, tinyint, varchar, ascii, timeuuid
1105                        const KNOWN_PRIMITIVES: &[&str] = &[
1106                            "int",
1107                            "bigint",
1108                            "counter",
1109                            "smallint",
1110                            "tinyint",
1111                            "text",
1112                            "varchar",
1113                            "ascii",
1114                            "uuid",
1115                            "timeuuid",
1116                            "boolean",
1117                            "blob",
1118                            "float",
1119                            "double",
1120                            "decimal",
1121                            "varint",
1122                            "timestamp",
1123                            "date",
1124                            "time",
1125                            "duration",
1126                            "inet",
1127                        ];
1128                        let is_container = inner_type.contains('<');
1129                        let is_primitive = KNOWN_PRIMITIVES.contains(&inner_type.as_str());
1130                        if !is_container && !is_primitive {
1131                            // Bare identifier that is neither a container nor a primitive —
1132                            // this is an unregistered UDT name.
1133                            return Err(Error::schema(format!(
1134                            "frozen<{inner}>: UDT '{inner}' not found in registry for keyspace '{}'; \
1135                             register it before reading",
1136                            self.keyspace,
1137                            inner = inner_type,
1138                        )));
1139                        }
1140                        // Non-collection / primitive frozen type — recurse normally.
1141                        // The recursive call now returns 4 elements; we only need value + offset.
1142                        let mut inner_column = column.clone();
1143                        inner_column.data_type = inner_type.clone();
1144                        let (inner_val, _inner_ts, _inner_exp, inner_off) = self
1145                            .parse_cell_value_schema_order(
1146                                data,
1147                                offset,
1148                                &inner_column,
1149                                None,
1150                                // Frozen-inner recursion: resolve the tag locally from
1151                                // `inner_column.data_type` (bounded, off the per-cell scan
1152                                // hot path).
1153                                None,
1154                                _reader,
1155                            )?;
1156                        (inner_val, inner_off)
1157                    };
1158
1159                    offset = new_offset;
1160
1161                    // Wrap in Frozen
1162                    Value::Frozen(Box::new(inner_value))
1163                } else if type_str.starts_with("tuple<") {
1164                    // Tuple types: parse fixed number of elements
1165                    self.parse_tuple_value(data, &mut offset, type_str, column, _reader)?
1166                }
1167                // Non-frozen collections: list, set, map
1168                // TODO(Issue #162, Task 3): Multi-cell collection parsing
1169                //
1170                // Collections in V5CompressedLegacy are stored as MULTIPLE CELLS with path identifiers,
1171                // NOT as single blob values. The current single-cell parser cannot handle this.
1172                //
1173                // Format (from sstabledump analysis):
1174                //   {"name": "scores", "deletion_info": {...}},  // Collection tombstone
1175                //   {"name": "scores", "path": ["uuid1"], "value": 23},  // Element 1
1176                //   {"name": "scores", "path": ["uuid2"], "value": 99},  // Element 2
1177                //
1178                // Required implementation:
1179                //   1. Parse cell path (clustering key bytes) for each collection element
1180                //   2. Detect collection tombstone cell (has deletion_info, no path/value)
1181                //   3. Read N element cells (each with path + value)
1182                //   4. Aggregate elements into Value::List/Set/Map based on column type
1183                //   5. Handle different path encodings:
1184                //      - list<T>: path is UUID bytes (timeuuid for ordering)
1185                //      - set<T>: path is serialized element value (key), value is empty
1186                //      - map<K,V>: path is serialized key, value is serialized value
1187                //
1188                // This is a fundamental architectural change requiring cell-level parsing
1189                // before column-level aggregation. For now, return stub to unblock downstream work.
1190                else if type_str.starts_with("list<")
1191                    || type_str.starts_with("set<")
1192                    || type_str.starts_with("map<")
1193                {
1194                    warn!(
1195                    "V5CompressedLegacy: Non-frozen collection '{}' type '{}' requires multi-cell parsing (not yet implemented). \
1196                     Collections are stored as multiple cells with path identifiers, requiring cell-level aggregation. \
1197                     Returning empty collection as placeholder. See Issue #162 Task 3 for implementation plan.",
1198                    column.name, column.data_type
1199                );
1200
1201                    // Return empty collection based on type
1202                    if type_str.starts_with("list<") {
1203                        Value::List(Vec::new())
1204                    } else if type_str.starts_with("set<") {
1205                        Value::Set(Vec::new())
1206                    } else {
1207                        Value::Map(Vec::new())
1208                    }
1209                }
1210                // Issue #1080 / roborev job 1363: marshal-form frozen UDT. When the
1211                // schema is DERIVED FROM the on-disk header (rather than supplied as a
1212                // CQL short form), `column.data_type` is the authoritative marshal
1213                // string `org.apache.cassandra.db.marshal.FrozenType(...UserType...)`,
1214                // which does NOT start with CQL `frozen<` and so misses the arm above.
1215                // Decode it structurally from that marshal type (same authoritative
1216                // path as the supplied-schema header fallback) instead of blobbing it.
1217                // `marshal_is_top_level_frozen_udt` accepts ONLY a top-level
1218                // `FrozenType(UserType(...))` (NOT a frozen collection that contains a
1219                // UDT, e.g. `FrozenType(ListType(UserType(...)))` — roborev 1365), and a
1220                // non-frozen top-level UDT is routed to the complex branch by
1221                // `is_complex_column`, so reaching here means a single-cell frozen UDT
1222                // → wrap in `Value::Frozen` (consistent with the CQL `frozen<` arm).
1223                else if Self::marshal_is_top_level_frozen_udt(&column.data_type) {
1224                    let (udt_value, new_offset) = self.decode_frozen_udt_from_header_type(
1225                        data,
1226                        offset,
1227                        &column.data_type,
1228                        column,
1229                    )?;
1230                    offset = new_offset;
1231                    Value::Frozen(Box::new(udt_value))
1232                }
1233                // TODO(Issue #162): UDT parsing requires schema registry access
1234                // For now, UDTs fall through to blob. Future implementation will:
1235                // - Extract UDT name from type_str
1236                // - Look up UDT definition in schema registry
1237                // - Parse fields according to UDT schema
1238                // - Return Value::Udt(UdtValue)
1239
1240                // Default: treat as VInt-length-prefixed blob (unknown/`varint`).
1241                // Shares the `decode_vint_blob` closure with the literal-`blob`
1242                // `CellKind::Blob` arm — identical decode pre-J1 (both hit `_ =>`).
1243                else {
1244                    decode_vint_blob(&mut offset)?
1245                }
1246            }
1247        };
1248
1249        Ok((value, cell_timestamp, cell_expiration, offset))
1250    }
1251}
