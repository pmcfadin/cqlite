15    /// Issue #1080: is this on-disk SerializationHeader marshal type a TOP-LEVEL
16    /// frozen (or bare) UDT — `FrozenType(UserType(...))` or `UserType(...)` — that
17    /// may be decoded as a single UDT blob via `decode_frozen_udt_from_header_type`?
18    ///
19    /// Returns FALSE for a frozen COLLECTION that merely CONTAINS a UDT, e.g.
20    /// `FrozenType(ListType(UserType(...)))` or `FrozenType(MapType(...UserType...))`
21    /// — those must go through the collection decoders, NOT the scalar UDT path; a
22    /// substring `UserType(` test would misroute them and corrupt the value or break
23    /// the row loop, dropping trailing columns (roborev job 1365).
24    ///
25    /// We match ONLY the fully-qualified marshal form (the single shape real
26    /// SerializationHeaders carry, preserved verbatim by `convert_marshal_type_to_cql`)
27    /// and require `UserType(` to be the IMMEDIATE inner type, after an optional
28    /// single `FrozenType(` wrapper. The whole decode chain
29    /// (`parse_udt_type_definition` + `parse_cassandra_type_with_depth`) keys on the
30    /// same qualified marker at every nesting level (roborev jobs 1359/1361).
31    pub(super) fn marshal_is_top_level_frozen_udt(marshal_type: &str) -> bool {
32        const FROZEN: &str = "org.apache.cassandra.db.marshal.frozentype(";
33        const USERTYPE: &str = "org.apache.cassandra.db.marshal.usertype(";
34        let lower = marshal_type.trim().to_lowercase();
35        // Strip at most one leading FrozenType( wrapper; then UserType( must be the
36        // immediate inner type (rejecting FrozenType(ListType(UserType(...))) etc.).
37        let inner = lower.strip_prefix(FROZEN).unwrap_or(&lower);
38        inner.starts_with(USERTYPE)
39    }

41    /// Issue #1080: decode a single-cell frozen UDT value using the AUTHORITATIVE
42    /// on-disk SerializationHeader marshal type (the fully-qualified
43    /// `FrozenType(UserType(...))` marshal string), used when no `UdtRegistry` is
44    /// wired and the supplied schema short form (`frozen<person_type>`) carries no
45    /// field defs.
46    ///
47    /// `parse_udt_type_definition` does a case-insensitive find for the qualified
48    /// `UserType(` marker, so
49    /// the `FrozenType(...)` wrapper is transparently handled and nested
50    /// `FrozenType(UserType(...))` fields resolve to `CqlType::Frozen(CqlType::Udt)`
51    /// (decoded recursively by `parse_udt_field_value`). Returns the decoded UDT
52    /// value (UNWRAPPED — the caller wraps it in `Value::Frozen`) and the new offset
53    /// AFTER the VInt-prefixed blob, so trailing columns stay byte-aligned.
54    pub(super) fn decode_frozen_udt_from_header_type(
55        &self,
56        data: &[u8],
57        mut offset: usize,
58        header_type: &str,
59        column: &crate::schema::Column,
60    ) -> Result<(Value, usize)> {
61        let udt_def = Self::parse_udt_type_definition(header_type)?;
62
63        // Read the VInt-prefixed blob length (same framing as the registry /
64        // marshal-format UDT blocks in the frozen< arm).
65        let (remaining, blob_len_raw) = parse_vuint(&data[offset..]).map_err(|e| {
66            Error::corruption(format!(
67                "Frozen UDT (column '{}', on-disk header type): failed to parse blob length: {:?}",
68                column.name, e
69            ))
70        })?;
71        if blob_len_raw > MAX_CELL_VALUE_LENGTH {
72            return Err(Error::corruption(format!(
73                "Frozen UDT (column '{}', on-disk header type): blob_len {} exceeds maximum {}",
74                column.name, blob_len_raw, MAX_CELL_VALUE_LENGTH
75            )));
76        }
77        let blob_len = blob_len_raw as usize;
78        let len_bytes_consumed = data[offset..].len() - remaining.len();
79        offset += len_bytes_consumed;
80
81        if offset + blob_len > data.len() {
82            return Err(Error::corruption(format!(
83                "Frozen UDT (column '{}', on-disk header type): need {} bytes but only {} available",
84                column.name,
85                blob_len,
86                data.len() - offset
87            )));
88        }
89
90        let udt_data = &data[offset..offset + blob_len];
91        // ROOT nesting depth: this is a COLUMN-level decode (issue #3631; see
92        // `parse_udt_value`, which deliberately offers no zero-depth overload).
93        let (udt_value, n) = self.parse_udt_value(udt_data, 0, &udt_def, column, 0)?;
94        // #3811 (finding C): `parse_udt_value` REPORTS; this caller used to drop it.
95        Self::require_fully_consumed(n, udt_data.len(), &column.name, "frozen UDT")?;
96        offset += blob_len;
97
98        Ok((udt_value, offset))
99    }

101    /// Extract inner type from a frozen type string (CQL or Cassandra internal format).
102    ///
103    /// Accepts both the CQL short form `frozen<T>` and the authoritative on-disk
104    /// marshal form `org.apache.cassandra.db.marshal.FrozenType(T)`. Multicell-UDT
105    /// field types resolve from the `UserType(...)` marshal string, where frozen
106    /// fields are spelled `FrozenType(...)` (e.g. `frozen<list<int>>` →
107    /// `FrozenType(ListType(Int32Type))`), so we mirror how
108    /// `extract_collection_element_type` supports both forms. In both cases the
109    /// inner substring is sliced from the ORIGINAL-CASE input so nested marshal
110    /// types keep their case and re-normalize correctly on recursion.
111    pub(super) fn extract_frozen_inner_type(&self, type_str: &str) -> Result<String> {
112        let type_lower = type_str.to_lowercase();
113
114        // Cassandra internal format: org.apache.cassandra.db.marshal.FrozenType(T)
115        let internal_prefix_lower = "org.apache.cassandra.db.marshal.frozentype(";
116        if type_lower.starts_with(internal_prefix_lower) && type_lower.ends_with(')') {
117            let inner = &type_str[internal_prefix_lower.len()..type_str.len() - 1];
118            if inner.is_empty() {
119                return Err(Error::schema(format!("Empty frozen type: {}", type_str)));
120            }
121            return Ok(inner.to_string());
122        }
123
124        // CQL format: frozen<T> — match the prefix/suffix case-insensitively
125        // (parse_value_from_raw_bytes routes here off a lowercased guard but
126        // passes the ORIGINAL-CASE string, so a mixed-case `Frozen<...>` reaches
127        // this branch and must still be accepted) while slicing from the
128        // original string to preserve nested-type case.
129        if type_lower.starts_with("frozen<") && type_lower.ends_with('>') {
130            let inner = &type_str[7..type_str.len() - 1];
131            if inner.is_empty() {
132                return Err(Error::schema(format!("Empty frozen type: {}", type_str)));
133            }
134            return Ok(inner.to_string());
135        }
136
137        Err(Error::schema(format!(
138            "Invalid frozen type format: {}",
139            type_str
140        )))
141    }

143    /// Check if a type string represents a UDT (User-Defined Type).
144    /// Detects Cassandra's internal format: org.apache.cassandra.db.marshal.UserType(...)
145    pub(super) fn is_udt_type(type_str: &str) -> bool {
146        // ASCII case-insensitive substring match without allocating a lowercased
147        // copy. The marshal name is pure ASCII so byte-window comparison is safe.
148        const TARGET: &[u8] = b"org.apache.cassandra.db.marshal.usertype";
149        let bytes = type_str.as_bytes();
150        if bytes.len() < TARGET.len() {
151            return false;
152        }
153        bytes
154            .windows(TARGET.len())
155            .any(|w| w.iter().zip(TARGET).all(|(a, b)| a.eq_ignore_ascii_case(b)))
156    }

158    /// Parse a UDT value from binary data using the given UDT definition.
159    /// UDT binary format (frozen):
160    /// - For each field in schema order:
161    ///   - [4 bytes BE i32]: field length (-1 = null, 0 = empty, >0 = data length)
162    ///   - [N bytes]: field data (if length > 0)
163    // NOTE: this UDT decoder is purely structural — it reads the i32-length-prefixed
164    // field layout using only the [`UdtTypeDef`] field types. It does NOT need an
165    // [`SSTableReader`] (the previous `reader` param was threaded through but never
166    // dereferenced), so it is reader-free and unit-testable in isolation (issue #1080).
167    ///
168    /// `depth` is the CURRENT type-nesting level, threaded from the caller rather
169    /// than restarted here (issue #3631). This function recurses back into itself
170    /// through `parse_simple_udt_field_value_at`, so a `0` written here would make
171    /// `MAX_TYPE_NESTING_DEPTH` bound nothing across alternating collection/UDT
172    /// layers: each frozen-UDT hop would reset the counter and a cyclic
173    /// `UdtRegistry` — which the registry type permits even though CQL does not —
174    /// would recurse until the stack is exhausted. A caller genuinely at the root
175    /// (a column-level decode, or a test) writes the `0` at its own call site, where
176    /// a reviewer can see it.
177    pub(super) fn parse_udt_value(
178        &self,
179        data: &[u8],
180        offset: usize,
181        udt_def: &UdtTypeDef,
182        _column: &crate::schema::Column,
183        depth: usize,
184    ) -> Result<(Value, usize)> {
185        if depth > MAX_TYPE_NESTING_DEPTH {
186            return Err(Error::corruption(format!(
187                "UDT '{}': type nesting depth {} exceeds maximum {}",
188                udt_def.name, depth, MAX_TYPE_NESTING_DEPTH
189            )));
190        }
191        // Validate field count to prevent memory exhaustion
192        if udt_def.fields.len() > MAX_UDT_FIELD_COUNT {
193            return Err(Error::schema(format!(
194                "UDT '{}' has {} fields, exceeds maximum {}",
195                udt_def.name,
196                udt_def.fields.len(),
197                MAX_UDT_FIELD_COUNT
198            )));
199        }
200
201        let mut current_offset = offset;
202        let mut fields = Vec::with_capacity(udt_def.fields.len());
203
204        tracing::debug!(
205            "V5CompressedLegacy: Parsing UDT '{}' with {} fields at offset {}",
206            udt_def.name,
207            udt_def.fields.len(),
208            offset
209        );
210
211        for field_def in &udt_def.fields {
212            // Check bounds for field length (4 bytes)
213            if current_offset + 4 > data.len() {
214                // Trailing fields can be omitted (implicit null)
215                tracing::debug!(
216                    "V5CompressedLegacy: UDT field '{}' omitted (implicit null), remaining fields omitted",
217                    field_def.name
218                );
219                // Fill remaining fields with null
220                while fields.len() < udt_def.fields.len() {
221                    let remaining_field = &udt_def.fields[fields.len()];
222                    fields.push(UdtField {
223                        name: remaining_field.name.clone(),
224                        value: None,
225                    });
226                }
227                break;
228            }
229
230            // Read field length (4 bytes big-endian i32)
231            let field_len = i32::from_be_bytes([
232                data[current_offset],
233                data[current_offset + 1],
234                data[current_offset + 2],
235                data[current_offset + 3],
236            ]);
237            current_offset += 4;
238
239            let field_value = if field_len == -1 {
240                // Null field
241                tracing::debug!("V5CompressedLegacy: UDT field '{}' is null", field_def.name);
242                None
243            } else if field_len == 0 {
244                // Empty field - create empty value based on type
245                tracing::debug!(
246                    "V5CompressedLegacy: UDT field '{}' is empty",
247                    field_def.name
248                );
249                // A ZERO-LENGTH field is decoded from its DECLARED TYPE, exactly like
250                // a field WITH data (issue #3631). `create_empty_value_for_type`'s
251                // `_ =>` arm was an empty BLOB, so an empty `int`, an empty `tuple`
252                // and an empty nested UDT all surfaced as `Blob([])` — the same
253                // silent degradation as the non-empty arms, one length apart. The
254                // Cassandra rule (`isEmptyValueMeaningless` and, more precisely, the
255                // per-serializer `accessor.isEmpty(value) ? null : …` guard) is
256                // stated once, in `typed_value.rs::empty_is_a_value`.
257                Some(self.parse_simple_udt_field_value_at(&[], &field_def.field_type, depth)?)
258            } else {
259                // Field with data. `checked_component_len` owns BOTH the negative
260                // rejection and the bounds test, so no loop can have one without
261                // the other (issue #3612, R3-F1).
262                let field_len = Self::checked_component_len(
263                    field_len,
264                    &field_def.name,
265                    current_offset,
266                    data.len(),
267                )?;
268
269                let field_data = &data[current_offset..current_offset + field_len];
270                current_offset += field_len;
271
272                tracing::debug!(
273                    "V5CompressedLegacy: UDT field '{}' has {} bytes of data",
274                    field_def.name,
275                    field_len
276                );
277
278                // ONE per-field entry (issue #3631). `parse_udt_field_value` used to
279                // live below with its own ~140-line dispatch, a `Value::Blob` `_ =>`
280                // arm for every collection/tuple type, a `date` arm missing
281                // Cassandra's `SimpleDateType` epoch offset, and a nested-UDT arm that
282                // called back into this function with a hard `0` depth.
283                // `parse_simple_udt_field_value_at` expresses all of it once, threads
284                // `depth`, routes through the single exhaustion assert, and returns an
285                // explicit `Error` naming a type it cannot decode (#3631 criterion 5).
286                let value =
287                    self.parse_simple_udt_field_value_at(field_data, &field_def.field_type, depth)?;
288                Some(value)
289            };
290
291            fields.push(UdtField {
292                name: field_def.name.clone(),
293                value: field_value,
294            });
295        }
296
297        let udt_value = UdtValue {
298            type_name: udt_def.name.clone(),
299            keyspace: udt_def.keyspace.clone(),
300            fields,
301        };
302
303        Ok((Value::Udt(Box::new(udt_value)), current_offset))
304    }

306    /// Parse a CounterContext structure and return the total counter value.
307    ///
308    /// Counter cells in Cassandra store a CounterContext, not a raw i64 value.
309    /// The CounterContext tracks counter updates across multiple replicas (shards).
310    ///
311    /// Format (from Cassandra's CounterContext.java):
312    /// ```text
313    /// [header_size: 2-byte BE signed short]    <- Number of shards (negative if cleanup needed)
314    /// [indices: 2 bytes * |header_size|]       <- Shard type indicators (negative = global)
315    /// [shards: 32 bytes each]:
316    ///     [counter_id: 16 bytes UUID]          <- Replica's CounterId
317    ///     [clock: 8-byte BE unsigned long]     <- Logical clock
318    ///     [count: 8-byte BE signed long]       <- The actual counter value for this shard
319    /// ```
320    ///
321    /// The counter value is the sum of all shard counts, matching Cassandra's `total()` function.
322    ///
323    /// Returns (total_value, bytes_consumed)
324    pub(super) fn parse_counter_context(
325        data: &[u8],
326        offset: usize,
327        column_name: &str,
328    ) -> Result<(i64, usize)> {
329        // Constants from CounterContext.java
330        const HEADER_SIZE_LENGTH: usize = 2;
331        const HEADER_ELT_LENGTH: usize = 2;
332        const COUNTER_ID_LENGTH: usize = 16;
333        const CLOCK_LENGTH: usize = 8;
334        const COUNT_LENGTH: usize = 8;
335        const STEP_LENGTH: usize = COUNTER_ID_LENGTH + CLOCK_LENGTH + COUNT_LENGTH; // 32
336
337        // Maximum reasonable shard count to prevent DoS from corrupted data
338        // A typical Cassandra cluster has at most 100-500 nodes, so 1024 is generous
339        const MAX_COUNTER_SHARDS: usize = 1024;
340
341        let mut pos = offset;
342
343        // Read header_size (2-byte BE signed short)
344        if pos + HEADER_SIZE_LENGTH > data.len() {
345            return Err(Error::corruption(format!(
346                "Counter '{}': need {} bytes for header_size at offset {}, only {} available",
347                column_name,
348                HEADER_SIZE_LENGTH,
349                pos,
350                data.len() - pos
351            )));
352        }
353        let header_size_raw = i16::from_be_bytes([data[pos], data[pos + 1]]);
354        // Negative header_size indicates local shards need cleanup (CASSANDRA-1938).
355        // The absolute value gives the actual shard count.
356        let shard_count = header_size_raw.unsigned_abs() as usize;
357        pos += HEADER_SIZE_LENGTH;
358
359        // Validate shard count to prevent DoS from corrupted data
360        if shard_count > MAX_COUNTER_SHARDS {
361            return Err(Error::corruption(format!(
362                "Counter '{}': unreasonable shard count {} (max {})",
363                column_name, shard_count, MAX_COUNTER_SHARDS
364            )));
365        }
366
367        tracing::debug!(
368            "V5CompressedLegacy: Counter '{}' header_size={}, shard_count={}",
369            column_name,
370            header_size_raw,
371            shard_count
372        );
373
374        // Handle empty counter context (0 shards = counter value of 0)
375        if shard_count == 0 {
376            return Ok((0, HEADER_SIZE_LENGTH));
377        }
378
379        // Skip header indices (2 bytes per shard)
380        let indices_size = HEADER_ELT_LENGTH * shard_count;
381        if pos + indices_size > data.len() {
382            return Err(Error::corruption(format!(
383                "Counter '{}': need {} bytes for indices at offset {}, only {} available",
384                column_name,
385                indices_size,
386                pos,
387                data.len() - pos
388            )));
389        }
390        pos += indices_size;
391
392        // Calculate expected body size
393        let body_size = STEP_LENGTH * shard_count;
394        if pos + body_size > data.len() {
395            return Err(Error::corruption(format!(
396                "Counter '{}': need {} bytes for {} shards at offset {}, only {} available",
397                column_name,
398                body_size,
399                shard_count,
400                pos,
401                data.len() - pos
402            )));
403        }
404
405        // Sum count values from all shards (matching Cassandra's total() function)
406        let mut total: i64 = 0;
407        for shard_idx in 0..shard_count {
408            // Skip counter_id (16 bytes) and clock (8 bytes), read count (8 bytes)
409            let count_offset = pos + (shard_idx * STEP_LENGTH) + COUNTER_ID_LENGTH + CLOCK_LENGTH;
410            let count = i64::from_be_bytes([
411                data[count_offset],
412                data[count_offset + 1],
413                data[count_offset + 2],
414                data[count_offset + 3],
415                data[count_offset + 4],
416                data[count_offset + 5],
417                data[count_offset + 6],
418                data[count_offset + 7],
419            ]);
420            // Use checked_add to detect overflow (unlike Java which silently wraps)
421            total = total.checked_add(count).ok_or_else(|| {
422                Error::corruption(format!(
423                    "Counter '{}': integer overflow when summing shard {} (total={}, count={})",
424                    column_name, shard_idx, total, count
425                ))
426            })?;
427
428            tracing::trace!(
429                "V5CompressedLegacy: Counter '{}' shard {} count={}",
430                column_name,
431                shard_idx,
432                count
433            );
434        }
435
436        // Total bytes consumed
437        let consumed = HEADER_SIZE_LENGTH + indices_size + body_size;
438
439        Ok((total, consumed))
440    }

442    /// Parse a nested UDT from registry definition (Issue #238)
443    /// Used when parsing UDT fields that are themselves UDTs
444    ///
445    /// # ONE per-field dispatch (issue #3631)
446    /// The per-field `match` this function used to carry was the SECOND copy of the
447    /// same ~100-line dispatch: it resolved a nested UDT name itself, wrapped
448    /// `frozen` itself, and had FOUR `Value::Blob` arms for a name it could not
449    /// resolve plus a `_ =>` that degraded every collection-typed field. All of it
450    /// is now one call to `parse_simple_udt_field_value_at`, which resolves through
451    /// the very same `self.udt_registry`, threads `depth`, and returns an explicit
452    /// `Error` naming an unresolvable type instead of silently handing back bytes
453    /// (#3631 criterion 5).
454    ///
455    /// The old `registry: &UdtRegistry` parameter is GONE: every caller passed
456    /// `self.udt_registry`'s contents, and the one delegate resolves through that
457    /// same field, so threading it was a second path to one fact.
458    ///
459    /// `depth` is threaded, never restarted: this function is re-entered through the
460    /// delegate above, so a hard `0`/`1` here is what let a chain of frozen-UDT hops
461    /// reset `MAX_TYPE_NESTING_DEPTH` indefinitely.
462    pub(super) fn parse_nested_udt_from_registry(
463        &self,
464        data: &[u8],
465        udt_def: &crate::types::UdtTypeDef,
466        depth: usize,
467    ) -> Result<Value> {
468        if depth > MAX_TYPE_NESTING_DEPTH {
469            return Err(Error::corruption(format!(
470                "Nested UDT '{}': type nesting depth {} exceeds maximum {}",
471                udt_def.name, depth, MAX_TYPE_NESTING_DEPTH
472            )));
473        }
474        if udt_def.fields.len() > MAX_UDT_FIELD_COUNT {
475            return Err(Error::schema(format!(
476                "UDT '{}' has {} fields, exceeds maximum {}",
477                udt_def.name,
478                udt_def.fields.len(),
479                MAX_UDT_FIELD_COUNT
480            )));
481        }
482        let mut current_offset = 0;
483        let mut fields = Vec::with_capacity(udt_def.fields.len());
484
485        for field_def in &udt_def.fields {
486            // Check bounds for field length (4 bytes BE i32)
487            if current_offset + 4 > data.len() {
488                // Trailing fields are implicit null
489                while fields.len() < udt_def.fields.len() {
490                    let remaining_field = &udt_def.fields[fields.len()];
491                    fields.push(UdtField {
492                        name: remaining_field.name.clone(),
493                        value: None,
494                    });
495                }
496                break;
497            }
498
499            // Read field length (4 bytes big-endian i32)
500            let field_len = i32::from_be_bytes([
501                data[current_offset],
502                data[current_offset + 1],
503                data[current_offset + 2],
504                data[current_offset + 3],
505            ]);
506            current_offset += 4;
507
508            let field_value = if field_len == -1 {
509                None
510            } else if field_len == 0 {
511                // Zero-length: decoded from the DECLARED type, see
512                // `typed_value.rs::empty_is_a_value` (issue #3631).
513                Some(self.parse_simple_udt_field_value_at(&[], &field_def.field_type, depth)?)
514            } else {
515                let field_len = Self::checked_component_len(
516                    field_len,
517                    &field_def.name,
518                    current_offset,
519                    data.len(),
520                )?;
521
522                let field_data = &data[current_offset..current_offset + field_len];
523                current_offset += field_len;
524
525                Some(self.parse_simple_udt_field_value_at(
526                    field_data,
527                    &field_def.field_type,
528                    depth,
529                )?)
530            };
531
532            fields.push(UdtField {
533                name: field_def.name.clone(),
534                value: field_value,
535            });
536        }
537
538        // Issue #3811 (census finding C): `data` IS the whole nested UDT value —
539        // every caller slices it with `checked_component_len` — so the field loop
540        // must have reached its end. Trailing bytes and a partial component-length
541        // header both leave `current_offset` short; TupleType.split rule 1 (a
542        // genuinely short encoding) leaves it EQUAL and stays accepted.
543        //
544        // #3631: this is ALSO what makes `typed_value.rs::parse_typed_udt` able to
545        // report `data.len()` as the consumed count — an `Ok` here is a proof of it,
546        // not an assumption.
547        Self::require_fully_consumed(current_offset, data.len(), &udt_def.name, "nested UDT")?;
548        Ok(Value::Udt(Box::new(UdtValue {
549            type_name: udt_def.name.clone(),
550            keyspace: udt_def.keyspace.clone(),
551            fields,
552        })))

555    /// Returns true if the column type is a complex column (non-frozen collection).
556    /// Complex columns are stored as multiple cells with cell paths, unlike
557    /// frozen collections which are stored as a single cell with blob value.
558    ///
559    /// Issue #221: This is critical for proper parsing - complex columns have
560    /// a different format: [complex_deletion_time?] [cell_count] [cells...]
561    pub(super) fn is_complex_column(data_type: &str) -> bool {
562        // J1 (issue #1635): this is now called ONCE per column at
563        // `RowColumnResolution::build` time (its result cached on
564        // `ColumnToParse.is_complex`), NOT per cell. The per-cell-loop
565        // `TYPE_NORMALIZE_CALLS` gauge is therefore no longer recorded here — the
566        // per-cell decode path performs zero type normalizations after J1.
567        let dt = data_type.to_lowercase();
568        // Non-frozen collections start directly with list/set/map (CQL syntax)
569        // or org.apache.cassandra.db.marshal.ListType/SetType/MapType (internal syntax)
570        // Collections containing frozen element types (e.g., list<frozen<...>>) are still complex
571        // collections because the outer collection is not frozen - only the elements are.
572        // Only frozen<list<...>> etc. are not complex (they're single-cell frozen types)
573
574        // Check for frozen collections (which are NOT complex)
575        if dt.starts_with("frozen<")
576            || dt.starts_with("org.apache.cassandra.db.marshal.frozentype(")
577        {
578            return false;
579        }
580
581        // Check for CQL-style collection types
582        if dt.starts_with("list<") || dt.starts_with("set<") || dt.starts_with("map<") {
583            return true;
584        }
585
586        // Check for Cassandra internal collection types
587        if dt.starts_with("org.apache.cassandra.db.marshal.listtype(")
588            || dt.starts_with("org.apache.cassandra.db.marshal.settype(")
589            || dt.starts_with("org.apache.cassandra.db.marshal.maptype(")
590        {
591            return true;
592        }
593
594        // Issue #927: a TOP-LEVEL non-frozen UDT is a first-class multi-cell
595        // complex column — each field is stored as its own cell whose cell_path
596        // is the 2-byte (signed ShortType) field index. Frozen UDTs were already
597        // excluded above (frozen<...> / FrozenType(...)), and a UDT nested inside
598        // a collection (e.g. ListType(UserType(...))) is matched by its outer
599        // list/set/map branch, so a bare `usertype(` prefix here is unambiguous.
600        if dt.starts_with("org.apache.cassandra.db.marshal.usertype(") {
601            return true;
602        }
603
604        false
605    }

607    /// Parse the OUTERMOST `UserType(ks,hexname,hexfield:type,...)` marshal string
608    /// into its DECLARED field order, returning `(field_name, field_marshal_type)`
609    /// pairs (issue #927). The field type is kept as its raw marshal string so the
610    /// per-field value bytes can be decoded with [`parse_value_from_raw_bytes`].
611    ///
612    /// Reuses the same paren-matching / hex-name decoding as
613    /// [`parse_udt_type_definition`]; the two differ only in that this preserves
614    /// the raw type string instead of converting it to a `CqlType`.
615    pub(super) fn udt_field_marshal_types(type_str: &str) -> Result<Vec<(String, String)>> {
616        // The ONE `UserType(` locator + argument split (`udt/marshal_name.rs`),
617        // shared with `parse_udt_type_definition_with_depth`. This used to carry an
618        // independent copy of the find-and-walk-parens code, and so an independent
619        // copy of its package-SUFFIX hole: a substring `find` of the qualified
620        // marker accepted `my.org.apache.cassandra.db.marshal.UserType(…)`
621        // (roborev job 76).
622        let parts = Self::marshal_user_type_args(type_str)?;
623
624        let mut fields = Vec::with_capacity(parts.len().saturating_sub(2));
625        for field_def in parts.iter().skip(2) {
626            let field_def = field_def.trim();
627            if field_def.is_empty() {
628                continue;
629            }
630            let colon_idx = field_def.find(':').ok_or_else(|| {
631                Error::schema(format!(
632                    "Invalid UDT field definition (missing colon): {}",
633                    field_def
634                ))
635            })?;
636            let field_name = Self::decode_hex_name(&field_def[..colon_idx])?;
637            let field_type = field_def[colon_idx + 1..].trim().to_string();
638            fields.push((field_name, field_type));
639        }
640        Ok(fields)
641    }
642
643    /// Extract `(keyspace, type_name)` from the outermost `UserType(...)` marshal
644    /// string (issue #927). The keyspace is the first arg; the name is the
645    /// hex-decoded second arg.
646    pub(super) fn udt_keyspace_and_name(type_str: &str) -> Result<(String, String)> {
647        let udt_def = Self::parse_udt_type_definition(type_str)?;
648        Ok((udt_def.keyspace, udt_def.name))
649    }
