4    /// Read a single length-prefixed element from a frozen collection blob.
5    ///
6    /// `blob_end` is the exclusive upper byte index bounding the collection.
7    /// `element_desc` appears in error messages (e.g. `"list 'col' element 3"`).
8    pub(super) fn read_frozen_element(
9        &self,
10        data: &[u8],
11        offset: &mut usize,
12        blob_end: usize,
13        type_str: &str,
14        element_desc: &str,
15        depth: usize,
16    ) -> Result<Value> {
17        if *offset + 4 > blob_end {
18            return Err(Error::corruption(format!(
19                "Frozen {}: not enough bytes for length",
20                element_desc
21            )));
22        }
23        let len_i32 = i32::from_be_bytes([
24            data[*offset],
25            data[*offset + 1],
26            data[*offset + 2],
27            data[*offset + 3],
28        ]);
29        if len_i32 < 0 {
30            return Err(Error::corruption(format!(
31                "Frozen {}: negative length {}",
32                element_desc, len_i32
33            )));
34        }
35        let len = len_i32 as usize;
36        *offset += 4;
37
38        if *offset + len > blob_end {
39            return Err(Error::corruption(format!(
40                "Frozen {}: needs {} bytes but only {} available",
41                element_desc,
42                len,
43                blob_end - *offset
44            )));
45        }
46
47        let elem_data = &data[*offset..*offset + len];
48        let value = self.parse_value_from_raw_bytes(elem_data, type_str, element_desc, depth)?;
49        *offset += len;
50        Ok(value)
51    }

53    /// Parse a frozen list or set (cell-level, with VUInt blob_len prefix).
54    ///
55    /// The cell layout on disk is:
56    ///   [VUInt blob_len][i32 BE element_count][i32 BE elem_len][elem_bytes]...
57    ///
58    /// `as_set = true` wraps the result in `Value::Set`; otherwise `Value::List`.
59    fn parse_frozen_sequence_value(
60        &self,
61        data: &[u8],
62        mut offset: usize,
63        element_type: &str,
64        column: &crate::schema::Column,
65        as_set: bool,
66    ) -> Result<(Value, usize)> {
67        let kind = if as_set { "set" } else { "list" };
68        let (count, blob_end) = Self::read_frozen_preamble(data, &mut offset, kind, &column.name)?;
69
70        tracing::debug!(
71            "V5CompressedLegacy: Frozen {} '{}' with {} elements, element_type='{}'",
72            kind,
73            column.name,
74            count,
75            element_type
76        );
77
78        let mut elements = Vec::with_capacity(count);
79        for i in 0..count {
80            let desc = format!("{} '{}' element {}", kind, column.name, i);
81            let value =
82                self.read_frozen_element(data, &mut offset, blob_end, element_type, &desc, 0)?;
83            tracing::debug!("V5CompressedLegacy: Frozen {kind} element {i}: {value:?}");
84            elements.push(value);
85        }
86
87        Self::require_frozen_extent(offset, blob_end, kind, &column.name)?; // #3811 (F)
88        if as_set {
89            let xs = self.frozen_set_members_never_null(elements, kind);
90            Ok((Value::Set(xs), blob_end))
91        } else {
92            Ok((Value::List(elements), blob_end))
93        }
94    }

121    /// Parse a frozen list or set (raw, nested inside an already-bounded blob).
122    ///
123    /// Called when parsing nested collections inside an already-bounded frozen
124    /// blob.  There is NO VUInt cell-value-length prefix — the caller has
125    /// already bounded the data slice.  `as_set = true` produces `Value::Set`.
126    fn parse_frozen_sequence_value_raw(
127        &self,
128        data: &[u8],
129        mut offset: usize,
130        element_type: &str,
131        column_name: &str,
132        as_set: bool,
133        depth: usize,
134    ) -> Result<(Value, usize)> {
135        let kind = if as_set { "set" } else { "list" };
136        let count = Self::read_frozen_count(data, &mut offset, data.len(), kind, column_name)?;
137
138        tracing::debug!(
139            "V5CompressedLegacy: Parsing frozen {} '{}' with {} elements (raw)",
140            kind,
141            column_name,
142            count
143        );
144
145        let mut elements = Vec::with_capacity(count);
146        for i in 0..count {
147            // Each element in a frozen collection: [i32 BE len][element bytes]
148            if offset + 4 > data.len() {
149                return Err(Error::corruption(format!(
150                    "Frozen {} '{}': not enough bytes for element {} length",
151                    kind, column_name, i
152                )));
153            }
154            let elem_len_i32 = i32::from_be_bytes([
155                data[offset],
156                data[offset + 1],
157                data[offset + 2],
158                data[offset + 3],
159            ]);
160            if elem_len_i32 < 0 {
161                return Err(Error::corruption(format!(
162                    "Frozen {} '{}': negative element {} length {}",
163                    kind, column_name, i, elem_len_i32
164                )));
165            }
166            let elem_len = elem_len_i32 as usize;
167            offset += 4;
168
169            if offset + elem_len > data.len() {
170                return Err(Error::corruption(format!(
171                    "Frozen {} '{}': element {} needs {} bytes but only {} available",
172                    kind,
173                    column_name,
174                    i,
175                    elem_len,
176                    data.len() - offset
177                )));
178            }
179
180            let elem_data = &data[offset..offset + elem_len];
181            let elem_value =
182                self.parse_value_from_raw_bytes(elem_data, element_type, column_name, depth)?;
183            elements.push(elem_value);
184            offset += elem_len;
185        }
186
187        if as_set {
188            let xs = self.frozen_set_members_never_null(elements, kind);
189            Ok((Value::Set(xs), offset))
190        } else {
191            Ok((Value::List(elements), offset))
192        }
193    }

219    /// Parse tuple value from binary data at the cell level.
220    ///
221    /// Cell-level layout (written by `write_cell`):
222    /// ```text
223    /// [VUInt blob_len]
224    /// for each element (schema-ordered, from type string):
225    ///   [i32 BE element_len]  (-1 = null, 0 = empty, >0 = byte count)
226    ///   [element_len bytes]   (only present when element_len > 0)
227    /// ```
228    ///
229    /// Element count and types are derived exclusively from the schema type string
230    /// (no-heuristics mandate, Issue #28).
231    pub(super) fn parse_tuple_value(
232        &self,
233        data: &[u8],
234        offset: &mut usize,
235        type_str: &str,
236        column: &crate::schema::Column,
237    ) -> Result<Value> {
238        // Extract element types from schema (schema-aware, no heuristics)
239        let element_types = self.extract_tuple_element_types(type_str)?;
240
241        if element_types.is_empty() {
242            return Err(Error::schema(format!("Empty tuple type: {}", type_str)));
243        }
244
245        // Read the VUInt outer blob length to bound the tuple bytes
246        let (remaining, blob_len_raw) = parse_vuint(&data[*offset..]).map_err(|e| {
247            Error::corruption(format!(
248                "Tuple '{}': failed to parse outer blob length as VUInt: {:?}",
249                column.name, e
250            ))
251        })?;
252        if blob_len_raw > MAX_CELL_VALUE_LENGTH {
253            return Err(Error::corruption(format!(
254                "Tuple '{}': blob_len {} exceeds maximum {}",
255                column.name, blob_len_raw, MAX_CELL_VALUE_LENGTH
256            )));
257        }
258        let blob_len = blob_len_raw as usize;
259        let len_bytes_consumed = data[*offset..].len() - remaining.len();
260        *offset += len_bytes_consumed;
261
262        if *offset + blob_len > data.len() {
263            return Err(Error::corruption(format!(
264                "Tuple '{}': blob_len {} exceeds available data {}",
265                column.name,
266                blob_len,
267                data.len() - *offset
268            )));
269        }
270
271        let blob_end = *offset + blob_len;
272
273        // Parse each element using the schema-derived element type and the
274        // [i32 BE len][bytes] wire format (same as UDT fields and frozen
275        // collection elements — see type-mapping-complex.md).
276        let elements =
277            self.parse_tuple_elements_raw(data, offset, blob_end, &element_types, &column.name, 0)?;
278
279        // #3811 (F): the removed comment claimed advancing to blob_end "protects
280        // against trailing bytes"; it made them unobservable. Refuse, THEN advance
281        // — the caller needs the STREAM position here, not the consumed count.
282        Self::require_frozen_extent(*offset, blob_end, "tuple", &column.name)?;
283        *offset = blob_end;
284
285        Ok(Value::Tuple(elements))
286    }

288    /// Parse tuple elements from an already-bounded raw byte slice.
289    ///
290    /// Each element is encoded as `[i32 BE len][bytes]` with -1 meaning null.
291    /// Element types are taken from `element_types` in order (schema-aware).
292    ///
293    /// `blob_end` is the exclusive upper byte index bounding the tuple data.
294    pub(super) fn parse_tuple_elements_raw(
295        &self,
296        data: &[u8],
297        offset: &mut usize,
298        blob_end: usize,
299        element_types: &[String],
300        column_name: &str,
301        depth: usize,
302    ) -> Result<Vec<Value>> {
303        let mut elements = Vec::with_capacity(element_types.len());
304
305        for (idx, elem_type) in element_types.iter().enumerate() {
306            let elem_desc = format!("tuple '{}' element {}", column_name, idx);
307
308            // Need at least 4 bytes for the element length
309            if *offset + 4 > blob_end {
310                // Trailing elements are implicitly null (matches UDT behaviour)
311                tracing::debug!(
312                    "Tuple '{}': element {} beyond blob_end, treating as null",
313                    column_name,
314                    idx
315                );
316                elements.push(Value::Null);
317                continue;
318            }
319
320            // Read element length (4-byte big-endian i32)
321            let elem_len_i32 = i32::from_be_bytes([
322                data[*offset],
323                data[*offset + 1],
324                data[*offset + 2],
325                data[*offset + 3],
326            ]);
327            *offset += 4;
328
329            if elem_len_i32 == -1 {
330                // Null element
331                elements.push(Value::Null);
332                continue;
333            }
334
335            if elem_len_i32 < -1 {
336                return Err(Error::corruption(format!(
337                    "{}: invalid negative element length {}",
338                    elem_desc, elem_len_i32
339                )));
340            }
341
342            let elem_len = elem_len_i32 as usize;
343
344            if *offset + elem_len > blob_end {
345                return Err(Error::corruption(format!(
346                    "{}: needs {} bytes but only {} available in blob",
347                    elem_desc,
348                    elem_len,
349                    blob_end - *offset
350                )));
351            }
352
353            let elem_data = &data[*offset..*offset + elem_len];
354            let value =
355                self.parse_value_from_raw_bytes(elem_data, elem_type, &elem_desc, depth + 1)?;
356            *offset += elem_len;
357
358            elements.push(value);
359        }
360
361        Ok(elements)
362    }

364    /// Extract tuple element types from a `tuple<T1, T2, ...>` (CQL short) or
365    /// `org.apache.cassandra.db.marshal.TupleType(T1, T2, ...)` (Cassandra marshal)
366    /// type string. Marshal support (issue #1081) is required so a tuple-typed
367    /// multicell-UDT field, whose type arrives in marshal form, decodes its
368    /// elements instead of falling through to the blob default.
369    pub(super) fn extract_tuple_element_types(&self, type_str: &str) -> Result<Vec<String>> {
370        let type_lower = type_str.to_lowercase();
371
372        // Determine the inner element-list content based on format. Slice from the
373        // ORIGINAL-CASE `type_str` so nested element marshal types keep their case.
374        let inner = if type_lower.starts_with("org.apache.cassandra.db.marshal.tupletype(")
375            && type_str.ends_with(')')
376        {
377            let prefix = "org.apache.cassandra.db.marshal.TupleType(";
378            &type_str[prefix.len()..type_str.len() - 1]
379        } else if type_lower.starts_with("tuple<") && type_str.ends_with('>') {
380            &type_str[6..type_str.len() - 1]
381        } else {
382            return Err(Error::schema(format!(
383                "Invalid tuple type format: {}",
384                type_str
385            )));
386        };
387
388        if inner.is_empty() {
389            return Ok(Vec::new());
390        }
391
392        // Split by top-level comma, handling both CQL angle brackets (`<`/`>`)
393        // and marshal parentheses (`(`/`)`) so nested composite element types
394        // (e.g. `ListType(UTF8Type)`) are not split internally.
395        let mut types = Vec::new();
396        let mut current = String::new();
397        let mut depth = 0i32;
398
399        for ch in inner.chars() {
400            match ch {
401                '<' | '(' => {
402                    depth += 1;
403                    current.push(ch);
404                }
405                '>' | ')' => {
406                    if depth == 0 {
407                        return Err(Error::schema(format!(
408                            "Unmatched '{}' in tuple type: {}",
409                            ch, type_str
410                        )));
411                    }
412                    depth -= 1;
413                    current.push(ch);
414                }
415                ',' if depth == 0 => {
416                    types.push(current.trim().to_string());
417                    current.clear();
418                }
419                _ => {
420                    current.push(ch);
421                }
422            }
423        }
424
425        if !current.is_empty() {
426            types.push(current.trim().to_string());
427        }
428
429        Ok(types)
430    }

665    #[test]
666    fn test_tuple_int_text_parsing() {
667        // Test parse_tuple_elements_raw with constructed binary data.
668        //
669        // Wire format for each tuple element: [i32 BE elem_len][elem_bytes]
670        // Null element: [i32 BE -1] (no following bytes)
671        //
672        // Tuple: (int=42, text="hi")
673        //   element 0: [0x00, 0x00, 0x00, 0x04][42 as i32 BE] -> [0,0,0,4][0,0,0,42]
674        //   element 1: [0x00, 0x00, 0x00, 0x02]["hi"] -> [0,0,0,2][0x68,0x69]
675        let parser =
676            V5CompressedLegacyParser::new("test".to_string(), "table".to_string(), 0, 0, None);
677
678        let mut data = Vec::new();
679        // element 0: int 42
680        data.extend_from_slice(&4i32.to_be_bytes()); // length
681        data.extend_from_slice(&42i32.to_be_bytes()); // value
682                                                      // element 1: text "hi"
683        let hi = b"hi";
684        data.extend_from_slice(&(hi.len() as i32).to_be_bytes()); // length
685        data.extend_from_slice(hi); // value
686
687        let element_types = vec!["int".to_string(), "text".to_string()];
688        let mut offset = 0usize;
689        let blob_end = data.len();
690        let elements = parser
691            .parse_tuple_elements_raw(&data, &mut offset, blob_end, &element_types, "col", 0)
692            .unwrap();
693
694        assert_eq!(elements.len(), 2);
695        assert_eq!(elements[0], Value::Integer(42));
696        assert_eq!(elements[1], Value::text("hi".to_string()));
697        assert_eq!(
698            offset, blob_end,
699            "offset should reach blob_end after parsing all elements"
700        );
701
702        // Also test null element: (int=null, text="ok")
703        let mut data2 = Vec::new();
704        data2.extend_from_slice(&(-1i32).to_be_bytes()); // null element 0
705        let ok = b"ok";
706        data2.extend_from_slice(&(ok.len() as i32).to_be_bytes());
707        data2.extend_from_slice(ok);
708
709        let mut offset2 = 0usize;
710        let blob_end2 = data2.len();
711        let elements2 = parser
712            .parse_tuple_elements_raw(&data2, &mut offset2, blob_end2, &element_types, "col", 0)
713            .unwrap();
714
715        assert_eq!(elements2.len(), 2);
716        assert_eq!(elements2[0], Value::Null);
717        assert_eq!(elements2[1], Value::text("ok".to_string()));
718    }
