1//! Marshal collection-element type extraction (issue #1340).
2//!
3//! A `frozen<collection>` column's authoritative on-disk SerializationHeader
4//! marshal type (`RowColumnResolution.header_type`) carries the full element type
5//! for the collection, e.g. `frozen<list<frozen<person>>>` is written as
6//! `FrozenType(ListType(FrozenType(UserType(ks,person,field-types...))))`. The
7//! query/compaction decoders route a frozen collection through a per-element
8//! decoder keyed on the CQL short form (`frozen<person>`), which cannot resolve
9//! the inner UDT's fields without a wired `UdtRegistry`. Threading the marshal
10//! element type down lets an inner `frozen<UDT>` decode to a typed
11//! `Value::Frozen(Value::Udt(..))` from the file's OWN header — the most
12//! authoritative metadata, exactly what the no-heuristics mandate (#28) asks for.
13//!
14//! This module owns the small paren-aware extractor that peels the outer
15//! `FrozenType(...)` wrapper and pulls the `ListType`/`SetType` element type or
16//! the `MapType` key/value types out of the marshal string. It NEVER panics on a
17//! malformed marshal string — it returns `None` so callers fall back to the CQL
18//! short form (then the registry, then an opaque `Value::Blob`).
19
20use super::*;

22/// Marshal element type(s) extracted from a frozen collection's marshal type.
23///
24/// Borrows directly from the input `header_type` str (no owned `String`s) so the
25/// per-cell frozen-decode path stays allocation-free (<128MB budget).
26#[derive(Debug, Clone, PartialEq, Eq)]
27pub(super) enum MarshalCollectionElements<'a> {
28    /// `ListType(X)` / `SetType(X)` element marshal type `X`.
29    Sequence(&'a str),
30    /// `MapType(K, V)` key and value marshal types.
31    Map(&'a str, &'a str),
32}
33
34const FROZEN_LOWER: &str = "org.apache.cassandra.db.marshal.frozentype(";
35const LIST_LOWER: &str = "org.apache.cassandra.db.marshal.listtype(";
36const SET_LOWER: &str = "org.apache.cassandra.db.marshal.settype(";
37const MAP_LOWER: &str = "org.apache.cassandra.db.marshal.maptype(";
38
39/// If `s` (case-insensitively) starts with `prefix_lower` — which MUST end in the
40/// opening `(` — and the paren opened by that prefix is closed by the LAST byte of
41/// `s`, return the balanced inner content (original case). Returns `None` for any
42/// prefix mismatch, unbalanced parens, or trailing bytes after the matching close.
43/// Never panics.
44fn balanced_inner<'a>(s: &'a str, prefix_lower: &str) -> Option<&'a str> {
45    let head = s.get(..prefix_lower.len())?;
46    if !head.eq_ignore_ascii_case(prefix_lower) {
47        return None;
48    }
49    let bytes = s.as_bytes();
50    let inner_start = prefix_lower.len();
51    // The prefix's own `(` is already open, so start at depth 1.
52    let mut depth: usize = 1;
53    let mut i = inner_start;
54    while i < bytes.len() {
55        match bytes[i] {
56            b'(' => depth += 1,
57            b')' => {
58                depth -= 1;
59                if depth == 0 {
60                    return if i == bytes.len() - 1 {
61                        Some(&s[inner_start..i])
62                    } else {
63                        // Trailing bytes after the matching close: not a clean
64                        // single top-level type — reject rather than guess.
65                        None
66                    };
67                }
68            }
69            _ => {}
70        }
71        i += 1;
72    }
73    None
74}
75
76/// Split a `MapType` inner (`K, V`) at the first depth-0 comma, honoring both
77/// paren `()` and angle `<>` nesting. Returns `None` if there is no top-level
78/// comma. Never panics.
79fn split_top_level_comma(inner: &str) -> Option<(&str, &str)> {
80    let mut depth: usize = 0;
81    for (i, ch) in inner.char_indices() {
82        match ch {
83            '(' | '<' => depth += 1,
84            ')' | '>' => depth = depth.saturating_sub(1),
85            ',' if depth == 0 => {
86                return Some((&inner[..i], &inner[i + 1..]));
87            }
88            _ => {}
89        }
90    }
91    None
92}

94impl V5CompressedLegacyParser {
95    /// Extract the marshal element type(s) for a frozen collection from the
96    /// column's authoritative SerializationHeader marshal type (issue #1340).
97    ///
98    /// Peels at most one leading `FrozenType(...)` wrapper, then matches
99    /// `ListType`/`SetType` (→ [`MarshalCollectionElements::Sequence`]) or
100    /// `MapType` (→ [`MarshalCollectionElements::Map`]). Returns `None` — never
101    /// panics — for any non-collection or malformed marshal string, so the caller
102    /// falls back to the CQL short form.
103    ///
104    /// Parse this ONCE per frozen cell (not per element): the result borrows from
105    /// `header_type`, so the per-element hot loop gains no allocations (<128MB budget).
106    pub(super) fn extract_marshal_collection_elements(
107        header_type: &str,
108    ) -> Option<MarshalCollectionElements<'_>> {
109        let s = header_type.trim();
110        // Strip at most one outer FrozenType(...) wrapper (a frozen collection
111        // column's header is `FrozenType(ListType(...))` etc.); tolerate a bare
112        // `ListType(...)` too.
113        let inner = balanced_inner(s, FROZEN_LOWER).map(str::trim).unwrap_or(s);
114
115        if let Some(elem) =
116            balanced_inner(inner, LIST_LOWER).or_else(|| balanced_inner(inner, SET_LOWER))
117        {
118            return Some(MarshalCollectionElements::Sequence(elem.trim()));
119        }
120        if let Some(kv) = balanced_inner(inner, MAP_LOWER) {
121            let (k, v) = split_top_level_comma(kv)?;
122            return Some(MarshalCollectionElements::Map(k.trim(), v.trim()));
123        }
124        None
125    }
126
127    /// Pick the element type to decode with, honoring the no-heuristics precedence
128    /// (issue #28): prefer the authoritative marshal element type ONLY when it is a
129    /// UDT-bearing type (`UserType(...)` at any nesting), otherwise keep the CQL
130    /// short form (which already decodes every primitive/collection element and
131    /// preserves the existing byte-parity behavior). The registry and `Value::Blob`
132    /// fallbacks live downstream in `parse_value_from_raw_bytes`.
133    pub(super) fn prefer_udt_marshal_element<'a>(
134        marshal: Option<&'a str>,
135        schema_short: &'a str,
136    ) -> &'a str {
137        match marshal {
138            Some(m) if Self::is_udt_type(m) => m,
139            _ => schema_short,
140        }
141    }
142
143    /// THE ONE RULE FOR A MAP KEY'S DECODE TYPE, shared by BOTH map readers
144    /// (issue #3612, roborev round 8 finding 1).
145    ///
146    /// The multicell reader (`complex_column`'s map branch) and the frozen reader
147    /// (`cell_value_complex`'s `map<` branch) must hand the SAME string to the same
148    /// decoder, or they produce `Value` keys that compare and hash differently for
149    /// one CQL value. They previously did not, and the difference was invisible for
150    /// UDT keys and visible for COLLECTION keys:
151    ///
152    /// * multicell `map<frozen<set<frozen<U>>>, int>` → Cassandra records
153    ///   `MapType(FrozenType(SetType(UserType(..))),..)`, so the key marshal KEEPS
154    ///   its `FrozenType` — a multicell map key must be explicitly frozen;
155    /// * frozen `frozen<map<frozen<set<frozen<U>>>, int>>` → Cassandra records
156    ///   `FrozenType(MapType(SetType(UserType(..)),..))`, with the INNER marker
157    ///   OMITTED, because everything inside a frozen collection is already frozen.
158    ///
159    /// Decoding those two strings yields `Frozen(Set(Udt))` and `Set(Udt)`. Hence
160    /// the strip below: it normalizes the marshal form to the shape the frozen
161    /// reader already receives, so both readers decode an identical string and
162    /// parity holds BY CONSTRUCTION rather than by a second, value-level
163    /// normalizer bolted onto one side (which is what round 3 did, and what left
164    /// this hole one nesting level down).
165    ///
166    /// # WHY MARSHAL BEATS SCHEMA AT ALL — from CASSANDRA, not from our own code
167    ///
168    /// `SerializationHeader.getType` (cassandra-5.0.8,
169    /// `src/java/org/apache/cassandra/db/SerializationHeader.java`) is:
170    ///
171    /// ```text
172    /// public AbstractType<?> getType(ColumnMetadata column)
173    /// {
174    ///     return typeMap == null ? column.type : typeMap.get(column.name.bytes);
175    /// }
176    /// ```
177    ///
178    /// Cassandra's OWN read path decodes with the header's recorded type and falls
179    /// back to the live schema's `column.type` only when the header carries no type
180    /// map. The recorded type is not a guess — it is what the writer put beside
181    /// these bytes, and after an `ALTER` the on-disk bytes conform to IT, not to the
182    /// current schema. Decoding old bytes with a newer schema is how you mis-decode
183    /// them.
184    ///
185    /// This does NOT contradict CLAUDE.md's "schema, else `Statistics.db`". That
186    /// rule is about where type information may come from AT ALL — prefer declared
187    /// metadata, never infer a type from byte patterns (issue #28). It is not a
188    /// claim that a user-supplied schema overrides the writer's recorded type for
189    /// the same column; `getType` settles that question the other way.
190    ///
191    /// We are deliberately NARROWER than `getType`: the marshal is taken ONLY when
192    /// it is UDT-bearing, so no non-UDT map key changes behaviour. One consequence,
193    /// correct rather than unfortunate: a user-supplied schema that DISAGREES with
194    /// the recorded type on a UDT-keyed multicell map is ignored — exactly the case
195    /// Cassandra resolves in the header's favour.
196    ///
197    /// The strip is applied ONLY to the marshal form. When
198    /// `prefer_udt_marshal_element` keeps the SCHEMA short form — every non
199    /// UDT-bearing key — both readers already receive that same unstripped string
200    /// (e.g. `frozen<set<int>>` on both sides), so stripping it would break the
201    /// parity it is meant to preserve.
202    ///
203    /// On the FROZEN reader this call is a NO-OP in both branches, by
204    /// construction: its marshal key never carries the outer marker (Cassandra
205    /// omits it, measured on `f_map_tuple_udt` and `fcm`), and its schema branch is
206    /// untouched. It is wired there anyway so the rule has ONE home and cannot
207    /// drift into two opinions.
208    pub(super) fn map_key_type_for_decode(marshal: Option<&str>, schema_short: &str) -> String {
209        match marshal {
210            // This guard MUST stay identical to `prefer_udt_marshal_element`'s, since
211            // it decides the same choice; the `debug_assert` pins them together so a
212            // change to one cannot silently diverge from the other.
213            Some(m) if Self::is_udt_type(m) => {
214                debug_assert!(
215                    std::ptr::eq(
216                        Self::prefer_udt_marshal_element(marshal, schema_short),
217                        m as &str
218                    ),
219                    "map_key_type_for_decode's guard diverged from prefer_udt_marshal_element"
220                );
221                Self::strip_one_outer_frozen_marshal(m).to_string()
222            }
223            _ => schema_short.to_string(),
224        }
225    }
226
227    /// Strip at most ONE outer `FrozenType(..)` from a MARSHAL type string,
228    /// case-insensitively; return the input unchanged if it carries none.
229    fn strip_one_outer_frozen_marshal(marshal: &str) -> &str {
230        const PREFIX: &str = "org.apache.cassandra.db.marshal.frozentype(";
231        let t = marshal.trim();
232        let lower = t.to_ascii_lowercase();
233        if lower.starts_with(PREFIX) && lower.ends_with(')') {
234            return t[PREFIX.len()..t.len() - 1].trim();
235        }
236        t
237    }
238}

359    fn unresolvable_inner_type_decodes_to_blob_no_panic() {
360        // Spec Req 1 scenario 4 (no-heuristics #28): with NO header marshal type
361        // and NO wired registry, an unresolvable inner UDT short-name must stay an
362        // opaque `Value::Blob` carrying the exact bytes — never a byte-pattern
363        // guess, never a panic.
364        let parser = V5CompressedLegacyParser::new("ks".to_string(), "tbl".to_string(), 0, 0, None);
365        let bytes = [0xDE, 0xAD, 0xBE, 0xEF];
366        let val = parser
367            .parse_value_from_raw_bytes(&bytes, "some_unregistered_udt", "col", 0)
368            .expect("unresolved UDT must not error");
369        assert_eq!(val, Value::blob(bytes.to_vec()));
370    }
