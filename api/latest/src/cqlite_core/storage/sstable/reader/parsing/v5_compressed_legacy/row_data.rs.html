3impl V5CompressedLegacyParser {
4    /// Parse row data (header + cells) and return cells with new offset
5    ///
6    /// V5CompressedLegacy format stores cells WITHOUT column names in schema column order.
7    /// Schema is REQUIRED to determine which column each value belongs to.
8    ///
9    /// Returns: `ParsedRow` = `(cells, row_header, new_offset, is_static)` where
10    /// `is_static` is `true` when the row's `EXTENDED_IS_STATIC` flag was set.
11    /// Static rows must be merged into clustering rows by the caller, not emitted directly.
12    #[allow(clippy::too_many_arguments)]
13    pub(super) fn parse_row_data_with_offset(
14        &self,
15        data: &[u8],
16        offset: usize,
17        schema: Option<&TableSchema>,
18        reader: &crate::storage::sstable::reader::types::SSTableReader,
19        want_cell_metadata: bool,
20        resolution: &RowColumnResolution,
21        // Issue #1741 (Finding 1): read-side shadow for per-cell tombstone/TTL
22        // filtering. `Some` only on user-facing SELECT reads (`read_shadowing`);
23        // `None` for every physical consumer (compaction / delta-scan / tests), which
24        // stay byte-unchanged.
25        shadow: Option<&PartitionShadow>,
26    ) -> Result<ParsedRow> {
27        self.parse_row_data_with_offset_impl(
28            data,
29            offset,
30            schema,
31            reader,
32            want_cell_metadata,
33            None,
34            resolution,
35            shadow,
36        )
37    }
38
39    /// Implementation of [`Self::parse_row_data_with_offset`] with an optional
40    /// per-column complex-element collector for the compaction read path
41    /// (epic #899). When `compaction_complex_out` is `Some`, every complex
42    /// column's per-element cells + complex deletion are captured into it
43    /// alongside the normal collapsed-`Value` cells. On user-facing reads it is
44    /// `None` and behavior is byte-unchanged.
45    #[allow(clippy::too_many_arguments)]
46    pub(super) fn parse_row_data_with_offset_impl(
47        &self,
48        data: &[u8],
49        mut offset: usize,
50        schema: Option<&TableSchema>,
51        reader: &crate::storage::sstable::reader::types::SSTableReader,
52        want_cell_metadata: bool,
53        mut compaction_complex_out: Option<&mut CompactionComplexColumns>,
54        resolution: &RowColumnResolution,
55        shadow: Option<&PartitionShadow>,
56    ) -> Result<ParsedRow> {
57        // Cells map keyed by the interned column-name handle (issue #1334): the
58        // name is a schema-owned `Arc<str>` shared across every cell/row, so
59        // populating a cell with its name is an `Arc::clone` refcount bump, NOT a
60        // per-cell, per-row heap `String` allocation.
61        let mut cells: HashMap<Arc<str>, Value> = HashMap::new();
62        // Parallel per-cell write metadata map (populated alongside `cells`).
63        // Only allocated when the caller actually needs WRITETIME/TTL metadata
64        // (i.e. `want_cell_metadata == true`).  On the normal read path this stays
65        // `None` so that zero HashMap allocations or inserts occur per cell.
66        let mut cell_meta: Option<HashMap<String, CellWriteMetadata>> = if want_cell_metadata {
67            Some(HashMap::new())
68        } else {
69            None
70        };
71
72        // DS4 (Issue #700): Per-column complex collection metadata.  Only allocated when
73        // want_cell_metadata is true (same gate as cell_meta to avoid hot-path overhead).
74        let mut complex_col_meta: Option<HashMap<String, ComplexColumnMeta>> = if want_cell_metadata
75        {
76            Some(HashMap::new())
77        } else {
78            None
79        };
80
81        let schema = schema.ok_or_else(|| {
82            Error::schema(format!(
83                "V5CompressedLegacy: Schema required for {}.{} (cells stored without column names)",
84                self.keyspace, self.table_name
85            ))
86        })?;
87
88        debug!(
89            "V5CompressedLegacy: Starting row data parse at offset {} with {} schema columns",
90            offset,
91            schema.columns.len()
92        );
93
94        // ISSUE #213 FIX: Correct parsing order for V5CompressedLegacy format
95        //
96        // The CORRECT format from Cassandra UnfilteredSerializer.java is:
97        //   1. [row_flags: u8]
98        //   2. [extended_flags: u8 if 0x80 set]
99        //   3. [clustering_prefix: variable]  ← BEFORE row_size!
100        //   4. [row_size: VInt]
101        //   5. [prev_size: VInt]
102        //   6. [row_body: timestamp, ttl, deletion, bitmap, cells]
103        //
104        // The previous code parsed row_size BEFORE clustering, which caused
105        // clustering key bytes to be misinterpreted as row_size (often 0).
106
107        // Step 1: Parse row flags (1-2 bytes)
108        let (row_flags, extended_flags, flags_size) = self.parse_row_flags(data, offset)?;
109        offset += flags_size;
110
111        // Issue #258 fix: Check if this is a static row (no clustering prefix)
112        // Per Cassandra UnfilteredSerializer.java lines 114-122, 190-191:
113        // Static rows have the IS_STATIC bit (0x01) set in extended flags and
114        // do NOT have a clustering prefix - skip directly to row_size.
115        let is_static = extended_flags
116            .map(|ef| (ef & EXTENDED_IS_STATIC) != 0)
117            .unwrap_or(false);
118
119        // Step 2: Parse clustering prefix BEFORE row_size (Issue #213 fix)
120        // This is the critical change - clustering comes AFTER flags but BEFORE row_size
121        // EXCEPT for static rows which have no clustering prefix at all.
122        let (clustering_values, offset) = if !is_static {
123            self.parse_clustering_prefix(data, offset, schema)?
124        } else {
125            log::debug!(
126                "V5CompressedLegacy: Static row detected (extended_flags=0x{:02x}), skipping clustering prefix",
127                extended_flags.unwrap_or(0)
128            );
129            (vec![], offset)
130        };
131
132        log::debug!(
133            "V5CompressedLegacy: Parsed {} clustering values after flags, now at offset {} (is_static={})",
134            clustering_values.len(),
135            offset,
136            is_static
137        );
138
139        // Issue #229 FIX: Add clustering key values to cells HashMap
140        //
141        // Cassandra stores clustering keys separately from regular columns, but they
142        // must be included in the result for proper query output. Without this fix,
143        // tables with clustering keys show fallback column names because the clustering
144        // values weren't being added to the cells HashMap.
145        for (i, _ck) in schema.clustering_keys.iter().enumerate() {
146            if i < clustering_values.len() {
147                // Issue #1334: reuse the interned clustering-key name handle
148                // (an `Arc::clone`) rather than cloning the schema `String`.
149                if let Some(name) = resolution.clustering_name(i) {
150                    cells.insert(Arc::clone(name), clustering_values[i].clone());
151                }
152            }
153        }
154
155        // Issue #1741 (Finding 1): per-cell shadow context for this row. `Some` only
156        // on the user-facing SELECT path. `covering` is the deletion (µs) that shadows
157        // any data older than it (partition tombstone folded with the open range
158        // tombstone when this row's clustering falls inside it); `now` is the read
159        // clock for per-cell TTL expiry. A data cell / whole collection whose
160        // effective write ts <= `covering`, or which is TTL-expired at `now`, is
161        // dropped below (no new per-cell allocation — the insert is made conditional).
162        let cell_ctx: Option<(Option<i64>, i64)> =
163            shadow.map(|sh| sh.cell_context(&clustering_values));
164
165        // Step 3: Parse row metadata (row_size, prev_size, timestamps, etc.)
166        //
167        // CRITICAL (Issue #237): Save offset where row_size VInt STARTS.
168        // The row_size value is measured from AFTER this VInt is consumed.
169        // Formula: next_offset = (row_metadata_offset + row_size_vint_len) + row_size
170        // This offset is right after the clustering prefix (which was already parsed).
171        let row_metadata_offset = offset;
172        let (mut row_header, row_size) =
173            self.parse_row_metadata(data, offset, row_flags, extended_flags)?;
174
175        // CRITICAL VALIDATION: row_size must be reasonable
176        //
177        // In V5CompressedLegacy format, row_size should never exceed the block size (typically 16KB).
178        // If row_size is unreasonably large, it indicates either:
179        // 1. Partition tombstone or deletion marker (no actual row data)
180        // 2. Format parsing error (landed at wrong offset)
181        // 3. Corrupted data
182        //
183        // In all cases, we should skip this partition rather than panic.
184        const MAX_REASONABLE_ROW_SIZE: u64 = 1_000_000; // 1MB max (very generous)
185        if row_size > MAX_REASONABLE_ROW_SIZE {
186            return Err(Error::corruption(format!(
187                "V5CompressedLegacy: Unreasonably large row_size={} at offset {} (max: {}). Likely partition tombstone or format error.",
188                row_size,
189                offset,
190                MAX_REASONABLE_ROW_SIZE
191            )));
192        }
193
194        // Row payloads can span multiple compressed chunks in V5CompressedLegacy format.
195        // The reader has already stitched all chunks together (see get_all_entries()),
196        // so row_size is valid across chunk boundaries. We MUST NOT validate against
197        // individual chunk sizes as rows naturally span chunks in Cassandra's format.
198        // This is NOT corruption - it's the intended file layout.
199
200        log::debug!(
201            "V5CompressedLegacy: Parsed row metadata at offset {}: header_size={} bytes, row_size={} bytes, timestamp={:?}, ttl={:?}, deletion={:?}",
202            offset, row_header.header_size, row_size, row_header.timestamp, row_header.ttl, row_header.local_deletion_time
203        );
204
205        // Row tombstone detection (Issue #191, Phase 2) + coexistence (Issue #932).
206        //
207        // A row with ROW_HAS_DELETION (0x10) carries a row-level deletion. HISTORICALLY
208        // such a row had NO cells, so the parser skipped cell parsing entirely. But a
209        // row deletion can COEXIST with surviving cells (cells written strictly after
210        // the deletion — issue #932): the writer emits HAS_DELETION + HAS_TIMESTAMP +
211        // the surviving cells. We must therefore skip cell parsing ONLY when there are
212        // no cell bytes after the row header; otherwise fall through to the normal cell
213        // loop so the surviving cells are read (the row header still carries the
214        // deletion via `local_deletion_time` / `marked_for_delete_at`).
215        //
216        // Calculate offset after row data (based on row_size from header).
217        //
218        // CRITICAL FIX (Issue #237): row_size is measured from AFTER the row_size VInt,
219        // not from where it starts. This matches Cassandra's getFilePointer() semantics:
220        //   next_position = row_size_value + position_after_reading_row_size_vint
221        let after_row_offset =
222            (row_metadata_offset + row_header.row_size_vint_len) + row_size as usize;
223        // Cell data (if any) begins right after the row header. When that start
224        // reaches the row boundary there are no cells — a pure row tombstone.
225        let cell_data_start = row_metadata_offset + row_header.header_size;
226        let has_cell_bytes = cell_data_start < after_row_offset;
227
228        if row_header.local_deletion_time.is_some() && !has_cell_bytes {
229            log::debug!(
230                "V5CompressedLegacy: Pure row tombstone (deletion_time={:?}), skipping cell parsing",
231                row_header.local_deletion_time
232            );
233
234            // Validate we have enough data
235            if after_row_offset > data.len() {
236                let remaining = data
237                    .len()
238                    .saturating_sub(row_metadata_offset + row_header.row_size_vint_len);
239                return Err(Error::corruption(format!(
240                    "V5CompressedLegacy: Not enough bytes for row data at offset {} (need {}, have {})",
241                    row_metadata_offset + row_header.row_size_vint_len,
242                    row_size,
243                    remaining
244                )));
245            }
246
247            let next_offset = after_row_offset;
248            log::debug!(
249                "V5CompressedLegacy: Skipped tombstoned row, next offset = {}",
250                next_offset
251            );
252
253            // Return empty cells for tombstoned row (no cell metadata, no complex meta)
254            return Ok((
255                cells,
256                cell_meta,
257                Some(row_header),
258                next_offset,
259                is_static,
260                None,
261            ));
262        }
263
264        // Advance offset past row metadata to start of cell data
265        let mut offset = offset + row_header.header_size;
266
267        log::debug!(
268            "V5CompressedLegacy: Cell data starts at offset {}, first 32 bytes: {}",
269            offset,
270            hex::encode(&data[offset..std::cmp::min(offset + 32, data.len())])
271        );
272
273        // Cell flags validation: First byte should be valid cell flags (0x00-0x1F) for simple cells
274        // Common flags: 0x00 (basic cell), 0x08 (USE_ROW_TIMESTAMP), 0x04 (HAS_EMPTY_VALUE)
275        // Deleted cells have 0x01 (IS_DELETED), expiring cells have 0x02 (IS_EXPIRING)
276        //
277        // NOTE: For complex columns (non-frozen collections), the first byte is a VInt for the
278        // cell count, which may have values > 0x1F. This is normal and not an error.
279        // The validation below is only accurate for tables with all simple cells.
280        if offset < data.len() {
281            let first_byte = data[offset];
282            if first_byte <= 0x1F {
283                debug!(
284                    "V5CompressedLegacy: Valid cell flags 0x{:02x} at offset {} after row header",
285                    first_byte, offset
286                );
287            } else {
288                debug!(
289                    "V5CompressedLegacy: First byte 0x{:02x} at offset {} (> 0x1F) - may be VInt for complex column cell count",
290                    first_byte, offset
291                );
292            }
293        }
294
295        // CRITICAL: V5CompressedLegacy format stores cells WITHOUT column names
296        // or column IDs in the binary data. Cells appear in SCHEMA DEFINITION ORDER
297        // (the order columns were defined in CREATE TABLE), NOT alphabetical order.
298        //
299        // NULL/missing columns are handled by:
300        // - Checking for cell marker (0x08) before attempting to parse
301        // - If no marker found or parse fails, column is NULL (not present)
302        // - Continue to next column in schema order
303        //
304        // This implementation uses schema definition order directly, which is the
305        // correct approach per Cassandra 5.0 SerializationHeader semantics.
306
307        // CRITICAL FIX (Issue #191): cells are stored in SERIALIZATION HEADER column
308        // order (Statistics.db serialization header — alphabetical by
309        // ColumnIdentifier/comparator), NOT CQL schema order. Partition/clustering
310        // keys carry no cell data and are excluded; dropped columns (present on disk
311        // but absent from the supplied schema) are RETAINED in order so their bytes
312        // are consumed and trailing columns stay byte-aligned (issue #1080 Part 2).
313        //
314        // Issue #702: the on-disk column group differs by row kind, so the resolution
315        // exposes a separate ordering for static vs regular rows.
316        //
317        // Issue #1046 (the true hoist): this header→schema resolution is CONSTANT for
318        // every row in the block, so it is built ONCE (in `RowColumnResolution::build`
319        // at the top of the per-block driver) and reused here. The per-row body now
320        // performs ZERO schema-lookup allocations — no per-row `HashMap`, no per-row
321        // `String` clone, no per-row `Vec` of columns.
322        let columns_in_order: &[ColumnToParse] = resolution.columns_for(is_static);
323
324        // Apply the missing_columns_bitmap INLINE (no per-row `Vec`): a column at
325        // on-disk index `idx` is present iff `idx >= 64` (beyond the u64 bitmap,
326        // treated present) or its bit is clear. The bitmap is indexed by the ON-DISK
327        // column order, which is exactly `columns_in_order` (dropped columns retained),
328        // so the index alignment is preserved.
329        let missing_bitmap = row_header.missing_columns_bitmap;
330        let is_present = |idx: usize| -> bool {
331            match missing_bitmap {
332                Some(bitmap) => idx >= 64 || (bitmap & (1u64 << idx)) == 0,
333                None => true,
334            }
335        };
336
337        log::debug!("V5CompressedLegacy: Parsing cells in SERIALIZATION HEADER ORDER starting at offset {} (row header was {} bytes, {} on-disk columns, bitmap={:?})", offset, row_header.header_size, columns_in_order.len(), missing_bitmap);
338        log::debug!(
339            "V5CompressedLegacy: Cell data hex (first 64 bytes): {}",
340            hex::encode(&data[offset..std::cmp::min(offset + 64, data.len())])
341        );
342
343        // Issue #221: Check if row has complex deletion info for non-frozen collections
344        let has_complex_deletion = (row_flags & ROW_HAS_COMPLEX_DELETION) != 0;
345        if has_complex_deletion {
346            log::debug!("V5CompressedLegacy: Row has HAS_COMPLEX_DELETION flag (0x40) set");
347        }
348
349        // Issue #1741 read-side shadowing aggregate. Computed from scalars the cell
350        // parser already returns (no new allocation, no extra HashMap): the max data
351        // cell write timestamp, the max expiring-cell expiry, and whether any data
352        // cell is live-forever. Stashed on `row_header` after the loop and consulted
353        // by the read emit paths to hide partition/range-tombstone-shadowed and
354        // TTL-expired rows. Kept off the write/compaction reconciliation path.
355        let mut agg_max_cell_ts: Option<i64> = None;
356        let mut agg_max_expires_at: Option<i64> = None;
357        let mut agg_has_live_forever = false;
358
359        for (col_idx, ctp) in columns_in_order.iter().enumerate() {
360            // Skip columns marked MISSING by the row's bitmap (inline, no per-row
361            // allocation). `col_idx` is the ON-DISK column index — exactly what the
362            // bitmap is indexed by — so this is identical to the prior pre-filtered
363            // `columns_to_parse` Vec, just without materializing it.
364            if !is_present(col_idx) {
365                continue;
366            }
367            let header_type: Option<&str> = ctp.header_type;
368
369            // A column present on disk but ABSENT from the supplied schema is a
370            // DROPPED column (issue #1080 Part 2): its bytes MUST be consumed to
371            // keep the trailing columns aligned, but it is NOT emitted. We decode
372            // it with a synthetic Column whose `data_type` is the AUTHORITATIVE
373            // on-disk header marshal string (the only type metadata we have), then
374            // discard the value. `emit` gates every cell/metadata insertion below.
375            let emit = ctp.schema.is_some();
376            // Synthetic column used ONLY to consume a DROPPED column's bytes; built
377            // from the on-disk header marshal type. Initialized lazily (deferred
378            // binding) so the common schema-present path pays no allocation — the
379            // dropped-column path is rare. The holder outlives the borrow below.
380            let dropped_column_holder;
381            let column: &crate::schema::Column = match ctp.schema {
382                Some(c) => c,
383                None => {
384                    dropped_column_holder = crate::schema::Column {
385                        name: format!("__dropped_col_{col_idx}"),
386                        data_type: header_type.unwrap_or("blob").to_string(),
387                        nullable: true,
388                        default: None,
389                        is_static,
390                    };
391                    &dropped_column_holder
392                }
393            };
394
395            // Issue #1081: the AUTHORITATIVE complex-ness / complex-decode type.
396            // Prefer the on-disk SerializationHeader marshal type (carries
397            // `UserType(...)` for a non-frozen UDT, which the supplied schema's
398            // bare short form cannot express); fall back to the supplied schema
399            // type only on the header-empty path (no-heuristics: both are
400            // authoritative metadata, never guessed from bytes — issue #28). For a
401            // dropped column the synthetic `column.data_type` IS the header marshal
402            // type, so this resolves identically either way.
403            let complex_type: &str = header_type.unwrap_or(&column.data_type);
404
405            if offset >= data.len() {
406                log::debug!(
407                    "V5CompressedLegacy: Reached end of data at column {} ('{}'), parsed {}/{} on-disk cells",
408                    col_idx,
409                    column.name,
410                    cells.len(),
411                    columns_in_order.len()
412                );
413                break;
414            }
415
416            // Issue #221: Branch based on column type - complex columns need special parsing
417            // Issue #693: simple columns return 4-tuple including cell timestamp / expiration;
418            //             complex columns return 2-tuple and inherit the row-level timestamp.
419            if Self::is_complex_column(complex_type) {
420                log::debug!(
421                    "V5CompressedLegacy: Column '{}' is complex (non-frozen collection / multicell UDT), using parse_complex_column",
422                    column.name
423                );
424                // Epic #899: on the compaction read path collect per-element
425                // cells + the real complex deletion into `compaction_complex_out`
426                // (otherwise this is the user-facing read path: collapsed value
427                // only, byte-unchanged). Elements inherit the row liveness
428                // timestamp when they carry USE_ROW_TIMESTAMP.
429                let parse_result = if compaction_complex_out.is_some() {
430                    let row_ts = row_header.timestamp.unwrap_or(0);
431                    let mut element_buf = Vec::new();
432                    self.parse_complex_column_inner(
433                        data,
434                        offset,
435                        column,
436                        complex_type,
437                        has_complex_deletion,
438                        row_ts,
439                        Some(&mut element_buf),
440                        // Compaction / delta read path: NO per-element filtering (it
441                        // needs every element for reconciliation) — byte-unchanged.
442                        None,
443                    )
444                    .map(|(value, new_offset, col_meta)| {
445                        if emit {
446                            if let Some(ref mut out) = compaction_complex_out {
447                                // Capture the whole-collection collapsed value for the
448                                // byte-neutral Phase A output path (roborev #863).
449                                out.insert(
450                                    column.name.clone(),
451                                    (col_meta.complex_deletion, element_buf, value.clone()),
452                                );
453                            }
454                        }
455                        (value, new_offset, col_meta)
456                    })
457                } else {
458                    // Issue #1741 (per-element filtering): on the user-facing SELECT
459                    // read path (`cell_ctx.is_some()`), thread the covering deletion +
460                    // read clock + row-liveness ts into the element loop so shadowed /
461                    // TTL-expired elements are skipped from the emitted container.
462                    // `None` when no shadow context is active (tombstone-free reads and
463                    // physical consumers) — byte-unchanged.
464                    let element_filter = cell_ctx.map(|(cover, now)| ElementShadow {
465                        cover,
466                        now,
467                        row_ts: row_header.timestamp,
468                        // Issue #1741: the effective row expiry a USE_ROW_TTL
469                        // collection element inherits — computed EXACTLY like the
470                        // scalar USE_ROW_TTL cell path (see the `None if use_row_ttl`
471                        // arm below): the pk-liveness localExpirationTime
472                        // (`liveness_expires_at_seconds`, year-2106-safe i64 from
473                        // HAS_TTL), falling back to the row `local_deletion_time`
474                        // re-read UNSIGNED on oa/da when the GC clock is post-2038.
475                        row_expires_at: row_header.liveness_expires_at_seconds.or_else(|| {
476                            row_header.local_deletion_time.map(|s| {
477                                if self.has_uint_deletion_time() {
478                                    (s as u32) as i64
479                                } else {
480                                    s as i64
481                                }
482                            })
483                        }),
484                    });
485                    self.parse_complex_column(
486                        data,
487                        offset,
488                        column,
489                        complex_type,
490                        has_complex_deletion,
491                        reader,
492                        element_filter,
493                    )
494                };
495                match parse_result {
496                    Ok((value, new_offset, col_meta)) => {
497                        log::debug!(
498                            "V5CompressedLegacy:   ✓ Complex column {} '{}' = {:?}, consumed {} bytes",
499                            col_idx, column.name, value, new_offset - offset
500                        );
501                        // Normal read-path (WRITETIME/TTL queries): use the row-level liveness
502                        // timestamp unchanged.  Cassandra's WRITETIME(non_frozen_collection) on
503                        // the standard read path returns the row timestamp, not per-element max.
504                        // The delta-scan path computes its own max-element-writetime from
505                        // ComplexColumnMeta (stored separately below) and never reads this field
506                        // for collection columns.  Do NOT mutate this with max_element_writetime
507                        // here — that would silently change WRITETIME(col) on the ordinary path
508                        // (roborev Finding 1).
509                        if emit {
510                            // Issue #1741 (Finding 2): the collection's effective max
511                            // write ts is the newest of its element-level timestamps and
512                            // the row liveness ts inherited by USE_ROW_TIMESTAMP elements.
513                            // Folding `max_element_writetime` (not just the row ts) means a
514                            // collection element NEWER than a partition/range tombstone
515                            // keeps the row visible even when the row ts predates the
516                            // tombstone. `max_element_writetime` folds ALL live elements
517                            // (including shadow-dropped ones), so a wholly-shadowed
518                            // collection still contributes its element ts here.
519                            let coll_eff_ts = col_meta
520                                .max_element_writetime
521                                .max(row_header.timestamp.unwrap_or(i64::MIN));
522                            // Issue #1741 (per-element filtering): the shadowed/expired
523                            // elements have ALREADY been skipped from `value` inside
524                            // `parse_complex_column_inner`. The collection now reads as
525                            // ABSENT (null) iff the read-side filter EMPTIED it (every
526                            // element was shadowed/expired: `shadow_filtered_element_count
527                            // > 0` AND the container is empty). A collection empty for any
528                            // OTHER reason (genuinely empty / all element-tombstones, where
529                            // `shadow_filtered_element_count == 0`) keeps its prior
530                            // behavior, so tombstone-free reads and physical consumers are
531                            // byte-unchanged. `cover == None` never filters an element, so
532                            // a non-shadowing read never treats a collection as absent.
533                            let collection_absent = col_meta.shadow_filtered_element_count > 0
534                                && Self::complex_value_is_empty(&value);
535                            // Fold the collection into the ROW aggregate REGARDLESS of
536                            // absence (so a wholly-shadowed collection is still recognised
537                            // as shadowed by `row_hidden`, and Finding 2's newest-element ts
538                            // keeps a surviving row visible). Its expiring elements
539                            // contribute their expiry; `has_live_forever_element` is already
540                            // computed POST-filter, so a wholly-shadowed collection never
541                            // sets live-forever.
542                            if coll_eff_ts != i64::MIN {
543                                agg_max_cell_ts = Some(
544                                    agg_max_cell_ts.map_or(coll_eff_ts, |m| m.max(coll_eff_ts)),
545                                );
546                            }
547                            if let Some(e) = col_meta.max_element_expires_at {
548                                agg_max_expires_at =
549                                    Some(agg_max_expires_at.map_or(e, |m| m.max(e)));
550                            }
551                            if col_meta.has_live_forever_element {
552                                agg_has_live_forever = true;
553                            }
554                            if !collection_absent {
555                                if let Some(ref mut meta_map) = cell_meta {
556                                    let row_ts = row_header.timestamp.unwrap_or(0);
557                                    meta_map.insert(
558                                        column.name.clone(),
559                                        CellWriteMetadata {
560                                            write_timestamp_micros: row_ts,
561                                            expiration: None,
562                                        },
563                                    );
564                                }
565                                // DS4 (Issue #700): Store ComplexColumnMeta for delta-scan.
566                                if let Some(ref mut ccm_map) = complex_col_meta {
567                                    ccm_map.insert(column.name.clone(), col_meta);
568                                }
569                                // Issue #1334: interned name handle (Arc::clone), no alloc.
570                                cells.insert(Arc::clone(&ctp.name), value);
571                            }
572                        }
573                        offset = new_offset;
574                    }
575                    Err(e) => {
576                        log::debug!(
577                            "V5CompressedLegacy:   ✗ Complex column {} '{}' at offset {} FAILED: {}",
578                            col_idx, column.name, offset, e
579                        );
580                        break;
581                    }
582                }
583            } else {
584                // Issue #1741: peek the cell flags (offset points at the flags byte)
585                // to detect USE_ROW_TTL (0x10), which makes a cell with no explicit
586                // expiry inherit the ROW's expiry rather than being live-forever.
587                let cell_flags = data.get(offset).copied().unwrap_or(0);
588                match self.parse_cell_value_schema_order(data, offset, column, header_type, reader)
589                {
590                    Ok((value, cell_own_ts, cell_exp, new_offset)) => {
591                        log::debug!(
592                            "V5CompressedLegacy:   ✓ Column {} '{}' ({}) = {:?}, consumed {} bytes",
593                            col_idx,
594                            column.name,
595                            column.data_type,
596                            value,
597                            new_offset - offset
598                        );
599                        // Only compute and store per-cell metadata when the caller requested it.
600                        // On the normal read hot-path (want_cell_metadata == false), cell_meta is
601                        // None and this entire block is skipped — zero allocations per cell.
602                        // `emit` is false for a DROPPED column (issue #1080 Part 2): we still
603                        // advanced `offset` to consume its bytes, but emit no cell/metadata.
604                        if emit {
605                            // Issue #1741 read-side shadowing aggregate (scalars only,
606                            // no allocation). A cell tombstone is not live data. A live
607                            // cell contributes its effective write timestamp and, if
608                            // expiring (explicit per-cell TTL, or USE_ROW_TTL inheriting
609                            // the row's expiry), its expiry; otherwise it is live-forever.
610                            // Computed BEFORE the metadata block, which moves `cell_exp`.
611                            let mut dropped = false;
612                            if !matches!(value, Value::Tombstone(_)) {
613                                let eff_ts = cell_own_ts.or(row_header.timestamp);
614                                // A USE_ROW_TTL cell inherits the ROW's expiry. For a
615                                // TTL-bearing INSERT that expiry is the pk-liveness
616                                // localExpirationTime (`liveness_expires_at_seconds`,
617                                // from HAS_TTL); `local_deletion_time` is set only by
618                                // HAS_DELETION (row tombstone), so it is the fallback.
619                                let use_row_ttl = (cell_flags & 0x10) != 0;
620                                let eff_exp: Option<i64> = match cell_exp.as_ref() {
621                                    Some(e) => Some(e.expires_at_seconds),
622                                    None if use_row_ttl => {
623                                        // Liveness expiry is already the year-2106-safe
624                                        // i64 (unsigned-reinterpreted at decode, #1741 F1).
625                                        // The `local_deletion_time` fallback is stored as
626                                        // the `(u32) as i32` on-disk representation, so on
627                                        // oa/da a post-2038 GC clock must be re-read UNSIGNED
628                                        // here too rather than sign-extended negative.
629                                        row_header.liveness_expires_at_seconds.or_else(|| {
630                                            row_header.local_deletion_time.map(|s| {
631                                                if self.has_uint_deletion_time() {
632                                                    (s as u32) as i64
633                                                } else {
634                                                    s as i64
635                                                }
636                                            })
637                                        })
638                                    }
639                                    None => None,
640                                };
641                                // Issue #1741 (Finding 1): per-cell shadow/TTL filter.
642                                // Drop this data cell from the EMITTED map when its
643                                // effective write ts is shadowed by the covering deletion
644                                // (`eff_ts <= cover`) or it is TTL-expired at `now`.
645                                // `cell_ctx` is `None` for physical reads → never drops →
646                                // byte-unchanged.
647                                if let Some((cover, now)) = cell_ctx {
648                                    dropped = PartitionShadow::cell_shadowed_or_expired(
649                                        cover, now, eff_ts, eff_exp,
650                                    );
651                                }
652                                // Fold this cell into the ROW aggregate that drives the
653                                // row-level `row_hidden` decision. A dropped (shadowed/
654                                // expired) cell STILL contributes its ts + expiry — so a
655                                // row reduced to nothing is recognised as shadowed/expired
656                                // (its max ts stays `<= covering` / its expiry stays past)
657                                // rather than looking like an empty/truncated parse — but
658                                // it is NEVER counted live-forever (it is not live).
659                                if let Some(t) = eff_ts {
660                                    agg_max_cell_ts = Some(agg_max_cell_ts.map_or(t, |m| m.max(t)));
661                                }
662                                match eff_exp {
663                                    Some(e) => {
664                                        agg_max_expires_at =
665                                            Some(agg_max_expires_at.map_or(e, |m| m.max(e)));
666                                    }
667                                    None => {
668                                        if !dropped {
669                                            agg_has_live_forever = true;
670                                        }
671                                    }
672                                }
673                            }
674                            if dropped {
675                                offset = new_offset;
676                                continue;
677                            }
678                            // Only compute and store per-cell metadata when the caller requested it.
679                            // On the normal read hot-path (want_cell_metadata == false), cell_meta is
680                            // None and this entire block is skipped — zero allocations per cell.
681                            if let Some(ref mut meta_map) = cell_meta {
682                                // Resolve effective write timestamp:
683                                // use cell's own timestamp when present, else row-level liveness timestamp.
684                                let effective_ts = cell_own_ts
685                                    .unwrap_or_else(|| row_header.timestamp.unwrap_or(0));
686                                // Resolve expiration: cell-level wins; fall back to row-level TTL when
687                                // the cell used USE_ROW_TTL (cell_exp is None in that case).
688                                // USE_ROW_TTL path: row_header.ttl is the row-level TTL in seconds.
689                                // row_header.local_deletion_time is the corresponding expires_at (seconds).
690                                let row_level_exp =
691                                    match (row_header.ttl, row_header.local_deletion_time) {
692                                        (Some(ttl_s), Some(ldt_s)) => Some(CellExpiration {
693                                            ttl_seconds: ttl_s,
694                                            expires_at_seconds: ldt_s as i64,
695                                        }),
696                                        _ => None,
697                                    };
698                                let effective_exp = cell_exp.or(row_level_exp);
699                                meta_map.insert(
700                                    column.name.clone(),
701                                    CellWriteMetadata {
702                                        write_timestamp_micros: effective_ts,
703                                        expiration: effective_exp,
704                                    },
705                                );
706                            }
707                            // Issue #1334: interned name handle (Arc::clone), no String alloc.
708                            cells.insert(Arc::clone(&ctp.name), value);
709                        }
710                        offset = new_offset;
711                    }
712                    Err(e) => {
713                        log::debug!(
714                            "V5CompressedLegacy:   ✗ Column {} '{}' ({}) at offset {} FAILED: {}",
715                            col_idx,
716                            column.name,
717                            column.data_type,
718                            offset,
719                            e
720                        );
721                        // CRITICAL FIX: Stop parsing remaining columns when we hit an error
722                        // The offset doesn't advance here, but we exit the loop cleanly
723                        // rather than continuing with invalid offset
724                        break;
725                    }
726                }
727            }
728        }
729
730        // Issue #1741: stash the read-side shadowing aggregate onto the header.
731        row_header.max_data_cell_timestamp = agg_max_cell_ts;
732        row_header.max_data_cell_expires_at = agg_max_expires_at;
733        row_header.has_live_forever_data_cell = agg_has_live_forever;
734
735        log::debug!(
736            "V5CompressedLegacy: Parsed {}/{} on-disk columns (missing columns are NULL)",
737            cells.len(),
738            columns_in_order.len()
739        );
740        log::debug!(
741            "V5CompressedLegacy: Cells HashMap keys: {:?}",
742            cells.keys().collect::<Vec<_>>()
743        );
744
745        debug!("V5CompressedLegacy: Parsed total of {} cells", cells.len());
746
747        // Calculate offset after cell data (based on row_size from header)
748        //
749        // CRITICAL (Issue #237): row_size is measured from AFTER the row_size VInt,
750        // not from where it starts. This matches Cassandra's getFilePointer() semantics:
751        //   next_position = row_size_value + position_after_reading_row_size_vint
752        //
753        // Formula: (row_metadata_offset + row_size_vint_len) + row_size
754        //
755        // There is NO trailing field in V5CompressedLegacy format - the next partition/row
756        // starts immediately after row_size bytes from this position.
757        let row_size_counted_from = row_metadata_offset + row_header.row_size_vint_len;
758        let after_cells_offset = row_size_counted_from + row_size as usize;
759
760        // Validate we have enough data
761        if after_cells_offset > data.len() {
762            let remaining = data.len().saturating_sub(row_size_counted_from);
763            return Err(Error::corruption(format!(
764                "V5CompressedLegacy: Not enough bytes for row data at offset {} (need {}, have {})",
765                row_size_counted_from, row_size, remaining
766            )));
767        }
768
769        // No trailing field - next partition/row starts immediately
770        let next_offset = after_cells_offset;
771
772        debug!(
773            "V5CompressedLegacy: Row complete - row_size={} bytes, next offset = {} (counted from {}, is_static={})",
774            row_size, next_offset, row_size_counted_from, is_static
775        );
776
777        Ok((
778            cells,
779            cell_meta,
780            Some(row_header),
781            next_offset,
782            is_static,
783            complex_col_meta,
784        ))
785    }

787    /// Test-only helper that parses the cell header (flags + conditional temporal
788    /// metadata) and returns the offset at which the value bytes begin.
789    ///
790    /// This mirrors the logic in `parse_cell_value_schema_order` for the conditional
791    /// sections (Steps 1-3), but stops before the value parse.  It is used by the
792    /// S1 audit verification tests (Issue #623) to confirm that:
793    ///   - USE_ROW_TIMESTAMP (0x08) causes the timestamp VInt to be ABSENT
794    ///   - USE_ROW_TTL (0x10) without IS_EXPIRING causes LDT/TTL to be ABSENT
795    ///
796    /// Returns `(flags, value_start_offset)`.
797    #[cfg(test)]
798    pub(super) fn parse_cell_header_end_offset(
799        &self,
800        data: &[u8],
801        start_offset: usize,
802    ) -> Result<(u8, usize)> {
803        const CELL_IS_DELETED: u8 = 0x01;
804        const CELL_IS_EXPIRING: u8 = 0x02;
805        const CELL_USE_ROW_TIMESTAMP: u8 = 0x08;
806        const CELL_USE_ROW_TTL: u8 = 0x10;
807
808        if start_offset >= data.len() {
809            return Err(Error::corruption(
810                "cell_header_end_offset: no flags byte".to_string(),
811            ));
812        }
813        let flags = data[start_offset];
814        let mut offset = start_offset + 1;
815
816        let is_deleted = (flags & CELL_IS_DELETED) != 0;
817        let is_expiring = (flags & CELL_IS_EXPIRING) != 0;
818        let use_row_timestamp = (flags & CELL_USE_ROW_TIMESTAMP) != 0;
819        let use_row_ttl = (flags & CELL_USE_ROW_TTL) != 0;
820
821        // Step 1: skip timestamp VInt if not using row timestamp.
822        // Skip-only: byte advancement is identical for vint/vuint, but use the
823        // UNSIGNED variant to match the writer encoding (roborev #863).
824        if !use_row_timestamp {
825            let (remaining, _ts_delta) = parse_vuint(&data[offset..]).map_err(|e| {
826                Error::corruption(format!(
827                    "cell_header_end_offset: failed to parse timestamp VInt: {:?}",
828                    e
829                ))
830            })?;
831            offset += data[offset..].len() - remaining.len();
832        }
833        // Step 2: skip LDT VUInt if not using row TTL and (deleted or expiring)
834        if !use_row_ttl && (is_deleted || is_expiring) {
835            let (remaining, _ldt_delta) = parse_vuint(&data[offset..]).map_err(|e| {
836                Error::corruption(format!(
837                    "cell_header_end_offset: failed to parse LDT VUInt: {:?}",
838                    e
839                ))
840            })?;
841            offset += data[offset..].len() - remaining.len();
842        }
843        // Step 3: skip TTL VUInt if not using row TTL and expiring
844        if !use_row_ttl && is_expiring {
845            let (remaining, _ttl_delta) = parse_vuint(&data[offset..]).map_err(|e| {
846                Error::corruption(format!(
847                    "cell_header_end_offset: failed to parse TTL VUInt: {:?}",
848                    e
849                ))
850            })?;
851            offset += data[offset..].len() - remaining.len();
852        }
853
854        Ok((flags, offset))
855    }
