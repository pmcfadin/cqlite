3impl V5CompressedLegacyParser {
4    /// Within-partition clustering-slice variant of [`parse_block_emit`] (Issue
5    /// #954, Epic #951).
6    ///
7    /// When `row_body_window` is `Some((body_start, body_end))` (byte offsets
8    /// into `data`, in the SAME domain as `data` indices) the FIRST partition's
9    /// row-body parse is bounded to that window:
10    ///   - after the partition header is decoded, the row cursor is fast-forwarded
11    ///     to `max(after_header, body_start)` (skipping rows that precede the
12    ///     requested clustering slice), and
13    ///   - the row loop stops once the cursor reaches `body_end` (skipping rows
14    ///     after the slice).
15    ///
16    /// `body_start`/`body_end` are the byte extent of the row-index block(s) that
17    /// the authoritative BTI row index resolved as covering the requested
18    /// clustering range, so this decodes O(matched rows + index-block slack)
19    /// rather than the whole partition. The post-scan `evaluate_leaf` backstop
20    /// trims the block-granularity over-read, so the returned rows are a SUPERSET
21    /// of the exact slice and the final query output is byte-identical.
22    ///
23    /// The start fast-forward is only applied when the schema has NO static
24    /// columns (the caller enforces this): a static row precedes the clustering
25    /// rows and must be merged into each clustering row, so skipping past it would
26    /// drop static values. The end bound is always safe to apply.
27    ///
28    /// Every row this method actually decodes bumps the `work_counters`
29    /// `rows_decoded` counter so a test can prove the decode was bounded to the
30    /// slice. With `row_body_window == None` this is byte-for-byte
31    /// [`parse_block_emit`]'s original behaviour, and the counter is still bumped
32    /// (the seek path reports its full-partition decode too).
33    pub fn parse_block_emit_windowed<F>(
34        &self,
35        data: &[u8],
36        schema: Option<&TableSchema>,
37        reader: &crate::storage::sstable::reader::types::SSTableReader,
38        row_body_window: Option<(usize, usize)>,
39        mut emit: F,
40    ) -> Result<()>
41    where
42        F: FnMut((TableId, RowKey, ScanRow)) -> Result<std::ops::ControlFlow<()>>,
43    {
44        if data.is_empty() {
45            return Ok(());
46        }
47
48        // V5CompressedLegacy format stores cells WITHOUT column names,
49        // relying on schema to interpret the binary data. Schema is REQUIRED.
50        let schema = schema.ok_or_else(|| {
51            Error::schema(format!(
52                "V5CompressedLegacy format requires schema for {}.{} (cells lack column names in binary data)",
53                self.keyspace, self.table_name
54            ))
55        })?;
56
57        // Issue #1046: build the header→schema column resolution ONCE per block.
58        let resolution = RowColumnResolution::build(schema, reader);
59
60        tracing::debug!(
61            "V5CompressedLegacy: Parsing block for {}.{} ({} bytes)",
62            self.keyspace,
63            self.table_name,
64            data.len()
65        );
66        tracing::debug!(
67            "V5CompressedLegacy: Schema has {} columns",
68            schema.columns.len()
69        );
70        for (i, col) in schema.columns.iter().enumerate() {
71            tracing::debug!("  Column {}: {} ({})", i, col.name, col.data_type);
72        }
73        tracing::debug!(
74            "V5CompressedLegacy: First 64 bytes of data: {}",
75            hex::encode(&data[..std::cmp::min(64, data.len())])
76        );
77        debug!(
78            "V5CompressedLegacy: Parsing block for {}.{} ({} bytes)",
79            self.keyspace,
80            self.table_name,
81            data.len()
82        );
83
84        let mut emitted: usize = 0;
85        let mut offset = 0;
86        // Issue #1741 (F2): read-time TTL clock — captured ONCE per parser (== once per
87        // read/scan operation) in `V5CompressedLegacyParser::new`. The windowed scan
88        // driver reuses this parser across every block; sampling `self.now_secs` here
89        // (not the wall clock per block) keeps a boundary-crossing scan consistent.
90        let now_secs = self.now_secs;
91        let table_id = TableId::new(format!("{}.{}", self.keyspace, self.table_name));
92
93        // Cassandra partition key size limits (used in header validation)
94        // - CASSANDRA_MAX_KEY_SIZE: 64KB limit per Apache Cassandra specification
95        // - FORMAT_MAX_KEY_SIZE: u8 max value - V5CompressedLegacy format limitation
96        const CASSANDRA_MAX_KEY_SIZE: usize = 65536; // 64KB per Cassandra spec
97        const FORMAT_MAX_KEY_SIZE: usize = 255; // u8 max value - format limitation
98
99        // Parse ALL partitions in block (Issue #2 fix: previously only parsed one partition)
100        let mut partition_index = 0;
101        let mut skipped_partitions = 0;
102        while offset < data.len() {
103            // Cooperative cancellation (issue #2264): an uncompressed, index-less
104            // SSTable is returned to the scan as ONE contiguous block, so this loop
105            // is the 400k+-partition hot loop that the compaction streaming read
106            // (and thus a Flight `do_get`) spends its whole time in. Poll the
107            // reader's cancel token at a bounded interval so a disconnected client
108            // abandons the walk within milliseconds instead of running to
109            // completion under the coarse ~1–2 min backstop. Every 256 partitions
110            // keeps the relaxed-atomic load negligible against the per-partition
111            // parse cost.
112            if partition_index & 0xFF == 0 {
113                reader.scan_cancel.check()?;
114            }
115            tracing::debug!(
116                "V5CompressedLegacy: === PARTITION {} at offset {} (block size: {}) ===",
117                partition_index,
118                offset,
119                data.len()
120            );
121
122            // CRITICAL FIX (Issue #164): Validate partition header format before attempting parse
123            //
124            // Most compressed blocks contain EXACTLY ONE partition. After parsing the first
125            // partition's row data and trailing VInt, we should NOT assume there's another
126            // partition just because offset < data.len().
127            //
128            // Partition header format validation:
129            // - Byte 0: Flags (typically 0x00, sometimes has partition-level flags)
130            // - Byte 1: Partition key length (u8, typically 16 for UUID)
131            // - Bytes 2+: Partition key data
132            //
133            // If we don't see a valid partition header structure, we've reached the end
134            // of partitions in this block (remaining bytes are likely padding or metadata).
135            if offset >= data.len() {
136                break; // End of block
137            }
138
139            // Check if this looks like a partition header (flags byte + reasonable key length)
140            // Partition keys can be up to 64KB per Cassandra spec (composite keys, text, etc.)
141            if offset + 2 > data.len() {
142                tracing::debug!(
143                    "V5CompressedLegacy: Not enough bytes for partition header at offset {} (need 2, have {}), stopping",
144                    offset,
145                    data.len() - offset
146                );
147                break;
148            }
149
150            let flags = data[offset];
151            let key_len = data[offset + 1] as usize;
152
153            // Validate partition header:
154            // - Key length must be non-zero and within format's limit (u8 max = 255 bytes)
155            //   Note: Cassandra spec allows 64KB keys, but V5CompressedLegacy format uses u8 length
156            // - Must have enough bytes for the header (size depends on format version)
157            //
158            // VG3: oa format (hasUIntDeletionTime) uses a compact DeletionTime:
159            //   LIVE = 1 byte; DELETED = 12 bytes.  The minimum is therefore 1 byte.
160            // nb format always uses 12 bytes (4 + 8).
161            // NOTE: No heuristic validation of flags (Issue #258, #28 no-heuristics mandate)
162            let deletion_time_min = if self.has_uint_deletion_time() { 1 } else { 12 };
163            let header_min_size = 1 + 1 + key_len + deletion_time_min;
164            if key_len == 0
165                || key_len > FORMAT_MAX_KEY_SIZE.min(CASSANDRA_MAX_KEY_SIZE)
166                || offset + header_min_size > data.len()
167            {
168                tracing::warn!(
169                    "V5CompressedLegacy: Skipping malformed partition header at offset {} \
170                     (flags=0x{:02x}, key_len={}, need {} bytes, have {}, partition={}): header validation failed",
171                    offset,
172                    flags,
173                    key_len,
174                    header_min_size,
175                    data.len() - offset,
176                    partition_index
177                );
178                // Try to skip to next potential partition boundary
179                skipped_partitions += 1;
180                offset += 1; // Minimal forward progress to avoid infinite loop
181                continue; // Skip this partition, try next
182            }
183
184            // Try to parse partition header
185            match self.parse_partition_header_full(data, offset) {
186                Ok((partition_key, new_offset, partition_deletion)) => {
187                    let header_size = new_offset - offset;
188                    offset = new_offset;
189
190                    // Issue #1741: per-partition read-side shadowing, active ONLY for
191                    // user-facing query reads (`self.read_shadowing`). Captures the
192                    // partition-level deletion and tracks the open range tombstone as
193                    // rows are walked, so partition/range-tombstone-shadowed and
194                    // TTL-expired rows are hidden (matching Cassandra SELECT). `None`
195                    // for physical consumers (verify / get_all_entries) which must see
196                    // every on-disk row. Un-gated (not behind write-support).
197                    let mut shadow = self.read_shadowing.then(|| {
198                        PartitionShadow::open(
199                            now_secs,
200                            partition_deletion,
201                            clustering_reversed_flags(schema),
202                        )
203                    });
204
205                    // Issue #954: when a within-partition clustering-slice window is
206                    // supplied, fast-forward the row cursor of the FIRST partition to
207                    // the first row-index block covering the slice (skipping rows
208                    // before it). Applied only to `partition_index == 0` because the
209                    // seek decodes exactly one target partition. The end bound
210                    // (`body_end`) stops the row loop below. The caller guarantees a
211                    // start fast-forward is only requested when the schema has no
212                    // static columns, so this never skips a static prefix.
213                    let row_body_end = match row_body_window {
214                        Some((body_start, body_end)) if partition_index == 0 => {
215                            if body_start > offset && body_start <= data.len() {
216                                // Issue #1741 (Finding 1): a range tombstone that
217                                // OPENS before `body_start` can cover rows inside the
218                                // requested clustering slice. Fast-forwarding straight
219                                // to `body_start` would skip those markers, so the open
220                                // range would never be fed into `PartitionShadow` and a
221                                // slice read could return rows a full scan correctly
222                                // hides. When shadowing is active, replay the markers
223                                // from the partition body start up to `body_start` into
224                                // the shadow FSM first. If priming cannot faithfully
225                                // reconstruct the state (unrepresentable marker or an
226                                // undecodable row), fall back to a full-partition decode
227                                // from the body start rather than skip the markers: the
228                                // post-scan clustering backstop still trims the rows
229                                // before the slice, so correctness holds and only the
230                                // fast path is lost. When shadowing is off (physical
231                                // read) there is nothing to prime — keep the byte-for-
232                                // byte fast-forward (no-RT fast path unaffected).
233                                //
234                                // Issue #1741 (Finding 2): priming decodes the pre-window
235                                // prefix, regressing the row-index fast-forward from
236                                // O(slice) to O(prefix+slice) on a wide partition. Pay it
237                                // ONLY when a range tombstone can actually reach the slice.
238                                // Partition-deletion shadowing needs NO prefix decode (it is
239                                // captured from the header into `shadow`). Range tombstones:
240                                // when the SSTable's authoritative EncodingStats prove there
241                                // are no deletions (hence no range tombstones), keep the
242                                // O(slice) `offset = body_start` fast-forward and skip
243                                // priming. Priming itself is now marker-ONLY — it skips row
244                                // bodies via framing (no cell decode), feeding only the RT
245                                // markers into the FSM — so even when it runs it never
246                                // re-decodes the prefix cells.
247                                let primed = match shadow.as_mut() {
248                                    Some(_) if !self.sstable_may_have_range_tombstones() => true,
249                                    Some(sh) => self.prime_shadow_before_window(
250                                        data, offset, body_start, schema, sh,
251                                    ),
252                                    None => true,
253                                };
254                                if primed {
255                                    offset = body_start;
256                                }
257                            }
258                            Some(body_end)
259                        }
260                        _ => None,
261                    };
262
263                    tracing::debug!(
264                        "V5CompressedLegacy: Partition {} - Parsed partition key: {} bytes (header consumed {} bytes, now at offset {})",
265                        partition_index,
266                        partition_key.0.len(),
267                        header_size,
268                        offset
269                    );
270                    tracing::debug!(
271                        "V5CompressedLegacy: Partition {} - Row data starts at offset {}, remaining: {} bytes",
272                        partition_index,
273                        offset,
274                        data.len() - offset
275                    );
276                    tracing::debug!(
277                        "V5CompressedLegacy: Partition {} - Row data hex (first 128 bytes): {}",
278                        partition_index,
279                        hex::encode(&data[offset..std::cmp::min(offset + 128, data.len())])
280                    );
281
282                    debug!(
283                        "V5CompressedLegacy: Parsed partition key: {} bytes, now at offset {}",
284                        partition_key.0.len(),
285                        offset
286                    );
287
288                    // Parse ALL rows in this partition (Issue #166 fix: multi-row partition support)
289                    //
290                    // V5CompressedLegacy partitions can contain multiple rows with different clustering keys.
291                    // We use structural parsing (peek_is_partition_header) to detect partition boundaries,
292                    // not flag value heuristics (Issue #258, #28 no-heuristics mandate).
293                    // We parse rows in a loop until we encounter:
294                    // - End of block (offset >= data.len())
295                    // - END_OF_PARTITION marker (flags == 0x01, Issue #229 fix)
296                    // - Next partition header (detected via peek_is_partition_header)
297                    // - Parse error (invalid row data)
298
299                    // Issue #480 FIX: Static cell handling
300                    //
301                    // Cassandra static rows are stored once per partition (before clustering rows).
302                    // They should NOT be emitted as separate result entries — instead their column
303                    // values must be merged into each clustering row that follows in the partition.
304                    //
305                    // We accumulate static cells here and inject them into every clustering row.
306                    // Issue #1642 (K3): positional `RowCells`, matching the decoder emit.
307                    let mut static_cells: RowCells = Vec::new();
308                    let mut row_count = 0;
309                    loop {
310                        // Issue #954: stop at the clustering-slice end bound. The
311                        // row-index block extent (`body_end`) is the authoritative
312                        // upper byte bound of the rows that may fall in the requested
313                        // clustering range; rows past it are outside the slice, so we
314                        // stop decoding (the post-scan backstop already trims any
315                        // block-granularity over-read within the window).
316                        if let Some(body_end) = row_body_end {
317                            if offset >= body_end {
318                                break;
319                            }
320                        }
321
322                        // Issue #229 FIX: Check for END_OF_PARTITION marker BEFORE attempting row parse
323                        //
324                        // Per Cassandra's UnfilteredSerializer.java (lines 102, 730-732):
325                        // When END_OF_PARTITION (0x01) is set in the flags byte, nothing follows.
326                        // The partition is complete and we should move to the next partition.
327                        if offset < data.len() && Self::is_end_of_partition(data[offset]) {
328                            tracing::debug!(
329                                "V5CompressedLegacy: Partition {} complete via END_OF_PARTITION marker at offset {} ({} rows parsed)",
330                                partition_index, offset, row_count
331                            );
332                            offset += 1; // Skip the END_OF_PARTITION marker byte
333                            break; // Move to next partition
334                        }
335
336                        // Issue #229 FIX: Check for range tombstone marker
337                        //
338                        // Per Cassandra's UnfilteredSerializer.java (lines 103, 735-738):
339                        // When IS_MARKER (0x02) is set, this is a range tombstone boundary, not a row.
340                        // We skip these markers for now (full implementation would parse deletion ranges).
341                        if offset < data.len() && Self::is_range_tombstone_marker(data[offset]) {
342                            tracing::debug!(
343                                "V5CompressedLegacy: Range tombstone marker at offset {} (partition {}), skipping",
344                                offset, partition_index
345                            );
346                            // Issue #1741: when read-side shadowing is active, decode
347                            // the marker (bounds + deletion time) and feed the
348                            // range-tombstone FSM so covered rows are shadowed;
349                            // otherwise (physical read) only advance past it.
350                            if let Some(sh) = shadow.as_mut() {
351                                match self.parse_range_tombstone_marker_full(data, offset, schema) {
352                                    Ok((
353                                        bound_values,
354                                        bound_kind,
355                                        del_primary,
356                                        del_secondary,
357                                        next_offset,
358                                    )) => {
359                                        if let Err(e) = sh.feed_range_marker(
360                                            bound_values,
361                                            bound_kind,
362                                            del_primary,
363                                            del_secondary,
364                                        ) {
365                                            tracing::debug!(
366                                                "V5CompressedLegacy: range tombstone FSM error at offset {}: {}",
367                                                offset, e
368                                            );
369                                            break; // Unrepresentable marker, end partition
370                                        }
371                                        offset = next_offset;
372                                        continue; // Continue to next row/marker
373                                    }
374                                    Err(e) => {
375                                        tracing::debug!(
376                                            "V5CompressedLegacy: Failed to parse range tombstone marker at offset {}: {}",
377                                            offset, e
378                                        );
379                                        break; // Can't parse marker, end partition
380                                    }
381                                }
382                            }
383                            match self.skip_range_tombstone_marker(data, offset, schema) {
384                                Ok(next_offset) => {
385                                    offset = next_offset;
386                                    continue;
387                                }
388                                Err(e) => {
389                                    tracing::debug!(
390                                        "V5CompressedLegacy: Failed to skip range tombstone marker at offset {}: {}",
391                                        offset, e
392                                    );
393                                    break;
394                                }
395                            }
396                        }
397
398                        match self.parse_row_data_with_offset(
399                            data,
400                            offset,
401                            Some(schema),
402                            reader,
403                            false,
404                            &resolution,
405                            shadow.as_ref(),
406                        ) {
407                            Ok((
408                                mut cells,
409                                _row_cell_meta,
410                                row_header_opt,
411                                next_offset,
412                                is_static,
413                                _complex_meta,
414                            )) => {
415                                // Update offset to point to the next row or partition
416                                offset = next_offset;
417                                row_count += 1;
418
419                                tracing::debug!(
420                                    "V5CompressedLegacy: Partition {} Row {} - Parsed {} cells, now at offset {} (is_static={})",
421                                    partition_index,
422                                    row_count,
423                                    cells.len(),
424                                    offset,
425                                    is_static
426                                );
427
428                                if let Some(ref header) = row_header_opt {
429                                    tracing::debug!(
430                                        "V5CompressedLegacy: Row {} metadata - timestamp={:?}, ttl={:?}, deletion={:?}",
431                                        row_count,
432                                        header.timestamp, header.ttl, header.local_deletion_time
433                                    );
434                                }
435
436                                debug!(
437                                    "V5CompressedLegacy: Parsed {} cells from row {} (is_static={})",
438                                    cells.len(),
439                                    row_count,
440                                    is_static
441                                );
442
443                                // Issue #480 FIX: Static row handling
444                                //
445                                // Static rows are stored once per partition and contain values for
446                                // STATIC columns (e.g. `static_data TEXT STATIC`). They must NOT
447                                // be emitted as standalone result rows. Instead, store the static
448                                // column values and merge them into each subsequent clustering row.
449                                if is_static {
450                                    tracing::debug!(
451                                        "V5CompressedLegacy: Partition {} - Storing {} static cells for merging into clustering rows",
452                                        partition_index,
453                                        cells.len()
454                                    );
455                                    // Issue #1741 (Finding 1): a static row can itself be
456                                    // shadowed by the partition tombstone (static write ts
457                                    // <= markedForDeleteAt) or expired by its own TTL. If
458                                    // so its cells are STALE and must NOT be merged into a
459                                    // surviving clustering row, or a SELECT would resurface
460                                    // the deleted/expired static value. Static rows carry no
461                                    // clustering key, so an open range tombstone never covers
462                                    // them (empty clustering). No-op when shadowing is off.
463                                    let static_hidden = shadow.as_ref().is_some_and(|sh| {
464                                        row_header_opt
465                                            .as_ref()
466                                            .is_some_and(|h| sh.row_hidden(h, &[]))
467                                    });
468                                    static_cells = if static_hidden { Vec::new() } else { cells };
469                                    // Do NOT push to results — static rows are not result rows
470                                    // Continue to next row/marker in partition
471                                } else {
472                                    // Merge static cells into this clustering row (Issue #480;
473                                    // positional, clustering-row-wins, issue #1642).
474                                    merge_static_cells(&mut cells, &static_cells);
475
476                                    // Issue #1741: hide rows shadowed by a partition or
477                                    // range tombstone, or expired by TTL, matching a
478                                    // Cassandra SELECT. Active only for query reads
479                                    // (shadow is `Some`). A row reduced to only its
480                                    // primary key by per-cell filtering is hidden here
481                                    // too: the dropped cells still fold into the row
482                                    // aggregate, so `row_hidden` sees it shadowed/expired.
483                                    let hidden = shadow.as_ref().is_some_and(|sh| {
484                                        row_header_opt.as_ref().is_some_and(|h| {
485                                            let clustering = if sh.needs_clustering() {
486                                                extract_clustering_values(&cells, schema)
487                                            } else {
488                                                Vec::new()
489                                            };
490                                            sh.row_hidden(h, &clustering)
491                                        })
492                                    });
493
494                                    // Issue #505/#932: row-tombstone display rule now
495                                    // lives in the shared `build_display_row` helper.
496                                    let row_value =
497                                        build_display_row(cells, row_header_opt.as_ref(), schema);
498
499                                    // Issue #954: count each clustering row actually
500                                    // decoded out of Data.db so a slice query can be
501                                    // proven to decode O(matched rows + index block),
502                                    // not the whole partition. Counted at the row
503                                    // grain (here), distinct from the per-partition
504                                    // `partitions_decoded`. Gated `not(tombstones)`
505                                    // because the mutator is compiled only there.
506                                    #[cfg(not(feature = "tombstones"))]
507                                    crate::storage::sstable::work_counters::add_rows_decoded(1);
508
509                                    if !hidden {
510                                        match emit((
511                                            table_id.clone(),
512                                            partition_key.clone(),
513                                            row_value,
514                                        ))? {
515                                            std::ops::ControlFlow::Continue(()) => emitted += 1,
516                                            // Consumer dropped (streaming receiver gone): stop parsing.
517                                            std::ops::ControlFlow::Break(()) => return Ok(()),
518                                        }
519                                    }
520                                }
521
522                                // Check if we're at the end of the partition
523                                if offset >= data.len() {
524                                    debug!(
525                                        "V5CompressedLegacy: Partition {} complete: {} rows parsed (end of block)",
526                                        partition_index, row_count
527                                    );
528                                    break; // End of block
529                                }
530
531                                // CRITICAL FIX (Issue #166): NO HEURISTICS - Try-parse approach
532                                //
533                                // Instead of guessing based on byte patterns (e.g., checking if flags <= 0x20
534                                // or validating key_len ranges), we ACTUALLY TRY TO PARSE the next structure.
535                                //
536                                // Why heuristics fail:
537                                // - Row with small value (e.g., boolean=0x0A) can look like key_len
538                                // - Row flags=0x00 or 0x20 pass "<= 0x20" checks meant for partitions
539                                // - Any byte-pattern guessing will eventually fail on edge cases
540                                //
541                                // The only reliable approach: try to parse as partition header.
542                                // If that succeeds, it's a partition. If it fails, continue with rows.
543                                if self.peek_is_partition_header(data, offset) {
544                                    debug!(
545                                        "V5CompressedLegacy: Partition {} complete: {} rows parsed (next partition detected at offset {})",
546                                        partition_index, row_count, offset
547                                    );
548                                    break; // Next partition starts here
549                                }
550
551                                // Peek failed - not a partition header, so continue parsing rows
552                                debug!(
553                                    "V5CompressedLegacy: Partition {} - Continuing to row {} at offset {} (peek confirmed this is NOT a partition header)",
554                                    partition_index, row_count + 1, offset
555                                );
556                            }
557                            Err(e) => {
558                                // End of valid data in partition
559                                debug!(
560                                    "V5CompressedLegacy: Partition {} ended after {} rows: {}",
561                                    partition_index, row_count, e
562                                );
563                                if row_count == 0 {
564                                    // If we couldn't parse even one row, log as error
565                                    tracing::error!(
566                                        "V5CompressedLegacy: Partition {} - Failed to parse first row at offset {}: {}",
567                                        partition_index, offset, e
568                                    );
569                                }
570                                break; // End of valid data in partition
571                            }
572                        }
573                    }
574
575                    partition_index += 1;
576                }
577                Err(e) => {
578                    tracing::warn!(
579                        "V5CompressedLegacy: Failed to parse partition header at offset {} \
580                         (partition={}): {}. Attempting to continue to next partition.",
581                        offset,
582                        partition_index,
583                        e
584                    );
585                    // Try to skip forward to find next partition
586                    skipped_partitions += 1;
587                    offset += 1;
588                    continue; // Skip this partition, try next
589                }
590            }
591        }
592
593        if skipped_partitions > 0 {
594            tracing::warn!(
595                "V5CompressedLegacy: Successfully parsed {} entries, skipped {} malformed partitions",
596                emitted,
597                skipped_partitions
598            );
599        }
600
601        debug!(
602            "V5CompressedLegacy: Parsed {} total entries from block",
603            emitted
604        );
605
606        Ok(())
607    }

609    /// Issue #1741 (Finding 1): replay the range-tombstone (and any other)
610    /// unfiltereds between the partition body start (`start`) and the
611    /// clustering-slice window start (`body_start`) into `shadow`, so an open
612    /// range tombstone that begins BEFORE the slice correctly shadows covered
613    /// rows inside it. Rows in this prefix are NOT emitted — only markers move
614    /// the FSM. `body_start` aligns to the start of an unfiltered (a BTI row-index
615    /// block boundary), so markers before it are fully contained in the prefix.
616    ///
617    /// Returns `true` when the prefix was replayed cleanly all the way to
618    /// `body_start` (the caller may then fast-forward). Returns `false` when the
619    /// state cannot be faithfully reconstructed (an unrepresentable marker, an
620    /// undecodable row framing, an early END_OF_PARTITION, or a non-advancing
621    /// parse); the caller then declines the fast-forward and decodes the full
622    /// partition, which is still correct because the post-scan backstop trims rows
623    /// before the slice.
624    ///
625    /// Issue #1741 (Finding 2): this is a MARKER-ONLY scan. Data rows in the prefix
626    /// are skipped via their framing (`skip_row_framing`) WITHOUT decoding any cell
627    /// values — only range-tombstone markers move the FSM. It therefore adds no
628    /// per-cell decode / allocation over the pre-window prefix; only the (cheap)
629    /// row framing (flags + clustering + row_size VInts) is parsed to advance. It
630    /// runs solely on the slice-read fast-forward branch when shadowing is active
631    /// AND the SSTable may contain range tombstones (see the call site gate).
632    fn prime_shadow_before_window(
633        &self,
634        data: &[u8],
635        start: usize,
636        body_start: usize,
637        schema: &TableSchema,
638        shadow: &mut PartitionShadow,
639    ) -> bool {
640        let mut offset = start;
641        while offset < body_start {
642            if offset >= data.len() {
643                return false;
644            }
645            let flags = data[offset];
646            // An END_OF_PARTITION before the window means the window is not in this
647            // partition — bail to the safe full-decode path.
648            if Self::is_end_of_partition(flags) {
649                return false;
650            }
651            if Self::is_range_tombstone_marker(flags) {
652                match self.parse_range_tombstone_marker_full(data, offset, schema) {
653                    Ok((bound_values, bound_kind, del_primary, del_secondary, next_offset)) => {
654                        if shadow
655                            .feed_range_marker(bound_values, bound_kind, del_primary, del_secondary)
656                            .is_err()
657                        {
658                            return false;
659                        }
660                        if next_offset <= offset {
661                            return false; // non-advancing parse guard
662                        }
663                        offset = next_offset;
664                        continue;
665                    }
666                    Err(_) => return false,
667                }
668            }
669            // A data row in the prefix: advance past it via framing ONLY (no cell
670            // decode), so the fast-forward stays O(prefix framing), not O(prefix
671            // cells). The prefix carries no static row (the caller only requests a
672            // start fast-forward when the schema has no static columns).
673            match self.skip_row_framing(data, offset, schema) {
674                Ok(next_offset) => {
675                    if next_offset <= offset {
676                        return false; // non-advancing parse guard
677                    }
678                    offset = next_offset;
679                }
680                Err(_) => return false,
681            }
682        }
683        true
684    }

686    /// Issue #1741 (Finding 2): compute the offset immediately after the row at
687    /// `offset` by parsing ONLY its framing — flags, clustering prefix, and the
688    /// `row_size` VInt — never decoding cell values. Mirrors the authoritative
689    /// offset arithmetic in `parse_row_data_with_offset`
690    /// (`next = row_metadata_offset + row_size_vint_len + row_size`) so a marker-only
691    /// prefix scan can skip data rows cheaply. Returns an error on truncation or an
692    /// out-of-range `row_size`, which the caller treats as "cannot prime".
693    pub(super) fn skip_row_framing(
694        &self,
695        data: &[u8],
696        offset: usize,
697        schema: &TableSchema,
698    ) -> Result<usize> {
699        let (row_flags, extended_flags, flags_size) = self.parse_row_flags(data, offset)?;
700        let after_flags = offset + flags_size;
701        let is_static = extended_flags
702            .map(|ef| (ef & EXTENDED_IS_STATIC) != 0)
703            .unwrap_or(false);
704        // Static rows carry no clustering prefix; regular rows do.
705        let row_metadata_offset = if is_static {
706            after_flags
707        } else {
708            let (_clustering, after_clustering) =
709                self.parse_clustering_prefix(data, after_flags, schema)?;
710            after_clustering
711        };
712        let (row_header, row_size) =
713            self.parse_row_metadata(data, row_metadata_offset, row_flags, extended_flags)?;
714        let next_offset = (row_metadata_offset + row_header.row_size_vint_len) + row_size as usize;
715        if next_offset > data.len() {
716            return Err(Error::corruption(
717                "prime_shadow: row framing extends past decompressed block".to_string(),
718            ));
719        }
720        Ok(next_offset)
721    }

723    /// Parse all partitions in a decompressed block, returning per-row timestamps.
724    ///
725    /// This is the compaction-specific variant of [`parse_block`].  It returns
726    /// `(TableId, RowKey, ScanRow, row_timestamp_micros)` so that the k-way merger
727    /// can perform timestamp-accurate last-write-wins ordering rather than
728    /// falling back to `SystemTime::now()`.
729    ///
730    /// Row tombstones are emitted as `Value::Tombstone(RowTombstone)` with their
731    /// actual `deletion_time`.  Cell tombstones within live rows are stored as
732    /// `Value::Tombstone(CellTombstone)` inside the `Value::Map`, again carrying
733    /// the cell-level deletion timestamp.
734    ///
735    /// The `row_timestamp_micros` in the returned tuple is the row-level write
736    /// timestamp decoded from the `HAS_TIMESTAMP` field in the row header
737    /// (`min_timestamp + delta`).  For row tombstones the same timestamp also
738    /// appears in `TombstoneInfo::deletion_time`.
739    ///
740    /// Normal user-facing scan/get paths should use [`parse_block`] instead.
741    /// (Issue #505)
742    pub fn parse_block_with_timestamps(
743        &self,
744        data: &[u8],
745        schema: Option<&TableSchema>,
746        reader: &crate::storage::sstable::reader::types::SSTableReader,
747    ) -> Result<Vec<(TableId, RowKey, ScanRow, i64)>> {
748        // Thin wrapper that collects the streaming emit variant into a Vec, so
749        // every existing caller/test is byte-for-byte unchanged (issue #827).
750        let mut results: Vec<(TableId, RowKey, ScanRow, i64)> = Vec::new();
751        self.parse_block_with_timestamps_emit(data, schema, reader, |entry| {
752            results.push(entry);
753            Ok(std::ops::ControlFlow::Continue(()))
754        })?;
755        Ok(results)
756    }

758    /// Streaming variant of [`parse_block_with_timestamps`]: invokes `emit` for
759    /// each parsed `(TableId, RowKey, ScanRow, row_timestamp_micros)` entry rather
760    /// than collecting into a `Vec`, so the compaction read path can forward
761    /// rows into a bounded channel without materialising the whole block at once
762    /// (issue #827). Returning `ControlFlow::Break` from `emit` stops parsing
763    /// early — used when the streaming consumer is dropped.
764    ///
765    /// The tombstone/timestamp semantics are byte-identical to
766    /// [`parse_block_with_timestamps`] (Issue #505/#533): a row tombstone is
767    /// emitted as `Value::Tombstone` carrying its `markedForDeleteAt`, and the
768    /// fourth tuple element is the row write timestamp for live rows.
769    pub fn parse_block_with_timestamps_emit<F>(
770        &self,
771        data: &[u8],
772        schema: Option<&TableSchema>,
773        reader: &crate::storage::sstable::reader::types::SSTableReader,
774        mut emit: F,
775    ) -> Result<()>
776    where
777        F: FnMut((TableId, RowKey, ScanRow, i64)) -> Result<std::ops::ControlFlow<()>>,
778    {
779        if data.is_empty() {
780            return Ok(());
781        }
782
783        let schema = schema.ok_or_else(|| {
784            Error::schema(format!(
785                "V5CompressedLegacy (compaction) format requires schema for {}.{}",
786                self.keyspace, self.table_name
787            ))
788        })?;
789
790        let mut offset = 0;
791        let mut skipped_partitions = 0;
792
793        // Wrap `emit` so a Break is observable here as well as inside the
794        // one-partition parser (which stops its inner row loop on Break). This
795        // lets the outer loop terminate promptly when the consumer is dropped.
796        // `Cell` so the wrapping closure can borrow it shared while the outer
797        // loop also reads it between calls.
798        let broke = std::cell::Cell::new(false);
799        let mut tracking_emit = |entry| -> Result<std::ops::ControlFlow<()>> {
800            let flow = emit(entry)?;
801            if matches!(flow, std::ops::ControlFlow::Break(())) {
802                broke.set(true);
803            }
804            Ok(flow)
805        };
806
807        while offset < data.len() {
808            match self.parse_one_partition_with_timestamps(
809                &data[offset..],
810                Some(schema),
811                reader,
812                // The whole block is present; never request a refill. A trailing
813                // parse failure is terminal here (matches the legacy
814                // `Err(_) => break` behaviour of the original loop).
815                //
816                // Compaction NEVER shadows: this parser is built without
817                // `read_shadowing`, so the merger sees the raw tombstones/expired
818                // cells it needs to reconcile across generations (#1741).
819                true,
820                &mut tracking_emit,
821            )? {
822                ParseStep::Emitted(consumed) => {
823                    if consumed == 0 {
824                        // Defensive: avoid an infinite loop on a zero-byte
825                        // partition (should not happen — a header is >= 2 bytes).
826                        skipped_partitions += 1;
827                        offset += 1;
828                    } else {
829                        offset += consumed;
830                    }
831                }
832                // `at_final_chunk = true` collapses NeedMore into Done: there is
833                // no further chunk to append, so a truncated tail is end-of-data.
834                ParseStep::NeedMore | ParseStep::Done => break,
835            }
836            // Propagate an early Break from `emit` (consumer dropped).
837            if broke.get() {
838                break;
839            }
840        }
841
842        if skipped_partitions > 0 {
843            tracing::warn!(
844                "V5CompressedLegacy (compaction): skipped {} malformed partitions",
845                skipped_partitions
846            );
847        }
848
849        Ok(())
850    }
851
852    /// Parse exactly ONE partition from the front of `data`, emitting each row
853    /// via `emit`, and report how the parse terminated (issue #827).
854    ///
855    /// This isolates the body of the outer partition loop so the sliding-window
856    /// compaction driver can drain one partition at a time and `drain(0..consumed)`
857    /// from its window between calls. The crucial distinction over the legacy
858    /// monolithic loop is `NeedMore` vs `Done`:
859    ///
860    /// - [`ParseStep::Emitted(consumed)`] — a full partition was parsed and
861    ///   terminated by an END_OF_PARTITION marker or a confirmed next-partition
862    ///   header. `consumed` bytes may be drained from the window.
863    /// - [`ParseStep::NeedMore`] — `data` is (possibly) truncated mid-partition
864    ///   and `!at_final_chunk`, so the caller must append the next chunk and
865    ///   retry. NEVER returned when `at_final_chunk` is true.
866    /// - [`ParseStep::Done`] — genuine end of partitions, or (when
867    ///   `at_final_chunk`) a trailing truncation that cannot be resolved by more
868    ///   data. Terminal.
869    ///
870    /// `at_final_chunk` flips a mid-partition parse failure between a refill
871    /// request (`NeedMore`) and a terminal stop. The legacy code conflated
872    /// parse-error with end-of-partitions (`Err(_) => break`); doing that
873    /// mid-stream would silently drop every partition after a chunk boundary, so
874    /// we return `NeedMore` whenever the buffer may simply be truncated and we
875    /// are not yet at the final chunk.
876    /// Read-side shadowing (issue #1741) is driven by `self.read_shadowing`: when the
877    /// parser was built for a user-facing query read, rows shadowed by a partition/
878    /// range tombstone or expired by TTL are dropped before they reach `emit`. The
879    /// compaction read path builds the parser WITHOUT shadowing (`read_shadowing ==
880    /// false`) — it needs the raw tombstones/expired cells preserved to reconcile
881    /// across generations. Un-gated: read correctness does not depend on `write-support`.
882    pub fn parse_one_partition_with_timestamps<F>(
883        &self,
884        data: &[u8],
885        schema: Option<&TableSchema>,
886        reader: &crate::storage::sstable::reader::types::SSTableReader,
887        at_final_chunk: bool,
888        emit: &mut F,
889    ) -> Result<ParseStep>
890    where
891        F: FnMut((TableId, RowKey, ScanRow, i64)) -> Result<std::ops::ControlFlow<()>>,
892    {
893        if data.is_empty() {
894            return Ok(ParseStep::Done);
895        }
896
897        let schema = schema.ok_or_else(|| {
898            Error::schema(format!(
899                "V5CompressedLegacy (compaction) format requires schema for {}.{}",
900                self.keyspace, self.table_name
901            ))
902        })?;
903
904        // Issue #1640 (K1): thin adapter over the single sliding-window partition
905        // driver. `TimestampPolicy` supplies the three per-consumer hooks
906        // (read-side shadow open, range-tombstone FSM feed, static-merge +
907        // display-row build); the driver owns the framing skeleton + ParseStep +
908        // pending buffering this function used to hand-roll.
909        let mut policy = TimestampPolicy::new(self);
910        self.drive_partition_sliding(data, schema, reader, at_final_chunk, &mut policy, |row| {
911            emit(row)
912        })
913    }
914}

916/// Issue #1640 (K1): [`SlidingPartitionPolicy`] for the streaming-scan /
917/// compaction-read `(TableId, RowKey, ScanRow, row_timestamp)` path
918/// (`parse_block_with_timestamps` family). Read-side shadowing is active only
919/// when the parser was built for a user-facing query read (`read_shadowing`);
920/// physical consumers leave `shadow` `None` and see every on-disk row.
921struct TimestampPolicy<'a> {
922    parser: &'a V5CompressedLegacyParser,
923    table_id: TableId,
924    partition_key: RowKey,
925    /// Issue #1741 per-partition read-side shadow; `None` for physical reads.
926    shadow: Option<PartitionShadow>,
927    /// Issue #480 static cells accumulated for merge into each clustering row.
928    /// Issue #1642 (K3): positional `RowCells`, matching the decoder emit.
929    static_cells: RowCells,
930}
931
932impl<'a> TimestampPolicy<'a> {
933    fn new(parser: &'a V5CompressedLegacyParser) -> Self {
934        Self {
935            parser,
936            table_id: TableId::new(format!("{}.{}", parser.keyspace, parser.table_name)),
937            partition_key: RowKey::new(Vec::new()),
938            shadow: None,
939            static_cells: Vec::new(),
940        }
941    }
942}
943
944impl SlidingPartitionPolicy for TimestampPolicy<'_> {
945    type Row = (TableId, RowKey, ScanRow, i64);
946
947    fn on_partition_open(
948        &mut self,
949        partition_key: RowKey,
950        partition_deletion: Option<(i64, i32)>,
951        schema: &TableSchema,
952        _pending: &mut Vec<Self::Row>,
953    ) {
954        self.partition_key = partition_key;
955        // Issue #1741 (F2): reuse the parser's once-per-read `now_secs`, not a
956        // per-block wall-clock sample, so all blocks of one scan share one `now`.
957        self.shadow = self.parser.read_shadowing.then(|| {
958            PartitionShadow::open(
959                self.parser.now_secs,
960                partition_deletion,
961                clustering_reversed_flags(schema),
962            )
963        });
964    }
965
966    fn on_range_marker(
967        &mut self,
968        data: &[u8],
969        offset: usize,
970        schema: &TableSchema,
971        _pending: &mut Vec<Self::Row>,
972    ) -> MarkerOutcome {
973        // Issue #1741: on the user-facing scan path decode the marker and feed
974        // the range-tombstone FSM so covered rows are shadowed; the physical path
975        // (shadow == None) only advances past it.
976        if let Some(sh) = self.shadow.as_mut() {
977            match self
978                .parser
979                .parse_range_tombstone_marker_full(data, offset, schema)
980            {
981                Ok((bv, bk, dp, ds, next_offset)) => {
982                    if sh.feed_range_marker(bv, bk, dp, ds).is_err() {
983                        return MarkerOutcome::Stop;
984                    }
985                    MarkerOutcome::Advanced(next_offset)
986                }
987                Err(_) => MarkerOutcome::Stop,
988            }
989        } else {
990            match self
991                .parser
992                .skip_range_tombstone_marker(data, offset, schema)
993            {
994                Ok(next_offset) => MarkerOutcome::Advanced(next_offset),
995                Err(_) => MarkerOutcome::Stop,
996            }
997        }
998    }

1000    fn on_data_row(
1001        &mut self,
1002        data: &[u8],
1003        offset: usize,
1004        schema: &TableSchema,
1005        reader: &crate::storage::sstable::reader::types::SSTableReader,
1006        resolution: &RowColumnResolution,
1007        pending: &mut Vec<Self::Row>,
1008    ) -> Option<usize> {
1009        match self.parser.parse_row_data_with_offset(
1010            data,
1011            offset,
1012            Some(schema),
1013            reader,
1014            false,
1015            resolution,
1016            self.shadow.as_ref(),
1017        ) {
1018            Ok((
1019                mut cells,
1020                _row_cell_meta,
1021                row_header_opt,
1022                next_offset,
1023                is_static,
1024                _complex_meta,
1025            )) => {
1026                // Issue #932 (K1): the ONE row-write-timestamp coexistence rule.
1027                let row_ts = row_write_timestamp(&row_header_opt);
1028
1029                if is_static {
1030                    // Issue #1741 (Finding 1): drop a shadowed/expired static row's
1031                    // cells so a surviving clustering row does not resurface stale
1032                    // static data. No-op when shadowing is off.
1033                    let static_hidden = self.shadow.as_ref().is_some_and(|sh| {
1034                        row_header_opt
1035                            .as_ref()
1036                            .is_some_and(|h| sh.row_hidden(h, &[]))
1037                    });
1038                    self.static_cells = if static_hidden { Vec::new() } else { cells };
1039                } else {
1040                    // Positional, clustering-row-wins merge (issue #1642).
1041                    merge_static_cells(&mut cells, &self.static_cells);
1042
1043                    // Issue #1741: hide rows shadowed by a partition/range tombstone
1044                    // or expired by TTL (matching a Cassandra SELECT). No-op on the
1045                    // physical path (shadow == None).
1046                    let hidden = self.shadow.as_ref().is_some_and(|sh| {
1047                        row_header_opt.as_ref().is_some_and(|h| {
1048                            let clustering = if sh.needs_clustering() {
1049                                extract_clustering_values(&cells, schema)
1050                            } else {
1051                                Vec::new()
1052                            };
1053                            sh.row_hidden(h, &clustering)
1054                        })
1055                    });
1056
1057                    // Issue #505/#932: row-tombstone display rule lives in the
1058                    // shared `build_display_row` helper.
1059                    let row_value = build_display_row(cells, row_header_opt.as_ref(), schema);
1060
1061                    if !hidden {
1062                        pending.push((
1063                            self.table_id.clone(),
1064                            self.partition_key.clone(),
1065                            row_value,
1066                            row_ts,
1067                        ));
1068                    }
1069                }
1070                Some(next_offset)
1071            }
1072            Err(_) => None,
1073        }
1074    }
