13impl V5CompressedLegacyParser {
14    /// Within-partition clustering-slice variant of [`parse_block_emit`] (Issue
15    /// #954, Epic #951).
16    ///
17    /// When `row_body_window` is `Some((body_start, body_end))` (byte offsets
18    /// into `data`, in the SAME domain as `data` indices) the FIRST partition's
19    /// row-body parse is bounded to that window:
20    ///   - after the partition header is decoded, the row cursor is fast-forwarded
21    ///     to `max(after_header, body_start)` (skipping rows that precede the
22    ///     requested clustering slice), and
23    ///   - the row loop stops once the cursor reaches `body_end` (skipping rows
24    ///     after the slice).
25    ///
26    /// `body_start`/`body_end` are the byte extent of the row-index block(s) that
27    /// the authoritative BTI row index resolved as covering the requested
28    /// clustering range, so this decodes O(matched rows + index-block slack)
29    /// rather than the whole partition. The post-scan `evaluate_leaf` backstop
30    /// trims the block-granularity over-read, so the returned rows are a SUPERSET
31    /// of the exact slice and the final query output is byte-identical.
32    ///
33    /// The start fast-forward is only applied when the schema has NO static
34    /// columns (the caller enforces this): a static row precedes the clustering
35    /// rows and must be merged into each clustering row, so skipping past it would
36    /// drop static values. The end bound is always safe to apply.
37    ///
38    /// Every row this method actually decodes bumps the `work_counters`
39    /// `rows_decoded` counter so a test can prove the decode was bounded to the
40    /// slice. With `row_body_window == None` this is byte-for-byte
41    /// [`parse_block_emit`]'s original behaviour, and the counter is still bumped
42    /// (the seek path reports its full-partition decode too).
43    pub fn parse_block_emit_windowed<F>(
44        &self,
45        data: &[u8],
46        extent: BufferExtent,
47        schema: Option<&TableSchema>,
48        reader: &crate::storage::sstable::reader::types::SSTableReader,
49        row_body_window: Option<(usize, usize)>,
50        mut emit: F,
51    ) -> Result<()>
52    where
53        F: FnMut((TableId, RowKey, ScanRow)) -> Result<std::ops::ControlFlow<()>>,
54    {
55        if data.is_empty() {
56            return Ok(());
57        }
58
59        // V5CompressedLegacy format stores cells WITHOUT column names,
60        // relying on schema to interpret the binary data. Schema is REQUIRED.
61        let schema = schema.ok_or_else(|| {
62            Error::schema(format!(
63                "V5CompressedLegacy format requires schema for {}.{} (cells lack column names in binary data)",
64                self.keyspace, self.table_name
65            ))
66        })?;
67
68        // Issue #1046: build the header→schema column resolution ONCE per block.
69        let resolution = RowColumnResolution::build(schema, reader);
70
71        tracing::debug!(
72            "V5CompressedLegacy: Parsing block for {}.{} ({} bytes)",
73            self.keyspace,
74            self.table_name,
75            data.len()
76        );
77        tracing::debug!(
78            "V5CompressedLegacy: Schema has {} columns",
79            schema.columns.len()
80        );
81        for (i, col) in schema.columns.iter().enumerate() {
82            tracing::debug!("  Column {}: {} ({})", i, col.name, col.data_type);
83        }
84        tracing::debug!(
85            "V5CompressedLegacy: First 64 bytes of data: {}",
86            hex::encode(&data[..std::cmp::min(64, data.len())])
87        );
88        debug!(
89            "V5CompressedLegacy: Parsing block for {}.{} ({} bytes)",
90            self.keyspace,
91            self.table_name,
92            data.len()
93        );
94
95        let mut emitted: usize = 0;
96        let mut offset = 0;
97        // Issue #1741 (F2): read-time TTL clock — captured ONCE per parser (== once per
98        // read/scan operation) in `V5CompressedLegacyParser::new`. The windowed scan
99        // driver reuses this parser across every block; sampling `self.now_secs` here
100        // (not the wall clock per block) keeps a boundary-crossing scan consistent.
101        let now_secs = self.now_secs;
102        let table_id = TableId::new(format!("{}.{}", self.keyspace, self.table_name));
103
104        // Parse ALL partitions in block (Issue #2 fix: previously only parsed one partition)
105        let mut partition_index = 0;
106        let mut skipped_partitions = 0;
107        // Issue #3928 (finding C2): the ONE header-tolerance state for this
108        // walk. It starts ATTRIBUTABLE — including on a call that supplied a
109        // `row_body_window`, because a window is not yet an uncertainty; it
110        // becomes one only when its endpoint is REACHED, which is where
111        // `bounded_out()` is called. Before that, partition 0's own header is as
112        // attributable as on an unbounded walk.
113        let tolerance = HeaderTolerance::for_extent(extent);
114        // Issue #3928 (finding B1): LABELLED so the row-body bound can terminate
115        // the WHOLE walk, not just the row loop — see the bound check below.
116        'partitions: while offset < data.len() {
117            // Cooperative cancellation (issue #2264): an uncompressed, index-less
118            // SSTable is returned to the scan as ONE contiguous block, so this loop
119            // is the 400k+-partition hot loop that the compaction streaming read
120            // (and thus a Flight `do_get`) spends its whole time in. Poll the
121            // reader's cancel token at a bounded interval so a disconnected client
122            // abandons the walk within milliseconds instead of running to
123            // completion under the coarse ~1–2 min backstop. Every 256 partitions
124            // keeps the relaxed-atomic load negligible against the per-partition
125            // parse cost.
126            if partition_index & 0xFF == 0 {
127                reader.scan_cancel.check()?;
128            }
129            tracing::debug!(
130                "V5CompressedLegacy: === PARTITION {} at offset {} (block size: {}) ===",
131                partition_index,
132                offset,
133                data.len()
134            );
135
136            // Issue #3928: the partition-HEADER arm — bounds, the issue-#164
137            // structural validation, and the real parse — lives in ONE place
138            // (`partition_header_arm.rs`), because on a PROVEN-COMPLETE buffer it
139            // must REFUSE rather than skip a byte and resynchronise: the resync
140            // both drops the partition and can invent one out of misaligned
141            // bytes. `?` propagates that refusal; `Resync`/`EndOfBlock` are only
142            // reachable on a `BufferExtent::Window`.
143            match self.block_partition_header(data, offset, tolerance, partition_index)? {
144                HeaderStep::Parsed(partition_key, new_offset, partition_deletion) => {
145                    let header_size = new_offset - offset;
146                    offset = new_offset;
147
148                    // Issue #1741: per-partition read-side shadowing, active ONLY for
149                    // user-facing query reads (`self.read_shadowing`). Captures the
150                    // partition-level deletion and tracks the open range tombstone as
151                    // rows are walked, so partition/range-tombstone-shadowed and
152                    // TTL-expired rows are hidden (matching Cassandra SELECT). `None`
153                    // for physical consumers (verify / get_all_entries) which must see
154                    // every on-disk row. Un-gated (not behind write-support).
155                    let mut shadow = self.read_shadowing.then(|| {
156                        PartitionShadow::open(
157                            now_secs,
158                            partition_deletion,
159                            clustering_reversed_flags(schema),
160                        )
161                    });
162
163                    // Issue #954: when a within-partition clustering-slice window is
164                    // supplied, fast-forward the row cursor of the FIRST partition to
165                    // the first row-index block covering the slice (skipping rows
166                    // before it). Applied only to `partition_index == 0` because the
167                    // seek decodes exactly one target partition. The end bound
168                    // (`body_end`) stops the row loop below. The caller guarantees a
169                    // start fast-forward is only requested when the schema has no
170                    // static columns, so this never skips a static prefix.
171                    let row_body_end = match row_body_window {
172                        Some((body_start, body_end)) if partition_index == 0 => {
173                            if body_start > offset && body_start <= data.len() {
174                                // Issue #1741 (Finding 1): a range tombstone that
175                                // OPENS before `body_start` can cover rows inside the
176                                // requested clustering slice. Fast-forwarding straight
177                                // to `body_start` would skip those markers, so the open
178                                // range would never be fed into `PartitionShadow` and a
179                                // slice read could return rows a full scan correctly
180                                // hides. When shadowing is active, replay the markers
181                                // from the partition body start up to `body_start` into
182                                // the shadow FSM first. If priming cannot faithfully
183                                // reconstruct the state (unrepresentable marker or an
184                                // undecodable row), fall back to a full-partition decode
185                                // from the body start rather than skip the markers: the
186                                // post-scan clustering backstop still trims the rows
187                                // before the slice, so correctness holds and only the
188                                // fast path is lost. When shadowing is off (physical
189                                // read) there is nothing to prime — keep the byte-for-
190                                // byte fast-forward (no-RT fast path unaffected).
191                                //
192                                // Issue #1741 (Finding 2): priming decodes the pre-window
193                                // prefix, regressing the row-index fast-forward from
194                                // O(slice) to O(prefix+slice) on a wide partition. Pay it
195                                // ONLY when a range tombstone can actually reach the slice.
196                                // Partition-deletion shadowing needs NO prefix decode (it is
197                                // captured from the header into `shadow`). Range tombstones:
198                                // when the SSTable's authoritative EncodingStats prove there
199                                // are no deletions (hence no range tombstones), keep the
200                                // O(slice) `offset = body_start` fast-forward and skip
201                                // priming. Priming itself is now marker-ONLY — it skips row
202                                // bodies via framing (no cell decode), feeding only the RT
203                                // markers into the FSM — so even when it runs it never
204                                // re-decodes the prefix cells.
205                                let primed = match shadow.as_mut() {
206                                    Some(_) if !self.sstable_may_have_range_tombstones() => true,
207                                    Some(sh) => self.prime_shadow_before_window(
208                                        data, offset, body_start, schema, sh,
209                                    ),
210                                    None => true,
211                                };
212                                if primed {
213                                    offset = body_start;
214                                }
215                            }
216                            Some(body_end)
217                        }
218                        _ => None,
219                    };
220
221                    tracing::debug!(
222                        "V5CompressedLegacy: Partition {} - Parsed partition key: {} bytes (header consumed {} bytes, now at offset {})",
223                        partition_index,
224                        partition_key.0.len(),
225                        header_size,
226                        offset
227                    );
228                    tracing::debug!(
229                        "V5CompressedLegacy: Partition {} - Row data starts at offset {}, remaining: {} bytes",
230                        partition_index,
231                        offset,
232                        data.len() - offset
233                    );
234                    tracing::debug!(
235                        "V5CompressedLegacy: Partition {} - Row data hex (first 128 bytes): {}",
236                        partition_index,
237                        hex::encode(&data[offset..std::cmp::min(offset + 128, data.len())])
238                    );
239
240                    debug!(
241                        "V5CompressedLegacy: Parsed partition key: {} bytes, now at offset {}",
242                        partition_key.0.len(),
243                        offset
244                    );
245
246                    // Parse ALL rows in this partition (Issue #166 fix: multi-row partition support)
247                    //
248                    // V5CompressedLegacy partitions can contain multiple rows with different clustering keys.
249                    // We use structural parsing (peek_is_partition_header) to detect partition boundaries,
250                    // not flag value heuristics (Issue #258, #28 no-heuristics mandate).
251                    // We parse rows in a loop until we encounter:
252                    // - End of block (offset >= data.len())
253                    // - END_OF_PARTITION marker (flags == 0x01, Issue #229 fix)
254                    // - Next partition header (detected via peek_is_partition_header)
255                    // - Parse error (invalid row data)
256
257                    // Issue #480 FIX: Static cell handling
258                    //
259                    // Cassandra static rows are stored once per partition (before clustering rows).
260                    // They should NOT be emitted as separate result entries — instead their column
261                    // values must be merged into each clustering row that follows in the partition.
262                    //
263                    // We accumulate static cells here and inject them into every clustering row.
264                    // Issue #1642 (K3): positional `RowCells`, matching the decoder emit.
265                    let mut static_cells: RowCells = Vec::new();
266                    let mut row_count = 0;
267                    // Issue #3095: Cassandra's `partition.hasNext()` — clustering
268                    // rows only (the static row is delivered out of band by
269                    // `partition.staticRow()`), plus proof that this call really
270                    // saw the partition's END. A partition body that is only
271                    // PARTIALLY present in `data` must never be mistaken for an
272                    // empty one, so the static-only row below is emitted solely
273                    // when the walk reached `END_OF_PARTITION` or the next
274                    // partition header.
275                    let mut emitted_clustering_row = false;
276                    let mut partition_complete = false;
277                    loop {
278                        // Issue #954: stop at the clustering-slice end bound. The
279                        // row-index block extent (`body_end`) is the authoritative
280                        // upper byte bound of the rows that may fall in the requested
281                        // clustering range; rows past it are outside the slice, so we
282                        // stop decoding (the post-scan backstop already trims any
283                        // block-granularity over-read within the window).
284                        if let Some(body_end) = row_body_end {
285                            if offset >= body_end {
286                                // Issue #3928 (finding B1): TERMINATE THE WHOLE
287                                // WALK, not just this row loop.
288                                //
289                                // The bound is the end of what the caller ASKED
290                                // to read (the row-index block extent the
291                                // authoritative BTI/promoted index resolved for
292                                // the requested clustering range). Reaching it
293                                // used to break only the inner loop, so the outer
294                                // partition loop then walked the unrequested
295                                // remainder byte-by-byte as candidate headers.
296                                // That both read partitions nobody asked for and
297                                // could FABRICATE one from misaligned row payload
298                                // — AC2's own subject — and on a wide partition it
299                                // could rescan the same remainder per call.
300                                //
301                                // Measured on the PRISTINE BIG fixture with a
302                                // window ending at offset 62: the walk emitted 99
303                                // FOREIGN partition keys, i.e. the whole table,
304                                // for a request scoped to ~32 bytes. Pinned by
305                                // `a_bounded_walk_stops_at_its_bound_and_reads_no_further_partition`.
306                                //
307                                // Nothing after this point in the arm is owed to
308                                // the caller: the #3095 static-only emission
309                                // below is already gated on
310                                // `row_body_end.is_none()`, and a bounded read's
311                                // rows are trimmed by the post-scan clustering
312                                // backstop either way.
313                                break 'partitions;
314                            }
315                        }
316
317                        // Issue #229 FIX: Check for END_OF_PARTITION marker BEFORE attempting row parse
318                        //
319                        // Per Cassandra's UnfilteredSerializer.java (lines 102, 730-732):
320                        // When END_OF_PARTITION (0x01) is set in the flags byte, nothing follows.
321                        // The partition is complete and we should move to the next partition.
322                        if offset < data.len() && Self::is_end_of_partition(data[offset]) {
323                            tracing::debug!(
324                                "V5CompressedLegacy: Partition {} complete via END_OF_PARTITION marker at offset {} ({} rows parsed)",
325                                partition_index, offset, row_count
326                            );
327                            offset += 1; // Skip the END_OF_PARTITION marker byte
328                            partition_complete = true;
329                            break; // Move to next partition
330                        }
331
332                        // Issue #229 FIX: Check for range tombstone marker
333                        //
334                        // Per Cassandra's UnfilteredSerializer.java (lines 103, 735-738):
335                        // When IS_MARKER (0x02) is set, this is a range tombstone boundary, not a row.
336                        if offset < data.len() && Self::is_range_tombstone_marker(data[offset]) {
337                            // Issue #1741: when read-side shadowing is active, decode
338                            // the marker (bounds + deletion time) and feed the
339                            // range-tombstone FSM so covered rows are shadowed;
340                            // otherwise (physical read) only advance past it.
341                            if let Some(sh) = shadow.as_mut() {
342                                match self.parse_range_tombstone_marker_full(data, offset, schema) {
343                                    Ok((
344                                        bound_values,
345                                        bound_kind,
346                                        del_primary,
347                                        del_secondary,
348                                        next_offset,
349                                    )) => {
350                                        if let Err(e) = sh.feed_range_marker(
351                                            bound_values,
352                                            bound_kind,
353                                            del_primary,
354                                            del_secondary,
355                                        ) {
356                                            // Issue #3721: FRAMED marker (body
357                                            // continues at `next_offset`) with an
358                                            // unrepresentable bound kind; `break`
359                                            // dropped every later row and said `Ok`.
360                                            // #3928 (I2) a fortiori: a `return`
361                                            // reaches no header arm — see that arm,
362                                            // "Reconciling #3721 and #3928".
363                                            return Err(range_marker_error::range_marker_refused(
364                                                e,
365                                                &partition_index,
366                                                offset,
367                                                next_offset,
368                                            ));
369                                        }
370                                        offset = next_offset;
371                                        continue; // Continue to next row/marker
372                                    }
373                                    // Issue #3721 (job 78): corruption, never a
374                                    // terminator (`range_marker_error` docs). The
375                                    // cause used to be dropped into `debug!`.
376                                    //
377                                    // Issue #3782 x #3721 (job 98): DELIBERATELY UNCONDITIONAL,
378                                    // and the alternative was MEASURED to be worse.
379                                    //
380                                    // roborev job 98 is right that an index-NARROWED window can
381                                    // legitimately end inside a valid marker, and that this
382                                    // refusal is `Error::Corruption`, which
383                                    // `indexed_walk_falls_back` does not recognise (it matches
384                                    // `is_column_decode` only) — so such a read fails instead of
385                                    // retracting. That is a real, conservative false refusal.
386                                    //
387                                    // But gating this on `extent.is_complete()` REINTRODUCED
388                                    // #3721's defect on the SELECT path: the scan passes `Window`
389                                    // even with the whole partition buffered, so all three
390                                    // `d9_select_marker_parse_failure` cases went back to
391                                    // returning `Ok` with the marker and every later row silently
392                                    // dropped. A false refusal on a narrowed read is recoverable;
393                                    // a silent short answer from `SELECT` is the defect itself.
394                                    //
395                                    // The correct fix is a RETRACTION SIGNAL the indexed walk can
396                                    // recognise — not tolerance, which would merely make the
397                                    // narrowed read silently short instead. That needs a TYPED
398                                    // discriminator (matching the message text would violate the
399                                    // no-heuristics rule, issue #28), so it is left as a named
400                                    // residual rather than invented here.
401                                    //
402                                    // Issue #3928 (I2) a fortiori: a `return` ends the walk at
403                                    // every extent, so the header arm is never re-entered with
404                                    // the cursor ON the marker. That reconciliation, and the
405                                    // SECOND residual it leaves, are recorded in
406                                    // `partition_header_arm.rs`, "Reconciling #3721 and #3928".
407                                    Err(cause) => {
408                                        return Err(unparseable(cause, &partition_index, offset))
409                                    }
410                                }
411                            }
412                            match self.skip_range_tombstone_marker(data, offset, schema) {
413                                Ok(next_offset) => {
414                                    offset = next_offset;
415                                    continue;
416                                }
417                                // Same decision, and the same two residuals, on the
418                                // PHYSICAL path (no shadowing); #3928's I2 invariant
419                                // holds here for the same reason (a `return` ends the
420                                // walk at every extent).
421                                Err(cause) => {
422                                    return Err(unparseable(cause, &partition_index, offset))
423                                }
424                            }
425                        }
426
427                        match self.parse_row_data_with_offset(
428                            data,
429                            offset,
430                            Some(schema),
431                            reader,
432                            false,
433                            &resolution,
434                            shadow.as_ref(),
435                        ) {
436                            Ok((
437                                mut cells,
438                                _row_cell_meta,
439                                row_header_opt,
440                                next_offset,
441                                is_static,
442                                _complex_meta,
443                            )) => {
444                                // Update offset to point to the next row or partition
445                                offset = next_offset;
446                                row_count += 1;
447
448                                tracing::debug!(
449                                    "V5CompressedLegacy: Partition {} Row {} - Parsed {} cells, now at offset {} (is_static={})",
450                                    partition_index,
451                                    row_count,
452                                    cells.len(),
453                                    offset,
454                                    is_static
455                                );
456
457                                if let Some(ref header) = row_header_opt {
458                                    tracing::debug!(
459                                        "V5CompressedLegacy: Row {} metadata - timestamp={:?}, ttl={:?}, deletion={:?}",
460                                        row_count,
461                                        header.timestamp, header.ttl, header.local_deletion_time
462                                    );
463                                }
464
465                                debug!(
466                                    "V5CompressedLegacy: Parsed {} cells from row {} (is_static={})",
467                                    cells.len(),
468                                    row_count,
469                                    is_static
470                                );
471
472                                // Issue #480 FIX: Static row handling
473                                //
474                                // Static rows are stored once per partition and contain values for
475                                // STATIC columns (e.g. `static_data TEXT STATIC`). They must NOT
476                                // be emitted as standalone result rows. Instead, store the static
477                                // column values and merge them into each subsequent clustering row.
478                                if is_static {
479                                    tracing::debug!(
480                                        "V5CompressedLegacy: Partition {} - Storing {} static cells for merging into clustering rows",
481                                        partition_index,
482                                        cells.len()
483                                    );
484                                    // Issue #1741 (Finding 1): a static row can itself be
485                                    // shadowed by the partition tombstone (static write ts
486                                    // <= markedForDeleteAt) or expired by its own TTL. If
487                                    // so its cells are STALE and must NOT be merged into a
488                                    // surviving clustering row, or a SELECT would resurface
489                                    // the deleted/expired static value. Static rows carry no
490                                    // clustering key, so an open range tombstone never covers
491                                    // them (empty clustering). No-op when shadowing is off.
492                                    let static_hidden = shadow.as_ref().is_some_and(|sh| {
493                                        row_header_opt
494                                            .as_ref()
495                                            .is_some_and(|h| sh.row_hidden(h, &[]))
496                                    });
497                                    static_cells = if static_hidden { Vec::new() } else { cells };
498                                    // Do NOT push to results — static rows are not result rows
499                                    // Continue to next row/marker in partition
500                                } else {
501                                    // Issue #1741: hide rows shadowed by a partition or
502                                    // range tombstone, or expired by TTL, matching a
503                                    // Cassandra SELECT. Active only for query reads
504                                    // (shadow is `Some`). A row reduced to only its
505                                    // primary key by per-cell filtering is hidden here
506                                    // too: the dropped cells still fold into the row
507                                    // aggregate, so `row_hidden` sees it shadowed/expired.
508                                    let hidden = shadow.as_ref().is_some_and(|sh| {
509                                        row_header_opt.as_ref().is_some_and(|h| {
510                                            let clustering = if sh.needs_clustering() {
511                                                extract_clustering_values(&cells, schema)
512                                            } else {
513                                                Vec::new()
514                                            };
515                                            sh.row_hidden(h, &clustering)
516                                        })
517                                    });
518
519                                    // Issue #505/#932: row-tombstone display rule lives
520                                    // in the shared `build_display_row` helper. Issue
521                                    // #480/#1642: static cells are merged in
522                                    // (positional, clustering-row-wins).
523                                    //
524                                    // Issue #3095: on a user-facing SELECT read the
525                                    // tombstone decision is taken over the row's OWN
526                                    // cells FIRST, so a static value cannot revive a
527                                    // row-tombstoned row (see
528                                    // `build_display_row_read_path`). Physical consumers
529                                    // keep the historical inject-then-decide order, so
530                                    // their byte-pinned output is unchanged.
531                                    let row_value = if shadow.is_some() {
532                                        build_display_row_read_path(
533                                            cells,
534                                            &static_cells,
535                                            row_header_opt.as_ref(),
536                                            schema,
537                                        )
538                                    } else {
539                                        merge_static_cells(&mut cells, &static_cells);
540                                        build_display_row(cells, row_header_opt.as_ref(), schema)
541                                    };
542
543                                    // Issue #954: count each clustering row actually
544                                    // decoded out of Data.db so a slice query can be
545                                    // proven to decode O(matched rows + index block),
546                                    // not the whole partition. Counted at the row
547                                    // grain (here), distinct from the per-partition
548                                    // `partitions_decoded`. Gated `not(tombstones)`
549                                    // because the mutator is compiled only there.
550                                    #[cfg(not(feature = "tombstones"))]
551                                    crate::storage::sstable::work_counters::add_rows_decoded(1);
552
553                                    if !hidden {
554                                        // Issue #3095 (roborev + rust-reviewer
555                                        // BLOCKER): only a VISIBLE row counts as one
556                                        // of Cassandra's `partition.hasNext()` rows —
557                                        // a `ScanRow::Marker` (pure row tombstone) is
558                                        // suppressed downstream by every user-facing
559                                        // consumer, so counting it would hide the
560                                        // static row of a partition whose clustering
561                                        // rows are all deleted. See `row_is_visible`.
562                                        emitted_clustering_row |= row_is_visible(&row_value);
563                                        match emit((
564                                            table_id.clone(),
565                                            partition_key.clone(),
566                                            row_value,
567                                        ))? {
568                                            std::ops::ControlFlow::Continue(()) => emitted += 1,
569                                            // Consumer dropped (streaming receiver gone): stop parsing.
570                                            std::ops::ControlFlow::Break(()) => return Ok(()),
571                                        }
572                                    }
573                                }
574
575                                // Check if we're at the end of the partition
576                                if offset >= data.len() {
577                                    debug!(
578                                        "V5CompressedLegacy: Partition {} complete: {} rows parsed (end of block)",
579                                        partition_index, row_count
580                                    );
581                                    break; // End of block
582                                }
583
584                                // CRITICAL FIX (Issue #166): NO HEURISTICS - Try-parse approach
585                                //
586                                // Instead of guessing based on byte patterns (e.g., checking if flags <= 0x20
587                                // or validating key_len ranges), we ACTUALLY TRY TO PARSE the next structure.
588                                //
589                                // Why heuristics fail:
590                                // - Row with small value (e.g., boolean=0x0A) can look like key_len
591                                // - Row flags=0x00 or 0x20 pass "<= 0x20" checks meant for partitions
592                                // - Any byte-pattern guessing will eventually fail on edge cases
593                                //
594                                // The only reliable approach: try to parse as partition header.
595                                // If that succeeds, it's a partition. If it fails, continue with rows.
596                                if self.peek_is_partition_header(data, offset) {
597                                    debug!(
598                                        "V5CompressedLegacy: Partition {} complete: {} rows parsed (next partition detected at offset {})",
599                                        partition_index, row_count, offset
600                                    );
601                                    partition_complete = true;
602                                    break; // Next partition starts here
603                                }
604
605                                // Peek failed - not a partition header, so continue parsing rows
606                                debug!(
607                                    "V5CompressedLegacy: Partition {} - Continuing to row {} at offset {} (peek confirmed this is NOT a partition header)",
608                                    partition_index, row_count + 1, offset
609                                );
610                            }
611                            Err(e) => {
612                                // End of valid data in partition
613                                debug!(
614                                    "V5CompressedLegacy: Partition {} ended after {} rows: {}",
615                                    partition_index, row_count, e
616                                );
617                                if row_count == 0 {
618                                    // If we couldn't parse even one row, log as error
619                                    tracing::error!(
620                                        "V5CompressedLegacy: Partition {} - Failed to parse first row at offset {}: {}",
621                                        partition_index, offset, e
622                                    );
623                                }
624                                // Issue #3782: on a PROVEN-complete buffer no further
625                                // bytes can finish this row, so the failure is
626                                // truncation/corruption — DATA LOSS — and is reported;
627                                // swallowing it made a SELECT over a fixture with ONE
628                                // corrupted clustering byte return 23 of 100 rows.
629                                // Otherwise the tolerant break STAYS: a chunk-covering
630                                // window can legitimately cut a row at its tail. See
631                                // `BufferExtent`.
632                                // Issue #3721 composes INSIDE that gate: on a complete
633                                // buffer the failure is authoritative, but it is still
634                                // two different facts — a per-column decode failure
635                                // (propagate, naming the column) versus the ordinary
636                                // end-of-partition-body signal (break). Only
637                                // `column_decode_error` can tell them apart, so returning
638                                // `Err(e)` unconditionally here would report a normal
639                                // partition end as corruption.
640                                // #3782 FIRST and unconditionally: on a proven-complete
641                                // buffer ANY failure is data loss, whatever its kind.
642                                if extent.is_complete() {
643                                    return Err(e);
644                                }
645                                // #3721 for the incomplete case: a per-column decode
646                                // failure must STILL reach the caller, because the caller
647                                // owns the tolerance decision (the BTI point read remembers
648                                // it and retries with more bytes; the driver uses
649                                // `at_final_chunk`). Gating this on `is_complete()` put that
650                                // decision back INSIDE the parse — contradicting #3782's own
651                                // rule — and silently reintroduced job 80's phantom absence,
652                                // caught by `issue_3721_bti_point_read_absence`.
653                                column_decode_error::end_of_partition_or_bail(
654                                    e,
655                                    partition_index,
656                                    row_count,
657                                    offset,
658                                )?;
659                                // Issue #3928 round 5 (I2): END THE WALK. `Ok` from
660                                // `end_of_partition_or_bail` establishes only "not
661                                // `Error::ColumnDecode`" and consumed nothing, so the
662                                // cursor still sits where the failed row parse STARTED —
663                                // the only exit here leaving it UNCONFIRMED (both boundary
664                                // exits set `partition_complete`), reachable only when
665                                // `!extent.is_complete()`, i.e. MID-ROW under the straddle
666                                // protocol. How this composes with #3721 above, and the
667                                // measurement in BOTH directions:
668                                // `partition_header_arm.rs`, "Reconciling #3721 and #3928".
669                                break 'partitions; // End of valid data in partition
670                            }
671                        }
672                    }
673
674                    // Issue #3095: Cassandra's static-content-on-an-empty-partition
675                    // rule. `SelectStatement.processPartition()` (cassandra-5.0.8,
676                    // L1099-1120): with NO clustering rows and a non-empty
677                    // out-of-band `partition.staticRow()`, the query returns EXACTLY
678                    // ONE row — clustering + REGULAR columns null
679                    // (`default: result.add((ByteBuffer) null)`), STATIC columns
680                    // populated — and that branch `return`s, making it mutually
681                    // exclusive with the per-row loop.
682                    //
683                    // Gated on `read_shadowing` (user-facing SELECT reads only), so a
684                    // PHYSICAL consumer still sees exactly the on-disk unfiltereds and
685                    // sstabledump/compaction parity is unchanged; on
686                    // `partition_complete`, so a partially-present partition body is
687                    // never mistaken for an empty one; and on a clustering-slice
688                    // window being absent, since a slice read decodes only part of the
689                    // partition (a static-bearing schema never takes the row-index
690                    // fast-forward today, so this is a belt-and-braces invariant).
691                    //
692                    // RESIDUAL, stated rather than left implicit (issue #3095 review):
693                    // `partition_complete` is established PER BLOCK here — it is set
694                    // only by this block's `END_OF_PARTITION` byte or by the next
695                    // partition's header appearing in THIS block. So a static-only
696                    // partition whose `END_OF_PARTITION` byte lands in the NEXT
697                    // decompressed block yields 0 rows where Cassandra returns 1. The
698                    // direction is FAIL-CLOSED (a row withheld, never a phantom row or
699                    // a wrong value), and the shape is narrow: a static-only partition
700                    // is a partition header + one static row, so its body straddling a
701                    // block boundary requires the boundary to fall inside those few
702                    // bytes. Closing it needs an `at_final_block`-style signal threaded
703                    // from every caller (`data_access::big_promoted`,
704                    // `data_access::bti_point`, the Summary-guided walk), i.e. the
705                    // `at_final_chunk` contract `drive_partition_sliding` already has —
706                    // which is why the SLIDING path (the full-scan route, and the one
707                    // the Flight fast arm drives) does NOT have this residual.
708                    //
709                    // `static_cells` is already empty when the static row was shadowed
710                    // by the partition tombstone or expired by its own TTL (#1741
711                    // Finding 1), so a stale static row cannot resurface. The
712                    // clustering/regular-restriction half of
713                    // `returnStaticContentOnPartitionWithNoRows()` is enforced
714                    // downstream: this row's clustering AND regular columns are null,
715                    // so any restriction on one of them (the only way
716                    // `queriesFullPartitions()` becomes false) rejects it under the
717                    // three-valued predicate rule that keeps a row only when the
718                    // predicate is definitely True.
719                    if self.read_shadowing
720                        && partition_complete
721                        && row_body_end.is_none()
722                        && !emitted_clustering_row
723                        && !static_cells.is_empty()
724                    {
725                        // Non-empty cells holding a static (non-key) column, so the
726                        // shared display rule yields `ScanRow::Row`; no header is
727                        // needed (a shadowed/tombstoned static row already emptied
728                        // `static_cells`).
729                        let row_value = build_display_row(static_cells, None, schema);
730                        match emit((table_id.clone(), partition_key.clone(), row_value))? {
731                            std::ops::ControlFlow::Continue(()) => emitted += 1,
732                            std::ops::ControlFlow::Break(()) => return Ok(()),
733                        }
734                    }
735
736                    partition_index += 1;
737                    // Issue #3721: `column_decode_error`, "Where an
738                    // INDEX-POSITIONED walk must stop".
739                    if row_body_window.is_some() {
740                        break;
741                    }
742                }
743                HeaderStep::EndOfBlock => break,
744                HeaderStep::Resync => {
745                    skipped_partitions += 1;
746                    offset += 1; // Minimal forward progress to avoid an infinite loop
747                    continue; // Skip this partition, try the next offset
748                }
749            }
750        }
751
752        if skipped_partitions > 0 {
753            tracing::warn!(
754                "V5CompressedLegacy: Successfully parsed {} entries, skipped {} malformed partitions",
755                emitted,
756                skipped_partitions
757            );
758        }
759
760        debug!(
761            "V5CompressedLegacy: Parsed {} total entries from block",
762            emitted
763        );
764
765        Ok(())
766    }

768    /// Issue #1741 (Finding 1): replay the range-tombstone (and any other)
769    /// unfiltereds between the partition body start (`start`) and the
770    /// clustering-slice window start (`body_start`) into `shadow`, so an open
771    /// range tombstone that begins BEFORE the slice correctly shadows covered
772    /// rows inside it. Rows in this prefix are NOT emitted — only markers move
773    /// the FSM. `body_start` aligns to the start of an unfiltered (a BTI row-index
774    /// block boundary), so markers before it are fully contained in the prefix.
775    ///
776    /// Returns `true` when the prefix was replayed cleanly all the way to
777    /// `body_start` (the caller may then fast-forward). Returns `false` when the
778    /// state cannot be faithfully reconstructed (an unrepresentable marker, an
779    /// undecodable row framing, an early END_OF_PARTITION, or a non-advancing
780    /// parse); the caller then declines the fast-forward and decodes the full
781    /// partition, which is still correct because the post-scan backstop trims rows
782    /// before the slice.
783    ///
784    /// Issue #1741 (Finding 2): this is a MARKER-ONLY scan. Data rows in the prefix
785    /// are skipped via their framing (`skip_row_framing`) WITHOUT decoding any cell
786    /// values — only range-tombstone markers move the FSM. It therefore adds no
787    /// per-cell decode / allocation over the pre-window prefix; only the (cheap)
788    /// row framing (flags + clustering + row_size VInts) is parsed to advance. It
789    /// runs solely on the slice-read fast-forward branch when shadowing is active
790    /// AND the SSTable may contain range tombstones (see the call site gate).
791    fn prime_shadow_before_window(
792        &self,
793        data: &[u8],
794        start: usize,
795        body_start: usize,
796        schema: &TableSchema,
797        shadow: &mut PartitionShadow,
798    ) -> bool {
799        let mut offset = start;
800        while offset < body_start {
801            if offset >= data.len() {
802                return false;
803            }
804            let flags = data[offset];
805            // An END_OF_PARTITION before the window means the window is not in this
806            // partition — bail to the safe full-decode path.
807            if Self::is_end_of_partition(flags) {
808                return false;
809            }
810            if Self::is_range_tombstone_marker(flags) {
811                match self.parse_range_tombstone_marker_full(data, offset, schema) {
812                    Ok((bound_values, bound_kind, del_primary, del_secondary, next_offset)) => {
813                        if shadow
814                            .feed_range_marker(bound_values, bound_kind, del_primary, del_secondary)
815                            .is_err()
816                        {
817                            return false;
818                        }
819                        if next_offset <= offset {
820                            return false; // non-advancing parse guard
821                        }
822                        offset = next_offset;
823                        continue;
824                    }
825                    Err(_) => return false,
826                }
827            }
828            // A data row in the prefix: advance past it via framing ONLY (no cell
829            // decode), so the fast-forward stays O(prefix framing), not O(prefix
830            // cells). The prefix carries no static row (the caller only requests a
831            // start fast-forward when the schema has no static columns).
832            match self.skip_row_framing(data, offset, schema) {
833                Ok(next_offset) => {
834                    if next_offset <= offset {
835                        return false; // non-advancing parse guard
836                    }
837                    offset = next_offset;
838                }
839                Err(_) => return false,
840            }
841        }
842        true
843    }

845    /// Issue #1741 (Finding 2): compute the offset immediately after the row at
846    /// `offset` by parsing ONLY its framing — flags, clustering prefix, and the
847    /// `row_size` VInt — never decoding cell values. Mirrors the authoritative
848    /// offset arithmetic in `parse_row_data_with_offset`
849    /// (`next = row_metadata_offset + row_size_vint_len + row_size`) so a marker-only
850    /// prefix scan can skip data rows cheaply. Returns an error on truncation or an
851    /// out-of-range `row_size`, which the caller treats as "cannot prime".
852    pub(super) fn skip_row_framing(
853        &self,
854        data: &[u8],
855        offset: usize,
856        schema: &TableSchema,
857    ) -> Result<usize> {
858        let (row_flags, extended_flags, flags_size) = self.parse_row_flags(data, offset)?;
859        let after_flags = offset + flags_size;
860        let is_static = extended_flags
861            .map(|ef| (ef & EXTENDED_IS_STATIC) != 0)
862            .unwrap_or(false);
863        // Static rows carry no clustering prefix; regular rows do.
864        let row_metadata_offset = if is_static {
865            after_flags
866        } else {
867            let (_clustering, after_clustering) =
868                self.parse_clustering_prefix(data, after_flags, schema)?;
869            after_clustering
870        };
871        let (row_header, row_size) =
872            self.parse_row_metadata(data, row_metadata_offset, row_flags, extended_flags)?;
873        let body_start = row_metadata_offset + row_header.row_size_vint_len;
874        if row_size > data.len().saturating_sub(body_start) as u64 {
875            return Err(Error::corruption(
876                "prime_shadow: row framing extends past decompressed block".to_string(),
877            ));
878        }
879        Ok(body_start + row_size as usize)
880    }

882    /// Parse all partitions in a decompressed block, returning per-row timestamps.
883    ///
884    /// This is the compaction-specific variant of [`parse_block`].  It returns
885    /// `(TableId, RowKey, ScanRow, row_timestamp_micros)` so that the k-way merger
886    /// can perform timestamp-accurate last-write-wins ordering rather than
887    /// falling back to `SystemTime::now()`.
888    ///
889    /// Row tombstones are emitted as `Value::Tombstone(RowTombstone)` with their
890    /// actual `deletion_time`.  Cell tombstones within live rows are stored as
891    /// `Value::Tombstone(CellTombstone)` inside the `Value::Map`, again carrying
892    /// the cell-level deletion timestamp.
893    ///
894    /// The `row_timestamp_micros` in the returned tuple is the row-level write
895    /// timestamp decoded from the `HAS_TIMESTAMP` field in the row header
896    /// (`min_timestamp + delta`).  For row tombstones the same timestamp also
897    /// appears in `TombstoneInfo::deletion_time`.
898    ///
899    /// Normal user-facing scan/get paths should use [`parse_block`] instead.
900    /// (Issue #505)
901    pub fn parse_block_with_timestamps(
902        &self,
903        data: &[u8],
904        schema: Option<&TableSchema>,
905        reader: &crate::storage::sstable::reader::types::SSTableReader,
906    ) -> Result<Vec<(TableId, RowKey, ScanRow, i64)>> {
907        // Thin wrapper that collects the streaming emit variant into a Vec, so
908        // every existing caller/test is byte-for-byte unchanged (issue #827).
909        let mut results: Vec<(TableId, RowKey, ScanRow, i64)> = Vec::new();
910        self.parse_block_with_timestamps_emit(data, schema, reader, |entry| {
911            results.push(entry);
912            Ok(std::ops::ControlFlow::Continue(()))
913        })?;
914        Ok(results)
915    }

917    /// Streaming variant of [`parse_block_with_timestamps`]: invokes `emit` for
918    /// each parsed `(TableId, RowKey, ScanRow, row_timestamp_micros)` entry rather
919    /// than collecting into a `Vec`, so the compaction read path can forward
920    /// rows into a bounded channel without materialising the whole block at once
921    /// (issue #827). Returning `ControlFlow::Break` from `emit` stops parsing
922    /// early — used when the streaming consumer is dropped.
923    ///
924    /// The tombstone/timestamp semantics are byte-identical to
925    /// [`parse_block_with_timestamps`] (Issue #505/#533): a row tombstone is
926    /// emitted as `Value::Tombstone` carrying its `markedForDeleteAt`, and the
927    /// fourth tuple element is the row write timestamp for live rows.
928    pub fn parse_block_with_timestamps_emit<F>(
929        &self,
930        data: &[u8],
931        schema: Option<&TableSchema>,
932        reader: &crate::storage::sstable::reader::types::SSTableReader,
933        mut emit: F,
934    ) -> Result<()>
935    where
936        F: FnMut((TableId, RowKey, ScanRow, i64)) -> Result<std::ops::ControlFlow<()>>,
937    {
938        if data.is_empty() {
939            return Ok(());
940        }
941
942        let schema = schema.ok_or_else(|| {
943            Error::schema(format!(
944                "V5CompressedLegacy format requires schema for {}.{}",
945                self.keyspace, self.table_name
946            ))
947        })?;
948
949        let mut offset = 0;
950        let mut skipped_partitions = 0;
951
952        // Wrap `emit` so a Break is observable here as well as inside the
953        // one-partition parser (which stops its inner row loop on Break). This
954        // lets the outer loop terminate promptly when the consumer is dropped.
955        // `Cell` so the wrapping closure can borrow it shared while the outer
956        // loop also reads it between calls.
957        let broke = std::cell::Cell::new(false);
958        let mut tracking_emit = |entry| -> Result<std::ops::ControlFlow<()>> {
959            let flow = emit(entry)?;
960            if matches!(flow, std::ops::ControlFlow::Break(())) {
961                broke.set(true);
962            }
963            Ok(flow)
964        };
965
966        while offset < data.len() {
967            match self.parse_one_partition_with_timestamps(
968                &data[offset..],
969                Some(schema),
970                reader,
971                // The whole block is present; never request a refill. A trailing
972                // parse failure is terminal here (matches the legacy
973                // `Err(_) => break` behaviour of the original loop).
974                //
975                // Compaction NEVER shadows: this parser is built without
976                // `read_shadowing`, so the merger sees the raw tombstones/expired
977                // cells it needs to reconcile across generations (#1741).
978                true,
979                &mut tracking_emit,
980            )? {
981                ParseStep::Emitted(consumed) => {
982                    if consumed == 0 {
983                        // Defensive: avoid an infinite loop on a zero-byte
984                        // partition (should not happen — a header is >= 2 bytes).
985                        skipped_partitions += 1;
986                        offset += 1;
987                    } else {
988                        offset += consumed;
989                    }
990                }
991                // `at_final_chunk = true` collapses NeedMore into Done: there is
992                // no further chunk to append, so a truncated tail is end-of-data.
993                ParseStep::NeedMore | ParseStep::Done => break,
994            }
995            // Propagate an early Break from `emit` (consumer dropped).
996            if broke.get() {
997                break;
998            }
999        }
1000
1001        if skipped_partitions > 0 {
1002            tracing::warn!(
1003                "V5CompressedLegacy: skipped {} malformed partitions",
1004                skipped_partitions
1005            );
1006        }
1007
1008        Ok(())
1009    }

1011    /// Parse exactly ONE partition from the front of `data`, emitting each row
1012    /// via `emit`, and report how the parse terminated (issue #827).
1013    ///
1014    /// This isolates the body of the outer partition loop so the sliding-window
1015    /// compaction driver can drain one partition at a time and `drain(0..consumed)`
1016    /// from its window between calls. The crucial distinction over the legacy
1017    /// monolithic loop is `NeedMore` vs `Done`:
1018    ///
1019    /// - [`ParseStep::Emitted(consumed)`] — a full partition was parsed and
1020    ///   terminated by an END_OF_PARTITION marker or a confirmed next-partition
1021    ///   header. `consumed` bytes may be drained from the window.
1022    /// - [`ParseStep::NeedMore`] — `data` is (possibly) truncated mid-partition
1023    ///   and `!at_final_chunk`, so the caller must append the next chunk and
1024    ///   retry. NEVER returned when `at_final_chunk` is true.
1025    /// - [`ParseStep::Done`] — genuine end of partitions, or (when
1026    ///   `at_final_chunk`) a trailing truncation that cannot be resolved by more
1027    ///   data. Terminal.
1028    ///
1029    /// `at_final_chunk` flips a mid-partition parse failure between a refill
1030    /// request (`NeedMore`) and a terminal stop. The legacy code conflated
1031    /// parse-error with end-of-partitions (`Err(_) => break`); doing that
1032    /// mid-stream would silently drop every partition after a chunk boundary, so
1033    /// we return `NeedMore` whenever the buffer may simply be truncated and we
1034    /// are not yet at the final chunk.
1035    /// Read-side shadowing (issue #1741) is driven by `self.read_shadowing`: when the
1036    /// parser was built for a user-facing query read, rows shadowed by a partition/
1037    /// range tombstone or expired by TTL are dropped before they reach `emit`. The
1038    /// compaction read path builds the parser WITHOUT shadowing (`read_shadowing ==
1039    /// false`) — it needs the raw tombstones/expired cells preserved to reconcile
1040    /// across generations. Un-gated: read correctness does not depend on `write-support`.
1041    pub fn parse_one_partition_with_timestamps<F>(
1042        &self,
1043        data: &[u8],
1044        schema: Option<&TableSchema>,
1045        reader: &crate::storage::sstable::reader::types::SSTableReader,
1046        at_final_chunk: bool,
1047        emit: &mut F,
1048    ) -> Result<ParseStep>
1049    where
1050        F: FnMut((TableId, RowKey, ScanRow, i64)) -> Result<std::ops::ControlFlow<()>>,
1051    {
1052        if data.is_empty() {
1053            return Ok(ParseStep::Done);
1054        }
1055
1056        let schema = schema.ok_or_else(|| {
1057            Error::schema(format!(
1058                "V5CompressedLegacy format requires schema for {}.{}",
1059                self.keyspace, self.table_name
1060            ))
1061        })?;
1062
1063        // Issue #1640 (K1): thin adapter over the single sliding-window partition
1064        // driver. `TimestampPolicy` supplies the three per-consumer hooks
1065        // (read-side shadow open, range-tombstone FSM feed, static-merge +
1066        // display-row build); the driver owns the framing skeleton + ParseStep +
1067        // pending buffering this function used to hand-roll.
1068        let mut policy = TimestampPolicy::new(self);
1069        self.drive_partition_sliding(data, schema, reader, at_final_chunk, &mut policy, |row| {
1070            emit(row)
1071        })
1072    }
