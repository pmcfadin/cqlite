1//! Parser for Cassandra 5.0 V5CompressedLegacy decompressed blocks
2//!
3//! This format uses **u8 length prefixes** (NOT VInt) for partition keys and strings,
4//! and simplified encoding optimized for compression. This differs from the newer
5//! V5_0NewBig and V5_0Bti formats which use pure VInt encoding.
6//!
7//! ## Partition Key Size Constraints
8//!
9//! **Apache Cassandra Specification**: Partition keys can be up to 64KB (65536 bytes).
10//! **V5CompressedLegacy Format Limitation**: Uses u8 for key length field, limiting keys to 255 bytes max.
11//!
12//! This means V5CompressedLegacy format cannot represent partition keys larger than 255 bytes,
13//! even though Cassandra allows keys up to 64KB. Tables with larger keys would use a different format.
14//!
15//! Based on format research in docs/V5_COMPRESSED_LEGACY_FORMAT_SPEC.md
16//!
17//! ## Format Structure
18//!
19//! ```text
20//! Decompressed Block:
21//! ├─ [0x00] Partition flags (u8)
22//! ├─ [0x01] Partition key length (u8, NOT VInt)
23//! ├─ [0x02..] Raw partition key bytes
24//! ├─ [+0] Partition deletion time (i32 big-endian)
25//! ├─ [+4] Unknown 8-byte field
26//! ├─ [+8] Row data begins
27//! │  ├─ Row header (flags, timestamp, row_size)
28//! │  ├─ Cells:
29//! │  │  ├─ Type tag or flags (u8)
30//! │  │  ├─ Column name length (u8)
31//! │  │  ├─ Column name bytes
32//! │  │  ├─ Value length (varies)
33//! │  │  └─ Value bytes
34//! │  └─ Trailing 4-byte field (NOT included in row_size)
35//! └─ [Next partition or end of block]
36//! ```

100/// One on-disk column to decode, in serialization-header order.
101///
102/// `schema` is the resolved supplied-schema `Column` (drives column identity /
103/// emit name) or `None` for a DROPPED / evolved-away column present on disk but
104/// absent from the supplied schema — its bytes are still consumed to keep the
105/// trailing columns byte-aligned (issue #1080 Part 2), but no cell is emitted.
106///
107/// `header_type` is the AUTHORITATIVE on-disk SerializationHeader marshal type
108/// (`ColumnInfo.column_type`), the no-heuristics source of truth (issue #28) used
109/// to decide complex-ness and to decode complex values. It is `None` only on the
110/// header-empty fallback path (synthetic SSTables) where the supplied schema type
111/// is all we have.
112///
113/// Both fields borrow: `schema` from the per-call `TableSchema`, `header_type`
114/// from `reader.header`. The owning [`RowColumnResolution`] therefore borrows
115/// from both for `'a`.
116pub(super) struct ColumnToParse<'a> {
117    pub(super) schema: Option<&'a crate::schema::Column>,
118    pub(super) header_type: Option<&'a str>,
119    /// The column's emit name, interned ONCE per block as a shared `Arc<str>`
120    /// (issue #1334). Populating a decoded cell with its name is then an
121    /// `Arc::clone` refcount bump instead of a per-cell, per-row heap `String`
122    /// allocation. For an emitted column this is the supplied-schema column
123    /// name; for a DROPPED column (never emitted) it mirrors the on-disk header
124    /// name and is unused.
125    pub(super) name: Arc<str>,
126}
127
128/// Pre-resolved header→schema column ordering for a whole SSTable block.
129///
130/// Issue #1046 (the true hoist): the header→schema-column resolution is CONSTANT
131/// for every row in an SSTable — the serialization header is fixed and the
132/// supplied schema does not change between rows. Building it once per block and
133/// reusing it across every partition/row means the per-row decode performs ZERO
134/// schema-lookup allocations (no per-row `HashMap`, no per-row `String` clone, no
135/// per-row `Vec` of columns). The expensive `O(header_cols × schema_cols)` name
136/// resolution + the `Vec` allocations happen ONCE per block, not once per row.
137///
138/// Two orderings are precomputed because the on-disk column group differs by row
139/// kind (Cassandra's `missing_columns_bitmap` covers only the static columns for
140/// a static row and only the regular columns for a regular row — issue #702):
141///   - `regular`: header columns with `is_static == false`
142///   - `static_`: header columns with `is_static == true`
143///
144/// The per-row missing-columns bitmap filter is applied by iterating the chosen
145/// slice and skipping bitmapped-out indices INLINE (no per-row `Vec`), so the
146/// row body still allocates nothing for column resolution.
147///
148/// Lifetime: borrows the header-type strings from `reader.header` and the
149/// `Column`s from the supplied `schema`; both outlive the block's row loops
150/// (resolution is built at the top of each `parse_block_emit*` / per-partition
151/// driver, where `reader` and `schema` are in scope). This is a per-BLOCK hoist —
152/// allocations scale with block count, not row count.
153pub(super) struct RowColumnResolution<'a> {
154    /// On-disk regular (non-static) columns in serialization-header order.
155    regular: Vec<ColumnToParse<'a>>,
156    /// On-disk static columns in serialization-header order.
157    static_: Vec<ColumnToParse<'a>>,
158    /// Clustering-key names interned once per block as shared `Arc<str>` handles
159    /// (issue #1334), positionally aligned with `schema.clustering_keys`. Reused
160    /// so the per-row clustering-key cell insert is an `Arc::clone`, not a
161    /// `String` allocation.
162    clustering: Vec<Arc<str>>,
163}
164
165impl<'a> RowColumnResolution<'a> {
166    /// Build the resolution ONCE for a block from the on-disk serialization
167    /// header (`reader.header`) and the supplied `schema`.
168    ///
169    /// Resolution is an exact column-name match, keep-FIRST on a duplicate name
170    /// (Cassandra schemas cannot have duplicate column names, so this is purely
171    /// defensive and matches the prior per-row `HashMap`/`iter().find()`
172    /// semantics). On the header-empty fallback path (synthetic SSTables) the
173    /// supplied schema order is used directly, every column schema-present by
174    /// construction.
175    pub(super) fn build(
176        schema: &'a TableSchema,
177        reader: &'a crate::storage::sstable::reader::types::SSTableReader,
178    ) -> Self {
179        if !reader.header.columns.is_empty() {
180            // O(header_cols × schema_cols) name resolution, but ONCE per block via
181            // a borrowed-key map (keys are `&str` slices into `schema.columns[].name`,
182            // no `String` clones; values are `&Column`). Both borrow for `'a`.
183            let mut resolve_lookup: HashMap<&'a str, &'a crate::schema::Column> =
184                HashMap::with_capacity(schema.columns.len());
185            for col in &schema.columns {
186                resolve_lookup.entry(col.name.as_str()).or_insert(col);
187            }
188
189            let build_for = |want_static: bool| -> Vec<ColumnToParse<'a>> {
190                reader
191                    .header
192                    .columns
193                    .iter()
194                    .filter(|col_info| {
195                        !col_info.is_primary_key
196                            && !col_info.is_clustering
197                            && col_info.is_static == want_static
198                    })
199                    .map(|col_info| {
200                        let schema = resolve_lookup.get(col_info.name.as_str()).copied();
201                        // Intern the emit name ONCE per block (issue #1334): the
202                        // supplied-schema column name when matched, else the
203                        // on-disk header name for a DROPPED column (never emitted).
204                        let name: Arc<str> = Arc::from(
205                            schema
206                                .map(|c| c.name.as_str())
207                                .unwrap_or(col_info.name.as_str()),
208                        );
209                        ColumnToParse {
210                            schema,
211                            header_type: Some(col_info.column_type.as_str()),
212                            name,
213                        }
214                    })
215                    .collect()
216            };
217
218            RowColumnResolution {
219                regular: build_for(false),
220                static_: build_for(true),
221                clustering: schema
222                    .clustering_keys
223                    .iter()
224                    .map(|k| Arc::from(k.name.as_str()))
225                    .collect(),
226            }
227        } else {
228            // Fallback to schema order when header is empty (shouldn't happen for
229            // real SSTables). Filter out partition/clustering keys (regular columns
230            // only carry cell data) and split by row kind.
231            log::warn!("V5CompressedLegacy: reader.header.columns is empty, falling back to schema order (may cause column misalignment)");
232            let partition_key_names: std::collections::HashSet<&str> = schema
233                .partition_keys
234                .iter()
235                .map(|k| k.name.as_str())
236                .collect();
237            let clustering_key_names: std::collections::HashSet<&str> = schema
238                .clustering_keys
239                .iter()
240                .map(|k| k.name.as_str())
241                .collect();
242            let build_for = |want_static: bool| -> Vec<ColumnToParse<'a>> {
243                schema
244                    .columns
245                    .iter()
246                    .filter(|col| {
247                        !partition_key_names.contains(col.name.as_str())
248                            && !clustering_key_names.contains(col.name.as_str())
249                            && col.is_static == want_static
250                    })
251                    .map(|col| ColumnToParse {
252                        schema: Some(col),
253                        header_type: None,
254                        name: Arc::from(col.name.as_str()),
255                    })
256                    .collect()
257            };
258            RowColumnResolution {
259                regular: build_for(false),
260                static_: build_for(true),
261                clustering: schema
262                    .clustering_keys
263                    .iter()
264                    .map(|k| Arc::from(k.name.as_str()))
265                    .collect(),
266            }
267        }
268    }
269
270    /// The interned clustering-key name at position `i` (issue #1334), or `None`
271    /// when out of range. Cloning the returned handle into the cells map is an
272    /// `Arc::clone`, not a `String` allocation.
273    pub(super) fn clustering_name(&self, i: usize) -> Option<&Arc<str>> {
274        self.clustering.get(i)
275    }
276
277    /// The pre-resolved on-disk column ordering for a row of the given kind.
278    pub(super) fn columns_for(&self, is_static: bool) -> &[ColumnToParse<'a>] {
279        if is_static {
280            &self.static_
281        } else {
282            &self.regular
283        }
284    }
285}

305/// Outcome of [`V5CompressedLegacyParser::parse_one_partition_with_timestamps`].
306///
307/// The sliding-window compaction-read driver (issue #827) feeds the parser a
308/// `data` slice that grows as decompressed chunks are appended and shrinks as
309/// confirmed partitions are drained. Because a single partition can straddle a
310/// compression-chunk boundary, the bounded parser must distinguish "I parsed a
311/// complete partition" from "I ran out of buffer mid-partition and need more
312/// bytes" — conflating the two would silently drop trailing partitions.
313#[derive(Debug, PartialEq, Eq)]
314pub enum ParseStep {
315    /// A full partition was parsed; `usize` bytes of `data` were consumed and
316    /// can be drained from the front of the sliding window.
317    Emitted(usize),
318    /// `data` appears truncated mid-partition. The caller should append the
319    /// next decompressed chunk and retry. Only returned when `!at_final_chunk`.
320    NeedMore,
321    /// Genuine end of partitions in `data` (no more bytes to consume).
322    Done,
323}

325/// Row header data extracted from V5CompressedLegacy row
326#[derive(Debug, Clone)]
327struct RowHeader {
328    /// Row-level timestamp (after delta decoding from min_timestamp)
329    timestamp: Option<i64>,
330    /// Row-level TTL (after delta decoding from min_ttl)
331    ttl: Option<i32>,
332    /// Liveness local deletion time in SECONDS: absolute epoch-second expiry for a
333    /// TTL-bearing INSERT row (the `pk_liveness.ttl()` clock, from `HAS_TTL`).
334    /// This is the `expires_at` for the row liveness — distinct from
335    /// `local_deletion_time` which is the GC-grace clock for row tombstones.
336    /// `None` when `HAS_TTL` was not set.
337    ///
338    /// Stored as `i64` (not `i32`): on oa/da (`hasUIntDeletionTime`) a post-2038 expiry
339    /// is decoded as an UNSIGNED 32-bit value in `[2^31, 2^32)`; keeping it `i64`
340    /// preserves the large positive second count so the read-time TTL filter does not
341    /// wrap it negative and hide a still-live row (#1741 F1). See the decode site in
342    /// `row_framing.rs` for the reinterpretation.
343    liveness_expires_at_seconds: Option<i64>,
344    /// Row-level local deletion time in SECONDS (after delta decoding from
345    /// min_local_deletion_time). This is the GC-grace clock, NOT the reconciliation
346    /// timestamp; do not use it for last-write-wins comparisons.
347    local_deletion_time: Option<i32>,
348    /// Row-level `markedForDeleteAt` reconciliation timestamp in MICROSECONDS
349    /// (absolute = min_timestamp + delta). For a row tombstone this is the
350    /// authoritative deletion timestamp used by the compaction merger for
351    /// timestamp-based last-write-wins shadowing (Issue #505). `None` when the
352    /// row has no HAS_DELETION flag.
353    marked_for_delete_at: Option<i64>,
354    /// Number of bytes consumed by the header
355    header_size: usize,
356    /// Length of the row_size VInt in bytes (needed for offset calculation)
357    /// row_size is measured from AFTER this VInt is consumed
358    row_size_vint_len: usize,
359    /// Bitmask of missing columns from Cassandra's Columns.Serializer format.
360    /// bit=1 means column missing, bit=0 means column present.
361    /// None when HAS_ALL_COLUMNS flag is set.
362    missing_columns_bitmap: Option<u64>,
363    /// Issue #1741 read-side shadowing aggregate: max write timestamp (µs) across
364    /// this row's decoded DATA cells (excludes partition/clustering pseudo-cells;
365    /// a cell inheriting the row timestamp contributes the row liveness timestamp,
366    /// and a non-frozen collection contributes its newest element ts). Includes
367    /// shadow/TTL-dropped cells so a fully-reduced row is still recognised as
368    /// shadowed. `None` when the row decoded no data cell (e.g. a truncated parse).
369    /// Combined with `timestamp` to decide whether a tombstone shadows the WHOLE row.
370    max_data_cell_timestamp: Option<i64>,
371    /// Issue #1741: max effective expiry (epoch seconds) across this row's expiring
372    /// DATA cells (explicit per-cell TTL or a `USE_ROW_TTL` cell inheriting the
373    /// row's expiry; includes expired cells). `None` when none is expiring.
374    max_data_cell_expires_at: Option<i64>,
375    /// Issue #1741: `true` when at least one decoded DATA cell is LIVE (not
376    /// shadowed, not expired) and live-forever (no TTL). Keeps the row visible
377    /// regardless of liveness expiry. A shadowed/expired cell never sets this.
378    has_live_forever_data_cell: bool,
379}
380
381impl RowHeader {
382    /// Whether this row header describes a row tombstone (HAS_DELETION was set).
383    ///
384    /// Detected via `local_deletion_time` being present, which is only set when the
385    /// HAS_DELETION (0x10) flag was decoded (Issue #505).
386    fn is_row_tombstone(&self) -> bool {
387        self.local_deletion_time.is_some()
388    }
389
390    /// Build the `Value::Tombstone(RowTombstone)` for this header.
391    ///
392    /// The reconciliation `deletion_time` is `marked_for_delete_at` (the
393    /// `markedForDeleteAt` field, MICROSECONDS — same clock as HAS_TIMESTAMP), NOT
394    /// the row write `timestamp` and NOT `local_deletion_time` (seconds, GC clock).
395    /// For a pure row tombstone HAS_TIMESTAMP is absent, so using the write
396    /// timestamp would yield epoch 0 and lose every merge comparison (Issue #505).
397    ///
398    /// Falls back to `local_deletion_time` promoted to microseconds only if
399    /// `marked_for_delete_at` is somehow absent while a deletion was recorded
400    /// (should not happen given HAS_DELETION always writes both VInts), keeping the
401    /// merger's LWW ordering meaningful rather than defaulting to 0.
402    fn row_tombstone(&self) -> Value {
403        let deletion_time = self.row_tombstone_deletion_time();
404        Value::Tombstone(TombstoneInfo {
405            deletion_time,
406            tombstone_type: TombstoneType::RowTombstone,
407            // Carry the on-disk `localDeletionTime` (GC clock, seconds) so the
408            // compaction merge→rewrite path can preserve it (#873). Absent for a
409            // non-tombstone header, hence the `0` fallback.
410            local_deletion_time: self.local_deletion_time.unwrap_or(0) as i64,
411            ttl: None,
412            range_start: None,
413            range_end: None,
414        })
415    }
416
417    /// Authoritative reconciliation timestamp (microseconds) for a row tombstone.
418    fn row_tombstone_deletion_time(&self) -> i64 {
419        match self.marked_for_delete_at {
420            Some(ts) => ts,
421            // Defensive fallback: promote the seconds-based local deletion time to
422            // microseconds so ordering remains non-zero and monotonic.
423            None => self
424                .local_deletion_time
425                .map(|s| (s as i64).saturating_mul(1_000_000))
426                .unwrap_or(0),
427        }
428    }
429
430    /// Issue #1741: the row's maximum write timestamp (µs) across its liveness and
431    /// decoded data cells. Used for partition/range-tombstone shadowing.
432    fn max_write_timestamp(&self) -> i64 {
433        let mut m = self.timestamp.unwrap_or(i64::MIN);
434        if let Some(c) = self.max_data_cell_timestamp {
435            if c > m {
436                m = c;
437            }
438        }
439        m
440    }
441
442    /// Issue #1741: `true` when a deletion at `deleted_at_micros` shadows the WHOLE
443    /// row — i.e. every piece of the row's data is older than (or equal to) the
444    /// deletion. Follows Cassandra `DeletionTime.deletes(ts) = ts <= markedForDeleteAt`.
445    ///
446    /// Fail-safe: a row with NO authoritative timestamp (the `i64::MIN` sentinel —
447    /// no liveness and no decodable data cell, e.g. an empty/undecodable row) is
448    /// NOT shadowed. We only hide a row when authoritative metadata proves it
449    /// predates the deletion, never by guessing (no-heuristics mandate, issue #28).
450    fn shadowed_by_deletion_at(&self, deleted_at_micros: i64) -> bool {
451        let max_ts = self.max_write_timestamp();
452        max_ts != i64::MIN && max_ts <= deleted_at_micros
453    }
454
455    /// Issue #1741: read-time TTL expiry. `true` iff the row carries a TTL somewhere
456    /// AND every piece of its data (primary-key liveness + cells) has expired at
457    /// `now_secs`, matching what a Cassandra `SELECT` hides. Returns `false` for any
458    /// row with no TTL at all (never touches the tombstone-free common case).
459    fn row_liveness_expired(&self, now_secs: i64) -> bool {
460        let has_ttl =
461            self.liveness_expires_at_seconds.is_some() || self.max_data_cell_expires_at.is_some();
462        if !has_ttl {
463            return false;
464        }
465        // A live-forever data cell (written with no TTL) keeps the row visible.
466        if self.has_live_forever_data_cell {
467            return false;
468        }
469        // Primary-key liveness still live? Present iff HAS_TIMESTAMP; live-forever
470        // when it carries no TTL expiry, otherwise live until its expiry passes.
471        let liveness_live = self.timestamp.is_some()
472            && self
473                .liveness_expires_at_seconds
474                .is_none_or(|s| s > now_secs);
475        if liveness_live {
476            return false;
477        }
478        // Any expiring data cell still live keeps the row visible.
479        if self.max_data_cell_expires_at.is_some_and(|e| e > now_secs) {
480            return false;
481        }
482        true
483    }
484}

486/// Result of parsing a single complex cell (an element of a list/set, or a
487/// key/value entry of a map).  See [`V5CompressedLegacyParser::parse_complex_cell_value`].
488///
489/// `is_deleted` carries the authoritative IS_DELETED (0x01) cell flag so that
490/// collection parsers can distinguish element-level tombstones from live
491/// elements that merely have an empty value (Issue #493).
492struct ComplexCellParse {
493    /// Decoded cell value, or `None` if the cell was deleted or had an empty value.
494    value: Option<Value>,
495    /// Raw cell-path bytes (the element value for sets, the key for maps).
496    path_bytes: Vec<u8>,
497    /// Whether the cell carries the IS_DELETED (0x01) flag (an element tombstone).
498    is_deleted: bool,
499    /// Whether the cell carries the HAS_EMPTY_VALUE (0x04) flag (an on-disk
500    /// empty-value cell — e.g. a SET member whose identity lives entirely in the
501    /// cell_path). Surfaced so the compaction writer can reproduce the SAME
502    /// on-disk emptiness byte-for-byte rather than re-deriving it from the
503    /// decoded value (epic #899, Phase C: a SET element's decoded member is the
504    /// path bytes, NOT a cell value).
505    has_empty_value: bool,
506    /// Offset immediately following the parsed cell.
507    next_offset: usize,
508    /// Per-element writetime decoded from the cell's own timestamp field, in µs since
509    /// Unix epoch (absolute, after delta decoding from min_timestamp).  `None` when the
510    /// element inherited the row-level timestamp (USE_ROW_TIMESTAMP flag 0x08).
511    ///
512    /// Used by `parse_complex_column_inner` to compute the max element writetime
513    /// for a collection column (Issue #700, DS4).
514    element_writetime: Option<i64>,
515    /// Per-element TTL in seconds when the cell is expiring (`IS_EXPIRING`

622/// Issue #932: does the decoded cell map hold any NON-primary-key data cell?
623///
624/// Primary-key (partition + clustering) columns are surfaced into the cell map
625/// as pseudo-cells (#229) so the read-back path can recover the clustering
626/// identity; they are NOT row data. A row carrying `HAS_DELETION` is a PURE row
627/// tombstone only when no such data cell survives — otherwise the row deletion
628/// COEXISTS with surviving (strictly-newer) cells and the row displays as live.
629fn row_has_non_key_cell(cells: &HashMap<Arc<str>, Value>, schema: &TableSchema) -> bool {
630    cells.keys().any(|name| {
631        let name: &str = name;
632        !schema.partition_keys.iter().any(|k| k.name == name)
633            && !schema.clustering_keys.iter().any(|c| c.name == name)
634    })
635}

637/// Issue #932/#1741: build the user-facing `ScanRow` display value for a parsed
638/// clustering row from its decoded `cells` and row header. Shared by every
639/// user-facing emit path so the row-tombstone display rule lives in ONE place:
640/// a `HAS_DELETION` row that carries NO surviving non-key cell displays as a pure
641/// `Tombstone` marker (suppressed downstream by `filter_tombstone`); a row that
642/// still carries surviving cells displays as a live `Row` (the deletion shadows
643/// only already-absent older cells). An empty cell set becomes a null marker.
644fn build_display_row(
645    cells: HashMap<Arc<str>, Value>,
646    row_header_opt: Option<&RowHeader>,
647    schema: &TableSchema,
648) -> ScanRow {
649    let row_tombstone = row_header_opt.filter(|h| h.is_row_tombstone());
650    let has_data_cell = row_has_non_key_cell(&cells, schema);
651    if row_tombstone.is_some() && !has_data_cell {
652        ScanRow::Marker(
653            row_tombstone
654                .map(|h| h.row_tombstone())
655                .unwrap_or(Value::Null),
656        )
657    } else if cells.is_empty() {
658        ScanRow::Marker(Value::Null)
659    } else {
660        // Issue #1334: carry the interned `Arc<str>` name handles straight into
661        // the row carrier; emit-time alphabetical ordering preserved exactly.
662        let mut row_cells: RowCells = cells.into_iter().collect();
663        row_cells.sort_by(|a, b| a.0.as_ref().cmp(b.0.as_ref()));
664        ScanRow::Row(row_cells)
665    }
666}

668/// Issue #1741: extract a clustering row's clustering-key values (in schema
669/// clustering order) from its decoded cell map. Only called when a range
670/// tombstone is currently OPEN in the partition, so the per-row clone is off the
671/// tombstone-free hot path. Returns fewer values than the clustering arity only
672/// for a malformed/partial row (missing clustering pseudo-cells).
673fn extract_clustering_values(cells: &HashMap<Arc<str>, Value>, schema: &TableSchema) -> Vec<Value> {

675        .clustering_keys
676        .iter()

843    /// Return `true` when the version gates indicate `hasUIntDeletionTime` (oa / da).
844    ///
845    /// Authority: BigFormat.java:409 — `hasUintDeletionTime = version.compareTo("oa") >= 0`
846    #[inline]
847    fn has_uint_deletion_time(&self) -> bool {
848        match self.version_gates.as_ref() {
849            VersionGates::Big(g) => g.has_uint_deletion_time,

854    /// Issue #1741 (Finding 2): `true` when this SSTable's authoritative
855    /// EncodingStats prove it carries NO deletions of any kind — hence NO range
856    /// tombstones. A clustering-slice read can then keep the O(slice) row-index
857    /// fast-forward and skip prefix priming entirely (a range tombstone opening
858    /// before the slice is impossible).
859    ///
860    /// `min_local_deletion_time` is `EncodingStats.minLocalDeletionTime`, the MIN
861    /// of every cell's `localDeletionTime`. A live cell contributes the LIVE
862    /// sentinel `Cell.NO_DELETION_TIME == Integer.MAX_VALUE`; a partition/row/
863    /// range/cell tombstone OR an expiring cell contributes a smaller value. So
864    /// the min equals `Integer.MAX_VALUE` iff the SSTable has no deletion and no
865    /// TTL. This OVER-approximates range-tombstone presence (a cell tombstone or
866    /// TTL also trips it), which is safe: priming then runs and stays correct. No
867    /// stats (`min == 0` from the `build_v5_parser` fallback) conservatively primes.
868    /// No heuristics — authoritative metadata only (issue #28).
869    #[inline]
870    fn sstable_may_have_range_tombstones(&self) -> bool {
871        // Integer.MAX_VALUE — Cassandra `Cell.NO_DELETION_TIME` LIVE sentinel.
872        const NO_DELETION_TIME: i64 = i32::MAX as i64;
873        self.min_local_deletion_time != NO_DELETION_TIME
874    }

876    /// Try to parse partition header at offset WITHOUT consuming it.
877    ///
878    /// This performs a full parse attempt to determine if the bytes at offset
879    /// represent a valid partition header. This is the NO-HEURISTICS approach:
880    /// we actually try to parse the structure instead of guessing based on byte patterns.
881    ///
882    /// # Arguments
883    /// * `data` - Binary data buffer
884    /// * `offset` - Offset to check
885    ///
886    /// # Returns
887    /// * `true` if a valid partition header can be parsed at this offset
888    /// * `false` if parsing fails (likely a row header or invalid data)
889    ///
890    /// # Visibility
891    /// Exposed for integration testing to validate partition boundary detection
892    #[doc(hidden)]
893    pub fn peek_is_partition_header(&self, data: &[u8], offset: usize) -> bool {
894        // Issue #229 FIX: Check for END_OF_PARTITION marker FIRST
895        //
896        // The END_OF_PARTITION marker (0x01) can be misinterpreted as a valid partition
897        // header because parse_partition_header doesn't validate flags semantically.
898        // We must explicitly reject END_OF_PARTITION (0x01) and IS_MARKER (0x02) here.
899        if offset < data.len() {
900            let flags = data[offset];
901            if Self::is_end_of_partition(flags) || Self::is_range_tombstone_marker(flags) {
902                return false; // These are markers, not partition headers
903            }
904        }
905
906        // Try to actually parse the partition header
907        self.parse_partition_header(data, offset).is_ok()
908    }
