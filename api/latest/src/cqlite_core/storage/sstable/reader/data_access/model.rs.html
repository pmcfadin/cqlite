15/// A single-column clustering-key restriction pushed down to a within-partition
16/// seek (Issue #954, Epic #951).
17///
18/// Carries the decoded clustering bound `Value`(s) for the FIRST clustering
19/// column (multi-column prefixes are a documented follow-up). Each bound is one
20/// clustering component; an empty `Vec` means that side is OPEN (unbounded). The
21/// `_inclusive` flags distinguish `>=`/`<=` (inclusive) from `>`/`<` (exclusive)
22/// — they are advisory for byte-window selection (block selection is
23/// over-inclusive by block granularity; the post-scan `evaluate_leaf` backstop
24/// applies the exact bound), so a slightly wider byte window never changes the
25/// final result.
26///
27/// `ck = v` is represented as `start == end == [v]`, both inclusive.
28#[derive(Debug, Clone)]
29pub struct ClusteringSlice {
30    /// Lower bound clustering component(s); empty = unbounded below.
31    pub start: Vec<Value>,
32    /// Whether the lower bound is inclusive (`>=`/`=`) vs exclusive (`>`).
33    pub start_inclusive: bool,
34    /// Upper bound clustering component(s); empty = unbounded above.
35    pub end: Vec<Value>,
36    /// Whether the upper bound is inclusive (`<=`/`=`) vs exclusive (`<`).
37    pub end_inclusive: bool,
38}

40/// The within-partition row-body byte window selected by the BTI row index for a
41/// clustering slice (Issue #954). Offsets are RELATIVE to the partition start —
42/// the same domain the parser sees for `window[within..]`.
43#[cfg(not(feature = "tombstones"))]
44#[derive(Debug, Clone, Copy)]
45pub(super) struct ClusteringRowWindow {
46    /// First byte of the row body to parse (inclusive), relative to partition
47    /// start. `0` decodes from the partition body start (used when statics exist).
48    pub(super) body_start_rel: usize,
49    /// Exclusive end of the row body to parse, relative to partition start;
50    /// `usize::MAX` means "to the partition end" (clamped by the caller).
51    pub(super) body_end_rel: usize,
52}

54/// Length of the all-`0xFF` sentinel used to represent an OPEN upper clustering
55/// bound (+∞) for byte-comparable block selection (Issue #954). Any separator in
56/// `Rows.db` is shorter or sorts below an all-`0xFF` run of this length, so it
57/// reliably selects through the last block. 64 bytes comfortably exceeds any
58/// realistic single-column clustering separator width.
59#[cfg(not(feature = "tombstones"))]
60pub(super) const MAX_OSS50_BOUND_SENTINEL_LEN: usize = 64;

62/// Normalize a CQL [`ClusteringSlice`] into the `(physical_lower, physical_upper)`
63/// byte-comparable bounds that [`select_row_index_blocks_for_range`] consumes
64/// (issue #954 High-severity correctness fix).
65///
66/// [`select_row_index_blocks_for_range`] selects blocks purely in **physical**
67/// (on-disk, byte-comparable) order — the order `Rows.db` separators are stored
68/// in. The encoder ([`encode_clustering_bound_oss50_with_order`]) already inverts
69/// every byte of a DESC component (`ReversedType` / `ByteSource.invert`), so for a
70/// DESC first clustering column an ASCENDING byte order corresponds to a
71/// DESCENDING CQL value order. The consequence:
72///
73/// - **ASC** (`is_reversed[0] == false`): CQL `start` (the `>=`/`>` lower value
74///   bound) IS the physical-lower bound and CQL `end` (the `<=`/`<` upper value
75///   bound) IS the physical-upper bound. No swap.
76/// - **DESC** (`is_reversed[0] == true`): the CQL lower-value bound encodes to the
77///   physically GREATER bytes and the CQL upper-value bound to the physically
78///   SMALLER bytes, so the roles SWAP. A CQL `ck >= v` (lower) selects the rows
79///   with the largest values, which sit at the physical-LOW byte side through
80///   `enc(v)`; a CQL `ck < v` (upper) selects the physical-HIGH side. The open
81///   sentinels swap accordingly: an open CQL lower (`-∞` value) maps to physical
82///   `+∞` and an open CQL upper (`+∞` value) maps to physical `-∞`.
83///
84/// Block selection is over-inclusive by block granularity and the post-scan
85/// `evaluate_leaf` backstop re-applies the exact CQL bound by VALUE, so the
86/// inclusivity of each bound does not need separate handling here — the physical
87/// window only has to be a SUPERSET of the matching rows. (Swapping the roles is
88/// what guarantees the superset for DESC; the previous code built
89/// `[enc(lower), +∞]` for DESC, which excluded the matching low-byte rows and the
90/// backstop could not recover rows that were never decoded.)
91///
92/// Returns:
93/// - `Ok(Some((lower, upper)))` — usable physical bounds.
94/// - `Ok(None)` — a bound's type is not byte-comparable-encodable here, so the
95///   narrowing is unsafe and the caller must decode the whole partition.
96/// - `Err(_)` — never (kept `Result` for call-site symmetry); reserved.
97#[cfg(not(feature = "tombstones"))]
98pub(super) fn physical_byte_bounds_for_slice(
99    slice: &ClusteringSlice,
100    is_reversed: &[bool],
101) -> Result<Option<(Vec<u8>, Vec<u8>)>> {
102    use crate::storage::sstable::bti::encode_clustering_bound_oss50_with_order;
103
104    // The physical-low sentinel: empty sorts before every separator (-∞).
105    let neg_inf = Vec::<u8>::new();
106    // The physical-high sentinel: an all-0xFF run sorts after every separator (+∞).
107    let pos_inf = || vec![0xFFu8; MAX_OSS50_BOUND_SENTINEL_LEN];
108
109    // Encode a closed CQL bound to its physical byte-comparable form; `None`
110    // bubbles up as an un-encodable-bound fallback at the call site.
111    let encode = |values: &[Value]| -> Option<Vec<u8>> {
112        encode_clustering_bound_oss50_with_order(values, is_reversed).ok()
113    };
114
115    // The FIRST clustering column's order decides the value↔byte direction. A
116    // missing entry (no schema / fewer entries) is ascending, matching the encoder.
117    let first_desc = is_reversed.first().copied().unwrap_or(false);
118
119    // CQL lower bound (`>=`/`>` / equality lower) → its physical byte image.
120    let cql_lower_bytes = if slice.start.is_empty() {
121        None // open CQL lower (value -∞)
122    } else {
123        match encode(&slice.start) {
124            Some(b) => Some(b),
125            None => return Ok(None),
126        }
127    };
128    // CQL upper bound (`<=`/`<` / equality upper) → its physical byte image.
129    let cql_upper_bytes = if slice.end.is_empty() {
130        None // open CQL upper (value +∞)
131    } else {
132        match encode(&slice.end) {
133            Some(b) => Some(b),
134            None => return Ok(None),
135        }
136    };
137
138    let (phys_lower, phys_upper) = if first_desc {
139        // DESC: CQL upper-value bound → physical-lower bytes; CQL lower-value bound
140        // → physical-upper bytes. Open CQL upper → physical -∞; open CQL lower →
141        // physical +∞.
142        let lower = cql_upper_bytes.unwrap_or(neg_inf);
143        let upper = cql_lower_bytes.unwrap_or_else(pos_inf);
144        (lower, upper)
145    } else {
146        // ASC: CQL lower-value bound → physical-lower bytes; CQL upper-value bound
147        // → physical-upper bytes (the original mapping).
148        let lower = cql_lower_bytes.unwrap_or(neg_inf);
149        let upper = cql_upper_bytes.unwrap_or_else(pos_inf);
150        (lower, upper)
151    };
152
153    Ok(Some((phys_lower, phys_upper)))
154}

201/// Compare two table IDs, handling both qualified (keyspace.table) and unqualified (table) formats.
202///
203/// This function allows flexible matching:
204/// - "keyspace.table" matches "keyspace.table" (exact match)
205/// - "table" matches "keyspace.table" (unqualified matches qualified)
206/// - "keyspace.table" matches "table" (qualified matches unqualified)
207/// - "table" matches "table" (exact match)
208///
209/// This is necessary because:
210/// - Dataset mode SSTables store qualified table_ids (e.g., "test_basic.simple_table")
211/// - Queries can use either qualified ("test_basic.simple_table") or unqualified ("simple_table") names
212/// - Production SSTables may use unqualified table_ids
213pub(in crate::storage::sstable::reader) fn table_ids_match(
214    entry_table_id: &TableId,
215    query_table_id: &TableId,
216) -> bool {
217    let entry_name = entry_table_id.name();
218    let query_name = query_table_id.name();
219
220    // Fast path: exact match
221    if entry_name == query_name {
222        return true;
223    }
224
225    // Extract unqualified table names for comparison
226    let entry_unqualified = if let Some(dot_pos) = entry_name.rfind('.') {
227        &entry_name[dot_pos + 1..]
228    } else {
229        entry_name
230    };
231
232    let query_unqualified = if let Some(dot_pos) = query_name.rfind('.') {
233        &query_name[dot_pos + 1..]
234    } else {
235        query_name
236    };
237
238    // Match if unqualified names are the same
239    entry_unqualified == query_unqualified
240}

242/// Stricter table-id match used by the BTI point-lookup guard (issue #831 review).
243///
244/// [`table_ids_match`] matches on the unqualified table name, so it treats
245/// `ks_a.users` and `ks_b.users` as equal — fine for index lookups that are
246/// already scoped to one table, but too permissive as a defensive guard against
247/// a fully-qualified wrong-keyspace query. When BOTH ids are qualified
248/// (`keyspace.table`), this requires exact `keyspace.table` equality; it only
249/// falls back to the permissive unqualified match when one side lacks a
250/// keyspace (preserving qualified-vs-unqualified flexibility).
251pub(super) fn table_ids_match_strict(entry_table_id: &TableId, query_table_id: &TableId) -> bool {
252    let entry_qualified = entry_table_id.name().contains('.');
253    let query_qualified = query_table_id.name().contains('.');
254    if entry_qualified && query_qualified {
255        entry_table_id.name() == query_table_id.name()
256    } else {
257        table_ids_match(entry_table_id, query_table_id)
258    }
259}

261/// Seek-path table-id consistency check (issue #1284, hardened per #1284 review).
262///
263/// The single-partition SEEK (`bti_decompress_and_parse_target_all`) builds
264/// `entry_table_id` from THIS reader's AUTHORITATIVE serialization header
265/// (`{header.keyspace}.{header.table_name}`). `query_table_id` is the query's id.
266///
267/// `fully_qualified_match` records HOW the manager resolved this reader for the
268/// query (`resolve_reader_list`):
269///   - `true`  — the query's fully-qualified `keyspace.table` key matched the
270///     reader map EXACTLY (path-authoritative: this reader genuinely IS the
271///     queried table), or the query was unqualified.
272///   - `false` — the query was fully qualified but reached this reader via the
273///     bare/unqualified NAME fallback (its keyspace did NOT match any map key).
274///
275/// [`table_ids_match_strict`] additionally requires the *keyspace* to match when
276/// BOTH ids are fully qualified — the correct `get()` defensive guard (#831). For
277/// a seek it falsely REJECTS every row when the reader's HEADER keyspace differs
278/// from the query keyspace even though resolution was an exact FQ match (e.g. a
279/// writer-produced header, or a header whose embedded keyspace is not the
280/// path-derived one), silently zeroing the result and forcing a full-scan
281/// fallback (#1284).
282///
283/// The relaxation (accept on a consistent TABLE NAME despite a header-keyspace
284/// divergence) is therefore gated on `fully_qualified_match`:
285///   - Exact FQ match (or unqualified query): the reader is authoritatively the
286///     target table, so a header-keyspace divergence is benign — accept when the
287///     table names agree, reject a different table name (the #831 wrong-table
288///     guard survives via [`table_ids_match`]). This is #1284's actual goal.
289///   - FQ query reached via the bare-name FALLBACK: the keyspace genuinely did
290///     NOT match, so a relaxed name-only check could surface rows from ANOTHER
291///     keyspace whose table name collides — keep STRICT keyspace matching
292///     ([`table_ids_match_strict`]) and REJECT the wrong-keyspace rows.
293///
294/// This is no-heuristic: it relies only on the authoritative header identity, the
295/// query id, and the authoritative resolution mode — never on guessing.
296///
297/// Used by BOTH the single-partition SEEK (`bti_collect_partition_rows`,
298/// `not(tombstones)` only) and the `get()` POINT-LOOKUP decoder
299/// (`bti_decompress_and_parse_target`, all feature builds, issue #1321). The
300/// point-lookup caller exists in every build, so this helper is NOT gated.
301pub(super) fn table_header_consistent_for_seek(
302    entry_table_id: &TableId,
303    query_table_id: &TableId,
304    fully_qualified_match: bool,
305) -> bool {
306    // Only relax to the name-only consistency check for a fully-qualified query
307    // when resolution was an EXACT fully-qualified match (the reader is
308    // path-authoritatively the queried table). A fully-qualified query that
309    // reached this reader via the bare-name fallback keeps STRICT keyspace
310    // matching, so it can never return rows from a different keyspace whose table
311    // name collides (#1284 review correctness fix).
312    if fully_qualified_match && query_table_id.name().contains('.') {
313        table_ids_match(entry_table_id, query_table_id)
314    } else {
315        table_ids_match_strict(entry_table_id, query_table_id)
316    }
317}

319/// Per-iteration decision for the BTI chunk-targeted point-lookup loop.
320#[derive(Debug, PartialEq, Eq)]
321pub(super) enum BtiLookupStep {
322    /// The full partition-key prefix is buffered and matches the queried key —
323    /// parse the partition.
324    Parse,
325    /// The header/key prefix straddles a chunk boundary and is not yet fully
326    /// buffered — read the next chunk before parsing (chunk-targeted path only).
327    PullNextChunk,
328    /// Treat the partition as absent: either the buffered key prefix does not
329    /// match, or a whole-section window is structurally too short to grow.
330    Absent,
331}
332
333/// Decide what the BTI point-lookup loop should do for the current window state.
334///
335/// Pure so the chunk-straddle control flow is unit-testable without a real
336/// multi-chunk BTI fixture (DataOffset partitions are narrow and fit in one
337/// chunk, so a boundary-straddling header cannot be produced by the available
338/// fixtures). Crucially, when the key prefix is not yet buffered on the
339/// chunk-targeted path this returns [`BtiLookupStep::PullNextChunk`] — it must
340/// NOT lead to parsing a truncated header (issue #831 review).
341pub(super) fn bti_lookup_step(
342    key_prefix_available: bool,
343    key_matches: bool,
344    chunk_targeted: bool,
345) -> BtiLookupStep {
346    if key_prefix_available {
347        if key_matches {
348            BtiLookupStep::Parse
349        } else {
350            BtiLookupStep::Absent
351        }
352    } else if chunk_targeted {
353        BtiLookupStep::PullNextChunk
354    } else {
355        BtiLookupStep::Absent
356    }
357}

359/// Sort a result slice in ascending Cassandra token order.
360///
361/// The authoritative ordering for SSTable partitions is ascending Murmur3 token, with
362/// equal-token ties broken by raw key bytes (lexicographic). This matches the on-disk
363/// physical order (spec §5, Appendix B §313) and the write engine's `DecoratedKey::cmp`.
364///
365/// Computes each key's token once to avoid O(n log n) recomputation inside the comparator.
366pub(super) fn sort_by_token_order(results: &mut Vec<(RowKey, ScanRow)>) {
367    // Map to (token, RowKey, ScanRow), sort, then reassemble.
368    let mut tagged: Vec<(i64, RowKey, ScanRow)> = results
369        .drain(..)
370        .map(|(k, v)| {
371            let t = cassandra_murmur3_token(k.as_bytes());
372            (t, k, v)
373        })
374        .collect();
375    tagged.sort_by(|a, b| a.0.cmp(&b.0).then_with(|| a.1.cmp(&b.1)));
376    results.extend(tagged.into_iter().map(|(_, k, v)| (k, v)));
377}

379/// Sort `(RowKey, ScanRow, CellMeta)` triples by Cassandra Murmur3 token order.
380pub(super) fn sort_by_token_order_with_meta(
381    results: &mut Vec<(
382        RowKey,
383        ScanRow,
384        std::collections::HashMap<String, CellWriteMetadata>,
385    )>,
386) {
387    let mut tagged: Vec<(
388        i64,
389        RowKey,
390        ScanRow,
391        std::collections::HashMap<String, CellWriteMetadata>,
392    )> = results
393        .drain(..)
394        .map(|(k, v, m)| {
395            let t = cassandra_murmur3_token(k.as_bytes());
396            (t, k, v, m)
397        })
398        .collect();
399    tagged.sort_by(|a, b| a.0.cmp(&b.0).then_with(|| a.1.cmp(&b.1)));
400    results.extend(tagged.into_iter().map(|(_, k, v, m)| (k, v, m)));
401}

412    fn test_table_ids_match_strict_keyspace_aware() {
413        let a = TableId::new("ks_a.users".to_string());
414        let b = TableId::new("ks_b.users".to_string());
415        // Both qualified, different keyspace, same table name → must NOT match
416        // (the permissive helper would match these).
417        assert!(table_ids_match(&a, &b), "permissive helper matches on name");
418        assert!(
419            !table_ids_match_strict(&a, &b),
420            "strict guard must reject a wrong-keyspace same-name query"
421        );
422        // Both qualified, identical → match.
423        let a2 = TableId::new("ks_a.users".to_string());
424        assert!(table_ids_match_strict(&a, &a2));
425        // One side unqualified → fall back to permissive name match.
426        let unq = TableId::new("users".to_string());
427        assert!(table_ids_match_strict(&a, &unq));
428        assert!(table_ids_match_strict(&unq, &a));

431    /// Issue #1284 (hardened per review): the SEEK consistency check relaxes to a
432    /// name-only match across a header-keyspace divergence ONLY when resolution
433    /// was an EXACT fully-qualified match. A fully-qualified query that reached
434    /// the reader via the bare-name FALLBACK keeps STRICT keyspace matching, so a
435    /// wrong-keyspace query can never return another keyspace's same-named rows.
436    /// A different table name always rejects (the #831 wrong-table guard).
437    #[test]
438    fn test_table_header_consistent_for_seek_gates_on_resolution_mode() {
439        // The seek builds `entry` from the SSTable's authoritative header.
440        let header = TableId::new("ks_a.users".to_string());
441
442        // 1. EXACT fully-qualified match, header keyspace MATCHES, same table:
443        //    ACCEPT (seek engages — the genuine #1284 goal, no regression).
444        let query_same = TableId::new("ks_a.users".to_string());
445        assert!(
446            table_header_consistent_for_seek(&header, &query_same, /*fq_match=*/ true),
447            "exact FQ match with a consistent header keyspace+table must accept (seek engages)"
448        );
449
450        // 1b. EXACT fully-qualified match, header keyspace DIFFERS but resolution
451        //     was path-authoritative (the benign #1284 divergence): ACCEPT on the
452        //     consistent table name. The strict guard alone would wrongly reject.
453        let query_div_exact = TableId::new("ks_b.users".to_string());
454        assert!(
455            !table_ids_match_strict(&header, &query_div_exact),
456            "precondition: the strict guard rejects a different-keyspace same-name query"
457        );
458        assert!(
459            table_header_consistent_for_seek(&header, &query_div_exact, /*fq_match=*/ true),
460            "Issue #1284: an EXACT fully-qualified resolution must accept rows when only the \
461             header keyspace diverges and the table name is consistent"
462        );
463
464        // 2. Fully-qualified query, header keyspace DIFFERS, reached via the
465        //    bare-name FALLBACK (fq_match = false), SAME table name: REJECT — the
466        //    keyspaces genuinely differ, so accepting would surface another
467        //    keyspace's same-named rows (#1284 review correctness fix).
468        let query_other_ks = TableId::new("ks_b.users".to_string());
469        assert!(
470            !table_header_consistent_for_seek(&header, &query_other_ks, /*fq_match=*/ false),
471            "Issue #1284 review: a fully-qualified query resolved via the bare-name fallback must \
472             NOT accept rows from a reader whose header keyspace differs (no wrong-keyspace rows)"
473        );
474
475        // 3. DIFFERENT table name: REJECT regardless of resolution mode (the #831
476        //    wrong-table guard must survive the relaxation).
477        let query_wrong_table = TableId::new("ks_b.accounts".to_string());
478        assert!(
479            !table_header_consistent_for_seek(&header, &query_wrong_table, /*fq_match=*/ true),
480            "Issue #831: a genuinely different table name must still be rejected (exact match)"
481        );
482        assert!(
483            !table_header_consistent_for_seek(
484                &header,
485                &query_wrong_table,
486                /*fq_match=*/ false
487            ),
488            "Issue #831: a genuinely different table name must still be rejected (fallback)"
489        );
490
491        // 4. UNqualified query: defers to the strict guard's permissive name match
492        //    (resolution mode irrelevant — there is no keyspace to mismatch).
493        let query_unqualified = TableId::new("users".to_string());
494        assert!(table_header_consistent_for_seek(
495            &header,
496            &query_unqualified,
497            /*fq_match=*/ false
498        ));

507    /// Issue #1321: the `get()` POINT-LOOKUP decoder
508    /// (`bti_decompress_and_parse_target`) now applies the SAME
509    /// resolution-mode-aware guard the SEEK path adopted in #1284 — replacing its
510    /// previous unconditional `table_ids_match_strict`. This exercises the exact
511    /// row-acceptance predicate the get path evaluates per emitted partition:
512    ///   - ACCEPT: an EXACT fully-qualified resolution with a consistent table name
513    ///     even when the reader's header keyspace diverges (the #1321 goal);
514    ///   - REJECT (fallback): a fully-qualified query that reached the reader via
515    ///     the bare-name fallback whose header keyspace differs — no wrong-keyspace
516    ///     rows on get() either;
517    ///   - REJECT (different table): a genuinely different table name (#831).
518    #[test]
519    fn test_get_point_lookup_guard_gates_on_resolution_mode() {
520        // The get() decoder builds `entry` (`tid`) from the parser-emitted partition,
521        // whose table id is this reader's authoritative serialization header.
522        let header = TableId::new("ks_a.users".to_string());
523
524        // ACCEPT: exact FQ match, header keyspace diverges, same table name. The old
525        // strict-only get guard would WRONGLY reject this (the #1321 false rejection).
526        let query_div_exact = TableId::new("ks_b.users".to_string());
527        assert!(
528            !table_ids_match_strict(&header, &query_div_exact),
529            "precondition: the old get() guard (strict) rejects this divergence"
530        );
531        assert!(
532            table_header_consistent_for_seek(&header, &query_div_exact, /*fq_match=*/ true),
533            "Issue #1321: get() must accept rows when resolution was an exact FQ match \
534             and only the header keyspace diverges"
535        );
536
537        // REJECT (fallback): fully-qualified query resolved via the bare-name
538        // fallback, header keyspace differs — must NOT surface another keyspace's rows.
539        assert!(
540            !table_header_consistent_for_seek(&header, &query_div_exact, /*fq_match=*/ false),
541            "Issue #1321: a fully-qualified get() resolved via the bare-name fallback must \
542             still REJECT a wrong-keyspace reader (strict keyspace match preserved)"
543        );
544
545        // REJECT (different table): a genuinely different table never matches,
546        // regardless of resolution mode (#831 wrong-table guard survives).
547        let query_wrong_table = TableId::new("ks_a.accounts".to_string());
548        assert!(
549            !table_header_consistent_for_seek(&header, &query_wrong_table, /*fq_match=*/ true),
550            "Issue #831: get() must still reject a different table name (exact match)"
551        );
552        assert!(
553            !table_header_consistent_for_seek(
554                &header,
555                &query_wrong_table,
556                /*fq_match=*/ false
557            ),
558            "Issue #831: get() must still reject a different table name (fallback)"
559        );
560
561        // Sanity: the existing strict default (fq_match=false) the unchanged per-reader
562        // `get()` callers pass is exactly today's behavior for a consistent FQ query.
563        let query_same = TableId::new("ks_a.users".to_string());
564        assert!(table_header_consistent_for_seek(
565            &header,
566            &query_same,
567            /*fq_match=*/ false
568        ));
569    }

640    /// DESC: the CQL lower/upper roles SWAP into physical order, and the result is
641    /// still a well-ordered `[phys_lower, phys_upper]` (the bug produced a
642    /// REVERSED, empty-selecting range).
643    #[cfg(not(feature = "tombstones"))]
644    #[test]
645    fn physical_bounds_desc_swaps_roles() {
646        // ck >= 100 AND ck < 110 on a DESC column.
647        let slice = int_slice(Some(100), true, Some(110), false);
648        let (lower, upper) = physical_byte_bounds_for_slice(&slice, &[true])
649            .expect("ok")
650            .expect("encodable");
651        // Physical-lower comes from the CQL UPPER bound (enc_desc(110)); physical-
652        // upper from the CQL LOWER bound (enc_desc(100)).
653        assert_eq!(
654            lower,
655            enc_int(110, true),
656            "DESC physical-lower must come from the CQL upper bound (110)"
657        );
658        assert_eq!(
659            upper,
660            enc_int(100, true),
661            "DESC physical-upper must come from the CQL lower bound (100)"
662        );
663        // The whole point: under DESC the swapped bounds are well-ordered. With the
664        // un-swapped (buggy) mapping the range would be [enc_desc(100),
665        // enc_desc(110)] which is REVERSED (enc_desc(100) > enc_desc(110)) and
666        // `select_row_index_blocks_for_range` returns EMPTY → dropped rows.
667        assert!(
668            lower < upper,
669            "DESC swapped physical bounds must be ordered (lower < upper); \
670             got lower={lower:?} upper={upper:?}"
671        );
672        assert!(
673            enc_int(100, true) > enc_int(110, true),
674            "sanity: under DESC, enc(100) sorts AFTER enc(110) in physical bytes — \
675             the un-swapped mapping would build a reversed (empty) range"
676        );
677    }

679    /// DESC single-bound `ck >= v` (open CQL upper): the matching values (v and
680    /// larger) all sort to the physical-LOW byte side (DESC inverts bytes), so the
681    /// physical window is `[-∞, enc_desc(v)]`. (The buggy un-swapped code built
682    /// `[enc_desc(v), +∞]`, which EXCLUDED exactly those low-byte matching rows.)
683    #[cfg(not(feature = "tombstones"))]
684    #[test]
685    fn physical_bounds_desc_lower_bound_only() {
686        // ck >= 290, open above.
687        let slice = int_slice(Some(290), true, None, false);
688        let (lower, upper) = physical_byte_bounds_for_slice(&slice, &[true])
689            .expect("ok")
690            .expect("encodable");
691        assert_eq!(
692            lower,
693            Vec::<u8>::new(),
694            "DESC `ck >= 290`: open CQL upper → physical -∞ (the matching large \
695             values sort to the LOW physical-byte side)"
696        );
697        assert_eq!(
698            upper,
699            enc_int(290, true),
700            "DESC `ck >= 290`: physical-upper = enc_desc(290) (the boundary value)"
701        );
702        assert!(lower < upper, "must be ordered");
703        // Crucial: enc_desc(290) sorts ABOVE enc_desc(299), so [-∞, enc_desc(290)]
704        // includes enc_desc(299) — the buggy [enc_desc(290), +∞] would NOT.
705        assert!(
706            enc_int(299, true) < enc_int(290, true),
707            "sanity: larger DESC value has smaller bytes"
708        );
709        assert!(
710            enc_int(299, true).as_slice() <= upper.as_slice(),
711            "the physical window must include enc_desc(299), a matching row"
712        );
713    }

715    /// DESC single-bound `ck < v` (open CQL lower): the matching values (smaller
716    /// than v) sort to the physical-HIGH byte side, so the physical window is
717    /// `[enc_desc(v), +∞]`.
718    #[cfg(not(feature = "tombstones"))]
719    #[test]
720    fn physical_bounds_desc_upper_bound_only() {
721        // ck < 20, open below.
722        let slice = int_slice(None, false, Some(20), false);
723        let (lower, upper) = physical_byte_bounds_for_slice(&slice, &[true])
724            .expect("ok")
725            .expect("encodable");
726        assert_eq!(
727            lower,
728            enc_int(20, true),
729            "DESC `ck < 20`: physical-lower = enc_desc(20) (the boundary value)"
730        );
731        assert_eq!(
732            upper,
733            vec![0xFFu8; MAX_OSS50_BOUND_SENTINEL_LEN],
734            "DESC `ck < 20`: open CQL lower → physical +∞ sentinel"
735        );
736        assert!(lower < upper, "must be ordered");
737        // A matching small value (ck=0) has the LARGEST DESC bytes, inside the
738        // window; the buggy [-∞, enc_desc(20)] mapping would exclude it.
739        assert!(
740            enc_int(0, true).as_slice() >= lower.as_slice(),
741            "the physical window must include enc_desc(0), a matching row"
742        );
743    }
