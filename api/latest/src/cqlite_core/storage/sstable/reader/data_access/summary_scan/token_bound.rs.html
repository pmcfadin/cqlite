1//! The half-open token bound pushed into the Summary-guided walk (issue #2413),
2//! split out of `summary_scan/mod.rs` so the membership rule and its pins sit in
3//! one small file (campsite rule, epic #1116).
4
5/// Half-open `(start_excl, end_incl]` token bound pushed into the per-SSTable walk
6/// (issue #2413 Option A). Mirrors the flight `TokenFilter` half-open semantics
7/// exactly (including the `start == end` FULL-ring convention, #2228); the flight
8/// crate constructs one of these from its `TokenFilter` and a grid test pins that
9/// the two agree, so the membership rule lives in ONE place.
10///
11/// Wrapping is DERIVED from the two endpoints ([`Self::is_wraparound`]), never
12/// carried as a third field: a caller-supplied flag can disagree with the bounds
13/// it describes, and Cassandra has no such flag to disagree with (issue #3634).
14#[derive(Debug, Clone, Copy)]
15pub struct ScanTokenBound {
16    /// Exclusive lower bound.
17    pub start_excl: i64,
18    /// Inclusive upper bound.
19    pub end_incl: i64,
20}
21
22impl ScanTokenBound {
23    /// Whether this is a ring-wraparound segment, derived exactly as
24    /// `Range.isWrapAround(left, right)` does at `cassandra-5.0.8`
25    /// (`src/java/org/apache/cassandra/dht/Range.java`):
26    /// `left.compareTo(right) >= 0`.
27    ///
28    /// The `>=` is load-bearing, and is why the FULL-ring convention (#2228)
29    /// needs no special case: equal endpoints wrap, and the wraparound arm of
30    /// [`Self::contains`] is then `t > s || t <= s`, which every token satisfies.
31    pub fn is_wraparound(&self) -> bool {
32        self.start_excl >= self.end_incl
33    }
34
35    /// Whether `token` is inside this half-open `(start, end]` range.
36    ///
37    /// Transcribed from `Range.contains(left, right, point)` at `cassandra-5.0.8`.
38    pub fn contains(&self, token: i64) -> bool {
39        if self.is_wraparound() {
40            token > self.start_excl || token <= self.end_incl
41        } else {
42            token > self.start_excl && token <= self.end_incl
43        }
44    }
45
46    /// Whether every remaining (token-ascending) partition is guaranteed to be
47    /// ABOVE this range, so a forward walk can stop. Only sound for a
48    /// non-wraparound range (a wraparound range has in-range tokens at both ends
49    /// of the ring, so a forward walk cannot early-stop) — which subsumes the
50    /// full ring, since equal endpoints wrap.
51    pub(super) fn can_stop_past(&self, token: i64) -> bool {
52        !self.is_wraparound() && token > self.end_incl
53    }
54}

56#[cfg(test)]
57mod tests {
58    use super::ScanTokenBound;
59
60    #[test]
61    fn contains_non_wraparound_half_open() {
62        let b = ScanTokenBound {
63            start_excl: 10,
64            end_incl: 20,
65        };
66        assert!(!b.contains(10), "start is exclusive");
67        assert!(b.contains(11));
68        assert!(b.contains(20), "end is inclusive");
69        assert!(!b.contains(21));
70    }
71
72    #[test]
73    fn contains_equal_endpoints_is_full_ring() {
74        let b = ScanTokenBound {
75            start_excl: 5,
76            end_incl: 5,
77        };
78        assert!(
79            b.is_wraparound(),
80            "equal endpoints wrap (Range.isWrapAround `>=`)"
81        );
82        for t in [i64::MIN, -1, 0, 5, 6, i64::MAX] {
83            assert!(b.contains(t), "equal endpoints cover every token (#2228)");
84        }
85    }
86
87    #[test]
88    fn contains_wraparound() {
89        let b = ScanTokenBound {
90            start_excl: 100,
91            end_incl: -100,
92        };
93        assert!(b.is_wraparound(), "start > end wraps");
94        assert!(b.contains(101), "above start is in range");
95        assert!(b.contains(-100), "at/below end is in range");
96        assert!(!b.contains(0), "the interior gap is excluded");
97    }
98
99    #[test]
100    fn can_stop_past_only_non_wraparound_above_end() {
101        let fwd = ScanTokenBound {
102            start_excl: 10,
103            end_incl: 20,
104        };
105        assert!(!fwd.can_stop_past(20), "at end still in range");
106        assert!(fwd.can_stop_past(21), "past end can stop");
107        let wrap = ScanTokenBound {
108            start_excl: 100,
109            end_incl: -100,
110        };
111        assert!(
112            !wrap.can_stop_past(i64::MAX),
113            "a wraparound range never early-stops a forward walk"
114        );
115    }
116
117    /// #3634: the `start != end` guard the flag-carrying form needed is SUBSUMED
118    /// by the derived predicate, so the full ring must still never early-stop a
119    /// forward walk — an early stop there would silently truncate the scan.
120    #[test]
121    fn a_full_ring_never_early_stops() {
122        for s in [i64::MIN, -1, 0, 1, i64::MAX] {
123            let full = ScanTokenBound {
124                start_excl: s,
125                end_incl: s,
126            };
127            for t in [
128                i64::MIN,
129                s.saturating_sub(1),
130                s,
131                s.saturating_add(1),
132                i64::MAX,
133            ] {
134                assert!(
135                    !full.can_stop_past(t),
136                    "a full ring ({s}, {s}] admits tokens above {t}, so the walk \
137                     must not stop"
138                );
139            }
140        }
141    }
142
143    /// The derived form must agree with the flag-carrying form wherever the flag
144    /// was CONSISTENT with its endpoints — i.e. every shape a live caller built.
145    #[test]
146    fn can_stop_past_matches_the_consistent_flag_form() {
147        for (start_excl, end_incl) in [(10, 20), (100, -100), (5, 5), (i64::MIN, i64::MAX)] {
148            let b = ScanTokenBound {
149                start_excl,
150                end_incl,
151            };
152            for t in [i64::MIN, -101, -100, 0, 20, 21, 100, i64::MAX] {
153                let consistent_flag = start_excl > end_incl;
154                let previous = !consistent_flag && start_excl != end_incl && t > end_incl;
155                assert_eq!(
156                    b.can_stop_past(t),
157                    previous,
158                    "({start_excl}, {end_incl}] at {t}"
159                );
160            }
161        }
162    }
