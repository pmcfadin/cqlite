1//! BTI ("da") POINT-READ path (issue #1599 / G3 split of `bti.rs`, campsite #1116).
2//!
3//! Extracted from `bti.rs`: the trie-resolved single-partition point lookup
4//! (`bti_point_lookup`) and its chunk-targeted decode machinery — offset resolution
5//! via the one [`locate`](super::super::SSTableReader::locate) façade, the
6//! positional (`read_at`) chunk/section fetch, the growing-window single-partition
7//! parse, and the prefix-collision key guard. The whole-Data.db BTI scan, the
8//! single-partition seek, and clustering-slice narrowing stay in `bti.rs`.

25impl SSTableReader {
26    /// BTI ("da") point lookup: resolve a partition key via the Partitions.db
27    /// trie, decode the partition at the resolved offset, and return its row
28    /// `Value` (issue #831).
29    ///
30    /// Correctness invariants (see issue #831 / #755):
31    ///
32    /// - **Offset domain**: the trie returns an *uncompressed* Data.db offset, so
33    ///   we decode the partition out of the DECOMPRESSED data section, never via
34    ///   `read_value_at_offset`/`get_cached_data` (which seek raw file bytes).
35    /// - **Own decompression**: `requires_chunk_stitching()` is `false` for BTI,
36    ///   so this path decompresses the chunk-compressed Data.db itself via the
37    ///   reader's CompressionInfo + compression_reader. Because the trie already
38    ///   resolved the EXACT uncompressed offset of the target partition, this only
39    ///   decompresses the chunk that contains that offset and continues forward
40    ///   chunk-by-chunk ONLY until the target partition is fully parsed — it never
41    ///   decompresses earlier chunks or the rest of the file (issue #831 perf
42    ///   finding). The whole-section `point_read_whole_section` fallback (one
43    ///   positioned read of the entire data section) is used only when chunk
44    ///   targeting is impossible (no/zero `chunk_length`).
45    /// - **Prefix-collision guard**: the trie may return a candidate for a
46    ///   prefix-colliding key, so the decoded partition key is verified to equal
47    ///   the queried key before any row is returned.
48    ///
49    /// `fully_qualified_match` is the authoritative resolution-mode signal threaded
50    /// from the manager's `resolve_reader_list` (issue #1321, mirroring #1284's
51    /// seek path): `true` iff the query's fully-qualified `keyspace.table` key
52    /// matched this reader's map slot EXACTLY (or the query was unqualified),
53    /// `false` iff a fully-qualified query reached this reader via the bare-name
54    /// fallback. It gates the per-row table-consistency guard in
55    /// `bti_decompress_and_parse_target`: an exact FQ match may relax across a
56    /// header-keyspace divergence, while a fallback keeps strict keyspace matching.
57    ///
58    /// Returns `(row, oracle_pruned)` (issue #2163): `oracle_pruned` is `true`
59    /// ONLY for the Step 1 trie-miss branch — this SSTable was excluded from the
60    /// read by the presence oracle BEFORE any decode was attempted. Every other
61    /// `None` outcome (a decoded-but-non-matching prefix-collision candidate, a
62    /// tombstone) is NOT an oracle exclusion and reports `oracle_pruned = false`,
63    /// so the caller (`get_with_resolution`) emits `cqlite.read.sstables_pruned`
64    /// and runs the opt-in false-negative verification ONLY for a genuine
65    /// definitive negative — never for a row this reader actually examined.
66    pub(super) async fn bti_point_lookup(
67        &self,
68        table_id: &TableId,
69        key: &RowKey,
70        fully_qualified_match: bool,
71    ) -> Result<(Option<ScanRow>, bool)> {
72        // 1. Resolve the uncompressed Data.db offset via the one `locate` façade
73        //    (issue #1599 / G3). For a BTI reader `locate` runs the C5 step (a no-op
74        //    — BTI has no Summary bound, so nothing is short-circuited or recorded)
75        //    then the `Partitions.db` trie, which is the AUTHORITATIVE presence
76        //    oracle and emits the single READ_BLOOM_CHECKS; the bloom filter is never
77        //    consulted. A trie miss is definitive absence. BTI records no size, so
78        //    `locate` returns `(offset, 0)` and we use only the offset.
79        let offset = match self.locate(key.as_bytes()).await? {
80            Some((off, _size)) => off as usize,
81            // Not in this SSTable (authoritative trie miss) — an oracle exclusion.
82            None => return Ok((None, true)),
83        };
84
85        // 2. Obtain a DECOMPRESSED window that contains the target partition.
86        //
87        //    `window_base` is the uncompressed offset of the window's first byte
88        //    and `window` holds the decompressed bytes from there onward. The
89        //    target partition starts at `offset - window_base` inside `window`
90        //    (INVARIANT 1: the trie offset indexes the uncompressed data section).
91        //
92        //    For the chunk-targeted path the window starts at the chunk that
93        //    contains `offset` (so `window_base = target_chunk * chunk_length`);
94        //    for the whole-section fallback the window starts at offset 0
95        //    (`window_base = 0`). Either way the parse below uses the same
96        //    `within = offset - window_base` index.
97        let schema_opt = self.get_table_schema(None);
98        let parser = self.build_v5_parser(true);
99
100        let found = self
101            .bti_decompress_and_parse_target(
102                offset,
103                key,
104                table_id,
105                fully_qualified_match,
106                schema_opt.as_ref(),
107                &parser,
108            )
109            .await?;
110
111        // Steps 2+ decoded (or attempted to decode) actual partition data — the
112        // trie admitted this SSTable, so any `None` here (prefix-collision
113        // candidate that didn't match, or a tombstone) is NOT an oracle exclusion.
114        match found {
115            Some(value) => {
116                if !self.filter_tombstone(&value) {
117                    return Ok((None, false));
118                }
119                Ok((Some(value), false))
120            }
121            None => Ok((None, false)),
122        }
123    }
124
125    /// Compute the chunk that contains uncompressed `offset`, the uncompressed
126    /// offset of that chunk's start, and the within-chunk index — given the
127    /// CompressionInfo `chunk_length` (issue #831).
128    ///
129    /// Returns `(target_chunk, window_base, within)` where
130    /// `window_base = target_chunk * chunk_length` and `within = offset - window_base`.
131    /// Pure arithmetic so it can be unit-tested independently of any I/O.
132    #[inline]
133    fn bti_chunk_target(offset: usize, chunk_length: usize) -> (usize, usize, usize) {
134        let target_chunk = offset / chunk_length;
135        let window_base = target_chunk * chunk_length;
136        let within = offset - window_base;
137        (target_chunk, window_base, within)
138    }

140    /// Positional (`pread`) read of the ENTIRE uncompressed data section for the
141    /// point-read whole-section fallback (issue #1573, C2) — used when chunk
142    /// targeting is impossible (no/zero `chunk_length`, i.e. an uncompressed BTI or
143    /// nb-without-CompressionInfo Data.db). Reads `[header_size, file_size)` in one
144    /// positioned read and, when a `CRC.db` is present (uncompressed BIG), verifies
145    /// the covering chunks BEFORE the bytes are parsed — preserving the CRC-then-use
146    /// ordering the cursor path enforced via `read_uncompressed_data_block`.
147    pub(super) async fn point_read_whole_section(&self) -> Result<Vec<u8>> {
148        let header_size = self.calculate_header_size() as u64;
149        // Authoritative file length straight from the positional source (== the
150        // reader's `file_size`; using the source keeps the read self-consistent).
151        let end = self.point_source.len();
152        let len = end.saturating_sub(header_size);
153        let mut whole = vec![0u8; len as usize];
154        if len > 0 {
155            // Read the section in BOUNDED windows rather than one section-sized
156            // `read_exact_at`. A `DirectReadAt` backend allocates a per-call
157            // aligned bounce buffer as large as the requested range, so a single
158            // whole-section read would transiently ~double resident memory vs the
159            // <128MB target for a large section. Windowing caps the bounce buffer
160            // at ~`WHOLE_SECTION_READ_WINDOW` regardless of backend (issue #1573
161            // roborev); `whole` itself is the returned data and is unavoidable.
162            const WHOLE_SECTION_READ_WINDOW: usize = 1 << 20; // 1 MiB
163            let mut filled = 0usize;
164            while filled < whole.len() {
165                let win_end = (filled + WHOLE_SECTION_READ_WINDOW).min(whole.len());
166                self.point_source
167                    .read_exact_at(header_size + filled as u64, &mut whole[filled..win_end])?;
168                filled = win_end;
169            }
170        }
171        // CRC-verify the covering chunk(s) when a CRC.db is present (no-op for BTI
172        // and compressed tables) BEFORE returning the bytes. The section is already
173        // resident in `whole`, so verify against those in-memory bytes rather than
174        // re-reading the identical range from `point_source` — the section is
175        // transferred from disk EXACTLY ONCE (issue #1573 roborev), preserving the
176        // CRC-before-use ordering and the CRC algorithm unchanged.
177        if self.crc_reader.is_some() {
178            // POINT intent (issue #2876): the only I/O the verifier can still need
179            // is the straddling-chunk re-read below `header_size`, and this is a
180            // point path, so it stays on the advised `MADV_RANDOM` mapping (#2210) —
181            // the same plane the section itself was just read from.
182            let point_source = self.point_source.clone();
183            self.verify_uncompressed_section_in_buffer(point_source.as_ref(), header_size, &whole)
184                .await?;
185        }
186        Ok(super::super::chunk_source::counted_raw_chunk(whole, None))
187    }

189    /// Decompress only the chunk(s) needed to fully parse the target partition at
190    /// uncompressed `offset`, then parse and return its row value (issue #831).
191    ///
192    /// Chunk targeting (the fast path): when `CompressionInfo` with a non-zero
193    /// `chunk_length` is present, the chunk containing `offset` is
194    /// `target_chunk = offset / chunk_length`; we seek that chunk via its
195    /// `chunk_offsets` entry, set the cursor's chunk index to `target_chunk`, then
196    /// decompress forward chunk-by-chunk, appending each into `window`. After each
197    /// appended chunk we attempt to parse the FIRST partition at `window[within..]`
198    /// (`within = offset % chunk_length`). The stop condition (correctness-critical
199    /// — never return a truncated parse):
200    ///   - parse returns `Ok` AND the emit closure fired (a COMPLETE partition was
201    ///     decoded) -> stop and return what the closure captured;
202    ///   - parse returns `Err` (buffer truncated mid-partition) OR the closure
203    ///     never fired -> append the next chunk and retry;
204    ///   - `read_next_block()` returns `None` (EOF) and still not parsed -> stop
205    ///     (the caller treats `None` as "absent", matching prior behaviour).
206    ///
207    /// Fallbacks (preserve prior behaviour exactly): when `compression_info` is
208    /// `None` (uncompressed BTI Data.db) or `chunk_length` is 0/absent, this
209    /// reads the WHOLE section in one positioned read via
210    /// `point_read_whole_section` (`window_base = 0`, CRC-verified when a CRC.db
211    /// is present) and runs the same single-partition parse.
212    ///
213    /// Chunks are fetched with positioned (`read_at`) reads on the shared
214    /// `point_source` — no per-lookup `open(2)`, no `ScanCursor`, and no mutex.
215    /// `chunk_index` is a plain local (a lookup is single-threaded within
216    /// itself), so concurrent lookups run in parallel without serialization;
217    /// safety comes from `read_at` taking `&self` (issue #1573, superseding the
218    /// per-scan-cursor approach of issue #815).
219    pub(super) async fn bti_decompress_and_parse_target(
220        &self,
221        offset: usize,
222        key: &RowKey,
223        table_id: &TableId,
224        // Issue #1321: authoritative resolution mode (see `bti_point_lookup`).
225        // Gates the per-row table-consistency guard exactly like the seek path
226        // (#1284): an EXACT fully-qualified resolution may accept rows across a
227        // benign header-keyspace divergence on a consistent table name, while a
228        // fully-qualified query resolved via the bare-name fallback keeps STRICT
229        // keyspace matching (no wrong-keyspace rows).
230        fully_qualified_match: bool,
231        schema_opt: Option<&crate::schema::TableSchema>,
232        parser: &crate::storage::sstable::reader::parsing::V5CompressedLegacyParser,
233    ) -> Result<Option<ScanRow>> {
234        use crate::storage::sstable::compression::Compression;
235
236        // Issue #1573 (C2): the point path fetches chunks via positioned reads on
237        // the shared `point_source` — no per-lookup `open(2)`, no cursor, no mutex.
238        // `chunk_index` is a plain local (a lookup is single-threaded within
239        // itself); concurrency safety comes from `read_at` being `&self`.
240
241        // Determine the chunk-targeting parameters. `chunk_length == 0` (or no
242        // CompressionInfo) means we cannot chunk-target -> whole-section fallback.
243        let chunk_length = self
244            .compression_info
245            .as_ref()
246            .map(|ci| ci.chunk_length as usize)
247            .filter(|&len| len > 0);
248
249        let mut chunk_index = 0usize;
250        let (target_chunk, window_base, mut window) = match chunk_length {
251            Some(len) => {
252                let (target_chunk, window_base, _within) = Self::bti_chunk_target(offset, len);
253                // Positioned reads resolve their own offset from the chunk index, so
254                // no pre-seek is needed — just start at `target_chunk`.
255                chunk_index = target_chunk;
256                (target_chunk, window_base, Vec::<u8>::new())
257            }
258            None => {
259                // Whole-section fallback (uncompressed BTI, or chunk_length absent/0):
260                // one positioned read of the whole data section, CRC-verified when a
261                // CRC.db is present (see `point_read_whole_section`).
262                let whole = self.point_read_whole_section().await?;
263                (0usize, 0usize, whole)
264            }
265        };
266
267        // `within` is the start of the target partition inside `window`.
268        if offset < window_base {
269            return Err(Error::corruption(format!(
270                "BTI point lookup: resolved offset {} precedes window base {} (chunk {})",
271                offset, window_base, target_chunk
272            )));
273        }
274        let within = offset - window_base;
275
276        // For the chunk-targeted path we still need to populate `window`. For the
277        // whole-section fallback `window` is already complete.
278        let chunk_targeted = chunk_length.is_some();
279
280        // #3721 (job 80): last parse failure that cannot mean absence.
281        let mut undecodable: Option<Error> = None;
282
283        loop {
284            // If chunk-targeted, append the next chunk before each parse attempt
285            // (the whole-section fallback already has all bytes in `window`).
286            if chunk_targeted {
287                // Single decode plane (issue #1598, G2): positioned read → CRC →
288                // decompress → B1 cache via `ChunkSource`. A shared-cache hit skips
289                // the decompressor; a miss reads + CRC-checks (guardrail #1411) the
290                // compressed bytes before decompressing them (issue #1567 cache key
291                // = the ABSOLUTE chunk index in the NS_BTI_CHUNK namespace).
292                let compression_opt = self
293                    .compression_reader
294                    .as_ref()
295                    .map(|cr| Compression::new(*cr.algorithm()))
296                    .transpose()?;
297                let comp_info = self.compression_info.as_deref().ok_or_else(|| {
298                    Error::corruption(
299                        "BTI chunk_targeted path requires CompressionInfo but it is absent",
300                    )
301                })?;
302                let chunk_source = super::super::chunk_source::ChunkSource::new(
303                    self.point_source.as_ref(),
304                    comp_info,
305                    compression_opt.as_ref(),
306                    &self.chunk_cache,
307                    self.stats.file_size,
308                    0, // NB/BTI: chunk offsets are absolute from Data.db byte 0
309                    super::NS_BTI_CHUNK,
310                    self.chunk_cache_id,
311                );
312                match chunk_source.chunk(chunk_index)? {
313                    Some(decompressed_chunk) => {
314                        chunk_index += 1;
315                        window.extend_from_slice(&decompressed_chunk);
316                    }
317                    None => {
318                        // EOF (#3721 job 80: a remembered failure is not absence).
319                        return point_read_absence_or_remembered(&mut undecodable);
320                    }
321                }
322            }
323
324            // Need at least the partition header to attempt a match.
325            if within >= window.len() {
326                if chunk_targeted {
327                    // Not enough bytes yet; pull the next chunk.
328                    continue;
329                }
330                // Whole-section window can't grow: offset is past the data.
331                return Err(Error::corruption(format!(
332                    "BTI trie resolved Data.db offset {} beyond decompressed data section ({} bytes)",
333                    offset,
334                    window.len()
335                )));
336            }
337
338            // INVARIANT 3 + chunk-straddle gate. The parse/pull/absent decision is
339            // factored into the pure `bti_lookup_step` so the chunk-straddle control
340            // flow is unit-testable without a multi-chunk fixture (issue #831 review):
341            // when the header/key prefix is not yet fully buffered we must NOT invoke
342            // the parser on a truncated header (it can skip bytes and emit a later
343            // false-positive entry), and must read the next chunk first.
344            let key_available =
345                Self::bti_partition_key_bytes_available(&window, within, key.as_bytes());
346            let key_matches =
347                key_available && self.bti_partition_key_matches(&window, within, key.as_bytes());
348            match bti_lookup_step(key_available, key_matches, chunk_targeted) {
349                BtiLookupStep::Parse => { /* full key prefix buffered and matches */ }
350                BtiLookupStep::PullNextChunk => continue,
351                BtiLookupStep::Absent => {
352                    if key_available {
353                        debug!(
354                            "BTI trie candidate at offset {} did not match queried key \
355                             (prefix collision); treating as absent",
356                            offset
357                        );
358                    }
359                    return Ok(None);
360                }
361            }
362
363            // Attempt to parse the FIRST partition at window[within..]. The parser
364            // detects the next partition boundary / 0x01 end-of-partition marker and
365            // stops; we break after the first emitted entry. A COMPLETE partition
366            // means: parse returned Ok AND the closure emitted OUR queried partition
367            // key (see `emitted_our_key` below).
368            let mut found: Option<ScanRow> = None;
369            let mut emitted_our_key = false;
370            // #3782: a chunk-covering WINDOW — a truncated tail here is this
371            // reader's straddle signal ("pull the next chunk"), not corruption.
372            let parse_result = parser.parse_block_emit(
373                &window[within..],
374                // Issue #3782 x #3721: the extent MUST come from the same finality state the
375                // guard below uses (`chunk_targeted`), or the two mechanisms cancel.
376                //
377                // `Window` unconditionally is what broke `issue_3721_bti_point_read_absence`:
378                // on the WHOLE-SECTION path every byte is already present, but a tolerant
379                // extent makes the parse SWALLOW the decode failure and return `Ok(())`, so
380                // `point_read_remember_or_bail` remembers nothing and
381                // `point_read_absence_or_remembered` answers `Ok(None)` — job 80's phantom
382                // ABSENCE, reintroduced through an AUTO-MERGED line, not a resolved conflict.
383                //
384                // Chunk-targeted, more bytes can arrive, so `Window` is right there and the
385                // #1572 straddle retry stays intact.
386                if chunk_targeted {
387                    BufferExtent::Window
388                } else {
389                    BufferExtent::Complete
390                },
391                schema_opt,
392                self,
393                |(tid, entry_key, entry_value)| {
394                    // Did the parser decode OUR queried partition key? This is the
395                    // authoritative "the partition was fully buffered and parsed
396                    // cleanly" signal: `bti_partition_key_matches` already confirmed
397                    // our key's raw bytes sit at `within`, so a COMPLETE parse must
398                    // re-decode exactly those bytes. A DIFFERENT decoded key means the
399                    // window tail was truncated mid-partition and the parser resynced
400                    // onto garbage (the chunk-straddle case handled below).
401                    let is_our_key = entry_key.as_bytes() == key.as_bytes();
402                    if is_our_key {
403                        emitted_our_key = true;
404                    }
405                    // Verify BOTH the emitted table id is consistent with the
406                    // queried table AND the parser-decoded partition key equals the
407                    // queried key. The table check is resolution-mode-aware (issue
408                    // #1321, mirroring the seek path #1284): an EXACT fully-qualified
409                    // resolution accepts a consistent table name across a benign
410                    // header-keyspace divergence; a fully-qualified query resolved via
411                    // the bare-name fallback keeps STRICT keyspace matching so it can
412                    // never return another keyspace's same-named rows. A genuinely
413                    // different table name is always rejected (issue #831).
414                    if table_header_consistent_for_seek(&tid, table_id, fully_qualified_match)
415                        && is_our_key
416                    {
417                        found = Some(entry_value);
418                    }
419                    Ok(std::ops::ControlFlow::Break(()))
420                },
421            );
422
423            match parse_result {
424                Ok(()) if emitted_our_key => {
425                    // The partition at `within` parsed COMPLETELY: our queried key was
426                    // decoded from a fully-buffered window. Return the row when the
427                    // table guard also passed (`found`), else `None` — a genuine
428                    // soft-miss (schema-unavailable / benign table-guard rejection)
429                    // the caller resolves via `scan_for_key`.
430                    return Ok(found);
431                }
432                _ => {
433                    // The parse did NOT decode our key: an `Err`, the closure never
434                    // firing, or a FOREIGN key. Chunk-targeted, the latter two are the
435                    // #1572 STRADDLE case, answered by the next chunk (absence would
436                    // wrongly fall back to a whole-file `scan_for_key`).
437                    // #3721 (job 80): an `Err` is a THIRD state and a decode failure
438                    // is not absence — see `point_read_remember_or_bail`, which also
439                    // says why the straddle retry stays exactly as it was.
440                    point_read_remember_or_bail(parse_result, chunk_targeted, &mut undecodable)?;
441                    if chunk_targeted {
442                        continue;
443                    }
444                    // Every byte present, and no absence-ruling-out failure: prior
445                    // behaviour (closure never fired, or a foreign key decoded).
446                    return Ok(None);
447                }
448            }
449        }
450    }

452    /// Collect-ALL-rows variant of [`bti_decompress_and_parse_target`] for the
453    /// within-SSTable seek (`scan_single_partition`, Issue #953 / #951).
454    ///
455    /// [`bti_decompress_and_parse_target`] stops after the FIRST emitted row of the
456    /// decoded partition — correct for a `get()` point lookup that returns a single
457    /// `Value`, but WRONG for `scan_partition`, which must hand the query layer
458    /// EVERY clustering row of the partition so it can apply clustering predicates.
459    /// A `WHERE pk = ?` over a table with multiple clustering rows per partition
460    /// would otherwise drop every row after the first whenever the seek succeeds
461    /// (the original #953 bug — see the multi-row regression test).
462    ///
463    /// This variant reuses the identical window-building (chunk targeting or
464    /// whole-section fallback), the identical prefix-collision key re-verification,
465    /// and the identical `parse_block_emit` decode that the user-facing scan path
466    /// runs — but instead of breaking after the first row it COLLECTS every row the
467    /// parser emits for the ONE target partition. The emit closure keeps each
468    /// `Value` whose decoded key equals the queried key (and whose table id
469    /// matches) and `Break`s the instant the parser emits a row with a DIFFERENT
470    /// partition key.
471    ///
472    /// Bounding the decompression window (Issue #953 / #951 MEDIUM fix). The seek
473    /// must materialize ONLY the chunks covering the target partition — never
474    /// stitch to EOF (for a head-of-file point lookup on a large SSTable that would
475    /// decompress nearly the whole `Data.db`, full-table I/O for one partition).
476    /// The bound is AUTHORITATIVE, not a heuristic boundary scan:
477    ///
478    ///   - **`decode_end_bound = Some(end)`** — the caller resolved the SUCCESSOR
479    ///     partition's uncompressed start offset (next trie/index entry). The
480    ///     target partition occupies `[offset, end)`, so we pull chunks only until
481    ///     `window.len() >= end - window_base` (or EOF) and then parse ONCE over a
482    ///     window that fully contains the partition. Because the WHOLE `[offset,
483    ///     end)` extent is decompressed before parsing, a row/cell that spans
484    ///     multiple compression chunks is present in full — no mid-stream
485    ///     truncation, no boundary guessing. This is the exact bound for every
486    ///     non-last partition in both BTI (`da`) and BIG (`nb`).
487    ///
488    ///   - **`decode_end_bound = None`** — `offset` is the LAST partition (no successor).
489    ///     The end is then the authoritative data-section length
490    ///     (`CompressionInfo.data_length`); we buffer to that length (or EOF) and
491    ///     parse once. If that length is unavailable (no usable `CompressionInfo`),
492    ///     we CANNOT bound the last partition authoritatively, so we return
493    ///     `Ok(None)` and the caller falls back to the safe full-scan + retain path
494    ///     (correctness over optimization). The previous row-count *stability
495    ///     guard* — itself a heuristic that could falsely accept a next-partition
496    ///     boundary while the target partition was incomplete (a single large
497    ///     multi-chunk cell, static/range-marker regions, or a truncated tail
498    ///     parsed as garbage headers) — has been REMOVED entirely.
499    ///
500    /// The whole-section fallback (uncompressed BTI) already has every byte so its
501    /// first parse holds the whole partition regardless of the bound. This yields
502    /// byte-for-byte the same rows as the full-scan path filtered down to
503    /// `partition_key`.
504    ///
505    /// Bounding the PARSE input (Issue #3890). Bounding the DECOMPRESSION is not
506    /// the same as bounding the PARSE: chunks are fixed-size, so the buffered
507    /// window overruns the partition's end by up to one chunk, and the parser this
508    /// hands off to is a MULTI-partition block walker. Both call sites below
509    /// therefore pass the partition's authoritative exclusive end down to
510    /// [`bti_collect_partition_rows`](Self::bti_collect_partition_rows), which
511    /// slices the parser's input at it. That end is `partition_end_bound` — the
512    /// UN-narrowed successor offset — never `decode_end_bound`, which an engaged
513    /// #954 clustering slice tightens to a row-index block extent INSIDE the
514    /// partition (bounding the parse there would truncate a row that starts just
515    /// before the block end; the clamp to the buffered window makes the bound a
516    /// no-op in that case, preserving the #954 behaviour exactly). See that
517    /// function's doc comment for what the overrun actually caused.
518    ///
519    /// Returns:
520    /// - `Ok(Some(rows))` — the partition's rows (empty when the trie/index
521    ///   candidate was a prefix collision for an absent key). The caller wraps each
522    ///   in a `(RowKey, ScanRow)` and applies the same tombstone suppression the scan
523    ///   path applies.
524    /// - `Ok(None)` — could not bound the (last) partition authoritatively; the
525    ///   caller must fall back to a full scan + retain.
526    #[cfg(not(feature = "tombstones"))]
527    pub(super) async fn bti_decompress_and_parse_target_all(
528        &self,
529        offset: usize,
530        // The end of the byte extent to DECOMPRESS: the successor partition's
531        // uncompressed start, TIGHTENED by an engaged #954 clustering slice to that
532        // slice's row-index block extent. `None` = last partition.
533        decode_end_bound: Option<usize>,
534        // Issue #3890: the end of the target PARTITION itself — the successor
535        // offset BEFORE any clustering-slice narrowing, so it is the authoritative
536        // extent to bound the PARSE by. `None` = last partition (no successor); the
537        // data-section length is then used, and if that is also unavailable no bound
538        // is invented (#28 no-heuristics) and today's whole-window parse stands.
539        partition_end_bound: Option<usize>,
540        // Issue #954: when `Some((start_rel, end_rel))`, bound the partition's
541        // row-body parse to that within-partition byte window (relative to the
542        // partition start) so only the clustering slice's row-index block(s) are
543        // decoded. `None` decodes the whole partition (the #953 behaviour).
544        row_body_window: Option<(usize, usize)>,
545        key: &RowKey,
546        table_id: &TableId,
547        // See `bti_collect_partition_rows`: `true` iff the manager resolved this
548        // reader by an exact fully-qualified `keyspace.table` match (or the query
549        // was unqualified). Threaded into the seek table-consistency guard (#1284).
550        fully_qualified_match: bool,
551        schema_opt: Option<&crate::schema::TableSchema>,
552        parser: &crate::storage::sstable::reader::parsing::V5CompressedLegacyParser,
553    ) -> Result<Option<Vec<ScanRow>>> {
554        // Issue #1573 (C2): positioned reads on the shared `point_source` — no
555        // per-lookup `open(2)`, no cursor, no mutex. `chunk_index` is a plain local.
556
557        // Determine the chunk-targeting parameters. `chunk_length == 0` (or no
558        // CompressionInfo) means we cannot chunk-target -> whole-section fallback.
559        let chunk_length = self
560            .compression_info
561            .as_ref()
562            .map(|ci| ci.chunk_length as usize)
563            .filter(|&len| len > 0);
564
565        let mut chunk_index = 0usize;
566        let (target_chunk, window_base, mut window) = match chunk_length {
567            Some(len) => {
568                let (target_chunk, window_base, _within) = Self::bti_chunk_target(offset, len);
569                // Positioned reads resolve their own offset from the chunk index.
570                chunk_index = target_chunk;
571                (target_chunk, window_base, Vec::<u8>::new())
572            }
573            None => {
574                // Whole-section fallback (uncompressed BTI, or chunk_length absent/0):
575                // one positioned read of the whole data section, CRC-verified when a
576                // CRC.db is present (see `point_read_whole_section`).
577                let whole = self.point_read_whole_section().await?;
578                (0usize, 0usize, whole)
579            }
580        };
581
582        if offset < window_base {
583            return Err(Error::corruption(format!(
584                "BTI single-partition seek: resolved offset {} precedes window base {} (chunk {})",
585                offset, window_base, target_chunk
586            )));
587        }
588        let within = offset - window_base;
589        let chunk_targeted = chunk_length.is_some();
590
591        if chunk_targeted {
592            // Resolve the AUTHORITATIVE exclusive end of the target partition in
593            // the UNCOMPRESSED offset domain. Non-last partitions are bounded by
594            // the successor partition's start (`end_bound`); the LAST partition is
595            // bounded by the data-section length. When NEITHER is known we cannot
596            // bound the last partition without re-introducing a heuristic, so we
597            // return `Ok(None)` and let the caller fall back to a full scan.
598            let data_length = self
599                .compression_info
600                .as_ref()
601                .map(|ci| ci.data_length as usize)
602                .filter(|&len| len > offset);
603            let end_offset = match decode_end_bound.or(data_length) {
604                Some(end) => end,
605                None => {
606                    debug!(
607                        "BTI single-partition seek: last partition at offset {} has no \
608                         authoritative end (no successor, no usable data_length); falling \
609                         back to full scan",
610                        offset
611                    );
612                    return Ok(None);
613                }
614            };
615            // Issue #3890: the PARSE bound is the partition's OWN authoritative
616            // extent, which equals `decode_end_bound` only when no #954 clustering
617            // slice narrowed it. Same domain shift as `needed` below.
618            let partition_end_within = partition_end_bound
619                .or(data_length)
620                .map(|end| end.saturating_sub(window_base));
621
622            // Step 1: buffer enough chunks to expose the partition header, then run
623            // the prefix-collision / chunk-straddle gate. This bails out cheaply
624            // (without decompressing the rest of the partition) when the trie/index
625            // candidate is a prefix collision for an absent key.
626            loop {
627                // Pull a chunk if the header is not yet (fully) buffered.
628                if within + 2 > window.len()
629                    || !Self::bti_partition_key_bytes_available(&window, within, key.as_bytes())
630                {
631                    match self
632                        .bti_pull_decompressed_chunk(&mut chunk_index, &mut window)
633                        .await?
634                    {
635                        true => continue, // chunk appended; re-check the header
636                        false => {
637                            // EOF before the header is buffered: nothing decodable
638                            // at the resolved offset.
639                            return Ok(Some(Vec::new()));
640                        }
641                    }
642                }
643
644                let key_matches = self.bti_partition_key_matches(&window, within, key.as_bytes());
645                if !key_matches {
646                    debug!(
647                        "BTI seek candidate at offset {} did not match queried key \
648                         (prefix collision); treating as absent",
649                        offset
650                    );
651                    return Ok(Some(Vec::new()));
652                }
653                break; // header buffered AND key matches
654            }
655
656            // Step 2: buffer EXACTLY the chunks covering `[offset, end_offset)` —
657            // never stitch to EOF (the #953 MEDIUM finding: a head-of-file lookup
658            // would otherwise decompress the whole file). `end_offset` is in the
659            // same uncompressed-offset domain as `window_base + window.len()`, so
660            // the window holds the whole partition once `window.len()` reaches
661            // `end_offset - window_base` (or EOF — a stale end never reads past
662            // EOF). Decompressing the FULL extent before parsing means a row/cell
663            // that spans multiple compression chunks is present in full, so the
664            // single parse below collects every target row without truncation.
665            let needed = end_offset.saturating_sub(window_base);
666            while window.len() < needed {
667                if !self
668                    .bti_pull_decompressed_chunk(&mut chunk_index, &mut window)
669                    .await?
670                {
671                    break; // EOF: window holds all available bytes.
672                }
673            }
674            return self
675                .bti_collect_partition_rows(
676                    &window,
677                    within,
678                    partition_end_within,
679                    row_body_window,
680                    key,
681                    table_id,
682                    fully_qualified_match,
683                    schema_opt,
684                    parser,
685                )
686                .map(|(rows, _complete)| Some(rows));
687        }
688
689        // Whole-section fallback (uncompressed BTI): every byte is already present,
690        // so the first parse is authoritative.
691        if within >= window.len() {
692            return Err(Error::corruption(format!(
693                "BTI trie resolved Data.db offset {} beyond decompressed data section ({} bytes)",
694                offset,
695                window.len()
696            )));
697        }
698        // Issue #3890: `window_base` is 0 on this arm (the whole data section was
699        // read), so `partition_end_bound` is already a `window` index. When it is
700        // absent (the LAST partition) the data section's own length IS the
701        // authoritative end and `window` holds exactly that section, so `None` —
702        // parse to `window.len()` — is authoritative here too; no bound is invented.
703        self.bti_collect_partition_rows(
704            &window,
705            within,
706            partition_end_bound,
707            row_body_window,
708            key,
709            table_id,
710            fully_qualified_match,
711            schema_opt,
712            parser,
713        )
714        .map(|(rows, _complete)| Some(rows))
715    }
716
717    /// Read the next compressed chunk from `cursor`, decompress it (if the reader
718    /// has a compression algorithm), and append the decompressed bytes to
719    /// `window`. Returns `true` when a chunk was appended, `false` at EOF.
720    ///
721    /// Shared by the chunk-targeted seek so the header-buffering and
722    /// partition-bounding loops use one decompression code path; each call bumps
723    /// `work_counters::chunks_decompressed` so a test can prove the seek bounded
724    /// its decompression to the target partition's chunk span (Issue #953/#951).
725    ///
726    /// Issue #1750 (regression fix): this now fetches through the CACHING
727    /// [`ChunkSource::chunk`] — read → CRC → decompress → B1 cache — exactly like
728    /// the legacy `get()` point-lookup path (`bti_decompress_and_parse_target`)
729    /// did. Before #1750 retired the `is_simple_id_lookup` fork, a `WHERE pk = ?`
730    /// point read routed to `get()` and populated/served the shared B1
731    /// [`DecompressedChunkCache`]; rerouting it to the modern seek path made the
732    /// seek's chunk fetch UNCACHED (`decompress_only`), so a repeated point read
733    /// re-decompressed every chunk and `Database::stats().memory_stats` reported a
734    /// structural zero B1 hit rate (the `dead_cache_delete` regression). Routing
735    /// through `chunk()` keyed in the shared `NS_BTI_CHUNK` namespace makes a warm
736    /// repeat read a refcount-bump cache HIT — restoring the pre-#1750 caching
737    /// behavior — while remaining the SAME single decode plane (issue #1598, G2).
738    ///
739    /// `work_counters::chunks_decompressed` is bumped once per chunk the seek
740    /// materializes into its `window` (a cache miss decompresses; a hit serves the
741    /// resident bytes). The issue_953/#951 bound tests reset the counter per query
742    /// and read each partition's chunks cold within the process, so materialized ==
743    /// decompressed there and the "seek bounded its chunk span" assertions are
744    /// unchanged; the counter still forbids a stitch-to-EOF regression.
745    #[cfg(not(feature = "tombstones"))]
746    async fn bti_pull_decompressed_chunk(
747        &self,
748        chunk_index: &mut usize,
749        window: &mut Vec<u8>,
750    ) -> Result<bool> {
751        use crate::storage::sstable::compression::Compression;
752        // Issue #1573 (C2): positioned chunk fetch — no cursor, no mutex, no
753        // per-lookup open. CRC is verified inside `ChunkSource::chunk` BEFORE
754        // decompression (guardrail #1411).
755        //
756        // Single decode plane (issue #1598, G2) WITH the shared B1 cache (issue
757        // #1750): `ChunkSource::chunk` does read → CRC → decompress → cache, keyed
758        // by the ABSOLUTE chunk index in the NS_BTI_CHUNK namespace — the same key
759        // space the legacy `get()` point read used, so a warm repeat read hits.
760        let compression_opt = self
761            .compression_reader
762            .as_ref()
763            .map(|cr| Compression::new(*cr.algorithm()))
764            .transpose()?;
765        let comp_info = self.compression_info.as_deref().ok_or_else(|| {
766            Error::corruption(
767                "BTI seek chunk-targeted path requires CompressionInfo but it is absent",
768            )
769        })?;
770        let chunk_source = super::super::chunk_source::ChunkSource::new(
771            self.point_source.as_ref(),
772            comp_info,
773            compression_opt.as_ref(),
774            &self.chunk_cache,
775            self.stats.file_size,
776            0, // NB/BTI: chunk offsets are absolute from Data.db byte 0
777            super::NS_BTI_CHUNK,
778            self.chunk_cache_id,
779        );
780        match chunk_source.chunk(*chunk_index)? {
781            Some(decompressed_chunk) => {
782                *chunk_index += 1;
783                // Issue #953/#951: count every chunk the seek materializes so a
784                // bound test can prove the window is bounded to the target
785                // partition's chunk span, not stitched to EOF.
786                super::super::super::work_counters::add_chunk_decompressed();
787                window.extend_from_slice(&decompressed_chunk);
788                Ok(true)
789            }
790            None => Ok(false),
791        }
792    }
793
794    /// Returns true when the `[flags][key_len: u8][key bytes]` prefix at `within`
795    /// is fully present in `window` AND `key_len` equals `expected_key.len()`.
796    ///
797    /// Used by the chunk-targeted BTI lookup to decide whether the INVARIANT-3
798    /// key match can be evaluated yet, or whether more chunk bytes must be pulled
799    /// first (issue #831).
800    fn bti_partition_key_bytes_available(
801        window: &[u8],
802        within: usize,
803        _expected_key: &[u8],
804    ) -> bool {
805        // Need flags + key_len byte first.
806        if within + 2 > window.len() {
807            return false;
808        }
809        let key_len = window[within + 1] as usize;
810        // The declared key bytes must all be buffered. (Whether `key_len` equals
811        // the expected length is decided by the subsequent match check, which
812        // fails fast on a mismatch — here we only require the bytes be present.)
813        within + 2 + key_len <= window.len()
814    }
815
816    /// Verify the on-disk partition-key bytes at `offset` in the decompressed
817    /// data section equal `expected_key` (issue #831, INVARIANT 3).
818    ///
819    /// Reads the `[flags][key_len: u8][key bytes]` prefix. Returns `false` (rather
820    /// than erroring) on any structural mismatch so the caller can treat the trie
821    /// candidate as absent.
822    fn bti_partition_key_matches(
823        &self,
824        decompressed: &[u8],
825        offset: usize,
826        expected_key: &[u8],
827    ) -> bool {
828        // Need at least flags + key_len.
829        if offset + 2 > decompressed.len() {
830            return false;
831        }
832        let key_len = decompressed[offset + 1] as usize;
833        let key_start = offset + 2;
834        let key_end = key_start + key_len;
835        if key_end > decompressed.len() {
836            return false;
837        }
838        &decompressed[key_start..key_end] == expected_key
839    }
840}

846    /// The chunk-index arithmetic must match `CompressionInfo`'s definitions:
847    /// `target_chunk = off / chunk_length`, `window_base = target_chunk *
848    /// chunk_length`, `within = off - window_base` (== `off % chunk_length`).
849    #[test]
850    fn bti_chunk_target_arithmetic() {
851        // Single-chunk case (simple_table fixture shape): chunk_length 16384,
852        // offset 0/63/125 all land in chunk 0 with within == offset.
853        let chunk_length = 16384;
854        for off in [0usize, 63, 125] {
855            let (chunk, base, within) = SSTableReader::bti_chunk_target(off, chunk_length);
856            assert_eq!(chunk, 0, "off {off} must be in chunk 0");
857            assert_eq!(base, 0, "chunk 0 window base must be 0");
858            assert_eq!(within, off, "within must equal offset in chunk 0");
859        }
860
861        // Multi-chunk arithmetic with a small chunk_length to exercise the math.
862        let cl = 100usize;
863        // Exactly on a chunk boundary.
864        assert_eq!(SSTableReader::bti_chunk_target(100, cl), (1, 100, 0));
865        assert_eq!(SSTableReader::bti_chunk_target(200, cl), (2, 200, 0));
866        // Inside chunk 1.
867        assert_eq!(SSTableReader::bti_chunk_target(150, cl), (1, 100, 50));
868        // Just before a boundary.
869        assert_eq!(SSTableReader::bti_chunk_target(99, cl), (0, 0, 99));
870        // Within always equals off % chunk_length, base = chunk * chunk_length.
871        for off in [0usize, 1, 99, 100, 101, 250, 999] {
872            let (chunk, base, within) = SSTableReader::bti_chunk_target(off, cl);
873            assert_eq!(within, off % cl);
874            assert_eq!(base, chunk * cl);
875            assert_eq!(base + within, off);
876        }
877    }

879    /// `bti_partition_key_bytes_available` drives the growing-window stop
880    /// condition: while the `[flags][key_len][key bytes]` prefix is NOT yet fully
881    /// buffered it returns false (the chunk-targeted loop pulls another chunk);
882    /// once the declared key bytes have all arrived it returns true (the
883    /// INVARIANT-3 key match can be evaluated). This is the SYNTHETIC spanning
884    /// test: the key prefix straddles a simulated chunk boundary and the window
885    /// grows one byte at a time across it.
886    ///
887    /// NOTE: a full multi-chunk-spanning parse against a real
888    /// `V5CompressedLegacyParser` has NO real BTI DataOffset fixture — these are
889    /// narrow partitions that fit within a single chunk — so the spanning *parse*
890    /// path is only exercised structurally here via the byte-availability gate
891    /// that decides when a parse may even be attempted. This calls the real
892    /// associated function (no I/O), so a regression in its boundary math is
893    /// caught.
894    #[test]
895    fn bti_partition_key_bytes_available_growing_window() {
896        // Header at within=0: [flags=0x00][key_len=4][k0 k1 k2 k3]. Simulate a
897        // window that grows from 0 bytes up to the full prefix; availability must
898        // flip to true exactly when all 4 declared key bytes are buffered.
899        let expected_key = [0xAA, 0xBB, 0xCC, 0xDD];
900        let within = 0usize;
901        let full = {
902            let mut v = vec![0x00u8, expected_key.len() as u8];
903            v.extend_from_slice(&expected_key);
904            v
905        };
906
907        let avail = |len: usize| {
908            SSTableReader::bti_partition_key_bytes_available(&full[..len], within, &expected_key)
909        };
910
911        // Not enough for flags+key_len yet.
912        assert!(!avail(0));
913        assert!(!avail(1));
914        // flags+key_len present but key bytes not fully buffered.
915        assert!(!avail(2));
916        assert!(!avail(3)); // 1 key byte
917        assert!(!avail(4)); // 2 key bytes
918        assert!(!avail(5)); // 3 key bytes
919                            // All 4 key bytes buffered -> available (boundary fully crossed).
920        assert!(avail(6));
921        assert!(avail(full.len()));
922
923        // A non-zero `within` (target partition not at window start) must use the
924        // same relative math.
925        let mut padded = vec![0x77u8, 0x88];
926        padded.extend_from_slice(&full);
927        assert!(!SSTableReader::bti_partition_key_bytes_available(
928            &padded[..2 + 5],
929            2,
930            &expected_key
931        ));
932        assert!(SSTableReader::bti_partition_key_bytes_available(
933            &padded,
934            2,
935            &expected_key
936        ));
937    }
