1//! Data access methods for SSTableReader.
2//!
3//! This module contains all methods related to reading data from SSTables,
4//! split by read path:
5//! - [`model`] — shared model types and free helpers (clustering slice,
6//!   table-id matching, token-order sorting, the `scan_for_key` counter).
7//! - [`bti`] — BTI ("da") trie-resolved point lookups, single-partition seeks,
8//!   clustering-slice narrowing, and the whole-Data.db BTI scan.
9//! - [`sequential`] — sequential / index-driven range and full scans, the
10//!   `scan_for_key` fallback, cell-metadata scan, and the streaming scan.
11//! - [`compaction`] — timestamp-preserving compaction iteration and the bounded
12//!   streaming compaction driver.
13//!
14//! `mod.rs` keeps the shared entry points and the I/O / parser infrastructure
15//! every read path builds on (`get`, `build_v5_parser`, `new_scan_cursor`,
16//! `read_next_block`, the offset-read helpers, the chunk-stitch helpers, and the
17//! delta-scan bridge), plus the public-surface re-exports.

180    /// Return `true` when Data.db uses the V5CompressedLegacy NB chunked format and
181    /// therefore requires all chunks to be stitched before parsing.
182    ///
183    /// The correct predicate is:
184    ///   data_format == V5CompressedLegacy  AND  is_nb_format()
185    ///
186    /// Rationale:
187    /// - `V5CompressedLegacy` identifies the row serialization format (u16 length
188    ///   prefixes, legacy encoding) used by all Cassandra 5 'nb' SSTables.
189    /// - `is_nb_format()` identifies the chunked-compression read path. It intentionally
190    ///   EXCLUDES `V5_0Uncompressed`, which uses the same row format but stores data as
191    ///   a single contiguous block (no chunk boundaries, no stitching needed).
192    /// - Using `is_compressed` (compression_reader.is_some()) would be wrong for NB
193    ///   format because the per-chunk decompression is handled inside `stitch_and_parse_all_chunks`,
194    ///   and `is_compressed` may differ from `is_nb_format` for edge-case versions.
195    pub(super) fn requires_chunk_stitching(&self) -> bool {
196        let data_format = self.header.cassandra_version.data_format();
197        matches!(data_format, DataFormat::V5CompressedLegacy)
198            && self.header.cassandra_version.is_nb_format()
199    }

201    /// Test-only: return the fully stitched + decompressed Data.db data section
202    /// for a V5CompressedLegacy ("nb") SSTable, or `None` for any other format.
203    ///
204    /// This exposes the exact on-disk bytes (post-decompression) that the
205    /// production scan path parses, so the Issue #1623 corpus-differential test
206    /// can locate REAL Cassandra `writeUnsignedVInt` length prefixes and route
207    /// those literal bytes through [`crate::parser::vint::parse_vint_length`].
208    /// It replicates the seek + stitch preamble of `sequential_scan` exactly.
209    #[cfg(test)]
210    pub(crate) async fn stitched_data_section_for_tests(&self) -> Result<Option<Vec<u8>>> {
211        if !self.requires_chunk_stitching() {
212            return Ok(None);
213        }
214        let cursor = self.new_scan_cursor().await?;
215        let header_size = self.calculate_header_size();
216        {
217            let mut file_guard = cursor.file.lock().await;
218            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
219        }
220        let stitched = self.stitch_all_chunks(&cursor).await?;
221        Ok(Some(stitched))
222    }

224    /// Stitch all compressed chunks and parse as a single buffer (V5CompressedLegacy)
225    ///
226    /// This helper method extracts the stitching logic from get_all_entries so it can be
227    /// reused by sequential_scan and other methods that need to handle V5CompressedLegacy
228    /// format where partitions can span chunk boundaries.
229    ///
230    /// `read_shadowing` (issue #1741): `true` for user-facing SELECT scans
231    /// (`scan_for_key`, `sequential_scan`), `false` for the physical
232    /// `get_all_entries` (integrity verification / data-manager) so it counts every
233    /// on-disk row.
234    pub(super) async fn stitch_and_parse_all_chunks(
235        &self,
236        cursor: &ScanCursor,
237        schema: Option<&crate::schema::TableSchema>,
238        read_shadowing: bool,
239        scan_cancel: &crate::storage::scan_cancel::ScanCancel,
240    ) -> Result<Vec<(TableId, RowKey, ScanRow)>> {
241        let stitched_buffer = self
242            .stitch_all_chunks_cancellable(cursor, scan_cancel)
243            .await?;
244        let parser = self.build_v5_parser(read_shadowing);
245
246        // Get schema (use provided schema or reader's schema)
247        let reader_schema;
248        let table_schema = if let Some(s) = schema {
249            Some(s)
250        } else {
251            reader_schema = self.get_table_schema(None);
252            reader_schema.as_ref()
253        };
254
255        // Parse the stitched decompressed buffer
256        // #3782: EVERY chunk of the data section — see `BufferExtent`.
257        let entries =
258            parser.parse_block(&stitched_buffer, BufferExtent::Complete, table_schema, self)?;
259        tracing::debug!(
260            "stitch_and_parse_all_chunks: Parsed {} entries from stitched buffer",
261            entries.len()
262        );
263
264        Ok(entries)
265    }

311    /// Read, decompress, and concatenate every compressed chunk of the data
312    /// section into a single buffer.
313    ///
314    /// V5CompressedLegacy partitions can span chunk boundaries, so the whole
315    /// data section must be stitched before parsing. The returned buffer is
316    /// bounded by the *uncompressed data-section size* — it scales with on-disk
317    /// bytes, not row count (issue #790).
318    ///
319    /// Precondition: the caller has seeked `cursor`'s file to the start of the
320    /// data section (the cursor's chunk index starts at 0 when freshly minted).
321    ///
322    /// Existing callers use the reader's own [`SSTableReader::scan_cancel`]
323    /// field (unchanged pre-#2346 behaviour); this is a thin wrapper over
324    /// [`Self::stitch_all_chunks_cancellable`], which the per-call seam
325    /// (`sequential_scan`) drives with its caller-supplied token instead.
326    pub(super) async fn stitch_all_chunks(&self, cursor: &ScanCursor) -> Result<Vec<u8>> {
327        self.stitch_all_chunks_cancellable(cursor, &self.scan_cancel)
328            .await
329    }

331    /// [`Self::stitch_all_chunks`] with an explicit PER-CALL cancel token
332    /// (issue #2346). `scan_cancel` is polled every 256 chunks so a cancelled
333    /// caller abandons the stitch — the chunk-read/decompress loop is the
334    /// I/O-bound phase of the stitched scan path, matching the compaction
335    /// stream's per-256-chunk cadence (`stream_all_partitions_for_compaction`).
336    pub(super) async fn stitch_all_chunks_cancellable(
337        &self,
338        cursor: &ScanCursor,
339        scan_cancel: &crate::storage::scan_cancel::ScanCancel,
340    ) -> Result<Vec<u8>> {
341        use crate::storage::sstable::compression::Compression;
342
343        // #3853 scan-lifetime madvise seam, DEFENCE IN DEPTH — deliberate
344        // belt-and-braces, not a substitute for the entry-point guards.
345        //
346        // This function is the SINGLE funnel every stitched whole-section read
347        // passes through (`stitch_all_chunks` delegates here;
348        // `stitch_and_parse_all_chunks{,_with_metadata}` reach it through one of
349        // the two), so a guard here means a FUTURE caller is covered without
350        // anyone remembering to wire it. Under an already-guarded caller it is
351        // free: the count goes 1 -> 2 on begin and 2 -> 1 on drop, neither of
352        // which is an advice transition, so no syscall is issued and nothing is
353        // released early.
354        //
355        // It does NOT make the entry-point guards redundant. An entry point may
356        // do index/Summary work BEFORE the stitch and hold borrows into the
357        // mapping AFTER it, and the guard should cover that whole span; a guard
358        // that begins and ends inside the stitch would release while the caller
359        // is still reading.
360        //
361        // Its LIMIT, stated honestly: this closes only walks that go THROUGH
362        // this funnel. A raw block-loop walk — the shape `get_all_entries`' and
363        // `scan_for_key`'s non-stitching branches already have — bypasses it
364        // entirely and still needs its own guard. This is a NARROWING of the
365        // hole, not a closure of it.
366        let _scan = self.begin_scan();
367
368        // Pre-allocate buffer for ~2.5MB (estimated max size for test data)
369        let mut stitched_buffer = Vec::with_capacity(2_500_000);
370
371        // Incompressible-chunk fallback (Bug #639, epic #970): Cassandra stores a
372        // chunk RAW (not compressed) when its compressed length would meet or
373        // exceed `max_compressed_length`. `ChunkDecompressor::decompress_chunk`
374        // already honours this, but the stitch path did not — it blindly tried to
375        // LZ4/Snappy/etc-decode a raw chunk, which fails on the `incompressible`
376        // fixture. Mirror the writer rule here: when the (CRC-stripped) chunk
377        // length >= max_compressed_length, the bytes are already plaintext.
378        // Authority: CompressedSequentialWriter.java:160-177.
379        let max_compressed_length = self
380            .compression_info
381            .as_ref()
382            .map(|ci| ci.max_compressed_length as usize)
383            .unwrap_or(usize::MAX);
384
385        let mut chunk_count = 0;
386        while let Some(compressed_chunk) = self.read_next_block(cursor).await? {
387            // Cooperative checkpoint (issue #2346): poll the per-call token so a
388            // cancelled stitched scan abandons the I/O/decompress walk promptly
389            // instead of stitching the entire data section first, yielding every
390            // 256 chunks so the chokepoint timeout can elapse here too (#1695).
391            scan_cancel.checkpoint(chunk_count).await?;
392            let decompressed_chunk = if compressed_chunk.len() >= max_compressed_length {
393                // Stored uncompressed by Cassandra — the raw bytes pass through, and
394                // are COUNTED at the plane's one raw-chunk boundary (issue #1701 F3).
395                super::chunk_source::count_raw_chunk(
396                    &compressed_chunk,
397                    self.compression_reader.as_ref().map(|r| r.algorithm()),
398                );
399                compressed_chunk
400            } else if let Some(compression_reader) = &self.compression_reader {
401                // Single decode plane (issue #1598, G2): route the stitch-path
402                // decompress through ChunkSource so it is the ONLY query-path module
403                // that calls Compression::decompress. Behavior-identical (no cache,
404                // no counter) to the prior inline call.
405                let compression = Compression::new(*compression_reader.algorithm())?;
406                super::chunk_source::ChunkSource::decompress_only(
407                    Some(&compression),
408                    compressed_chunk,
409                )
410                .map_err(|e| {
411                    Error::corruption(format!(
412                        "stitch_all_chunks: Failed to decompress chunk {}: {}",
413                        chunk_count, e
414                    ))
415                })?
416            } else {
417                // No compression (should not happen for V5CompressedLegacy)
418                tracing::warn!("stitch_all_chunks: No compression reader, using raw chunk data");
419                compressed_chunk
420            };
421
422            stitched_buffer.extend_from_slice(&decompressed_chunk);
423            chunk_count += 1;
424        }
425
426        tracing::debug!(
427            "stitch_all_chunks: Stitched {} chunks, total buffer: {} bytes",
428            chunk_count,
429            stitched_buffer.len()
430        );
431
432        Ok(stitched_buffer)
433    }

435    /// Build a [`V5CompressedLegacyParser`] configured from this reader's header,
436    /// statistics (EncodingStats), version gates, and UDT registry.
437    ///
438    /// `read_shadowing` (issue #1741) MUST be `true` for user-facing query reads
439    /// (`scan`/`scan_stream`/`scan_with_cell_metadata`/point `get`) so the emit path
440    /// applies SELECT-semantic partition/range-tombstone shadowing and TTL expiry, and
441    /// `false` for PHYSICAL consumers that must see every on-disk row — integrity
442    /// verification via `get_all_entries`, `sstable_data_manager`, delta-scan, and the
443    /// compaction read path (which reconciles tombstones itself across generations).
444    ///
445    /// [`V5CompressedLegacyParser`]: crate::storage::sstable::reader::parsing::V5CompressedLegacyParser
446    pub(super) fn build_v5_parser(
447        &self,
448        read_shadowing: bool,
449    ) -> crate::storage::sstable::reader::parsing::V5CompressedLegacyParser {
450        let keyspace = self.header.keyspace.clone();
451        let table_name = self.header.table_name.clone();
452
453        // Extract EncodingStats from statistics_reader (if available)
454        let (min_timestamp, min_local_deletion_time, min_ttl) =
455            if let Some(stats_reader) = &self.statistics_reader {
456                let ts_stats = &stats_reader.statistics().timestamp_stats;
457                (
458                    ts_stats.min_timestamp,
459                    ts_stats.min_deletion_time,
460                    ts_stats.min_ttl,
461                )
462            } else {
463                (0, 0, None)
464            };
465
466        let parser = crate::storage::sstable::reader::parsing::V5CompressedLegacyParser::new(
467            keyspace,
468            table_name,
469            min_timestamp,
470            min_local_deletion_time,
471            min_ttl,
472        )
473        // VG1: thread VersionGates from SSTableReader down to row parser so
474        // that VG3 can flip gate-sensitive code paths without re-deriving gates.
475        .with_version_gates(self.version_gates.clone())
476        // Issue #1741: SELECT-semantic read shadowing (see fn docs).
477        .with_read_shadowing(read_shadowing);
478        // Add UDT registry if available for UDT-aware collection parsing (Issue #238)
479        if let Some(ref registry) = self.udt_registry {
480            parser.with_udt_registry(registry.clone())
481        } else {
482            parser
483        }
484    }

486    /// Read value at a specific offset with caching — the POINT-intent entry.
487    ///
488    /// Reads Data.db through the reader's dedicated `MADV_RANDOM` point mapping
489    /// ([`point_source`](Self::point_source), issue #2210), which is the right
490    /// advice for a scattered fault.
491    ///
492    /// Note (issue #2876, roborev job 4634): the index-driven range scan in
493    /// `sequential.rs` also uses THIS point-intent entry, and that is deliberate.
494    /// It looks like a scan, but `Index::get_range` yields entries in raw key-BYTE
495    /// order while Data.db is laid out in Murmur3 TOKEN order — uncorrelated for the
496    /// default partitioner — so its access really is scattered and `MADV_RANDOM` is
497    /// correct. The genuinely sequential walks (the Summary-guided walk, the full
498    /// index scan/stream, the windowed scan) reach the unadvised scan plane through
499    /// the positional helpers below, which take their plane from the caller.
500    pub async fn read_value_at_offset(&self, offset: u64, size: u32) -> Result<Option<ScanRow>> {

505    /// Shared body of the two offset-read entry points, parameterized by the
506    /// positional plane its caller's read intent selects (issue #2876).
507    ///
508    /// `source` is threaded all the way down — through the `CRC.db` verifier AND
509    /// the byte read / compressed-window decode — so a read never mixes planes:
510    /// no helper below hardcodes one, which is what made an index-driven scan's CRC
511    /// reads land on the advised point mapping while its bodies came off the scan
512    /// mapping (roborev #2882, Finding 1) and, symmetrically, made a genuine point
513    /// lookup lose the advised mapping entirely (Finding 2).
514    async fn read_value_at_offset_via(
515        &self,
516        source: &dyn super::read_at::ReadAt,
517        offset: u64,
518        size: u32,
519    ) -> Result<Option<ScanRow>> {
520        // Size must be non-zero for offset-based reading
521        if size == 0 {
522            return Err(Error::corruption(format!(
523                "Cannot read value at offset {} with size=0. This should have been caught earlier and handled via sequential scan.",
524                offset
525            )));
526        }
527
528        // Read-time CRC verification for uncompressed BIG SSTables (issue #1396).
529        // The index-based scan and point-lookup paths reach Data.db here (bypassing
530        // read_next_block / read_uncompressed_data_block), so verify the CRC.db
531        // chunk(s) covering [offset, offset+size) BEFORE returning any bytes. A
532        // mismatch is a typed Error::Corruption naming the chunk + offset (never
533        // wrong values / never a silent result). No-op when no CRC.db is present
534        // (compressed tables / BTI / absent-CRC.db warn-and-proceed). The CRC read
535        // uses the CALLER's plane (issue #2876): a scan's covering-chunk reads are
536        // part of its sequential walk, not scattered point faults.
537        self.verify_uncompressed_range(source, offset, size).await?;
538
539        // Read + decompress in ONE decode plane (issue #1598, G2). `get_cached_data`
540        // returns the final DECODED window: for a compressed Data.db it routes through
541        // the CRC-validated `read_compressed_offset_window` (issue #1773 — per-chunk
542        // inline CRC32 checked BEFORE decompression, whose sole `Compression::decompress`
543        // now resolves inside `ChunkSource`), or the CRC.db-verified raw bytes for an
544        // uncompressed one. There is no second decode here — doing so used to
545        // double-decompress the compressed path (and skip its inline CRC, the #1411 bug).
546        let data = self.get_cached_data(source, offset, size).await?;
547
548        // Preserve raw data until schema is available. Pre-#1334 this offset-read
549        // placeholder returned a bare `Value::Blob` of the row's raw value bytes,
550        // which the schema-aware decode path then schema-decoded and the no-schema query
551        // layer surfaced as a synthetic single-column "data" row. Carry that RAW
552        // provenance explicitly (issue #1334): a schema-aware consumer decodes the
553        // bytes; a no-schema consumer surfaces a single "data" blob — with no
554        // downstream guessing from the value's shape.
555        let row = ScanRow::RawRow(data.to_vec());
556
557        // Extract write time from value (placeholder - would need to be parsed from SSTable)
558        let _write_time = std::time::SystemTime::now()
559            .duration_since(std::time::UNIX_EPOCH)
560            .map(|d| d.as_micros() as i64)
561            .unwrap_or_else(|e| {
562                warn!("Failed to get system time: {}; using fallback value 0", e);
563                0
564            });
565
566        // Filter out tombstones and expired data
567        if !self.filter_tombstone(&row) {
568            return Ok(None);
569        }
570
571        Ok(Some(row))
572    }

574    /// Read block with caching support and hit/miss tracking.
575    ///
576    /// Wired to the shared [`DecompressedChunkCache`] (issue #1567): the cache is
577    /// consulted BEFORE the file read, so a repeat read of a resident region is a
578    /// refcount-bump hit that touches the file zero times and re-decompresses
579    /// nothing. Keyed by the index-resolved `block_offset` in the BIG-point-read
580    /// namespace ([`NS_BIG_POINT`]), which is disjoint from the chunk-index
581    /// namespaces used by the windowed-scan and BTI sites. The returned buffer is
582    /// already CRC-verified: [`read_value_at_offset`](Self::read_value_at_offset)
583    /// runs `verify_uncompressed_range` before calling this, so a hit returns
584    /// bytes verified when they were first inserted.
585    ///
586    /// `source` is the positional plane the CALLER's read intent selected (issue
587    /// #2876) — the advised point mapping for a point lookup, the unadvised scan
588    /// mapping for the index-driven scan. It is threaded in rather than read off
589    /// `self` precisely because this helper serves BOTH intents.
590    ///
591    /// [`DecompressedChunkCache`]: crate::storage::cache::DecompressedChunkCache
592    async fn get_cached_data(
593        &self,
594        source: &dyn super::read_at::ReadAt,
595        block_offset: u64,
596        size: u32,
597    ) -> Result<Vec<u8>> {
598        // The shared B1 cache tracks its own hit/miss counters (issue #1567). The
599        // ranged BIG-point key carries `size` as its aux discriminant so two reads at
600        // the same offset with different sizes can never alias (roborev #1567).
601        let key = crate::storage::cache::ChunkKey::with_aux(
602            self.chunk_cache_id ^ NS_BIG_POINT,
603            block_offset,
604            size as u64,
605        );
606        if let Some(hit) = self.chunk_cache.get(&key) {
607            return Ok(hit.to_vec());
608        }
609
610        // Produce the final DECOMPRESSED, integrity-verified window for this offset.
611        let data = if let Some(comp_info) = self.compression_info.as_deref() {
612            // COMPRESSED offset read (issue #1773): validate the inline per-chunk CRC32
613            // before decompression via the shared CRC-enforcing chunk reader (multi-chunk
614            // assembly, fail-closed past-EOF), NOT a raw read + blind LZ4 decode (#1411);
615            // the single decompress resolves inside `ChunkSource` (#1598). Count one
616            // backing read HERE (point-read site only), symmetric with the uncompressed
617            // branch below (#2167) — scan callers of the shared helper must NOT bump it.
618            model::CHUNK_READ_CALLS.fetch_add(1, Ordering::Relaxed);
619            self.read_compressed_offset_window(source, comp_info, block_offset, size)
620                .await?
621        } else {
622            // UNCOMPRESSED offset read. Positioned read on the caller's positional
623            // plane (issue #1573/C2 for the lock-free part, issue #2876 for the
624            // plane): no cursor mutex is held across this I/O, so concurrent reads
625            // do not convoy. `get_cached_data` is reached both by genuine point
626            // lookups (`big_point.rs`) and by every index-driven scan
627            // (`sequential.rs`), so the plane comes from the caller's intent — a
628            // point read keeps the dedicated `MADV_RANDOM` mapping (issue #2210), a
629            // scan takes the unadvised one, for which that readahead suppression is
630            // a deliberate loss. The covering CRC.db chunks were already verified on
631            // the SAME plane by `verify_uncompressed_range` (issue #1396), so these
632            // bytes are integrity-checked too.
633            model::CHUNK_READ_CALLS.fetch_add(1, Ordering::Relaxed);
634            let mut buffer = vec![0u8; size as usize];
635            source.read_exact_at(block_offset, &mut buffer)?;
636            // cqlite.read.bytes (issue #1701 roborev F3): an UNCOMPRESSED offset read
637            // never reaches the decode plane — there is nothing to decompress — so its
638            // bytes are counted at the plane's raw-chunk boundary here. Without this a
639            // point read (or an index-driven partition read) of an uncompressed
640            // SSTable reported rows and a duration but ZERO bytes. The cache-hit
641            // early return above stays uncounted: it read no Data.db.
642            super::chunk_source::count_raw_chunk(&buffer, None);
643            buffer
644        };
645
646        // Insert into the shared cache (converts the Vec to Arc<[u8]> once) and
647        // return an owned copy (this site's callers consume a Vec).
648        let arc = self.chunk_cache.insert(key, data);
649        Ok(arc.to_vec())
650    }

652    /// Verify the `CRC.db` chunk(s) covering the uncompressed Data.db byte range
653    /// `[offset, offset + size)` against their stored per-chunk CRC32 (issue
654    /// #1396), on the offset-read path used by the index-based scan and point
655    /// lookups.
656    ///
657    /// A partition read touches a sub-range of one or more `chunk_size` blocks; a
658    /// chunk's CRC can only be checked over the WHOLE chunk, so this reads each
659    /// covering chunk (bounded to one `chunk_size` block at a time) and compares.
660    /// Each chunk is verified at most once per reader lifetime (memoized in
661    /// [`SSTableReader::verified_uncompressed_chunks`]), keeping the cost at the
662    /// budgeted one CRC32 pass per chunk even when many partitions share a chunk.
663    ///
664    /// No-op when this reader has no `CRC.db` (compressed tables carry inline
665    /// per-chunk CRCs; BTI ships none; an absent `CRC.db` is warn-and-proceed).
666    ///
667    /// `source` is the positional plane the CALLER's read intent selected (issue
668    /// #2876). A chunk CRC read is I/O on the same Data.db bytes the caller is
669    /// about to consume, so it belongs on the caller's plane: a scan's
670    /// covering-chunk reads are part of its sequential walk (the advised
671    /// `MADV_RANDOM` point mapping would suppress readahead over a whole 64 KiB
672    /// chunk), while a point lookup's stay on the advised mapping (issue #2210).
673    /// This is why the plane is a parameter and never read off `self` here —
674    /// hardcoding it split one logical read across two planes (roborev #2882).
675    async fn verify_uncompressed_range(
676        &self,
677        source: &dyn super::read_at::ReadAt,
678        offset: u64,
679        size: u32,
680    ) -> Result<()> {
681        let Some(crc) = self.crc_reader.as_deref() else {
682            return Ok(());
683        };
684        if size == 0 {
685            return Ok(());
686        }
687        let file_size = self.stats.file_size;
688        // Fix 3 (issue #1396): a corrupt on-disk offset/size can overflow
689        // `offset + size` (debug panic / wrapped range that misattributes CRC
690        // chunks) or point past EOF. Use checked arithmetic and reject a range
691        // that overflows or exceeds the Data.db length as typed corruption
692        // BEFORE deriving any chunk index. `size >= 1` here (0 handled above),
693        // so `end > offset` and `end - 1` never underflows.
694        let end = offset
695            .checked_add(size as u64)
696            .filter(|end| *end <= file_size)
697            .ok_or_else(|| {
698                Error::corruption(format!(
699                    "uncompressed read range [0x{offset:x}, +{size}) overflows or exceeds the \
700                     Data.db length {file_size}; refusing to verify a corrupt offset"
701                ))
702            })?;
703        // Each covering chunk is CRC'd over its on-disk bytes via a positioned
704        // read on the caller's plane (issue #1573 C2 for the lock-free part, issue
705        // #2876 for the plane): the verifier never holds a cursor mutex across I/O,
706        // so it neither convoys concurrent point reads nor disturbs any scan
707        // cursor's position.
708        self.verify_covering_chunks(crc, offset, end, |lo, hi, chunk| {
709            let mut buf = vec![0u8; (hi - lo) as usize];
710            source.read_exact_at(lo, &mut buf).map_err(|e| {
711                Error::corruption(format!(
712                    "failed to read uncompressed chunk {chunk} at Data.db offset 0x{lo:x} for CRC verification: {e}"
713                ))
714            })?;
715            Ok(crc32fast::hash(&buf))
716        })
717    }
718
719    /// In-buffer counterpart of [`Self::verify_uncompressed_range`] for the
720    /// whole-section point-read fallback (issue #1573): the section is ALREADY
721    /// resident in `section`, covering Data.db `[base, base + section.len())`, so
722    /// each fully-contained covering chunk is CRC-checked directly from memory —
723    /// the section is transferred from disk EXACTLY ONCE (the caller's windowed
724    /// read), not a second time for CRC. A chunk that straddles below `base` (only
725    /// possible when `base` is not `CRC.db`-chunk-aligned, e.g. a nonzero header
726    /// that does not fall on a chunk boundary) re-reads just that one chunk so the
727    /// CRC-before-use guarantee holds regardless of header alignment. Same CRC32
728    /// algorithm, chunk layout, memoization, and mismatch/typed-error semantics as
729    /// [`Self::verify_uncompressed_range`] — including its `source` contract: the
730    /// straddling-chunk re-read uses the CALLER's positional plane (issue #2876),
731    /// never a hardcoded one.
732    async fn verify_uncompressed_section_in_buffer(
733        &self,
734        source: &dyn super::read_at::ReadAt,
735        base: u64,
736        section: &[u8],
737    ) -> Result<()> {
738        let Some(crc) = self.crc_reader.as_deref() else {
739            return Ok(());
740        };
741        if section.is_empty() {
742            return Ok(());
743        }
744        let file_size = self.stats.file_size;
745        let end = base
746            .checked_add(section.len() as u64)
747            .filter(|end| *end <= file_size)
748            .ok_or_else(|| {
749                Error::corruption(format!(
750                    "uncompressed section [0x{base:x}, +{}) overflows or exceeds the Data.db \
751                     length {file_size}; refusing to verify a corrupt section",
752                    section.len()
753                ))
754            })?;
755        self.verify_covering_chunks(crc, base, end, |lo, hi, chunk| {
756            if lo >= base && hi <= end {
757                // Chunk fully resident in the already-read buffer — CRC from
758                // memory, no second I/O.
759                let s = (lo - base) as usize;
760                let e = (hi - base) as usize;
761                Ok(crc32fast::hash(&section[s..e]))
762            } else {
763                // Chunk extends below `base` (unaligned header) — read just it.
764                let mut buf = vec![0u8; (hi - lo) as usize];
765                source.read_exact_at(lo, &mut buf).map_err(|e| {
766                    Error::corruption(format!(
767                        "failed to read uncompressed chunk {chunk} at Data.db offset 0x{lo:x} for CRC verification: {e}"
768                    ))
769                })?;
770                Ok(crc32fast::hash(&buf))
771            }
772        })
773    }

775    /// Walk the `CRC.db` chunk(s) covering Data.db `[start, end)` (`end` already
776    /// bounds-checked `<= file_size` by the caller, `end > start`), skipping chunks
777    /// already verified for this reader, comparing each covering chunk's CRC32 to
778    /// `CRC.db`, and memoizing successes so a chunk is verified at most once per
779    /// reader lifetime. `compute(lo, hi, chunk)` yields the CRC32 of the chunk's
780    /// on-disk bytes `[lo, hi)`; callers supply either a positioned disk read or an
781    /// in-buffer slice, which is what lets the whole-section fallback read its bytes
782    /// exactly once. This is the single place the chunk geometry, mismatch error,
783    /// and memoization live — the two public verify entry points differ ONLY in how
784    /// they source each chunk's bytes. Module-visible so the windowed scan's
785    /// synchronous uncompressed piece reader shares the same memoized path (#1940).
786    pub(in crate::storage::sstable::reader) fn verify_covering_chunks(
787        &self,
788        crc: &super::crc::CrcDb,
789        start: u64,
790        end: u64,
791        mut compute: impl FnMut(u64, u64, u64) -> Result<u32>,
792    ) -> Result<()> {
793        let cs = crc.chunk_size() as u64;
794        if cs == 0 {
795            return Err(Error::corruption(
796                "CRC.db chunk size is zero; cannot verify uncompressed read",
797            ));
798        }
799        let file_size = self.stats.file_size;
800        let first = start / cs;
801        let last = (end - 1) / cs;
802        for chunk in first..=last {
803            // Skip chunks already verified for this reader.
804            {
805                let seen = self
806                    .verified_uncompressed_chunks
807                    .lock()
808                    .unwrap_or_else(|e| e.into_inner());
809                if seen.contains(&chunk) {
810                    continue;
811                }
812            }
813            let lo = chunk * cs;
814            let hi = ((chunk + 1) * cs).min(file_size);
815            if hi <= lo {
816                break; // range extends past EOF; nothing real to verify
817            }
818            let computed = compute(lo, hi, chunk)?;
819            let expected = crc.crc_for_chunk(chunk as usize)?;
820            if computed != expected {
821                return Err(Error::corruption(format!(
822                    "uncompressed CRC32 mismatch for chunk {} at Data.db offset 0x{:x} \
823                     ({} bytes): expected=0x{:08x}, computed=0x{:08x} (CRC.db)",
824                    chunk,
825                    lo,
826                    hi - lo,
827                    expected,
828                    computed
829                )));
830            }
831            self.verified_uncompressed_chunks
832                .lock()
833                .unwrap_or_else(|e| e.into_inner())
834                .insert(chunk);
835        }
836        Ok(())
837    }

839    /// The SINGLE CRC-checked chokepoint for reading a raw byte range from an
840    /// *uncompressed* Data.db section (issue #1396).
841    ///
842    /// EVERY uncompressed offset read MUST flow through here so that no public
843    /// read path can silently bypass `CRC.db` verification and hand corrupt
844    /// bytes to the parser. It runs the shared, memoized verifier
845    /// [`Self::verify_uncompressed_range`] over the covering chunk(s) BEFORE any
846    /// bytes are returned; a mismatch is the typed [`Error::Corruption`] naming
847    /// the failing chunk + Data.db offset. The verifier is a no-op when this
848    /// reader has no `CRC.db` (compressed tables carry inline per-chunk CRCs,
849    /// BTI ships none, an absent `CRC.db` is warn-and-proceed), so this helper is
850    /// also the correct accessor for the raw-read step of compressed offset reads.
851    ///
852    /// `offset` is an ABSOLUTE Data.db file offset (post-header). `file` is the
853    /// handle to read the range from — the shared point-read handle
854    /// ([`Self::file`]) or a scan-local cursor's private handle (issue #815).
855    /// CRC verification reads positionally via `source` instead, so it is
856    /// independent of `file` and never disturbs a scan cursor's position.
857    ///
858    /// `source` is the positional plane the CALLER's read intent selects (issue
859    /// #2876), threaded through for the CRC chunk reads exactly as in
860    /// [`Self::verify_uncompressed_range`]. Every caller today is a scan walk
861    /// (`summary_scan.rs`, `full_index_scan.rs`, `full_index_stream.rs`) and passes
862    /// the unadvised scan mapping; a future point caller passes the advised one and
863    /// gets the right advice without editing this helper.
864    ///
865    /// SCOPE NOTE (#2876): `source` governs the **CRC chunk reads only**. The payload
866    /// bytes below are read through `file` — a seek-based [`BlockSource`], which is NOT
867    /// an mmap and therefore carries no `madvise` advice on any backend. So the
868    /// uncompressed path never suffered the MADV_RANDOM readahead suppression this
869    /// issue fixes (that regression was mmap-specific, hence compressed-walk-specific),
870    /// and threading the plane here closes the CRC-read half completely: after this
871    /// change NO read on an uncompressed scan touches an advised mapping. The
872    /// separate cost on this path is the shared `file` mutex serializing concurrent
873    /// scans (#815's domain), which is deliberately out of scope here.
874    ///
875    /// Future uncompressed offset reads MUST call this instead of doing their own
876    /// `seek` + `read_exact`, so the CRC check can never be forgotten again.
877    pub(in crate::storage::sstable::reader) async fn read_uncompressed_verified(
878        &self,
879        source: &dyn super::read_at::ReadAt,
880        file: &tokio::sync::Mutex<super::source::BlockSource>,
881        offset: u64,
882        len: usize,
883    ) -> Result<Vec<u8>> {
884        use tokio::io::AsyncReadExt;
885
886        let size = u32::try_from(len).map_err(|_| {
887            Error::corruption(format!(
888                "uncompressed read length {len} exceeds u32 range for CRC verification \
889                 at Data.db offset 0x{offset:x}"
890            ))
891        })?;
892
893        // Issue #2819 (BLOCKER): attribute the UNCOMPRESSED body page-in to
894        // `stream_cold_fault`, but time ONLY the actual IO — NOT the
895        // `file.lock().await` below on the reader-wide mutex. The warm registry
896        // (#2356) shares one `Arc<SSTableReader>` across concurrent `do_get`s that
897        // admission (#2420) runs at once, so a PEER scan's lock-wait must never be
898        // attributed to cold-IO (it would inflate `stream_cold_fault` on a WARM run
899        // and break the cold−warm delta = cold-IO meaning). Mirrors the compressed
900        // path's "time only the positional read". `verify_uncompressed_range` reads
901        // CRC.db via the positional `source` (NO shared mutex) — genuine cold IO —
902        // so it is timed; the body read is timed under the already-held lock. No
903        // `stream_decompress` (correctly absent).
904        //
905        // Issue #2819 (L1): capture the sink ONCE here, BEFORE any `.await`, and
906        // build both timers from that captured `Option` — so no thread-local read
907        // happens post-await (correct even if the future resumes on another
908        // executor thread). `None` (zero cost) with no flight sink installed.
909        let cold_sink = crate::observability::stream_subphase::current();
910        {
911            let _cold = crate::observability::stream_subphase::scoped_captured(
912                &cold_sink,
913                crate::observability::StreamSubPhase::ColdFault,
914            );
915            // Verify the covering CRC.db chunk(s) BEFORE returning any bytes.
916            self.verify_uncompressed_range(source, offset, size).await?;
917        }
918
919        let mut buf = vec![0u8; len];
920        {
921            // Lock-wait is acquired OUTSIDE the cold-fault timer (peer-scan mutex
922            // contention is not cold-IO — the BLOCKER).
923            let mut guard = file.lock().await;
924            let _cold = crate::observability::stream_subphase::scoped_captured(
925                &cold_sink,
926                crate::observability::StreamSubPhase::ColdFault,
927            );
928            guard.seek(SeekFrom::Start(offset)).await?;
929            guard.read_exact(&mut buf).await?;
930        }
931        Ok(buf)
932    }

986    /// Prepare for a delta-scan pass: stitch all compressed chunks of the data
987    /// section and return the decompressed buffer together with a pre-configured
988    /// parser.
989    ///
990    /// Uses its own per-scan cursor (issue #815), so it no longer needs the
991    /// caller to serialize against concurrent reads. This method is gated on the
992    /// `delta-scan` feature and is the only bridge between the SSTableReader
993    /// internals and the `delta_scan` module, which cannot access private
994    /// helpers directly.
995    ///
996    /// The `schema` parameter is not used here — it is threaded through the
997    /// caller's `parse_block_emit_delta` invocation instead.  The parser is
998    /// built via `build_v5_parser()` which handles version-gates and UDT
999    /// registry without needing the schema at construction time.
1000    #[cfg(feature = "delta-scan")]
1001    pub async fn prepare_delta_scan(
1002        &self,
1003    ) -> Result<(Vec<u8>, super::parsing::V5CompressedLegacyParser)> {
1004        use tokio::io::AsyncSeekExt;
1005
1006        // #3853 scan-lifetime madvise seam. This function reads the WHOLE data
1007        // section (seek past the header, then stitch every chunk), so by the
1008        // seam's own definition it is a scan and holds a guard.
1009        //
1010        // The guard's scope is THIS BODY, deliberately, and not the caller's
1011        // (`delta_scan::scan.rs`): the return value is an OWNED `Vec<u8>` plus a
1012        // parser, and the caller parses that owned buffer — no borrow into the
1013        // mapping survives the return. So releasing when the stitch completes is
1014        // the RSS control working, not a premature release.
1015        let _scan = self.begin_scan();
1016
1017        // Seek the per-scan cursor to the start of the data section.
1018        let cursor = self.new_scan_cursor().await?;
1019        let header_size = self.calculate_header_size();
1020        {
1021            let mut file_guard = cursor.file.lock().await;
1022            file_guard
1023                .seek(std::io::SeekFrom::Start(header_size as u64))
1024                .await?;
1025        }
1026
1027        // Stitch all compressed chunks (bounded by uncompressed data-section size).
1028        let stitched = self.stitch_all_chunks(&cursor).await?;
1029
1030        // Build a parser (re-using the existing builder so version-gates and
1031        // UDT registry are threaded through correctly).
1032        //
1033        // #3782: this buffer IS the whole data section, and no extent has to be
1034        // declared for it — the delta-scan consumer is `parse_block_emit_delta`,
1035        // which refuses a partition-header or row decode failure
1036        // UNCONDITIONALLY (`Error::corruption`, `block_emit.rs`). So there is no
1037        // defaulted-lossy behaviour for this site to inherit; adding an extent
1038        // argument the parse never consults would assert a contract that does
1039        // not exist.
1040        let parser = self.build_v5_parser(false);
1041
1042        Ok((stitched, parser))
1043    }
1044}
