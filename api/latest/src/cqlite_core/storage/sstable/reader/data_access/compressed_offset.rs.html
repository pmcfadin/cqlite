1//! CRC-validated compressed offset-read window (issue #1773).
2//!
3//! The compressed offset-read point-lookup path
4//! ([`read_value_at_offset`](super::super::SSTableReader::read_value_at_offset) →
5//! `get_cached_data`) used to read `size` raw bytes at an offset and LZ4-decompress
6//! them WITHOUT validating the trailing 4-byte inline per-chunk CRC32 — re-introducing
7//! the #1411 CRC bypass on a latent path (unreachable today via `get`, but live the
8//! moment `find_entry` hits for a compressed table). This module carries the helper
9//! that routes that case through the shared CRC-enforcing chunk reader.
10
11use std::sync::atomic::Ordering;
12
13use crate::storage::sstable::reader::SSTableReader;
14use crate::{Error, Result};
15
16impl SSTableReader {
17    /// Decompress the uncompressed byte window `[block_offset, block_offset + size)`
18    /// out of a COMPRESSED Data.db, validating each covering chunk's authoritative
19    /// inline per-chunk CRC32 BEFORE decompression (issue #1773).
20    ///
21    /// Reuses [`read_compressed_chunk_at`] — the positional, CRC-enforcing chunk
22    /// reader also used by the BTI point-read path — so the CRC-then-decompress
23    /// ordering (guardrail #1411) is honored on this path too: a bit-flipped chunk
24    /// surfaces the SAME typed `Error::InvalidFormat` (naming the chunk index +
25    /// offset) that scan / `scan_for_key` return, never garbage. No heuristics: the
26    /// authoritative CRC trailer is checked, never inferred.
27    ///
28    /// [`read_compressed_chunk_at`]: super::super::block_io::read_compressed_chunk_at
29    pub(super) async fn read_compressed_offset_window(
30        &self,
31        comp_info: &crate::storage::sstable::compression_info::CompressionInfo,
32        block_offset: u64,
33        size: u32,
34    ) -> Result<Vec<u8>> {
35        use crate::storage::sstable::compression::Compression;
36
37        let compression = self
38            .compression_reader
39            .as_ref()
40            .map(|r| Compression::new(*r.algorithm()))
41            .transpose()?;
42
43        read_compressed_offset_window_impl(
44            self.point_source.as_ref(),
45            comp_info,
46            compression.as_ref(),
47            self.stats.file_size,
48            block_offset,
49            size,
50        )
51    }
52}

54/// Decompression core of [`SSTableReader::read_compressed_offset_window`], factored
55/// out of the reader method so it can be exercised directly against a hand-built
56/// [`CompressionInfo`] + `ReadAt` source (no reader/SQL roundtrip) — see the
57/// `max_compressed_length == 0` regression test below.
58///
59/// [`CompressionInfo`]: crate::storage::sstable::compression_info::CompressionInfo
60pub(super) fn read_compressed_offset_window_impl(
61    point_source: &dyn super::super::read_at::ReadAt,
62    comp_info: &crate::storage::sstable::compression_info::CompressionInfo,
63    compression: Option<&crate::storage::sstable::compression::Compression>,
64    file_size: u64,
65    block_offset: u64,
66    size: u32,
67) -> Result<Vec<u8>> {
68    use super::super::block_io;
69
70    let chunk_length = comp_info.chunk_length as usize;
71    if chunk_length == 0 {
72        return Err(Error::corruption(
73            "CompressionInfo chunk_length is zero; cannot map a Data.db offset to a \
74             compressed chunk"
75                .to_string(),
76        ));
77    }
78
79    // Fail CLOSED on a corrupt `CompressionInfo` whose `max_compressed_length == 0`
80    // (issue #2524, mirroring the #1869 fix in the sibling `compressed_partition_window`):
81    // otherwise the raw-chunk fallback below (`compressed.len() >= max_compressed_length`)
82    // is ALWAYS true, so EVERY still-LZ4-compressed chunk would be returned verbatim as
83    // "raw/incompressible" plaintext — never decompressed — and the inline CRC32 (computed
84    // over the genuine on-disk bytes) would still pass, silently handing back garbage rows.
85    // A valid Cassandra `CompressionInfo` never records a zero `max_compressed_length`
86    // (it is `i32::MAX` when minCompressRatio=0, the default; CompressionParams.java:186-189).
87    let max_compressed_length = comp_info.max_compressed_length as usize;
88    if max_compressed_length == 0 {
89        return Err(Error::corruption(
90            "compressed offset read: CompressionInfo max_compressed_length is zero; \
91             cannot distinguish compressed chunks from raw/incompressible ones"
92                .to_string(),
93        ));
94    }
95
96    let start = block_offset as usize;
97    let end = start.checked_add(size as usize).ok_or_else(|| {
98        Error::corruption(format!(
99            "compressed offset read range [{start}, +{size}) overflows"
100        ))
101    })?;
102    // `size >= 1` (callers reject size == 0), so `end > start` and `end - 1`
103    // never underflows; `first_chunk <= last_chunk`.
104    let first_chunk = start / chunk_length;
105    let last_chunk = (end - 1) / chunk_length;
106
107    let mut assembled =
108        Vec::with_capacity(last_chunk.saturating_sub(first_chunk).saturating_add(1) * chunk_length);
109    for chunk_idx in first_chunk..=last_chunk {
110        // CRC-validated compressed bytes (guardrail #1411): the trailing inline
111        // per-chunk CRC32 is checked HERE, before decompression. header_offset is
112        // 0 for nb/BTI (CompressionInfo chunk offsets are absolute from Data.db
113        // byte 0), matching the cursor and BTI point-read paths (issue #1573 C2).
114        //
115        // Fail CLOSED (issue #1773 roborev): for a VALID offset+size every chunk
116        // in `first_chunk..=last_chunk` exists, so a `None` (EOF) here means the
117        // requested range does not actually fit in this compressed Data.db — a
118        // corrupt/out-of-range offset or size. Returning the partial `assembled`
119        // bytes as `Ok` would silently truncate; this must surface as a typed,
120        // non-recoverable corruption error instead.
121        let Some(compressed) =
122            block_io::read_compressed_chunk_at(point_source, comp_info, chunk_idx, file_size, 0)?
123        else {
124            return Err(Error::corruption(format!(
125                "compressed offset read requires chunk {chunk_idx} past EOF for range \
126                 [{start}, {end}) — corrupt or out-of-range offset/size"
127            )));
128        };
129
130        // Incompressible-chunk fallback (Bug #639, epic #970): Cassandra stores a
131        // chunk RAW (not compressed) when its compressed length would meet or
132        // exceed `max_compressed_length`; those bytes are already plaintext.
133        // Authority: CompressedSequentialWriter.java:160-177.
134        let decompressed = if compressed.len() >= max_compressed_length {
135            compressed
136        } else if let Some(compression) = compression {
137            // Single decode plane (issue #1598, G2): the actual decompress call
138            // resolves inside `ChunkSource::decompress_only`, so this module holds
139            // zero query-path decompress call sites (the architecture test proves
140            // exactly one such module). CRC is already validated above by
141            // `read_compressed_chunk_at` (guardrail #1411), so we never decode bytes
142            // that failed their inline CRC32.
143            let out = super::super::chunk_source::ChunkSource::decompress_only(
144                Some(compression),
145                compressed,
146            )?;
147            super::model::DECOMPRESS_CALLS.fetch_add(1, Ordering::Relaxed);
148            out
149        } else {
150            // No compression reader (should not happen: this path is only reached
151            // for a compressed Data.db, which always carries one). Mirror
152            // `stitch_all_chunks`' warn! (issue #2167) so an unexpected reach —
153            // returning still-compressed bytes as if plaintext — is visible in logs.
154            tracing::warn!(
155                "read_compressed_offset_window: No compression reader, using raw chunk data"
156            );
157            compressed
158        };
159        assembled.extend_from_slice(&decompressed);
160    }
161
162    // Slice the requested window out of the assembled uncompressed chunk bytes.
163    //
164    // Fail CLOSED (issue #1773 roborev): for a VALID offset+size the assembled
165    // chunks fully cover `[start, end)`. If they do not — a short final chunk, or
166    // `within_start` itself past what decoded — the offset/size is corrupt or
167    // out-of-range; return a typed corruption error rather than truncating to
168    // `Ok(partial)` / `Ok(empty)`.
169    let window_base = first_chunk * chunk_length;
170    let within_start = start - window_base;
171    let requested_end = end - window_base;
172    if requested_end > assembled.len() {
173        return Err(Error::corruption(format!(
174            "compressed offset read range [{start}, {end}) is short by {} bytes after \
175             decompressing chunks {first_chunk}..={last_chunk} (assembled {} bytes) — \
176             corrupt or out-of-range offset/size",
177            requested_end - assembled.len(),
178            assembled.len()
179        )));
180    }
181    Ok(assembled[within_start..requested_end].to_vec())
182}

184#[cfg(test)]
185mod tests {
186    use crate::storage::sstable::compression_info::CompressionInfo;
187    use crate::storage::sstable::reader::read_at::ReadAt;
188    use crate::Error;
189
190    /// Minimal `ReadAt` double. The `max_compressed_length == 0` guard fires BEFORE
191    /// any chunk read, so `read_at` is never reached — it exists only to satisfy the
192    /// signature (and panics if the guard ever regresses to touch the source).
193    struct NeverReadAt;
194    impl ReadAt for NeverReadAt {
195        fn read_at(&self, _offset: u64, _buf: &mut [u8]) -> crate::Result<usize> {
196            panic!(
197                "read_at must not be reached: the max_compressed_length==0 guard fails closed \
198                    before any chunk I/O"
199            );
200        }
201        fn len(&self) -> u64 {
202            0
203        }
204    }
205
206    /// A hand-built `CompressionInfo` with `max_compressed_length` corrupted to `0`.
207    /// One 64-byte chunk; the chunk_offsets/data_length are immaterial because the
208    /// guard rejects the metadata before mapping any offset to a chunk.
209    fn comp_info_zero_max() -> CompressionInfo {
210        CompressionInfo {
211            algorithm: "LZ4Compressor".to_string(),
212            option_pairs: Vec::new(),
213            chunk_length: 64,
214            max_compressed_length: 0, // corrupt: a valid CompressionInfo never records 0
215            data_length: 64,
216            chunk_offsets: vec![0],
217        }
218    }
219
220    /// Issue #2524 — a corrupt/malformed `CompressionInfo` whose `max_compressed_length
221    /// == 0` MUST fail closed with a typed `Error::Corruption`, never silently return
222    /// still-compressed bytes as plaintext. Pre-fix, the raw-chunk fallback test
223    /// `compressed.len() >= max_compressed_length` was `>= 0` — ALWAYS true — so every
224    /// still-LZ4-compressed chunk was returned verbatim as "raw" plaintext, and the
225    /// inline CRC32 (over the genuine on-disk bytes) still matched, failing OPEN.
226    /// Mirrors the #1869 fix + test for the sibling `compressed_partition_window`.
227    #[test]
228    fn zero_max_compressed_length_fails_closed() {
229        let ci = comp_info_zero_max();
230        let src = NeverReadAt;
231
232        let err = super::read_compressed_offset_window_impl(&src, &ci, None, 0, 0, 16).expect_err(
233            "CompressionInfo.max_compressed_length == 0 must fail closed with a typed \
234                 corruption error, not silently return still-compressed bytes as plaintext",
235        );
236        match err {
237            Error::Corruption(m) => assert!(
238                m.contains("max_compressed_length is zero"),
239                "unexpected corruption text: {m}"
240            ),
241            other => panic!("expected Corruption(max_compressed_length zero), got {other:?}"),
242        }
243    }
