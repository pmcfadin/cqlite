1//! Summary-guided streaming partition enumeration (issue #2412 §C / #2413
2//! Option A, Stage 4).
3//!
4//! The #2361 streaming walk ([`full_index_stream`](super::full_index_stream))
5//! emits each partition as the index resolves it, but sources its entries from
6//! the resident `Vec<PartitionIndexEntry>` — which requires the whole `Index.db`
7//! to be materialised first ([`IndexReader::ensure_materialized`](crate::storage::sstable::index_reader::IndexReader::ensure_materialized),
8//! the ~500MB-resident structure #2385 retired at open). This module sources the
9//! SAME walk from a FORWARD-STREAMED `Index.db`
10//! ([`IndexEntryStream`](crate::storage::sstable::index_reader::IndexEntryStream)),
11//! so a warm BIG reader never materialises the full map (spec R3/R4).
12//!
13//! Two extra wins over the materialising walk:
14//! - **No resident `Vec`.** Entry memory is one refill window, partition-count
15//!   independent — the property the warm registry (Stage 5) relies on.
16//! - **Token pushdown (#2413).** A token-range split begins the walk at the
17//!   `Summary.db` sample covering the range start
18//!   ([`SummaryReader::scan_start_position_for_token`](crate::storage::sstable::summary_reader::SummaryReader::scan_start_position_for_token))
19//!   and stops once the walk passes the range end, so out-of-range partition
20//!   bodies are never read/decoded. A compaction consumer passes no range and
21//!   walks the full ring.
22//!
23//! Walk contract preserved from #2361: the `(token, key)` order guard
24//! ([`check_token_order`](super::full_index_stream::check_token_order)),
25//! per-partition structural coverage
26//! ([`partition_slice_fully_consumed`](SSTableReader::partition_slice_fully_consumed),
27//! Signal B), cancel-aware polling, and the `stream_walk_partitions_parsed`
28//! work-probe. The completeness signal (Signal A, mid-entry truncation) moves to
29//! the streaming terminus ([`IndexEntryStream::truncated_tail`](crate::storage::sstable::index_reader::IndexEntryStream)).
30//!
31//! No-heuristics (issue #28): the start offset is an authoritative `Summary.db`
32//! sample position; every entry is authoritative `Index.db` framing; each
33//! partition body is bounded by the SUCCESSOR entry's offset (last by the
34//! data-section end) — never a guessed size.

81/// The token range applied to a walk whose signature does not take one.
82///
83/// [`SSTableReader::stream_all_partitions_for_query`] pushes its
84/// [`ScanTokenBound`] into the Summary-guided walk, which skips an out-of-range
85/// partition body rather than decoding it. Its full-ring fallback routes take no
86/// bound, so this gate applies the range to their emit instead. Correctness is
87/// then a property of the call rather than of which route the reader chose, which
88/// matters most for a BTI (`da`) reader: it fails the Summary-guided gate on every
89/// call, so the fallback is its only route (issue #3358).
90///
91/// The verdict is computed once per partition, not once per row. A token is a
92/// property of the partition key, and a walk emits a partition's rows
93/// consecutively, so a per-row hash would charge every row for an answer that
94/// changes at a partition boundary.
95struct TokenGate {
96    bound: Option<ScanTokenBound>,
97    /// The key `admits` below was computed for.
98    last: Option<std::sync::Arc<[u8]>>,
99    admits: bool,
100}
101
102impl TokenGate {
103    fn new(bound: Option<ScanTokenBound>) -> Self {
104        // `admits` starts true so an unbounded walk answers without touching
105        // `last` at all.
106        Self {
107            bound,
108            last: None,
109            admits: true,
110        }
111    }
112
113    /// Whether this row's partition is inside the range.
114    fn admits(&mut self, key: &std::sync::Arc<[u8]>) -> bool {
115        let Some(bound) = self.bound else {
116            return true;
117        };
118        if self.last.as_ref().is_none_or(|last| last != key) {
119            let token = crate::util::cassandra_murmur3::cassandra_murmur3_token(key);
120            self.admits = bound.contains(token);
121            self.last = Some(key.clone());
122        }
123        self.admits
124    }
125}

127impl SSTableReader {
128    /// The `Index.db` path SIBLING of this reader's CURRENT `Data.db` path (issue
129    /// #2412 §C, #2383-aware). Derived from [`Self::file_path`] (the ArcSwap that a
130    /// #2383 inode-rebind updates) by swapping the `-Data.db` component suffix, so a
131    /// fresh streaming `Index.db` open follows a snapshot rebind rather than
132    /// re-opening a torn-down path. Falls back to the reader's recorded open-time
133    /// index path when the `Data.db` name is not the expected `*-Data.db` shape.
134    fn current_index_db_path(&self) -> std::path::PathBuf {
135        let data = self.file_path();
136        // Same `-Data.db` → `-Index.db` sibling rule the #2383 rebind uses to repoint
137        // the lazy IndexReader, so both stay on the rebound generation (issue #2356).
138        crate::storage::sstable::reader::index_db_sibling(&data).unwrap_or_else(|| {
139            self.index_reader
140                .as_ref()
141                .map(|ir| ir.index_path())
142                .unwrap_or(data)
143        })
144    }

146    /// Shared Summary-guided FORWARD `Index.db` walk (issue #2412 §C / #2413
147    /// Option A): stream in-range partition SLICES from a `Summary.db`-guided start
148    /// offset WITHOUT materialising the resident partition map, invoking `decode`
149    /// on each in-range partition's exact `Data.db` byte slice.
150    ///
151    /// Both emit shapes are built on this: the read-shadowing ScanRow decoder
152    /// ([`stream_partitions_summary_guided`](Self::stream_partitions_summary_guided),
153    /// the non-stitching `V5_0Uncompressed` model) and the full-fidelity
154    /// CompactionRow decoder
155    /// ([`stream_partitions_summary_guided_compaction`](Self::stream_partitions_summary_guided_compaction),
156    /// the chunk-stitching `nb` merge model). Factoring the walk here means the
157    /// order guard, coverage check, work-probe, and range/terminus logic are
158    /// defined ONCE — the two decoders differ only in how a proven-complete slice
159    /// becomes rows.
160    ///
161    /// `token_bound`: `None` = full ring from offset 0; `Some(bound)` = begin at
162    /// the sample covering `start_excl` and stop once past `end_incl`.
163    ///
164    /// Returns [`FullIndexStreamOutcome::FellBack`] (nothing emitted) when there is
165    /// no usable summary/index to stream from. Any inconsistency AFTER the first
166    /// emit is a hard [`Error`] (fail-closed): a streaming walk cannot fall back.
167    async fn walk_in_range_partition_slices<D>(
168        &self,
169        scan_cancel: &ScanCancel,
170        token_bound: Option<ScanTokenBound>,
171        parser: &crate::storage::sstable::reader::parsing::V5CompressedLegacyParser,
172        schema: Option<&crate::schema::TableSchema>,
173        decode: &mut D,
174    ) -> Result<FullIndexStreamOutcome>
175    where
176        D: FnMut(&[u8]) -> Result<ControlFlow<()>>,
177    {
178        let (Some(_index_reader), Some(summary)) =
179            (self.index_reader.as_ref(), self.summary_reader.as_ref())
180        else {
181            return Ok(FullIndexStreamOutcome::FellBack);
182        };
183        if summary.get_entries().is_empty() {
184            return Ok(FullIndexStreamOutcome::FellBack);
185        }
186
187        // Exclusive end of the last partition (uncompressed data-section domain),
188        // identical to the materialising walk's derivation.
189        let data_section_end = match self.compression_info.as_deref() {
190            Some(ci) => ci.data_length,
191            None => self
192                .stats
193                .file_size
194                .saturating_sub(self.actual_header_size as u64),
195        };
196        if data_section_end == 0 {
197            return Ok(FullIndexStreamOutcome::FellBack);
198        }
199
200        // Summary-guided start offset: the covering sample for a token range's
201        // exclusive lower bound, else the index start for a full scan.
202        //
203        // WRAPAROUND (roborev endgame finding, High — silent data loss): a
204        // wraparound range (`start_excl > end_incl`, the ring segment crossing the
205        // min-token boundary) is `(start_excl, MAX] ∪ [MIN, end_incl]` — its
206        // in-range partitions live in TWO disjoint physical regions of the
207        // token-ordered `Index.db`: a HIGH-token tail (`token > start_excl`) and a
208        // LOW-token head (`token <= end_incl`). This walk is a single FORWARD pass
209        // to EOF (`Index.db` is not circular), so starting at
210        // `scan_start_position_for_token(start_excl)` (the HIGH segment's start)
211        // can NEVER reach the LOW segment's entries — they physically precede that
212        // offset and are silently skipped entirely. For a wraparound range the walk
213        // MUST start at the index's true beginning (offset 0) so BOTH segments are
214        // reachable; `ScanTokenBound::contains` (the per-entry filter below) already
215        // selects exactly the two segments, and `can_stop_past` is unconditionally
216        // `false` for a wraparound bound (verified: the walk never early-stops
217        // before EOF), so the pair is coherent — a full walk, filtered.
218        //
219        // Wrapping is DERIVED from the endpoints (#3634), so the FULL ring
220        // (`start_excl == end_incl`, #2228) takes this arm too — as it must: it
221        // admits every token, so the walk has to start at the true beginning.
222        let start_position = match token_bound {
223            Some(bound) if bound.is_wraparound() => 0,
224            Some(bound) => summary.scan_start_position_for_token(bound.start_excl),
225            None => 0,
226        };
227
228        // Derive the `Index.db` path from the reader's CURRENT (possibly
229        // #2383-rebound) `Data.db` path, NOT the index_reader's recorded open-time
230        // path: a warm reader rebound across a snapshot teardown keeps its already
231        // -open `Data.db` FD (valid even after the snapshot dir is cleared) and
232        // swaps only `file_path`, but the streaming index read does a FRESH
233        // `File::open` — so it must follow the rebind or ENOENT on the dead
234        // snapshot path (#2352). Snapshot components are same-inode hardlinks, so
235        // the rebound path's `Index.db` is byte-identical to the open-time one.
236        let index_db_path = self.current_index_db_path();
237        let mut stream = crate::storage::sstable::index_reader::IndexEntryStream::open(
238            &index_db_path,
239            start_position,
240        )
241        .await?;
242
243        // One-entry lookahead so each partition body is bounded by its SUCCESSOR's
244        // offset (the last by the data-section end), exactly as the materialising
245        // walk bounds `entries[i]` by `entries[i+1]`.
246        let mut prev_key: Option<(i64, Vec<u8>)> = None;
247        let mut index = 0usize;
248        let mut emitted_any = false;
249        // Coalesced, chunk-aligned compressed-scan window (issue #2877): serves
250        // consecutive in-range partitions from one decompressed window instead
251        // of one `read_compressed_offset_window` call per partition. Unused
252        // (never filled) on the uncompressed branch.
253        let mut compressed_window = CompressedScanWindow::new();
254        let Some(mut current) = stream.next().await? else {
255            // Zero entries from the start offset: nothing in range (or empty).
256            return Ok(FullIndexStreamOutcome::Streamed);
257        };
258
259        loop {
260            scan_cancel.check()?;
261
262            let cur_key: Vec<u8> = current
263                .raw_key
264                .as_deref()
265                .map(|k| k.to_vec())
266                .unwrap_or_default();
267            // Empty partition keys round-trip corrupt through BOTH read paths
268            // (issue #2302/#2325): before any emit, fall back safely; after an
269            // emit, fail closed rather than feed the merger a corrupt shape.
270            if cur_key.is_empty() {
271                if emitted_any {
272                    return Err(Error::corruption(
273                        "walk_in_range_partition_slices: empty partition key mid-walk \
274                         (issue #2302/#2325)"
275                            .to_string(),
276                    ));
277                }
278                return Ok(FullIndexStreamOutcome::FellBack);
279            }
280
281            let token = cassandra_murmur3_token(&cur_key);
282
283            // (token, key) order guard — the merge-input contract (issue #2361).
284            check_token_order(
285                prev_key.as_ref().map(|(t, k)| (*t, k.as_slice())),
286                token,
287                &cur_key,
288                index,
289            )?;
290
291            // Read the successor to bound the current partition (and to decide the
292            // range stop). The successor of the LAST entry is the data-section end.
293            let next = stream.next().await?;
294            let end = next
295                .as_ref()
296                .map(|n| n.data_offset)
297                .unwrap_or(data_section_end);
298
299            // Token pushdown (#2413): decode only in-range bodies.
300            let in_range = token_bound.map(|b| b.contains(token)).unwrap_or(true);
301            if in_range {
302                let start = current.data_offset;
303                if end <= start {
304                    return Err(Error::corruption(format!(
305                        "walk_in_range_partition_slices: non-ascending offset at entry \
306                         {index} (offset {start} >= successor {end}, issue #2412)"
307                    )));
308                }
309                let span = end - start;
310                let Ok(size) = u32::try_from(span) else {
311                    return Err(Error::corruption(format!(
312                        "walk_in_range_partition_slices: partition {index} span {span} \
313                         overflows u32 (issue #2412)"
314                    )));
315                };
316
317                // SCAN intent (issue #2876): this forward Summary-guided walk reads
318                // Data.db in mostly-ascending order, so every read — the partition
319                // body AND (uncompressed) its covering `CRC.db` chunks — goes
320                // through the reader's never-`MADV_RANDOM` `scan_positional_source`,
321                // not the `MADV_RANDOM` point mapping whose suppressed readahead (#2210)
322                // would cost this walk ~one 4 KiB fault per partition.
323                //
324                // The #2877 coalescing window is the CONSUMER of that plane, never a
325                // bypass of it: it makes the SAME scan-plane reads fewer and larger
326                // (chunk-aligned, ramped to 4 MiB), which is precisely what the
327                // scan mapping's kernel readahead rewards. Issuing them on the
328                // advised point plane instead would reinstate the #2876 field
329                // regression while every per-PR test stayed green — the combined
330                // interaction `cqlite-core/tests/issue_2877_scan_chunk_coalescing.rs`
331                // pins (CASSANDRA-15452's lesson: a userspace scan buffer only helps
332                // when it reads the plane that actually reads ahead).
333                let scan_source = self.scan_positional_source.clone();
334                let raw: std::borrow::Cow<'_, [u8]> =
335                    if let Some(ci) = self.compression_info.as_deref() {
336                        std::borrow::Cow::Borrowed(
337                            compressed_window
338                                .slice(self, scan_source.as_ref(), ci, start, end, data_section_end)
339                                .await?,
340                        )
341                    } else {
342                        std::borrow::Cow::Owned(
343                            self.read_uncompressed_verified(
344                                scan_source.as_ref(),
345                                &self.file,
346                                start + self.actual_header_size as u64,
347                                size as usize,
348                            )
349                            .await?,
350                        )
351                    };
352                let raw: &[u8] = &raw;
353
354                // Structural coverage (Signal B): the slice must decode as exactly
355                // one complete partition. Mid-walk failure = fail-closed.
356                if !self.partition_slice_fully_consumed(parser, raw, schema)? {
357                    return Err(Error::corruption(format!(
358                        "walk_in_range_partition_slices: partition {index} slice not fully \
359                         consumed (truncated/corrupt body, issue #2412)"
360                    )));
361                }
362
363                // Work-probe (issue #2398): one partition body decoded. A narrow
364                // token range must keep this near its in-range slice, not O(all).
365                crate::storage::sstable::work_counters::add_stream_walk_partition_parsed();
366                emitted_any = true;
367                match decode(raw)? {
368                    ControlFlow::Continue(()) => {}
369                    ControlFlow::Break(()) => return Ok(FullIndexStreamOutcome::Streamed),
370                }
371            }
372
373            // Advance. Stop early once a non-wraparound range is fully past its end.
374            if let Some(bound) = token_bound {
375                if bound.can_stop_past(token) {
376                    return Ok(FullIndexStreamOutcome::Streamed);
377                }
378            }
379            match next {
380                Some(n) => {
381                    prev_key = Some((token, cur_key));
382                    current = n;
383                    index += 1;
384                }
385                None => break,
386            }
387        }
388
389        // Terminus completeness (Signal A, issue #2302): for a FULL scan a
390        // mid-entry-truncated `Index.db` tail must be detectable — after an emit
391        // the streaming walk cannot fall back, so this is a hard fail-closed error
392        // (never a silent under-enumeration). A token-range scan does not enumerate
393        // the tail, so a truncated tail beyond its range is not its concern.
394        if token_bound.is_none() && stream.truncated_tail() {
395            return Err(Error::corruption(
396                "walk_in_range_partition_slices: Index.db tail truncated mid-entry — \
397                 full-scan enumeration is incomplete (Signal A, issue #2302/#2412)"
398                    .to_string(),
399            ));
400        }
401
402        Ok(FullIndexStreamOutcome::Streamed)
403    }

405    /// Read-shadowing ScanRow decoder over the shared walk (the non-stitching
406    /// `V5_0Uncompressed` model): each in-range partition slice is decoded with the
407    /// read-shadowing parser (`build_v5_parser(true)`) and its surviving rows are
408    /// emitted as `(RowKey, ScanRow)` — byte-identical to
409    /// [`stream_all_partitions_via_full_index`](Self::iterate_all_partitions_via_full_index)'s
410    /// per-partition emit, but streamed (no resident `Vec`) + token-scoped.
411    ///
412    /// `now_secs` (issue #3058): when `Some`, the caller's request-scoped
413    /// read-time TTL clock is pinned onto the read-shadowing parser instead of
414    /// the ambient one it samples at construction, so a caller that already
415    /// captured ONE reconciliation `now` (the Flight producer) expires TTL cells
416    /// against exactly that instant. `None` keeps the ambient sample.
417    ///
418    /// `caller_schema` (issue #3058): the AUTHORITATIVE table schema, when the
419    /// caller has one (the Flight producer's ticket-DDL schema). It must take
420    /// precedence over the reader's own four-tier lookup exactly as the
421    /// compaction sibling's `schema` parameter does — an `nb` SSTable header
422    /// carries no embedded schema, so decoding it with the reader-derived one
423    /// loses the clustering-key columns (they surface as NULL). `None` keeps the
424    /// reader-derived resolution.
425    pub(in crate::storage::sstable::reader) async fn stream_partitions_summary_guided<F>(
426        &self,
427        scan_cancel: &ScanCancel,
428        token_bound: Option<ScanTokenBound>,
429        now_secs: Option<i64>,
430        caller_schema: Option<&crate::schema::TableSchema>,
431        emit: &mut F,
432    ) -> Result<FullIndexStreamOutcome>
433    where
434        F: FnMut((RowKey, ScanRow)) -> Result<ControlFlow<()>>,
435    {
436        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
437        let parser = self.build_v5_parser(true);
438        let parser = match now_secs {
439            Some(now) => parser.with_now_secs(now),
440            None => parser,
441        };
442        let reader_schema = caller_schema
443            .cloned()
444            .or_else(|| self.get_table_schema(None));
445        let schema = reader_schema.as_ref();
446        self.walk_in_range_partition_slices(scan_cancel, token_bound, &parser, schema, &mut |raw| {
447            // #3782: `walk_in_range_partition_slices` runs
448            // `partition_slice_fully_consumed` per slice before calling this
449            // decode, so `raw` is a proven-complete partition extent.
450            let parsed = parser.parse_block(raw, BufferExtent::Complete, schema, self)?;
451            for (_table_id, row_key, value) in parsed {
452                if self.filter_tombstone(&value) {
453                    match emit((row_key, value))? {
454                        ControlFlow::Continue(()) => {}
455                        ControlFlow::Break(()) => return Ok(ControlFlow::Break(())),
456                    }
457                }
458            }
459            Ok(ControlFlow::Continue(()))
460        })
461        .await
462    }

464    /// Full-fidelity CompactionRow decoder over the shared walk (the chunk-stitching
465    /// `nb` merge model): each in-range partition slice is decoded with the
466    /// compaction parser (`build_v5_parser(false)`) via
467    /// `parse_one_partition_for_compaction` — byte-identical to
468    /// [`drain_compaction_window`](Self::stream_all_partitions_for_compaction)'s
469    /// per-partition [`CompactionRow`](super::super::compaction_row::CompactionRow)
470    /// emit (preserving cell timestamps / tombstone markers the k-way merger's LWW
471    /// reconciliation needs), but streamed + token-scoped. `at_final_chunk = true`
472    /// because `raw` is exactly one complete, coverage-proven partition slice.
473    async fn stream_partitions_summary_guided_compaction<F>(
474        &self,
475        scan_cancel: &ScanCancel,
476        token_bound: Option<ScanTokenBound>,
477        schema: Option<&crate::schema::TableSchema>,
478        emit: &mut F,
479    ) -> Result<FullIndexStreamOutcome>
480    where
481        F: FnMut(super::super::compaction_row::CompactionRow) -> Result<ControlFlow<()>>,
482    {
483        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
484        use crate::storage::sstable::reader::parsing::ParseStep;
485        let parser = self.build_v5_parser(false);
486        let owned_schema = schema.cloned().or_else(|| self.get_table_schema(None));
487        let sch = owned_schema.as_ref();
488        self.walk_in_range_partition_slices(scan_cancel, token_bound, &parser, sch, &mut |raw| {
489            let mut broke = false;
490            let step = parser.parse_one_partition_for_compaction(
491                raw,
492                sch,
493                self,
494                true,
495                &mut |row: super::super::compaction_row::CompactionRow| match emit(row)? {
496                    ControlFlow::Continue(()) => Ok(ControlFlow::Continue(())),
497                    ControlFlow::Break(()) => {
498                        broke = true;
499                        Ok(ControlFlow::Break(()))
500                    }
501                },
502            )?;
503            // The coverage check already proved `raw` decodes as exactly one
504            // complete partition, so a non-`Emitted` step here is a structural
505            // inconsistency — fail closed rather than silently drop the partition.
506            if !matches!(step, ParseStep::Emitted(_)) {
507                return Err(Error::corruption(
508                    "stream_partitions_summary_guided_compaction: coverage-proven slice \
509                     did not decode to a terminated partition (issue #2412)"
510                        .to_string(),
511                ));
512            }
513            if broke {
514                return Ok(ControlFlow::Break(()));
515            }
516            Ok(ControlFlow::Continue(()))
517        })
518        .await
519    }
520
521    /// Query-serve partition enumeration for the WARM reader-based merge (issue
522    /// #2412 §C / #2413 Option A) — the analogue of
523    /// [`stream_all_partitions_for_compaction`](Self::stream_all_partitions_for_compaction)
524    /// that the flight `do_get` warm path drives (`from_readers::drive_query_stream`).
525    ///
526    /// Routing MIRRORS `stream_all_partitions_for_compaction`'s per-mechanism emit
527    /// shape so the merger reconciles identically, but STREAMS the index (no
528    /// resident `Vec`, spec R4) and pushes the token range into the walk (#2413):
529    /// - Chunk-stitching (`nb`) readers use the full-fidelity CompactionRow decoder
530    ///   ([`stream_partitions_summary_guided_compaction`](Self::stream_partitions_summary_guided_compaction)) —
531    ///   byte-identical to `drain_compaction_window`'s emit, so cross-generation
532    ///   LWW reconciliation is preserved.
533    /// - Non-stitching (`V5_0Uncompressed`) readers use the read-shadowing ScanRow
534    ///   decoder ([`stream_partitions_summary_guided`](Self::stream_partitions_summary_guided)) —
535    ///   byte-identical to `stream_all_partitions_cancellable`'s emit.
536    ///
537    /// A `FellBack` (no usable summary/index) routes on to the full-ring
538    /// `stream_all_partitions_for_compaction`. Compaction consumers do NOT call
539    /// this (they use the path-based stream directly, no token range), so they keep
540    /// full-ring byte-parity walks unchanged (spec R3).
541    pub async fn stream_all_partitions_for_query<F>(
542        &self,
543        schema: Option<&crate::schema::TableSchema>,
544        scan_cancel: &ScanCancel,
545        token_bound: Option<ScanTokenBound>,
546        mut emit: F,
547    ) -> Result<()>
548    where
549        F: FnMut(super::super::compaction_row::CompactionRow) -> Result<ControlFlow<()>>,
550    {
551        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
552        let summary_usable = self
553            .summary_reader
554            .as_ref()
555            .map(|s| !s.get_entries().is_empty())
556            .unwrap_or(false);
557        if self.index_reader.is_some() && self.bti_partitions_db.is_none() && summary_usable {
558            let outcome = if self.requires_chunk_stitching() {
559                // `nb` merge model: full-fidelity CompactionRows (cell timestamps /
560                // tombstone markers preserved for the merger's LWW reconciliation).
561                self.stream_partitions_summary_guided_compaction(
562                    scan_cancel,
563                    token_bound,
564                    schema,
565                    &mut emit,
566                )
567                .await?
568            } else {
569                // `V5_0Uncompressed` model: read-shadowing rows adapted to
570                // CompactionRow (row_timestamp 0), matching the non-stitching
571                // compaction stream's emit exactly.
572                //
573                // `caller_schema` is the caller's AUTHORITATIVE `schema` (issue
574                // #3097): the merge arm previously passed `None` here and resolved
575                // the decode schema from the reader's own four-tier lookup, which
576                // for a `V5_0Uncompressed` reader (an `nb` header carries no
577                // embedded schema) is a header-derived schema whose clustering
578                // columns bear the placeholder name `clustering_key` — so a ticket
579                // whose DDL declares a real clustering key (`ck`) decoded that
580                // column under the wrong name (surfacing as NULL to a projected
581                // `SELECT`). Passing `Some(schema)` makes the merge arm honour the
582                // caller's schema EXACTLY as the chunk-stitching sibling
583                // (`stream_partitions_summary_guided_compaction`) and the fast arm
584                // (`query_rows::drive_query_rows`) already do. `stream_partitions_
585                // summary_guided`/`walk_in_range_partition_slices` still fall back
586                // to the reader-derived lookup when the caller passes `None`
587                // (e.g. `stream_all_partitions_for_query(None, …)`), so the no-
588                // caller-schema behaviour is preserved. Compaction never routes
589                // through this method (it uses `stream_all_partitions_for_
590                // compaction` directly), so byte-parity walks are untouched.
591                self.stream_partitions_summary_guided(
592                    scan_cancel,
593                    token_bound,
594                    None,
595                    schema,
596                    &mut |(k, v)| {
597                        let row =
598                            super::super::compaction_row::CompactionRow::from_legacy_value(k, v, 0);
599                        emit(row)
600                    },
601                )
602                .await?
603            };
604            if matches!(outcome, FullIndexStreamOutcome::Streamed) {
605                return Ok(());
606            }
607            tracing::warn!(
608                "stream_all_partitions_for_query: Summary-guided streaming fell back \
609                 (no usable Summary.db/Index.db to stream); using the full-ring \
610                 compaction stream (issue #2412)."
611            );
612        }
613        // Full-ring fallback: the routes below take no `token_bound` in their
614        // signatures, so the range is applied to their emit instead of being
615        // dropped (issue #3358). A BTI (`da`) reader reaches this fallback on
616        // EVERY call — `bti_partitions_db.is_some()` fails the gate above — so
617        // before this it was not a fallback for that format but the only route,
618        // and `stream_all_partitions_for_query(.., Some(bound), ..)` returned the
619        // whole ring. An `nb` generation holding the same rows returned exactly
620        // the range, which `issue_2412_wraparound_scan.rs` pins with
621        // `assert_eq!`, so the two formats disagreed about what the parameter
622        // means.
623        //
624        // The gate is applied here, once, rather than inside each route: it is
625        // the boundary the caller's range was handed to, and the two routes below
626        // reach three different walk implementations between them.
627        //
628        // QUERY-ARM caller-schema fidelity (issue #3097): the summary-guided walk
629        // above honours the caller's authoritative `schema`, but this fallback
630        // fires when `Summary.db` is absent/unusable (or the walk `FellBack`). For
631        // a non-stitching `V5_0Uncompressed` reader, `stream_all_partitions_for_
632        // compaction` delegates its non-stitch branch to
633        // `stream_all_partitions_cancellable`, whose full-index/sequential routes
634        // resolve the decode schema from the READER only — so a ticket-DDL
635        // clustering key (`ck`) would again surface as NULL exactly as in the
636        // summary-guided branch this issue fixed. So the QUERY arm threads its
637        // caller `schema` through `stream_all_partitions_cancellable` directly for
638        // that branch (adapting `ScanRow` → `CompactionRow` byte-identically to the
639        // compaction non-stitch branch). Compaction never routes through this
640        // method, and its own `stream_all_partitions_cancellable` call passes
641        // `None`, so compaction's effective decode schema is unchanged (#3097).
642        //
643        // Chunk-stitching (`nb`) and BTI (`da`) readers keep routing through
644        // `stream_all_partitions_for_compaction`, whose stitch/BTI decode already
645        // honours the passed `schema` (`schema.cloned().or_else(...)`), so no
646        // divergence is introduced there.
647        let mut gate = TokenGate::new(token_bound);
648        let mut emit = move |row: super::super::compaction_row::CompactionRow| {
649            if !gate.admits(&row.key.0) {
650                return Ok(ControlFlow::Continue(()));
651            }
652            emit(row)
653        };
654        if self.requires_chunk_stitching() || self.bti_partitions_db.is_some() {
655            return self
656                .stream_all_partitions_for_compaction(schema, scan_cancel, emit)
657                .await;
658        }
659        self.stream_all_partitions_cancellable(scan_cancel, schema, |(key, value)| {
660            let row = super::super::compaction_row::CompactionRow::from_legacy_value(key, value, 0);
661            emit(row)
662        })
663        .await
664    }
665}

667#[cfg(test)]
668mod tests {
669    /// The gate's own tests. The end-to-end pin, over a Cassandra-written BTI
670    /// generation through `stream_all_partitions_for_query`, is
671    /// `tests/issue_3358_bti_query_token_bound.rs`; these cover the parts that
672    /// file cannot isolate.
673    mod token_gate {
674        use super::super::{ScanTokenBound, TokenGate};
675        use crate::util::cassandra_murmur3::cassandra_murmur3_token;
676        use std::sync::Arc;
677
678        fn key(bytes: &[u8]) -> Arc<[u8]> {
679            Arc::from(bytes.to_vec().into_boxed_slice())
680        }
681
682        #[test]
683        fn no_bound_admits_every_key() {
684            let mut gate = TokenGate::new(None);
685            for bytes in [&b"a"[..], b"b", b"a"] {
686                assert!(gate.admits(&key(bytes)));
687            }
688        }
689
690        #[test]
691        fn a_bound_admits_its_own_partition_only() {
692            let inside = key(b"inside");
693            let outside = key(b"outside");
694            let token = cassandra_murmur3_token(&inside);
695            let bound = ScanTokenBound {
696                start_excl: token - 1,
697                end_incl: token,
698            };
699            assert!(
700                !bound.contains(cassandra_murmur3_token(&outside)),
701                "the two keys must fall either side of the bound, or this case \
702                 cannot discriminate"
703            );
704            let mut gate = TokenGate::new(Some(bound));
705            assert!(gate.admits(&inside));
706            assert!(!gate.admits(&outside));
707        }
708
709        /// The memo holds one key's verdict, so it must be recomputed the moment
710        /// the key changes — including when the walk returns to a key it has seen.
711        #[test]
712        fn the_memo_follows_the_partition_boundary() {
713            let inside = key(b"inside");
714            let outside = key(b"outside");
715            let token = cassandra_murmur3_token(&inside);
716            let mut gate = TokenGate::new(Some(ScanTokenBound {
717                start_excl: token - 1,
718                end_incl: token,
719            }));
720            let verdicts: Vec<bool> = [&inside, &inside, &outside, &outside, &inside]
721                .into_iter()
722                .map(|k| gate.admits(k))
723                .collect();
724            assert_eq!(verdicts, vec![true, true, false, false, true]);
725        }
726    }
