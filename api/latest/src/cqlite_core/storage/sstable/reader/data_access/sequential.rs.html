33/// Classify a `parse_block` failure on the compressed (`V5CompressedLegacy`)
34/// point-lookup path (`scan_for_key`) as a soft-miss vs. a fatal error.
35///
36/// Issue #1411: the scan path ([`SSTableReader::stitch_and_parse_all_chunks`])
37/// propagates every `parse_block` error via `?`. The point-lookup path mirrors
38/// that so `get()` and `scan()` agree on which failures are fatal — with ONE
39/// deliberate exception, the case the original blanket `Err(_) => Ok(None)`
40/// existed to protect: this reader has no schema for the table, so it cannot
41/// serve the key and the caller must fall through to the next SSTable reader.
42///
43/// Returns `true` (soft-miss → `Ok(None)`) ONLY when both hold:
44/// - `schema_present == false` — the reader supplied no schema, and
45/// - the parser reported [`Error::Schema`] — which `V5CompressedLegacy`'s
46///   `parse_block` raises ("requires schema for <ks>.<table>") *before* it
47///   inspects any bytes when no schema is available.
48///
49/// Every other failure is non-recoverable and must propagate: real data
50/// corruption / malformed blocks (`Corruption`, `InvalidFormat`, `Parse`, I/O,
51/// etc.), and — critically — a deep schema/type-resolution `Error::Schema`
52/// raised when a schema IS present (a UDT/type that cannot be resolved is a real
53/// error, not a missing key). Requiring `!schema_present` prevents that class
54/// from ever being masked as "not found".
55fn is_parse_soft_miss(schema_present: bool, err: &Error) -> bool {
56    !schema_present && matches!(err, Error::Schema(_))
57}

60    /// Scan a range of keys
61    ///
62    /// # Arguments
63    /// * `table_id` - The table to scan
64    /// * `start_key` - Optional start key for range scan
65    /// * `end_key` - Optional end key for range scan
66    /// * `limit` - Optional limit on number of results
67    /// * `schema` - Optional table schema for schema-aware parsing. When provided,
68    ///   enables accurate type detection and avoids heuristic-based parsing.
69    ///   Strongly recommended for Cassandra 5.0+ formats.
70    ///
71    /// # Error counting (issue #1704)
72    ///
73    /// A failure is counted ONCE into
74    /// `cqlite.errors.total{category, subsystem="reader"}` at this boundary. This
75    /// is the top-level exit seam of the MATERIALIZING scan (the streaming
76    /// surfaces are counted at [`JoinedStream::recv`], and `scan_delta` at its own
77    /// terminal send), so the inner steps it delegates to — `sequential_scan`,
78    /// `bti_scan_with_metadata`, the index walk — deliberately do NOT count: they
79    /// are also reached from `iterate_all_partitions` and the full-index stream,
80    /// and counting there too would report one failed scan two or three times.
81    /// The category comes from the classifier via
82    /// [`crate::observability::record_result`], never from this call site, and the
83    /// `Err` is returned unchanged.
84    pub async fn scan(
85        &self,
86        table_id: &TableId,
87        start_key: Option<&RowKey>,
88        end_key: Option<&RowKey>,
89        limit: Option<usize>,
90        schema: Option<&crate::schema::TableSchema>,
91    ) -> Result<Vec<(RowKey, ScanRow)>> {
92        crate::observability::record_result(
93            "reader",
94            self.scan_inner(table_id, start_key, end_key, limit, schema)
95                .await,
96        )
97    }

99    /// Implementation of [`scan`](Self::scan); see there for the contract. Kept
100    /// separate so the error-counting seam is a single, unmissable wrapper rather
101    /// than a `record_error` at each of this function's several early returns.
102    async fn scan_inner(
103        &self,
104        table_id: &TableId,
105        start_key: Option<&RowKey>,
106        end_key: Option<&RowKey>,
107        limit: Option<usize>,
108        schema: Option<&crate::schema::TableSchema>,
109    ) -> Result<Vec<(RowKey, ScanRow)>> {
110        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
111        tracing::debug!("SSTableReader::scan - Starting scan");
112        tracing::debug!("SSTableReader::scan - File path: {:?}", self.file_path());
113        tracing::debug!("SSTableReader::scan - Table ID: {}", table_id);
114        tracing::debug!("SSTableReader::scan - Start key: {:?}", start_key);
115        tracing::debug!("SSTableReader::scan - End key: {:?}", end_key);
116        tracing::debug!("SSTableReader::scan - Limit: {:?}", limit);
117        tracing::debug!("SSTableReader::scan - Has schema: {}", schema.is_some());
118        tracing::debug!("SSTableReader::scan - Has index: {}", self.index.is_some());
119        tracing::debug!(
120            "SSTableReader::scan - Has bloom filter: {}",
121            self.bloom_filter.is_some()
122        );
123
124        // Issue #660: BTI ("da") readers have no Index.db/Summary.db. A full scan
125        // walks the whole (chunk-compressed) Data.db once and parses every
126        // partition — the same partition decode the point-lookup path proves
127        // correct, but emitting ALL partitions instead of stopping at the first.
128        if self.bti_partitions_db.is_some() {
129            let entries = self
130                .bti_scan_with_metadata(start_key, end_key, limit, schema, true, None)
131                .await?;
132            return Ok(entries.into_iter().map(|(k, v, _meta)| (k, v)).collect());
133        }
134
135        let mut results = Vec::new();
136
137        // Use index for efficient range scan if available
138        if let Some(index) = &self.index {
139            tracing::debug!("SSTableReader::scan - Using index-based scan");
140            let entries = index.get_range(table_id, start_key, end_key)?;
141            tracing::debug!(
142                "SSTableReader::scan - Index returned {} entries",
143                entries.len()
144            );
145
146            // Issue #256 FIX: Fall back to sequential scan when index returns no entries
147            //
148            // This handles BTI (Big Trie Index) format where parsing may be incomplete or
149            // where the index format is not yet fully supported. Without this check, tables
150            // using BTI format return 0 rows because:
151            // 1. The index exists (so we take the index-based path)
152            // 2. But get_range() returns 0 entries (BTI parsing incomplete)
153            // 3. The has_zero_size check never triggers (no entries to check)
154            // 4. The for loop iterates 0 times, returning empty results
155            //
156            // Sequential scan correctly parses Data.db directly, bypassing index issues.
157            if entries.is_empty() {
158                tracing::debug!(
159                    "SSTableReader::scan - Index returned 0 entries (BTI format or incomplete parsing), falling back to sequential scan"
160                );
161                return self
162                    .sequential_scan(
163                        table_id,
164                        start_key,
165                        end_key,
166                        limit,
167                        schema,
168                        &self.scan_cancel,
169                    )
170                    .await;
171            }
172
173            // Check if any entry has size=0 (Cassandra 5.0 format)
174            let has_zero_size = entries.iter().any(|e| e.size == 0);
175            if has_zero_size {
176                tracing::debug!("SSTableReader::scan - Index reports size=0 for some entries, using sequential scan fallback");
177                return self
178                    .sequential_scan(
179                        table_id,
180                        start_key,
181                        end_key,
182                        limit,
183                        schema,
184                        &self.scan_cancel,
185                    )
186                    .await;
187            }
188
189            // Collect ALL index entries (limit applied after sort — BLOCKING-1).
190            for (i, entry) in entries.iter().enumerate() {
191                // Index offsets are relative to data section start - adjust for header
192                let file_offset = entry.offset + self.actual_header_size as u64;
193                tracing::debug!(
194                    "SSTableReader::scan - Processing index entry {}: index_offset={}, file_offset={}, size={}",
195                    i, entry.offset, file_offset, entry.size
196                );
197
198                // POINT intent, deliberately (issue #2876, roborev job 4634).
199                //
200                // This legacy index path looks like a scan but does NOT read
201                // sequentially: `Index::get_range` walks `sorted_keys`, i.e. raw
202                // key-BYTE order (index.rs:270-277), while Data.db is laid out in
203                // Murmur3 TOKEN order. For the default partitioner those two orders
204                // are uncorrelated, so these reads are genuinely scattered and
205                // `MADV_RANDOM` is the CORRECT advice — the same reasoning that put
206                // point lookups on the advised plane in #2210. Routing them to the
207                // scan plane would invite readahead that mostly fetches pages this
208                // walk never touches.
209                //
210                // Do NOT "fix" this by sorting `entries` by `entry.offset` before
211                // reading to make the access sequential: `results` is sorted into
212                // token order AFTER the loop (`sort_by_token_order`, see below) and
213                // `limit` is applied AFTER that sort, so LIMIT N must mean "the N
214                // token-smallest partitions". Read order is independent of result
215                // order here, so an offset-ordered read is safe in principle — but
216                // it is a separate optimization with its own regression-test burden,
217                // NOT part of removing a wrong advice from the true scan walks.
218                // Tracked as follow-up; this line stays on the point plane.
219                if let Some(value) = self.read_value_at_offset(file_offset, entry.size).await? {
220                    tracing::debug!(
221                        "SSTableReader::scan - Successfully read value at offset {}",
222                        entry.offset
223                    );
224                    results.push((entry.key.clone(), value));
225                } else {
226                    tracing::debug!("SSTableReader::scan - Value at offset {} was filtered out (tombstone or expired)", entry.offset);
227                }
228            }
229        } else {
230            // Fallback to sequential scan.  sequential_scan() already returns results in
231            // token order (NON-BLOCKING-1: avoid double-sort — return directly).
232            tracing::debug!("SSTableReader::scan - No index, falling back to sequential scan");
233            let seq_results = self
234                .sequential_scan(
235                    table_id,
236                    start_key,
237                    end_key,
238                    limit,
239                    schema,
240                    &self.scan_cancel,
241                )
242                .await?;
243            tracing::debug!(
244                "SSTableReader::scan - Sequential scan returned {} results",
245                seq_results.len()
246            );
247            tracing::debug!(
248                "SSTableReader::scan - Returning {} final results",
249                seq_results.len()
250            );
251            return Ok(seq_results);
252        }
253
254        // Index-based path: sort by Murmur3 token order (ascending token, then key bytes).
255        // This matches the on-disk physical order (spec §5, Appendix B §313) and the write
256        // engine's DecoratedKey::cmp.  Compute each key's token once before sorting to
257        // avoid O(n log n) recomputation inside the comparator.
258        sort_by_token_order(&mut results);
259        // Limit applied AFTER sort so LIMIT N returns the N token-smallest partitions.
260        if let Some(lim) = limit {
261            results.truncate(lim);
262        }
263
264        tracing::debug!(
265            "SSTableReader::scan - Returning {} final results",
266            results.len()
267        );
268        Ok(results)
269    }

271    /// Get all entries in the SSTable.
272    ///
273    /// # Tombstone contract (Issue #505)
274    ///
275    /// This is a **user-facing** accessor: row tombstones are filtered out via
276    /// [`Self::filter_tombstone`] and never appear in the returned entries. The
277    /// underlying `parse_block` path emits `Value::Tombstone(RowTombstone)` for
278    /// deleted rows, but those are suppressed here so callers see exactly the live
279    /// rows (matching the previous `Value::Null` suppression behaviour).
280    ///
281    /// The compaction k-way merger must instead use
282    /// [`Self::iterate_all_partitions_for_compaction`], which preserves
283    /// `Value::Tombstone` entries (with their authoritative deletion timestamps)
284    /// so that tombstone-shadowing semantics can be applied during the merge.
285    pub async fn get_all_entries(&self) -> Result<Vec<(TableId, RowKey, ScanRow)>> {
286        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
287
288        // Issue #660: BTI ("da") tables have no Index.db; route through the
289        // whole-Data.db BTI scan, which resolves schema via get_table_schema
290        // (header/registry) and decodes every partition. It mints its own
291        // per-scan cursor, as does the non-BTI path below (issue #815).
292        if self.bti_partitions_db.is_some() {
293            let table_id = TableId::new(format!(
294                "{}.{}",
295                self.header.keyspace, self.header.table_name
296            ));
297            let entries = self
298                .bti_scan_with_metadata(None, None, None, None, false, None)
299                .await?;
300            return Ok(entries
301                .into_iter()
302                .map(|(k, v, _meta)| (table_id.clone(), k, v))
303                .collect());
304        }
305
306        // Issue #815: independent per-scan cursor — no cross-scan serialization.
307        let cursor = self.new_scan_cursor().await?;
308
309        let mut results = Vec::new();
310
311        // Reset to beginning of data section
312        let header_size = self.calculate_header_size();
313        {
314            let mut file_guard = cursor.file.lock().await;
315            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
316        }
317
318        if self.requires_chunk_stitching() {
319            // V5CompressedLegacy: Row payloads can span multiple compressed chunks
320            // We must decompress and stitch all chunks together before parsing
321            tracing::debug!(
322                "V5CompressedLegacy format detected, decompressing and stitching all chunks before parsing"
323            );
324
325            // Use shared stitching helper method. Physical enumeration
326            // (`get_all_entries`) honours the reader's own cancel field —
327            // unchanged pre-#2346 behaviour.
328            let entries = self
329                .stitch_and_parse_all_chunks(&cursor, None, false, &self.scan_cancel)
330                .await?;
331            results.extend(entries);
332        } else {
333            // Other formats: Read and parse blocks individually
334            while let Some(block) = self.read_next_block(&cursor).await? {
335                let entries = self.parse_block_entries(&block, None, false)?;
336                results.extend(entries);
337            }
338        }
339
340        // Issue #505: suppress row tombstones from user-facing output. The compaction
341        // path (iterate_all_partitions_for_compaction) bypasses this filter.
342        results.retain(|(_tid, _key, value)| self.filter_tombstone(value));
343
344        Ok(results)
345    }

347    pub(super) async fn scan_for_key(
348        &self,
349        table_id: &TableId,
350        key: &RowKey,
351    ) -> Result<Option<ScanRow>> {
352        // Issue #831: record the call so tests can assert the BTI point-lookup
353        // path never reaches the sequential scan.
354        SCAN_FOR_KEY_CALLS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
355
356        // #3853 scan-lifetime madvise seam. This site is NOT one of issue
357        // #3853's nine entry points and is a DELIBERATE extension, because a
358        // reviewer will otherwise ask why a point lookup takes a SCAN guard.
359        // Three reasons, and a fourth that follows from the code below:
360        //
361        // (a) It demonstrably reads the WHOLE data section. Both branches do:
362        //     the stitching branch stitches every chunk, and the non-stitching
363        //     block loop runs to EOF. By the seam's own invariant that makes it
364        //     a scan, whatever the caller's intent was.
365        // (b) There is no point-plane degradation to trade against. The arming
366        //     gate (`scan_lifetime::resolve`) enables the seam ONLY when
367        //     `!Arc::ptr_eq(point_plane_mmap, scan_mmap)`, so advising the scan
368        //     mapping provably cannot touch the #2210 `MADV_RANDOM` POINT
369        //     mapping — they are different allocations or the seam is disabled.
370        // (c) The `DONTNEED` at the end is actively DESIRABLE here: a whole-file
371        //     read performed to answer ONE key is precisely the resident-set
372        //     footprint worth releasing.
373        // (d) A guard here covers BOTH branches, whereas the funnel guard in
374        //     `stitch_all_chunks_cancellable` covers only the stitching one.
375        //
376        // The guard is taken before the early `Ok(None)` soft-miss returns and
377        // before the found-key early return, so every exit releases via `Drop`.
378        let _scan = self.begin_scan();
379
380        // Issue #815: independent per-scan cursor — no cross-scan serialization.
381        let cursor = self.new_scan_cursor().await?;
382
383        let header_size = self.calculate_header_size();
384
385        // For V5CompressedLegacy NB format, partitions can span chunk boundaries.
386        // The block-by-block parser will miss any partition whose bytes cross a
387        // chunk boundary.  Use the same stitched-buffer path that sequential_scan()
388        // uses so that get() and scan() share a consistent view of the data.
389        // (Issue #517)
390        if self.requires_chunk_stitching() {
391            tracing::debug!(
392                "scan_for_key: V5CompressedLegacy NB detected, using stitched buffer for key lookup"
393            );
394            // `stitch_all_chunks` reads from the CURRENT cursor position forward,
395            // so its precondition is "seeked to the data-section start" (the fresh
396            // cursor's chunk index already starts at 0). Each call uses its own
397            // cursor (issue #815), so there is no cross-call position to reset.
398            {
399                let mut file_guard = cursor.file.lock().await;
400                file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
401            }
402
403            // Issue #1411: separate the chunk-integrity stitch from the schema-aware
404            // parse so their failure classes are not conflated. `stitch_all_chunks`
405            // drives the authoritative per-chunk CRC32 check
406            // (`block_io::read_nb_format_chunk_data`) + decompression; a CRC mismatch,
407            // decompression failure, or corrupt offset is corruption and MUST surface
408            // (propagate via `?`, matching `scan`), never be masked as a missing key.
409            // Only the `parse_block` step below may soft-miss (schema unavailable /
410            // wrong table type) so the caller can try the next reader. Previously
411            // `stitch_and_parse_all_chunks` combined both under a blanket
412            // `Err(_) => Ok(None)` that swallowed CRC corruption on point lookups.
413            let stitched_buffer = self.stitch_all_chunks(&cursor).await?;
414
415            // Pass the reader's own schema so that V5CompressedLegacy rows can be fully
416            // parsed and their partition RowKeys emitted.  Without a schema, parse_row_v5
417            // fails for all rows in a partition, causing no entries to be pushed and making
418            // the key comparison always miss even when the key exists.
419            let schema_opt = self.get_table_schema(None);
420            // Issue #1741: point lookups apply SELECT-semantic read shadowing, so
421            // build the parser with read_shadowing = true. The stitch/parse split
422            // (issue #1411) is preserved: CRC/decompress failures already surfaced
423            // via `stitch_all_chunks` above; only `parse_block` may soft-miss.
424            let parser = self.build_v5_parser(true);
425            // #3782: `stitch_all_chunks` returned the WHOLE data section.
426            let all_entries = match parser.parse_block(
427                &stitched_buffer,
428                BufferExtent::Complete,
429                schema_opt.as_ref(),
430                self,
431            ) {
432                Ok(entries) => entries,
433                // Issue #1411 (roborev): the scan path (`stitch_and_parse_all_chunks`)
434                // propagates EVERY `parse_block` error via `?`. Mirror that here so
435                // `get()` and `scan()` agree on which errors are fatal. The ONLY
436                // legitimate soft-miss — the case the original blanket
437                // `Err(_) => Ok(None)` existed to protect — is classified by
438                // `is_parse_soft_miss`: this reader has no schema for the table, so it
439                // cannot serve the key and the caller must try the next SSTable reader.
440                // Every other failure (real corruption / malformed block, or a deep
441                // schema/type-resolution error when a schema IS present) MUST surface.
442                Err(e) if is_parse_soft_miss(schema_opt.is_some(), &e) => {
443                    tracing::debug!(
444                        "scan_for_key: no schema for this reader ({}); soft-miss so the caller tries the next reader: {}",
445                        table_id,
446                        e
447                    );
448                    return Ok(None);
449                }
450                Err(e) => return Err(e),
451            };
452
453            // NOTE: The SSTableIndex is built from 16-byte Murmur3 *digests*, not raw keys,
454            // so find_entry() always misses and falls through to this path.  For a found key
455            // we stop early (O(found position)); for a key not present we must scan the whole
456            // stitched buffer — O(file size).  This O(file) miss cost is an existing
457            // limitation of the digest-index design and is tracked separately as a follow-up.
458            //
459            // NON-BLOCKING-2: Table-id matching is intentionally skipped in the stitching path
460            // (consistent with sequential_scan's stitching path).  The V5CompressedLegacy parser
461            // returns entries tagged with the table_id from the SSTable header, which may hold
462            // default or incorrect values when headers use bare keyspace/table names rather than
463            // the query's fully-qualified form.  Since all entries in this stitch buffer come from
464            // the single SSTable being queried, skipping the check is correct and safe.
465            for (_, entry_key, entry_value) in all_entries {
466                if entry_key == *key {
467                    // Early-return on first match (BLOCKING-2: don't parse the rest of the file).
468                    if !self.filter_tombstone(&entry_value) {
469                        return Ok(None);
470                    }
471                    return Ok(Some(entry_value));
472                }
473            }
474
475            return Ok(None);
476        }
477
478        {
479            let mut file_guard = cursor.file.lock().await;
480            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
481        }
482
483        // Sequential scan through blocks
484        while let Some(block) = self.read_next_block(&cursor).await? {
485            let entries = self.parse_block_entries(&block, None, true)?;
486
487            for (entry_table_id, entry_key, entry_value) in entries {
488                if table_ids_match(&entry_table_id, table_id) && entry_key == *key {
489                    // Extract write time from entry metadata
490                    let _write_time = self.extract_write_time_from_entry(&entry_key, &entry_value);
491
492                    // Filter out tombstones and expired data
493                    if !self.filter_tombstone(&entry_value) {
494                        return Ok(None);
495                    }
496
497                    return Ok(Some(entry_value));
498                }
499            }
500        }
501
502        Ok(None)
503    }

505    /// `scan_cancel` is an explicit PER-CALL cancellation token (issue #2346).
506    /// Every existing caller passes `&self.scan_cancel` (the reader's own field,
507    /// unchanged pre-#2346 semantics); the compaction path
508    /// ([`SSTableReader::iterate_all_partitions_cancellable`]) passes its
509    /// caller-supplied token instead, so a shared/cached reader's two concurrent
510    /// scans cancel independently.
511    pub(in crate::storage::sstable::reader) async fn sequential_scan(
512        &self,
513        table_id: &TableId,
514        start_key: Option<&RowKey>,
515        end_key: Option<&RowKey>,
516        limit: Option<usize>,
517        schema: Option<&crate::schema::TableSchema>,
518        scan_cancel: &ScanCancel,
519    ) -> Result<Vec<(RowKey, ScanRow)>> {
520        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
521        tracing::debug!(
522            "SSTableReader::sequential_scan - starting: table_id={table_id}, has_schema={}",
523            schema.is_some()
524        );
525
526        // Issue #3109: BTI (`da`) readers decode through the authoritative trie
527        // walk, NEVER the block loop below. `iterate_all_partitions` is the one
528        // caller that reaches here with a `da` reader (both its index branches are
529        // gated on `bti_partitions_db.is_none()`), and that loop's state machine
530        // drops `read_shadowing` (and fails outright on a schema-required fixture).
531        // Posture is otherwise identical: both apply `filter_tombstone` + the key
532        // range + sort-then-`limit`. The PER-CALL `scan_cancel` is threaded through
533        // (#2264/#2346) — the non-cancellable wrapper polls the READER's own field
534        // instead, so a cancelled walk would stitch+parse the whole data section
535        // and return `Ok(every row)`.
536        if self.bti_partitions_db.is_some() {
537            let entries = self
538                .bti_scan_with_metadata_cancellable(
539                    start_key,
540                    end_key,
541                    limit,
542                    schema,
543                    true,
544                    None,
545                    scan_cancel,
546                )
547                .await?;
548            return Ok(entries.into_iter().map(|(k, v, _meta)| (k, v)).collect());
549        }
550
551        // Issue #815: each scan uses its own cursor (private file position and
552        // chunk index), so concurrent scans on this reader run in parallel
553        // without the per-scan serialization #805 introduced for correctness.
554        let cursor = self.new_scan_cursor().await?;
555
556        let mut results = Vec::new();
557
558        let header_size = self.calculate_header_size();
559        tracing::debug!(
560            "SSTableReader::sequential_scan - Header size: {} bytes",
561            header_size
562        );
563
564        {
565            let mut file_guard = cursor.file.lock().await;
566            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
567            tracing::debug!(
568                "SSTableReader::sequential_scan - Seeked to start of data section at offset {}",
569                header_size
570            );
571        }
572
573        // CRITICAL FIX: V5CompressedLegacy partitions can span chunk boundaries.
574        // We must stitch all chunks together before parsing to avoid dropping partitions.
575        // Use `requires_chunk_stitching()` as the single source of truth for whether
576        // stitching is needed (BLOCKING-3: unified predicate).
577        //
578        // Note: We intentionally skip table_id matching in the stitching path because the
579        // parser may return incorrect table_ids from header defaults.  Since sequential_scan
580        // is called with a specific table_id, all entries from this SSTable match it.
581        if self.requires_chunk_stitching() {
582            tracing::debug!(
583                "SSTableReader::sequential_scan - V5CompressedLegacy NB detected, using stitched buffer"
584            );
585
586            // Stitch all chunks together (reuse logic from get_all_entries).
587            // Thread the PER-CALL cancel token (issue #2346) so a cancelled
588            // caller abandons the stitched walk promptly instead of blocking
589            // until the entire data section is stitched+parsed — the
590            // chunk-stitch loop polls at the same 256-chunk cadence as the
591            // non-stitching branch below.
592            let all_entries = self
593                .stitch_and_parse_all_chunks(&cursor, schema, true, scan_cancel)
594                .await?;
595            tracing::debug!(
596                "SSTableReader::sequential_scan - Stitched parsing returned {} total entries",
597                all_entries.len()
598            );
599
600            // Apply key-range filter and tombstone filter; collect ALL matching entries
601            // before sorting.  Limit is applied AFTER sort so that LIMIT N returns the N
602            // token-smallest partitions, not the first N encountered in parse order.
603            // (BLOCKING-1: limit-after-order)
604            // Work-probe (issue #2398, roborev 1692): same-partition rows arrive
605            // consecutively (Data.db is laid out partition-by-partition), so a
606            // changed `RowKey` (the partition key) marks a new partition body
607            // decoded — matching the "once per partition" semantics of the two
608            // index-driven walks. Tracked here, BEFORE the filters below, so it
609            // counts decode work regardless of a later tombstone/range skip.
610            let mut prev_partition_key: Option<RowKey> = None;
611            for (idx, (_entry_table_id, entry_key, entry_value)) in
612                all_entries.into_iter().enumerate()
613            {
614                // Cooperative checkpoint (issue #2346): the chunk-stitch loop
615                // poll covers the I/O phase, but `parse_block` materialises every
616                // entry in one shot — checkpoint here so a cancelled caller does
617                // not walk a huge already-parsed result set to completion, and so
618                // the query-engine's ONE `max_execution_time` timeout can elapse
619                // mid-walk (issue #1695: the yield, at the same 256-entry cadence,
620                // is what makes this future `Pending`).
621                scan_cancel.checkpoint(idx).await?;
622                if prev_partition_key.as_ref() != Some(&entry_key) {
623                    crate::storage::sstable::work_counters::add_stream_walk_partition_parsed();
624                    prev_partition_key = Some(entry_key.clone());
625                }
626                if let Some(start) = start_key {
627                    if &entry_key < start {
628                        continue;
629                    }
630                }
631
632                if let Some(end) = end_key {
633                    if &entry_key > end {
634                        continue;
635                    }
636                }
637
638                if !self.filter_tombstone(&entry_value) {
639                    continue;
640                }
641
642                results.push((entry_key, entry_value));
643            }
644
645            tracing::debug!(
646                "SSTableReader::sequential_scan - Filtered to {} results before limit (limit: {:?})",
647                results.len(),
648                limit
649            );
650
651            // Sort by Murmur3 token order (spec §5, Appendix B §313), then truncate to limit.
652            sort_by_token_order(&mut results);
653            if let Some(lim) = limit {
654                results.truncate(lim);
655            }
656
657            tracing::debug!(
658                "SSTableReader::sequential_scan - Returning {} results after sort+limit",
659                results.len()
660            );
661            return Ok(results);
662        }
663
664        // Non-stitching path for other formats
665        let mut block_count = 0;
666        // Work-probe (issue #2398, roborev 1692): scoped OUTSIDE the block loop so
667        // a partition split across a block boundary is not double-counted; see the
668        // stitching branch above for the "changed key = new partition" rationale.
669        let mut prev_partition_key: Option<RowKey> = None;
670        while let Some(block) = self.read_next_block(&cursor).await? {
671            // Cooperative checkpoint (issue #2264): an index-less (Summary.db
672            // absent) SSTable materialises EVERY partition here in one pass; poll
673            // the token per block so a cancelled Flight `do_get` abandons the walk
674            // promptly instead of running to completion under the ~1–2 min backstop.
675            // A whole block is coarse, so every iteration is a correct yield point
676            // for the chokepoint timeout too (issue #1695).
677            scan_cancel.checkpoint_now().await?;
678            block_count += 1;
679            tracing::debug!(
680                "SSTableReader::sequential_scan - Read block {}, size {} bytes",
681                block_count,
682                block.len()
683            );
684
685            let entries = self.parse_block_entries_with_schema(&block, schema, true)?;
686            tracing::debug!(
687                "SSTableReader::sequential_scan - Block {} contains {} entries",
688                block_count,
689                entries.len()
690            );
691
692            for (i, (entry_table_id, entry_key, entry_value)) in entries.iter().enumerate() {
693                // Cooperative cancellation (issue #2264, roborev): the per-block
694                // poll above fires ONCE per `read_next_block` call, but an
695                // uncompressed/BTI-direct block returns the WHOLE data section as
696                // one contiguous unit (per `read_next_block_impl`'s doc comment) —
697                // so for that shape `entries` alone can number in the hundreds of
698                // thousands. Poll again here at the SAME 256-entry cadence used
699                // elsewhere so materialisation honours the interval regardless of
700                // how large a single block turned out to be, independent of
701                // whichever inner parser branch produced `entries`.
702                scan_cancel.checkpoint(i).await?;
703                tracing::debug!(
704                    "SSTableReader::sequential_scan - Block {} entry {}: table_id='{}', key={:?}",
705                    block_count,
706                    i,
707                    entry_table_id,
708                    entry_key
709                );
710
711                // Work-probe (issue #2398, roborev 1692): count decode work BEFORE
712                // any table-id/range/tombstone filter, matching the other sites.
713                if prev_partition_key.as_ref() != Some(entry_key) {
714                    crate::storage::sstable::work_counters::add_stream_walk_partition_parsed();
715                    prev_partition_key = Some(entry_key.clone());
716                }
717
718                // Match table IDs - supports both qualified (keyspace.table) and unqualified (table) formats
719                // This allows queries with either format to match SSTables stored with either format
720                if !table_ids_match(entry_table_id, table_id) {
721                    tracing::debug!("SSTableReader::sequential_scan - Skipping entry: table_id mismatch ('{}' != '{}')",
722                              entry_table_id, table_id);
723                    continue;
724                }
725
726                // Check key range
727                if let Some(start) = start_key {
728                    if entry_key < start {
729                        tracing::debug!(
730                            "SSTableReader::sequential_scan - Skipping entry: key < start_key"
731                        );
732                        continue;
733                    }
734                }
735
736                if let Some(end) = end_key {
737                    if entry_key > end {
738                        tracing::debug!(
739                            "SSTableReader::sequential_scan - Skipping entry: key > end_key"
740                        );
741                        continue;
742                    }
743                }
744
745                // Extract write time from entry metadata
746                let _write_time = self.extract_write_time_from_entry(entry_key, entry_value);
747
748                // Filter out tombstones and expired data
749                if !self.filter_tombstone(entry_value) {
750                    tracing::debug!("SSTableReader::sequential_scan - Skipping entry: filtered out (tombstone or expired)");
751                    continue;
752                }
753
754                tracing::debug!("SSTableReader::sequential_scan - Including entry in results");
755                results.push((entry_key.clone(), entry_value.clone()));
756            }
757        }
758
759        tracing::debug!(
760            "SSTableReader::sequential_scan - Finished scanning {} blocks",
761            block_count
762        );
763        tracing::debug!(
764            "SSTableReader::sequential_scan - {} results before sort+limit",
765            results.len()
766        );
767
768        // Sort by Murmur3 token order (spec §5, Appendix B §313), then apply limit.
769        // Limit is applied AFTER sort so that LIMIT N returns the N token-smallest
770        // partitions (BLOCKING-1: limit-after-order).
771        sort_by_token_order(&mut results);
772        if let Some(lim) = limit {
773            results.truncate(lim);
774        }
775
776        tracing::debug!(
777            "SSTableReader::sequential_scan - Returning {} results after sort+limit",
778            results.len()
779        );
780        Ok(results)
781    }
782
783    /// Scan a range of keys AND return per-cell write metadata.
784    ///
785    /// Used when `ProjectionFlags::include_cell_metadata` is set (issue #693).
786    /// Falls through to `stitch_and_parse_all_chunks_with_metadata` for
787    /// V5CompressedLegacy format (the common path for real SSTables).
788    /// Returns `None` as the metadata for non-V5 formats (they do not carry
789    /// per-cell timestamps in a way the parser currently surfaces).
790    pub async fn scan_with_cell_metadata(
791        &self,
792        table_id: &TableId,
793        start_key: Option<&RowKey>,
794        end_key: Option<&RowKey>,
795        limit: Option<usize>,
796        schema: Option<&crate::schema::TableSchema>,
797    ) -> Result<
798        Vec<(
799            RowKey,
800            ScanRow,
801            std::collections::HashMap<String, CellWriteMetadata>,
802        )>,
803    > {
804        tracing::debug!("SSTableReader::scan_with_cell_metadata - Starting");
805        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
806
807        // Issue #660: BTI ("da") metadata scan — same whole-Data.db walk as the
808        // plain BTI scan, but surfaces per-cell write metadata for WRITETIME/TTL.
809        if self.bti_partitions_db.is_some() {
810            return self
811                .bti_scan_with_metadata(start_key, end_key, limit, schema, true, None)
812                .await;
813        }
814
815        // Issue #815: independent per-scan cursor — no cross-scan serialization.
816        let cursor = self.new_scan_cursor().await?;
817
818        let header_size = self.calculate_header_size();
819        {
820            let mut file_guard = cursor.file.lock().await;
821            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
822        }
823
824        // V5CompressedLegacy (stitching) path — the common path for Cassandra 5.0 SSTables.
825        if self.requires_chunk_stitching() {
826            let all_entries = self
827                .stitch_and_parse_all_chunks_with_metadata(&cursor, schema)
828                .await?;
829
830            let mut results = Vec::new();
831            for (_entry_table_id, entry_key, entry_value, cell_meta) in all_entries {
832                if let Some(start) = start_key {
833                    if &entry_key < start {
834                        continue;
835                    }
836                }
837                if let Some(end) = end_key {
838                    if &entry_key > end {
839                        continue;
840                    }
841                }
842                if !self.filter_tombstone(&entry_value) {
843                    continue;
844                }
845                results.push((entry_key, entry_value, cell_meta));
846            }
847
848            sort_by_token_order_with_meta(&mut results);
849            if let Some(lim) = limit {
850                results.truncate(lim);
851            }
852
853            tracing::debug!(
854                "SSTableReader::scan_with_cell_metadata - Returning {} results (stitched path)",
855                results.len()
856            );
857            return Ok(results);
858        }
859
860        // Non-stitching path: fall back to regular scan + empty metadata.
861        // Per-cell metadata is not yet surfaced for block-entry formats.
862        let plain = self
863            .sequential_scan(
864                table_id,
865                start_key,
866                end_key,
867                limit,
868                schema,
869                &self.scan_cancel,
870            )
871            .await?;
872        Ok(plain
873            .into_iter()
874            .map(|(k, v)| (k, v, std::collections::HashMap::new()))
875            .collect())
876    }
877}
