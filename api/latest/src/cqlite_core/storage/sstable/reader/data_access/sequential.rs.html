27/// Classify a `parse_block` failure on the compressed (`V5CompressedLegacy`)
28/// point-lookup path (`scan_for_key`) as a soft-miss vs. a fatal error.
29///
30/// Issue #1411: the scan path ([`SSTableReader::stitch_and_parse_all_chunks`])
31/// propagates every `parse_block` error via `?`. The point-lookup path mirrors
32/// that so `get()` and `scan()` agree on which failures are fatal — with ONE
33/// deliberate exception, the case the original blanket `Err(_) => Ok(None)`
34/// existed to protect: this reader has no schema for the table, so it cannot
35/// serve the key and the caller must fall through to the next SSTable reader.
36///
37/// Returns `true` (soft-miss → `Ok(None)`) ONLY when both hold:
38/// - `schema_present == false` — the reader supplied no schema, and
39/// - the parser reported [`Error::Schema`] — which `V5CompressedLegacy`'s
40///   `parse_block` raises ("requires schema for <ks>.<table>") *before* it
41///   inspects any bytes when no schema is available.
42///
43/// Every other failure is non-recoverable and must propagate: real data
44/// corruption / malformed blocks (`Corruption`, `InvalidFormat`, `Parse`, I/O,
45/// etc.), and — critically — a deep schema/type-resolution `Error::Schema`
46/// raised when a schema IS present (a UDT/type that cannot be resolved is a real
47/// error, not a missing key). Requiring `!schema_present` prevents that class
48/// from ever being masked as "not found".
49fn is_parse_soft_miss(schema_present: bool, err: &Error) -> bool {
50    !schema_present && matches!(err, Error::Schema(_))
51}

53impl SSTableReader {
54    /// Scan a range of keys
55    ///
56    /// # Arguments
57    /// * `table_id` - The table to scan
58    /// * `start_key` - Optional start key for range scan
59    /// * `end_key` - Optional end key for range scan
60    /// * `limit` - Optional limit on number of results
61    /// * `schema` - Optional table schema for schema-aware parsing. When provided,
62    ///   enables accurate type detection and avoids heuristic-based parsing.
63    ///   Strongly recommended for Cassandra 5.0+ formats.
64    pub async fn scan(
65        &self,
66        table_id: &TableId,
67        start_key: Option<&RowKey>,
68        end_key: Option<&RowKey>,
69        limit: Option<usize>,
70        schema: Option<&crate::schema::TableSchema>,
71    ) -> Result<Vec<(RowKey, ScanRow)>> {
72        tracing::debug!("SSTableReader::scan - Starting scan");
73        tracing::debug!("SSTableReader::scan - File path: {:?}", self.file_path());
74        tracing::debug!("SSTableReader::scan - Table ID: {}", table_id);
75        tracing::debug!("SSTableReader::scan - Start key: {:?}", start_key);
76        tracing::debug!("SSTableReader::scan - End key: {:?}", end_key);
77        tracing::debug!("SSTableReader::scan - Limit: {:?}", limit);
78        tracing::debug!("SSTableReader::scan - Has schema: {}", schema.is_some());
79        tracing::debug!("SSTableReader::scan - Has index: {}", self.index.is_some());
80        tracing::debug!(
81            "SSTableReader::scan - Has bloom filter: {}",
82            self.bloom_filter.is_some()
83        );
84
85        // Issue #660: BTI ("da") readers have no Index.db/Summary.db. A full scan
86        // walks the whole (chunk-compressed) Data.db once and parses every
87        // partition — the same partition decode the point-lookup path proves
88        // correct, but emitting ALL partitions instead of stopping at the first.
89        if self.bti_partitions_db.is_some() {
90            let entries = self
91                .bti_scan_with_metadata(start_key, end_key, limit, schema, true)
92                .await?;
93            return Ok(entries.into_iter().map(|(k, v, _meta)| (k, v)).collect());
94        }
95
96        let mut results = Vec::new();
97
98        // Use index for efficient range scan if available
99        if let Some(index) = &self.index {
100            tracing::debug!("SSTableReader::scan - Using index-based scan");
101            let entries = index.get_range(table_id, start_key, end_key)?;
102            tracing::debug!(
103                "SSTableReader::scan - Index returned {} entries",
104                entries.len()
105            );
106
107            // Issue #256 FIX: Fall back to sequential scan when index returns no entries
108            //
109            // This handles BTI (Big Trie Index) format where parsing may be incomplete or
110            // where the index format is not yet fully supported. Without this check, tables
111            // using BTI format return 0 rows because:
112            // 1. The index exists (so we take the index-based path)
113            // 2. But get_range() returns 0 entries (BTI parsing incomplete)
114            // 3. The has_zero_size check never triggers (no entries to check)
115            // 4. The for loop iterates 0 times, returning empty results
116            //
117            // Sequential scan correctly parses Data.db directly, bypassing index issues.
118            if entries.is_empty() {
119                tracing::debug!(
120                    "SSTableReader::scan - Index returned 0 entries (BTI format or incomplete parsing), falling back to sequential scan"
121                );
122                return self
123                    .sequential_scan(
124                        table_id,
125                        start_key,
126                        end_key,
127                        limit,
128                        schema,
129                        &self.scan_cancel,
130                    )
131                    .await;
132            }
133
134            // Check if any entry has size=0 (Cassandra 5.0 format)
135            let has_zero_size = entries.iter().any(|e| e.size == 0);
136            if has_zero_size {
137                tracing::debug!("SSTableReader::scan - Index reports size=0 for some entries, using sequential scan fallback");
138                return self
139                    .sequential_scan(
140                        table_id,
141                        start_key,
142                        end_key,
143                        limit,
144                        schema,
145                        &self.scan_cancel,
146                    )
147                    .await;
148            }
149
150            // Collect ALL index entries (limit applied after sort — BLOCKING-1).
151            for (i, entry) in entries.iter().enumerate() {
152                // Index offsets are relative to data section start - adjust for header
153                let file_offset = entry.offset + self.actual_header_size as u64;
154                tracing::debug!(
155                    "SSTableReader::scan - Processing index entry {}: index_offset={}, file_offset={}, size={}",
156                    i, entry.offset, file_offset, entry.size
157                );
158
159                if let Some(value) = self.read_value_at_offset(file_offset, entry.size).await? {
160                    tracing::debug!(
161                        "SSTableReader::scan - Successfully read value at offset {}",
162                        entry.offset
163                    );
164                    results.push((entry.key.clone(), value));
165                } else {
166                    tracing::debug!("SSTableReader::scan - Value at offset {} was filtered out (tombstone or expired)", entry.offset);
167                }
168            }
169        } else {
170            // Fallback to sequential scan.  sequential_scan() already returns results in
171            // token order (NON-BLOCKING-1: avoid double-sort — return directly).
172            tracing::debug!("SSTableReader::scan - No index, falling back to sequential scan");
173            let seq_results = self
174                .sequential_scan(
175                    table_id,
176                    start_key,
177                    end_key,
178                    limit,
179                    schema,
180                    &self.scan_cancel,
181                )
182                .await?;
183            tracing::debug!(
184                "SSTableReader::scan - Sequential scan returned {} results",
185                seq_results.len()
186            );
187            tracing::debug!(
188                "SSTableReader::scan - Returning {} final results",
189                seq_results.len()
190            );
191            return Ok(seq_results);
192        }
193
194        // Index-based path: sort by Murmur3 token order (ascending token, then key bytes).
195        // This matches the on-disk physical order (spec §5, Appendix B §313) and the write
196        // engine's DecoratedKey::cmp.  Compute each key's token once before sorting to
197        // avoid O(n log n) recomputation inside the comparator.
198        sort_by_token_order(&mut results);
199        // Limit applied AFTER sort so LIMIT N returns the N token-smallest partitions.
200        if let Some(lim) = limit {
201            results.truncate(lim);
202        }
203
204        tracing::debug!(
205            "SSTableReader::scan - Returning {} final results",
206            results.len()
207        );
208        Ok(results)
209    }

211    /// Get all entries in the SSTable.
212    ///
213    /// # Tombstone contract (Issue #505)
214    ///
215    /// This is a **user-facing** accessor: row tombstones are filtered out via
216    /// [`Self::filter_tombstone`] and never appear in the returned entries. The
217    /// underlying `parse_block` path emits `Value::Tombstone(RowTombstone)` for
218    /// deleted rows, but those are suppressed here so callers see exactly the live
219    /// rows (matching the previous `Value::Null` suppression behaviour).
220    ///
221    /// The compaction k-way merger must instead use
222    /// [`Self::iterate_all_partitions_for_compaction`], which preserves
223    /// `Value::Tombstone` entries (with their authoritative deletion timestamps)
224    /// so that tombstone-shadowing semantics can be applied during the merge.
225    pub async fn get_all_entries(&self) -> Result<Vec<(TableId, RowKey, ScanRow)>> {
226        // Issue #660: BTI ("da") tables have no Index.db; route through the
227        // whole-Data.db BTI scan, which resolves schema via get_table_schema
228        // (header/registry) and decodes every partition. It mints its own
229        // per-scan cursor, as does the non-BTI path below (issue #815).
230        if self.bti_partitions_db.is_some() {
231            let table_id = TableId::new(format!(
232                "{}.{}",
233                self.header.keyspace, self.header.table_name
234            ));
235            let entries = self
236                .bti_scan_with_metadata(None, None, None, None, false)
237                .await?;
238            return Ok(entries
239                .into_iter()
240                .map(|(k, v, _meta)| (table_id.clone(), k, v))
241                .collect());
242        }
243
244        // Issue #815: independent per-scan cursor — no cross-scan serialization.
245        let cursor = self.new_scan_cursor().await?;
246
247        let mut results = Vec::new();
248
249        // Reset to beginning of data section
250        let header_size = self.calculate_header_size();
251        {
252            let mut file_guard = cursor.file.lock().await;
253            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
254        }
255
256        if self.requires_chunk_stitching() {
257            // V5CompressedLegacy: Row payloads can span multiple compressed chunks
258            // We must decompress and stitch all chunks together before parsing
259            tracing::debug!(
260                "V5CompressedLegacy format detected, decompressing and stitching all chunks before parsing"
261            );
262
263            // Use shared stitching helper method. Physical enumeration
264            // (`get_all_entries`) honours the reader's own cancel field —
265            // unchanged pre-#2346 behaviour.
266            let entries = self
267                .stitch_and_parse_all_chunks(&cursor, None, false, &self.scan_cancel)
268                .await?;
269            results.extend(entries);
270        } else {
271            // Other formats: Read and parse blocks individually
272            while let Some(block) = self.read_next_block(&cursor).await? {
273                let entries = self.parse_block_entries(&block, None, false)?;
274                results.extend(entries);
275            }
276        }
277
278        // Issue #505: suppress row tombstones from user-facing output. The compaction
279        // path (iterate_all_partitions_for_compaction) bypasses this filter.
280        results.retain(|(_tid, _key, value)| self.filter_tombstone(value));
281
282        Ok(results)
283    }

285    /// Streaming scan (issue #790): yield `(RowKey, ScanRow)` entries lazily
286    /// through a bounded channel instead of materializing the whole result in a
287    /// `Vec`. Live heap is bounded by `buffer_size` rows (plus the stitched
288    /// data-section buffer) rather than growing O(rows).
289    ///
290    /// Entries are yielded in on-disk order — token order for a single SSTable —
291    /// matching the order of the materializing [`scan`](Self::scan) path. The
292    /// bounded channel applies backpressure: parsing pauses when the consumer
293    /// falls behind and stops entirely if the consumer is dropped.
294    ///
295    /// In-flight bound (chunk-stitching SSTables): the windowed pipeline (issue
296    /// #1143) materializes one confirmed partition at a time and batches its rows to
297    /// amortize the cross-thread wake, so against a stalled consumer the resident
298    /// `(RowKey, ScanRow)` count is the SUM of three inherent terms, not one constant:
299    /// `buffer_size` (this channel) `+ max_partition_size` (the one fully-materialized
300    /// confirmed partition — a pre-existing #1156 windowed-scan term, inherent to any
301    /// row-materializing partition scan) `+`
302    /// [`MAX_INFLIGHT_BATCH_ROWS`](super::super::scan_stream_windowed::MAX_INFLIGHT_BATCH_ROWS)
303    /// (the FIXED, BOUNDED amount the issue-#1143 batching subsystem may run ahead,
304    /// regardless of `buffer_size`, holding even for `buffer_size == 1`).
305    /// `MAX_INFLIGHT_BATCH_ROWS` bounds the batching subsystem alone, NOT the
306    /// `max_partition_size` materialization term. Non-stitching SSTables parse a
307    /// whole block before forwarding its rows, so the resident `(RowKey, ScanRow)`
308    /// count is bounded by `buffer_size + (one parsed block's entries)`.
309    pub fn scan_stream(
310        self: std::sync::Arc<Self>,
311        table_id: TableId,
312        start_key: Option<RowKey>,
313        end_key: Option<RowKey>,
314        schema: Option<crate::schema::TableSchema>,
315        buffer_size: usize,
316    ) -> mpsc::Receiver<Result<(RowKey, ScanRow)>> {
317        // A directly-invoked reader scan is a top-level scan OPERATION: acquire one
318        // admission permit (issue #1594). Callers that fan out to multiple readers
319        // (`SSTableManager::scan_stream`) instead hold ONE permit for the whole
320        // operation and open each sub-scan `Exempt` via `scan_stream_admitted`.
321        self.scan_stream_admitted(
322            table_id,
323            start_key,
324            end_key,
325            schema,
326            buffer_size,
327            ScanAdmission::Acquire,
328        )
329    }

331    /// [`scan_stream`](Self::scan_stream) with an explicit admission context
332    /// (issue #1594). Admission is per top-level scan OPERATION: a cross-generation
333    /// fan-out merge passes [`ScanAdmission::Exempt`] for each sub-scan while
334    /// holding ONE permit for the whole operation, so a single query's fan-out to
335    /// `N > cap` generations can never hold-and-wait on itself (the deadlock a
336    /// per-sub-scan permit introduced). A direct scan passes
337    /// [`ScanAdmission::Acquire`].
338    pub(crate) fn scan_stream_admitted(
339        self: std::sync::Arc<Self>,
340        table_id: TableId,
341        start_key: Option<RowKey>,
342        end_key: Option<RowKey>,
343        schema: Option<crate::schema::TableSchema>,
344        buffer_size: usize,
345        admission: ScanAdmission,
346    ) -> mpsc::Receiver<Result<(RowKey, ScanRow)>> {
347        let (tx, rx) = mpsc::channel(buffer_size.max(1));
348        tokio::spawn(async move {
349            if let Err(e) = self
350                .run_scan_stream(table_id, start_key, end_key, schema, tx.clone(), admission)
351                .await
352            {
353                // Surface the error to the consumer as a terminal stream item.
354                let _ = tx.send(Err(e)).await;
355            }
356        });
357        rx
358    }

360    async fn run_scan_stream(
361        self: std::sync::Arc<Self>,
362        table_id: TableId,
363        start_key: Option<RowKey>,
364        end_key: Option<RowKey>,
365        schema: Option<crate::schema::TableSchema>,
366        tx: mpsc::Sender<Result<(RowKey, ScanRow)>>,
367        admission: ScanAdmission,
368    ) -> Result<()> {
369        // Admission control (issue #1594, F4): a top-level scan operation acquires
370        // ONE blocking-pool permit here, at the top, BEFORE opening the cursor or
371        // spawning any `spawn_blocking` work, and holds it via this RAII guard for
372        // the whole scan (released on every exit — success, error, cancellation).
373        // A fan-out merge's sub-scan is `Exempt` (the merge holds the operation's
374        // single permit), so it acquires none — a fan-out to `N > cap` generations
375        // can never hold-and-wait on itself. No other permit/lock is held while
376        // awaiting admission, so this single-permit acquisition cannot deadlock.
377        let _admission = match admission {
378            ScanAdmission::Acquire => Some(scan_admission::admit().await),
379            ScanAdmission::Exempt => None,
380        };
381
382        // Issue #1577 (owner-chosen fix, 2026-07-06): BTI (`da`) readers MUST use
383        // the SAME per-reader decode path `SSTableReader::scan` uses — the trie-walk
384        // `bti_scan_with_metadata` — NOT the block-by-block `read_next_block` +
385        // `parse_block_entries_with_schema` decoder below. The block-by-block path
386        // diverges for BTI: it can under/over-produce, reorder, or (as here) fail
387        // outright ("Blob fallback not allowed for V5_0Bti"), while D1's LIMIT
388        // pushdown (`capped_fallback_scan`) trusts the streamed first-`cap` rows to
389        // be byte-identical to `scan`'s first-`cap` rows. Driving the identical
390        // authoritative decoder makes `scan_stream` PREFIX-AUTHORITATIVE with `scan`
391        // for BTI BY CONSTRUCTION (same rows, same `sort_by_token_order`, same
392        // key-range + `filter_tombstone` filtering — all applied INSIDE
393        // `bti_scan_with_metadata`), so no runtime reconcile against `scan` is
394        // needed. This gates on the SAME `self.bti_partitions_db.is_some()`
395        // condition `scan` uses, so the two can never disagree on which readers are
396        // BTI. BTI decode fully materializes the (small, index-less) reconciled
397        // table before streaming — mirrored by `scan_stream_materializes` returning
398        // `true` for BTI, so a bounded LIMIT consumer charges the true decoded count
399        // rather than assuming a lazy decode-stop.
400        if self.bti_partitions_db.is_some() {
401            let entries = self
402                .bti_scan_with_metadata(
403                    start_key.as_ref(),
404                    end_key.as_ref(),
405                    None,
406                    schema.as_ref(),
407                    true,
408                )
409                .await?;
410            for (entry_key, entry_value, _meta) in entries {
411                // `bti_scan_with_metadata` already applied the key-range and
412                // tombstone filters and token-ordered the rows; forward as-is so
413                // the stream is byte-identical to `scan`'s BTI path.
414                if tx.send(Ok((entry_key, entry_value))).await.is_err() {
415                    return Ok(()); // consumer dropped
416                }
417            }
418            return Ok(());
419        }
420
421        // Issue #815: independent per-scan cursor — no cross-scan serialization.
422        // Issue #1577 (rust-reviewer nit): created only for the non-BTI path — the
423        // BTI branch above mints its own cursor inside `bti_scan_with_metadata` and
424        // returned already, so opening+seeking one here for BTI was a wasted
425        // open(2)+seek. Mirrors how `scan`/`get_all_entries`/`scan_with_cell_metadata`
426        // gate cursor creation after their BTI early-return.
427        let cursor = self.new_scan_cursor().await?;
428
429        // Position at the start of the data section (mirrors sequential_scan).
430        let header_size = self.calculate_header_size();
431        {
432            let mut file_guard = cursor.file.lock().await;
433            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
434        }
435
436        if self.requires_chunk_stitching() {
437            // Issue #1143: SLIDING-WINDOW stitch+parse on a bounded blocking->async
438            // pipeline instead of stitching the ENTIRE Data.db into one growing
439            // `Vec<u8>` before parsing (which thrashed the allocator and blew up
440            // read p99 under concurrent write load). Keeps only a `window` bounded
441            // by `max_partition_size + one_chunk`. Same bounded driver as the
442            // compaction read path (issue #827); scan output (key-range + tombstone
443            // + `table_ids_match` filters, dropped timestamp) is parity-identical to
444            // `parse_stitched_stream`. Full rationale + the in-flight batching bound
445            // live in the `scan_stream_windowed` module docs.
446            self.run_scan_stream_windowed(
447                table_id,
448                start_key,
449                end_key,
450                schema,
451                &cursor,
452                WindowedOut::PerRow(tx.clone()),
453            )
454            .await
455        } else {
456            // Non-stitching formats already read block-by-block; emit per block so
457            // only one block's entries are live at a time.
458            while let Some(block) = self.read_next_block(&cursor).await? {
459                let entries =
460                    self.parse_block_entries_with_schema(&block, schema.as_ref(), true)?;
461                for (entry_table_id, entry_key, entry_value) in entries {
462                    if !table_ids_match(&entry_table_id, &table_id) {
463                        continue;
464                    }
465                    if let Some(ref start) = start_key {
466                        if &entry_key < start {
467                            continue;
468                        }
469                    }
470                    if let Some(ref end) = end_key {
471                        if &entry_key > end {
472                            continue;
473                        }
474                    }
475                    if !self.filter_tombstone(&entry_value) {
476                        continue;
477                    }
478                    if tx.send(Ok((entry_key, entry_value))).await.is_err() {
479                        return Ok(()); // consumer dropped
480                    }
481                }
482            }
483            Ok(())
484        }
485    }

487    /// Batched streaming scan (issue #1592, Epic F/F2): the additive companion to
488    /// [`scan_stream`](Self::scan_stream) whose channel item is a `Vec` BATCH of
489    /// `(RowKey, ScanRow)` entries rather than a single entry. Forwarding one batch
490    /// per channel send collapses the one-async-wake-per-row cost the internal
491    /// windowed pipeline (issue #1143) was designed to amortize but the public
492    /// forwarder then re-flattened away.
493    ///
494    /// Order and content are identical to [`scan_stream`](Self::scan_stream):
495    /// flattening the batches yields exactly the per-row stream. Backpressure is
496    /// preserved — the channel is bounded (in BATCHES) and every send observes it.
497    pub fn scan_stream_batched(
498        self: std::sync::Arc<Self>,

505        self.scan_stream_batched_admitted(
506            table_id,
507            start_key,
508            end_key,
509            schema,
510            buffer_size,
511            ScanAdmission::Acquire,
512        )

515    /// [`scan_stream_batched`](Self::scan_stream_batched) with an explicit
516    /// admission context (issue #1594), mirroring
517    /// [`scan_stream_admitted`](Self::scan_stream_admitted).
518    pub(crate) fn scan_stream_batched_admitted(
519        self: std::sync::Arc<Self>,
520        table_id: TableId,
521        start_key: Option<RowKey>,
522        end_key: Option<RowKey>,
523        schema: Option<crate::schema::TableSchema>,
524        buffer_size: usize,
525        admission: ScanAdmission,
526    ) -> mpsc::Receiver<Result<Vec<(RowKey, ScanRow)>>> {
527        // The public channel carries BATCHES; sizing it to
528        // `ceil(buffer_size / BATCH_EMIT_ROWS)` batches keeps the resident-row
529        // budget of this channel comparable to the per-row surface's `buffer_size`
530        // rather than `buffer_size * BATCH_EMIT_ROWS`.
531        let cap = buffer_size.div_ceil(BATCH_EMIT_ROWS).max(1);
532        let (tx, rx) = mpsc::channel(cap);
533        tokio::spawn(async move {
534            if let Err(e) = self
535                .run_scan_stream_batched(
536                    table_id,
537                    start_key,
538                    end_key,
539                    schema,
540                    tx.clone(),
541                    admission,
542                )
543                .await
544            {
545                let _ = tx.send(Err(e)).await;
546            }
547        });
548        rx

551    async fn run_scan_stream_batched(
552        self: std::sync::Arc<Self>,
553        table_id: TableId,
554        start_key: Option<RowKey>,
555        end_key: Option<RowKey>,
556        schema: Option<crate::schema::TableSchema>,
557        tx: mpsc::Sender<Result<Vec<(RowKey, ScanRow)>>>,
558        admission: ScanAdmission,
559    ) -> Result<()> {
560        // Admission control (issue #1594, F4): identical discipline to the per-row
561        // `run_scan_stream` — one permit per top-level scan operation, held via RAII.
562        let _admission = match admission {
563            ScanAdmission::Acquire => Some(scan_admission::admit().await),
564            ScanAdmission::Exempt => None,
565        };
566
567        let cursor = self.new_scan_cursor().await?;
568        let header_size = self.calculate_header_size();
569        {
570            let mut file_guard = cursor.file.lock().await;
571            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
572        }
573
574        if self.requires_chunk_stitching() {
575            // Forward the windowed driver's internal batches straight through — no
576            // flatten, no re-batch (issue #1592). Same driver, same order/content
577            // as the per-row path; only the output surface differs.
578            self.run_scan_stream_windowed(
579                table_id,
580                start_key,
581                end_key,
582                schema,
583                &cursor,
584                WindowedOut::Batched(tx.clone()),
585            )
586            .await
587        } else {
588            // Non-stitching formats read block-by-block; emit surviving entries
589            // in BATCH_EMIT_ROWS-capped batches (issue #1592). A block with more
590            // than BATCH_EMIT_ROWS surviving entries is split across multiple
591            // batches so every emitted batch respects `batch.len() <=
592            // BATCH_EMIT_ROWS` (the resident-row budget behind the channel cap);
593            // the remainder carries over to the next block so we still wake the
594            // consumer at most once per full batch.
595            //
596            // Confirmed rows from successfully-parsed prior blocks live in `batch`
597            // until it fills. If a LATER block's read/parse errors mid-scan, flush
598            // those confirmed rows to the consumer BEFORE surfacing the terminal
599            // error — matching both the per-row `run_scan_stream` contract (each row
600            // is sent the instant it is parsed) and the stitching path's
601            // `flush_pending` (issue #1143 / #1592, roborev Finding 1). Propagating
602            // the error via `?` here would silently drop up to BATCH_EMIT_ROWS-1
603            // confirmed rows.
604            let mut batch: Vec<(RowKey, ScanRow)> = Vec::with_capacity(BATCH_EMIT_ROWS);
605            loop {
606                let block = match self.read_next_block(&cursor).await {
607                    Ok(Some(block)) => block,
608                    Ok(None) => break,
609                    Err(e) => {
610                        if !batch.is_empty() {
611                            let _ = tx.send(Ok(std::mem::take(&mut batch))).await;
612                        }
613                        return Err(e);
614                    }
615                };
616                let entries =
617                    match self.parse_block_entries_with_schema(&block, schema.as_ref(), true) {
618                        Ok(entries) => entries,
619                        Err(e) => {
620                            if !batch.is_empty() {
621                                let _ = tx.send(Ok(std::mem::take(&mut batch))).await;
622                            }
623                            return Err(e);
624                        }
625                    };
626                for (entry_table_id, entry_key, entry_value) in entries {
627                    if !table_ids_match(&entry_table_id, &table_id) {
628                        continue;
629                    }
630                    if let Some(ref start) = start_key {
631                        if &entry_key < start {
632                            continue;
633                        }
634                    }
635                    if let Some(ref end) = end_key {
636                        if &entry_key > end {
637                            continue;
638                        }
639                    }
640                    if !self.filter_tombstone(&entry_value) {
641                        continue;
642                    }
643                    batch.push((entry_key, entry_value));
644                    if batch.len() >= BATCH_EMIT_ROWS {
645                        if tx.send(Ok(std::mem::take(&mut batch))).await.is_err() {
646                            return Ok(()); // consumer dropped
647                        }
648                        batch.reserve(BATCH_EMIT_ROWS);
649                    }
650                }
651            }
652            if !batch.is_empty() && tx.send(Ok(batch)).await.is_err() {
653                return Ok(()); // consumer dropped
654            }
655            Ok(())
656        }
657    }

659    pub(super) async fn scan_for_key(
660        &self,
661        table_id: &TableId,
662        key: &RowKey,
663    ) -> Result<Option<ScanRow>> {
664        // Issue #831: record the call so tests can assert the BTI point-lookup
665        // path never reaches the sequential scan.
666        SCAN_FOR_KEY_CALLS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
667
668        // Issue #815: independent per-scan cursor — no cross-scan serialization.
669        let cursor = self.new_scan_cursor().await?;
670
671        let header_size = self.calculate_header_size();
672
673        // For V5CompressedLegacy NB format, partitions can span chunk boundaries.
674        // The block-by-block parser will miss any partition whose bytes cross a
675        // chunk boundary.  Use the same stitched-buffer path that sequential_scan()
676        // uses so that get() and scan() share a consistent view of the data.
677        // (Issue #517)
678        if self.requires_chunk_stitching() {
679            tracing::debug!(
680                "scan_for_key: V5CompressedLegacy NB detected, using stitched buffer for key lookup"
681            );
682            // `stitch_all_chunks` reads from the CURRENT cursor position forward,
683            // so its precondition is "seeked to the data-section start" (the fresh
684            // cursor's chunk index already starts at 0). Each call uses its own
685            // cursor (issue #815), so there is no cross-call position to reset.
686            {
687                let mut file_guard = cursor.file.lock().await;
688                file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
689            }
690
691            // Issue #1411: separate the chunk-integrity stitch from the schema-aware
692            // parse so their failure classes are not conflated. `stitch_all_chunks`
693            // drives the authoritative per-chunk CRC32 check
694            // (`block_io::read_nb_format_chunk_data`) + decompression; a CRC mismatch,
695            // decompression failure, or corrupt offset is corruption and MUST surface
696            // (propagate via `?`, matching `scan`), never be masked as a missing key.
697            // Only the `parse_block` step below may soft-miss (schema unavailable /
698            // wrong table type) so the caller can try the next reader. Previously
699            // `stitch_and_parse_all_chunks` combined both under a blanket
700            // `Err(_) => Ok(None)` that swallowed CRC corruption on point lookups.
701            let stitched_buffer = self.stitch_all_chunks(&cursor).await?;
702
703            // Pass the reader's own schema so that V5CompressedLegacy rows can be fully
704            // parsed and their partition RowKeys emitted.  Without a schema, parse_row_v5
705            // fails for all rows in a partition, causing no entries to be pushed and making
706            // the key comparison always miss even when the key exists.
707            let schema_opt = self.get_table_schema(None);
708            // Issue #1741: point lookups apply SELECT-semantic read shadowing, so
709            // build the parser with read_shadowing = true. The stitch/parse split
710            // (issue #1411) is preserved: CRC/decompress failures already surfaced
711            // via `stitch_all_chunks` above; only `parse_block` may soft-miss.
712            let parser = self.build_v5_parser(true);
713            let all_entries = match parser.parse_block(&stitched_buffer, schema_opt.as_ref(), self)
714            {
715                Ok(entries) => entries,
716                // Issue #1411 (roborev): the scan path (`stitch_and_parse_all_chunks`)
717                // propagates EVERY `parse_block` error via `?`. Mirror that here so
718                // `get()` and `scan()` agree on which errors are fatal. The ONLY
719                // legitimate soft-miss — the case the original blanket
720                // `Err(_) => Ok(None)` existed to protect — is classified by
721                // `is_parse_soft_miss`: this reader has no schema for the table, so it
722                // cannot serve the key and the caller must try the next SSTable reader.
723                // Every other failure (real corruption / malformed block, or a deep
724                // schema/type-resolution error when a schema IS present) MUST surface.
725                Err(e) if is_parse_soft_miss(schema_opt.is_some(), &e) => {
726                    tracing::debug!(
727                        "scan_for_key: no schema for this reader ({}); soft-miss so the caller tries the next reader: {}",
728                        table_id,
729                        e
730                    );
731                    return Ok(None);
732                }
733                Err(e) => return Err(e),
734            };
735
736            // NOTE: The SSTableIndex is built from 16-byte Murmur3 *digests*, not raw keys,
737            // so find_entry() always misses and falls through to this path.  For a found key
738            // we stop early (O(found position)); for a key not present we must scan the whole
739            // stitched buffer — O(file size).  This O(file) miss cost is an existing
740            // limitation of the digest-index design and is tracked separately as a follow-up.
741            //
742            // NON-BLOCKING-2: Table-id matching is intentionally skipped in the stitching path
743            // (consistent with sequential_scan's stitching path).  The V5CompressedLegacy parser
744            // returns entries tagged with the table_id from the SSTable header, which may hold
745            // default or incorrect values when headers use bare keyspace/table names rather than
746            // the query's fully-qualified form.  Since all entries in this stitch buffer come from
747            // the single SSTable being queried, skipping the check is correct and safe.
748            for (_, entry_key, entry_value) in all_entries {
749                if entry_key == *key {
750                    // Early-return on first match (BLOCKING-2: don't parse the rest of the file).
751                    if !self.filter_tombstone(&entry_value) {
752                        return Ok(None);
753                    }
754                    return Ok(Some(entry_value));
755                }
756            }
757
758            return Ok(None);
759        }
760
761        {
762            let mut file_guard = cursor.file.lock().await;
763            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
764        }
765
766        // Sequential scan through blocks
767        while let Some(block) = self.read_next_block(&cursor).await? {
768            let entries = self.parse_block_entries(&block, None, true)?;
769
770            for (entry_table_id, entry_key, entry_value) in entries {
771                if table_ids_match(&entry_table_id, table_id) && entry_key == *key {
772                    // Extract write time from entry metadata
773                    let _write_time = self.extract_write_time_from_entry(&entry_key, &entry_value);
774
775                    // Filter out tombstones and expired data
776                    if !self.filter_tombstone(&entry_value) {
777                        return Ok(None);
778                    }
779
780                    return Ok(Some(entry_value));
781                }
782            }
783        }
784
785        Ok(None)
786    }

788    /// `scan_cancel` is an explicit PER-CALL cancellation token (issue #2346).
789    /// Every existing caller passes `&self.scan_cancel` (the reader's own field,
790    /// unchanged pre-#2346 semantics); the compaction path
791    /// ([`SSTableReader::iterate_all_partitions_cancellable`]) passes its
792    /// caller-supplied token instead, so a shared/cached reader's two concurrent
793    /// scans cancel independently.
794    pub(in crate::storage::sstable::reader) async fn sequential_scan(
795        &self,
796        table_id: &TableId,
797        start_key: Option<&RowKey>,
798        end_key: Option<&RowKey>,
799        limit: Option<usize>,
800        schema: Option<&crate::schema::TableSchema>,
801        scan_cancel: &ScanCancel,
802    ) -> Result<Vec<(RowKey, ScanRow)>> {
803        tracing::debug!("SSTableReader::sequential_scan - Starting sequential scan");
804        tracing::debug!("SSTableReader::sequential_scan - Table ID: {}", table_id);
805        tracing::debug!(
806            "SSTableReader::sequential_scan - Has schema: {}",
807            schema.is_some()
808        );
809
810        // Issue #815: each scan uses its own cursor (private file position and
811        // chunk index), so concurrent scans on this reader run in parallel
812        // without the per-scan serialization #805 introduced for correctness.
813        let cursor = self.new_scan_cursor().await?;
814
815        let mut results = Vec::new();
816
817        let header_size = self.calculate_header_size();
818        tracing::debug!(
819            "SSTableReader::sequential_scan - Header size: {} bytes",
820            header_size
821        );
822
823        {
824            let mut file_guard = cursor.file.lock().await;
825            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
826            tracing::debug!(
827                "SSTableReader::sequential_scan - Seeked to start of data section at offset {}",
828                header_size
829            );
830        }
831
832        // CRITICAL FIX: V5CompressedLegacy partitions can span chunk boundaries.
833        // We must stitch all chunks together before parsing to avoid dropping partitions.
834        // Use `requires_chunk_stitching()` as the single source of truth for whether
835        // stitching is needed (BLOCKING-3: unified predicate).
836        //
837        // Note: We intentionally skip table_id matching in the stitching path because the
838        // parser may return incorrect table_ids from header defaults.  Since sequential_scan
839        // is called with a specific table_id, all entries from this SSTable match it.
840        if self.requires_chunk_stitching() {
841            tracing::debug!(
842                "SSTableReader::sequential_scan - V5CompressedLegacy NB detected, using stitched buffer"
843            );
844
845            // Stitch all chunks together (reuse logic from get_all_entries).
846            // Thread the PER-CALL cancel token (issue #2346) so a cancelled
847            // caller abandons the stitched walk promptly instead of blocking
848            // until the entire data section is stitched+parsed — the
849            // chunk-stitch loop polls at the same 256-chunk cadence as the
850            // non-stitching branch below.
851            let all_entries = self
852                .stitch_and_parse_all_chunks(&cursor, schema, true, scan_cancel)
853                .await?;
854            tracing::debug!(
855                "SSTableReader::sequential_scan - Stitched parsing returned {} total entries",
856                all_entries.len()
857            );
858
859            // Apply key-range filter and tombstone filter; collect ALL matching entries
860            // before sorting.  Limit is applied AFTER sort so that LIMIT N returns the N
861            // token-smallest partitions, not the first N encountered in parse order.
862            // (BLOCKING-1: limit-after-order)
863            for (idx, (_entry_table_id, entry_key, entry_value)) in
864                all_entries.into_iter().enumerate()
865            {
866                // Cooperative cancellation (issue #2346): the chunk-stitch loop
867                // poll covers the I/O phase, but `parse_block` materialises every
868                // entry in one shot — poll here at the same 256-entry cadence as
869                // the non-stitching branch so a cancelled caller does not walk a
870                // huge already-parsed result set to completion.
871                if idx & 0xFF == 0 {
872                    scan_cancel.check()?;
873                }
874                if let Some(start) = start_key {
875                    if &entry_key < start {
876                        continue;
877                    }
878                }
879
880                if let Some(end) = end_key {
881                    if &entry_key > end {
882                        continue;
883                    }
884                }
885
886                if !self.filter_tombstone(&entry_value) {
887                    continue;
888                }
889
890                results.push((entry_key, entry_value));
891            }
892
893            tracing::debug!(
894                "SSTableReader::sequential_scan - Filtered to {} results before limit (limit: {:?})",
895                results.len(),
896                limit
897            );
898
899            // Sort by Murmur3 token order (spec §5, Appendix B §313), then truncate to limit.
900            sort_by_token_order(&mut results);
901            if let Some(lim) = limit {
902                results.truncate(lim);
903            }
904
905            tracing::debug!(
906                "SSTableReader::sequential_scan - Returning {} results after sort+limit",
907                results.len()
908            );
909            return Ok(results);
910        }
911
912        // Non-stitching path for other formats
913        let mut block_count = 0;
914        while let Some(block) = self.read_next_block(&cursor).await? {
915            // Cooperative cancellation (issue #2264): an index-less (Summary.db
916            // absent) SSTable materialises EVERY partition here in one pass; poll
917            // the token per block so a cancelled Flight `do_get` abandons the walk
918            // promptly instead of running to completion under the ~1–2 min backstop.
919            scan_cancel.check()?;
920            block_count += 1;
921            tracing::debug!(
922                "SSTableReader::sequential_scan - Read block {}, size {} bytes",
923                block_count,
924                block.len()
925            );
926
927            let entries = self.parse_block_entries_with_schema(&block, schema, true)?;
928            tracing::debug!(
929                "SSTableReader::sequential_scan - Block {} contains {} entries",
930                block_count,
931                entries.len()
932            );
933
934            for (i, (entry_table_id, entry_key, entry_value)) in entries.iter().enumerate() {
935                // Cooperative cancellation (issue #2264, roborev): the per-block
936                // poll above fires ONCE per `read_next_block` call, but an
937                // uncompressed/BTI-direct block returns the WHOLE data section as
938                // one contiguous unit (per `read_next_block_impl`'s doc comment) —
939                // so for that shape `entries` alone can number in the hundreds of
940                // thousands. Poll again here at the SAME 256-entry cadence used
941                // elsewhere so materialisation honours the interval regardless of
942                // how large a single block turned out to be, independent of
943                // whichever inner parser branch produced `entries`.
944                if i & 0xFF == 0 {
945                    scan_cancel.check()?;
946                }
947                tracing::debug!(
948                    "SSTableReader::sequential_scan - Block {} entry {}: table_id='{}', key={:?}",
949                    block_count,
950                    i,
951                    entry_table_id,
952                    entry_key
953                );
954
955                // Match table IDs - supports both qualified (keyspace.table) and unqualified (table) formats
956                // This allows queries with either format to match SSTables stored with either format
957                if !table_ids_match(entry_table_id, table_id) {
958                    tracing::debug!("SSTableReader::sequential_scan - Skipping entry: table_id mismatch ('{}' != '{}')",
959                              entry_table_id, table_id);
960                    continue;
961                }
962
963                // Check key range
964                if let Some(start) = start_key {
965                    if entry_key < start {
966                        tracing::debug!(
967                            "SSTableReader::sequential_scan - Skipping entry: key < start_key"
968                        );
969                        continue;
970                    }
971                }
972
973                if let Some(end) = end_key {
974                    if entry_key > end {
975                        tracing::debug!(
976                            "SSTableReader::sequential_scan - Skipping entry: key > end_key"
977                        );
978                        continue;
979                    }
980                }
981
982                // Extract write time from entry metadata
983                let _write_time = self.extract_write_time_from_entry(entry_key, entry_value);
984
985                // Filter out tombstones and expired data
986                if !self.filter_tombstone(entry_value) {
987                    tracing::debug!("SSTableReader::sequential_scan - Skipping entry: filtered out (tombstone or expired)");
988                    continue;
989                }
990
991                tracing::debug!("SSTableReader::sequential_scan - Including entry in results");
992                results.push((entry_key.clone(), entry_value.clone()));
993            }
994        }
995
996        tracing::debug!(
997            "SSTableReader::sequential_scan - Finished scanning {} blocks",
998            block_count
999        );
1000        tracing::debug!(
1001            "SSTableReader::sequential_scan - {} results before sort+limit",
1002            results.len()
1003        );
1004
1005        // Sort by Murmur3 token order (spec §5, Appendix B §313), then apply limit.
1006        // Limit is applied AFTER sort so that LIMIT N returns the N token-smallest
1007        // partitions (BLOCKING-1: limit-after-order).
1008        sort_by_token_order(&mut results);
1009        if let Some(lim) = limit {
1010            results.truncate(lim);
1011        }
1012
1013        tracing::debug!(
1014            "SSTableReader::sequential_scan - Returning {} results after sort+limit",
1015            results.len()
1016        );
1017        Ok(results)
1018    }

1028        &self,
1029        table_id: &TableId,
1030        start_key: Option<&RowKey>,
1031        end_key: Option<&RowKey>,

1034    ) -> Result<
1035        Vec<(

1039        )>,
1040    > {

1042

1044        // plain BTI scan, but surfaces per-cell write metadata for WRITETIME/TTL.

1046            return self

1048                .await;
1049        }
1050
1051        // Issue #815: independent per-scan cursor — no cross-scan serialization.
1052        let cursor = self.new_scan_cursor().await?;

1056            let mut file_guard = cursor.file.lock().await;
1057            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
1058        }

1060        // V5CompressedLegacy (stitching) path — the common path for Cassandra 5.0 SSTables.

1063                .stitch_and_parse_all_chunks_with_metadata(&cursor, schema)
1064                .await?;
1065

1067            for (_entry_table_id, entry_key, entry_value, cell_meta) in all_entries {
1068                if let Some(start) = start_key {
1069                    if &entry_key < start {
1070                        continue;
1071                    }
1072                }
1073                if let Some(end) = end_key {
1074                    if &entry_key > end {
1075                        continue;
1076                    }
1077                }
1078                if !self.filter_tombstone(&entry_value) {
1079                    continue;
1080                }
1081                results.push((entry_key, entry_value, cell_meta));
1082            }
1083
1084            sort_by_token_order_with_meta(&mut results);
1085            if let Some(lim) = limit {
1086                results.truncate(lim);
1087            }
1088
1089            tracing::debug!(
1090                "SSTableReader::scan_with_cell_metadata - Returning {} results (stitched path)",
1091                results.len()
1092            );

1097        // Per-cell metadata is not yet surfaced for block-entry formats.
1098        let plain = self
1099            .sequential_scan(
1100                table_id,
1101                start_key,
1102                end_key,
1103                limit,
1104                schema,
1105                &self.scan_cancel,
1106            )
1107            .await?;
1108        Ok(plain
1109            .into_iter()
1110            .map(|(k, v)| (k, v, std::collections::HashMap::new()))
1111            .collect())
1112    }
