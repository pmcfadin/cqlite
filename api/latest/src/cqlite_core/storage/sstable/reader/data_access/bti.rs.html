1//! BTI ("da") read paths: trie-resolved point lookups, single-partition seeks,
2//! clustering-slice narrowing, and the whole-Data.db BTI scan.
3//!
4//! BTI SSTables carry no Index.db/Summary.db. The Partitions.db trie is the
5//! authoritative present/absent oracle; every method here resolves an
6//! uncompressed Data.db offset via the trie and decodes only the chunk window
7//! that holds the target partition (issue #831 / #909 / #953 / #954).

46    /// Single-partition *seek* for the partition-targeted lookup fast path,
47    /// clustering-slice-aware (Issue #953 + #954, Epic #951).
48    ///
49    /// Where [`scan`](Self::scan) decodes EVERY partition in this SSTable and the
50    /// caller retains one, this resolves the target partition's `Data.db` offset
51    /// from the authoritative index (the BTI Partitions.db trie or the BIG
52    /// `Index.db`) and decodes ONLY that partition — the same per-partition decode
53    /// `scan` runs (`parse_block_emit` over the chunk-targeted decompressed
54    /// window), so its output is byte-for-byte identical to filtering the full
55    /// `scan` result down to `partition_key`.
56    ///
57    /// Offset domains (no-heuristics: authoritative resolved offsets only):
58    /// - **BTI ("da")** — `lookup_partition_via_bti_trie` returns the UNCOMPRESSED
59    ///   `Data.db` offset; a trie miss is authoritative absence.
60    /// - **BIG (`nb`)** — `lookup_partition_with_index` returns the partition's
61    ///   offset into the (uncompressed) data section. A hit is authoritative
62    ///   present; a MISS returns `Ok(None)` (the `Index.db` may be digest-keyed or
63    ///   incomplete, exactly the `get()` fallback rationale at #517) so the caller
64    ///   re-checks via a full scan rather than risk a false negative.
65    ///
66    /// Prefix-collision / wrong-offset guard: the decode
67    /// (`bti_decompress_and_parse_target_all`) re-verifies the decoded partition
68    /// key equals `partition_key` before collecting any row, so a BTI
69    /// prefix-collision candidate or a stale/mismatched index offset decodes to
70    /// nothing and is reported as absent — never a wrong partition. Every
71    /// clustering row of the matched partition is collected (not just the first),
72    /// so a multi-row partition returns all rows.
73    ///
74    /// Compiled only for the default (`not(tombstones)`) build: the manager's
75    /// seek-driven `scan_partition` exists only there, so under `tombstones` this
76    /// would be dead code.
77    ///
78    /// When `clustering` is `Some(slice)` AND this reader is BTI (`da`) with a
79    /// per-partition row index (`Rows.db`), the target partition's authoritative
80    /// row index is consulted to resolve the byte extent of the row-index
81    /// block(s) covering the requested clustering range, and ONLY that byte window
82    /// is decoded — so a `WHERE pk = ? AND ck </>/= ?` slice over a wide partition
83    /// decodes O(matched rows + index block slack) rather than the whole
84    /// partition. The post-scan `evaluate_leaf` backstop trims the
85    /// block-granularity over-read, so the returned rows are a superset of the
86    /// exact slice and the final query output is byte-identical to the
87    /// full-partition decode + post-filter.
88    ///
89    /// Returns `Ok(Some((rows, clustering_seek_engaged)))`:
90    /// - `clustering_seek_engaged == true` only when the clustering row-index
91    ///   narrowing actually bounded the decode (BTI wide partition with a usable
92    ///   row index and an encodable bound). The caller reports
93    ///   [`AccessPath::ClusteringSlice`](crate::query::access_path::AccessPath::ClusteringSlice)
94    ///   in that case.
95    /// - `clustering_seek_engaged == false` when the partition was decoded in full
96    ///   (no clustering slice, a NARROW BTI partition with no row index, the BIG
97    ///   format, or an un-encodable bound). Results are still correct — the caller
98    ///   reports the honest `PartitionLookup` path, NOT a fake clustering slice.
99    ///
100    /// `Ok(None)` mirrors [`scan_single_partition`]: the seek is not applicable
101    /// (no authoritative offset) and the caller must fall back to a full scan +
102    /// retain.
103    ///
104    /// [`scan_single_partition`]: crate::storage::sstable::SSTableReader
105    #[cfg(not(feature = "tombstones"))]
106    pub(crate) async fn scan_single_partition_clustering(
107        &self,
108        table_id: &TableId,
109        partition_key: &[u8],
110        clustering: Option<&ClusteringSlice>,
111        // `true` iff the manager resolved this reader by an EXACT fully-qualified
112        // `keyspace.table` match (or the query is unqualified). `false` means a
113        // fully-qualified query reached this reader via the bare-name fallback, in
114        // which case the seek guard keeps STRICT keyspace matching so it can never
115        // return rows from a different keyspace whose table name collides (#1284).
116        fully_qualified_match: bool,
117        schema: Option<&crate::schema::TableSchema>,
118    ) -> Result<Option<(Vec<(RowKey, ScanRow)>, bool)>> {
119        // 1. Resolve the partition's uncompressed Data.db offset, and record
120        //    whether THIS path's "decoded nothing" is authoritative absence (BTI
121        //    trie) or merely inconclusive (BIG Index.db).
122        let is_bti = self.bti_partitions_db.is_some();
123        // Resolve the target partition's UNCOMPRESSED `Data.db` start offset.
124        let offset = if is_bti {
125            match self.lookup_partition_via_bti_trie(partition_key)? {
126                // Trie hit: candidate uncompressed offset (re-verified on decode).
127                Some(off) => off,
128                // Trie miss is AUTHORITATIVE absence for BTI (no rows, no seek).
129                None => return Ok(Some((Vec::new(), false))),
130            }
131        } else {
132            match self.lookup_partition_with_index(partition_key).await? {
133                // Index.db hit: a candidate offset into the data section. (The
134                // `data_size` is 0 in writer-produced Index.db, so we do NOT use it
135                // as a bound — the successor offset below is the authoritative end.)
136                Some((off, _size)) => off,
137                // No Index.db hit: cannot seek authoritatively (the index may be
138                // digest-keyed / incomplete, exactly the get() #517 rationale).
139                // Fall back to a full scan.
140                None => return Ok(None),
141            }
142        };
143
144        // AUTHORITATIVE end bound (issue #953 / #951 MEDIUM): the target partition
145        // occupies `[offset, end)`, where `end` is the SUCCESSOR partition's start
146        // offset (next trie/index entry) — never a boundary-scan heuristic. It
147        // bounds BOTH the decompression and (issue #3890) the parse; the callee's
148        // doc comment states which is which and why they differ under a #954
149        // clustering narrowing. `None` = the LAST partition (no successor): the
150        // callee then uses the authoritative data-section length, or falls back to
151        // the safe full-scan path when even that is unknown.
152        let end_bound = self
153            .successor_partition_offset(offset, partition_key)
154            .await?
155            .map(|e| e as usize);
156
157        let schema_opt = self.get_table_schema(schema);
158
159        // Issue #954 / #1184: resolve the within-partition row-body byte window for a
160        // single-column clustering slice from the authoritative index (BTI `Rows.db`
161        // trie or BIG promoted `IndexInfo` blocks). The unified resolver lives in
162        // `big_promoted.rs` (campsite: keeps this over-threshold file from growing).
163        let (row_body_window, decode_end_bound, clustering_engaged) = self
164            .resolve_clustering_seek_window(
165                is_bti,
166                partition_key,
167                offset,
168                clustering,
169                schema_opt.as_ref(),
170                end_bound,
171            )
172            .await?;
173
174        // 2. Decode ONLY the target partition at the resolved offset. Issue #3721:
175        //    the decode step lives in `clustering_seek_decode` because a per-column
176        //    decode failure under an INDEX-POSITIONED narrowing must retract that
177        //    narrowing and re-read the full partition — a decision that needs BOTH
178        //    the narrowed and the authoritative bounds, which only this call has.
179        //    `clustering_engaged` comes back FALSE when the narrowing was retracted,
180        //    so the caller's reported `AccessPath` stays honest.
181        let key = RowKey::from(partition_key.to_vec());
182        let (decoded, clustering_engaged) = self
183            .decode_clustering_seek_target(
184                table_id,
185                partition_key,
186                &key,
187                is_bti,
188                fully_qualified_match,
189                offset,
190                // Issue #3890 x #3721: the UN-narrowed successor offset is BOTH
191                // the retraction bound and the authoritative PARSE bound —
192                // `decode_end_bound` may point INSIDE the partition, which is
193                // right for the DECOMPRESSION and wrong for the PARSE. Passed
194                // ONCE, as `full_end_bound`, so the two uses cannot drift apart.
195                end_bound,
196                decode_end_bound,
197                row_body_window,
198                clustering_engaged,
199                schema_opt.as_ref(),
200            )
201            .await?;
202        let decoded_rows = match decoded {
203            // Authoritatively bounded decode (rows may be empty for an absent key).
204            Some(rows) => rows,
205            // The seek could not bound the target partition authoritatively (the
206            // LAST partition with an unknown data-section length): fall back to the
207            // safe full scan + retain for correctness, per the #953 mandate.
208            None => return Ok(None),
209        };
210
211        // 3. Record the per-partition decode (Issue #953 / #958): exactly ONE
212        //    partition is decoded for a hit regardless of how many clustering rows
213        //    it yields — `partitions_decoded` counts partitions, not rows. This is
214        //    the signal that proves the within-SSTable seek (vs a full
215        //    parse-then-retain). A non-empty decode means the partition exists.
216        if !decoded_rows.is_empty() {
217            super::super::super::work_counters::add_partition_decoded();
218            // Tombstone suppression matches the user-facing scan path
219            // (`sequential_scan`/`bti_scan_with_metadata` both apply it),
220            // applied per-row so a row tombstone is dropped while live rows in
221            // the same partition survive.
222            let rows: Vec<(RowKey, ScanRow)> = decoded_rows
223                .into_iter()
224                .filter(|value| self.filter_tombstone(value))
225                .map(|value| (key.clone(), value))
226                .collect();
227            return Ok(Some((rows, clustering_engaged)));
228        }
229
230        // Decoded nothing at the resolved offset. Whether that is authoritative
231        // depends on HOW the offset was resolved (Constraint #4: never return a
232        // wrong/empty result from an unsupported/inconclusive seek):
233        //
234        // - **BTI** — the trie is the authoritative present/absent oracle and the
235        //   decode re-verified the key, so "decoded nothing" means the trie
236        //   candidate was a prefix-collision for an absent key. AUTHORITATIVE
237        //   empty: the caller does NOT fall back.
238        // - **BIG** — the `Index.db` offset is only a candidate position; its
239        //   promoted-index / chunk layout is not as load-bearing as the BTI trie,
240        //   so a failed decode at the resolved offset is INCONCLUSIVE (a partition
241        //   that straddles a chunk boundary, or a stale offset, can fail the
242        //   chunk-targeted decode yet be found by a full parse). Fall back to a
243        //   full scan rather than risk a false negative.
244        if is_bti {
245            // Authoritative absence (prefix-collision candidate for an absent key).
246            // No rows decoded, so report the clustering seek as NOT engaged.
247            Ok(Some((Vec::new(), false)))
248        } else {
249            Ok(None)
250        }
251    }
252
253    /// Resolve the within-partition row-body byte window covering a single-column
254    /// clustering slice, using the target partition's authoritative BTI row index
255    /// (Issue #954, Epic #951).
256    ///
257    /// For a WIDE BTI partition the `Partitions.db` trie points at a per-partition
258    /// `TrieIndexEntry` in `Rows.db`; that entry's row-index trie maps clustering
259    /// **separators** to row-index BLOCK offsets (relative to the partition
260    /// start). [`select_row_index_blocks_for_range`] applies the authoritative
261    /// separator-floor semantics to pick exactly the blocks whose key interval
262    /// intersects `[start, end]`, so the returned byte window is the smallest
263    /// authoritative extent that can contain the requested clustering range.
264    ///
265    /// Returns `Ok(Some(window))` only when the narrowing is authoritative and
266    /// useful:
267    /// - the reader is BTI with a `Rows.db`,
268    /// - the partition is WIDE (`Partitions.db` returned a `RowsOffset`, i.e. the
269    ///   partition has a row index — a NARROW partition has no per-partition row
270    ///   index to seek within),
271    /// - the clustering bound(s) encode to the OSS50 byte-comparable form, and
272    /// - the selected block set is non-empty.
273    ///
274    /// Returns `Ok(None)` (decode the whole partition, report `PartitionLookup`)
275    /// for every other case — a NARROW partition, an empty `Rows.db`, an
276    /// un-encodable bound, a slice that selects no block, or (issue #3002) an entry
277    /// whose row-index ROOT failed structural validation. This is the honest
278    /// fallback: correctness is preserved by decoding the full partition and
279    /// letting the post-scan backstop filter.
280    #[cfg(not(feature = "tombstones"))]
281    pub(super) fn bti_clustering_row_window(
282        &self,
283        partition_key: &[u8],
284        slice: &ClusteringSlice,
285        schema: Option<&crate::schema::TableSchema>,
286    ) -> Result<Option<ClusteringRowWindow>> {
287        use crate::storage::sstable::bti::{
288            encode_partition_key_for_bti_trie, lookup_partition_in_bti_slice,
289            resolve_rows_db_entry, rows_floor_block, rows_strict_ceiling_block,
290            BtiPartitionLocation,
291        };
292
293        let (Some(partitions_db), Some(rows_db)) = (&self.bti_partitions_db, &self.bti_rows_db)
294        else {
295            return Ok(None);
296        };
297
298        // Resolve the partition's location. Only a WIDE partition (RowsOffset) has
299        // a per-partition row index we can seek within; a NARROW partition
300        // (DataOffset) has none, so decode it in full.
301        // Issue #1574 (C3): walk the resident trie buffer in place (no whole-file copy).
302        // Issue #1575 (C4): encode the raw key then walk with the pre-encoded key via
303        // the single `lookup_partition_in_bti_slice` primitive every BTI lookup shares.
304        let encoded = encode_partition_key_for_bti_trie(partition_key);
305        let rows_offset = match lookup_partition_in_bti_slice(partitions_db.as_slice(), &encoded)
306            .map_err(|e| {
307                Error::corruption(format!(
308                    "BTI clustering seek: Partitions.db trie lookup failed (key len={}): {}",
309                    partition_key.len(),
310                    e
311                ))
312            })? {
313            Some(BtiPartitionLocation::RowsOffset(off)) => off as usize,
314            // NARROW partition or absent key: no row index to narrow with.
315            Some(BtiPartitionLocation::DataOffset(_)) | None => return Ok(None),
316        };
317
318        // Resolve the per-partition row-index entry ONCE (issue #1647 / L1: the
319        // pre-L1 path resolved it HERE and again inside `iterate_rows_for_partition`,
320        // discarding the first). `header.trie_root` roots the O(key-length)
321        // floor/ceiling walks below, so a clustering read never materializes every
322        // row-index block.
323        let header = resolve_rows_db_entry(rows_db.as_slice(), rows_offset).map_err(|e| {
324            Error::corruption(format!(
325                "BTI clustering seek: Rows.db entry at RowsOffset({rows_offset}) unreadable: {e}"
326            ))
327        })?;
328        // ROOT UNUSABLE (issue #3002): the entry's resolved root failed structural
329        // validation — it is not the last-written node before the entry, so walking
330        // from it would return a structurally valid but BOGUS window that silently
331        // drops rows (exactly the pre-#3002 fail-open). Take the honest "cannot
332        // narrow" fallback: `Ok(None)` ⇒ decode the whole partition and let the
333        // post-scan backstop filter. This is DISTINCT from the #1968 implicit-first
334        // signal below, which keeps a real (narrowed-END) window rooted at rel 0.
335        // `header.data_position`/`block_count` are unaffected and stay usable by the
336        // point-lookup / successor-walk paths.
337        let root = match &header.trie_root {
338            Ok(root) => root.offset(),
339            Err(rejection) => {
340                // OPERATOR SIGNAL (#3002): this fallback is otherwise invisible — it
341                // shows up only as unexplained clustering-read latency, because every
342                // slice over the affected partitions decodes in full. Count it with
343                // the violated invariant as the bounded attribute so a dashboard can
344                // name the cause; the `debug!` keeps the offsets, which never go on a
345                // metric label.
346                crate::observability::add_counter(
347                    crate::observability::catalog::READ_BTI_ROWS_ROOT_REJECTED,
348                    1,
349                    &[(
350                        crate::observability::catalog::attr::ROWS_ROOT_REJECT_REASON,
351                        rejection.reason.label().into(),
352                    )],
353                );
354                debug!(
355                    "BTI clustering seek: {rejection}; decoding the full partition (no \
356                     narrowing) — a Rows.db row index written by CQLite <= 0.16 must be \
357                     rewritten (re-flush/re-compact), see issue #3002"
358                );
359                return Ok(None);
360            }
361        };
362
363        // Per-column reverse order for the FIRST clustering column (single-column
364        // scope per #954). A missing/absent schema treats it as ascending.
365        let is_reversed: Vec<bool> = schema
366            .map(|s| {
367                s.clustering_keys
368                    .iter()
369                    .map(|c| matches!(c.order, crate::schema::ClusteringOrder::Desc))
370                    .collect()
371            })
372            .unwrap_or_default();
373
374        // Encode the CQL bounds into the PHYSICAL byte-comparable order the row
375        // index uses, normalizing for a DESC first clustering column (issue #954
376        // High-severity correctness fix). The floor/ceiling walks operate purely in
377        // physical (on-disk, byte-comparable) order, so the CQL lower/upper bounds
378        // must be mapped to the physical-lower/physical-upper sides before the walk.
379        // For a DESC column those roles SWAP (see `physical_byte_bounds_for_slice`).
380        // An un-encodable bound makes the narrowing unsafe → decode the whole
381        // partition (honest fallback).
382        let Some((start_bytes, end_bytes)) = physical_byte_bounds_for_slice(slice, &is_reversed)?
383        else {
384            return Ok(None);
385        };
386
387        // Locate the row-body window with two O(key-length) separator walks instead
388        // of materializing every row-index block then linearly filtering (issue
389        // #1647 / L1). Both mirror Cassandra `RowIndexReader.separatorFloor`:
390        //   - floor(start): the block whose half-open interval `[s_i, s_{i+1})`
391        //     contains the physical-lower bound (largest separator `<= start`), i.e.
392        //     the first block that can hold a row of the slice. `None` when `start`
393        //     sorts below the FIRST separator — the trie-implicit first block at the
394        //     partition body start (issue #1968), which no stored walk can return.
395        //   - strict_ceiling(end): the block with the smallest separator strictly
396        //     `> end` = the EXCLUSIVE window end (the successor of `floor(end)`).
397        //     `None` when the slice reaches the last block (window runs to the
398        //     partition end).
399        // These reproduce the pre-L1 `select_row_index_blocks_for_range` window
400        // (min selected-block start .. successor of max selected-block) EXACTLY, by
401        // construction, for every boundary class (see the rows_floor unit tests).
402        let floor_block =
403            rows_floor_block(rows_db.as_slice(), root, &start_bytes).map_err(|e| {
404                Error::corruption(format!(
405                "BTI clustering seek: Rows.db floor walk at RowsOffset({rows_offset}) failed: {e}"
406            ))
407            })?;
408        let ceil_block =
409            rows_strict_ceiling_block(rows_db.as_slice(), root, &end_bytes).map_err(|e| {
410                Error::corruption(format!(
411                    "BTI clustering seek: Rows.db ceiling walk at RowsOffset({rows_offset}) \
412                     failed: {e}"
413                ))
414            })?;
415
416        // The static row precedes the clustering rows and must be merged into each
417        // emitted clustering row, so we may only fast-forward PAST it when the table
418        // has NO static columns; otherwise decode from the partition body start
419        // (`body_start_rel = 0`) so the static prefix is seen (the END bound still
420        // narrows). (`test_da.wide_table` has no static columns, so the start narrows
421        // too.)
422        let has_static = schema
423            .map(|s| s.columns.iter().any(|c| c.is_static))
424            .unwrap_or(false);
425
426        // IMPLICIT FIRST BLOCK (issue #1968): a `start` below the FIRST stored
427        // separator selects a block no walk can return, so `rows_floor_block`'s
428        // `None` IS the implicit-first signal and the decode must begin at rel 0 or
429        // the earliest clustering rows are dropped. In a Cassandra-written trie read
430        // from the CORRECT root (issue #3002) this is now unreachable: the first
431        // block's separator is `ByteComparable.EMPTY` (`RowIndexWriter.add`), stored
432        // as the ROOT node's own payload, and nothing sorts below the empty key — the
433        // floor walk returns that block 0 entry as a genuine STORED floor. The `None`
434        // branch is retained for a trie whose first separator is NOT empty (a
435        // CQLite-written row index, or any bound below a non-empty first separator).
436        let includes_implicit_first_block = floor_block.is_none();
437
438        // The start narrows to the floor block only when NEITHER a static row NOR the
439        // implicit first block forces decoding from rel 0.
440        let narrows_start = !has_static && !includes_implicit_first_block;
441
442        // CORRECTNESS GUARD (no-heuristics, never wrong results): FLAG_OPEN_MARKER on
443        // the floor block's `IndexInfo` means a range tombstone is OPEN at the START
444        // of the narrowed window — a deletion opened in an earlier block still shadows
445        // rows inside the slice. Skipping earlier blocks would drop that range-open
446        // marker and risk resurrecting a deleted row (the post-scan backstop only
447        // FILTERS rows, it cannot re-apply a missed range tombstone). Per Cassandra
448        // `RowIndexReader.IndexInfo.openDeletion`, the marker on the floor block fully
449        // captures any deletion open at the window start, so this is the exact, tight
450        // guard: when it fires (only relevant when the start actually narrows) fall
451        // back to a full-partition decode. When the decode already starts at rel 0
452        // (static / implicit-first) no earlier block is skipped, so no guard is
453        // needed. (`test_da.wide_table` has no range tombstones.)
454        if narrows_start
455            && floor_block
456                .as_ref()
457                .map(|b| b.open_marker.is_some())
458                .unwrap_or(false)
459        {
460            debug!(
461                "BTI clustering seek: floor row-index block carries an open range-tombstone \
462                 marker; decoding full partition to preserve range-deletion semantics (no \
463                 narrowing)"
464            );
465            return Ok(None);
466        }
467
468        // body_start_rel: the floor block's offset when the start narrows, else the
469        // partition body start (rel 0). The block `data_offset` is relative to the
470        // partition start (the same domain the parser sees for `window[within..]`).
471        let body_start_rel = match &floor_block {
472            Some(b) if narrows_start => b.data_offset as usize,
473            _ => 0,
474        };
475        // body_end_rel: the exclusive successor block start, or the partition end
476        // (`usize::MAX`, clamped by the caller against the authoritative partition
477        // end / data-section length) when the slice reaches the last block.
478        let body_end_rel = ceil_block
479            .map(|b| b.data_offset as usize)
480            .unwrap_or(usize::MAX);
481
482        Ok(Some(ClusteringRowWindow {
483            body_start_rel,
484            body_end_rel,
485        }))
486    }

488    /// BTI ("da") full scan: decompress the whole Data.db section and parse
489    /// every partition in token order (issue #660).
490    ///
491    /// BTI SSTables carry no Index.db/Summary.db, so a range/full scan cannot
492    /// use the index path. Instead we stitch the entire (chunk-compressed) data
493    /// section into one buffer and run [`parse_block_with_cell_metadata`], which
494    /// walks ALL partitions — the same per-partition decode the point-lookup
495    /// path uses, but without stopping at the first match.
496    ///
497    /// Returns entries with per-cell write metadata so the WRITETIME/TTL scan
498    /// (`scan_with_cell_metadata`) and the plain `scan` (which drops the metadata)
499    /// can share a single implementation. Results are filtered by the optional
500    /// `[start_key, end_key]` range and tombstone-suppressed, then sorted into
501    /// Murmur3 token order and truncated to `limit` — identical post-processing
502    /// to the V5CompressedLegacy stitched path.
503    ///
504    /// Uses its own per-scan
505    /// [`ScanCursor`](crate::storage::sstable::reader::source::ScanCursor), so it
506    /// runs in parallel with other scans on this reader without serialization
507    /// (issue #815).
508    ///
509    /// [`parse_block_with_cell_metadata`]: crate::storage::sstable::reader::parsing::V5CompressedLegacyParser::parse_block_with_cell_metadata
510    ///
511    /// `read_shadowing` (issue #1741): `true` for user-facing SELECT scans
512    /// (`scan`, `scan_with_cell_metadata`), `false` for the physical
513    /// `get_all_entries` (integrity verification / data-manager) which must count
514    /// every on-disk row.
515    ///
516    /// `now_secs` (issue #3058, threaded here by #3109): a caller-pinned read-time
517    /// TTL clock. `Some` pins the decoder's expiry instant to the ONE
518    /// reconciliation instant the caller already captured for the request instead
519    /// of the ambient sample the parser takes at construction; `None` keeps the
520    /// ambient sample. Only consulted when `read_shadowing`.
521    pub(super) async fn bti_scan_with_metadata(
522        &self,
523        start_key: Option<&RowKey>,
524        end_key: Option<&RowKey>,
525        limit: Option<usize>,
526        schema: Option<&crate::schema::TableSchema>,
527        read_shadowing: bool,
528        now_secs: Option<i64>,
529    ) -> Result<
530        Vec<(
531            RowKey,
532            ScanRow,
533            std::collections::HashMap<String, CellWriteMetadata>,
534        )>,
535    > {
536        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
537        self.bti_scan_with_metadata_cancellable(
538            start_key,
539            end_key,
540            limit,
541            schema,
542            read_shadowing,
543            now_secs,
544            &self.scan_cancel,
545        )
546        .await
547    }

549    /// [`Self::bti_scan_with_metadata`] with an explicit PER-CALL cancellation
550    /// token (issue #2346/#2264), mirroring the
551    /// [`stitch_all_chunks`](Self::stitch_all_chunks) /
552    /// [`stitch_all_chunks_cancellable`](Self::stitch_all_chunks_cancellable) pair.
553    ///
554    /// `sequential_scan` is the seam that needs it: it takes a caller-supplied
555    /// token (the compaction path's
556    /// [`iterate_all_partitions_cancellable`](SSTableReader::iterate_all_partitions_cancellable)
557    /// drives it with one that is NOT the reader's own field), and its BTI branch
558    /// must honour that token. Routing a `da` reader through the non-cancellable
559    /// wrapper would poll the WRONG flag — a cancelled walk would stitch and parse
560    /// the entire data section and return `Ok(every row)`, reporting success for a
561    /// scan the caller abandoned.
562    ///
563    /// The token is polled in BOTH phases of the walk: every 256 chunks inside the
564    /// stitch (the I/O phase) and every 256 entries of the post-parse filter loop —
565    /// the same two-phase cadence `sequential_scan`'s stitched branch uses.
566    #[allow(clippy::too_many_arguments)]
567    pub(super) async fn bti_scan_with_metadata_cancellable(
568        &self,
569        start_key: Option<&RowKey>,
570        end_key: Option<&RowKey>,
571        limit: Option<usize>,
572        schema: Option<&crate::schema::TableSchema>,
573        read_shadowing: bool,
574        now_secs: Option<i64>,
575        scan_cancel: &crate::storage::scan_cancel::ScanCancel,
576    ) -> Result<
577        Vec<(
578            RowKey,
579            ScanRow,
580            std::collections::HashMap<String, CellWriteMetadata>,
581        )>,
582    > {
583        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
584        scan_cancel.check()?;
585        let cursor = self.new_scan_cursor().await?;
586
587        // Decompress the entire data section. Precondition for stitch_all_chunks:
588        // cursor's file seeked to data-section start (fresh cursor is at chunk 0).
589        let header_size = self.calculate_header_size();
590        {
591            let mut file_guard = cursor.file.lock().await;
592            // A5 read-work counter (SEEK_CALLS; consumer E4): scan seek-to-data-start
593            // before stitching the section. No-op in release (design.md Decision 1/2).
594            crate::storage::sstable::read_work_counters::record_seek();
595            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
596        }
597        let whole = self
598            .stitch_all_chunks_cancellable(&cursor, scan_cancel)
599            .await?;
600
601        // Resolve schema via the four-tier strategy (provided > header > registry).
602        // V5CompressedLegacy partition decode requires a schema (cells lack names).
603        let effective_schema = self.get_table_schema(schema);
604        let parser = self.build_v5_parser(read_shadowing);
605        // Issue #3058/#3109: honour the caller's pinned reconciliation clock, so a
606        // request that already sampled ONE `now` expires TTL cells at exactly that
607        // instant on this decoder too (mirrors `parse_block_entries_at_now` and the
608        // windowed driver).
609        let parser = match now_secs {
610            Some(now) => parser.with_now_secs(now),
611            None => parser,
612        };
613        // Issue #3782 (roborev job 48): `whole` is the ENTIRE data section —
614        // every chunk, stitched from a fresh cursor seeked to the data-section
615        // start above — so no further bytes can arrive to finish a row and a row
616        // decode error here is DATA LOSS. Before this declaration the `da` full
617        // scan (this is the route `scan`/`get_all_entries`/the batched streaming
618        // surface all take) inherited the tolerant break and returned `Ok` with
619        // 120 of 468 rows on a fixture with ONE compressed byte flipped inside a
620        // clustering value.
621        let parsed = parser.parse_block_with_cell_metadata(
622            &whole,
623            BufferExtent::Complete,
624            effective_schema.as_ref(),
625            self,
626        )?;
627
628        let mut results = Vec::new();
629        // Work-probe (issue #2398, threaded to BTI by #3109): "changed partition key
630        // = one more partition BODY decoded", counted BEFORE any range/tombstone
631        // filter exactly like the sibling walks (`sequential_scan`, the stitched
632        // walk, `run_scan_stream_batched`'s block loop). Without it the BTI decode
633        // would report ZERO scan work — including on the batched streaming surface,
634        // whose block loop counted before #3109 routed BTI readers here.
635        let mut prev_partition_key: Option<RowKey> = None;
636        for (idx, (_entry_table_id, entry_key, entry_value, cell_meta)) in
637            parsed.into_iter().enumerate()
638        {
639            // Cooperative cancellation (issue #2346/#2264): the stitch poll above
640            // covers the I/O phase, but `parse_block_with_cell_metadata`
641            // materialises every entry in one shot — poll here at the same
642            // 256-entry cadence as `sequential_scan`'s stitched branch so a
643            // cancelled caller does not walk a huge already-parsed result set to
644            // completion and then report success.
645            //
646            // KNOWN GAP for a QUERY TIMEOUT (#1695, roborev round 12), recorded here
647            // so it need not be re-derived: `scan_cancel` is the READER's SHARED
648            // token. #2264 trips it on a Flight client disconnect, and #2361 trips it
649            // when an iterator adapter drops — but NOTHING trips it when a streaming
650            // query's consumer goes away, so a timed-out scan over a BTI table keeps
651            // materialising this whole `results` Vec. The sibling producers were fixed
652            // by consulting their own channel (`tx.is_closed()` / racing
653            // `tx.closed()`), which is not available here: this function RETURNS a Vec
654            // and holds no sender.
655            //
656            // WHAT IS ACTUALLY MISSING, verified rather than guessed (roborev raised
657            // this twice, so the next person should not have to re-derive it):
658            // `bti_scan_with_metadata_cancellable` already exists and already takes a
659            // `&ScanCancel`, so the plumbing is NOT the gap. The gap is that the
660            // caller has no token it can legitimately pass:
661            //
662            //  * A CLONE of the reader's token is wrong — clones share state, so
663            //    tripping it cancels OTHER queries scanning the same reader. That is
664            //    the trap the materializing merges avoided with a per-call token.
665            //  * A FRESH per-call token tripped by a watcher on the output sender's
666            //    `closed()` handles the timeout case but silently DROPS #2264's Flight
667            //    cancellation, which trips the reader token WITHOUT necessarily
668            //    dropping the receiver — so a client disconnect would go back to the
669            //    ~1–2 min transport backstop. Losing existing cancellation to add new
670            //    cancellation is not a fix.
671            //
672            // So this needs one small API decision — a token that represents "either
673            // the reader's OR this stream's closure" (a linked/derived `ScanCancel`) —
674            // and that composition is the same primitive the deferred `JoinedStream` /
675            // `KWayMerger::step()` work needs. It is deliberately not improvised here:
676            // three partial fixes in this area is how the gap got this scattered, and a
677            // fourth guessing at token composition would be the worst of them.
678            scan_cancel.checkpoint(idx).await?;
679            if prev_partition_key.as_ref() != Some(&entry_key) {
680                crate::storage::sstable::work_counters::add_stream_walk_partition_parsed();
681                prev_partition_key = Some(entry_key.clone());
682            }
683            if let Some(start) = start_key {
684                if &entry_key < start {
685                    continue;
686                }
687            }
688            if let Some(end) = end_key {
689                if &entry_key > end {
690                    continue;
691                }
692            }
693            if !self.filter_tombstone(&entry_value) {
694                continue;
695            }
696            results.push((entry_key, entry_value, cell_meta));
697        }
698
699        sort_by_token_order_with_meta(&mut results);
700        if let Some(lim) = limit {
701            results.truncate(lim);
702        }
703
704        tracing::debug!(
705            "SSTableReader::bti_scan_with_metadata - Returning {} results",
706            results.len()
707        );
708        Ok(results)
709    }

711    /// The BTI (`da`) dispatch for BOTH public streaming surfaces — the per-row
712    /// `run_scan_stream` and the batched `run_scan_stream_batched` — in ONE place
713    /// (issue #3109).
714    ///
715    /// # Why a shared helper and not a third copy
716    ///
717    /// Issue #1577 established the rule: a BTI reader MUST decode through the SAME
718    /// authoritative trie-walk decoder [`Self::bti_scan_with_metadata`] that
719    /// [`SSTableReader::scan`] uses, NEVER the block-by-block `read_next_block` +
720    /// `parse_block_entries*` route. For `da` that route lands in the
721    /// `V5UncompressedOA` STATE MACHINE, which takes neither `read_shadowing` nor
722    /// `now_secs` and therefore silently DROPS both (see the "KNOWN FAIL-OPEN SEAM"
723    /// note in `parsing/block_entries.rs`, issue #3108) — so a `da` table read that
724    /// way is UNSHADOWED: partition/range tombstones and TTL expiry are not applied,
725    /// and on a schema-required fixture it fails outright ("Blob fallback not allowed
726    /// for V5_0Bti"). #1577 fixed the per-row surface; the batched surface was added
727    /// without the dispatch and reproduced the identical defect (#3109). Both now
728    /// call THIS function, gating on the SAME `bti_partitions_db.is_some()` condition
729    /// `scan` uses, so the three surfaces can never again disagree about which
730    /// readers are BTI or about the posture their rows are decoded under.
731    ///
732    /// The rows are exactly `scan`'s BTI rows: `bti_scan_with_metadata` has already
733    /// applied the key-range and tombstone filters and token-ordered them, so this
734    /// forwards them AS-IS and the stream stays prefix-authoritative with `scan`
735    /// (what D1's LIMIT pushdown relies on). BTI decode fully materializes the
736    /// (index-less) reconciled table before streaming — mirrored by
737    /// `scan_stream_materializes` returning `true` for BTI.
738    ///
739    /// `out` selects the emission shape only: one send per ROW for
740    /// [`WindowedOut::PerRow`], one send per `BATCH_EMIT_ROWS`-capped BATCH for
741    /// [`WindowedOut::Batched`] — the same cap and the same "flattening the batches
742    /// yields the per-row stream" contract the windowed driver honours, so the two
743    /// surfaces stay row-for-row identical for BTI by construction. A closed
744    /// consumer channel ends the scan cleanly (`Ok(())`), matching both callers.
745    pub(super) async fn stream_bti_scan(
746        &self,
747        start_key: Option<&RowKey>,
748        end_key: Option<&RowKey>,
749        schema: Option<&crate::schema::TableSchema>,
750        now_secs: Option<i64>,
751        out: &WindowedOut,
752    ) -> Result<()> {
753        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
754
755        // #1695 (roborev raised this in rounds 12, 14 and 15): race the
756        // materialization against OUR consumer's departure.
757        //
758        // `bti_scan_with_metadata` materializes the WHOLE table into `entries` before
759        // the first send below, so the send-failure checks in the arms are reached
760        // only after all the work is done. Its internal `scan_cancel.checkpoint(idx)`
761        // polls the READER-WIDE token, which #2264 trips on a Flight disconnect and
762        // #2361 on adapter teardown — but nothing trips when a timed-out streaming
763        // query drops its iterator. So a timed-out query kept decoding and sorting a
764        // whole BTI table.
765        //
766        // Racing the scan FUTURE is what fixes it, and it needs no new cancellation
767        // plumbing: the checkpoints inside are await points, so dropping the future
768        // there abandons the scan within one checkpoint interval. Crucially the
769        // reader-wide token is still passed through untouched, so #2264 and #2361 keep
770        // working — nothing is traded away for this. (An earlier attempt added a
771        // "derive a per-stream token from the reader's" API for the same purpose; it
772        // was removed once the race proved to give the same semantics with no new
773        // surface. Do not reintroduce it without a caller the race cannot serve.)
774        //
775        // `biased` so an already-departed consumer wins over a ready scan.
776        //
777        // RESIDUAL, deliberately: `sort_by_token_order_with_meta` after the loop is
778        // SYNCHRONOUS, so a departure during the sort is not observed until it ends.
779        // That is bounded by one sort rather than a whole table walk, and is the same
780        // uninterruptible-post-scan-stage limit documented at the chokepoint.
781        let scan = self.bti_scan_with_metadata(start_key, end_key, None, schema, true, now_secs);
782        let entries = match out {
783            WindowedOut::PerRow(tx) => tokio::select! {
784                biased;
785                _ = tx.closed() => return Ok(()),
786                result = scan => result?,
787            },
788            WindowedOut::Batched(tx) => tokio::select! {
789                biased;
790                _ = tx.closed() => return Ok(()),
791                result = scan => result?,
792            },
793        };
794
795        match out {
796            WindowedOut::PerRow(tx) => {
797                for (entry_key, entry_value, _meta) in entries {
798                    if tx.send(Ok((entry_key, entry_value))).await.is_err() {
799                        return Ok(()); // consumer dropped
800                    }
801                }
802            }
803            WindowedOut::Batched(tx) => {
804                let mut batch: Vec<(RowKey, ScanRow)> = Vec::with_capacity(BATCH_EMIT_ROWS);
805                for (entry_key, entry_value, _meta) in entries {
806                    batch.push((entry_key, entry_value));
807                    if batch.len() >= BATCH_EMIT_ROWS {
808                        if tx.send(Ok(std::mem::take(&mut batch))).await.is_err() {
809                            return Ok(()); // consumer dropped
810                        }
811                        batch.reserve(BATCH_EMIT_ROWS);
812                    }
813                }
814                if !batch.is_empty() && tx.send(Ok(batch)).await.is_err() {
815                    return Ok(()); // consumer dropped
816                }
817            }
818        }
819        Ok(())
820    }
