48    /// Single-partition *seek* for the partition-targeted lookup fast path,
49    /// clustering-slice-aware (Issue #953 + #954, Epic #951).
50    ///
51    /// Where [`scan`](Self::scan) decodes EVERY partition in this SSTable and the
52    /// caller retains one, this resolves the target partition's `Data.db` offset
53    /// from the authoritative index (the BTI Partitions.db trie or the BIG
54    /// `Index.db`) and decodes ONLY that partition — the same per-partition decode
55    /// `scan` runs (`parse_block_emit` over the chunk-targeted decompressed
56    /// window), so its output is byte-for-byte identical to filtering the full
57    /// `scan` result down to `partition_key`.
58    ///
59    /// Offset domains (no-heuristics: authoritative resolved offsets only):
60    /// - **BTI ("da")** — `lookup_partition_via_bti_trie` returns the UNCOMPRESSED
61    ///   `Data.db` offset; a trie miss is authoritative absence.
62    /// - **BIG (`nb`)** — `lookup_partition_with_index` returns the partition's
63    ///   offset into the (uncompressed) data section. A hit is authoritative
64    ///   present; a MISS returns `Ok(None)` (the `Index.db` may be digest-keyed or
65    ///   incomplete, exactly the `get()` fallback rationale at #517) so the caller
66    ///   re-checks via a full scan rather than risk a false negative.
67    ///
68    /// Prefix-collision / wrong-offset guard: the decode
69    /// (`bti_decompress_and_parse_target_all`) re-verifies the decoded partition
70    /// key equals `partition_key` before collecting any row, so a BTI
71    /// prefix-collision candidate or a stale/mismatched index offset decodes to
72    /// nothing and is reported as absent — never a wrong partition. Every
73    /// clustering row of the matched partition is collected (not just the first),
74    /// so a multi-row partition returns all rows.
75    ///
76    /// Compiled only for the default (`not(tombstones)`) build: the manager's
77    /// seek-driven `scan_partition` exists only there, so under `tombstones` this
78    /// would be dead code.
79    ///
80    /// When `clustering` is `Some(slice)` AND this reader is BTI (`da`) with a
81    /// per-partition row index (`Rows.db`), the target partition's authoritative
82    /// row index is consulted to resolve the byte extent of the row-index
83    /// block(s) covering the requested clustering range, and ONLY that byte window
84    /// is decoded — so a `WHERE pk = ? AND ck </>/= ?` slice over a wide partition
85    /// decodes O(matched rows + index block slack) rather than the whole
86    /// partition. The post-scan `evaluate_leaf` backstop trims the
87    /// block-granularity over-read, so the returned rows are a superset of the
88    /// exact slice and the final query output is byte-identical to the
89    /// full-partition decode + post-filter.
90    ///
91    /// Returns `Ok(Some((rows, clustering_seek_engaged)))`:
92    /// - `clustering_seek_engaged == true` only when the clustering row-index
93    ///   narrowing actually bounded the decode (BTI wide partition with a usable
94    ///   row index and an encodable bound). The caller reports
95    ///   [`AccessPath::ClusteringSlice`](crate::query::access_path::AccessPath::ClusteringSlice)
96    ///   in that case.
97    /// - `clustering_seek_engaged == false` when the partition was decoded in full
98    ///   (no clustering slice, a NARROW BTI partition with no row index, the BIG
99    ///   format, or an un-encodable bound). Results are still correct — the caller
100    ///   reports the honest `PartitionLookup` path, NOT a fake clustering slice.
101    ///
102    /// `Ok(None)` mirrors [`scan_single_partition`]: the seek is not applicable
103    /// (no authoritative offset) and the caller must fall back to a full scan +
104    /// retain.
105    ///
106    /// [`scan_single_partition`]: crate::storage::sstable::SSTableReader
107    #[cfg(not(feature = "tombstones"))]
108    pub(crate) async fn scan_single_partition_clustering(
109        &self,
110        table_id: &TableId,
111        partition_key: &[u8],
112        clustering: Option<&ClusteringSlice>,
113        // `true` iff the manager resolved this reader by an EXACT fully-qualified
114        // `keyspace.table` match (or the query is unqualified). `false` means a
115        // fully-qualified query reached this reader via the bare-name fallback, in
116        // which case the seek guard keeps STRICT keyspace matching so it can never
117        // return rows from a different keyspace whose table name collides (#1284).
118        fully_qualified_match: bool,
119        schema: Option<&crate::schema::TableSchema>,
120    ) -> Result<Option<(Vec<(RowKey, ScanRow)>, bool)>> {
121        // 1. Resolve the partition's uncompressed Data.db offset, and record
122        //    whether THIS path's "decoded nothing" is authoritative absence (BTI
123        //    trie) or merely inconclusive (BIG Index.db).
124        let is_bti = self.bti_partitions_db.is_some();
125        // Resolve the target partition's UNCOMPRESSED `Data.db` start offset.
126        let offset = if is_bti {
127            match self.lookup_partition_via_bti_trie(partition_key)? {
128                // Trie hit: candidate uncompressed offset (re-verified on decode).
129                Some(off) => off,
130                // Trie miss is AUTHORITATIVE absence for BTI (no rows, no seek).
131                None => return Ok(Some((Vec::new(), false))),
132            }
133        } else {
134            match self.lookup_partition_with_index(partition_key).await? {
135                // Index.db hit: a candidate offset into the data section. (The
136                // `data_size` is 0 in writer-produced Index.db, so we do NOT use it
137                // as a bound — the successor offset below is the authoritative end.)
138                Some((off, _size)) => off,
139                // No Index.db hit: cannot seek authoritatively (the index may be
140                // digest-keyed / incomplete, exactly the get() #517 rationale).
141                // Fall back to a full scan.
142                None => return Ok(None),
143            }
144        };
145
146        // AUTHORITATIVE end bound (issue #953 / #951 MEDIUM): the target partition
147        // occupies `[offset, end)`, where `end` is the SUCCESSOR partition's start
148        // offset (next trie/index entry) — never a boundary-scan heuristic. It
149        // bounds BOTH the decompression and (issue #3890) the parse; the callee's
150        // doc comment states which is which and why they differ under a #954
151        // clustering narrowing. `None` = the LAST partition (no successor): the
152        // callee then uses the authoritative data-section length, or falls back to
153        // the safe full-scan path when even that is unknown.
154        let end_bound = self
155            .successor_partition_offset(offset, partition_key)
156            .await?
157            .map(|e| e as usize);
158
159        let schema_opt = self.get_table_schema(schema);
160
161        // Issue #954 / #1184: resolve the within-partition row-body byte window for a
162        // single-column clustering slice from the authoritative index (BTI `Rows.db`
163        // trie or BIG promoted `IndexInfo` blocks). The unified resolver lives in
164        // `big_promoted.rs` (campsite: keeps this over-threshold file from growing).
165        let (row_body_window, decode_end_bound, clustering_engaged) = self
166            .resolve_clustering_seek_window(
167                is_bti,
168                partition_key,
169                offset,
170                clustering,
171                schema_opt.as_ref(),
172                end_bound,
173            )
174            .await?;
175
176        // 2. Decode ONLY the target partition at the resolved offset. Issue #3721:
177        //    the decode step lives in `clustering_seek_decode` because a per-column
178        //    decode failure under an INDEX-POSITIONED narrowing must retract that
179        //    narrowing and re-read the full partition — a decision that needs BOTH
180        //    the narrowed and the authoritative bounds, which only this call has.
181        //    `clustering_engaged` comes back FALSE when the narrowing was retracted,
182        //    so the caller's reported `AccessPath` stays honest.
183        let key = RowKey::from(partition_key.to_vec());
184        let (decoded, clustering_engaged) = self
185            .decode_clustering_seek_target(
186                table_id,
187                partition_key,
188                &key,
189                is_bti,
190                fully_qualified_match,
191                offset,
192                // Issue #3890 x #3721: the UN-narrowed successor offset is BOTH
193                // the retraction bound and the authoritative PARSE bound —
194                // `decode_end_bound` may point INSIDE the partition, which is
195                // right for the DECOMPRESSION and wrong for the PARSE. Passed
196                // ONCE, as `full_end_bound`, so the two uses cannot drift apart.
197                end_bound,
198                decode_end_bound,
199                row_body_window,
200                clustering_engaged,
201                schema_opt.as_ref(),
202            )
203            .await?;
204        let decoded_rows = match decoded {
205            // Authoritatively bounded decode (rows may be empty for an absent key).
206            Some(rows) => rows,
207            // The seek could not bound the target partition authoritatively (the
208            // LAST partition with an unknown data-section length): fall back to the
209            // safe full scan + retain for correctness, per the #953 mandate.
210            None => return Ok(None),
211        };
212
213        // 3. Record the per-partition decode (Issue #953 / #958): exactly ONE
214        //    partition is decoded for a hit regardless of how many clustering rows
215        //    it yields — `partitions_decoded` counts partitions, not rows. This is
216        //    the signal that proves the within-SSTable seek (vs a full
217        //    parse-then-retain). A non-empty decode means the partition exists.
218        if !decoded_rows.is_empty() {
219            super::super::super::work_counters::add_partition_decoded();
220            // Tombstone suppression matches the user-facing scan path
221            // (`sequential_scan`/`bti_scan_with_metadata` both apply it),
222            // applied per-row so a row tombstone is dropped while live rows in
223            // the same partition survive.
224            let rows: Vec<(RowKey, ScanRow)> = decoded_rows
225                .into_iter()
226                .filter(|value| self.filter_tombstone(value))
227                .map(|value| (key.clone(), value))
228                .collect();
229            return Ok(Some((rows, clustering_engaged)));
230        }
231
232        // Decoded nothing at the resolved offset. Whether that is authoritative
233        // depends on HOW the offset was resolved (Constraint #4: never return a
234        // wrong/empty result from an unsupported/inconclusive seek):
235        //
236        // - **BTI** — the trie is the authoritative present/absent oracle and the
237        //   decode re-verified the key, so "decoded nothing" means the trie
238        //   candidate was a prefix-collision for an absent key. AUTHORITATIVE
239        //   empty: the caller does NOT fall back.
240        // - **BIG** — the `Index.db` offset is only a candidate position; its
241        //   promoted-index / chunk layout is not as load-bearing as the BTI trie,
242        //   so a failed decode at the resolved offset is INCONCLUSIVE (a partition
243        //   that straddles a chunk boundary, or a stale offset, can fail the
244        //   chunk-targeted decode yet be found by a full parse). Fall back to a
245        //   full scan rather than risk a false negative.
246        if is_bti {
247            // Authoritative absence (prefix-collision candidate for an absent key).
248            // No rows decoded, so report the clustering seek as NOT engaged.
249            Ok(Some((Vec::new(), false)))
250        } else {
251            Ok(None)
252        }
253    }
254
255    /// Resolve the within-partition row-body byte window covering a single-column
256    /// clustering slice, using the target partition's authoritative BTI row index
257    /// (Issue #954, Epic #951).
258    ///
259    /// For a WIDE BTI partition the `Partitions.db` trie points at a per-partition
260    /// `TrieIndexEntry` in `Rows.db`; that entry's row-index trie maps clustering
261    /// **separators** to row-index BLOCK offsets (relative to the partition
262    /// start). [`select_row_index_blocks_for_range`] applies the authoritative
263    /// separator-floor semantics to pick exactly the blocks whose key interval
264    /// intersects `[start, end]`, so the returned byte window is the smallest
265    /// authoritative extent that can contain the requested clustering range.
266    ///
267    /// Returns `Ok(Some(window))` only when the narrowing is authoritative and
268    /// useful:
269    /// - the reader is BTI with a `Rows.db`,
270    /// - the partition is WIDE (`Partitions.db` returned a `RowsOffset`, i.e. the
271    ///   partition has a row index — a NARROW partition has no per-partition row
272    ///   index to seek within),
273    /// - the clustering bound(s) encode to the OSS50 byte-comparable form, and
274    /// - the selected block set is non-empty.
275    ///
276    /// Returns `Ok(None)` (decode the whole partition, report `PartitionLookup`)
277    /// for every other case — a NARROW partition, an empty `Rows.db`, an
278    /// un-encodable bound, a slice that selects no block, or (issue #3002) an entry
279    /// whose row-index ROOT failed structural validation. This is the honest
280    /// fallback: correctness is preserved by decoding the full partition and
281    /// letting the post-scan backstop filter.
282    #[cfg(not(feature = "tombstones"))]
283    pub(super) fn bti_clustering_row_window(
284        &self,
285        partition_key: &[u8],
286        slice: &ClusteringSlice,
287        schema: Option<&crate::schema::TableSchema>,
288    ) -> Result<Option<ClusteringRowWindow>> {
289        use crate::storage::sstable::bti::{
290            encode_partition_key_for_bti_trie, lookup_partition_in_bti_slice,
291            resolve_rows_db_entry, rows_floor_block, rows_strict_ceiling_block,
292            BtiPartitionLocation,
293        };
294
295        let (Some(partitions_db), Some(rows_db)) = (&self.bti_partitions_db, &self.bti_rows_db)
296        else {
297            return Ok(None);
298        };
299
300        // Resolve the partition's location. Only a WIDE partition (RowsOffset) has
301        // a per-partition row index we can seek within; a NARROW partition
302        // (DataOffset) has none, so decode it in full.
303        // Issue #1574 (C3): walk the resident trie buffer in place (no whole-file copy).
304        // Issue #1575 (C4): encode the raw key then walk with the pre-encoded key via
305        // the single `lookup_partition_in_bti_slice` primitive every BTI lookup shares.
306        let encoded = encode_partition_key_for_bti_trie(partition_key);
307        let rows_offset = match lookup_partition_in_bti_slice(partitions_db.as_slice(), &encoded)
308            .map_err(|e| {
309                Error::corruption(format!(
310                    "BTI clustering seek: Partitions.db trie lookup failed (key len={}): {}",
311                    partition_key.len(),
312                    e
313                ))
314            })? {
315            Some(BtiPartitionLocation::RowsOffset(off)) => off as usize,
316            // NARROW partition or absent key: no row index to narrow with.
317            Some(BtiPartitionLocation::DataOffset(_)) | None => return Ok(None),
318        };
319
320        // Resolve the per-partition row-index entry ONCE (issue #1647 / L1: the
321        // pre-L1 path resolved it HERE and again inside `iterate_rows_for_partition`,
322        // discarding the first). `header.trie_root` roots the O(key-length)
323        // floor/ceiling walks below, so a clustering read never materializes every
324        // row-index block.
325        let header = resolve_rows_db_entry(rows_db.as_slice(), rows_offset).map_err(|e| {
326            Error::corruption(format!(
327                "BTI clustering seek: Rows.db entry at RowsOffset({rows_offset}) unreadable: {e}"
328            ))
329        })?;
330        // ROOT UNUSABLE (issue #3002): the entry's resolved root failed structural
331        // validation — it is not the last-written node before the entry, so walking
332        // from it would return a structurally valid but BOGUS window that silently
333        // drops rows (exactly the pre-#3002 fail-open). Take the honest "cannot
334        // narrow" fallback: `Ok(None)` ⇒ decode the whole partition and let the
335        // post-scan backstop filter. This is DISTINCT from the #1968 implicit-first
336        // signal below, which keeps a real (narrowed-END) window rooted at rel 0.
337        // `header.data_position`/`block_count` are unaffected and stay usable by the
338        // point-lookup / successor-walk paths.
339        let root = match &header.trie_root {
340            Ok(root) => root.offset(),
341            Err(rejection) => {
342                // OPERATOR SIGNAL (#3002): this fallback is otherwise invisible — it
343                // shows up only as unexplained clustering-read latency, because every
344                // slice over the affected partitions decodes in full. Count it with
345                // the violated invariant as the bounded attribute so a dashboard can
346                // name the cause; the `debug!` keeps the offsets, which never go on a
347                // metric label.
348                crate::observability::add_counter(
349                    crate::observability::catalog::READ_BTI_ROWS_ROOT_REJECTED,
350                    1,
351                    &[(
352                        crate::observability::catalog::attr::ROWS_ROOT_REJECT_REASON,
353                        rejection.reason.label().into(),
354                    )],
355                );
356                debug!(
357                    "BTI clustering seek: {rejection}; decoding the full partition (no \
358                     narrowing) — a Rows.db row index written by CQLite <= 0.16 must be \
359                     rewritten (re-flush/re-compact), see issue #3002"
360                );
361                return Ok(None);
362            }
363        };
364
365        // Per-column reverse order for the FIRST clustering column (single-column
366        // scope per #954). A missing/absent schema treats it as ascending.
367        let (is_reversed, is_timeuuid) = clustering_flags(schema);
368
369        // Encode the CQL bounds into the PHYSICAL byte-comparable order the row
370        // index uses, normalizing for a DESC first clustering column (issue #954
371        // High-severity correctness fix). The floor/ceiling walks operate purely in
372        // physical (on-disk, byte-comparable) order, so the CQL lower/upper bounds
373        // must be mapped to the physical-lower/physical-upper sides before the walk.
374        // For a DESC column those roles SWAP (see `physical_byte_bounds_for_slice`).
375        // An un-encodable bound makes the narrowing unsafe → decode the whole
376        // partition (honest fallback).
377        let Some((start_bytes, end_bytes)) =
378            physical_byte_bounds_for_slice(slice, &is_reversed, &is_timeuuid)?
379        else {
380            return Ok(None);
381        };
382
383        // Locate the row-body window with two O(key-length) separator walks instead
384        // of materializing every row-index block then linearly filtering (issue
385        // #1647 / L1). Both mirror Cassandra `RowIndexReader.separatorFloor`:
386        //   - floor(start): the block whose half-open interval `[s_i, s_{i+1})`
387        //     contains the physical-lower bound (largest separator `<= start`), i.e.
388        //     the first block that can hold a row of the slice. `None` when `start`
389        //     sorts below the FIRST separator — the trie-implicit first block at the
390        //     partition body start (issue #1968), which no stored walk can return.
391        //   - strict_ceiling(end): the block with the smallest separator strictly
392        //     `> end` = the EXCLUSIVE window end (the successor of `floor(end)`).
393        //     `None` when the slice reaches the last block (window runs to the
394        //     partition end).
395        // These reproduce the pre-L1 `select_row_index_blocks_for_range` window
396        // (min selected-block start .. successor of max selected-block) EXACTLY, by
397        // construction, for every boundary class (see the rows_floor unit tests).
398        let floor_block =
399            rows_floor_block(rows_db.as_slice(), root, &start_bytes).map_err(|e| {
400                Error::corruption(format!(
401                "BTI clustering seek: Rows.db floor walk at RowsOffset({rows_offset}) failed: {e}"
402            ))
403            })?;
404        let ceil_block =
405            rows_strict_ceiling_block(rows_db.as_slice(), root, &end_bytes).map_err(|e| {
406                Error::corruption(format!(
407                    "BTI clustering seek: Rows.db ceiling walk at RowsOffset({rows_offset}) \
408                     failed: {e}"
409                ))
410            })?;
411
412        // The static row precedes the clustering rows and must be merged into each
413        // emitted clustering row, so we may only fast-forward PAST it when the table
414        // has NO static columns; otherwise decode from the partition body start
415        // (`body_start_rel = 0`) so the static prefix is seen (the END bound still
416        // narrows). (`test_da.wide_table` has no static columns, so the start narrows
417        // too.)
418        let has_static = schema
419            .map(|s| s.columns.iter().any(|c| c.is_static))
420            .unwrap_or(false);
421
422        // IMPLICIT FIRST BLOCK (issue #1968): a `start` below the FIRST stored
423        // separator selects a block no walk can return, so `rows_floor_block`'s
424        // `None` IS the implicit-first signal and the decode must begin at rel 0 or
425        // the earliest clustering rows are dropped. In a Cassandra-written trie read
426        // from the CORRECT root (issue #3002) this is now unreachable: the first
427        // block's separator is `ByteComparable.EMPTY` (`RowIndexWriter.add`), stored
428        // as the ROOT node's own payload, and nothing sorts below the empty key — the
429        // floor walk returns that block 0 entry as a genuine STORED floor. The `None`
430        // branch is retained for a trie whose first separator is NOT empty (a
431        // CQLite-written row index, or any bound below a non-empty first separator).
432        let includes_implicit_first_block = floor_block.is_none();
433
434        // The start narrows to the floor block only when NEITHER a static row NOR the
435        // implicit first block forces decoding from rel 0.
436        let narrows_start = !has_static && !includes_implicit_first_block;
437
438        // CORRECTNESS GUARD (no-heuristics, never wrong results): FLAG_OPEN_MARKER on
439        // the floor block's `IndexInfo` means a range tombstone is OPEN at the START
440        // of the narrowed window — a deletion opened in an earlier block still shadows
441        // rows inside the slice. Skipping earlier blocks would drop that range-open
442        // marker and risk resurrecting a deleted row (the post-scan backstop only
443        // FILTERS rows, it cannot re-apply a missed range tombstone). Per Cassandra
444        // `RowIndexReader.IndexInfo.openDeletion`, the marker on the floor block fully
445        // captures any deletion open at the window start, so this is the exact, tight
446        // guard: when it fires (only relevant when the start actually narrows) fall
447        // back to a full-partition decode. When the decode already starts at rel 0
448        // (static / implicit-first) no earlier block is skipped, so no guard is
449        // needed. (`test_da.wide_table` has no range tombstones.)
450        if narrows_start
451            && floor_block
452                .as_ref()
453                .map(|b| b.open_marker.is_some())
454                .unwrap_or(false)
455        {
456            debug!(
457                "BTI clustering seek: floor row-index block carries an open range-tombstone \
458                 marker; decoding full partition to preserve range-deletion semantics (no \
459                 narrowing)"
460            );
461            return Ok(None);
462        }
463
464        // body_start_rel: the floor block's offset when the start narrows, else the
465        // partition body start (rel 0). The block `data_offset` is relative to the
466        // partition start (the same domain the parser sees for `window[within..]`).
467        let body_start_rel = match &floor_block {
468            Some(b) if narrows_start => b.data_offset as usize,
469            _ => 0,
470        };
471        // body_end_rel: the exclusive successor block start, or the partition end
472        // (`usize::MAX`, clamped by the caller against the authoritative partition
473        // end / data-section length) when the slice reaches the last block.
474        let body_end_rel = ceil_block
475            .map(|b| b.data_offset as usize)
476            .unwrap_or(usize::MAX);
477
478        Ok(Some(ClusteringRowWindow {
479            body_start_rel,
480            body_end_rel,
481        }))
482    }
483
484    /// BTI ("da") full scan: decompress the whole Data.db section and parse
485    /// every partition in token order (issue #660).
486    ///
487    /// BTI SSTables carry no Index.db/Summary.db, so a range/full scan cannot
488    /// use the index path. Instead we stitch the entire (chunk-compressed) data
489    /// section into one buffer and run [`parse_block_with_cell_metadata`], which
490    /// walks ALL partitions — the same per-partition decode the point-lookup
491    /// path uses, but without stopping at the first match.
492    ///
493    /// Returns entries with per-cell write metadata so the WRITETIME/TTL scan
494    /// (`scan_with_cell_metadata`) and the plain `scan` (which drops the metadata)
495    /// can share a single implementation. Results are filtered by the optional
496    /// `[start_key, end_key]` range and tombstone-suppressed, then sorted into
497    /// Murmur3 token order and truncated to `limit` — identical post-processing
498    /// to the V5CompressedLegacy stitched path.
499    ///
500    /// Uses its own per-scan
501    /// [`ScanCursor`](crate::storage::sstable::reader::source::ScanCursor), so it
502    /// runs in parallel with other scans on this reader without serialization
503    /// (issue #815).
504    ///
505    /// [`parse_block_with_cell_metadata`]: crate::storage::sstable::reader::parsing::V5CompressedLegacyParser::parse_block_with_cell_metadata
506    ///
507    /// `read_shadowing` (issue #1741): `true` for user-facing SELECT scans
508    /// (`scan`, `scan_with_cell_metadata`), `false` for the physical
509    /// `get_all_entries` (integrity verification / data-manager) which must count
510    /// every on-disk row.
511    ///
512    /// `now_secs` (issue #3058, threaded here by #3109): a caller-pinned read-time
513    /// TTL clock. `Some` pins the decoder's expiry instant to the ONE
514    /// reconciliation instant the caller already captured for the request instead
515    /// of the ambient sample the parser takes at construction; `None` keeps the
516    /// ambient sample. Only consulted when `read_shadowing`.
517    pub(super) async fn bti_scan_with_metadata(
518        &self,
519        start_key: Option<&RowKey>,
520        end_key: Option<&RowKey>,
521        limit: Option<usize>,
522        schema: Option<&crate::schema::TableSchema>,
523        read_shadowing: bool,
524        now_secs: Option<i64>,
525    ) -> Result<
526        Vec<(
527            RowKey,
528            ScanRow,
529            std::collections::HashMap<String, CellWriteMetadata>,
530        )>,
531    > {
532        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
533        self.bti_scan_with_metadata_cancellable(
534            start_key,
535            end_key,
536            limit,
537            schema,
538            read_shadowing,
539            now_secs,
540            &self.scan_cancel,
541        )
542        .await
543    }
544
545    /// [`Self::bti_scan_with_metadata`] with an explicit PER-CALL cancellation
546    /// token (issue #2346/#2264), mirroring the
547    /// [`stitch_all_chunks`](Self::stitch_all_chunks) /
548    /// [`stitch_all_chunks_cancellable`](Self::stitch_all_chunks_cancellable) pair.
549    ///
550    /// `sequential_scan` is the seam that needs it: it takes a caller-supplied
551    /// token (the compaction path's
552    /// [`iterate_all_partitions_cancellable`](SSTableReader::iterate_all_partitions_cancellable)
553    /// drives it with one that is NOT the reader's own field), and its BTI branch
554    /// must honour that token. Routing a `da` reader through the non-cancellable
555    /// wrapper would poll the WRONG flag — a cancelled walk would stitch and parse
556    /// the entire data section and return `Ok(every row)`, reporting success for a
557    /// scan the caller abandoned.
558    ///
559    /// The token is polled in BOTH phases of the walk: every 256 chunks inside the
560    /// stitch (the I/O phase) and every 256 entries of the post-parse filter loop —
561    /// the same two-phase cadence `sequential_scan`'s stitched branch uses.
562    #[allow(clippy::too_many_arguments)]
563    pub(super) async fn bti_scan_with_metadata_cancellable(
564        &self,
565        start_key: Option<&RowKey>,
566        end_key: Option<&RowKey>,
567        limit: Option<usize>,
568        schema: Option<&crate::schema::TableSchema>,
569        read_shadowing: bool,
570        now_secs: Option<i64>,
571        scan_cancel: &crate::storage::scan_cancel::ScanCancel,
572    ) -> Result<
573        Vec<(
574            RowKey,
575            ScanRow,
576            std::collections::HashMap<String, CellWriteMetadata>,
577        )>,
578    > {
579        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
580        scan_cancel.check()?;
581        let cursor = self.new_scan_cursor().await?;
582
583        // Decompress the entire data section. Precondition for stitch_all_chunks:
584        // cursor's file seeked to data-section start (fresh cursor is at chunk 0).
585        let header_size = self.calculate_header_size();
586        {
587            let mut file_guard = cursor.file.lock().await;
588            // A5 read-work counter (SEEK_CALLS; consumer E4): scan seek-to-data-start
589            // before stitching the section. No-op in release (design.md Decision 1/2).
590            crate::storage::sstable::read_work_counters::record_seek();
591            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
592        }
593        let whole = self
594            .stitch_all_chunks_cancellable(&cursor, scan_cancel)
595            .await?;
596
597        // Resolve schema via the four-tier strategy (provided > header > registry).
598        // V5CompressedLegacy partition decode requires a schema (cells lack names).
599        let effective_schema = self.get_table_schema(schema);
600        let parser = self.build_v5_parser(read_shadowing);
601        // Issue #3058/#3109: honour the caller's pinned reconciliation clock, so a
602        // request that already sampled ONE `now` expires TTL cells at exactly that
603        // instant on this decoder too (mirrors `parse_block_entries_at_now` and the
604        // windowed driver).
605        let parser = match now_secs {
606            Some(now) => parser.with_now_secs(now),
607            None => parser,
608        };
609        // Issue #3782 (roborev job 48): `whole` is the ENTIRE data section —
610        // every chunk, stitched from a fresh cursor seeked to the data-section
611        // start above — so no further bytes can arrive to finish a row and a row
612        // decode error here is DATA LOSS. Before this declaration the `da` full
613        // scan (this is the route `scan`/`get_all_entries`/the batched streaming
614        // surface all take) inherited the tolerant break and returned `Ok` with
615        // 120 of 468 rows on a fixture with ONE compressed byte flipped inside a
616        // clustering value.
617        let parsed = parser.parse_block_with_cell_metadata(
618            &whole,
619            BufferExtent::Complete,
620            effective_schema.as_ref(),
621            self,
622        )?;
623
624        let mut results = Vec::new();
625        // Work-probe (issue #2398, threaded to BTI by #3109): "changed partition key
626        // = one more partition BODY decoded", counted BEFORE any range/tombstone
627        // filter exactly like the sibling walks (`sequential_scan`, the stitched
628        // walk, `run_scan_stream_batched`'s block loop). Without it the BTI decode
629        // would report ZERO scan work — including on the batched streaming surface,
630        // whose block loop counted before #3109 routed BTI readers here.
631        let mut prev_partition_key: Option<RowKey> = None;
632        for (idx, (_entry_table_id, entry_key, entry_value, cell_meta)) in
633            parsed.into_iter().enumerate()
634        {
635            // Cooperative cancellation (issue #2346/#2264): the stitch poll above
636            // covers the I/O phase, but `parse_block_with_cell_metadata`
637            // materialises every entry in one shot — poll here at the same
638            // 256-entry cadence as `sequential_scan`'s stitched branch so a
639            // cancelled caller does not walk a huge already-parsed result set to
640            // completion and then report success.
641            //
642            // KNOWN GAP for a QUERY TIMEOUT (#1695, roborev round 12), recorded here
643            // so it need not be re-derived: `scan_cancel` is the READER's SHARED
644            // token. #2264 trips it on a Flight client disconnect, and #2361 trips it
645            // when an iterator adapter drops — but NOTHING trips it when a streaming
646            // query's consumer goes away, so a timed-out scan over a BTI table keeps
647            // materialising this whole `results` Vec. The sibling producers were fixed
648            // by consulting their own channel (`tx.is_closed()` / racing
649            // `tx.closed()`), which is not available here: this function RETURNS a Vec
650            // and holds no sender.
651            //
652            // WHAT IS ACTUALLY MISSING, verified rather than guessed (roborev raised
653            // this twice, so the next person should not have to re-derive it):
654            // `bti_scan_with_metadata_cancellable` already exists and already takes a
655            // `&ScanCancel`, so the plumbing is NOT the gap. The gap is that the
656            // caller has no token it can legitimately pass:
657            //
658            //  * A CLONE of the reader's token is wrong — clones share state, so
659            //    tripping it cancels OTHER queries scanning the same reader. That is
660            //    the trap the materializing merges avoided with a per-call token.
661            //  * A FRESH per-call token tripped by a watcher on the output sender's
662            //    `closed()` handles the timeout case but silently DROPS #2264's Flight
663            //    cancellation, which trips the reader token WITHOUT necessarily
664            //    dropping the receiver — so a client disconnect would go back to the
665            //    ~1–2 min transport backstop. Losing existing cancellation to add new
666            //    cancellation is not a fix.
667            //
668            // So this needs one small API decision — a token that represents "either
669            // the reader's OR this stream's closure" (a linked/derived `ScanCancel`) —
670            // and that composition is the same primitive the deferred `JoinedStream` /
671            // `KWayMerger::step()` work needs. It is deliberately not improvised here:
672            // three partial fixes in this area is how the gap got this scattered, and a
673            // fourth guessing at token composition would be the worst of them.
674            scan_cancel.checkpoint(idx).await?;
675            if prev_partition_key.as_ref() != Some(&entry_key) {
676                crate::storage::sstable::work_counters::add_stream_walk_partition_parsed();
677                prev_partition_key = Some(entry_key.clone());
678            }
679            if let Some(start) = start_key {
680                if &entry_key < start {
681                    continue;
682                }
683            }
684            if let Some(end) = end_key {
685                if &entry_key > end {
686                    continue;
687                }
688            }
689            if !self.filter_tombstone(&entry_value) {
690                continue;
691            }
692            results.push((entry_key, entry_value, cell_meta));
693        }
694
695        sort_by_token_order_with_meta(&mut results);
696        if let Some(lim) = limit {
697            results.truncate(lim);
698        }
699
700        tracing::debug!(
701            "SSTableReader::bti_scan_with_metadata - Returning {} results",
702            results.len()
703        );
704        Ok(results)
705    }
706
707    /// The BTI (`da`) dispatch for BOTH public streaming surfaces — the per-row
708    /// `run_scan_stream` and the batched `run_scan_stream_batched` — in ONE place
709    /// (issue #3109).
710    ///
711    /// # Why a shared helper and not a third copy
712    ///
713    /// Issue #1577 established the rule: a BTI reader MUST decode through the SAME
714    /// authoritative trie-walk decoder [`Self::bti_scan_with_metadata`] that
715    /// [`SSTableReader::scan`] uses, NEVER the block-by-block `read_next_block` +
716    /// `parse_block_entries*` route. For `da` that route lands in the
717    /// `V5UncompressedOA` STATE MACHINE, which takes neither `read_shadowing` nor
718    /// `now_secs` and therefore silently DROPS both (see the "KNOWN FAIL-OPEN SEAM"
719    /// note in `parsing/block_entries.rs`, issue #3108) — so a `da` table read that
720    /// way is UNSHADOWED: partition/range tombstones and TTL expiry are not applied,
721    /// and on a schema-required fixture it fails outright ("Blob fallback not allowed
722    /// for V5_0Bti"). #1577 fixed the per-row surface; the batched surface was added
723    /// without the dispatch and reproduced the identical defect (#3109). Both now
724    /// call THIS function, gating on the SAME `bti_partitions_db.is_some()` condition
725    /// `scan` uses, so the three surfaces can never again disagree about which
726    /// readers are BTI or about the posture their rows are decoded under.
727    ///
728    /// The rows are exactly `scan`'s BTI rows: `bti_scan_with_metadata` has already
729    /// applied the key-range and tombstone filters and token-ordered them, so this
730    /// forwards them AS-IS and the stream stays prefix-authoritative with `scan`
731    /// (what D1's LIMIT pushdown relies on). BTI decode fully materializes the
732    /// (index-less) reconciled table before streaming — mirrored by
733    /// `scan_stream_materializes` returning `true` for BTI.
734    ///
735    /// `out` selects the emission shape only: one send per ROW for
736    /// [`WindowedOut::PerRow`], one send per `BATCH_EMIT_ROWS`-capped BATCH for
737    /// [`WindowedOut::Batched`] — the same cap and the same "flattening the batches
738    /// yields the per-row stream" contract the windowed driver honours, so the two
739    /// surfaces stay row-for-row identical for BTI by construction. A closed
740    /// consumer channel ends the scan cleanly (`Ok(())`), matching both callers.
741    pub(super) async fn stream_bti_scan(
742        &self,
743        start_key: Option<&RowKey>,
744        end_key: Option<&RowKey>,
745        schema: Option<&crate::schema::TableSchema>,
746        now_secs: Option<i64>,
747        out: &WindowedOut,
748    ) -> Result<()> {
749        let _scan = self.begin_scan(); // #3853 scan-lifetime madvise seam
750
751        // #1695 (roborev raised this in rounds 12, 14 and 15): race the
752        // materialization against OUR consumer's departure.
753        //
754        // `bti_scan_with_metadata` materializes the WHOLE table into `entries` before
755        // the first send below, so the send-failure checks in the arms are reached
756        // only after all the work is done. Its internal `scan_cancel.checkpoint(idx)`
757        // polls the READER-WIDE token, which #2264 trips on a Flight disconnect and
758        // #2361 on adapter teardown — but nothing trips when a timed-out streaming
759        // query drops its iterator. So a timed-out query kept decoding and sorting a
760        // whole BTI table.
761        //
762        // Racing the scan FUTURE is what fixes it, and it needs no new cancellation
763        // plumbing: the checkpoints inside are await points, so dropping the future
764        // there abandons the scan within one checkpoint interval. Crucially the
765        // reader-wide token is still passed through untouched, so #2264 and #2361 keep
766        // working — nothing is traded away for this. (An earlier attempt added a
767        // "derive a per-stream token from the reader's" API for the same purpose; it
768        // was removed once the race proved to give the same semantics with no new
769        // surface. Do not reintroduce it without a caller the race cannot serve.)
770        //
771        // `biased` so an already-departed consumer wins over a ready scan.
772        //
773        // RESIDUAL, deliberately: `sort_by_token_order_with_meta` after the loop is
774        // SYNCHRONOUS, so a departure during the sort is not observed until it ends.
775        // That is bounded by one sort rather than a whole table walk, and is the same
776        // uninterruptible-post-scan-stage limit documented at the chokepoint.
777        let scan = self.bti_scan_with_metadata(start_key, end_key, None, schema, true, now_secs);
778        let entries = match out {
779            WindowedOut::PerRow(tx) => tokio::select! {
780                biased;
781                _ = tx.closed() => return Ok(()),
782                result = scan => result?,
783            },
784            WindowedOut::Batched(tx) => tokio::select! {
785                biased;
786                _ = tx.closed() => return Ok(()),
787                result = scan => result?,
788            },
789        };
790
791        match out {
792            WindowedOut::PerRow(tx) => {
793                for (entry_key, entry_value, _meta) in entries {
794                    if tx.send(Ok((entry_key, entry_value))).await.is_err() {
795                        return Ok(()); // consumer dropped
796                    }
797                }
798            }
799            WindowedOut::Batched(tx) => {
800                let mut batch: Vec<(RowKey, ScanRow)> = Vec::with_capacity(BATCH_EMIT_ROWS);
801                for (entry_key, entry_value, _meta) in entries {
802                    batch.push((entry_key, entry_value));
803                    if batch.len() >= BATCH_EMIT_ROWS {
804                        if tx.send(Ok(std::mem::take(&mut batch))).await.is_err() {
805                            return Ok(()); // consumer dropped
806                        }
807                        batch.reserve(BATCH_EMIT_ROWS);
808                    }
809                }
810                if !batch.is_empty() && tx.send(Ok(batch)).await.is_err() {
811                    return Ok(()); // consumer dropped
812                }
813            }
814        }
815        Ok(())
816    }
