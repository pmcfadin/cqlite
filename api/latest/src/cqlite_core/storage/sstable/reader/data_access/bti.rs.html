33    /// Single-partition *seek* for the partition-targeted lookup fast path,
34    /// clustering-slice-aware (Issue #953 + #954, Epic #951).
35    ///
36    /// Where [`scan`](Self::scan) decodes EVERY partition in this SSTable and the
37    /// caller retains one, this resolves the target partition's `Data.db` offset
38    /// from the authoritative index (the BTI Partitions.db trie or the BIG
39    /// `Index.db`) and decodes ONLY that partition — the same per-partition decode
40    /// `scan` runs (`parse_block_emit` over the chunk-targeted decompressed
41    /// window), so its output is byte-for-byte identical to filtering the full
42    /// `scan` result down to `partition_key`.
43    ///
44    /// Offset domains (no-heuristics: authoritative resolved offsets only):
45    /// - **BTI ("da")** — `lookup_partition_via_bti_trie` returns the UNCOMPRESSED
46    ///   `Data.db` offset; a trie miss is authoritative absence.
47    /// - **BIG (`nb`)** — `lookup_partition_with_index` returns the partition's
48    ///   offset into the (uncompressed) data section. A hit is authoritative
49    ///   present; a MISS returns `Ok(None)` (the `Index.db` may be digest-keyed or
50    ///   incomplete, exactly the `get()` fallback rationale at #517) so the caller
51    ///   re-checks via a full scan rather than risk a false negative.
52    ///
53    /// Prefix-collision / wrong-offset guard: the decode
54    /// (`bti_decompress_and_parse_target_all`) re-verifies the decoded partition
55    /// key equals `partition_key` before collecting any row, so a BTI
56    /// prefix-collision candidate or a stale/mismatched index offset decodes to
57    /// nothing and is reported as absent — never a wrong partition. Every
58    /// clustering row of the matched partition is collected (not just the first),
59    /// so a multi-row partition returns all rows.
60    ///
61    /// Compiled only for the default (`not(tombstones)`) build: the manager's
62    /// seek-driven `scan_partition` exists only there, so under `tombstones` this
63    /// would be dead code.
64    ///
65    /// When `clustering` is `Some(slice)` AND this reader is BTI (`da`) with a
66    /// per-partition row index (`Rows.db`), the target partition's authoritative
67    /// row index is consulted to resolve the byte extent of the row-index
68    /// block(s) covering the requested clustering range, and ONLY that byte window
69    /// is decoded — so a `WHERE pk = ? AND ck </>/= ?` slice over a wide partition
70    /// decodes O(matched rows + index block slack) rather than the whole
71    /// partition. The post-scan `evaluate_leaf` backstop trims the
72    /// block-granularity over-read, so the returned rows are a superset of the
73    /// exact slice and the final query output is byte-identical to the
74    /// full-partition decode + post-filter.
75    ///
76    /// Returns `Ok(Some((rows, clustering_seek_engaged)))`:
77    /// - `clustering_seek_engaged == true` only when the clustering row-index
78    ///   narrowing actually bounded the decode (BTI wide partition with a usable
79    ///   row index and an encodable bound). The caller reports
80    ///   [`AccessPath::ClusteringSlice`](crate::query::access_path::AccessPath::ClusteringSlice)
81    ///   in that case.
82    /// - `clustering_seek_engaged == false` when the partition was decoded in full
83    ///   (no clustering slice, a NARROW BTI partition with no row index, the BIG
84    ///   format, or an un-encodable bound). Results are still correct — the caller
85    ///   reports the honest `PartitionLookup` path, NOT a fake clustering slice.
86    ///
87    /// `Ok(None)` mirrors [`scan_single_partition`]: the seek is not applicable
88    /// (no authoritative offset) and the caller must fall back to a full scan +
89    /// retain.
90    ///
91    /// [`scan_single_partition`]: crate::storage::sstable::SSTableReader
92    #[cfg(not(feature = "tombstones"))]
93    pub(crate) async fn scan_single_partition_clustering(
94        &self,
95        table_id: &TableId,
96        partition_key: &[u8],
97        clustering: Option<&ClusteringSlice>,
98        // `true` iff the manager resolved this reader by an EXACT fully-qualified
99        // `keyspace.table` match (or the query is unqualified). `false` means a
100        // fully-qualified query reached this reader via the bare-name fallback, in
101        // which case the seek guard keeps STRICT keyspace matching so it can never
102        // return rows from a different keyspace whose table name collides (#1284).
103        fully_qualified_match: bool,
104        schema: Option<&crate::schema::TableSchema>,
105    ) -> Result<Option<(Vec<(RowKey, ScanRow)>, bool)>> {
106        // 1. Resolve the partition's uncompressed Data.db offset, and record
107        //    whether THIS path's "decoded nothing" is authoritative absence (BTI
108        //    trie) or merely inconclusive (BIG Index.db).
109        let is_bti = self.bti_partitions_db.is_some();
110        // Resolve the target partition's UNCOMPRESSED `Data.db` start offset.
111        let offset = if is_bti {
112            match self.lookup_partition_via_bti_trie(partition_key)? {
113                // Trie hit: candidate uncompressed offset (re-verified on decode).
114                Some(off) => off,
115                // Trie miss is AUTHORITATIVE absence for BTI (no rows, no seek).
116                None => return Ok(Some((Vec::new(), false))),
117            }
118        } else {
119            match self.lookup_partition_with_index(partition_key).await? {
120                // Index.db hit: a candidate offset into the data section. (The
121                // `data_size` is 0 in writer-produced Index.db, so we do NOT use it
122                // as a bound — the successor offset below is the authoritative end.)
123                Some((off, _size)) => off,
124                // No Index.db hit: cannot seek authoritatively (the index may be
125                // digest-keyed / incomplete, exactly the get() #517 rationale).
126                // Fall back to a full scan.
127                None => return Ok(None),
128            }
129        };
130
131        // AUTHORITATIVE end bound (issue #953 / #951 MEDIUM): the target partition
132        // occupies `[offset, end)`, where `end` is the SUCCESSOR partition's start
133        // offset (next trie/index entry). Decompressing exactly the chunks covering
134        // that half-open range materializes every byte of the target partition —
135        // including a row/cell that SPANS multiple compression chunks — without
136        // reading the next partition. This replaces the previous next-partition
137        // *boundary-scan* heuristic (a row-count-stability guard that could falsely
138        // accept a boundary mid-partition); see `bti_decompress_and_parse_target_all`.
139        //
140        // `None` means `offset` is the LAST partition (no successor): the callee
141        // bounds the end with the authoritative data-section length, or falls back
142        // to the safe full-scan path when that length is unknown.
143        let end_bound = self.successor_partition_offset(offset)?.map(|e| e as usize);
144
145        let schema_opt = self.get_table_schema(schema);
146
147        // Issue #954 / #1184: resolve the within-partition row-body byte window for a
148        // single-column clustering slice from the authoritative index (BTI `Rows.db`
149        // trie or BIG promoted `IndexInfo` blocks). The unified resolver lives in
150        // `big_promoted.rs` (campsite: keeps this over-threshold file from growing).
151        let (row_body_window, decode_end_bound, clustering_engaged) = self
152            .resolve_clustering_seek_window(
153                is_bti,
154                partition_key,
155                offset,
156                clustering,
157                schema_opt.as_ref(),
158                end_bound,
159            )?;
160
161        // Issue #1184: an engaged BIG clustering narrowing decodes the selected block
162        // window via `big_promoted.rs` (partition-key-bytes guard, not the BTI strict
163        // table-id match that rejects writer-header SSTables). BTI keeps its decoder.
164        if !is_bti && clustering_engaged {
165            if let Some(rows) = self
166                .big_decode_clustering_window(
167                    partition_key,
168                    offset,
169                    decode_end_bound,
170                    row_body_window,
171                    schema_opt.as_ref(),
172                )
173                .await?
174            {
175                if !rows.is_empty() {
176                    super::super::super::work_counters::add_partition_decoded();
177                }
178                return Ok(Some((rows, true)));
179            }
180        }
181
182        // 2. Decode ONLY the target partition at the resolved offset, using the
183        //    SAME parser the scan path uses. `bti_decompress_and_parse_target_all`
184        //    chunk-targets the decompression (decodes just the chunk window that
185        //    holds the partition) and re-verifies the decoded key, so this is
186        //    O(1) PARTITIONS decoded regardless of the SSTable's partition count.
187        //
188        //    Issue #953 correctness fix: this collects EVERY clustering row of the
189        //    one target partition (bounded by the authoritative successor offset /
190        //    data-section length), not just the first row, so a `WHERE pk = ?`
191        //    over a multi-clustering-row
192        //    partition returns all rows — byte-identical to filtering the full
193        //    scan down to `partition_key`. The single-row `*_target` decoder is
194        //    still used by the `get()` point-lookup path, which returns one Value.
195        //
196        //    Issue #954: when `row_body_window` is set, the parse is bounded to the
197        //    clustering slice's row-index block extent so only O(slice) rows are
198        //    decoded (the post-scan backstop trims the block-granularity slack).
199        let parser = self.build_v5_parser(true);
200        let key = RowKey::from(partition_key.to_vec());
201        let decoded_rows = match self
202            .bti_decompress_and_parse_target_all(
203                offset as usize,
204                decode_end_bound,
205                row_body_window,
206                &key,
207                table_id,
208                fully_qualified_match,
209                schema_opt.as_ref(),
210                &parser,
211            )
212            .await?
213        {
214            // Authoritatively bounded decode (rows may be empty for an absent key).
215            Some(rows) => rows,
216            // The seek could not bound the target partition authoritatively (the
217            // LAST partition with an unknown data-section length): fall back to the
218            // safe full scan + retain for correctness, per the #953 mandate.
219            None => return Ok(None),
220        };
221
222        // 3. Record the per-partition decode (Issue #953 / #958): exactly ONE
223        //    partition is decoded for a hit regardless of how many clustering rows
224        //    it yields — `partitions_decoded` counts partitions, not rows. This is
225        //    the signal that proves the within-SSTable seek (vs a full
226        //    parse-then-retain). A non-empty decode means the partition exists.
227        if !decoded_rows.is_empty() {
228            super::super::super::work_counters::add_partition_decoded();
229            // Tombstone suppression matches the user-facing scan path
230            // (`sequential_scan`/`bti_scan_with_metadata` both apply it),
231            // applied per-row so a row tombstone is dropped while live rows in
232            // the same partition survive.
233            let rows: Vec<(RowKey, ScanRow)> = decoded_rows
234                .into_iter()
235                .filter(|value| self.filter_tombstone(value))
236                .map(|value| (key.clone(), value))
237                .collect();
238            return Ok(Some((rows, clustering_engaged)));
239        }
240
241        // Decoded nothing at the resolved offset. Whether that is authoritative
242        // depends on HOW the offset was resolved (Constraint #4: never return a
243        // wrong/empty result from an unsupported/inconclusive seek):
244        //
245        // - **BTI** — the trie is the authoritative present/absent oracle and the
246        //   decode re-verified the key, so "decoded nothing" means the trie
247        //   candidate was a prefix-collision for an absent key. AUTHORITATIVE
248        //   empty: the caller does NOT fall back.
249        // - **BIG** — the `Index.db` offset is only a candidate position; its
250        //   promoted-index / chunk layout is not as load-bearing as the BTI trie,
251        //   so a failed decode at the resolved offset is INCONCLUSIVE (a partition
252        //   that straddles a chunk boundary, or a stale offset, can fail the
253        //   chunk-targeted decode yet be found by a full parse). Fall back to a
254        //   full scan rather than risk a false negative.
255        if is_bti {
256            // Authoritative absence (prefix-collision candidate for an absent key).
257            // No rows decoded, so report the clustering seek as NOT engaged.
258            Ok(Some((Vec::new(), false)))
259        } else {
260            Ok(None)
261        }
262    }

264    /// Resolve the within-partition row-body byte window covering a single-column
265    /// clustering slice, using the target partition's authoritative BTI row index
266    /// (Issue #954, Epic #951).
267    ///
268    /// For a WIDE BTI partition the `Partitions.db` trie points at a per-partition
269    /// `TrieIndexEntry` in `Rows.db`; that entry's row-index trie maps clustering
270    /// **separators** to row-index BLOCK offsets (relative to the partition
271    /// start). [`select_row_index_blocks_for_range`] applies the authoritative
272    /// separator-floor semantics to pick exactly the blocks whose key interval
273    /// intersects `[start, end]`, so the returned byte window is the smallest
274    /// authoritative extent that can contain the requested clustering range.
275    ///
276    /// Returns `Ok(Some(window))` only when the narrowing is authoritative and
277    /// useful:
278    /// - the reader is BTI with a `Rows.db`,
279    /// - the partition is WIDE (`Partitions.db` returned a `RowsOffset`, i.e. the
280    ///   partition has a row index — a NARROW partition has no per-partition row
281    ///   index to seek within),
282    /// - the clustering bound(s) encode to the OSS50 byte-comparable form, and
283    /// - the selected block set is non-empty.
284    ///
285    /// Returns `Ok(None)` (decode the whole partition, report `PartitionLookup`)
286    /// for every other case — a NARROW partition, an empty `Rows.db`, an
287    /// un-encodable bound, or a slice that selects no block. This is the honest
288    /// fallback: correctness is preserved by decoding the full partition and
289    /// letting the post-scan backstop filter.
290    #[cfg(not(feature = "tombstones"))]
291    pub(super) fn bti_clustering_row_window(
292        &self,
293        partition_key: &[u8],
294        slice: &ClusteringSlice,
295        schema: Option<&crate::schema::TableSchema>,
296    ) -> Result<Option<ClusteringRowWindow>> {
297        use crate::storage::sstable::bti::{
298            iterate_rows_for_partition, lookup_raw_key_in_bti_partitions_db, resolve_rows_db_entry,
299            select_row_index_blocks_for_range, BtiPartitionLocation,
300        };
301
302        let (Some(partitions_db), Some(rows_db)) = (&self.bti_partitions_db, &self.bti_rows_db)
303        else {
304            return Ok(None);
305        };
306
307        // Resolve the partition's location. Only a WIDE partition (RowsOffset) has
308        // a per-partition row index we can seek within; a NARROW partition
309        // (DataOffset) has none, so decode it in full.
310        let mut cursor = std::io::Cursor::new(partitions_db.as_slice());
311        let rows_offset = match lookup_raw_key_in_bti_partitions_db(&mut cursor, partition_key)
312            .map_err(|e| {
313                Error::corruption(format!(
314                    "BTI clustering seek: Partitions.db trie lookup failed (key len={}): {}",
315                    partition_key.len(),
316                    e
317                ))
318            })? {
319            Some(BtiPartitionLocation::RowsOffset(off)) => off as usize,
320            // NARROW partition or absent key: no row index to narrow with.
321            Some(BtiPartitionLocation::DataOffset(_)) | None => return Ok(None),
322        };
323
324        // Resolve the per-partition row-index entry and enumerate its blocks in
325        // ascending byte-comparable (clustering) order.
326        let header = resolve_rows_db_entry(rows_db.as_slice(), rows_offset).map_err(|e| {
327            Error::corruption(format!(
328                "BTI clustering seek: Rows.db entry at RowsOffset({rows_offset}) unreadable: {e}"
329            ))
330        })?;
331        let (_header2, entries) = iterate_rows_for_partition(rows_db.as_slice(), rows_offset)
332            .map_err(|e| {
333                Error::corruption(format!(
334                    "BTI clustering seek: Rows.db trie at RowsOffset({rows_offset}) unreadable: {e}"
335                ))
336            })?;
337        // `iterate_rows_for_partition` re-resolves the header internally; keep the
338        // first `header` (identical) for `block_count`/`data_position`.
339        let _ = header;
340        if entries.is_empty() {
341            return Ok(None);
342        }
343
344        // Per-column reverse order for the FIRST clustering column (single-column
345        // scope per #954). A missing/absent schema treats it as ascending.
346        let is_reversed: Vec<bool> = schema
347            .map(|s| {
348                s.clustering_keys
349                    .iter()
350                    .map(|c| matches!(c.order, crate::schema::ClusteringOrder::Desc))
351                    .collect()
352            })
353            .unwrap_or_default();
354
355        // Encode the CQL bounds into the PHYSICAL byte-comparable order the row
356        // index uses, normalizing for a DESC first clustering column (issue #954
357        // High-severity correctness fix). `select_row_index_blocks_for_range`
358        // operates purely in physical (on-disk, byte-comparable) order, so the CQL
359        // lower/upper bounds must be mapped to the physical-lower/physical-upper
360        // sides before block selection. For a DESC column those roles SWAP (see
361        // `physical_byte_bounds_for_slice`). An un-encodable bound makes the
362        // narrowing unsafe → decode the whole partition (honest fallback).
363        let Some((start_bytes, end_bytes)) = physical_byte_bounds_for_slice(slice, &is_reversed)?
364        else {
365            return Ok(None);
366        };
367
368        // CORRECTNESS GUARD (no-heuristics, never wrong results): a row-index block
369        // carrying an `open_marker` (FLAG_OPEN_MARKER) means a range tombstone is
370        // OPEN at that block boundary — a deletion opened in an earlier block can
371        // still shadow rows inside the requested slice. Narrowing the decode skips
372        // the rows (and the range-marker bytes) before the slice, which would drop
373        // that open deletion and risk resurrecting a deleted row. The post-scan
374        // backstop only FILTERS rows, it cannot re-apply a missed range tombstone.
375        // So when ANY block in this partition's row index carries an open marker we
376        // fall back to a full-partition decode (correct, just unnarrowed). The
377        // common wide-partition slice (no range tombstones) is unaffected.
378        if entries.iter().any(|(_sep, b)| b.open_marker.is_some()) {
379            debug!(
380                "BTI clustering seek: partition row index has open range-tombstone marker(s); \
381                 decoding full partition to preserve range-deletion semantics (no narrowing)"
382            );
383            return Ok(None);
384        }
385
386        let blocks = select_row_index_blocks_for_range(&entries, &start_bytes, &end_bytes);
387
388        // IMPLICIT FIRST BLOCK (issue #1968). A `Rows.db` row-index trie stores a
389        // separator per block EXCEPT the first: the block covering keys BELOW the
390        // first separator lives at the partition body start and has NO trie entry
391        // (mirroring Cassandra `RowIndexReader.separatorFloor`, which returns the
392        // partition start for a key below the first separator). Consequently
393        // `select_row_index_blocks_for_range` — which operates purely over the
394        // stored (separator, block) entries — can NEVER return that implicit block.
395        //
396        // The requested range overlaps the implicit first block iff its
397        // physical-lower bound sorts strictly BELOW the first separator
398        // (`start_bytes < entries[0].sep`). The canonical trigger is an OPEN lower
399        // bound (`ck < N` / `ck <= N`), whose physical-lower sentinel is `-∞` = the
400        // empty slice `b""`, but a closed lower bound below the first separator
401        // (`ck >= 2 AND ck < 20`, `ck = 0`) hits it too. When the range includes
402        // that block the earliest clustering rows precede every stored block, so the
403        // decode MUST begin at the partition body start (rel 0) or those rows are
404        // silently dropped (the pre-fix bug returned `ck=8..19` for `ck < 20`).
405        // Bind the first stored entry ONCE and reuse it for both the predicate and
406        // the entirely-within-implicit-block window below, so the window end can
407        // never diverge from the predicate (no `usize::MAX` over-read fallback).
408        let first_entry = entries.first();
409        let includes_implicit_first_block = first_entry
410            .map(|(sep, _)| start_bytes.as_slice() < sep.as_slice())
411            .unwrap_or(false);
412
413        if blocks.is_empty() {
414            // `includes_implicit_first_block` is true only when `first_entry` is
415            // `Some`; the `if let` makes that invariant explicit and reuses the exact
416            // entry that satisfied the predicate (its start is the window end).
417            if let (true, Some((_sep, first_block))) = (includes_implicit_first_block, first_entry)
418            {
419                // The range lies ENTIRELY within the implicit first block (its
420                // physical-upper is also below the first separator, e.g. `ck <= 3`
421                // or `ck = 0`): no stored block overlaps, but the implicit block
422                // does. Narrow to [partition body start, first stored block start);
423                // the post-scan backstop trims to the exact predicate.
424                return Ok(Some(ClusteringRowWindow {
425                    body_start_rel: 0,
426                    body_end_rel: first_block.data_offset as usize,
427                }));
428            }
429            // No block (implicit or stored) overlaps the range. The slice may still
430            // select rows that share the floor block's separator boundary; to stay
431            // correct we fall back to a full-partition decode rather than risk
432            // dropping a row.
433            return Ok(None);
434        }
435
436        // Row-body byte window = [first selected block start, end of the LAST
437        // selected block). The block `data_offset` is relative to the partition
438        // start (the same domain the parser sees for `window[within..]`). The end
439        // is the start of the FIRST block AFTER the last selected one (or +∞ via
440        // the partition end when the last selected block is the partition's last).
441        // The static row precedes the clustering rows and must be merged into each
442        // emitted clustering row, so we may only fast-forward PAST it when the
443        // table has NO static columns. With a static column present, decode from
444        // the partition body start (`body_start_rel = 0`) so the static prefix is
445        // seen; the END bound still narrows the decode. (The acceptance fixture
446        // `test_da.wide_table` has no static columns, so the start narrows too.)
447        let has_static = schema
448            .map(|s| s.columns.iter().any(|c| c.is_static))
449            .unwrap_or(false);
450        // Start at the earliest selected STORED block — UNLESS the table has a
451        // static row, or the range also covers the implicit first block (issue
452        // #1968); either case must decode from the partition body start (rel 0) so
453        // the static prefix / the implicit block's rows are seen. The END bound
454        // still narrows the decode in both cases.
455        let body_start_rel = if has_static || includes_implicit_first_block {
456            0
457        } else {
458            blocks
459                .iter()
460                .map(|b| b.data_offset as usize)
461                .min()
462                .unwrap_or(0)
463        };
464        let last_selected_off = blocks.iter().map(|b| b.data_offset).max().unwrap_or(0);
465        // The exclusive end is the next block's start strictly greater than the
466        // last selected block; if none, the window runs to the partition end
467        // (`usize::MAX` is clamped by the caller against the authoritative
468        // partition end / data-section length).
469        let body_end_rel = entries
470            .iter()
471            .map(|(_sep, b)| b.data_offset)
472            .filter(|&off| off > last_selected_off)
473            .min()
474            .map(|off| off as usize)
475            .unwrap_or(usize::MAX);
476
477        Ok(Some(ClusteringRowWindow {
478            body_start_rel,
479            body_end_rel,
480        }))
481    }

483    /// BTI ("da") point lookup: resolve a partition key via the Partitions.db
484    /// trie, decode the partition at the resolved offset, and return its row
485    /// `Value` (issue #831).
486    ///
487    /// Correctness invariants (see issue #831 / #755):
488    ///
489    /// - **Offset domain**: the trie returns an *uncompressed* Data.db offset, so
490    ///   we decode the partition out of the DECOMPRESSED data section, never via
491    ///   `read_value_at_offset`/`get_cached_data` (which seek raw file bytes).
492    /// - **Own decompression**: `requires_chunk_stitching()` is `false` for BTI,
493    ///   so this path decompresses the chunk-compressed Data.db itself via the
494    ///   reader's CompressionInfo + compression_reader. Because the trie already
495    ///   resolved the EXACT uncompressed offset of the target partition, this only
496    ///   decompresses the chunk that contains that offset and continues forward
497    ///   chunk-by-chunk ONLY until the target partition is fully parsed — it never
498    ///   decompresses earlier chunks or the rest of the file (issue #831 perf
499    ///   finding). The whole-section `point_read_whole_section` fallback (one
500    ///   positioned read of the entire data section) is used only when chunk
501    ///   targeting is impossible (no/zero `chunk_length`).
502    /// - **Prefix-collision guard**: the trie may return a candidate for a
503    ///   prefix-colliding key, so the decoded partition key is verified to equal
504    ///   the queried key before any row is returned.
505    ///
506    /// `fully_qualified_match` is the authoritative resolution-mode signal threaded
507    /// from the manager's `resolve_reader_list` (issue #1321, mirroring #1284's
508    /// seek path): `true` iff the query's fully-qualified `keyspace.table` key
509    /// matched this reader's map slot EXACTLY (or the query was unqualified),
510    /// `false` iff a fully-qualified query reached this reader via the bare-name
511    /// fallback. It gates the per-row table-consistency guard in
512    /// `bti_decompress_and_parse_target`: an exact FQ match may relax across a
513    /// header-keyspace divergence, while a fallback keeps strict keyspace matching.
514    pub(super) async fn bti_point_lookup(
515        &self,
516        table_id: &TableId,
517        key: &RowKey,
518        fully_qualified_match: bool,
519    ) -> Result<Option<ScanRow>> {
520        // 1. Resolve the uncompressed Data.db offset via the trie.
521        let offset = match self.lookup_partition_via_bti_trie(key.as_bytes())? {
522            Some(off) => off as usize,
523            None => return Ok(None), // not in this SSTable
524        };
525
526        // 2. Obtain a DECOMPRESSED window that contains the target partition.
527        //
528        //    `window_base` is the uncompressed offset of the window's first byte
529        //    and `window` holds the decompressed bytes from there onward. The
530        //    target partition starts at `offset - window_base` inside `window`
531        //    (INVARIANT 1: the trie offset indexes the uncompressed data section).
532        //
533        //    For the chunk-targeted path the window starts at the chunk that
534        //    contains `offset` (so `window_base = target_chunk * chunk_length`);
535        //    for the whole-section fallback the window starts at offset 0
536        //    (`window_base = 0`). Either way the parse below uses the same
537        //    `within = offset - window_base` index.
538        let schema_opt = self.get_table_schema(None);
539        let parser = self.build_v5_parser(true);
540
541        let found = self
542            .bti_decompress_and_parse_target(
543                offset,
544                key,
545                table_id,
546                fully_qualified_match,
547                schema_opt.as_ref(),
548                &parser,
549            )
550            .await?;
551
552        match found {
553            Some(value) => {
554                if !self.filter_tombstone(&value) {
555                    return Ok(None);
556                }
557                Ok(Some(value))
558            }
559            None => Ok(None),
560        }
561    }
562
563    /// Compute the chunk that contains uncompressed `offset`, the uncompressed
564    /// offset of that chunk's start, and the within-chunk index — given the
565    /// CompressionInfo `chunk_length` (issue #831).
566    ///
567    /// Returns `(target_chunk, window_base, within)` where
568    /// `window_base = target_chunk * chunk_length` and `within = offset - window_base`.
569    /// Pure arithmetic so it can be unit-tested independently of any I/O.
570    #[inline]
571    fn bti_chunk_target(offset: usize, chunk_length: usize) -> (usize, usize, usize) {
572        let target_chunk = offset / chunk_length;
573        let window_base = target_chunk * chunk_length;
574        let within = offset - window_base;
575        (target_chunk, window_base, within)
576    }

578    /// Positional (`pread`) fetch of compressed chunk `chunk_idx` for the
579    /// POINT-READ path (issue #1573, C2). Reads via the shared `point_source` — no
580    /// per-lookup `open(2)`, no cursor, no mutex — CRC-checks the chunk, and
581    /// returns the compressed bytes (the caller decompresses). `Ok(None)` at EOF.
582    /// Only called on the chunk-targeted path, where `CompressionInfo` is present.
583    pub(super) fn point_read_compressed_chunk(&self, chunk_idx: usize) -> Result<Option<Vec<u8>>> {
584        let Some(ci) = self.compression_info.as_deref() else {
585            return Ok(None);
586        };
587        // header_offset is ALWAYS 0 for NB/BTI: CompressionInfo chunk offsets are
588        // absolute from Data.db byte 0 (any embedded header is part of the
589        // compressed data), exactly as the cursor path hardcodes in
590        // `read_next_block_impl`. Passing `actual_header_size` here would shift
591        // every chunk read and fail CRC.
592        super::super::block_io::read_compressed_chunk_at(
593            self.point_source.as_ref(),
594            ci,
595            chunk_idx,
596            self.stats.file_size,
597            0,
598        )
599    }

601    /// Positional (`pread`) read of the ENTIRE uncompressed data section for the
602    /// point-read whole-section fallback (issue #1573, C2) — used when chunk
603    /// targeting is impossible (no/zero `chunk_length`, i.e. an uncompressed BTI or
604    /// nb-without-CompressionInfo Data.db). Reads `[header_size, file_size)` in one
605    /// positioned read and, when a `CRC.db` is present (uncompressed BIG), verifies
606    /// the covering chunks BEFORE the bytes are parsed — preserving the CRC-then-use
607    /// ordering the cursor path enforced via `read_uncompressed_data_block`.
608    pub(super) async fn point_read_whole_section(&self) -> Result<Vec<u8>> {
609        let header_size = self.calculate_header_size() as u64;
610        // Authoritative file length straight from the positional source (== the
611        // reader's `file_size`; using the source keeps the read self-consistent).
612        let end = self.point_source.len();
613        let len = end.saturating_sub(header_size);
614        let mut whole = vec![0u8; len as usize];
615        if len > 0 {
616            // Read the section in BOUNDED windows rather than one section-sized
617            // `read_exact_at`. A `DirectReadAt` backend allocates a per-call
618            // aligned bounce buffer as large as the requested range, so a single
619            // whole-section read would transiently ~double resident memory vs the
620            // <128MB target for a large section. Windowing caps the bounce buffer
621            // at ~`WHOLE_SECTION_READ_WINDOW` regardless of backend (issue #1573
622            // roborev); `whole` itself is the returned data and is unavoidable.
623            const WHOLE_SECTION_READ_WINDOW: usize = 1 << 20; // 1 MiB
624            let mut filled = 0usize;
625            while filled < whole.len() {
626                let win_end = (filled + WHOLE_SECTION_READ_WINDOW).min(whole.len());
627                self.point_source
628                    .read_exact_at(header_size + filled as u64, &mut whole[filled..win_end])?;
629                filled = win_end;
630            }
631        }
632        // CRC-verify the covering chunk(s) when a CRC.db is present (no-op for BTI
633        // and compressed tables) BEFORE returning the bytes. The section is already
634        // resident in `whole`, so verify against those in-memory bytes rather than
635        // re-reading the identical range from `point_source` — the section is
636        // transferred from disk EXACTLY ONCE (issue #1573 roborev), preserving the
637        // CRC-before-use ordering and the CRC algorithm unchanged.
638        if self.crc_reader.is_some() {
639            self.verify_uncompressed_section_in_buffer(header_size, &whole)
640                .await?;
641        }
642        Ok(whole)
643    }

645    /// Decompress only the chunk(s) needed to fully parse the target partition at
646    /// uncompressed `offset`, then parse and return its row value (issue #831).
647    ///
648    /// Chunk targeting (the fast path): when `CompressionInfo` with a non-zero
649    /// `chunk_length` is present, the chunk containing `offset` is
650    /// `target_chunk = offset / chunk_length`; we seek that chunk via its
651    /// `chunk_offsets` entry, set the cursor's chunk index to `target_chunk`, then
652    /// decompress forward chunk-by-chunk, appending each into `window`. After each
653    /// appended chunk we attempt to parse the FIRST partition at `window[within..]`
654    /// (`within = offset % chunk_length`). The stop condition (correctness-critical
655    /// — never return a truncated parse):
656    ///   - parse returns `Ok` AND the emit closure fired (a COMPLETE partition was
657    ///     decoded) -> stop and return what the closure captured;
658    ///   - parse returns `Err` (buffer truncated mid-partition) OR the closure
659    ///     never fired -> append the next chunk and retry;
660    ///   - `read_next_block()` returns `None` (EOF) and still not parsed -> stop
661    ///     (the caller treats `None` as "absent", matching prior behaviour).
662    ///
663    /// Fallbacks (preserve prior behaviour exactly): when `compression_info` is
664    /// `None` (uncompressed BTI Data.db) or `chunk_length` is 0/absent, this
665    /// reads the WHOLE section in one positioned read via
666    /// `point_read_whole_section` (`window_base = 0`, CRC-verified when a CRC.db
667    /// is present) and runs the same single-partition parse.
668    ///
669    /// Chunks are fetched with positioned (`read_at`) reads on the shared
670    /// `point_source` — no per-lookup `open(2)`, no `ScanCursor`, and no mutex.
671    /// `chunk_index` is a plain local (a lookup is single-threaded within
672    /// itself), so concurrent lookups run in parallel without serialization;
673    /// safety comes from `read_at` taking `&self` (issue #1573, superseding the
674    /// per-scan-cursor approach of issue #815).
675    pub(super) async fn bti_decompress_and_parse_target(
676        &self,
677        offset: usize,
678        key: &RowKey,
679        table_id: &TableId,
680        // Issue #1321: authoritative resolution mode (see `bti_point_lookup`).
681        // Gates the per-row table-consistency guard exactly like the seek path
682        // (#1284): an EXACT fully-qualified resolution may accept rows across a
683        // benign header-keyspace divergence on a consistent table name, while a
684        // fully-qualified query resolved via the bare-name fallback keeps STRICT
685        // keyspace matching (no wrong-keyspace rows).
686        fully_qualified_match: bool,
687        schema_opt: Option<&crate::schema::TableSchema>,
688        parser: &crate::storage::sstable::reader::parsing::V5CompressedLegacyParser,
689    ) -> Result<Option<ScanRow>> {
690        use crate::storage::sstable::compression::Compression;
691
692        // Issue #1573 (C2): the point path fetches chunks via positioned reads on
693        // the shared `point_source` — no per-lookup `open(2)`, no cursor, no mutex.
694        // `chunk_index` is a plain local (a lookup is single-threaded within
695        // itself); concurrency safety comes from `read_at` being `&self`.
696
697        // Determine the chunk-targeting parameters. `chunk_length == 0` (or no
698        // CompressionInfo) means we cannot chunk-target -> whole-section fallback.
699        let chunk_length = self
700            .compression_info
701            .as_ref()
702            .map(|ci| ci.chunk_length as usize)
703            .filter(|&len| len > 0);
704
705        let mut chunk_index = 0usize;
706        let (target_chunk, window_base, mut window) = match chunk_length {
707            Some(len) => {
708                let (target_chunk, window_base, _within) = Self::bti_chunk_target(offset, len);
709                // Positioned reads resolve their own offset from the chunk index, so
710                // no pre-seek is needed — just start at `target_chunk`.
711                chunk_index = target_chunk;
712                (target_chunk, window_base, Vec::<u8>::new())
713            }
714            None => {
715                // Whole-section fallback (uncompressed BTI, or chunk_length absent/0):
716                // one positioned read of the whole data section, CRC-verified when a
717                // CRC.db is present (see `point_read_whole_section`).
718                let whole = self.point_read_whole_section().await?;
719                (0usize, 0usize, whole)
720            }
721        };
722
723        // `within` is the start of the target partition inside `window`.
724        if offset < window_base {
725            return Err(Error::corruption(format!(
726                "BTI point lookup: resolved offset {} precedes window base {} (chunk {})",
727                offset, window_base, target_chunk
728            )));
729        }
730        let within = offset - window_base;
731
732        // For the chunk-targeted path we still need to populate `window`. For the
733        // whole-section fallback `window` is already complete.
734        let chunk_targeted = chunk_length.is_some();
735
736        loop {
737            // If chunk-targeted, append the next chunk before each parse attempt
738            // (the whole-section fallback already has all bytes in `window`).
739            if chunk_targeted {
740                // Cache key: the ABSOLUTE chunk index about to be read (issue #1567).
741                // Shared cache is consulted before decompress; a hit skips the
742                // decompressor (the compressed bytes were still read + CRC-checked).
743                let this_chunk = chunk_index as u64;
744                match self.point_read_compressed_chunk(chunk_index)? {
745                    Some(compressed_chunk) => {
746                        chunk_index += 1;
747                        let key = self.chunk_cache_key(super::NS_BTI_CHUNK, this_chunk);
748                        let decompressed_chunk: std::sync::Arc<[u8]> = if let Some(hit) =
749                            self.chunk_cache.get(&key)
750                        {
751                            hit
752                        } else if let Some(compression_reader) = &self.compression_reader {
753                            let compression = Compression::new(*compression_reader.algorithm())?;
754                            let d = compression.decompress(&compressed_chunk).map_err(|e| {
755                                Error::corruption(format!(
756                                    "BTI point lookup: failed to decompress chunk: {}",
757                                    e
758                                ))
759                            })?;
760                            super::model::DECOMPRESS_CALLS
761                                .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
762                            self.chunk_cache.insert(key, d)
763                        } else {
764                            // No compression reader despite CompressionInfo:
765                            // treat raw chunk bytes as the decompressed data.
766                            self.chunk_cache.insert(key, compressed_chunk)
767                        };
768                        window.extend_from_slice(&decompressed_chunk);
769                    }
770                    None => {
771                        // EOF: no more chunks. If we never parsed a complete
772                        // partition, the partition is treated as absent (matching
773                        // the prior whole-section behaviour for an unparseable tail).
774                        return Ok(None);
775                    }
776                }
777            }
778
779            // Need at least the partition header to attempt a match.
780            if within >= window.len() {
781                if chunk_targeted {
782                    // Not enough bytes yet; pull the next chunk.
783                    continue;
784                }
785                // Whole-section window can't grow: offset is past the data.
786                return Err(Error::corruption(format!(
787                    "BTI trie resolved Data.db offset {} beyond decompressed data section ({} bytes)",
788                    offset,
789                    window.len()
790                )));
791            }
792
793            // INVARIANT 3 + chunk-straddle gate. The parse/pull/absent decision is
794            // factored into the pure `bti_lookup_step` so the chunk-straddle control
795            // flow is unit-testable without a multi-chunk fixture (issue #831 review):
796            // when the header/key prefix is not yet fully buffered we must NOT invoke
797            // the parser on a truncated header (it can skip bytes and emit a later
798            // false-positive entry), and must read the next chunk first.
799            let key_available =
800                Self::bti_partition_key_bytes_available(&window, within, key.as_bytes());
801            let key_matches =
802                key_available && self.bti_partition_key_matches(&window, within, key.as_bytes());
803            match bti_lookup_step(key_available, key_matches, chunk_targeted) {
804                BtiLookupStep::Parse => { /* full key prefix buffered and matches */ }
805                BtiLookupStep::PullNextChunk => continue,
806                BtiLookupStep::Absent => {
807                    if key_available {
808                        debug!(
809                            "BTI trie candidate at offset {} did not match queried key \
810                             (prefix collision); treating as absent",
811                            offset
812                        );
813                    }
814                    return Ok(None);
815                }
816            }
817
818            // Attempt to parse the FIRST partition at window[within..]. The parser
819            // detects the next partition boundary / 0x01 end-of-partition marker and
820            // stops; we break after the first emitted entry. A complete partition
821            // means: parse returned Ok AND the closure fired.
822            let mut found: Option<ScanRow> = None;
823            let mut emitted = false;
824            let parse_result = parser.parse_block_emit(
825                &window[within..],
826                schema_opt,
827                self,
828                |(tid, entry_key, entry_value)| {
829                    emitted = true;
830                    // Verify BOTH the emitted table id is consistent with the
831                    // queried table AND the parser-decoded partition key equals the
832                    // queried key. The table check is resolution-mode-aware (issue
833                    // #1321, mirroring the seek path #1284): an EXACT fully-qualified
834                    // resolution accepts a consistent table name across a benign
835                    // header-keyspace divergence; a fully-qualified query resolved via
836                    // the bare-name fallback keeps STRICT keyspace matching so it can
837                    // never return another keyspace's same-named rows. A genuinely
838                    // different table name is always rejected (issue #831).
839                    if table_header_consistent_for_seek(&tid, table_id, fully_qualified_match)
840                        && entry_key.as_bytes() == key.as_bytes()
841                    {
842                        found = Some(entry_value);
843                    }
844                    Ok(std::ops::ControlFlow::Break(()))
845                },
846            );
847
848            match parse_result {
849                Ok(()) if emitted => {
850                    // A COMPLETE partition was decoded — accept it and stop.
851                    return Ok(found);
852                }
853                _ => {
854                    // Either Err (truncated mid-partition) or the closure never
855                    // fired (no complete partition yet). For the chunk-targeted
856                    // path, pull the next chunk and retry; never accept a partial.
857                    if chunk_targeted {
858                        continue;
859                    }
860                    // Whole-section fallback already has every byte: a failure here
861                    // means the partition genuinely could not be parsed -> absent.
862                    return Ok(None);
863                }
864            }
865        }
866    }
867
868    /// Collect-ALL-rows variant of [`bti_decompress_and_parse_target`] for the
869    /// within-SSTable seek (`scan_single_partition`, Issue #953 / #951).
870    ///
871    /// [`bti_decompress_and_parse_target`] stops after the FIRST emitted row of the
872    /// decoded partition — correct for a `get()` point lookup that returns a single
873    /// `Value`, but WRONG for `scan_partition`, which must hand the query layer
874    /// EVERY clustering row of the partition so it can apply clustering predicates.
875    /// A `WHERE pk = ?` over a table with multiple clustering rows per partition
876    /// would otherwise drop every row after the first whenever the seek succeeds
877    /// (the original #953 bug — see the multi-row regression test).
878    ///
879    /// This variant reuses the identical window-building (chunk targeting or
880    /// whole-section fallback), the identical prefix-collision key re-verification,
881    /// and the identical `parse_block_emit` decode that the user-facing scan path
882    /// runs — but instead of breaking after the first row it COLLECTS every row the
883    /// parser emits for the ONE target partition. The emit closure keeps each
884    /// `Value` whose decoded key equals the queried key (and whose table id
885    /// matches) and `Break`s the instant the parser emits a row with a DIFFERENT
886    /// partition key.
887    ///
888    /// Bounding the decompression window (Issue #953 / #951 MEDIUM fix). The seek
889    /// must materialize ONLY the chunks covering the target partition — never
890    /// stitch to EOF (for a head-of-file point lookup on a large SSTable that would
891    /// decompress nearly the whole `Data.db`, full-table I/O for one partition).
892    /// The bound is AUTHORITATIVE, not a heuristic boundary scan:
893    ///
894    ///   - **`end_bound = Some(end)`** — the caller resolved the SUCCESSOR
895    ///     partition's uncompressed start offset (next trie/index entry). The
896    ///     target partition occupies `[offset, end)`, so we pull chunks only until
897    ///     `window.len() >= end - window_base` (or EOF) and then parse ONCE over a
898    ///     window that fully contains the partition. Because the WHOLE `[offset,
899    ///     end)` extent is decompressed before parsing, a row/cell that spans
900    ///     multiple compression chunks is present in full — no mid-stream
901    ///     truncation, no boundary guessing. This is the exact bound for every
902    ///     non-last partition in both BTI (`da`) and BIG (`nb`).
903    ///
904    ///   - **`end_bound = None`** — `offset` is the LAST partition (no successor).
905    ///     The end is then the authoritative data-section length
906    ///     (`CompressionInfo.data_length`); we buffer to that length (or EOF) and
907    ///     parse once. If that length is unavailable (no usable `CompressionInfo`),
908    ///     we CANNOT bound the last partition authoritatively, so we return
909    ///     `Ok(None)` and the caller falls back to the safe full-scan + retain path
910    ///     (correctness over optimization). The previous row-count *stability
911    ///     guard* — itself a heuristic that could falsely accept a next-partition
912    ///     boundary while the target partition was incomplete (a single large
913    ///     multi-chunk cell, static/range-marker regions, or a truncated tail
914    ///     parsed as garbage headers) — has been REMOVED entirely.
915    ///
916    /// The whole-section fallback (uncompressed BTI) already has every byte so its
917    /// first parse is authoritative regardless of the bound. This yields
918    /// byte-for-byte the same rows as the full-scan path filtered down to
919    /// `partition_key`.
920    ///
921    /// Returns:
922    /// - `Ok(Some(rows))` — the partition's rows (empty when the trie/index
923    ///   candidate was a prefix collision for an absent key). The caller wraps each
924    ///   in a `(RowKey, ScanRow)` and applies the same tombstone suppression the scan
925    ///   path applies.
926    /// - `Ok(None)` — could not bound the (last) partition authoritatively; the
927    ///   caller must fall back to a full scan + retain.
928    #[cfg(not(feature = "tombstones"))]
929    async fn bti_decompress_and_parse_target_all(
930        &self,
931        offset: usize,
932        end_bound: Option<usize>,
933        // Issue #954: when `Some((start_rel, end_rel))`, bound the partition's
934        // row-body parse to that within-partition byte window (relative to the
935        // partition start) so only the clustering slice's row-index block(s) are
936        // decoded. `None` decodes the whole partition (the #953 behaviour).
937        row_body_window: Option<(usize, usize)>,
938        key: &RowKey,
939        table_id: &TableId,
940        // See `bti_collect_partition_rows`: `true` iff the manager resolved this
941        // reader by an exact fully-qualified `keyspace.table` match (or the query
942        // was unqualified). Threaded into the seek table-consistency guard (#1284).
943        fully_qualified_match: bool,
944        schema_opt: Option<&crate::schema::TableSchema>,
945        parser: &crate::storage::sstable::reader::parsing::V5CompressedLegacyParser,
946    ) -> Result<Option<Vec<ScanRow>>> {
947        // Issue #1573 (C2): positioned reads on the shared `point_source` — no
948        // per-lookup `open(2)`, no cursor, no mutex. `chunk_index` is a plain local.
949
950        // Determine the chunk-targeting parameters. `chunk_length == 0` (or no
951        // CompressionInfo) means we cannot chunk-target -> whole-section fallback.
952        let chunk_length = self
953            .compression_info
954            .as_ref()
955            .map(|ci| ci.chunk_length as usize)
956            .filter(|&len| len > 0);
957
958        let mut chunk_index = 0usize;
959        let (target_chunk, window_base, mut window) = match chunk_length {
960            Some(len) => {
961                let (target_chunk, window_base, _within) = Self::bti_chunk_target(offset, len);
962                // Positioned reads resolve their own offset from the chunk index.
963                chunk_index = target_chunk;
964                (target_chunk, window_base, Vec::<u8>::new())
965            }
966            None => {
967                // Whole-section fallback (uncompressed BTI, or chunk_length absent/0):
968                // one positioned read of the whole data section, CRC-verified when a
969                // CRC.db is present (see `point_read_whole_section`).
970                let whole = self.point_read_whole_section().await?;
971                (0usize, 0usize, whole)
972            }
973        };
974
975        if offset < window_base {
976            return Err(Error::corruption(format!(
977                "BTI single-partition seek: resolved offset {} precedes window base {} (chunk {})",
978                offset, window_base, target_chunk
979            )));
980        }
981        let within = offset - window_base;
982        let chunk_targeted = chunk_length.is_some();
983
984        if chunk_targeted {
985            // Resolve the AUTHORITATIVE exclusive end of the target partition in
986            // the UNCOMPRESSED offset domain. Non-last partitions are bounded by
987            // the successor partition's start (`end_bound`); the LAST partition is
988            // bounded by the data-section length. When NEITHER is known we cannot
989            // bound the last partition without re-introducing a heuristic, so we
990            // return `Ok(None)` and let the caller fall back to a full scan.
991            let end_offset = match end_bound {
992                Some(end) => end,
993                None => match self
994                    .compression_info
995                    .as_ref()
996                    .map(|ci| ci.data_length as usize)
997                    .filter(|&len| len > offset)
998                {
999                    Some(len) => len,
1000                    None => {
1001                        debug!(
1002                            "BTI single-partition seek: last partition at offset {} has no \
1003                             authoritative end (no successor, no usable data_length); falling \
1004                             back to full scan",
1005                            offset
1006                        );
1007                        return Ok(None);
1008                    }
1009                },
1010            };
1011
1012            // Step 1: buffer enough chunks to expose the partition header, then run
1013            // the prefix-collision / chunk-straddle gate. This bails out cheaply
1014            // (without decompressing the rest of the partition) when the trie/index
1015            // candidate is a prefix collision for an absent key.
1016            loop {
1017                // Pull a chunk if the header is not yet (fully) buffered.
1018                if within + 2 > window.len()
1019                    || !Self::bti_partition_key_bytes_available(&window, within, key.as_bytes())
1020                {
1021                    match self
1022                        .bti_pull_decompressed_chunk(&mut chunk_index, &mut window)
1023                        .await?
1024                    {
1025                        true => continue, // chunk appended; re-check the header
1026                        false => {
1027                            // EOF before the header is buffered: nothing decodable
1028                            // at the resolved offset.
1029                            return Ok(Some(Vec::new()));
1030                        }
1031                    }
1032                }
1033
1034                let key_matches = self.bti_partition_key_matches(&window, within, key.as_bytes());
1035                if !key_matches {
1036                    debug!(
1037                        "BTI seek candidate at offset {} did not match queried key \
1038                         (prefix collision); treating as absent",
1039                        offset
1040                    );
1041                    return Ok(Some(Vec::new()));
1042                }
1043                break; // header buffered AND key matches
1044            }
1045
1046            // Step 2: buffer EXACTLY the chunks covering `[offset, end_offset)` —
1047            // never stitch to EOF (the #953 MEDIUM finding: a head-of-file lookup
1048            // would otherwise decompress the whole file). `end_offset` is in the
1049            // same uncompressed-offset domain as `window_base + window.len()`, so
1050            // the window holds the whole partition once `window.len()` reaches
1051            // `end_offset - window_base` (or EOF — a stale end never reads past
1052            // EOF). Decompressing the FULL extent before parsing means a row/cell
1053            // that spans multiple compression chunks is present in full, so the
1054            // single parse below collects every target row without truncation.
1055            let needed = end_offset.saturating_sub(window_base);
1056            while window.len() < needed {
1057                if !self
1058                    .bti_pull_decompressed_chunk(&mut chunk_index, &mut window)
1059                    .await?
1060                {
1061                    break; // EOF: window holds all available bytes.
1062                }
1063            }
1064            return self
1065                .bti_collect_partition_rows(
1066                    &window,
1067                    within,
1068                    row_body_window,
1069                    key,
1070                    table_id,
1071                    fully_qualified_match,
1072                    schema_opt,
1073                    parser,
1074                )
1075                .map(|(rows, _complete)| Some(rows));
1076        }
1077
1078        // Whole-section fallback (uncompressed BTI): every byte is already present,
1079        // so the first parse is authoritative.
1080        if within >= window.len() {
1081            return Err(Error::corruption(format!(
1082                "BTI trie resolved Data.db offset {} beyond decompressed data section ({} bytes)",
1083                offset,
1084                window.len()
1085            )));
1086        }
1087        self.bti_collect_partition_rows(
1088            &window,
1089            within,
1090            row_body_window,
1091            key,
1092            table_id,
1093            fully_qualified_match,
1094            schema_opt,
1095            parser,
1096        )
1097        .map(|(rows, _complete)| Some(rows))
1098    }
1099
1100    /// Read the next compressed chunk from `cursor`, decompress it (if the reader
1101    /// has a compression algorithm), and append the decompressed bytes to
1102    /// `window`. Returns `true` when a chunk was appended, `false` at EOF.
1103    ///
1104    /// Shared by the chunk-targeted seek so the header-buffering and
1105    /// partition-bounding loops use one decompression code path; each call bumps
1106    /// `work_counters::chunks_decompressed` so a test can prove the seek bounded
1107    /// its decompression to the target partition's chunk span (Issue #953/#951).
1108    #[cfg(not(feature = "tombstones"))]
1109    async fn bti_pull_decompressed_chunk(
1110        &self,
1111        chunk_index: &mut usize,
1112        window: &mut Vec<u8>,
1113    ) -> Result<bool> {
1114        use crate::storage::sstable::compression::Compression;
1115        // Issue #1573 (C2): positioned chunk fetch — no cursor, no mutex, no
1116        // per-lookup open. CRC is verified inside `point_read_compressed_chunk`
1117        // BEFORE we decompress here (guardrail #1411).
1118        match self.point_read_compressed_chunk(*chunk_index)? {
1119            Some(compressed_chunk) => {
1120                *chunk_index += 1;
1121                let decompressed_chunk = if let Some(compression_reader) = &self.compression_reader
1122                {
1123                    let compression = Compression::new(*compression_reader.algorithm())?;
1124                    compression.decompress(&compressed_chunk).map_err(|e| {
1125                        Error::corruption(format!(
1126                            "BTI single-partition seek: failed to decompress chunk: {}",
1127                            e
1128                        ))
1129                    })?
1130                } else {
1131                    // No compression reader despite CompressionInfo: treat the raw
1132                    // chunk bytes as already-decompressed data.
1133                    compressed_chunk
1134                };
1135                // Issue #953/#951: count every chunk the seek materializes so a
1136                // bound test can prove the decompression window is bounded to the
1137                // target partition's chunk span, not stitched to EOF.
1138                super::super::super::work_counters::add_chunk_decompressed();
1139                window.extend_from_slice(&decompressed_chunk);
1140                Ok(true)
1141            }
1142            None => Ok(false),
1143        }
1144    }
1145
1146    /// Parse the buffered `window` from `within`, collecting every row of the
1147    /// FIRST (target) partition and stopping at the next partition boundary.
1148    ///
1149    /// Returns `(rows, saw_next_partition)`:
1150    /// - `rows` — the target partition's row `Value`s (those whose decoded key
1151    ///   equals `key` and whose table id matches, issue #831 wrong-table guard),
1152    ///   in on-disk order.
1153    /// - `saw_next_partition` — `true` iff the parser emitted a fully-decoded row
1154    ///   whose partition key DIFFERS from `key`, at which point collection stops.
1155    ///
1156    /// Because the caller now decompresses the partition's AUTHORITATIVE byte
1157    /// extent `[offset, end)` before parsing (the successor offset / data-section
1158    /// length, issue #953 / #951), the window always fully contains the target
1159    /// partition — there is no mid-partition truncation to resolve. The
1160    /// `Break`-on-different-key behaviour is defence in depth: when the window's
1161    /// final chunk overruns slightly into the next partition (chunks are
1162    /// fixed-size, so the extent rounds up to a chunk boundary), the first
1163    /// different-key row terminates collection so no next-partition row is ever
1164    /// kept. The returned flag is currently informational; the caller does not loop
1165    /// on it (the bound is authoritative, not boundary-scanned).
1166    ///
1167    /// Issue #954: when `row_body_window` is `Some((start_rel, end_rel))` the
1168    /// parse is bounded to that within-partition byte window (relative to the
1169    /// partition start, i.e. the `window[within..]` slice domain) so only the
1170    /// clustering slice's row-index block(s) are decoded. `None` parses the whole
1171    /// partition (the #953 behaviour).
1172    #[cfg(not(feature = "tombstones"))]
1173    fn bti_collect_partition_rows(
1174        &self,
1175        window: &[u8],
1176        within: usize,
1177        row_body_window: Option<(usize, usize)>,
1178        key: &RowKey,
1179        table_id: &TableId,
1180        // Whether the manager resolved this reader by an EXACT fully-qualified
1181        // `keyspace.table` match (or an unqualified query). When `false` a
1182        // fully-qualified query reached this reader via the bare-name fallback, so
1183        // the seek guard keeps STRICT keyspace matching (#1284 review).
1184        fully_qualified_match: bool,
1185        schema_opt: Option<&crate::schema::TableSchema>,
1186        parser: &crate::storage::sstable::reader::parsing::V5CompressedLegacyParser,
1187    ) -> Result<(Vec<ScanRow>, bool)> {
1188        let mut rows: Vec<ScanRow> = Vec::new();
1189        let mut saw_next_partition = false;
1190        // Clamp the window's end to the available bytes (`usize::MAX` means "to the
1191        // partition end"); the start is already within-partition-relative, which is
1192        // the same domain as `window[within..]`.
1193        let clamped_window = row_body_window.map(|(start, end)| {
1194            let avail = window.len().saturating_sub(within);
1195            (start.min(avail), end.min(avail))
1196        });
1197        parser.parse_block_emit_windowed(
1198            &window[within..],
1199            schema_opt,
1200            self,
1201            clamped_window,
1202            |(tid, entry_key, entry_value)| {
1203                if entry_key.as_bytes() == key.as_bytes() {
1204                    // Header-authoritative table consistency: wrong-table rejected
1205                    // (#831); a keyspace-divergent same-table query is served ONLY
1206                    // when resolution was an exact fully-qualified match — a
1207                    // fallback-resolved query keeps strict keyspace matching so it
1208                    // never returns another keyspace's same-named rows (#1284).
1209                    if table_header_consistent_for_seek(&tid, table_id, fully_qualified_match) {
1210                        rows.push(entry_value);
1211                    }
1212                    Ok(std::ops::ControlFlow::Continue(()))
1213                } else {
1214                    // First row of the NEXT partition (the authoritative extent
1215                    // can overrun into it by up to one chunk). Stop here so no
1216                    // next-partition row is collected; the target partition's rows
1217                    // are already complete because its whole extent was buffered.
1218                    saw_next_partition = true;
1219                    Ok(std::ops::ControlFlow::Break(()))
1220                }
1221            },
1222        )?;
1223        Ok((rows, saw_next_partition))
1224    }
1225
1226    /// Returns true when the `[flags][key_len: u8][key bytes]` prefix at `within`
1227    /// is fully present in `window` AND `key_len` equals `expected_key.len()`.
1228    ///
1229    /// Used by the chunk-targeted BTI lookup to decide whether the INVARIANT-3
1230    /// key match can be evaluated yet, or whether more chunk bytes must be pulled
1231    /// first (issue #831).
1232    fn bti_partition_key_bytes_available(
1233        window: &[u8],
1234        within: usize,
1235        _expected_key: &[u8],
1236    ) -> bool {
1237        // Need flags + key_len byte first.
1238        if within + 2 > window.len() {
1239            return false;
1240        }
1241        let key_len = window[within + 1] as usize;
1242        // The declared key bytes must all be buffered. (Whether `key_len` equals
1243        // the expected length is decided by the subsequent match check, which
1244        // fails fast on a mismatch — here we only require the bytes be present.)
1245        within + 2 + key_len <= window.len()
1246    }
1247
1248    /// Verify the on-disk partition-key bytes at `offset` in the decompressed
1249    /// data section equal `expected_key` (issue #831, INVARIANT 3).
1250    ///
1251    /// Reads the `[flags][key_len: u8][key bytes]` prefix. Returns `false` (rather
1252    /// than erroring) on any structural mismatch so the caller can treat the trie
1253    /// candidate as absent.
1254    fn bti_partition_key_matches(
1255        &self,
1256        decompressed: &[u8],
1257        offset: usize,
1258        expected_key: &[u8],
1259    ) -> bool {
1260        // Need at least flags + key_len.
1261        if offset + 2 > decompressed.len() {
1262            return false;
1263        }
1264        let key_len = decompressed[offset + 1] as usize;
1265        let key_start = offset + 2;
1266        let key_end = key_start + key_len;
1267        if key_end > decompressed.len() {
1268            return false;
1269        }
1270        &decompressed[key_start..key_end] == expected_key
1271    }

1273    /// BTI ("da") full scan: decompress the whole Data.db section and parse
1274    /// every partition in token order (issue #660).
1275    ///
1276    /// BTI SSTables carry no Index.db/Summary.db, so a range/full scan cannot
1277    /// use the index path. Instead we stitch the entire (chunk-compressed) data
1278    /// section into one buffer and run [`parse_block_with_cell_metadata`], which
1279    /// walks ALL partitions — the same per-partition decode the point-lookup
1280    /// path uses, but without stopping at the first match.
1281    ///
1282    /// Returns entries with per-cell write metadata so the WRITETIME/TTL scan
1283    /// (`scan_with_cell_metadata`) and the plain `scan` (which drops the metadata)
1284    /// can share a single implementation. Results are filtered by the optional
1285    /// `[start_key, end_key]` range and tombstone-suppressed, then sorted into
1286    /// Murmur3 token order and truncated to `limit` — identical post-processing
1287    /// to the V5CompressedLegacy stitched path.
1288    ///
1289    /// Uses its own per-scan
1290    /// [`ScanCursor`](crate::storage::sstable::reader::source::ScanCursor), so it
1291    /// runs in parallel with other scans on this reader without serialization
1292    /// (issue #815).
1293    ///
1294    /// [`parse_block_with_cell_metadata`]: crate::storage::sstable::reader::parsing::V5CompressedLegacyParser::parse_block_with_cell_metadata
1295    ///
1296    /// `read_shadowing` (issue #1741): `true` for user-facing SELECT scans
1297    /// (`scan`, `scan_with_cell_metadata`), `false` for the physical
1298    /// `get_all_entries` (integrity verification / data-manager) which must count
1299    /// every on-disk row.
1300    pub(super) async fn bti_scan_with_metadata(
1301        &self,
1302        start_key: Option<&RowKey>,
1303        end_key: Option<&RowKey>,
1304        limit: Option<usize>,
1305        schema: Option<&crate::schema::TableSchema>,
1306        read_shadowing: bool,
1307    ) -> Result<
1308        Vec<(
1309            RowKey,
1310            ScanRow,
1311            std::collections::HashMap<String, CellWriteMetadata>,
1312        )>,
1313    > {
1314        let cursor = self.new_scan_cursor().await?;
1315
1316        // Decompress the entire data section. Precondition for stitch_all_chunks:
1317        // cursor's file seeked to data-section start (fresh cursor is at chunk 0).
1318        let header_size = self.calculate_header_size();
1319        {
1320            let mut file_guard = cursor.file.lock().await;
1321            // A5 read-work counter (SEEK_CALLS; consumer E4): scan seek-to-data-start
1322            // before stitching the section. No-op in release (design.md Decision 1/2).
1323            crate::storage::sstable::read_work_counters::record_seek();
1324            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
1325        }
1326        let whole = self.stitch_all_chunks(&cursor).await?;
1327
1328        // Resolve schema via the four-tier strategy (provided > header > registry).
1329        // V5CompressedLegacy partition decode requires a schema (cells lack names).
1330        let effective_schema = self.get_table_schema(schema);
1331        let parser = self.build_v5_parser(read_shadowing);
1332        let parsed =
1333            parser.parse_block_with_cell_metadata(&whole, effective_schema.as_ref(), self)?;
1334
1335        let mut results = Vec::new();
1336        for (_entry_table_id, entry_key, entry_value, cell_meta) in parsed {
1337            if let Some(start) = start_key {
1338                if &entry_key < start {
1339                    continue;
1340                }
1341            }
1342            if let Some(end) = end_key {
1343                if &entry_key > end {
1344                    continue;
1345                }
1346            }
1347            if !self.filter_tombstone(&entry_value) {
1348                continue;
1349            }
1350            results.push((entry_key, entry_value, cell_meta));
1351        }
1352
1353        sort_by_token_order_with_meta(&mut results);
1354        if let Some(lim) = limit {
1355            results.truncate(lim);
1356        }
1357
1358        log::debug!(
1359            "SSTableReader::bti_scan_with_metadata - Returning {} results",
1360            results.len()
1361        );
1362        Ok(results)
1363    }
1364}
