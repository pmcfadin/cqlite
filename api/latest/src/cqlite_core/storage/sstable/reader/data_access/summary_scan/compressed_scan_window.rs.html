1//! Chunk-aligned coalesced read window for the Summary-guided compressed scan
2//! (issue #2877), extracted from the parent
3//! [`summary_scan`](super) module (campsite rule, epic #1116).
4//!
5//! The parent module owns the WALK (`walk_in_range_partition_slices`); this
6//! module owns the one piece of state that walk carries for a COMPRESSED
7//! `Data.db`: [`CompressedScanWindow`], the sliding window of decompressed bytes
8//! that serves consecutive in-range partition slices from ONE coalesced,
9//! chunk-aligned `read_compressed_offset_window` call instead of one call per
10//! partition.
11
12use std::sync::atomic::{AtomicU64, Ordering};

14use crate::storage::sstable::compression_info::CompressionInfo;

19/// Steady-state target size (bytes) of the coalesced, chunk-aligned
20/// compressed-scan read window (issue #2877), mirroring the uncompressed
21/// non-stitching walk's `SEQUENTIAL_WINDOW_TARGET_BYTES` (`full_index_stream.rs`,
22/// 4 MiB) precedent. See [`CompressedScanWindow`] for why the window's end is
23/// always rounded UP to a `CompressionInfo.chunk_length` boundary rather than cut
24/// at the target byte count exactly, and why a window only REACHES this size
25/// after ramping up to it.
26const COMPRESSED_SCAN_WINDOW_TARGET_BYTES: u64 = 4 * 1024 * 1024;
27
28/// Hard ceiling on the ramped greedy FLOOR (issue #2877 roborev, blocker A): the
29/// ramp doubles `chunk_length` per refill but can never ask for more than the
30/// steady-state target, so `chunk_length << ramp` is clamped here and the ramp
31/// counter stops advancing once the clamp binds. Equal to
32/// [`COMPRESSED_SCAN_WINDOW_TARGET_BYTES`] by construction — named separately
33/// because it is the ramp's *invariant*, asserted directly by the unit tests
34/// below.
35///
36/// Scope, precisely (it bounds the FLOOR, not the WINDOW): a refill's length is
37/// `max(minimal, floor)` rounded UP to a `chunk_length` boundary, so the window
38/// can legitimately exceed this cap by (a) up to one `chunk_length` of alignment
39/// rounding and (b) however much a SINGLE partition larger than the floor needs
40/// (`minimal`, deliberately never `min`-ed — the large-partition case must not
41/// regress). Concretely, `CompressionInfo::parse` admits a `chunk_length` up to
42/// 256 MiB, and an 8 MiB `chunk_length` therefore yields an 8 MiB window even
43/// though the floor clamps at 4 MiB. What the cap DOES guarantee is that
44/// read-ahead — bytes fetched beyond what the current partition needs — is
45/// bounded by the target plus one chunk, which is what the memory derivation
46/// under "Preserved invariants" below spends (`+ one chunk_length`).
47const COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES: u64 = COMPRESSED_SCAN_WINDOW_TARGET_BYTES;
48
49/// Process-global count of coalesced window REFILLS — one per
50/// `read_compressed_offset_window` call the Summary-guided compressed scan makes
51/// (issue #2877). Mirrors
52/// [`DECOMPRESS_CALLS`](crate::storage::sstable::reader::data_access::DECOMPRESS_CALLS)'s
53/// always-on `Relaxed`-add shape (integration tests compile the lib without its
54/// `test` cfg, and one atomic add per multi-MiB read is free).
55///
56/// This is the observable proof of the RAMP: covering an N-chunk data section in
57/// O(log(target/chunk_length)) + N*chunk_length/target refills — rather than one
58/// refill per chunk boundary crossed — is only possible if the window actually
59/// grows to its steady-state target. Read/reset via
60/// [`SSTableReader::scan_window_refill_count`] /
61/// [`SSTableReader::reset_scan_window_counters`].
62static SCAN_WINDOW_REFILLS: AtomicU64 = AtomicU64::new(0);
63
64/// Process-global count of dead-prefix COMPACTIONS (the `Vec::drain` memmove that
65/// reclaims already-served bytes) performed by the coalesced window (issue #2877).
66///
67/// The oracle for roborev blocker B: compaction must be O(refills), NOT
68/// O(partitions). Draining the dead prefix for every partition served shifts the
69/// whole remaining window each time — quadratic byte copying across the thousands
70/// of narrow partitions one 4 MiB window holds, which is exactly the coalescing
71/// win this issue exists to deliver. Read/reset via
72/// [`SSTableReader::scan_window_prefix_compaction_count`] /
73/// [`SSTableReader::reset_scan_window_counters`].
74static SCAN_WINDOW_PREFIX_COMPACTIONS: AtomicU64 = AtomicU64::new(0);
75
76impl SSTableReader {
77    /// Coalesced compressed-scan window refills since the last
78    /// [`reset_scan_window_counters`](Self::reset_scan_window_counters) (issue
79    /// #2877). See [`SCAN_WINDOW_REFILLS`].
80    pub fn scan_window_refill_count() -> u64 {
81        SCAN_WINDOW_REFILLS.load(Ordering::Relaxed)
82    }
83
84    /// Coalesced compressed-scan window dead-prefix compactions (memmoves) since
85    /// the last [`reset_scan_window_counters`](Self::reset_scan_window_counters)
86    /// (issue #2877). See [`SCAN_WINDOW_PREFIX_COMPACTIONS`].
87    pub fn scan_window_prefix_compaction_count() -> u64 {
88        SCAN_WINDOW_PREFIX_COMPACTIONS.load(Ordering::Relaxed)
89    }
90
91    /// Reset both coalesced-window work counters (test/instrumentation harness).
92    pub fn reset_scan_window_counters() {
93        SCAN_WINDOW_REFILLS.store(0, Ordering::Relaxed);
94        SCAN_WINDOW_PREFIX_COMPACTIONS.store(0, Ordering::Relaxed);
95    }
96}

98/// A chunk-aligned sliding window over the DECOMPRESSED bytes of a compressed
99/// scan, keyed to the UNCOMPRESSED data-section offset domain (issue #2877).
100///
101/// `walk_in_range_partition_slices` used to call
102/// [`read_compressed_offset_window`](SSTableReader::read_compressed_offset_window)
103/// once PER PARTITION. That helper maps `[start, start+size)` onto
104/// `CompressionInfo.db` chunks and decompresses every chunk it touches with no
105/// cross-call memoisation — so a 16-64 KiB chunk holding many narrow partitions
106/// was read + decompressed once per partition it contains, entirely bypassing
107/// the Epic B decompressed-chunk cache on this hot path (the issue's root
108/// cause). This window instead accumulates consecutive in-range partitions and
109/// refills in ONE coalesced call whenever the current partition's span is not
110/// already covered.
111///
112/// # Why chunk-ALIGNED, not just chunk-sized
113///
114/// A window boundary that falls in the MIDDLE of a chunk would make that chunk
115/// get decompressed twice: once as the tail of window N, once again as the
116/// head of window N+1 (both calls independently map their own `[start, end)`
117/// onto `CompressionInfo.chunk_length`-sized chunks and decompress whichever
118/// they touch). Rounding every refill's end UP to the next
119/// `chunk_length` boundary — and always resuming the NEXT window exactly at the
120/// previous window's end — means windows tile the byte space with NO gaps and
121/// NO overlaps at chunk granularity, so each chunk is decompressed by EXACTLY
122/// ONE window's read call across the whole scan (the acceptance criterion this
123/// window exists to satisfy).
124///
125/// # Why the size RAMPS instead of jumping straight to the target
126///
127/// (Issue #2877 roborev, blocker A.) A flat "every fresh fill reads at least 4
128/// MiB" defeats the walk's own token pushdown and early termination: a `decode`
129/// closure that returns `ControlFlow::Break` after one row, or a narrow token
130/// range holding a handful of partitions, would still read AND LZ4-decompress 4
131/// MiB of mostly out-of-range partition bodies. So the greedy floor starts at ONE
132/// `chunk_length` and DOUBLES per refill ([`Self::ramped_floor`]), clamped at
133/// [`COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES`]:
134///
135/// - an early-terminating callback pays ~one chunk, not 4 MiB;
136/// - a narrow token range never matures the ramp (few refills ⇒ small floor), and
137///   its total over-read is bounded by what it has already legitimately read
138///   (each refill is at most double the previous floor);
139/// - a long full scan reaches the 4 MiB steady state after ~`log2(target /
140///   chunk_length)` refills — a few hundred KiB of ramp-up for a 16 KiB chunk —
141///   and is asymptotically identical to a flat 4 MiB window.
142///
143/// The floor is only ever a FLOOR: `minimal.max(floor)` keeps a single partition
144/// LARGER than the ramped size served in one go, so the large-partition case
145/// never regresses.
146///
147/// # Preserved invariants
148///
149/// - **CRC-before-decompress ordering** (guardrail #1411/#1773): unchanged —
150///   every chunk still goes through `read_compressed_offset_window`'s
151///   CRC-validated chunk reader, just fewer times.
152/// - **SCAN-plane reads** (issue #2876, and the reason the two fixes are
153///   inseparable): every refill is issued on the positional plane the WALK hands
154///   in — the reader's never-`MADV_RANDOM` `scan_positional_source` — never the
155///   `MADV_RANDOM` `point_source`. Coalescing and the read-intent split are
156///   complementary halves of one mechanism: bigger sequential reads only pay off
157///   on a mapping that reads ahead, and CASSANDRA-15452 is the upstream precedent
158///   for the failure mode (their userspace scan buffer was defeated by the layer
159///   underneath it). Pinned end-to-end by
160///   `cqlite-core/tests/issue_2877_scan_chunk_coalescing.rs`'s combined test,
161///   which counts reads PER PLANE and requires zero on the point plane.
162/// - **`partition_slice_fully_consumed`** (Signal B): unchanged — still checked
163///   per partition against the slice this window serves.
164/// - **Dead-prefix reclamation is O(refills), not O(partitions)** (issue #2877
165///   roborev, blocker B): the already-served prefix is dropped at REFILL time
166///   ([`Self::compact_prefix`]) rather than after every partition, so serving
167///   thousands of narrow partitions out of one window costs zero byte copying.
168///   Mechanism, precisely: a per-refill `Vec::drain` of the dead prefix plus an
169///   advance of [`Self::start`](Self#structfield.start). There is deliberately NO
170///   separate logical-`head` offset field — draining once per refill is already
171///   O(refills), and one field fewer means one fewer invariant to keep.
172///   Compaction drops only from the FRONT, so the chunk-aligned TAIL boundary
173///   (`self.start + bytes.len()`) is invariant under it and a chunk already paid
174///   for can never be re-decompressed.
175/// - **Memory bound** (updated for the lazy reclamation above): `largest
176///   partition span + max(COMPRESSED_SCAN_WINDOW_TARGET_BYTES, largest partition
177///   span) + one chunk_length`. Derivation: compaction runs BEFORE every append,
178///   and a straddling partition's live remainder is `have_end - start < end -
179///   start` — i.e. under its own span; the append itself adds at most
180///   `max(minimal, ramped floor)` rounded up to a chunk, and both `minimal` and
181///   the floor are bounded by `max(largest span, target)`. A gap-realign replaces
182///   the buffer outright, so it is bounded by the second term alone. Deferring
183///   reclamation to refill time therefore costs at most ONE extra partition span
184///   of residency versus draining per partition — never unbounded growth, and
185///   comfortably within the `<128MB` target.
186/// - **The ramp counts CONSECUTIVE partitions only**: a gap-realign (the walk
187///   skipped an out-of-range run — e.g. a wraparound scan's jump to its second
188///   segment) RESETS the ramp to one chunk. Sequential locality is exactly what
189///   the ramp is spending read-ahead on, and a jump proves it was just broken; the
190///   new region re-earns its window. So a sparse/narrow range that keeps jumping
191///   can never accumulate its way to a 4 MiB over-read, while a dense scan (one
192///   initial gap fill, then straddle-appends) ramps uninterrupted.
193pub(super) struct CompressedScanWindow {
194    /// Decompressed bytes of `[start, start + bytes.len())` in the UNCOMPRESSED
195    /// data-section offset domain. Empty until the first refill.
196    bytes: Vec<u8>,
197    /// Start offset of `bytes`, in the same domain. Meaningless while `filled`
198    /// is `false`. Advances only at refill time (see [`Self::compact_prefix`]),
199    /// so between refills it lags behind the partition being served — which is
200    /// precisely what makes per-partition serving copy-free.
201    start: u64,
202    /// Whether `bytes`/`start` hold a real window yet.
203    filled: bool,
204    /// Doubling shift count for the greedy floor (`chunk_length << ramp`, clamped
205    /// at [`COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES`]). Starts at 0 (one chunk),
206    /// advances one step per refill until the clamp binds, and RESETS to 0 on a
207    /// gap-realign (the consecutive run it was earning read-ahead for is broken).
208    ramp: u32,
209}
210
211impl CompressedScanWindow {
212    pub(super) fn new() -> Self {
213        Self {
214            bytes: Vec::new(),
215            start: 0,
216            filled: false,
217            ramp: 0,
218        }
219    }
220
221    /// Current greedy FLOOR for a refill: `chunk_length << ramp`, clamped at
222    /// [`COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES`].
223    ///
224    /// Overflow-safe by construction — `chunk_length` comes from
225    /// `CompressionInfo.db` and may be any `u32` (including a hostile 4 GiB-1):
226    /// the shift is applied to `1u64` under `checked_shl` (so a ramp ≥ 64 cannot
227    /// UB) and multiplied in with `saturating_mul`, then clamped. No arithmetic
228    /// path can panic, wrap, or exceed the cap.
229    fn ramped_floor(&self, chunk_length: u64) -> u64 {
230        let factor = 1u64.checked_shl(self.ramp).unwrap_or(u64::MAX);
231        chunk_length
232            .saturating_mul(factor)
233            .min(COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES)
234    }
235
236    /// Advance the ramp one doubling step, unless the clamp already binds (so the
237    /// counter itself stays small and `ramped_floor` is a fixed point at the cap).
238    fn advance_ramp(&mut self, chunk_length: u64) {
239        if self.ramped_floor(chunk_length) < COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES {
240            self.ramp = self.ramp.saturating_add(1);
241        }
242    }
243
244    /// Drop the dead prefix `[self.start, live_from)` — bytes the strictly
245    /// ascending walk can never ask for again — and advance `self.start` to
246    /// match. Called ONLY from a refill (issue #2877 roborev, blocker B), so the
247    /// per-partition serving path performs zero copying.
248    ///
249    /// Front-only: `self.start + bytes.len()` (the chunk-aligned tail boundary a
250    /// straddle-append resumes from) is invariant under this operation.
251    fn compact_prefix(&mut self, live_from: u64) {
252        if !self.filled || live_from <= self.start {
253            return;
254        }
255        let drop_n = usize::try_from(live_from - self.start)
256            .unwrap_or(usize::MAX)
257            .min(self.bytes.len());
258        if drop_n == 0 {
259            return;
260        }
261        self.bytes.drain(0..drop_n);
262        self.start += drop_n as u64;
263        SCAN_WINDOW_PREFIX_COMPACTIONS.fetch_add(1, Ordering::Relaxed);
264    }
265
266    /// Fetch chunk-aligned decompressed bytes starting at `aligned_from` via the
267    /// reader's existing `read_compressed_offset_window`, sized to cover at least
268    /// up to `need_until`, padded out to `floor` (the ramped greedy floor), and
269    /// rounded UP to a `chunk_length` boundary — never past `data_section_end`.
270    ///
271    /// `minimal.max(floor)` (never `min`): a single partition LARGER than the
272    /// ramped floor is still served by one call, so the large-partition case
273    /// cannot regress.
274    #[allow(clippy::too_many_arguments)]
275    async fn fetch_aligned(
276        reader: &SSTableReader,
277        source: &dyn ReadAt,
278        ci: &CompressionInfo,
279        chunk_length: u64,
280        aligned_from: u64,
281        need_until: u64,
282        data_section_end: u64,
283        floor: u64,
284    ) -> Result<Vec<u8>> {
285        let remaining = data_section_end.saturating_sub(aligned_from);
286        let minimal = need_until.saturating_sub(aligned_from);
287        let want = minimal.max(floor).min(remaining);
288        let raw_end = aligned_from.saturating_add(want);
289        let rounded_end = raw_end.div_ceil(chunk_length).saturating_mul(chunk_length);
290        let window_end = rounded_end.min(aligned_from.saturating_add(remaining));
291        let window_len = window_end.saturating_sub(aligned_from);
292        // Fail CLOSED on a zero-length request: `read_compressed_offset_window`
293        // documents `size >= 1` (it derives `last_chunk` from `end - 1`). A valid
294        // walk never asks for this (`need_until > aligned_from` and
295        // `need_until <= data_section_end`), so reaching it means the caller's
296        // offsets contradict `CompressionInfo.data_length`.
297        if window_len == 0 {
298            return Err(Error::corruption(format!(
299                "walk_in_range_partition_slices: coalesced compressed window at \
300                 {aligned_from} would be empty (need_until {need_until}, \
301                 data_section_end {data_section_end}, issue #2877)"
302            )));
303        }
304        let Ok(window_len_u32) = u32::try_from(window_len) else {
305            return Err(Error::corruption(format!(
306                "walk_in_range_partition_slices: coalesced compressed window length \
307                 {window_len} overflows u32 (issue #2877)"
308            )));
309        };
310        SCAN_WINDOW_REFILLS.fetch_add(1, Ordering::Relaxed);
311        // `source` is the walk's SCAN-intent plane (issue #2876), threaded through
312        // rather than read off the reader: the coalescing window must never widen
313        // reads onto the `MADV_RANDOM` point mapping, or the readahead the larger
314        // reads exist to earn is suppressed again (CASSANDRA-15452's lesson).
315        reader
316            .read_compressed_offset_window(source, ci, aligned_from, window_len_u32)
317            .await
318    }
319
320    /// Serve `[start, end)` from this window, refilling iff not already
321    /// covered. `data_section_end` bounds the last window so it never reads
322    /// past the data section.
323    ///
324    /// Two distinct refill shapes (issue #2877 roborev finding, High —
325    /// correctness): a naive "always resume exactly where the previous window
326    /// ended" is WRONG whenever the current partition does not start exactly
327    /// there.
328    /// - **Straddle** (`start` is still inside the buffered window but `end`
329    ///   runs past it — the common case once many small partitions have been
330    ///   served from one big window): APPEND new chunk-aligned bytes onto the
331    ///   tail (relative to the TRUE buffered end, computed before any
332    ///   compaction), preserving the already-decompressed live prefix, so a
333    ///   chunk already paid for is never re-decompressed.
334    /// - **Gap** (`start` is beyond the buffered window entirely — e.g. the
335    ///   out-of-range run SKIPPED between a compressed wraparound scan's two
336    ///   segments, which are never read so the window never advances for
337    ///   them; or the very first fill): REALIGN directly to the chunk
338    ///   containing `start`, discarding whatever was buffered. Blindly
339    ///   "continuing" from the stale tail here either UNDERFLOWS (`start -
340    ///   self.start` when the stale tail is ahead of `start`) or leaves the
341    ///   window short of `start` entirely (a false corruption error) —
342    ///   exactly the bug this fixes.
343    pub(super) async fn slice(
344        &mut self,
345        reader: &SSTableReader,
346        source: &dyn ReadAt,
347        ci: &CompressionInfo,
348        start: u64,
349        end: u64,
350        data_section_end: u64,
351    ) -> Result<&[u8]> {
352        let chunk_length = ci.chunk_length as u64;
353        if chunk_length == 0 {
354            return Err(Error::corruption(
355                "walk_in_range_partition_slices: CompressionInfo chunk_length is zero \
356                 (issue #2877)"
357                    .to_string(),
358            ));
359        }
360        if end <= start {
361            return Err(Error::corruption(format!(
362                "walk_in_range_partition_slices: empty/reversed partition slice \
363                 [{start}, {end}) (issue #2877)"
364            )));
365        }
366        // The TRUE buffered end — captured BEFORE any compaction, which only ever
367        // drops from the front and therefore leaves this value unchanged.
368        let have_end = if self.filled {
369            self.start + self.bytes.len() as u64
370        } else {
371            0
372        };
373
374        if !self.filled || start < self.start || end > have_end {
375            if self.filled && start >= self.start && start <= have_end {
376                // Straddle: CONSECUTIVE with what we already served, so the ramp
377                // carries over. Reclaim the dead prefix (O(refills), blocker B),
378                // then APPEND from the true buffered end.
379                let floor = self.ramped_floor(chunk_length);
380                self.compact_prefix(start);
381                let extra = Self::fetch_aligned(
382                    reader,
383                    source,
384                    ci,
385                    chunk_length,
386                    have_end,
387                    end,
388                    data_section_end,
389                    floor,
390                )
391                .await?;
392                self.bytes.extend_from_slice(&extra);
393            } else {
394                // Gap (or the very first fill, or a defensive `start <
395                // self.start`): realign fresh, discarding the stale buffer. The
396                // sequential run the ramp was paying for is BROKEN here, so the
397                // ramp resets — the new region re-earns its read-ahead one chunk
398                // at a time (issue #2877 roborev blocker A: a sparse range that
399                // keeps jumping must never accumulate into a 4 MiB over-read).
400                self.ramp = 0;
401                let floor = self.ramped_floor(chunk_length);
402                let aligned_start = (start / chunk_length) * chunk_length;
403                self.bytes = Self::fetch_aligned(
404                    reader,
405                    source,
406                    ci,
407                    chunk_length,
408                    aligned_start,
409                    end,
410                    data_section_end,
411                    floor,
412                )
413                .await?;
414                self.start = aligned_start;
415            }
416            self.filled = true;
417            self.advance_ramp(chunk_length);
418        }
419
420        // Post-refill invariant: `self.start <= start` (a straddle compaction
421        // lands exactly on `start`; a gap realign lands on the chunk BELOW it).
422        // Assert it rather than papering over a violation with a saturating
423        // subtraction, which would silently serve the WRONG bytes.
424        if start < self.start {
425            return Err(Error::corruption(format!(
426                "walk_in_range_partition_slices: coalesced compressed window starts at \
427                 {} but partition slice starts at {start} (issue #2877)",
428                self.start
429            )));
430        }
431        let lo = usize::try_from(start - self.start).unwrap_or(usize::MAX);
432        let hi = usize::try_from(end - self.start).unwrap_or(usize::MAX);
433        if hi > self.bytes.len() {
434            return Err(Error::corruption(format!(
435                "walk_in_range_partition_slices: coalesced compressed window short by \
436                 {} bytes (issue #2877)",
437                hi - self.bytes.len()
438            )));
439        }
440        Ok(&self.bytes[lo..hi])
441    }
442}

445mod tests {

448    /// The ramp must be overflow-safe and cap-bounded for EVERY
449    /// `CompressionInfo.chunk_length` a (possibly hostile) sidecar can encode and
450    /// every ramp value reachable by saturation — no panic, no wrap, never above
451    /// the cap (issue #2877 roborev, blocker A).
452    #[test]
453    fn ramped_floor_is_overflow_safe_and_capped() {
454        for chunk_length in [1u64, 4096, 16 * 1024, 64 * 1024, u32::MAX as u64] {
455            for ramp in [0u32, 1, 10, 22, 23, 63, 64, 200, u32::MAX] {
456                let w = CompressedScanWindow {
457                    bytes: Vec::new(),
458                    start: 0,
459                    filled: false,
460                    ramp,
461                };
462                let floor = w.ramped_floor(chunk_length);
463                assert!(
464                    floor <= COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES,
465                    "floor {floor} exceeded the cap for chunk_length {chunk_length} \
466                     at ramp {ramp}"
467                );
468                assert!(
469                    floor >= chunk_length.min(COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES),
470                    "floor {floor} fell below one chunk ({chunk_length}) at ramp {ramp}"
471                );
472            }
473        }
474    }

476    /// The floor DOUBLES per refill from one chunk up to the cap, then holds
477    /// steady — the ramp semantics an early-terminating scan and a long full scan
478    /// respectively rely on (issue #2877 roborev, blocker A).
479    #[test]
480    fn ramp_doubles_from_one_chunk_then_holds_at_cap() {
481        let chunk_length = 64 * 1024u64;
482        let mut w = CompressedScanWindow::new();
483        assert_eq!(
484            w.ramped_floor(chunk_length),
485            chunk_length,
486            "a fresh window's first refill must ask for ONE chunk, not the target"
487        );
488        let mut seen = Vec::new();
489        for _ in 0..12 {
490            seen.push(w.ramped_floor(chunk_length));
491            w.advance_ramp(chunk_length);
492        }
493        assert_eq!(
494            &seen[..7],
495            &[
496                64 * 1024,
497                128 * 1024,
498                256 * 1024,
499                512 * 1024,
500                1024 * 1024,
501                2048 * 1024,
502                COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES,
503            ],
504            "the floor must double per refill up to the cap: {seen:?}"
505        );
506        assert!(
507            seen[7..]
508                .iter()
509                .all(|f| *f == COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES),
510            "past the cap the floor must hold steady: {seen:?}"
511        );
512        // The COUNTER itself must stop, not merely the floor it computes: 64 KiB
513        // << 6 == 4 MiB == COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES, so `advance_ramp`
514        // takes exactly 6 steps and then becomes a no-op — even though the loop
515        // above called it 12 times. An upper-bound assert (`<= 32`) would pass with
516        // `advance_ramp`'s clamp guard DELETED (12 calls ⇒ ramp 12), pinning
517        // nothing; `assert_eq!` is what makes "the counter stops at the cap" a
518        // real claim.
519        const EXPECTED_RAMP_AT_CAP: u32 = 6;
520        assert_eq!(
521            w.ramp,
522            EXPECTED_RAMP_AT_CAP,
523            "the ramp COUNTER must stop advancing at the cap after exactly \
524             {EXPECTED_RAMP_AT_CAP} steps (chunk_length {chunk_length} << \
525             {EXPECTED_RAMP_AT_CAP} == {} == the cap \
526             {COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES}), got {}",
527            chunk_length << EXPECTED_RAMP_AT_CAP,
528            w.ramp
529        );
530    }

532    /// A chunk_length at or above the cap must clamp the ramped FLOOR to the cap
533    /// immediately and never advance the ramp (nothing to ramp toward).
534    ///
535    /// Note what this does NOT claim: the cap bounds the FLOOR, not the window. An
536    /// 8 MiB `chunk_length` (`CompressionInfo::parse` admits up to 256 MiB) still
537    /// yields an 8 MiB window, because every refill rounds UP to a `chunk_length`
538    /// boundary — asserted directly below so the distinction stays documented in
539    /// executable form.
540    #[test]
541    fn oversized_chunk_length_clamps_the_floor_immediately() {
542        let mut w = CompressedScanWindow::new();
543        let chunk_length = 8 * 1024 * 1024u64;
544        assert_eq!(
545            w.ramped_floor(chunk_length),
546            COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES,
547            "the FLOOR clamps at the cap"
548        );
549        w.advance_ramp(chunk_length);
550        assert_eq!(w.ramp, 0, "the ramp must not advance once the clamp binds");
551        // The resulting WINDOW is one whole chunk, i.e. ABOVE the cap: a refill's
552        // length is the floor rounded up to a chunk boundary.
553        let floor = w.ramped_floor(chunk_length);
554        let window_len = floor.div_ceil(chunk_length) * chunk_length;
555        assert_eq!(
556            window_len, chunk_length,
557            "an oversized chunk_length yields a window of one chunk ({chunk_length} \
558             bytes), which EXCEEDS the floor cap \
559             {COMPRESSED_SCAN_WINDOW_RAMP_MAX_BYTES} — the cap bounds read-AHEAD, \
560             not the window"
561        );
562    }

564    /// Compaction drops only from the FRONT: the chunk-aligned tail boundary a
565    /// straddle-append resumes from must be invariant (issue #2877 blocker B).
566    #[test]
567    fn compact_prefix_preserves_the_tail_boundary() {
568        let mut w = CompressedScanWindow {
569            bytes: (0u8..64).collect(),
570            start: 1000,
571            filled: true,
572            ramp: 0,
573        };
574        let tail_before = w.start + w.bytes.len() as u64;
575        w.compact_prefix(1040);
576        // There is NO separate logical-head field: reclamation is a per-refill
577        // `Vec::drain` of the dead prefix plus an advance of `self.start` (equally
578        // O(refills), one field fewer).
579        assert_eq!(
580            w.start, 1040,
581            "`self.start` must advance to the requested live-from offset"
582        );
583        assert_eq!(
584            w.start + w.bytes.len() as u64,
585            tail_before,
586            "the tail boundary must NOT move (a paid-for chunk must never be \
587             re-decompressed)"
588        );
589        assert_eq!(
590            w.bytes[0], 40,
591            "the surviving bytes must be the live suffix"
592        );
593        // Idempotent / no-op for a `live_from` at or below the current `start`.
594        // Asserted on
595        // this window's OWN state rather than on the process-global
596        // `scan_window_prefix_compaction_count()`: that counter is shared by every
597        // test in the lib binary, so snapshotting it here would go non-deterministic
598        // the moment any sibling test drives the compressed walk under cargo's
599        // parallel intra-binary threads (the recurring flake class of
600        // `big_locate_b4_repeat_zero_reprobe`). "Must not copy" IS local
601        // invariance — `bytes.len()` and `start` unchanged.
602        let len_before = w.bytes.len();
603        let start_before = w.start;
604        let first_byte_before = w.bytes[0];
605        w.compact_prefix(1000); // strictly below `start` ⇒ nothing to drop
606        w.compact_prefix(1040); // exactly `start` ⇒ nothing to drop
607        assert_eq!(
608            (w.bytes.len(), w.start, w.bytes[0]),
609            (len_before, start_before, first_byte_before),
610            "a no-op compaction must not copy: buffer length, start offset, and the \
611             first live byte must all be unchanged"
612        );
613    }
