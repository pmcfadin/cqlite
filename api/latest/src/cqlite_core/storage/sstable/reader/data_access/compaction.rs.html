1//! Compaction read paths: timestamp-preserving partition iteration, the bounded
2//! streaming compaction driver, and the partition-key enumeration the verifier's
3//! BTI cross-check relies on.
4//!
5//! Unlike the user-facing scans in `sequential`/`bti`, these paths PRESERVE
6//! `Value::Tombstone` entries (with their authoritative deletion timestamps) so
7//! the k-way merger can apply tombstone-shadowing semantics (Issue #505).
8
9use super::super::source::ScanCursor;
10use super::super::window_cursor::WindowCursor;
11use super::super::SSTableReader;
12use crate::storage::scan_cancel::ScanCancel;
13use crate::{Error, Result};
14use std::io::SeekFrom;
15use tokio::io::AsyncSeekExt;

17impl SSTableReader {
18    /// Stitch all compressed chunks and parse with per-row timestamps (for compaction).
19    ///
20    /// Identical to [`stitch_and_parse_all_chunks`] but delegates to
21    /// [`V5CompressedLegacyParser::parse_block_with_timestamps`] so that each
22    /// entry carries its actual row-level write timestamp rather than
23    /// `SystemTime::now()`.  Row and cell tombstones are emitted as
24    /// `Value::Tombstone` with their authoritative deletion timestamps.
25    ///
26    /// Used exclusively by the compaction k-way merger path (Issue #505).
27    ///
28    /// [`stitch_and_parse_all_chunks`]: crate::storage::sstable::SSTableReader
29    /// [`V5CompressedLegacyParser::parse_block_with_timestamps`]: crate::storage::sstable::reader::parsing::V5CompressedLegacyParser
30    async fn stitch_and_parse_all_chunks_for_compaction(
31        &self,
32        cursor: &ScanCursor,
33        schema: Option<&crate::schema::TableSchema>,
34    ) -> Result<Vec<super::super::compaction_row::CompactionRow>> {
35        tracing::debug!("stitch_and_parse_all_chunks_for_compaction: stitching chunks");
36
37        let mut stitched_buffer = Vec::with_capacity(2_500_000);
38        let mut chunk_count = 0;
39
40        // Incompressible-chunk fallback (Bug #639, epic #970, issue #1104):
41        // Cassandra stores a chunk RAW (not compressed) when its compressed length
42        // would meet or exceed `max_compressed_length`. `stitch_all_chunks` (the
43        // read/scan path) honours this, but the compaction stitch path did not —
44        // it blindly tried to LZ4/Snappy/etc-decode a raw chunk, which fails on the
45        // `incompressible` fixture with e.g. "the offset to copy is not contained
46        // in the decompressed buffer". Mirror the writer rule here: when the
47        // (CRC-stripped) chunk length >= max_compressed_length, the bytes are
48        // already plaintext. Authority: CompressedSequentialWriter.java:160-177.
49        let max_compressed_length = self
50            .compression_info
51            .as_ref()
52            .map(|ci| ci.max_compressed_length as usize)
53            .unwrap_or(usize::MAX);
54
55        while let Some(compressed_chunk) = self.read_next_block(cursor).await? {
56            use crate::storage::sstable::compression::Compression;
57            let decompressed_chunk = if compressed_chunk.len() >= max_compressed_length {
58                // Stored uncompressed by Cassandra — pass the raw bytes through.
59                tracing::debug!(
60                    "stitch_and_parse_all_chunks_for_compaction: chunk {} is incompressible (len={} >= max_compressed_length={}), using raw bytes",
61                    chunk_count,
62                    compressed_chunk.len(),
63                    max_compressed_length
64                );
65                compressed_chunk
66            } else if let Some(compression_reader) = &self.compression_reader {
67                let compression = Compression::new(*compression_reader.algorithm())?;
68                compression.decompress(&compressed_chunk).map_err(|e| {
69                    Error::corruption(format!(
70                        "stitch_and_parse_all_chunks_for_compaction: Failed to decompress chunk {}: {}",
71                        chunk_count, e
72                    ))
73                })?
74            } else {
75                compressed_chunk
76            };
77            stitched_buffer.extend_from_slice(&decompressed_chunk);
78            chunk_count += 1;
79        }
80
81        tracing::debug!(
82            "stitch_and_parse_all_chunks_for_compaction: {} chunks, {} bytes total",
83            chunk_count,
84            stitched_buffer.len()
85        );
86
87        let keyspace = self.header.keyspace.clone();
88        let table_name = self.header.table_name.clone();
89
90        let (min_timestamp, min_local_deletion_time, min_ttl) =
91            if let Some(stats_reader) = &self.statistics_reader {
92                let ts_stats = &stats_reader.statistics().timestamp_stats;
93                (
94                    ts_stats.min_timestamp,
95                    ts_stats.min_deletion_time,
96                    ts_stats.min_ttl,
97                )
98            } else {
99                (0, 0, None)
100            };
101
102        let parser = crate::storage::sstable::reader::parsing::V5CompressedLegacyParser::new(
103            keyspace,
104            table_name,
105            min_timestamp,
106            min_local_deletion_time,
107            min_ttl,
108        )
109        // VG1: thread VersionGates from SSTableReader down to row parser.
110        .with_version_gates(self.version_gates.clone());
111        let parser = if let Some(ref registry) = self.udt_registry {
112            parser.with_udt_registry(registry.clone())
113        } else {
114            parser
115        };
116
117        let reader_schema;
118        let table_schema = if let Some(s) = schema {
119            Some(s)
120        } else {
121            reader_schema = self.get_table_schema(None);
122            reader_schema.as_ref()
123        };
124
125        let entries = parser.parse_block_for_compaction(&stitched_buffer, table_schema, self)?;
126        tracing::debug!(
127            "stitch_and_parse_all_chunks_for_compaction: parsed {} entries",
128            entries.len()
129        );
130
131        Ok(entries)
132    }

134    /// Iterate all partitions with per-row timestamps, for use by the compaction merger.
135    ///
136    /// Returns `(RowKey, ScanRow, row_timestamp_micros)` for every row in the SSTable.
137    /// Unlike [`iterate_all_partitions`]:
138    ///
139    /// - Row tombstones are returned as `Value::Tombstone(RowTombstone)` carrying
140    ///   the actual deletion timestamp extracted from the on-disk row header.
141    /// - Cell tombstones within live rows are stored as `Value::Tombstone(CellTombstone)`
142    ///   inside the `Value::Map`, also carrying the actual cell-level deletion timestamp.
143    /// - The third tuple element is the decoded row-level write timestamp, so the
144    ///   merger can perform timestamp-accurate last-write-wins comparisons.
145    ///
146    /// Normal user-facing reads use [`scan`] / [`get`] / [`iterate_all_partitions`],
147    /// which apply tombstone filtering.  Do NOT use this method for user-visible queries.
148    ///
149    /// (Issue #505)
150    ///
151    /// [`iterate_all_partitions`]: crate::storage::sstable::SSTableReader
152    /// [`scan`]: crate::storage::sstable::SSTableReader::scan
153    /// [`get`]: crate::storage::sstable::SSTableReader::get
154    pub async fn iterate_all_partitions_for_compaction(
155        &self,
156        schema: Option<&crate::schema::TableSchema>,
157    ) -> Result<Vec<super::super::compaction_row::CompactionRow>> {
158        // Only the V5CompressedLegacy NB chunk-stitching path is supported here
159        // (that is the format the WriteEngine produces).  For other formats, fall
160        // back to iterate_all_partitions and attach timestamp 0 as a conservative
161        // default (LWW ordering then relies solely on run_index).
162        if self.requires_chunk_stitching() {
163            // We need schema; retrieve it once.
164            // `schema` is Option<&TableSchema>; clone it into an owned value so we
165            // can pass it to the async helper without borrow-checker issues.
166            let owned_schema = schema.cloned().or_else(|| self.get_table_schema(None));
167
168            // Reset chunk reader to start of data section (own per-scan cursor).
169            let cursor = self.new_scan_cursor().await?;
170            let header_size = self.calculate_header_size();
171            {
172                let mut file_guard = cursor.file.lock().await;
173                file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
174            }
175
176            let entries = self
177                .stitch_and_parse_all_chunks_for_compaction(&cursor, owned_schema.as_ref())
178                .await?;
179
180            return Ok(entries);
181        }
182
183        // Non-stitching fallback: use iterate_all_partitions and attach ts=0.
184        let entries = self.iterate_all_partitions().await?;
185        Ok(entries
186            .into_iter()
187            .map(|(key, value)| {
188                super::super::compaction_row::CompactionRow::from_legacy_value(key, value, 0)
189            })
190            .collect())
191    }

193    /// Count the distinct PARTITION keys decoded from `Data.db` (issue #970).
194    ///
195    /// This is a partition-granular count — one per partition, never per row —
196    /// used by the SSTable verifier's BTI cross-check (`verify.rs`). It must NOT
197    /// be confused with [`get_all_entries`], whose `RowKey`s carry
198    /// clustering/column/static suffixes and therefore over-count a multi-row
199    /// partition as many distinct keys.
200    ///
201    /// Both supported Data.db layouts surface the partition key (without any
202    /// clustering suffix) via the compaction read path:
203    ///
204    /// - BIG (`nb`, `V5CompressedLegacy` + `is_nb_format`):
205    ///   [`iterate_all_partitions_for_compaction`] emits one
206    ///   [`CompactionRow`](super::super::compaction_row::CompactionRow) per row, each
207    ///   carrying the partition key (`CompactionRow::key`). Distinct keys ==
208    ///   partition count.
209    /// - BTI (`da`): the compaction iterator's non-stitching fallback would route
210    ///   through [`iterate_all_partitions`], whose keys are row-granular. Instead
211    ///   we stitch the data section and run the compaction parser directly, which
212    ///   emits the same partition-key-only `CompactionRow::key`.
213    ///
214    /// No schema is required from the caller: the parser resolves it via the
215    /// reader's header/registry (`get_table_schema`).
216    ///
217    /// [`get_all_entries`]: crate::storage::sstable::SSTableReader::get_all_entries
218    /// [`iterate_all_partitions`]: crate::storage::sstable::SSTableReader
219    pub async fn distinct_partition_count(&self) -> Result<usize> {
220        Ok(self.distinct_partition_keys().await?.len())
221    }

223    /// Return the set of distinct **raw** partition keys decoded from `Data.db`,
224    /// one entry per partition (NOT per row), in first-seen order.
225    ///
226    /// Each key is the raw serialized partition key as stored on disk (e.g. the
227    /// 4-byte big-endian value for `pk int`, 16 bytes for a UUID) — the same form
228    /// accepted by
229    /// [`encode_partition_key_for_bti_trie`](crate::storage::sstable::bti::parser::encode_partition_key_for_bti_trie).
230    /// This is used by the verifier's BTI `Partitions.db` cross-check to compare
231    /// partition-key IDENTITY (issue #1103), not just partition count.
232    ///
233    /// The stitch/parse strategy mirrors [`Self::distinct_partition_count`]: we
234    /// route through `stitch_all_chunks` (not the generic
235    /// `iterate_all_partitions_for_compaction`) for BOTH BIG and BTI so the
236    /// incompressible/raw-chunk fallback is honoured (issue #970), and parse with
237    /// the compaction parser, which emits one `CompactionRow` per partition with
238    /// the partition key in `key` (partition-granular). No schema is required
239    /// from the caller: the parser resolves it via the reader's header/registry.
240    pub async fn distinct_partition_keys(&self) -> Result<Vec<Vec<u8>>> {
241        use std::collections::HashSet;
242
243        let cursor = self.new_scan_cursor().await?;
244        let header_size = self.calculate_header_size();
245        {
246            let mut file_guard = cursor.file.lock().await;
247            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
248        }
249        let whole = self.stitch_all_chunks(&cursor).await?;
250
251        let effective_schema = self.get_table_schema(None);
252        let parser = self.build_v5_parser(false);
253        let rows = parser.parse_block_for_compaction(&whole, effective_schema.as_ref(), self)?;
254
255        let mut seen: HashSet<Vec<u8>> = HashSet::new();
256        let mut keys: Vec<Vec<u8>> = Vec::new();
257        for r in &rows {
258            let k = r.key.as_bytes();
259            if seen.insert(k.to_vec()) {
260                keys.push(k.to_vec());
261            }
262        }
263        Ok(keys)
264    }

266    /// Return the distinct raw partition keys decoded from `Data.db` together with
267    /// the byte offset at which each partition begins in the DECOMPRESSED data
268    /// section (issue #1103).
269    ///
270    /// Each tuple is `(data_position, raw_partition_key)`, where `data_position`
271    /// is exactly the value a BTI `Partitions.db` leaf encodes as
272    /// [`BtiPartitionLocation::DataOffset`](crate::storage::sstable::bti::parser::BtiPartitionLocation::DataOffset)
273    /// (and the `data_position` recovered from a `RowsOffset` entry via
274    /// [`resolve_rows_db_entry`](crate::storage::sstable::bti::parser::resolve_rows_db_entry)).
275    /// The verifier's BTI cross-check resolves each leaf PAYLOAD back to its raw
276    /// partition key through this map so it catches a corruption that keeps the
277    /// emitted trie prefix but rewrites the payload to point at a DIFFERENT
278    /// partition (a same-count wrong-IDENTITY corruption the prefix-only compare
279    /// missed).
280    ///
281    /// The stitch/parse strategy mirrors [`Self::distinct_partition_keys`]; only
282    /// the parser entry point differs (it threads the partition-start offset).
283    pub async fn distinct_partition_keys_with_positions(&self) -> Result<Vec<(u64, Vec<u8>)>> {
284        use std::collections::HashSet;
285
286        let cursor = self.new_scan_cursor().await?;
287        let header_size = self.calculate_header_size();
288        {
289            let mut file_guard = cursor.file.lock().await;
290            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
291        }
292        let whole = self.stitch_all_chunks(&cursor).await?;
293
294        let effective_schema = self.get_table_schema(None);
295        let parser = self.build_v5_parser(false);
296
297        // `seen` dedups partition keys; the recorded position is the FIRST row's
298        // offset for a partition (a partition spans contiguous rows, so the first
299        // row's offset is the partition start). `result` preserves first-seen
300        // order and is the only place the position is read back.
301        let mut seen: HashSet<Vec<u8>> = HashSet::new();
302        let mut result: Vec<(u64, Vec<u8>)> = Vec::new();
303        parser.parse_block_for_compaction_emit_with_offset(
304            &whole,
305            effective_schema.as_ref(),
306            self,
307            |partition_start, row| {
308                let k = row.key.as_bytes().to_vec();
309                if seen.insert(k.clone()) {
310                    result.push((partition_start as u64, k));
311                }
312                Ok(std::ops::ControlFlow::Continue(()))
313            },
314        )?;
315
316        Ok(result)
317    }

319    /// Verifier-facing scan (issue #1282): return, in on-disk order, every
320    /// distinct partition's raw key together with its raw partition-level
321    /// `localDeletionTime` (when the partition carries a tombstone).
322    ///
323    /// Each element is `(raw_partition_key, partition_local_deletion_time)` where
324    /// the LDT is `Some(i32)` only for a DELETED partition (a live partition
325    /// carries the `DeletionTime.LIVE` sentinel and yields `None`). The `i32` is
326    /// exactly the value [`parse_partition_header_full`] decodes: for the legacy
327    /// signed `nb` form it is the genuine signed `i32 BE`; for the unsigned
328    /// `oa`/`da` form it is the wrapping `as u32 as i32` representation of the
329    /// on-disk `u32` (far-future values in `[2^31, 2^32)` therefore appear
330    /// negative and are LEGITIMATE — the caller must consult
331    /// [`SSTableReader::has_uint_deletion_time`] before interpreting a negative
332    /// value as corrupt).
333    ///
334    /// The ordering is the on-disk partition order, so the verifier can assert
335    /// ascending Murmur3 token order (Cassandra stores partitions in token order)
336    /// without a separate scan.
337    ///
338    /// [`parse_partition_header_full`]: crate::storage::sstable::reader::parsing::V5CompressedLegacyParser::parse_partition_header_full
339    pub async fn partition_verify_scan(&self) -> Result<Vec<(Vec<u8>, Option<i32>)>> {
340        use std::collections::HashSet;
341
342        let cursor = self.new_scan_cursor().await?;
343        let header_size = self.calculate_header_size();
344        {
345            let mut file_guard = cursor.file.lock().await;
346            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
347        }
348        let whole = self.stitch_all_chunks(&cursor).await?;
349
350        let effective_schema = self.get_table_schema(None);
351        let parser = self.build_v5_parser(false);
352
353        // First pass: recover the distinct partition-start offsets in on-disk
354        // order (a partition spans contiguous rows, so the first row's offset is
355        // the partition start). Reuses the same emit-with-offset parser the BTI
356        // identity cross-check relies on, so partition framing stays defined in
357        // exactly one place.
358        //
359        // Issue #1282 (roborev): dedup by partition BOUNDARY (the per-partition
360        // start offset), NOT by raw key. Deduping by key would silently DROP a
361        // duplicate partition key that appears again later on disk — precisely the
362        // corruption Cassandra's `Verifier` flags ("Key out of order" for a
363        // non-increasing `(token, key)` step). Keying on the boundary offset
364        // collapses the many rows WITHIN one partition (which all share that
365        // offset) to a single entry while letting each DISTINCT on-disk
366        // partition-start — including a duplicated key — reach the classifier.
367        let mut seen: HashSet<usize> = HashSet::new();
368        let mut starts: Vec<usize> = Vec::new();
369        parser.parse_block_for_compaction_emit_with_offset(
370            &whole,
371            effective_schema.as_ref(),
372            self,
373            |partition_start, _row| {
374                if seen.insert(partition_start) {
375                    starts.push(partition_start);
376                }
377                Ok(std::ops::ControlFlow::Continue(()))
378            },
379        )?;
380
381        // Second pass: for each partition start, decode the raw partition header
382        // (authoritative key + partition-level DeletionTime). The header decode is
383        // byte-exact (no wrapping for the signed `nb` form), so the verifier sees
384        // a genuine negative `nb` `localDeletionTime`.
385        let mut result: Vec<(Vec<u8>, Option<i32>)> = Vec::with_capacity(starts.len());
386        for start in starts {
387            let (row_key, _next, partition_deletion) =
388                parser.parse_partition_header_full(&whole, start)?;
389            let ldt = partition_deletion.map(|(_mfda, ldt)| ldt);
390            result.push((row_key.as_bytes().to_vec(), ldt));
391        }
392
393        Ok(result)
394    }

396    /// Verifier-facing scan (issue #1282, roborev follow-up): return, in on-disk
397    /// order, every partition's ordered list of decoded CLUSTERING-key tuples.
398    ///
399    /// Each element is `(partition_index, Vec<Vec<Value>>)`, where the inner
400    /// `Vec<Value>` is one clustering row's values in schema clustering order, and
401    /// the outer `Vec` holds those rows in the exact ORDER they appear on disk. The
402    /// verifier compares consecutive tuples within a partition using the
403    /// authoritative per-column [`crate::types::comparator::ComparatorType`] plus
404    /// each clustering column's ASC/DESC order, and flags a non-increasing step as
405    /// [`crate::storage::sstable::verify::VerifyErrorClass::OutOfOrderKeyOrRow`] —
406    /// the "row" half of that class (Cassandra's `Verifier` rejects out-of-order
407    /// clustering rows too).
408    ///
409    /// Rows carrying no clustering prefix (a static row, a partition-level
410    /// tombstone carrier, or an unclustered table's single row) are OMITTED from
411    /// the per-partition list: they do not participate in clustering order.
412    ///
413    /// Reuses the SAME on-disk-order emit-with-offset parser as
414    /// [`Self::partition_verify_scan`], so the clustering values come from the
415    /// authoritative schema-aware decode, not a heuristic. Uses only non-gated CQL
416    /// comparators (no `write-support` dependency). When the table has no clustering
417    /// columns the returned list is empty (nothing to order).
418    pub async fn partition_clustering_verify_scan(
419        &self,
420    ) -> Result<Vec<(usize, Vec<Vec<crate::types::Value>>)>> {
421        use crate::types::Value;
422        use std::collections::BTreeMap;
423
424        let cursor = self.new_scan_cursor().await?;
425        let header_size = self.calculate_header_size();
426        {
427            let mut file_guard = cursor.file.lock().await;
428            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
429        }
430        let whole = self.stitch_all_chunks(&cursor).await?;
431
432        let effective_schema = self.get_table_schema(None);
433
434        // A table with no clustering columns has no intra-partition row order to
435        // verify — return early (empty) without decoding rows.
436        let has_clustering = effective_schema
437            .as_ref()
438            .map(|s| !s.clustering_keys.is_empty())
439            .unwrap_or(false);
440        if !has_clustering {
441            return Ok(Vec::new());
442        }
443
444        let parser = self.build_v5_parser(false);
445
446        // Group clustering tuples by partition-start offset (partitions appear in
447        // ascending on-disk offset order; rows within a partition are emitted in
448        // on-disk order). A BTreeMap keyed by the boundary offset both dedups the
449        // partition boundary and preserves on-disk partition order.
450        let mut by_partition: BTreeMap<usize, Vec<Vec<Value>>> = BTreeMap::new();
451        let schema = effective_schema.as_ref();
452        parser.parse_block_for_compaction_emit_with_offset(
453            &whole,
454            schema,
455            self,
456            |partition_start, row| {
457                let bucket = by_partition.entry(partition_start).or_default();
458                // A partition with only non-clustering rows still records an
459                // (empty) bucket so its index stays stable.
460                if let Some(tuple) = Self::clustering_tuple_of_compaction_row(&row, schema) {
461                    bucket.push(tuple);
462                }
463                Ok(std::ops::ControlFlow::Continue(()))
464            },
465        )?;
466
467        Ok(by_partition.into_values().enumerate().collect())
468    }

470    /// Extract the clustering-key tuple of a compaction row in schema order, or
471    /// `None` when the row carries no clustering prefix (static row, partition-level
472    /// tombstone carrier). Used only by the verifier's clustering-order check.
473    ///
474    /// Clustering columns are surfaced on the compaction read path as pseudo
475    /// simple-cells (row tombstones carry them in their dedicated `clustering`
476    /// field). We rebuild the tuple positionally from the schema's clustering
477    /// columns so the comparison uses authoritative positions.
478    fn clustering_tuple_of_compaction_row(
479        row: &super::super::compaction_row::CompactionRow,
480        schema: Option<&crate::schema::TableSchema>,
481    ) -> Option<Vec<crate::types::Value>> {
482        use super::super::compaction_row::CompactionRowData;
483
484        let schema = schema?;
485        if schema.clustering_keys.is_empty() {
486            return None;
487        }
488
489        match &row.row_data {
490            CompactionRowData::Tombstone { clustering, .. } => {
491                if clustering.len() != schema.clustering_keys.len() {
492                    // A partial/absent clustering prefix is not a full point row
493                    // (Cassandra's row order is over full clustering keys); skip.
494                    None
495                } else {
496                    Some(clustering.iter().map(|(_, v)| v.clone()).collect())
497                }
498            }
499            CompactionRowData::Live { simple, .. } => {
500                // A clustering row surfaces every clustering column as a simple
501                // pseudo-cell. Rebuild the tuple in schema order; if not every
502                // clustering column is present the row has no full clustering
503                // prefix (e.g. a static row) and is omitted.
504                let mut values: Vec<crate::types::Value> =
505                    Vec::with_capacity(schema.clustering_keys.len());
506                for ck in &schema.clustering_keys {
507                    match simple.iter().find(|c| c.column == ck.name) {
508                        Some(cell) => values.push(cell.value.clone()),
509                        None => return None,
510                    }
511                }
512                Some(values)
513            }
514            // Range markers / partition deletes are not clustering point rows.
515            CompactionRowData::RangeMarker { .. } | CompactionRowData::PartitionDelete { .. } => {
516                None
517            }
518        }
519    }

521    /// Streaming compaction read (issue #827): yield `(RowKey, ScanRow, ts)`
522    /// entries via `emit` one partition at a time, so peak memory is bounded by
523    /// `max_partition_size + one_chunk` rather than by the total input size.
524    ///
525    /// This is the incremental counterpart of
526    /// [`iterate_all_partitions_for_compaction`], which fully materialises the
527    /// decompressed data section and parses every entry into a `Vec` before
528    /// returning. The k-way merge producer (`merge::producer_thread`) uses this
529    /// to forward entries into its bounded channel directly, so a source's
530    /// decompressed content is never fully resident.
531    ///
532    /// ## Sliding-window driver
533    ///
534    /// The V5CompressedLegacy chunk-stitching path keeps a sliding
535    /// [`WindowCursor`](super::super::window_cursor::WindowCursor) of decompressed
536    /// bytes. After refilling with each decompressed chunk it drains confirmed
537    /// partitions via `parse_one_partition_with_timestamps`, advancing the cursor
538    /// over the front after every `Emitted` (the reclaimed prefix is compacted once
539    /// per refill, not memmoved per partition — issue #1589), and stopping at
540    /// `NeedMore` to await the next chunk (a partition can straddle a chunk
541    /// boundary). At EOF a final drain pass runs with
542    /// `at_final_chunk = true` so the trailing (possibly truncated) partition is
543    /// terminal rather than requesting a refill that will never come.
544    ///
545    /// Returning `ControlFlow::Break` from `emit` stops the scan early
546    /// (consumer dropped). Tombstone / timestamp semantics are byte-identical to
547    /// the Vec variant (Issue #505/#533).
548    ///
549    /// `scan_cancel` is an explicit PER-CALL cancellation token (issue #2346),
550    /// not the reader's own [`SSTableReader::scan_cancel`] field: a cached/shared
551    /// `Arc<SSTableReader>` (a future warm-handle registry) may drive two
552    /// concurrent calls to this method with two INDEPENDENT tokens, so
553    /// cancellation cannot live as mutable per-reader state (`set_scan_cancel`
554    /// requires `&mut self`, uncallable through a shared `Arc`). Callers that
555    /// want the reader's own field's semantics (the pre-#2346 default) pass
556    /// `&self.scan_cancel` explicitly — see
557    /// [`SSTableReader::iterate_all_partitions_cancellable`] for the analogous
558    /// non-compaction seam.
559    pub async fn stream_all_partitions_for_compaction<F>(
560        &self,
561        schema: Option<&crate::schema::TableSchema>,
562        scan_cancel: &ScanCancel,
563        mut emit: F,
564    ) -> Result<()>
565    where
566        F: FnMut(super::super::compaction_row::CompactionRow) -> Result<std::ops::ControlFlow<()>>,
567    {
568        // Reset chunk reader to the start of the data section (mirrors
569        // iterate_all_partitions_for_compaction) using an own per-scan cursor.
570        let cursor = self.new_scan_cursor().await?;
571        let header_size = self.calculate_header_size();
572        {
573            let mut file_guard = cursor.file.lock().await;
574            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
575        }
576
577        // Non-stitching formats (uncompressed / non-nb BIG — the #2361 field
578        // case): TRUE-stream each partition as the index walk resolves it, rather
579        // than materialising the whole SSTable into one Vec before the first emit
580        // (issue #2361). This is what lets a downstream `LIMIT` early-break and a
581        // client-disconnect cancellation take effect promptly and keeps peak
582        // memory bounded on a 1M-partition table. No producer-side `limit`
583        // parameter (roborev round 2): a partition count is not a safe proxy for
584        // a row-level `LIMIT` (see `stream_all_partitions_cancellable`'s doc) —
585        // `LIMIT` is enforced purely downstream via the consumer's
586        // post-reconciliation break + cancel-aware Drop teardown. Issue #2346:
587        // the PER-CALL `scan_cancel` governs the whole walk.
588        if !self.requires_chunk_stitching() {
589            self.stream_all_partitions_cancellable(scan_cancel, |(key, value)| {
590                let row =
591                    super::super::compaction_row::CompactionRow::from_legacy_value(key, value, 0);
592                emit(row)
593            })
594            .await?;
595            return Ok(());
596        }
597
598        // Resolve the schema the parser needs (cells lack column names on disk).
599        let owned_schema = schema.cloned().or_else(|| self.get_table_schema(None));
600        let parser = self.build_v5_parser(false);
601
602        // Sliding window with a FRONT CURSOR (issue #1589): confirmed partitions are
603        // consumed by advancing the cursor, and the reclaimed prefix is compacted
604        // once per refill — not memmoved per partition as the old front-drain did.
605        let mut window = WindowCursor::new();
606        let mut broke = false;
607
608        use crate::storage::sstable::compression::Compression;
609
610        // Incompressible-chunk fallback (Bug #639, epic #970, issue #1104):
611        // Cassandra stores a chunk RAW when its compressed length would meet or
612        // exceed `max_compressed_length`. This is the path the real k-way merge
613        // producer streams through, so it must honour the rule exactly like
614        // `stitch_all_chunks`/`stitch_and_parse_all_chunks_for_compaction`: when
615        // the (CRC-stripped) chunk length >= max_compressed_length, the bytes are
616        // already plaintext. Authority: CompressedSequentialWriter.java:160-177.
617        let max_compressed_length = self
618            .compression_info
619            .as_ref()
620            .map(|ci| ci.max_compressed_length as usize)
621            .unwrap_or(usize::MAX);
622
623        let mut chunk_count = 0;
624        while let Some(compressed_chunk) = self.read_next_block(&cursor).await? {
625            // Cooperative cancellation (issue #2264, roborev round 3): a poll
626            // every 256 chunks catches the edge case `drain_compaction_window`'s
627            // per-PARTITION poll cannot — a single partition so wide it spans
628            // hundreds of chunks without ever completing (so the drain loop's
629            // counter never advances).
630            if chunk_count & 0xFF == 0 {
631                scan_cancel.check()?;
632            }
633            let decompressed_chunk = if compressed_chunk.len() >= max_compressed_length {
634                // Stored uncompressed by Cassandra — pass the raw bytes through.
635                tracing::debug!(
636                    "stream_all_partitions_for_compaction: chunk {} is incompressible (len={} >= max_compressed_length={}), using raw bytes",
637                    chunk_count,
638                    compressed_chunk.len(),
639                    max_compressed_length
640                );
641                compressed_chunk
642            } else if let Some(compression_reader) = &self.compression_reader {
643                let compression = Compression::new(*compression_reader.algorithm())?;
644                compression.decompress(&compressed_chunk).map_err(|e| {
645                    Error::corruption(format!(
646                        "stream_all_partitions_for_compaction: Failed to decompress chunk {}: {}",
647                        chunk_count, e
648                    ))
649                })?
650            } else {
651                compressed_chunk
652            };
653            // Refill the window: compact the already-consumed prefix ONCE
654            // (issue #1589), then append the freshly decompressed chunk.
655            window.refill(&decompressed_chunk);
656            chunk_count += 1;
657
658            // Not the final chunk yet: NeedMore means "await more bytes". Drain
659            // every confirmed partition from the front of the window.
660            self.drain_compaction_window(
661                &parser,
662                owned_schema.as_ref(),
663                &mut window,
664                false,
665                &mut emit,
666                &mut broke,
667                scan_cancel,
668            )?;
669            if broke {
670                return Ok(());
671            }
672        }
673
674        // EOF: final drain — a truncated/unterminated trailing partition is now
675        // terminal (Done), not a refill request.
676        if !broke {
677            self.drain_compaction_window(
678                &parser,
679                owned_schema.as_ref(),
680                &mut window,
681                true,
682                &mut emit,
683                &mut broke,
684                scan_cancel,
685            )?;
686        }
687
688        tracing::debug!(
689            "stream_all_partitions_for_compaction: drained {} chunks (final window {} bytes)",
690            chunk_count,
691            window.len()
692        );
693
694        Ok(())
695    }
696
697    /// Drain every confirmed partition from the front of the sliding `window`,
698    /// emitting each row via `emit` (issue #827). After each `Emitted` the
699    /// consumed prefix is removed so the window's peak size stays bounded by
700    /// `max_partition_size + one_chunk`. Stops at `NeedMore` / `Done` (await the
701    /// next chunk / genuine end) or when `emit` returns `Break` (sets `*broke`).
702    fn drain_compaction_window<F>(
703        &self,
704        parser: &crate::storage::sstable::reader::parsing::V5CompressedLegacyParser,
705        schema: Option<&crate::schema::TableSchema>,
706        window: &mut WindowCursor,
707        at_final_chunk: bool,
708        emit: &mut F,
709        broke: &mut bool,
710        scan_cancel: &ScanCancel,
711    ) -> Result<()>
712    where
713        F: FnMut(super::super::compaction_row::CompactionRow) -> Result<std::ops::ControlFlow<()>>,
714    {
715        use crate::storage::sstable::reader::parsing::ParseStep;
716        let mut drained: usize = 0;
717        loop {
718            if *broke || window.is_empty() {
719                return Ok(());
720            }
721            // Cooperative cancellation (issue #2264): for a chunk-stitched ('nb')
722            // SSTable this is the per-partition hot loop the compaction stream (and
723            // thus a Flight `do_get`) spends its time in. Poll the PER-CALL cancel
724            // token (issue #2346) at a bounded interval so a disconnected client
725            // abandons the walk within milliseconds instead of the coarse ~1–2 min
726            // backstop.
727            if drained & 0xFF == 0 {
728                scan_cancel.check()?;
729            }
730            drained += 1;
731            let mut local_break = false;
732            let step = parser.parse_one_partition_for_compaction(
733                window.as_slice(),
734                schema,
735                self,
736                at_final_chunk,
737                &mut |row: super::super::compaction_row::CompactionRow| match emit(row)? {
738                    std::ops::ControlFlow::Continue(()) => Ok(std::ops::ControlFlow::Continue(())),
739                    std::ops::ControlFlow::Break(()) => {
740                        local_break = true;
741                        Ok(std::ops::ControlFlow::Break(()))
742                    }
743                },
744            )?;
745            match step {
746                ParseStep::Emitted(consumed) => {
747                    let take = if consumed == 0 { 1 } else { consumed };
748                    // Advance the cursor over the confirmed partition (issue #1589):
749                    // NO memmove here — the reclaimed prefix is compacted once at the
750                    // next refill. `consume` clamps to the remaining length, matching
751                    // the prior `drain(0..take.min(window.len()))`.
752                    window.consume(take);
753                    if local_break {
754                        *broke = true;
755                        return Ok(());
756                    }
757                }
758                ParseStep::NeedMore | ParseStep::Done => return Ok(()),
759            }
760        }
761    }
