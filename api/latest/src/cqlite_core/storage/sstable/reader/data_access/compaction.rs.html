1//! Compaction read paths: timestamp-preserving partition iteration, the bounded
2//! streaming compaction driver, and the partition-key enumeration the verifier's
3//! BTI cross-check relies on.
4//!
5//! Unlike the user-facing scans in `sequential`/`bti`, these paths PRESERVE
6//! `Value::Tombstone` entries (with their authoritative deletion timestamps) so
7//! the k-way merger can apply tombstone-shadowing semantics (Issue #505).
8
9use super::super::source::ScanCursor;
10use super::super::window_cursor::WindowCursor;
11use super::super::SSTableReader;
12use crate::{Error, Result};
13use std::io::SeekFrom;
14use tokio::io::AsyncSeekExt;

16impl SSTableReader {
17    /// Stitch all compressed chunks and parse with per-row timestamps (for compaction).
18    ///
19    /// Identical to [`stitch_and_parse_all_chunks`] but delegates to
20    /// [`V5CompressedLegacyParser::parse_block_with_timestamps`] so that each
21    /// entry carries its actual row-level write timestamp rather than
22    /// `SystemTime::now()`.  Row and cell tombstones are emitted as
23    /// `Value::Tombstone` with their authoritative deletion timestamps.
24    ///
25    /// Used exclusively by the compaction k-way merger path (Issue #505).
26    ///
27    /// [`stitch_and_parse_all_chunks`]: crate::storage::sstable::SSTableReader
28    /// [`V5CompressedLegacyParser::parse_block_with_timestamps`]: crate::storage::sstable::reader::parsing::V5CompressedLegacyParser
29    async fn stitch_and_parse_all_chunks_for_compaction(
30        &self,
31        cursor: &ScanCursor,
32        schema: Option<&crate::schema::TableSchema>,
33    ) -> Result<Vec<super::super::compaction_row::CompactionRow>> {
34        log::debug!("stitch_and_parse_all_chunks_for_compaction: stitching chunks");
35
36        let mut stitched_buffer = Vec::with_capacity(2_500_000);
37        let mut chunk_count = 0;
38
39        // Incompressible-chunk fallback (Bug #639, epic #970, issue #1104):
40        // Cassandra stores a chunk RAW (not compressed) when its compressed length
41        // would meet or exceed `max_compressed_length`. `stitch_all_chunks` (the
42        // read/scan path) honours this, but the compaction stitch path did not —
43        // it blindly tried to LZ4/Snappy/etc-decode a raw chunk, which fails on the
44        // `incompressible` fixture with e.g. "the offset to copy is not contained
45        // in the decompressed buffer". Mirror the writer rule here: when the
46        // (CRC-stripped) chunk length >= max_compressed_length, the bytes are
47        // already plaintext. Authority: CompressedSequentialWriter.java:160-177.
48        let max_compressed_length = self
49            .compression_info
50            .as_ref()
51            .map(|ci| ci.max_compressed_length as usize)
52            .unwrap_or(usize::MAX);
53
54        while let Some(compressed_chunk) = self.read_next_block(cursor).await? {
55            use crate::storage::sstable::compression::Compression;
56            let decompressed_chunk = if compressed_chunk.len() >= max_compressed_length {
57                // Stored uncompressed by Cassandra — pass the raw bytes through.
58                log::debug!(
59                    "stitch_and_parse_all_chunks_for_compaction: chunk {} is incompressible (len={} >= max_compressed_length={}), using raw bytes",
60                    chunk_count,
61                    compressed_chunk.len(),
62                    max_compressed_length
63                );
64                compressed_chunk
65            } else if let Some(compression_reader) = &self.compression_reader {
66                let compression = Compression::new(*compression_reader.algorithm())?;
67                compression.decompress(&compressed_chunk).map_err(|e| {
68                    Error::corruption(format!(
69                        "stitch_and_parse_all_chunks_for_compaction: Failed to decompress chunk {}: {}",
70                        chunk_count, e
71                    ))
72                })?
73            } else {
74                compressed_chunk
75            };
76            stitched_buffer.extend_from_slice(&decompressed_chunk);
77            chunk_count += 1;
78        }
79
80        log::debug!(
81            "stitch_and_parse_all_chunks_for_compaction: {} chunks, {} bytes total",
82            chunk_count,
83            stitched_buffer.len()
84        );
85
86        let keyspace = self.header.keyspace.clone();
87        let table_name = self.header.table_name.clone();
88
89        let (min_timestamp, min_local_deletion_time, min_ttl) =
90            if let Some(stats_reader) = &self.statistics_reader {
91                let ts_stats = &stats_reader.statistics().timestamp_stats;
92                (
93                    ts_stats.min_timestamp,
94                    ts_stats.min_deletion_time,
95                    ts_stats.min_ttl,
96                )
97            } else {
98                (0, 0, None)
99            };
100
101        let parser = crate::storage::sstable::reader::parsing::V5CompressedLegacyParser::new(
102            keyspace,
103            table_name,
104            min_timestamp,
105            min_local_deletion_time,
106            min_ttl,
107        )
108        // VG1: thread VersionGates from SSTableReader down to row parser.
109        .with_version_gates(self.version_gates.clone());
110        let parser = if let Some(ref registry) = self.udt_registry {
111            parser.with_udt_registry(registry.clone())
112        } else {
113            parser
114        };
115
116        let reader_schema;
117        let table_schema = if let Some(s) = schema {
118            Some(s)
119        } else {
120            reader_schema = self.get_table_schema(None);
121            reader_schema.as_ref()
122        };
123
124        let entries = parser.parse_block_for_compaction(&stitched_buffer, table_schema, self)?;
125        log::debug!(
126            "stitch_and_parse_all_chunks_for_compaction: parsed {} entries",
127            entries.len()
128        );
129
130        Ok(entries)
131    }

133    /// Iterate all partitions with per-row timestamps, for use by the compaction merger.
134    ///
135    /// Returns `(RowKey, ScanRow, row_timestamp_micros)` for every row in the SSTable.
136    /// Unlike [`iterate_all_partitions`]:
137    ///
138    /// - Row tombstones are returned as `Value::Tombstone(RowTombstone)` carrying
139    ///   the actual deletion timestamp extracted from the on-disk row header.
140    /// - Cell tombstones within live rows are stored as `Value::Tombstone(CellTombstone)`
141    ///   inside the `Value::Map`, also carrying the actual cell-level deletion timestamp.
142    /// - The third tuple element is the decoded row-level write timestamp, so the
143    ///   merger can perform timestamp-accurate last-write-wins comparisons.
144    ///
145    /// Normal user-facing reads use [`scan`] / [`get`] / [`iterate_all_partitions`],
146    /// which apply tombstone filtering.  Do NOT use this method for user-visible queries.
147    ///
148    /// (Issue #505)
149    ///
150    /// [`iterate_all_partitions`]: crate::storage::sstable::SSTableReader
151    /// [`scan`]: crate::storage::sstable::SSTableReader::scan
152    /// [`get`]: crate::storage::sstable::SSTableReader::get
153    pub async fn iterate_all_partitions_for_compaction(
154        &self,
155        schema: Option<&crate::schema::TableSchema>,
156    ) -> Result<Vec<super::super::compaction_row::CompactionRow>> {
157        // Only the V5CompressedLegacy NB chunk-stitching path is supported here
158        // (that is the format the WriteEngine produces).  For other formats, fall
159        // back to iterate_all_partitions and attach timestamp 0 as a conservative
160        // default (LWW ordering then relies solely on run_index).
161        if self.requires_chunk_stitching() {
162            // We need schema; retrieve it once.
163            // `schema` is Option<&TableSchema>; clone it into an owned value so we
164            // can pass it to the async helper without borrow-checker issues.
165            let owned_schema = schema.cloned().or_else(|| self.get_table_schema(None));
166
167            // Reset chunk reader to start of data section (own per-scan cursor).
168            let cursor = self.new_scan_cursor().await?;
169            let header_size = self.calculate_header_size();
170            {
171                let mut file_guard = cursor.file.lock().await;
172                file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
173            }
174
175            let entries = self
176                .stitch_and_parse_all_chunks_for_compaction(&cursor, owned_schema.as_ref())
177                .await?;
178
179            return Ok(entries);
180        }
181
182        // Non-stitching fallback: use iterate_all_partitions and attach ts=0.
183        let entries = self.iterate_all_partitions().await?;
184        Ok(entries
185            .into_iter()
186            .map(|(key, value)| {
187                super::super::compaction_row::CompactionRow::from_legacy_value(key, value, 0)
188            })
189            .collect())
190    }

192    /// Count the distinct PARTITION keys decoded from `Data.db` (issue #970).
193    ///
194    /// This is a partition-granular count — one per partition, never per row —
195    /// used by the SSTable verifier's BTI cross-check (`verify.rs`). It must NOT
196    /// be confused with [`get_all_entries`], whose `RowKey`s carry
197    /// clustering/column/static suffixes and therefore over-count a multi-row
198    /// partition as many distinct keys.
199    ///
200    /// Both supported Data.db layouts surface the partition key (without any
201    /// clustering suffix) via the compaction read path:
202    ///
203    /// - BIG (`nb`, `V5CompressedLegacy` + `is_nb_format`):
204    ///   [`iterate_all_partitions_for_compaction`] emits one
205    ///   [`CompactionRow`](super::super::compaction_row::CompactionRow) per row, each
206    ///   carrying the partition key (`CompactionRow::key`). Distinct keys ==
207    ///   partition count.
208    /// - BTI (`da`): the compaction iterator's non-stitching fallback would route
209    ///   through [`iterate_all_partitions`], whose keys are row-granular. Instead
210    ///   we stitch the data section and run the compaction parser directly, which
211    ///   emits the same partition-key-only `CompactionRow::key`.
212    ///
213    /// No schema is required from the caller: the parser resolves it via the
214    /// reader's header/registry (`get_table_schema`).
215    ///
216    /// [`get_all_entries`]: crate::storage::sstable::SSTableReader::get_all_entries
217    /// [`iterate_all_partitions`]: crate::storage::sstable::SSTableReader
218    pub async fn distinct_partition_count(&self) -> Result<usize> {
219        Ok(self.distinct_partition_keys().await?.len())
220    }

222    /// Return the set of distinct **raw** partition keys decoded from `Data.db`,
223    /// one entry per partition (NOT per row), in first-seen order.
224    ///
225    /// Each key is the raw serialized partition key as stored on disk (e.g. the
226    /// 4-byte big-endian value for `pk int`, 16 bytes for a UUID) — the same form
227    /// accepted by
228    /// [`encode_partition_key_for_bti_trie`](crate::storage::sstable::bti::parser::encode_partition_key_for_bti_trie).
229    /// This is used by the verifier's BTI `Partitions.db` cross-check to compare
230    /// partition-key IDENTITY (issue #1103), not just partition count.
231    ///
232    /// The stitch/parse strategy mirrors [`Self::distinct_partition_count`]: we
233    /// route through `stitch_all_chunks` (not the generic
234    /// `iterate_all_partitions_for_compaction`) for BOTH BIG and BTI so the
235    /// incompressible/raw-chunk fallback is honoured (issue #970), and parse with
236    /// the compaction parser, which emits one `CompactionRow` per partition with
237    /// the partition key in `key` (partition-granular). No schema is required
238    /// from the caller: the parser resolves it via the reader's header/registry.
239    pub async fn distinct_partition_keys(&self) -> Result<Vec<Vec<u8>>> {
240        use std::collections::HashSet;
241
242        let cursor = self.new_scan_cursor().await?;
243        let header_size = self.calculate_header_size();
244        {
245            let mut file_guard = cursor.file.lock().await;
246            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
247        }
248        let whole = self.stitch_all_chunks(&cursor).await?;
249
250        let effective_schema = self.get_table_schema(None);
251        let parser = self.build_v5_parser(false);
252        let rows = parser.parse_block_for_compaction(&whole, effective_schema.as_ref(), self)?;
253
254        let mut seen: HashSet<Vec<u8>> = HashSet::new();
255        let mut keys: Vec<Vec<u8>> = Vec::new();
256        for r in &rows {
257            let k = r.key.as_bytes();
258            if seen.insert(k.to_vec()) {
259                keys.push(k.to_vec());
260            }
261        }
262        Ok(keys)
263    }

265    /// Return the distinct raw partition keys decoded from `Data.db` together with
266    /// the byte offset at which each partition begins in the DECOMPRESSED data
267    /// section (issue #1103).
268    ///
269    /// Each tuple is `(data_position, raw_partition_key)`, where `data_position`
270    /// is exactly the value a BTI `Partitions.db` leaf encodes as
271    /// [`BtiPartitionLocation::DataOffset`](crate::storage::sstable::bti::parser::BtiPartitionLocation::DataOffset)
272    /// (and the `data_position` recovered from a `RowsOffset` entry via
273    /// [`resolve_rows_db_entry`](crate::storage::sstable::bti::parser::resolve_rows_db_entry)).
274    /// The verifier's BTI cross-check resolves each leaf PAYLOAD back to its raw
275    /// partition key through this map so it catches a corruption that keeps the
276    /// emitted trie prefix but rewrites the payload to point at a DIFFERENT
277    /// partition (a same-count wrong-IDENTITY corruption the prefix-only compare
278    /// missed).
279    ///
280    /// The stitch/parse strategy mirrors [`Self::distinct_partition_keys`]; only
281    /// the parser entry point differs (it threads the partition-start offset).
282    pub async fn distinct_partition_keys_with_positions(&self) -> Result<Vec<(u64, Vec<u8>)>> {
283        use std::collections::HashSet;
284
285        let cursor = self.new_scan_cursor().await?;
286        let header_size = self.calculate_header_size();
287        {
288            let mut file_guard = cursor.file.lock().await;
289            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
290        }
291        let whole = self.stitch_all_chunks(&cursor).await?;
292
293        let effective_schema = self.get_table_schema(None);
294        let parser = self.build_v5_parser(false);
295
296        // `seen` dedups partition keys; the recorded position is the FIRST row's
297        // offset for a partition (a partition spans contiguous rows, so the first
298        // row's offset is the partition start). `result` preserves first-seen
299        // order and is the only place the position is read back.
300        let mut seen: HashSet<Vec<u8>> = HashSet::new();
301        let mut result: Vec<(u64, Vec<u8>)> = Vec::new();
302        parser.parse_block_for_compaction_emit_with_offset(
303            &whole,
304            effective_schema.as_ref(),
305            self,
306            |partition_start, row| {
307                let k = row.key.as_bytes().to_vec();
308                if seen.insert(k.clone()) {
309                    result.push((partition_start as u64, k));
310                }
311                Ok(std::ops::ControlFlow::Continue(()))
312            },
313        )?;
314
315        Ok(result)
316    }

318    /// Verifier-facing scan (issue #1282): return, in on-disk order, every
319    /// distinct partition's raw key together with its raw partition-level
320    /// `localDeletionTime` (when the partition carries a tombstone).
321    ///
322    /// Each element is `(raw_partition_key, partition_local_deletion_time)` where
323    /// the LDT is `Some(i32)` only for a DELETED partition (a live partition
324    /// carries the `DeletionTime.LIVE` sentinel and yields `None`). The `i32` is
325    /// exactly the value [`parse_partition_header_full`] decodes: for the legacy
326    /// signed `nb` form it is the genuine signed `i32 BE`; for the unsigned
327    /// `oa`/`da` form it is the wrapping `as u32 as i32` representation of the
328    /// on-disk `u32` (far-future values in `[2^31, 2^32)` therefore appear
329    /// negative and are LEGITIMATE — the caller must consult
330    /// [`SSTableReader::has_uint_deletion_time`] before interpreting a negative
331    /// value as corrupt).
332    ///
333    /// The ordering is the on-disk partition order, so the verifier can assert
334    /// ascending Murmur3 token order (Cassandra stores partitions in token order)
335    /// without a separate scan.
336    ///
337    /// [`parse_partition_header_full`]: crate::storage::sstable::reader::parsing::V5CompressedLegacyParser::parse_partition_header_full
338    pub async fn partition_verify_scan(&self) -> Result<Vec<(Vec<u8>, Option<i32>)>> {
339        use std::collections::HashSet;
340
341        let cursor = self.new_scan_cursor().await?;
342        let header_size = self.calculate_header_size();
343        {
344            let mut file_guard = cursor.file.lock().await;
345            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
346        }
347        let whole = self.stitch_all_chunks(&cursor).await?;
348
349        let effective_schema = self.get_table_schema(None);
350        let parser = self.build_v5_parser(false);
351
352        // First pass: recover the distinct partition-start offsets in on-disk
353        // order (a partition spans contiguous rows, so the first row's offset is
354        // the partition start). Reuses the same emit-with-offset parser the BTI
355        // identity cross-check relies on, so partition framing stays defined in
356        // exactly one place.
357        //
358        // Issue #1282 (roborev): dedup by partition BOUNDARY (the per-partition
359        // start offset), NOT by raw key. Deduping by key would silently DROP a
360        // duplicate partition key that appears again later on disk — precisely the
361        // corruption Cassandra's `Verifier` flags ("Key out of order" for a
362        // non-increasing `(token, key)` step). Keying on the boundary offset
363        // collapses the many rows WITHIN one partition (which all share that
364        // offset) to a single entry while letting each DISTINCT on-disk
365        // partition-start — including a duplicated key — reach the classifier.
366        let mut seen: HashSet<usize> = HashSet::new();
367        let mut starts: Vec<usize> = Vec::new();
368        parser.parse_block_for_compaction_emit_with_offset(
369            &whole,
370            effective_schema.as_ref(),
371            self,
372            |partition_start, _row| {
373                if seen.insert(partition_start) {
374                    starts.push(partition_start);
375                }
376                Ok(std::ops::ControlFlow::Continue(()))
377            },
378        )?;
379
380        // Second pass: for each partition start, decode the raw partition header
381        // (authoritative key + partition-level DeletionTime). The header decode is
382        // byte-exact (no wrapping for the signed `nb` form), so the verifier sees
383        // a genuine negative `nb` `localDeletionTime`.
384        let mut result: Vec<(Vec<u8>, Option<i32>)> = Vec::with_capacity(starts.len());
385        for start in starts {
386            let (row_key, _next, partition_deletion) =
387                parser.parse_partition_header_full(&whole, start)?;
388            let ldt = partition_deletion.map(|(_mfda, ldt)| ldt);
389            result.push((row_key.as_bytes().to_vec(), ldt));
390        }
391
392        Ok(result)
393    }

395    /// Verifier-facing scan (issue #1282, roborev follow-up): return, in on-disk
396    /// order, every partition's ordered list of decoded CLUSTERING-key tuples.
397    ///
398    /// Each element is `(partition_index, Vec<Vec<Value>>)`, where the inner
399    /// `Vec<Value>` is one clustering row's values in schema clustering order, and
400    /// the outer `Vec` holds those rows in the exact ORDER they appear on disk. The
401    /// verifier compares consecutive tuples within a partition using the
402    /// authoritative per-column [`crate::types::comparator::ComparatorType`] plus
403    /// each clustering column's ASC/DESC order, and flags a non-increasing step as
404    /// [`crate::storage::sstable::verify::VerifyErrorClass::OutOfOrderKeyOrRow`] —
405    /// the "row" half of that class (Cassandra's `Verifier` rejects out-of-order
406    /// clustering rows too).
407    ///
408    /// Rows carrying no clustering prefix (a static row, a partition-level
409    /// tombstone carrier, or an unclustered table's single row) are OMITTED from
410    /// the per-partition list: they do not participate in clustering order.
411    ///
412    /// Reuses the SAME on-disk-order emit-with-offset parser as
413    /// [`Self::partition_verify_scan`], so the clustering values come from the
414    /// authoritative schema-aware decode, not a heuristic. Uses only non-gated CQL
415    /// comparators (no `write-support` dependency). When the table has no clustering
416    /// columns the returned list is empty (nothing to order).
417    pub async fn partition_clustering_verify_scan(
418        &self,
419    ) -> Result<Vec<(usize, Vec<Vec<crate::types::Value>>)>> {
420        use crate::types::Value;
421        use std::collections::BTreeMap;
422
423        let cursor = self.new_scan_cursor().await?;
424        let header_size = self.calculate_header_size();
425        {
426            let mut file_guard = cursor.file.lock().await;
427            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
428        }
429        let whole = self.stitch_all_chunks(&cursor).await?;
430
431        let effective_schema = self.get_table_schema(None);
432
433        // A table with no clustering columns has no intra-partition row order to
434        // verify — return early (empty) without decoding rows.
435        let has_clustering = effective_schema
436            .as_ref()
437            .map(|s| !s.clustering_keys.is_empty())
438            .unwrap_or(false);
439        if !has_clustering {
440            return Ok(Vec::new());
441        }
442
443        let parser = self.build_v5_parser(false);
444
445        // Group clustering tuples by partition-start offset (partitions appear in
446        // ascending on-disk offset order; rows within a partition are emitted in
447        // on-disk order). A BTreeMap keyed by the boundary offset both dedups the
448        // partition boundary and preserves on-disk partition order.
449        let mut by_partition: BTreeMap<usize, Vec<Vec<Value>>> = BTreeMap::new();
450        let schema = effective_schema.as_ref();
451        parser.parse_block_for_compaction_emit_with_offset(
452            &whole,
453            schema,
454            self,
455            |partition_start, row| {
456                let bucket = by_partition.entry(partition_start).or_default();
457                // A partition with only non-clustering rows still records an
458                // (empty) bucket so its index stays stable.
459                if let Some(tuple) = Self::clustering_tuple_of_compaction_row(&row, schema) {
460                    bucket.push(tuple);
461                }
462                Ok(std::ops::ControlFlow::Continue(()))
463            },
464        )?;
465
466        Ok(by_partition.into_values().enumerate().collect())
467    }

469    /// Extract the clustering-key tuple of a compaction row in schema order, or
470    /// `None` when the row carries no clustering prefix (static row, partition-level
471    /// tombstone carrier). Used only by the verifier's clustering-order check.
472    ///
473    /// Clustering columns are surfaced on the compaction read path as pseudo
474    /// simple-cells (row tombstones carry them in their dedicated `clustering`
475    /// field). We rebuild the tuple positionally from the schema's clustering
476    /// columns so the comparison uses authoritative positions.
477    fn clustering_tuple_of_compaction_row(
478        row: &super::super::compaction_row::CompactionRow,
479        schema: Option<&crate::schema::TableSchema>,
480    ) -> Option<Vec<crate::types::Value>> {
481        use super::super::compaction_row::CompactionRowData;
482
483        let schema = schema?;
484        if schema.clustering_keys.is_empty() {
485            return None;
486        }
487
488        match &row.row_data {
489            CompactionRowData::Tombstone { clustering, .. } => {
490                if clustering.len() != schema.clustering_keys.len() {
491                    // A partial/absent clustering prefix is not a full point row
492                    // (Cassandra's row order is over full clustering keys); skip.
493                    None
494                } else {
495                    Some(clustering.iter().map(|(_, v)| v.clone()).collect())
496                }
497            }
498            CompactionRowData::Live { simple, .. } => {
499                // A clustering row surfaces every clustering column as a simple
500                // pseudo-cell. Rebuild the tuple in schema order; if not every
501                // clustering column is present the row has no full clustering
502                // prefix (e.g. a static row) and is omitted.
503                let mut values: Vec<crate::types::Value> =
504                    Vec::with_capacity(schema.clustering_keys.len());
505                for ck in &schema.clustering_keys {
506                    match simple.iter().find(|c| c.column == ck.name) {
507                        Some(cell) => values.push(cell.value.clone()),
508                        None => return None,
509                    }
510                }
511                Some(values)
512            }
513            // Range markers / partition deletes are not clustering point rows.
514            CompactionRowData::RangeMarker { .. } | CompactionRowData::PartitionDelete { .. } => {
515                None
516            }
517        }
518    }

520    /// Streaming compaction read (issue #827): yield `(RowKey, ScanRow, ts)`
521    /// entries via `emit` one partition at a time, so peak memory is bounded by
522    /// `max_partition_size + one_chunk` rather than by the total input size.
523    ///
524    /// This is the incremental counterpart of
525    /// [`iterate_all_partitions_for_compaction`], which fully materialises the
526    /// decompressed data section and parses every entry into a `Vec` before
527    /// returning. The k-way merge producer (`merge::producer_thread`) uses this
528    /// to forward entries into its bounded channel directly, so a source's
529    /// decompressed content is never fully resident.
530    ///
531    /// ## Sliding-window driver
532    ///
533    /// The V5CompressedLegacy chunk-stitching path keeps a sliding
534    /// [`WindowCursor`](super::super::window_cursor::WindowCursor) of decompressed
535    /// bytes. After refilling with each decompressed chunk it drains confirmed
536    /// partitions via `parse_one_partition_with_timestamps`, advancing the cursor
537    /// over the front after every `Emitted` (the reclaimed prefix is compacted once
538    /// per refill, not memmoved per partition — issue #1589), and stopping at
539    /// `NeedMore` to await the next chunk (a partition can straddle a chunk
540    /// boundary). At EOF a final drain pass runs with
541    /// `at_final_chunk = true` so the trailing (possibly truncated) partition is
542    /// terminal rather than requesting a refill that will never come.
543    ///
544    /// Returning `ControlFlow::Break` from `emit` stops the scan early
545    /// (consumer dropped). Tombstone / timestamp semantics are byte-identical to
546    /// the Vec variant (Issue #505/#533).
547    pub async fn stream_all_partitions_for_compaction<F>(
548        &self,
549        schema: Option<&crate::schema::TableSchema>,
550        mut emit: F,
551    ) -> Result<()>
552    where
553        F: FnMut(super::super::compaction_row::CompactionRow) -> Result<std::ops::ControlFlow<()>>,
554    {
555        // Reset chunk reader to the start of the data section (mirrors
556        // iterate_all_partitions_for_compaction) using an own per-scan cursor.
557        let cursor = self.new_scan_cursor().await?;
558        let header_size = self.calculate_header_size();
559        {
560            let mut file_guard = cursor.file.lock().await;
561            file_guard.seek(SeekFrom::Start(header_size as u64)).await?;
562        }
563
564        // Non-stitching formats are single-block / small: emit via the
565        // materialising iterator with ts=0 (matches the Vec-variant fallback).
566        if !self.requires_chunk_stitching() {
567            let entries = self.iterate_all_partitions().await?;
568            for (key, value) in entries {
569                let row =
570                    super::super::compaction_row::CompactionRow::from_legacy_value(key, value, 0);
571                match emit(row)? {
572                    std::ops::ControlFlow::Continue(()) => {}
573                    std::ops::ControlFlow::Break(()) => return Ok(()),
574                }
575            }
576            return Ok(());
577        }
578
579        // Resolve the schema the parser needs (cells lack column names on disk).
580        let owned_schema = schema.cloned().or_else(|| self.get_table_schema(None));
581        let parser = self.build_v5_parser(false);
582
583        // Sliding window with a FRONT CURSOR (issue #1589): confirmed partitions are
584        // consumed by advancing the cursor, and the reclaimed prefix is compacted
585        // once per refill — not memmoved per partition as the old front-drain did.
586        let mut window = WindowCursor::new();
587        let mut broke = false;
588
589        use crate::storage::sstable::compression::Compression;
590
591        // Incompressible-chunk fallback (Bug #639, epic #970, issue #1104):
592        // Cassandra stores a chunk RAW when its compressed length would meet or
593        // exceed `max_compressed_length`. This is the path the real k-way merge
594        // producer streams through, so it must honour the rule exactly like
595        // `stitch_all_chunks`/`stitch_and_parse_all_chunks_for_compaction`: when
596        // the (CRC-stripped) chunk length >= max_compressed_length, the bytes are
597        // already plaintext. Authority: CompressedSequentialWriter.java:160-177.
598        let max_compressed_length = self
599            .compression_info
600            .as_ref()
601            .map(|ci| ci.max_compressed_length as usize)
602            .unwrap_or(usize::MAX);
603
604        let mut chunk_count = 0;
605        while let Some(compressed_chunk) = self.read_next_block(&cursor).await? {
606            let decompressed_chunk = if compressed_chunk.len() >= max_compressed_length {
607                // Stored uncompressed by Cassandra — pass the raw bytes through.
608                log::debug!(
609                    "stream_all_partitions_for_compaction: chunk {} is incompressible (len={} >= max_compressed_length={}), using raw bytes",
610                    chunk_count,
611                    compressed_chunk.len(),
612                    max_compressed_length
613                );
614                compressed_chunk
615            } else if let Some(compression_reader) = &self.compression_reader {
616                let compression = Compression::new(*compression_reader.algorithm())?;
617                compression.decompress(&compressed_chunk).map_err(|e| {
618                    Error::corruption(format!(
619                        "stream_all_partitions_for_compaction: Failed to decompress chunk {}: {}",
620                        chunk_count, e
621                    ))
622                })?
623            } else {
624                compressed_chunk
625            };
626            // Refill the window: compact the already-consumed prefix ONCE
627            // (issue #1589), then append the freshly decompressed chunk.
628            window.refill(&decompressed_chunk);
629            chunk_count += 1;
630
631            // Not the final chunk yet: NeedMore means "await more bytes". Drain
632            // every confirmed partition from the front of the window.
633            self.drain_compaction_window(
634                &parser,
635                owned_schema.as_ref(),
636                &mut window,
637                false,
638                &mut emit,
639                &mut broke,
640            )?;
641            if broke {
642                return Ok(());
643            }
644        }
645
646        // EOF: final drain — a truncated/unterminated trailing partition is now
647        // terminal (Done), not a refill request.
648        if !broke {
649            self.drain_compaction_window(
650                &parser,
651                owned_schema.as_ref(),
652                &mut window,
653                true,
654                &mut emit,
655                &mut broke,
656            )?;
657        }
658
659        log::debug!(
660            "stream_all_partitions_for_compaction: drained {} chunks (final window {} bytes)",
661            chunk_count,
662            window.len()
663        );
664
665        Ok(())
666    }
667
668    /// Drain every confirmed partition from the front of the sliding `window`,
669    /// emitting each row via `emit` (issue #827). After each `Emitted` the
670    /// consumed prefix is removed so the window's peak size stays bounded by
671    /// `max_partition_size + one_chunk`. Stops at `NeedMore` / `Done` (await the
672    /// next chunk / genuine end) or when `emit` returns `Break` (sets `*broke`).
673    fn drain_compaction_window<F>(
674        &self,
675        parser: &crate::storage::sstable::reader::parsing::V5CompressedLegacyParser,
676        schema: Option<&crate::schema::TableSchema>,
677        window: &mut WindowCursor,
678        at_final_chunk: bool,
679        emit: &mut F,
680        broke: &mut bool,
681    ) -> Result<()>
682    where
683        F: FnMut(super::super::compaction_row::CompactionRow) -> Result<std::ops::ControlFlow<()>>,
684    {
685        use crate::storage::sstable::reader::parsing::ParseStep;
686        loop {
687            if *broke || window.is_empty() {
688                return Ok(());
689            }
690            let mut local_break = false;
691            let step = parser.parse_one_partition_for_compaction(
692                window.as_slice(),
693                schema,
694                self,
695                at_final_chunk,
696                &mut |row: super::super::compaction_row::CompactionRow| match emit(row)? {
697                    std::ops::ControlFlow::Continue(()) => Ok(std::ops::ControlFlow::Continue(())),
698                    std::ops::ControlFlow::Break(()) => {
699                        local_break = true;
700                        Ok(std::ops::ControlFlow::Break(()))
701                    }
702                },
703            )?;
704            match step {
705                ParseStep::Emitted(consumed) => {
706                    let take = if consumed == 0 { 1 } else { consumed };
707                    // Advance the cursor over the confirmed partition (issue #1589):
708                    // NO memmove here — the reclaimed prefix is compacted once at the
709                    // next refill. `consume` clamps to the remaining length, matching
710                    // the prior `drain(0..take.min(window.len()))`.
711                    window.consume(take);
712                    if local_break {
713                        *broke = true;
714                        return Ok(());
715                    }
716                }
717                ParseStep::NeedMore | ParseStep::Done => return Ok(()),
718            }
719        }
720    }
