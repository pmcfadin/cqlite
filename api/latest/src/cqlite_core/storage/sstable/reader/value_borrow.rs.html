1//! Active-window decode borrow source (issue #1644, K5 stage 2, design D1).
2//!
3//! [`WindowCursor::borrow`]/[`borrow_subslice`](super::window_cursor::WindowCursor)
4//! need a handle to the CURRENT window to decide borrow-vs-copy, but the
5//! scalar-value decode call graph (`row_decoder/raw_value.rs`,
6//! `row_decoder/cell_value.rs`, `parsing/raw_type_value.rs`,
7//! `parsing/complex_column.rs`, `row_decoder/udt.rs`, and the comparator family)
8//! is hundreds of `&[u8]`-in/`&[u8]`-out nom-style combinators several call
9//! levels below the one place ([`super::scan_stream_windowed`]'s
10//! `drain_scan_window`) that owns the [`WindowCursor`]. Design D1 explicitly
11//! rejects threading `(&Bytes, offset)` provenance through every one of those
12//! signatures — "hundreds of decode signatures ... the copy fallback is
13//! unavoidable regardless" — in favor of localizing the decision to one place.
14//!
15//! This module is THE localization mechanism: an [`ActiveWindowGuard`] installs
16//! the current window's `Bytes` backing (when [`Backing::Borrowed`]) in a
17//! `thread_local!` for the duration of exactly one
18//! `parse_one_partition_with_timestamps` call, and [`borrow_active`] lets any
19//! leaf decode site — WITHOUT any new parameter on its own signature — turn a
20//! `&[u8]` it already holds (guaranteed, by ordinary slicing, to be a subslice
21//! of the window's backing whenever a window is active) into a zero-copy
22//! [`Bytes`] view via [`WindowCursor::borrow_subslice`]'s pointer-range check.
23//!
24//! # Why a thread-local is safe here
25//! The entire windowed-scan parse (`drain_scan_window_blocking`) runs
26//! SYNCHRONOUSLY on a single dedicated `spawn_blocking` thread: there is no
27//! `.await` between installing the guard and dropping it, and no other task
28//! interleaves partition-parsing work on that thread while the guard is live.
29//! So "thread-local for the scope of one synchronous call" behaves exactly like
30//! an implicit extra parameter threaded through every frame — WITHOUT touching
31//! any of those frames' signatures. The RAII guard restores the previous value
32//! on drop (supporting nested/sequential partition parses safely), and every
33//! consumer degrades to a SAFE (never memory-unsafe, never silently wrong —
34//! only less zero-copy) owned copy when no window is active: the compaction
35//! driver, `get()`/point-lookup decode, and any other non-windowed caller never
36//! installs a guard, so [`borrow_active`] always copies for them, unchanged
37//! from pre-#1644 behavior.
38
39use super::window_cursor::WindowCursor;
40use bytes::Bytes;
41use std::cell::RefCell;
42
43thread_local! {
44    /// A SNAPSHOT `WindowCursor` over the current streaming-scan window's
45    /// active `Bytes` backing (via [`WindowCursor::from_borrowed_bytes`]),
46    /// installed for the duration of ONE partition parse by
47    /// [`ActiveWindowGuard`]. `None` when no window is active (compaction /
48    /// point-lookup / any non-windowed decode path) OR when the real window is
49    /// currently STITCHED (a straddling partition — no single parent `Bytes`
50    /// exists, so every decode call naturally falls back to a copy via
51    /// [`WindowCursor::borrow_subslice`]'s pointer-range check, per D1's
52    /// correctness-over-borrow policy).
53    static ACTIVE_WINDOW: RefCell<Option<WindowCursor>> = const { RefCell::new(None) };
54}
55
56/// RAII guard installing the active decode-borrow source for its scope,
57/// restoring the previous value on drop.
58pub(crate) struct ActiveWindowGuard {
59    previous: Option<WindowCursor>,
60}
61
62impl ActiveWindowGuard {
63    /// Install a snapshot of `window`'s current `Bytes` backing (via
64    /// [`WindowCursor::active_bytes`] — `None` when stitched) as the active
65    /// borrow source for the returned guard's scope.
66    pub(crate) fn install(window: &WindowCursor) -> Self {
67        let snapshot = window.active_bytes().map(WindowCursor::from_borrowed_bytes);
68        let previous = ACTIVE_WINDOW.with(|w| w.replace(snapshot));
69        Self { previous }
70    }
71}
72
73impl Drop for ActiveWindowGuard {
74    fn drop(&mut self) {
75        ACTIVE_WINDOW.with(|w| *w.borrow_mut() = self.previous.take());
76    }
77}
78
79/// Materialize `sub` — a byte slice a decode site already holds, ordinarily a
80/// subslice of the active window's logical bytes — as a [`Bytes`] payload
81/// (issue #1644).
82///
83/// Zero-copy (a refcounted view, via [`WindowCursor::borrow_subslice`]'s
84/// pointer-range check) when a window is active AND `sub`'s address range lies
85/// within it; an owned copy otherwise (no active window, a stitched straddle,
86/// or `sub` sourced from an unrelated buffer). The bounds check is a
87/// pointer-VALUE comparison only — it can only ever choose borrow-vs-copy,
88/// never affect memory safety, since the copy path dereferences only `sub`
89/// itself (a slice the caller already legitimately holds).
90pub(crate) fn borrow_active(sub: &[u8]) -> Bytes {
91    ACTIVE_WINDOW.with(|w| match w.borrow().as_ref() {
92        Some(snapshot) => snapshot.borrow_subslice(sub).into_bytes(),
93        None => {
94            #[cfg(feature = "scan-offload-probe")]
95            super::window_cursor::probe::note_bytes_copied_into_value(sub.len());
96            Bytes::copy_from_slice(sub)
97        }
98    })
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104
105    #[test]
106    fn borrow_active_without_a_guard_copies() {
107        let data = b"no active window".to_vec();
108        let b = borrow_active(&data);
109        assert_eq!(&b[..], &data[..]);
110    }
111
112    #[test]
113    fn borrow_active_within_installed_window_is_zero_copy() {
114        let mut w = WindowCursor::new();
115        w.refill_owned(Bytes::from_static(b"hello borrowed world"));
116        let guard = ActiveWindowGuard::install(&w);
117        let sub = &w.as_slice()[0..5];
118        let b = borrow_active(sub);
119        assert_eq!(&b[..], b"hello");
120        assert_eq!(b.as_ptr(), sub.as_ptr());
121        drop(guard);
122        // After the guard drops, the same call falls back to a copy.
123        let owned = borrow_active(sub);
124        assert_eq!(&owned[..], b"hello");
125    }
126
127    #[test]
128    fn borrow_active_for_stitched_window_copies() {
129        let mut w = WindowCursor::new();
130        w.refill(b"stitched, no bytes backing");
131        let _guard = ActiveWindowGuard::install(&w);
132        let sub = &w.as_slice()[0..8];
133        let b = borrow_active(sub);
134        assert_eq!(&b[..], b"stitched");
135    }
136
137    #[test]
138    fn nested_guards_restore_previous_on_drop() {
139        let mut outer = WindowCursor::new();
140        outer.refill_owned(Bytes::from_static(b"outer-window-bytes"));
141        let outer_guard = ActiveWindowGuard::install(&outer);
142        {
143            let mut inner = WindowCursor::new();
144            inner.refill_owned(Bytes::from_static(b"inner"));
145            let _inner_guard = ActiveWindowGuard::install(&inner);
146            let sub = &inner.as_slice()[0..5];
147            assert_eq!(&borrow_active(sub)[..], b"inner");
148        }
149        // Outer window is restored after the inner guard drops.
150        let sub = &outer.as_slice()[0..5];
151        let b = borrow_active(sub);
152        assert_eq!(&b[..], b"outer");
153        assert_eq!(b.as_ptr(), sub.as_ptr());
154        drop(outer_guard);
155    }
156}
