1//! Next-partition (successor) offset resolution for the single-partition seek
2//! window (issue #953 / #951).
3//!
4//! The within-SSTable single-partition seek bounds its decompression window to
5//! exactly one partition's byte extent `[target_offset, successor_offset)` using
6//! authoritative index/trie layout metadata (never a heuristic boundary scan).
7//! Both helpers are gated `#[cfg(not(feature = "tombstones"))]` like the seek path
8//! they serve: their only caller is `scan_single_partition`, which the `tombstones`
9//! build compiles out (it serves single-partition reads via a full scan + filter,
10//! not a seek).
11//!
12//! Split out of `partition_lookup.rs` (campsite / epic #1116) so the B4 key-cache
13//! wiring (issue #1570) does not push that file over the source-size threshold.
14
15#[cfg(not(feature = "tombstones"))]
16use super::SSTableReader;
17#[cfg(not(feature = "tombstones"))]
18use crate::{Error, Result};
19
20#[cfg(not(feature = "tombstones"))]
21impl SSTableReader {
22    /// Authoritatively resolve the UNCOMPRESSED `Data.db` offset of the partition
23    /// that immediately FOLLOWS the partition starting at `target_offset`, used to
24    /// bound the within-SSTable seek's decompression window to exactly one
25    /// partition's byte extent (issue #953 / #951 MEDIUM).
26    ///
27    /// The successor's start offset is the partition's exclusive END: a partition
28    /// occupies `[target_offset, successor_offset)`, so decompressing the chunks
29    /// covering that half-open range materializes every byte of the target
30    /// partition (including a row/cell that spans multiple compression chunks)
31    /// without reading any of the next partition. This is authoritative metadata
32    /// (the index/trie's own partition layout), NOT a heuristic boundary scan.
33    ///
34    /// Returns:
35    /// - `Ok(Some(off))` — the next partition's start offset (`off > target_offset`).
36    /// - `Ok(None)` — `target_offset` is the LAST partition (no successor); the
37    ///   caller bounds the end with the authoritative data-section length.
38    ///
39    /// Resolution is per index format:
40    /// - **BTI (`da`)** — the `Partitions.db` trie is enumerated in byte-comparable
41    ///   order (which equals `Data.db` layout order) and the resolved offsets are
42    ///   cached ([`bti_partition_offsets`]); the successor is the smallest cached
43    ///   offset strictly greater than `target_offset` (binary search).
44    /// - **BIG (`nb`)** — `Index.db` `partition_entries` are sorted by key (==
45    ///   `Data.db` order); the successor is the smallest `data_offset` strictly
46    ///   greater than `target_offset`.
47    ///
48    /// [`bti_partition_offsets`]: Self::bti_partition_offsets
49    pub(crate) fn successor_partition_offset(&self, target_offset: u64) -> Result<Option<u64>> {
50        if self.bti_partitions_db.is_some() {
51            let offsets = self.bti_partition_offsets()?;
52            // Smallest offset strictly greater than target_offset.
53            let idx = offsets.partition_point(|&o| o <= target_offset);
54            return Ok(offsets.get(idx).copied());
55        }
56
57        // BIG (`nb`): scan the sorted Index.db entries for the smallest data_offset
58        // strictly greater than target_offset. `partition_entries` are emitted in
59        // key (== Data.db) order, but we take the min over `> target` defensively
60        // rather than rely on positional adjacency.
61        if let Some(index_reader) = &self.index_reader {
62            let successor = index_reader
63                .get_partition_entries()
64                .iter()
65                .map(|e| e.data_offset)
66                .filter(|&o| o > target_offset)
67                .min();
68            return Ok(successor);
69        }
70
71        // No index available: cannot resolve a successor authoritatively.
72        Ok(None)
73    }
74
75    /// Enumerate and cache every partition's UNCOMPRESSED `Data.db` start offset
76    /// from the BTI `Partitions.db` trie, ascending (issue #953 / #951).
77    ///
78    /// Computed lazily once and memoised in [`bti_partition_offsets`]: the trie is
79    /// DFS-walked in byte-comparable order, each `BtiPartitionLocation` is resolved
80    /// to its `Data.db` offset (NARROW → `DataOffset` directly; WIDE → the
81    /// `RowsOffset`'s `TrieIndexEntry.data_position` via `Rows.db`), and the
82    /// resulting offsets are sorted ascending. The sort makes the cache a clean
83    /// successor index regardless of trie emission order.
84    ///
85    /// [`bti_partition_offsets`]: Self::bti_partition_offsets
86    fn bti_partition_offsets(&self) -> Result<&[u64]> {
87        use crate::storage::sstable::bti::{
88            iterate_partition_locations_in_bti_file, resolve_rows_db_entry, BtiPartitionLocation,
89        };
90
91        if let Some(cached) = self.bti_partition_offsets.get() {
92            return Ok(cached);
93        }
94
95        let Some(partitions_db) = &self.bti_partitions_db else {
96            // Not a BTI reader: no trie to enumerate. Cache an empty list so the
97            // successor lookup is consistently O(1) and returns no successor.
98            let _ = self.bti_partition_offsets.set(Vec::new());
99            return Ok(self
100                .bti_partition_offsets
101                .get()
102                .map(Vec::as_slice)
103                .unwrap_or(&[]));
104        };
105
106        let mut cursor = std::io::Cursor::new(partitions_db.as_slice());
107        // Offset-only enumeration (issue #1649): we need only the partition
108        // locations here, never the reconstructed token keys, so this path
109        // performs zero per-partition key-`Vec` allocations.
110        let locations = iterate_partition_locations_in_bti_file(&mut cursor).map_err(|e| {
111            Error::corruption(format!(
112                "BTI Partitions.db trie enumeration failed while resolving the \
113                 next-partition seek bound: {e}"
114            ))
115        })?;
116
117        let mut offsets = Vec::with_capacity(locations.len());
118        for location in locations {
119            let off = match location {
120                BtiPartitionLocation::DataOffset(off) => off,
121                BtiPartitionLocation::RowsOffset(rows_offset) => {
122                    let rows_db = self.bti_rows_db.as_ref().ok_or_else(|| {
123                        Error::corruption(format!(
124                            "BTI Partitions.db enumeration returned RowsOffset({rows_offset}) \
125                             but this reader has no Rows.db; the SSTable is structurally invalid \
126                             (Rows.db is required for wide partitions)."
127                        ))
128                    })?;
129                    let header = resolve_rows_db_entry(rows_db.as_slice(), rows_offset as usize)
130                        .map_err(|e| {
131                            Error::corruption(format!(
132                                "BTI Rows.db row-index entry at RowsOffset({rows_offset}) is \
133                                 unreadable while resolving the next-partition seek bound: {e}"
134                            ))
135                        })?;
136                    header.data_position
137                }
138            };
139            offsets.push(off);
140        }
141        offsets.sort_unstable();
142
143        // Another thread may have populated the cache between the `get` above and
144        // here; `set` fails in that case and we read the winning value back.
145        let _ = self.bti_partition_offsets.set(offsets);
146        Ok(self
147            .bti_partition_offsets
148            .get()
149            .map(Vec::as_slice)
150            .unwrap_or(&[]))
151    }
152}
