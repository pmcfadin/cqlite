13    /// Enhanced partition lookup using Index.db reader with promoted index support.
14    ///
15    /// `partition_key` must be the raw partition-key bytes as produced by
16    /// [`PartitionKey::to_bytes`](crate::storage::write_engine::mutation::PartitionKey::to_bytes):
17    ///
18    /// - **Single-component keys** — raw value bytes (UUID = 16 bytes, int = 4 BE bytes, etc.).
19    /// - **Multi-component (composite) keys** — `[len: u16 BE][value bytes][0x00]` per component,
20    ///   including a trailing `0x00` after the final component.
21    ///
22    /// The Index.db key_lookup map is keyed on these exact raw bytes (set when the BIG-format
23    /// parser was fixed in Issue #552).  The old digest-based path (which caused every lookup
24    /// to miss) has been removed.  On a miss the function returns `Ok(None)` so callers can
25    /// fall through to their existing sequential-scan fallback.
26    pub async fn lookup_partition_with_index(
27        &self,
28        partition_key: &[u8],
29    ) -> Result<Option<(u64, u32)>> {
30        use crate::observability::{self as obs, catalog};
31
32        let _span = tracing::debug_span!("sstable.partition_lookup.index").entered();
33        let format = self.sstable_format_label();
34
35        if let Some(index_reader) = &self.index_reader {
36            // Direct raw-key lookup — O(1) HashMap lookup.
37            // Index.db entries are keyed on the raw partition key bytes since #552;
38            // no Murmur3 digest computation is needed or correct here.
39            if let Some(entry) = index_reader.lookup_partition(partition_key) {
40                debug!(
41                    "Found partition via Index.db raw-key lookup: offset={}, size={}",
42                    entry.data_offset, entry.data_size
43                );
44                obs::add_counter(
45                    catalog::READ_PARTITION_LOOKUP,
46                    1,
47                    &[
48                        (catalog::attr::RESULT, "hit".into()),
49                        (catalog::attr::LOOKUP_ROUTE, "index".into()),
50                        (catalog::attr::SSTABLE_FORMAT, format.into()),
51                    ],
52                );
53                return Ok(Some((entry.data_offset, entry.data_size)));
54            } else {
55                debug!(
56                    "Partition not found in Index.db for raw key (len={})",
57                    partition_key.len()
58                );
59            }
60        } else {
61            debug!("No Index.db reader available for partition lookup");
62        }
63        obs::add_counter(
64            catalog::READ_PARTITION_LOOKUP,
65            1,
66            &[
67                (catalog::attr::RESULT, "miss".into()),
68                (catalog::attr::LOOKUP_ROUTE, "index".into()),
69                (catalog::attr::SSTABLE_FORMAT, format.into()),
70            ],
71        );
72        Ok(None)
73    }

75    /// BTI ("da") partition point lookup via the in-memory Partitions.db trie
76    /// (issue #831, building on the verified #755 primitive).
77    ///
78    /// Encodes `partition_key` into the BTI byte-comparable trie key and walks the
79    /// trie to resolve the partition's location. Returns:
80    ///
81    /// - `Ok(Some(offset))` — the UNCOMPRESSED Data.db byte offset of the
82    ///   partition. INVARIANT: this offset indexes into the DECOMPRESSED data
83    ///   section, never raw file bytes. The offset is resolved in one of two ways:
84    ///     - **NARROW** partition (`BtiPartitionLocation::DataOffset`) — the trie
85    ///       returns the Data.db offset directly.
86    ///     - **WIDE** partition (`BtiPartitionLocation::RowsOffset`) — the trie
87    ///       returns a positive offset into `Rows.db`; we deserialize that
88    ///       partition's `TrieIndexEntry` via [`resolve_rows_db_entry`] and use its
89    ///       recovered `data_position` (issue #909/#910). Both forms share the same
90    ///       uncompressed Data.db offset domain, so the caller treats them
91    ///       identically.
92    /// - `Ok(None)` — the reader is not BTI, or the key has no trie path (the
93    ///   partition is definitely absent from this SSTable).
94    /// - `Err(_)` — a structural trie parse error, or a `RowsOffset` was returned
95    ///   without an accompanying `Rows.db` (a structurally invalid BTI SSTable).
96    ///
97    /// Because the BTI trie uses path compression, a returned offset may be a
98    /// candidate for a *prefix-colliding* key. The caller (`bti_point_lookup`)
99    /// MUST verify the partition-key bytes at the resolved offset equal the
100    /// queried key before returning rows.
101    ///
102    /// [`resolve_rows_db_entry`]: crate::storage::sstable::bti::resolve_rows_db_entry
103    pub fn lookup_partition_via_bti_trie(&self, partition_key: &[u8]) -> Result<Option<u64>> {
104        use crate::observability::{self as obs, catalog};
105        use crate::storage::sstable::bti::{
106            lookup_raw_key_in_bti_partitions_db, resolve_rows_db_entry, BtiPartitionLocation,
107        };
108
109        let span = tracing::debug_span!(
110            "sstable.partition_lookup.bti_trie",
111            partition_shape = tracing::field::Empty,
112        );
113        let _entered = span.enter();
114
115        let Some(partitions_db) = &self.bti_partitions_db else {
116            // Not a BTI reader — no trie to consult (no descent, no count).
117            return Ok(None);
118        };
119
120        // A5 read-work counter (TRIE_WALKS; consumers C3/C4): one per real trie
121        // descent — counted only once we have a `Partitions.db` to descend, so a
122        // non-BTI reader that returned above never bumps it. No-op in release
123        // (design.md Decision 1/2).
124        crate::storage::sstable::read_work_counters::record_trie_walk();
125
126        let lookup = lookup_raw_key_in_bti_partitions_db(
127            &mut std::io::Cursor::new(partitions_db.as_slice()),
128            partition_key,
129        )
130        .map_err(|e| {
131            Error::corruption(format!(
132                "BTI Partitions.db trie lookup failed for partition key (len={}): {}",
133                partition_key.len(),
134                e
135            ))
136        });
137        let lookup = match lookup {
138            Ok(v) => v,
139            Err(e) => {
140                obs::record_error(&e, "reader");
141                return Err(e);
142            }
143        };
144
145        // The BTI Partitions.db trie IS the presence oracle for a BTI SSTable
146        // (BTI has no bloom filter). Emit READ_BLOOM_CHECKS exactly ONCE here — the
147        // single common path every BTI presence/point lookup funnels through
148        // (get / get_with_spec_readers / get_with_schema_context / point lookup /
149        // might_contain_partition). `cqlite.result` is hit for a trie HIT
150        // (maybe-present/found: a DataOffset/RowsOffset, possibly a prefix-collision
151        // candidate the caller re-verifies) and miss for a trie MISS (definitive
152        // absence). A trie parse error returns above without an outcome and is
153        // counted as an error, not a bloom check, so this never double counts.
154        // Emitting before the per-arm handling (which can still error on a missing
155        // Rows.db for a wide partition) guarantees a single emission per call.
156        obs::add_counter(
157            catalog::READ_BLOOM_CHECKS,
158            1,
159            &[
160                (
161                    catalog::attr::RESULT,
162                    if lookup.is_some() { "hit" } else { "miss" }.into(),
163                ),
164                (catalog::attr::SSTABLE_FORMAT, "bti".into()),
165            ],
166        );
167
168        match lookup {
169            Some(BtiPartitionLocation::DataOffset(off)) => {
170                span.record("partition_shape", "narrow");
171                obs::add_counter(
172                    catalog::READ_PARTITION_LOOKUP,
173                    1,
174                    &[
175                        (catalog::attr::RESULT, "hit".into()),
176                        (catalog::attr::LOOKUP_ROUTE, "bti_trie".into()),
177                        (catalog::attr::SSTABLE_FORMAT, "bti".into()),
178                    ],
179                );
180                debug!(
181                    "BTI trie resolved NARROW partition (key len={}) to Data.db offset {}",
182                    partition_key.len(),
183                    off
184                );
185                Ok(Some(off))
186            }
187            Some(BtiPartitionLocation::RowsOffset(rows_offset)) => {
188                // WIDE partition: the trie pointed at this partition's
189                // TrieIndexEntry inside Rows.db. Deserialize it to recover the
190                // partition's Data.db start position (`data_position`), which lives
191                // in the SAME uncompressed-offset domain the narrow path uses, so
192                // the caller can decode the partition identically (#909/#910).
193                span.record("partition_shape", "wide");
194                let rows_db = match self.bti_rows_db.as_ref().ok_or_else(|| {
195                    Error::corruption(format!(
196                        "BTI Partitions.db trie returned RowsOffset({}) for partition key \
197                         (len={}) but this reader has no Rows.db loaded; the SSTable is \
198                         structurally invalid (Rows.db is required for wide partitions).",
199                        rows_offset,
200                        partition_key.len()
201                    ))
202                }) {
203                    Ok(v) => v,
204                    Err(e) => {
205                        obs::record_error(&e, "reader");
206                        return Err(e);
207                    }
208                };
209
210                let header = match resolve_rows_db_entry(rows_db.as_slice(), rows_offset as usize)
211                    .map_err(|e| {
212                        Error::corruption(format!(
213                            "BTI Rows.db row-index entry at RowsOffset({}) is unreadable for \
214                             partition key (len={}): {}",
215                            rows_offset,
216                            partition_key.len(),
217                            e
218                        ))
219                    }) {
220                    Ok(v) => v,
221                    Err(e) => {
222                        obs::record_error(&e, "reader");
223                        return Err(e);
224                    }
225                };
226
227                obs::add_counter(
228                    catalog::READ_PARTITION_LOOKUP,
229                    1,
230                    &[
231                        (catalog::attr::RESULT, "hit".into()),
232                        (catalog::attr::LOOKUP_ROUTE, "bti_trie".into()),
233                        (catalog::attr::SSTABLE_FORMAT, "bti".into()),
234                    ],
235                );
236                debug!(
237                    "BTI trie resolved WIDE partition (key len={}) via RowsOffset {} -> Data.db \
238                     position {} ({} row-index blocks)",
239                    partition_key.len(),
240                    rows_offset,
241                    header.data_position,
242                    header.block_count
243                );
244                Ok(Some(header.data_position))
245            }
246            None => {
247                obs::add_counter(
248                    catalog::READ_PARTITION_LOOKUP,
249                    1,
250                    &[
251                        (catalog::attr::RESULT, "miss".into()),
252                        (catalog::attr::LOOKUP_ROUTE, "bti_trie".into()),
253                        (catalog::attr::SSTABLE_FORMAT, "bti".into()),
254                    ],
255                );
256                Ok(None)
257            }
258        }
259    }

261    /// Authoritatively resolve the UNCOMPRESSED `Data.db` offset of the partition
262    /// that immediately FOLLOWS the partition starting at `target_offset`, used to
263    /// bound the within-SSTable seek's decompression window to exactly one
264    /// partition's byte extent (issue #953 / #951 MEDIUM).
265    ///
266    /// The successor's start offset is the partition's exclusive END: a partition
267    /// occupies `[target_offset, successor_offset)`, so decompressing the chunks
268    /// covering that half-open range materializes every byte of the target
269    /// partition (including a row/cell that spans multiple compression chunks)
270    /// without reading any of the next partition. This is authoritative metadata
271    /// (the index/trie's own partition layout), NOT a heuristic boundary scan.
272    ///
273    /// Returns:
274    /// - `Ok(Some(off))` — the next partition's start offset (`off > target_offset`).
275    /// - `Ok(None)` — `target_offset` is the LAST partition (no successor); the
276    ///   caller bounds the end with the authoritative data-section length.
277    ///
278    /// Resolution is per index format:
279    /// - **BTI (`da`)** — the `Partitions.db` trie is enumerated in byte-comparable
280    ///   order (which equals `Data.db` layout order) and the resolved offsets are
281    ///   cached ([`bti_partition_offsets`]); the successor is the smallest cached
282    ///   offset strictly greater than `target_offset` (binary search).
283    /// - **BIG (`nb`)** — `Index.db` `partition_entries` are sorted by key (==
284    ///   `Data.db` order); the successor is the smallest `data_offset` strictly
285    ///   greater than `target_offset`.
286    ///
287    /// [`bti_partition_offsets`]: Self::bti_partition_offsets
288    ///
289    /// Gated `not(tombstones)` like the seek path it serves: the only caller is
290    /// `scan_single_partition`, which the `tombstones` build compiles out (it
291    /// serves single-partition reads via a full scan + filter, not a seek).
292    #[cfg(not(feature = "tombstones"))]
293    pub(crate) fn successor_partition_offset(&self, target_offset: u64) -> Result<Option<u64>> {
294        if self.bti_partitions_db.is_some() {
295            let offsets = self.bti_partition_offsets()?;
296            // Smallest offset strictly greater than target_offset.
297            let idx = offsets.partition_point(|&o| o <= target_offset);
298            return Ok(offsets.get(idx).copied());
299        }
300
301        // BIG (`nb`): scan the sorted Index.db entries for the smallest data_offset
302        // strictly greater than target_offset. `partition_entries` are emitted in
303        // key (== Data.db) order, but we take the min over `> target` defensively
304        // rather than rely on positional adjacency.
305        if let Some(index_reader) = &self.index_reader {
306            let successor = index_reader
307                .get_partition_entries()
308                .iter()
309                .map(|e| e.data_offset)
310                .filter(|&o| o > target_offset)
311                .min();
312            return Ok(successor);
313        }
314
315        // No index available: cannot resolve a successor authoritatively.
316        Ok(None)
317    }

319    /// Enumerate and cache every partition's UNCOMPRESSED `Data.db` start offset
320    /// from the BTI `Partitions.db` trie, ascending (issue #953 / #951).
321    ///
322    /// Computed lazily once and memoised in [`bti_partition_offsets`]: the trie is
323    /// DFS-walked in byte-comparable order, each `BtiPartitionLocation` is resolved
324    /// to its `Data.db` offset (NARROW → `DataOffset` directly; WIDE → the
325    /// `RowsOffset`'s `TrieIndexEntry.data_position` via `Rows.db`), and the
326    /// resulting offsets are sorted ascending. The sort makes the cache a clean
327    /// successor index regardless of trie emission order.
328    ///
329    /// [`bti_partition_offsets`]: Self::bti_partition_offsets
330    #[cfg(not(feature = "tombstones"))]
331    fn bti_partition_offsets(&self) -> Result<&[u64]> {
332        use crate::storage::sstable::bti::{
333            iterate_partitions_in_bti_file, resolve_rows_db_entry, BtiPartitionLocation,
334        };
335
336        if let Some(cached) = self.bti_partition_offsets.get() {
337            return Ok(cached);
338        }
339
340        let Some(partitions_db) = &self.bti_partitions_db else {
341            // Not a BTI reader: no trie to enumerate. Cache an empty list so the
342            // successor lookup is consistently O(1) and returns no successor.
343            let _ = self.bti_partition_offsets.set(Vec::new());
344            return Ok(self
345                .bti_partition_offsets
346                .get()
347                .map(Vec::as_slice)
348                .unwrap_or(&[]));
349        };
350
351        let mut cursor = std::io::Cursor::new(partitions_db.as_slice());
352        let entries = iterate_partitions_in_bti_file(&mut cursor).map_err(|e| {
353            Error::corruption(format!(
354                "BTI Partitions.db trie enumeration failed while resolving the \
355                 next-partition seek bound: {e}"
356            ))
357        })?;
358
359        let mut offsets = Vec::with_capacity(entries.len());
360        for (_token_key, location) in entries {
361            let off = match location {
362                BtiPartitionLocation::DataOffset(off) => off,
363                BtiPartitionLocation::RowsOffset(rows_offset) => {
364                    let rows_db = self.bti_rows_db.as_ref().ok_or_else(|| {
365                        Error::corruption(format!(
366                            "BTI Partitions.db enumeration returned RowsOffset({rows_offset}) \
367                             but this reader has no Rows.db; the SSTable is structurally invalid \
368                             (Rows.db is required for wide partitions)."
369                        ))
370                    })?;
371                    let header = resolve_rows_db_entry(rows_db.as_slice(), rows_offset as usize)
372                        .map_err(|e| {
373                            Error::corruption(format!(
374                                "BTI Rows.db row-index entry at RowsOffset({rows_offset}) is \
375                                 unreadable while resolving the next-partition seek bound: {e}"
376                            ))
377                        })?;
378                    header.data_position
379                }
380            };
381            offsets.push(off);
382        }
383        offsets.sort_unstable();
384
385        // Another thread may have populated the cache between the `get` above and
386        // here; `set` fails in that case and we read the winning value back.
387        let _ = self.bti_partition_offsets.set(offsets);
388        Ok(self
389            .bti_partition_offsets
390            .get()
391            .map(Vec::as_slice)
392            .unwrap_or(&[]))
393    }

395    /// Cheap presence oracle: can this SSTable possibly contain `partition_key`?
396    ///
397    /// Used to prune SSTables before a partition-targeted scan (the query engine's
398    /// `WHERE pk = ?` fast path). Returning `false` MUST be definitive — the SSTable
399    /// is then skipped entirely without being parsed — so this only ever returns
400    /// `false` for an authoritative "absent" signal:
401    ///
402    /// - **BTI ("da")** readers have no bloom filter; the Partitions.db trie is the
403    ///   authoritative present/absent oracle. A trie miss (`Ok(None)`) is definitive
404    ///   absence. A trie hit may be a prefix-collision candidate, which is a safe
405    ///   *false positive* here (the partition scan re-verifies the key). Any trie
406    ///   parse error is treated conservatively as "maybe present".
407    /// - **BIG-format** readers consult the bloom filter, which never reports false
408    ///   negatives: `might_contain == false` is definitive absence. With no bloom
409    ///   filter loaded we cannot prune, so we conservatively return `true`.
410    ///
411    /// `partition_key` must be the raw partition-key bytes (same encoding the bloom
412    /// filter and Index.db/BTI trie are keyed on).
413    pub fn might_contain_partition(&self, partition_key: &[u8]) -> bool {
414        use crate::observability::{self as obs, catalog};
415
416        if self.bti_partitions_db.is_some() {
417            // BTI: trie miss is authoritative absence; any error is conservative.
418            // `lookup_partition_via_bti_trie` is the single common path that emits
419            // READ_BLOOM_CHECKS for the BTI presence check — do NOT emit again here
420            // or the metric would be double counted.
421            return matches!(
422                self.lookup_partition_via_bti_trie(partition_key),
423                Ok(Some(_)) | Err(_)
424            );
425        }
426        // BIG: only record READ_BLOOM_CHECKS when a bloom filter actually exists.
427        // With no filter loaded we cannot prune, so we conservatively return `true`
428        // WITHOUT recording a check (a no-filter "hit" is not a real bloom check and
429        // would inflate the metric).
430        match &self.bloom_filter {
431            Some(bloom) => {
432                let present = bloom.might_contain(partition_key);
433                obs::add_counter(
434                    catalog::READ_BLOOM_CHECKS,
435                    1,
436                    &[
437                        (
438                            catalog::attr::RESULT,
439                            if present { "hit" } else { "miss" }.into(),
440                        ),
441                        (
442                            catalog::attr::SSTABLE_FORMAT,
443                            self.sstable_format_label().into(),
444                        ),
445                    ],
446                );
447                present
448            }
449            None => true,
450        }
451    }

453    /// Enhanced partition lookup using schema-driven key digest computation
454    pub async fn lookup_partition_with_schema_context(
455        &self,
456        partition_key: &[u8],
457        parsing_context: &ParsingContext,
458    ) -> Result<Option<(u64, u32)>> {
459        if let Some(index_reader) = &self.index_reader {
460            // Compute the schema-driven key digest for Index.db lookup
461            let key_digest =
462                self.compute_partition_key_digest_with_schema(partition_key, parsing_context)?;
463
464            // Use spec-compliant Index.db reader for partition lookup
465            if let Some(entry) = index_reader.lookup_partition(&key_digest) {
466                debug!(
467                    "Found partition via schema-driven Index.db: offset={}, size={}",
468                    entry.data_offset, entry.data_size
469                );
470                return Ok(Some((entry.data_offset, entry.data_size)));
471            }
472        }
473        Ok(None)
474    }

476    /// Enhanced partition iteration using Summary.db reader
477    ///
478    /// Note: Token-based range queries are not directly supported because Summary.db
479    /// does not store token values (Issue #218). Instead, this iterates all summary
480    /// entries and returns all partition data.
481    ///
482    /// For token-based filtering, compute tokens from partition keys after retrieval.
483    ///
484    /// ## Issue #500: Sequential-scan fallback for writer-produced SSTables
485    ///
486    /// The Summary.db → Index.db → Data.db lookup path depends on Index.db format
487    /// compatibility between writer and reader (digest format vs. raw-key format).
488    /// Locally written SSTables emit raw-key Index.db entries that the reader's
489    /// digest-based parser cannot resolve, so the lookup loop returns 0 entries
490    /// even though Summary.db enumerates the partitions correctly.
491    ///
492    /// When that happens we fall back to `sequential_scan`, which walks Data.db
493    /// directly. For V5CompressedLegacy NB SSTables (the format the writer emits),
494    /// `sequential_scan` uses the chunk-stitching path and returns every partition.
495    pub async fn iterate_all_partitions(&self) -> Result<Vec<(RowKey, ScanRow)>> {
496        if let Some(summary_reader) = &self.summary_reader {
497            let entries = summary_reader.get_entries();
498            let mut results = Vec::new();
499
500            for entry in entries {
501                // Use Summary.db entry to find the corresponding Index.db entry
502                if let Some(_index_reader) = &self.index_reader {
503                    // The summary entry provides a position in Index.db
504                    // We need to read the partition data from Data.db
505
506                    // For now, use the partition key from the summary entry
507                    let partition_key_bytes = &entry.partition_key;
508
509                    // Look up the partition in Index.db to get the actual data offset
510                    if let Some((data_offset, data_size)) = self
511                        .lookup_partition_with_index(partition_key_bytes)
512                        .await?
513                    {
514                        // Convert Index.db relative offset to absolute file offset
515                        // Index.db offsets are relative to data section start (after compression header)
516                        let absolute_offset = data_offset + self.actual_header_size as u64;
517
518                        // Read and parse the actual partition data from Data.db
519                        match self
520                            .parse_partition_at_offset(absolute_offset, data_size)
521                            .await?
522                        {
523                            Some(partition_entries) => {
524                                for (row_key, value) in partition_entries {
525                                    results.push((row_key, value));
526                                }
527                            }
528                            None => {
529                                debug!("Failed to parse partition at offset {}", absolute_offset);
530                            }
531                        }
532                    }
533                } else {
534                    log::error!("Index reader not available for partition iteration");
535                    return Err(Error::corruption(
536                        "Index reader required for partition iteration - synthetic data not allowed for Issue #35",
537                    ));
538                }
539            }
540
541            // Only trust the index-based path when EVERY summary entry was resolved.
542            // Partial resolution silently drops the unresolved entries; defaulting to
543            // `sequential_scan` in that case is strictly safer and still correct on
544            // real Cassandra SSTables (sequential_scan returns the same partitions
545            // when the index resolves them all).
546            if results.len() == entries.len() && !entries.is_empty() {
547                debug!("Partition iteration found {} entries", results.len());
548                return Ok(results);
549            }
550
551            debug!(
552                "Index.db lookup resolved {}/{} summary entries; \
553                 falling back to sequential_scan (Issue #500)",
554                results.len(),
555                entries.len()
556            );
557        }
558
559        // Fallback path: sequential walk of Data.db.
560        // Used when Summary.db is absent OR when the Index.db lookup loop returned
561        // no entries (Issue #500: writer-produced SSTables emit a raw-key Index.db
562        // format the reader's digest-based parser does not resolve).
563        let table_id = self.scan_table_id();
564        let schema = self.schema.as_deref();
565        self.sequential_scan(&table_id, None, None, None, schema)
566            .await
567    }

569    /// Build the TableId used for fallback `sequential_scan` from header metadata.
570    ///
571    /// The reader populates `header.keyspace` / `header.table_name` from either the
572    /// SSTable header or the parent directory path. When the V5CompressedLegacy
573    /// stitching path is used, table_id matching is skipped, so any non-empty value
574    /// is accepted; for other formats this returns the qualified `keyspace.table`
575    /// form so the scan filter matches.
576    fn scan_table_id(&self) -> TableId {
577        let keyspace = &self.header.keyspace;
578        let table_name = &self.header.table_name;
579        if !keyspace.is_empty() && !table_name.is_empty() {
580            TableId::from(format!("{}.{}", keyspace, table_name))
581        } else if !table_name.is_empty() {
582            TableId::from(table_name.as_str())
583        } else {
584            TableId::from("default")
585        }
586    }

637    /// Enhanced get method using spec readers for efficient lookup
638    pub async fn get_with_spec_readers(
639        &self,
640        table_id: &TableId,
641        key: &RowKey,
642    ) -> Result<Option<ScanRow>> {
643        use crate::observability::{self as obs, catalog};
644
645        // Issue #1034: BTI ("da") readers resolve partitions via the Partitions.db
646        // trie, which is the AUTHORITATIVE presence oracle. Branch to the trie path
647        // FIRST and skip the bloom/Index.db pre-check entirely. This mirrors `get()`
648        // (which routes BTI to `bti_point_lookup` → `lookup_partition_via_bti_trie`)
649        // and is required for correctness: a BTI bloom false negative must never
650        // short-circuit the trie lookup. Routing through `get()` also guarantees the
651        // BTI presence check emits READ_BLOOM_CHECKS exactly once (from
652        // `lookup_partition_via_bti_trie`) instead of being counted here AND again on
653        // the fallback path. BIG readers keep the bloom → Index.db behavior below.
654        if self.bti_partitions_db.is_some() {
655            return self.get(table_id, key).await;
656        }
657
658        // Step 1: Use bloom filter for existence check
659        if let Some(bloom_filter) = &self.bloom_filter {
660            let present = bloom_filter.might_contain(key.as_bytes());
661            obs::add_counter(
662                catalog::READ_BLOOM_CHECKS,
663                1,
664                &[
665                    (
666                        catalog::attr::RESULT,
667                        if present { "hit" } else { "miss" }.into(),
668                    ),
669                    (
670                        catalog::attr::SSTABLE_FORMAT,
671                        self.sstable_format_label().into(),
672                    ),
673                ],
674            );
675            if !present {
676                debug!("Bloom filter indicates key does not exist");
677                return Ok(None);
678            }
679        }
680
681        // Step 2: Use Index.db reader for precise partition lookup
682        if let Some((offset, size)) = self.lookup_partition_with_index(key.as_bytes()).await? {
683            debug!("Using Index.db lookup: offset={}, size={}", offset, size);
684            return self.read_value_at_offset(offset, size).await;
685        }
686
687        // Step 3: Fallback to existing methods
688        debug!("Falling back to legacy lookup methods");
689        self.get(table_id, key).await
690    }

692    /// Enhanced get method using spec readers with schema-driven key digest computation
693    pub async fn get_with_schema_context(
694        &self,
695        table_id: &TableId,
696        key: &RowKey,
697        parsing_context: &ParsingContext,
698    ) -> Result<Option<ScanRow>> {
699        use crate::observability::{self as obs, catalog};
700
701        // Issue #1034: BTI ("da") readers resolve partitions via the Partitions.db
702        // trie keyed on RAW partition-key bytes, not Index.db key digests, so the
703        // schema-driven digest path below does not apply. Branch to the trie path
704        // FIRST and skip the bloom/Index.db pre-check entirely, mirroring `get()`
705        // (which routes BTI to `bti_point_lookup` → `lookup_partition_via_bti_trie`).
706        // This keeps BTI correct (a bloom false negative can never short-circuit the
707        // authoritative trie lookup) and ensures the BTI presence check emits
708        // READ_BLOOM_CHECKS exactly once instead of being double counted across this
709        // helper and its fallback. BIG readers keep the bloom → Index.db behavior.
710        if self.bti_partitions_db.is_some() {
711            return self.get(table_id, key).await;
712        }
713
714        // Step 1: Use bloom filter for existence check
715        if let Some(bloom_filter) = &self.bloom_filter {
716            let present = bloom_filter.might_contain(key.as_bytes());
717            obs::add_counter(
718                catalog::READ_BLOOM_CHECKS,
719                1,
720                &[
721                    (
722                        catalog::attr::RESULT,
723                        if present { "hit" } else { "miss" }.into(),
724                    ),
725                    (
726                        catalog::attr::SSTABLE_FORMAT,
727                        self.sstable_format_label().into(),
728                    ),
729                ],
730            );
731            if !present {
732                debug!("Bloom filter indicates key does not exist");
733                return Ok(None);
734            }
735        }
736
737        // Step 2: Use Index.db reader for precise partition lookup with schema-driven digest
738        if let Some((offset, size)) = self
739            .lookup_partition_with_schema_context(key.as_bytes(), parsing_context)
740            .await?
741        {
742            debug!(
743                "Using schema-driven Index.db lookup: offset={}, size={}",
744                offset, size
745            );
746            return self.read_value_at_offset(offset, size).await;
747        }
748
749        // Step 3: Fallback to existing methods
750        debug!("Falling back to legacy lookup methods");
751        self.get(table_id, key).await
752    }
