12    /// Enhanced partition lookup using Index.db reader with promoted index support.
13    ///
14    /// `partition_key` must be the raw partition-key bytes as produced by
15    /// [`PartitionKey::to_bytes`](crate::storage::write_engine::mutation::PartitionKey::to_bytes):
16    ///
17    /// - **Single-component keys** — raw value bytes (UUID = 16 bytes, int = 4 BE bytes, etc.).
18    /// - **Multi-component (composite) keys** — `[len: u16 BE][value bytes][0x00]` per component,
19    ///   including a trailing `0x00` after the final component.
20    ///
21    /// The Index.db key_lookup map is keyed on these exact raw bytes (set when the BIG-format
22    /// parser was fixed in Issue #552).  The old digest-based path (which caused every lookup
23    /// to miss) has been removed.  On a miss the function returns `Ok(None)` so callers can
24    /// fall through to their existing sequential-scan fallback.
25    pub async fn lookup_partition_with_index(
26        &self,
27        partition_key: &[u8],
28    ) -> Result<Option<(u64, u32)>> {
29        use crate::observability::{self as obs, catalog};
30
31        let _span = tracing::debug_span!("sstable.partition_lookup.index").entered();
32        let format = self.sstable_format_label();
33
34        // B4 key→partition-offset cache (issue #1570): a repeated point read of the
35        // same present key returns the cached location without re-probing Index.db,
36        // so `INDEX_PROBES` stays 0 on the hit. Positive-only, so an absent key is
37        // never cached and never fabricates a hit. The cached location is the exact
38        // `(offset,size)` a fresh probe resolved (correctness guardrail).
39        if let Some(loc) = self.key_offset_cache.get(partition_key) {
40            debug!(
41                "B4 key-cache hit for partition (raw key len={}): offset={}, size={}",
42                partition_key.len(),
43                loc.data_offset,
44                loc.data_size
45            );
46            obs::add_counter(
47                catalog::READ_PARTITION_LOOKUP,
48                1,
49                &[
50                    (catalog::attr::RESULT, "hit".into()),
51                    (catalog::attr::LOOKUP_ROUTE, "index".into()),
52                    (catalog::attr::SSTABLE_FORMAT, format.into()),
53                ],
54            );
55            return Ok(Some((loc.data_offset, loc.data_size)));
56        }
57
58        if let Some(index_reader) = &self.index_reader {
59            // A5/B4 read-work counter (`INDEX_PROBES`; consumer B4): one per real
60            // `Index.db` probe. Counted only when a cache miss forces a real probe,
61            // so a B4 cache hit above records nothing. No-op in release.
62            crate::storage::sstable::read_work_counters::record_index_probe();
63            // Direct raw-key lookup — O(1) HashMap lookup.
64            // Index.db entries are keyed on the raw partition key bytes since #552;
65            // no Murmur3 digest computation is needed or correct here.
66            if let Some(entry) = index_reader.lookup_partition(partition_key) {
67                debug!(
68                    "Found partition via Index.db raw-key lookup: offset={}, size={}",
69                    entry.data_offset, entry.data_size
70                );
71                obs::add_counter(
72                    catalog::READ_PARTITION_LOOKUP,
73                    1,
74                    &[
75                        (catalog::attr::RESULT, "hit".into()),
76                        (catalog::attr::LOOKUP_ROUTE, "index".into()),
77                        (catalog::attr::SSTABLE_FORMAT, format.into()),
78                    ],
79                );
80                // B4: cache this present-key resolution so the next point read of
81                // the same key skips the Index.db probe (issue #1570).
82                self.key_offset_cache.insert(
83                    partition_key,
84                    crate::storage::cache::PartitionLoc::new(entry.data_offset, entry.data_size),
85                );
86                return Ok(Some((entry.data_offset, entry.data_size)));
87            } else {
88                debug!(
89                    "Partition not found in Index.db for raw key (len={})",
90                    partition_key.len()
91                );
92            }
93        } else {
94            debug!("No Index.db reader available for partition lookup");
95        }
96        obs::add_counter(
97            catalog::READ_PARTITION_LOOKUP,
98            1,
99            &[
100                (catalog::attr::RESULT, "miss".into()),
101                (catalog::attr::LOOKUP_ROUTE, "index".into()),
102                (catalog::attr::SSTABLE_FORMAT, format.into()),
103            ],
104        );
105        Ok(None)
106    }

108    /// BTI ("da") partition point lookup via the in-memory Partitions.db trie
109    /// (issue #831, building on the verified #755 primitive).
110    ///
111    /// Encodes `partition_key` into the BTI byte-comparable trie key and walks the
112    /// trie to resolve the partition's location. Returns:
113    ///
114    /// - `Ok(Some(offset))` — the UNCOMPRESSED Data.db byte offset of the
115    ///   partition. INVARIANT: this offset indexes into the DECOMPRESSED data
116    ///   section, never raw file bytes. The offset is resolved in one of two ways:
117    ///     - **NARROW** partition (`BtiPartitionLocation::DataOffset`) — the trie
118    ///       returns the Data.db offset directly.
119    ///     - **WIDE** partition (`BtiPartitionLocation::RowsOffset`) — the trie
120    ///       returns a positive offset into `Rows.db`; we deserialize that
121    ///       partition's `TrieIndexEntry` via [`resolve_rows_db_entry`] and use its
122    ///       recovered `data_position` (issue #909/#910). Both forms share the same
123    ///       uncompressed Data.db offset domain, so the caller treats them
124    ///       identically.
125    /// - `Ok(None)` — the reader is not BTI, or the key has no trie path (the
126    ///   partition is definitely absent from this SSTable).
127    /// - `Err(_)` — a structural trie parse error, or a `RowsOffset` was returned
128    ///   without an accompanying `Rows.db` (a structurally invalid BTI SSTable).
129    ///
130    /// Because the BTI trie uses path compression, a returned offset may be a
131    /// candidate for a *prefix-colliding* key. The caller (`bti_point_lookup`)
132    /// MUST verify the partition-key bytes at the resolved offset equal the
133    /// queried key before returning rows.
134    ///
135    /// [`resolve_rows_db_entry`]: crate::storage::sstable::bti::resolve_rows_db_entry
136    pub fn lookup_partition_via_bti_trie(&self, partition_key: &[u8]) -> Result<Option<u64>> {
137        use crate::storage::sstable::bti::encode_partition_key_for_bti_trie;
138
139        // Not a BTI reader — no trie to consult (no descent, no encode, no count).
140        if self.bti_partitions_db.is_none() {
141            return Ok(None);
142        }
143
144        // B4 key→partition-offset cache (issue #1570) — the OUTERMOST layer, checked
145        // BEFORE the C3 memo and BEFORE the encode, so a repeated point read of the
146        // same present key returns the cached offset WITHOUT a trie descent
147        // (`TRIE_WALKS` stays 0) AND WITHOUT re-encoding the key (no `KEY_HASH_CALLS`
148        // bump — the encode below is skipped on a hit, preserving C4's
149        // one-hash-per-read property). Unlike the single-entry C3 memo, the multi-key
150        // B4 cache serves a hit even across interleaved keys. Trie-hit-only
151        // (prefix-collision candidates included; the caller re-verifies the key bytes
152        // at the resolved offset), never a trie MISS: the cached offset is the exact
153        // offset a fresh descent resolves, so re-verification reaches the identical
154        // conclusion (correctness guardrail; mirrors the C3 memo). Emit the presence
155        // counters (found = true) so a cache-served lookup records the same presence
156        // decision a fresh descent does. A disabled cache (`block_cache.enabled=false`)
157        // is a genuine no-op `get`, so this falls through and the read re-walks.
158        if let Some(loc) = self.key_offset_cache.get(partition_key) {
159            Self::emit_bti_presence_counters(true);
160            return Ok(Some(loc.data_offset));
161        }
162
163        // Issue #1574 (C3) single walk: reuse the prune's resolution for the seek —
164        // a pure function of the immutable trie + key. Checked BEFORE the encode so
165        // a memo hit costs neither a trie descent NOR a Murmur3 key hash. A
166        // stale/absent slot simply misses and re-walks. The presence-oracle
167        // observability is preserved: a memo hit still records the presence decision.
168        if let Some(resolved) = self.bti_lookup_memo_get(partition_key) {
169            Self::emit_bti_presence_counters(resolved.is_some());
170            return Ok(resolved);
171        }
172
173        // Issue #1575 (C4): encode ONCE here (Murmur3 hash + byte-comparable
174        // encoding, `KEY_HASH_CALLS`). The multi-candidate prune path instead calls
175        // `lookup_partition_via_bti_trie_encoded` with a single hoisted encoding so N
176        // candidate generations share ONE hash rather than re-hashing per candidate.
177        let encoded = encode_partition_key_for_bti_trie(partition_key);
178        self.bti_trie_resolve(partition_key, &encoded)
179    }

181    /// Resolve a BTI partition using a PRE-ENCODED byte-comparable key, so the
182    /// Murmur3 hash + encoding is computed once per read and reused across every
183    /// candidate SSTable's prune instead of being recomputed per candidate
184    /// (issue #1575 / C4). `encoded` MUST equal
185    /// `encode_partition_key_for_bti_trie(partition_key)`; the caller
186    /// (`SSTableManager`'s candidate prune) computes it exactly once. Semantics are
187    /// otherwise identical to [`lookup_partition_via_bti_trie`]: same same-key memo,
188    /// same presence-oracle observability, same resolved offset. A non-BTI reader
189    /// returns `Ok(None)` (the `encoded` argument is simply unused).
190    pub fn lookup_partition_via_bti_trie_encoded(
191        &self,
192        partition_key: &[u8],
193        encoded: &[u8; 9],
194    ) -> Result<Option<u64>> {
195        if self.bti_partitions_db.is_none() {
196            return Ok(None);
197        }
198        // B4 key→partition-offset cache (issue #1570) — the OUTERMOST layer on the
199        // candidate-prune path too, checked BEFORE the C3 memo and the trie descent.
200        // The caller (`prune_candidates`) already hoisted the single encode into
201        // `encoded` (C4 / `KEY_HASH_CALLS == 1` per read); a cache hit here simply
202        // skips this candidate's trie descent (`TRIE_WALKS` stays 0 on the hit) — the
203        // multi-key cache covers interleaved keys the single-entry memo cannot. Same
204        // trie-hit-only correctness guardrail as the raw-key entry: the cached offset
205        // is the exact offset a fresh descent resolves. A disabled cache is a no-op
206        // `get`, so this falls through and the candidate re-walks.
207        if let Some(loc) = self.key_offset_cache.get(partition_key) {
208            Self::emit_bti_presence_counters(true);
209            return Ok(Some(loc.data_offset));
210        }
211        if let Some(resolved) = self.bti_lookup_memo_get(partition_key) {
212            Self::emit_bti_presence_counters(resolved.is_some());
213            return Ok(resolved);
214        }
215        self.bti_trie_resolve(partition_key, encoded)
216    }

218    /// Shared BTI trie descent for a pre-encoded key — the body behind both
219    /// [`lookup_partition_via_bti_trie`] and
220    /// [`lookup_partition_via_bti_trie_encoded`]. The caller has already confirmed
221    /// this is a BTI reader and consulted the same-key memo; this records the single
222    /// `TRIE_WALKS`, walks the resident trie buffer with the pre-encoded key, emits
223    /// the presence-oracle counters, and stores the memo. It never re-encodes the
224    /// key (no `KEY_HASH_CALLS`), which is the property C4 hoists.
225    fn bti_trie_resolve(&self, partition_key: &[u8], encoded: &[u8; 9]) -> Result<Option<u64>> {
226        use crate::observability::{self as obs, catalog};
227        use crate::storage::sstable::bti::{
228            lookup_partition_in_bti_slice, resolve_rows_db_entry, BtiPartitionLocation,
229        };
230
231        let span = tracing::debug_span!(
232            "sstable.partition_lookup.bti_trie",
233            partition_shape = tracing::field::Empty,
234        );
235        let _entered = span.enter();
236
237        let Some(partitions_db) = &self.bti_partitions_db else {
238            // Not a BTI reader — no trie to descend (defensive; caller pre-checks).
239            return Ok(None);
240        };
241
242        // A5 read-work counter (TRIE_WALKS; consumers C3/C4): one per real trie
243        // descent — counted only once we have a `Partitions.db` to descend, so a
244        // non-BTI reader that returned above never bumps it. No-op in release
245        // (design.md Decision 1/2).
246        crate::storage::sstable::read_work_counters::record_trie_walk();
247
248        // Issue #1574 (C3): walk the resident `Arc<Vec<u8>>` trie buffer in place —
249        // no per-lookup whole-file copy. Issue #1575 (C4): with the PRE-ENCODED key,
250        // so the encode is not repeated per candidate.
251        let lookup =
252            lookup_partition_in_bti_slice(partitions_db.as_slice(), encoded).map_err(|e| {
253                Error::corruption(format!(
254                    "BTI Partitions.db trie lookup failed for partition key (len={}): {}",
255                    partition_key.len(),
256                    e
257                ))
258            });
259        let lookup = match lookup {
260            Ok(v) => v,
261            Err(e) => {
262                obs::record_error(&e, "reader");
263                return Err(e);
264            }
265        };
266
267        // The BTI Partitions.db trie IS the presence oracle for a BTI SSTable
268        // (BTI has no bloom filter). Emit READ_BLOOM_CHECKS exactly ONCE here — the
269        // single common path every BTI presence/point lookup funnels through
270        // (get → locate → bti_point_lookup, might_contain_partition, the candidate
271        // prune). `cqlite.result` is hit for a trie HIT
272        // (maybe-present/found: a DataOffset/RowsOffset, possibly a prefix-collision
273        // candidate the caller re-verifies) and miss for a trie MISS (definitive
274        // absence). A trie parse error returns above without an outcome and is
275        // counted as an error, not a bloom check, so this never double counts.
276        // Emitting before the per-arm handling (which can still error on a missing
277        // Rows.db for a wide partition) guarantees a single emission per call.
278        obs::add_counter(
279            catalog::READ_BLOOM_CHECKS,
280            1,
281            &[
282                (
283                    catalog::attr::RESULT,
284                    if lookup.is_some() { "hit" } else { "miss" }.into(),
285                ),
286                (catalog::attr::SSTABLE_FORMAT, "bti".into()),
287            ],
288        );
289
290        match lookup {
291            Some(BtiPartitionLocation::DataOffset(off)) => {
292                span.record("partition_shape", "narrow");
293                obs::add_counter(
294                    catalog::READ_PARTITION_LOOKUP,
295                    1,
296                    &[
297                        (catalog::attr::RESULT, "hit".into()),
298                        (catalog::attr::LOOKUP_ROUTE, "bti_trie".into()),
299                        (catalog::attr::SSTABLE_FORMAT, "bti".into()),
300                    ],
301                );
302                debug!(
303                    "BTI trie resolved NARROW partition (key len={}) to Data.db offset {}",
304                    partition_key.len(),
305                    off
306                );
307                self.bti_lookup_memo_store(partition_key, Some(off));
308                // B4: cache this trie-hit resolution (a prefix-collision candidate,
309                // re-verified by the caller downstream) so a later interleaved read of
310                // the same key skips the trie descent (issue #1570).
311                self.key_offset_cache.insert(
312                    partition_key,
313                    crate::storage::cache::PartitionLoc::offset_only(off),
314                );
315                Ok(Some(off))
316            }
317            Some(BtiPartitionLocation::RowsOffset(rows_offset)) => {
318                // WIDE partition: the trie pointed at this partition's
319                // TrieIndexEntry inside Rows.db. Deserialize it to recover the
320                // partition's Data.db start position (`data_position`), which lives
321                // in the SAME uncompressed-offset domain the narrow path uses, so
322                // the caller can decode the partition identically (#909/#910).
323                span.record("partition_shape", "wide");
324                let rows_db = match self.bti_rows_db.as_ref().ok_or_else(|| {
325                    Error::corruption(format!(
326                        "BTI Partitions.db trie returned RowsOffset({}) for partition key \
327                         (len={}) but this reader has no Rows.db loaded; the SSTable is \
328                         structurally invalid (Rows.db is required for wide partitions).",
329                        rows_offset,
330                        partition_key.len()
331                    ))
332                }) {
333                    Ok(v) => v,
334                    Err(e) => {
335                        obs::record_error(&e, "reader");
336                        return Err(e);
337                    }
338                };
339
340                let header = match resolve_rows_db_entry(rows_db.as_slice(), rows_offset as usize)
341                    .map_err(|e| {
342                        Error::corruption(format!(
343                            "BTI Rows.db row-index entry at RowsOffset({}) is unreadable for \
344                             partition key (len={}): {}",
345                            rows_offset,
346                            partition_key.len(),
347                            e
348                        ))
349                    }) {
350                    Ok(v) => v,
351                    Err(e) => {
352                        obs::record_error(&e, "reader");
353                        return Err(e);
354                    }
355                };
356
357                obs::add_counter(
358                    catalog::READ_PARTITION_LOOKUP,
359                    1,
360                    &[
361                        (catalog::attr::RESULT, "hit".into()),
362                        (catalog::attr::LOOKUP_ROUTE, "bti_trie".into()),
363                        (catalog::attr::SSTABLE_FORMAT, "bti".into()),
364                    ],
365                );
366                debug!(
367                    "BTI trie resolved WIDE partition (key len={}) via RowsOffset {} -> Data.db \
368                     position {} ({} row-index blocks)",
369                    partition_key.len(),
370                    rows_offset,
371                    header.data_position,
372                    header.block_count
373                );
374                self.bti_lookup_memo_store(partition_key, Some(header.data_position));
375                // B4: cache this present WIDE-partition resolution (issue #1570).
376                self.key_offset_cache.insert(
377                    partition_key,
378                    crate::storage::cache::PartitionLoc::offset_only(header.data_position),
379                );
380                Ok(Some(header.data_position))
381            }
382            None => {
383                obs::add_counter(
384                    catalog::READ_PARTITION_LOOKUP,
385                    1,
386                    &[
387                        (catalog::attr::RESULT, "miss".into()),
388                        (catalog::attr::LOOKUP_ROUTE, "bti_trie".into()),
389                        (catalog::attr::SSTABLE_FORMAT, "bti".into()),
390                    ],
391                );
392                self.bti_lookup_memo_store(partition_key, None);
393                Ok(None)
394            }
395        }
396    }

398    /// Cheap presence oracle: can this SSTable possibly contain `partition_key`?
399    ///
400    /// Used to prune SSTables before a partition-targeted scan (the query engine's
401    /// `WHERE pk = ?` fast path). Returning `false` MUST be definitive — the SSTable
402    /// is then skipped entirely without being parsed — so this only ever returns
403    /// `false` for an authoritative "absent" signal:
404    ///
405    /// - **BTI ("da")** readers have no bloom filter; the Partitions.db trie is the
406    ///   authoritative present/absent oracle. A trie miss (`Ok(None)`) is definitive
407    ///   absence. A trie hit may be a prefix-collision candidate, which is a safe
408    ///   *false positive* here (the partition scan re-verifies the key). Any trie
409    ///   parse error is treated conservatively as "maybe present".
410    /// - **BIG-format** readers consult the bloom filter, which never reports false
411    ///   negatives: `might_contain == false` is definitive absence. With no bloom
412    ///   filter loaded we cannot prune, so we conservatively return `true`.
413    ///
414    /// `partition_key` must be the raw partition-key bytes (same encoding the bloom
415    /// filter and Index.db/BTI trie are keyed on).
416    pub fn might_contain_partition(&self, partition_key: &[u8]) -> bool {
417        use crate::observability::{self as obs, catalog};
418
419        if self.bti_partitions_db.is_some() {
420            // Issue #2163 (roborev r7): restore the C5 range short-circuit
421            // (`locate`'s Step 1) ahead of the trie probe. The r2 refactor
422            // removed `locate_encoded` (replaced by routing the candidate-prune
423            // path through these `might_contain_partition[_encoded]` helpers)
424            // and dropped this step for BTI, so an out-of-range key paid a
425            // pointless trie descent and never recorded
426            // `read_work_counters::record_range_short_circuit`. A no-op for a
427            // genuine BTI reader today (no Summary.db, so
428            // `partition_key_out_of_range` always returns `false`), but this
429            // restores exact step-ordering parity with `locate` for any
430            // degraded case and keeps the work-counter accounting consistent.
431            if self.partition_key_out_of_range(partition_key) {
432                crate::storage::sstable::read_work_counters::record_range_short_circuit();
433                return false;
434            }
435            // BTI: trie miss is authoritative absence; any error is conservative.
436            // `lookup_partition_via_bti_trie` is the single common path that emits
437            // READ_BLOOM_CHECKS for the BTI presence check — do NOT emit again here
438            // or the metric would be double counted.
439            let present = matches!(
440                self.lookup_partition_via_bti_trie(partition_key),
441                Ok(Some(_)) | Err(_)
442            );
443            if !present {
444                // Definitive trie miss → this SSTable is pruned from the read
445                // (issue #2163).
446                self.emit_sstable_pruned();
447            }
448            return present;
449        }
450        // BIG: only record READ_BLOOM_CHECKS when a bloom filter actually exists.
451        // With no filter loaded we cannot prune, so we conservatively return `true`
452        // WITHOUT recording a check (a no-filter "hit" is not a real bloom check and
453        // would inflate the metric).
454        match &self.bloom_filter {
455            Some(bloom) => {
456                let present = bloom.might_contain(partition_key);
457                obs::add_counter(
458                    catalog::READ_BLOOM_CHECKS,
459                    1,
460                    &[
461                        (
462                            catalog::attr::RESULT,
463                            if present { "hit" } else { "miss" }.into(),
464                        ),
465                        (
466                            catalog::attr::SSTABLE_FORMAT,
467                            self.sstable_format_label().into(),
468                        ),
469                    ],
470                );
471                if !present {
472                    // Bloom `might_contain == false` is a definitive negative →
473                    // this SSTable is pruned from the read (issue #2163).
474                    self.emit_sstable_pruned();
475                }
476                present
477            }
478            None => true,
479        }
480    }

492        obs::add_counter(
493            catalog::READ_SSTABLES_PRUNED,
494            1,
495            &[(
496                catalog::attr::SSTABLE_FORMAT,
497                self.sstable_format_label().into(),
498            )],
499        );

509    /// Candidate-prune presence check using a PRE-ENCODED BTI byte-comparable key,
510    /// so a multi-generation `WHERE pk = ?` read hashes+encodes the key ONCE (the
511    /// encoding is identical for every candidate) instead of once per candidate
512    /// (issue #1575 / C4).
513    ///
514    /// `encoded` MUST equal `encode_partition_key_for_bti_trie(partition_key)`,
515    /// computed once by the caller. For a BTI reader this consults the trie with the
516    /// pre-encoded key (no re-hash); for a BIG reader (no `Partitions.db`) it falls
517    /// back to the raw-key [`might_contain_partition`] path (the bloom filter is
518    /// keyed on the raw key — there is no BTI encoding to hoist), so a mixed or
519    /// non-BTI candidate set stays correct. Semantics match
520    /// [`might_contain_partition`]: `false` is definitive absence; a BTI trie hit may
521    /// be a safe prefix-collision false positive re-verified by the partition scan;
522    /// any trie parse error is treated conservatively as "maybe present".
523    pub fn might_contain_partition_encoded(&self, partition_key: &[u8], encoded: &[u8; 9]) -> bool {
524        if self.bti_partitions_db.is_some() {
525            // Issue #2163 (roborev r7): restore the C5 range short-circuit
526            // (`locate`'s Step 1) ahead of the trie probe — see the identical
527            // restoration + rationale in `might_contain_partition`'s BTI branch.
528            // This is the candidate-prune path's OWN entry point (called from
529            // `SSTableManager::prune_candidates` for every BTI candidate), so
530            // restoring it here is what actually re-covers the multi-generation
531            // prune loop the old `locate_encoded` used to gate.
532            if self.partition_key_out_of_range(partition_key) {
533                crate::storage::sstable::read_work_counters::record_range_short_circuit();
534                return false;
535            }
536            let present = matches!(
537                self.lookup_partition_via_bti_trie_encoded(partition_key, encoded),
538                Ok(Some(_)) | Err(_)
539            );
540            if !present {
541                // Definitive trie miss on the candidate-prune path → SSTable pruned
542                // (issue #2163). The BIG branch below delegates to
543                // `might_contain_partition`, which emits its own prune, so there is
544                // no double count.
545                self.emit_sstable_pruned();
546            }
547            return present;
548        }
549        // BIG: no BTI encoding to hoist — the bloom filter hashes the raw key.
550        self.might_contain_partition(partition_key)
551    }

553    /// Enhanced partition iteration using Summary.db reader
554    ///
555    /// Note: Token-based range queries are not directly supported because Summary.db
556    /// does not store token values (Issue #218). Instead, this iterates all summary
557    /// entries and returns all partition data.
558    ///
559    /// For token-based filtering, compute tokens from partition keys after retrieval.
560    ///
561    /// ## Issue #500: Sequential-scan fallback for writer-produced SSTables
562    ///
563    /// The Summary.db → Index.db → Data.db lookup path depends on Index.db format
564    /// compatibility between writer and reader (digest format vs. raw-key format).
565    /// Locally written SSTables emit raw-key Index.db entries that the reader's
566    /// digest-based parser cannot resolve, so the lookup loop returns 0 entries
567    /// even though Summary.db enumerates the partitions correctly.
568    ///
569    /// When that happens we fall back to `sequential_scan`, which walks Data.db
570    /// directly. For V5CompressedLegacy NB SSTables (the format the writer emits),
571    /// `sequential_scan` uses the chunk-stitching path and returns every partition.
572    pub async fn iterate_all_partitions(&self) -> Result<Vec<(RowKey, ScanRow)>> {
573        // Delegates to the per-call-token sibling with the reader's own field —
574        // byte-identical to the pre-#2346 behaviour of this method (every
575        // pre-existing caller keeps its exact cancellation semantics).
576        self.iterate_all_partitions_cancellable(&self.scan_cancel)
577            .await
578    }

580    /// Like [`Self::iterate_all_partitions`], but takes an explicit PER-CALL
581    /// cancellation token instead of always reading the reader's own
582    /// [`SSTableReader::scan_cancel`] field (issue #2346).
583    ///
584    /// A cached/shared `Arc<SSTableReader>` (a future warm-handle registry) has
585    /// no `&mut self` available to set a per-request cancel flag, so this seam
586    /// lets the compaction streaming path
587    /// ([`SSTableReader::stream_all_partitions_for_compaction`]'s non-stitching
588    /// fallback) drive two concurrent enumerations with two INDEPENDENT tokens.
589    /// `iterate_all_partitions` itself is UNCHANGED — it delegates here with
590    /// `&self.scan_cancel`, so every existing caller keeps identical behaviour.
591    pub(crate) async fn iterate_all_partitions_cancellable(
592        &self,
593        scan_cancel: &crate::storage::scan_cancel::ScanCancel,
594    ) -> Result<Vec<(RowKey, ScanRow)>> {
595        // Index-random-read path (issue #2302): when a `Index.db` is present on a
596        // BIG SSTable, enumerate EVERY partition via the full Index.db
597        // partition-offset table instead of the sparse `Summary.db` samples.
598        //
599        // The historical loop walked `Summary.db` (only ~1-in-128 partitions) AND
600        // passed `data_size = 0` to `parse_partition_at_offset` (Index.db never
601        // stores partition size), so it read zero bytes per entry, resolved zero
602        // partitions, and SILENTLY fell back to a full `sequential_scan` on EVERY
603        // read — even with complete, valid components. Each partition's span is
604        // bounded instead by the successor entry's offset (the last by the
605        // data-section end), authoritative on-disk structure rather than a size
606        // guess (issue #28). Index.db offsets are in the UNCOMPRESSED data-section
607        // domain: for an uncompressed reader that equals the raw file domain; for a
608        // compressed reader the per-partition slice is mapped to its covering
609        // compression chunk(s) and decompressed (`read_compressed_offset_window`).
610        if self.index_reader.is_some() && self.bti_partitions_db.is_none() {
611            match self
612                .iterate_all_partitions_via_full_index(scan_cancel)
613                .await?
614            {
615                Some(results) => return Ok(results),
616                None => {
617                    // Present components that did NOT fully resolve: never silent.
618                    tracing::warn!(
619                        "SSTable Index.db is present but iterate_all_partitions could not \
620                         resolve every partition through the index-random-read path; \
621                         falling back to a full sequential scan of Data.db (issue #2302). \
622                         This should not happen for a well-formed BIG SSTable — the emitted \
623                         Index.db/Summary.db may be malformed or the data-section length is \
624                         inconsistent with the index offsets."
625                    );
626                }
627            }
628        } else if self.bti_partitions_db.is_none() {
629            // BIG SSTable with NO usable `index_reader` (this branch is only reached
630            // when `index_reader.is_none()`): the full-index random-read path cannot
631            // run, so every iteration falls into the sequential Data.db scan below — a
632            // real per-read perf cliff. The WARN trigger keys on Index.db availability,
633            // NOT Summary.db presence (roborev job 1612): a present Summary.db only
634            // samples ~1-in-128 partitions and never gates the full-index path, so a
635            // present-Summary/missing-Index pair is the SAME silent degradation this
636            // issue exists to kill and must be named LOUD, never dropped silently into
637            // `sequential_scan`. The sub-cause (failed-to-open vs genuinely absent)
638            // only shapes the message.
639            if self.index_present_but_unloadable {
640                // The sibling Index.db EXISTS on disk but failed to open/parse
641                // (issue #2302, roborev job 1606): distinct from a genuinely absent
642                // file — the component is malformed/truncated.
643                tracing::warn!(
644                    "SSTable Index.db is present on disk but failed to open/parse \
645                     (iterate_all_partitions cannot use the index-random-read path) \
646                     and falls back to a full sequential scan of Data.db (a per-read \
647                     perf cliff, issue #2302). The Index.db component may be malformed \
648                     or truncated — regenerate the SSTable or restore an intact Index.db."
649                );
650            } else {
651                // Index.db genuinely absent. Whether or not Summary.db loaded, the
652                // full-index path is unavailable, so this is still the degraded
653                // full-scan cliff (issues #2295/#2302). Name the absent components so
654                // an operator can spot an incomplete snapshot directory served with
655                // only Data.db (or Data.db + Summary.db) instead of silently paying
656                // the full-scan cost on every read.
657                let missing = if self.summary_reader.is_none() {
658                    "Index.db and Summary.db"
659                } else {
660                    "Index.db"
661                };
662                tracing::warn!(
663                    "SSTable has no usable random-access partition index ({missing} \
664                     absent): iterate_all_partitions falls back to a full sequential \
665                     scan of Data.db, materializing and sorting every partition on \
666                     each read (a per-read perf cliff, issues #2295/#2302). For \
667                     snapshot reads, ensure the snapshot directory holds the full \
668                     SSTable component set (Index.db + Summary.db), not just Data.db."
669                );
670            }
671        }
672
673        // Fallback path: sequential walk of Data.db.
674        // Used when Summary.db is absent OR when the Index.db lookup loop returned
675        // no entries (Issue #500: writer-produced SSTables emit a raw-key Index.db
676        // format the reader's digest-based parser does not resolve).
677        let table_id = self.scan_table_id();
678        let schema = self.schema.as_deref();
679        self.sequential_scan(&table_id, None, None, None, schema, scan_cancel)
680            .await
681    }

683    /// Whether this reader has a **usable random-access partition index** — a
684    /// loaded `Index.db` (BIG format; issue #2302 resolves CQLite-written Index.db
685    /// too) or a `Partitions.db` trie (BTI format) — that lets
686    /// [`iterate_all_partitions`](Self::iterate_all_partitions) and point lookups
687    /// avoid a full sequential Data.db scan.
688    ///
689    /// This is a **capability probe**: it must report `true` only for the exact
690    /// states that route to the non-sequential path. Both point lookups
691    /// (`index_reader.lookup_partition` / the BTI trie) and `iterate_all_partitions`
692    /// (the full-`Index.db` walk / BTI) gate on `index_reader`/`bti_partitions_db`
693    /// — never on `summary_reader`.
694    ///
695    /// Issue #2302 (roborev job 1613): `summary_reader` is DELIBERATELY excluded. A
696    /// `Summary.db` only samples ~1-in-128 partitions and never gates the
697    /// random-access route, so a present-Summary / absent-Index BIG reader is
698    /// degraded — every read falls into `sequential_scan` and WARNs. Including
699    /// `summary_reader` here made the probe report "fast path available" for that
700    /// degraded reader (a fidelity lie). Narrowing to the truly-supporting state
701    /// keeps the probe honest.
702    ///
703    /// Snapshot-completeness probe (issue #2295): a directory served with only
704    /// `Data.db` (its index siblings absent) still opens but reports `false`; a
705    /// complete component set loads `Index.db` (→ `index_reader`) and reports
706    /// `true`. All current callers (this module's own tests + the #2295 snapshot
707    /// test) use this probe to mean "does the fast/index path apply," so a single
708    /// narrowed predicate covers every caller without a split API.
709    pub fn has_partition_index(&self) -> bool {
710        self.index_reader.is_some() || self.bti_partitions_db.is_some()
711    }

713    /// Build the TableId used for fallback `sequential_scan` from header metadata.
714    ///
715    /// The reader populates `header.keyspace` / `header.table_name` from either the
716    /// SSTable header or the parent directory path. When the V5CompressedLegacy
717    /// stitching path is used, table_id matching is skipped, so any non-empty value
718    /// is accepted; for other formats this returns the qualified `keyspace.table`
719    /// form so the scan filter matches.
720    pub(in crate::storage::sstable::reader) fn scan_table_id(&self) -> TableId {
721        let keyspace = &self.header.keyspace;
722        let table_name = &self.header.table_name;
723        if !keyspace.is_empty() && !table_name.is_empty() {
724            TableId::from(format!("{}.{}", keyspace, table_name))
725        } else if !table_name.is_empty() {
726            TableId::from(table_name.as_str())
727        } else {
728            TableId::from("default")
729        }
730    }
