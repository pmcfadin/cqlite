13    /// Enhanced partition lookup using Index.db reader with promoted index support.
14    ///
15    /// `partition_key` must be the raw partition-key bytes as produced by
16    /// [`PartitionKey::to_bytes`](crate::storage::write_engine::mutation::PartitionKey::to_bytes):
17    ///
18    /// - **Single-component keys** — raw value bytes (UUID = 16 bytes, int = 4 BE bytes, etc.).
19    /// - **Multi-component (composite) keys** — `[len: u16 BE][value bytes][0x00]` per component,
20    ///   including a trailing `0x00` after the final component.
21    ///
22    /// The Index.db key_lookup map is keyed on these exact raw bytes (set when the BIG-format
23    /// parser was fixed in Issue #552).  The old digest-based path (which caused every lookup
24    /// to miss) has been removed.  On a miss the function returns `Ok(None)` so callers can
25    /// fall through to their existing sequential-scan fallback.
26    pub async fn lookup_partition_with_index(
27        &self,
28        partition_key: &[u8],
29    ) -> Result<Option<(u64, u32)>> {
30        if let Some(index_reader) = &self.index_reader {
31            // Direct raw-key lookup — O(1) HashMap lookup.
32            // Index.db entries are keyed on the raw partition key bytes since #552;
33            // no Murmur3 digest computation is needed or correct here.
34            if let Some(entry) = index_reader.lookup_partition(partition_key) {
35                debug!(
36                    "Found partition via Index.db raw-key lookup: offset={}, size={}",
37                    entry.data_offset, entry.data_size
38                );
39                return Ok(Some((entry.data_offset, entry.data_size)));
40            } else {
41                debug!(
42                    "Partition not found in Index.db for raw key (len={})",
43                    partition_key.len()
44                );
45            }
46        } else {
47            debug!("No Index.db reader available for partition lookup");
48        }
49        Ok(None)
50    }

52    /// BTI ("da") partition point lookup via the in-memory Partitions.db trie
53    /// (issue #831, building on the verified #755 primitive).
54    ///
55    /// Encodes `partition_key` into the BTI byte-comparable trie key and walks the
56    /// trie to resolve the partition's location. Returns:
57    ///
58    /// - `Ok(Some(offset))` — the UNCOMPRESSED Data.db byte offset of the
59    ///   partition. INVARIANT: this offset indexes into the DECOMPRESSED data
60    ///   section, never raw file bytes. The offset is resolved in one of two ways:
61    ///     - **NARROW** partition (`BtiPartitionLocation::DataOffset`) — the trie
62    ///       returns the Data.db offset directly.
63    ///     - **WIDE** partition (`BtiPartitionLocation::RowsOffset`) — the trie
64    ///       returns a positive offset into `Rows.db`; we deserialize that
65    ///       partition's `TrieIndexEntry` via [`resolve_rows_db_entry`] and use its
66    ///       recovered `data_position` (issue #909/#910). Both forms share the same
67    ///       uncompressed Data.db offset domain, so the caller treats them
68    ///       identically.
69    /// - `Ok(None)` — the reader is not BTI, or the key has no trie path (the
70    ///   partition is definitely absent from this SSTable).
71    /// - `Err(_)` — a structural trie parse error, or a `RowsOffset` was returned
72    ///   without an accompanying `Rows.db` (a structurally invalid BTI SSTable).
73    ///
74    /// Because the BTI trie uses path compression, a returned offset may be a
75    /// candidate for a *prefix-colliding* key. The caller (`bti_point_lookup`)
76    /// MUST verify the partition-key bytes at the resolved offset equal the
77    /// queried key before returning rows.
78    ///
79    /// [`resolve_rows_db_entry`]: crate::storage::sstable::bti::resolve_rows_db_entry
80    pub fn lookup_partition_via_bti_trie(&self, partition_key: &[u8]) -> Result<Option<u64>> {
81        use crate::storage::sstable::bti::{
82            lookup_raw_key_in_bti_partitions_db, resolve_rows_db_entry, BtiPartitionLocation,
83        };
84
85        let Some(partitions_db) = &self.bti_partitions_db else {
86            // Not a BTI reader — no trie to consult.
87            return Ok(None);
88        };
89
90        let mut cursor = std::io::Cursor::new(partitions_db.as_slice());
91        match lookup_raw_key_in_bti_partitions_db(&mut cursor, partition_key).map_err(|e| {
92            Error::corruption(format!(
93                "BTI Partitions.db trie lookup failed for partition key (len={}): {}",
94                partition_key.len(),
95                e
96            ))
97        })? {
98            Some(BtiPartitionLocation::DataOffset(off)) => {
99                debug!(
100                    "BTI trie resolved NARROW partition (key len={}) to Data.db offset {}",
101                    partition_key.len(),
102                    off
103                );
104                Ok(Some(off))
105            }
106            Some(BtiPartitionLocation::RowsOffset(rows_offset)) => {
107                // WIDE partition: the trie pointed at this partition's
108                // TrieIndexEntry inside Rows.db. Deserialize it to recover the
109                // partition's Data.db start position (`data_position`), which lives
110                // in the SAME uncompressed-offset domain the narrow path uses, so
111                // the caller can decode the partition identically (#909/#910).
112                let rows_db = self.bti_rows_db.as_ref().ok_or_else(|| {
113                    Error::corruption(format!(
114                        "BTI Partitions.db trie returned RowsOffset({}) for partition key \
115                         (len={}) but this reader has no Rows.db loaded; the SSTable is \
116                         structurally invalid (Rows.db is required for wide partitions).",
117                        rows_offset,
118                        partition_key.len()
119                    ))
120                })?;
121
122                let header = resolve_rows_db_entry(rows_db.as_slice(), rows_offset as usize)
123                    .map_err(|e| {
124                        Error::corruption(format!(
125                            "BTI Rows.db row-index entry at RowsOffset({}) is unreadable for \
126                             partition key (len={}): {}",
127                            rows_offset,
128                            partition_key.len(),
129                            e
130                        ))
131                    })?;
132
133                debug!(
134                    "BTI trie resolved WIDE partition (key len={}) via RowsOffset {} -> Data.db \
135                     position {} ({} row-index blocks)",
136                    partition_key.len(),
137                    rows_offset,
138                    header.data_position,
139                    header.block_count
140                );
141                Ok(Some(header.data_position))
142            }
143            None => Ok(None),
144        }
145    }

147    /// Cheap presence oracle: can this SSTable possibly contain `partition_key`?
148    ///
149    /// Used to prune SSTables before a partition-targeted scan (the query engine's
150    /// `WHERE pk = ?` fast path). Returning `false` MUST be definitive — the SSTable
151    /// is then skipped entirely without being parsed — so this only ever returns
152    /// `false` for an authoritative "absent" signal:
153    ///
154    /// - **BTI ("da")** readers have no bloom filter; the Partitions.db trie is the
155    ///   authoritative present/absent oracle. A trie miss (`Ok(None)`) is definitive
156    ///   absence. A trie hit may be a prefix-collision candidate, which is a safe
157    ///   *false positive* here (the partition scan re-verifies the key). Any trie
158    ///   parse error is treated conservatively as "maybe present".
159    /// - **BIG-format** readers consult the bloom filter, which never reports false
160    ///   negatives: `might_contain == false` is definitive absence. With no bloom
161    ///   filter loaded we cannot prune, so we conservatively return `true`.
162    ///
163    /// `partition_key` must be the raw partition-key bytes (same encoding the bloom
164    /// filter and Index.db/BTI trie are keyed on).
165    pub fn might_contain_partition(&self, partition_key: &[u8]) -> bool {
166        if self.bti_partitions_db.is_some() {
167            // BTI: trie miss is authoritative absence; any error is conservative.
168            return matches!(
169                self.lookup_partition_via_bti_trie(partition_key),
170                Ok(Some(_)) | Err(_)
171            );
172        }
173        match &self.bloom_filter {
174            Some(bloom) => bloom.might_contain(partition_key),
175            None => true,
176        }
177    }

179    /// Enhanced partition lookup using schema-driven key digest computation
180    pub async fn lookup_partition_with_schema_context(
181        &self,
182        partition_key: &[u8],
183        parsing_context: &ParsingContext,
184    ) -> Result<Option<(u64, u32)>> {
185        if let Some(index_reader) = &self.index_reader {
186            // Compute the schema-driven key digest for Index.db lookup
187            let key_digest =
188                self.compute_partition_key_digest_with_schema(partition_key, parsing_context)?;
189
190            // Use spec-compliant Index.db reader for partition lookup
191            if let Some(entry) = index_reader.lookup_partition(&key_digest) {
192                debug!(
193                    "Found partition via schema-driven Index.db: offset={}, size={}",
194                    entry.data_offset, entry.data_size
195                );
196                return Ok(Some((entry.data_offset, entry.data_size)));
197            }
198        }
199        Ok(None)
200    }

202    /// Enhanced partition iteration using Summary.db reader
203    ///
204    /// Note: Token-based range queries are not directly supported because Summary.db
205    /// does not store token values (Issue #218). Instead, this iterates all summary
206    /// entries and returns all partition data.
207    ///
208    /// For token-based filtering, compute tokens from partition keys after retrieval.
209    ///
210    /// ## Issue #500: Sequential-scan fallback for writer-produced SSTables
211    ///
212    /// The Summary.db → Index.db → Data.db lookup path depends on Index.db format
213    /// compatibility between writer and reader (digest format vs. raw-key format).
214    /// Locally written SSTables emit raw-key Index.db entries that the reader's
215    /// digest-based parser cannot resolve, so the lookup loop returns 0 entries
216    /// even though Summary.db enumerates the partitions correctly.
217    ///
218    /// When that happens we fall back to `sequential_scan`, which walks Data.db
219    /// directly. For V5CompressedLegacy NB SSTables (the format the writer emits),
220    /// `sequential_scan` uses the chunk-stitching path and returns every partition.
221    pub async fn iterate_all_partitions(&self) -> Result<Vec<(RowKey, Value)>> {
222        if let Some(summary_reader) = &self.summary_reader {
223            let entries = summary_reader.get_entries();
224            let mut results = Vec::new();
225
226            for entry in entries {
227                // Use Summary.db entry to find the corresponding Index.db entry
228                if let Some(_index_reader) = &self.index_reader {
229                    // The summary entry provides a position in Index.db
230                    // We need to read the partition data from Data.db
231
232                    // For now, use the partition key from the summary entry
233                    let partition_key_bytes = &entry.partition_key;
234
235                    // Look up the partition in Index.db to get the actual data offset
236                    if let Some((data_offset, data_size)) = self
237                        .lookup_partition_with_index(partition_key_bytes)
238                        .await?
239                    {
240                        // Convert Index.db relative offset to absolute file offset
241                        // Index.db offsets are relative to data section start (after compression header)
242                        let absolute_offset = data_offset + self.actual_header_size as u64;
243
244                        // Read and parse the actual partition data from Data.db
245                        match self
246                            .parse_partition_at_offset(absolute_offset, data_size)
247                            .await?
248                        {
249                            Some(partition_entries) => {
250                                for (row_key, value) in partition_entries {
251                                    results.push((row_key, value));
252                                }
253                            }
254                            None => {
255                                debug!("Failed to parse partition at offset {}", absolute_offset);
256                            }
257                        }
258                    }
259                } else {
260                    log::error!("Index reader not available for partition iteration");
261                    return Err(Error::corruption(
262                        "Index reader required for partition iteration - synthetic data not allowed for Issue #35",
263                    ));
264                }
265            }
266
267            // Only trust the index-based path when EVERY summary entry was resolved.
268            // Partial resolution silently drops the unresolved entries; defaulting to
269            // `sequential_scan` in that case is strictly safer and still correct on
270            // real Cassandra SSTables (sequential_scan returns the same partitions
271            // when the index resolves them all).
272            if results.len() == entries.len() && !entries.is_empty() {
273                debug!("Partition iteration found {} entries", results.len());
274                return Ok(results);
275            }
276
277            debug!(
278                "Index.db lookup resolved {}/{} summary entries; \
279                 falling back to sequential_scan (Issue #500)",
280                results.len(),
281                entries.len()
282            );
283        }
284
285        // Fallback path: sequential walk of Data.db.
286        // Used when Summary.db is absent OR when the Index.db lookup loop returned
287        // no entries (Issue #500: writer-produced SSTables emit a raw-key Index.db
288        // format the reader's digest-based parser does not resolve).
289        let table_id = self.scan_table_id();
290        let schema = self.schema.as_deref();
291        self.sequential_scan(&table_id, None, None, None, schema)
292            .await
293    }

295    /// Build the TableId used for fallback `sequential_scan` from header metadata.
296    ///
297    /// The reader populates `header.keyspace` / `header.table_name` from either the
298    /// SSTable header or the parent directory path. When the V5CompressedLegacy
299    /// stitching path is used, table_id matching is skipped, so any non-empty value
300    /// is accepted; for other formats this returns the qualified `keyspace.table`
301    /// form so the scan filter matches.
302    fn scan_table_id(&self) -> TableId {
303        let keyspace = &self.header.keyspace;
304        let table_name = &self.header.table_name;
305        if !keyspace.is_empty() && !table_name.is_empty() {
306            TableId::from(format!("{}.{}", keyspace, table_name))
307        } else if !table_name.is_empty() {
308            TableId::from(table_name.as_str())
309        } else {
310            TableId::from("default")
311        }
312    }

363    /// Enhanced get method using spec readers for efficient lookup
364    pub async fn get_with_spec_readers(
365        &self,
366        table_id: &TableId,
367        key: &RowKey,
368    ) -> Result<Option<Value>> {
369        // Step 1: Use bloom filter for existence check
370        if let Some(bloom_filter) = &self.bloom_filter {
371            if !bloom_filter.might_contain(key.as_bytes()) {
372                debug!("Bloom filter indicates key does not exist");
373                return Ok(None);
374            }
375        }
376
377        // Step 2: Use Index.db reader for precise partition lookup
378        if let Some((offset, size)) = self.lookup_partition_with_index(key.as_bytes()).await? {
379            debug!("Using Index.db lookup: offset={}, size={}", offset, size);
380            return self.read_value_at_offset(offset, size).await;
381        }
382
383        // Step 3: Fallback to existing methods
384        debug!("Falling back to legacy lookup methods");
385        self.get(table_id, key).await
386    }
