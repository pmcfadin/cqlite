87/// Returns `true` when memory-mapped reads are force-enabled via the
88/// `CQLITE_USE_MMAP` environment variable.
89///
90/// Accepts `1`, `true`, `yes`, `on` (case-insensitive). Any other value — or
91/// an unset variable — leaves the decision to [`Config`]. This is an opt-in
92/// escape hatch for ad-hoc local use without threading a custom config.
93fn mmap_enabled_via_env() -> bool {
94    std::env::var("CQLITE_USE_MMAP")
95        .ok()
96        .as_deref()
97        .map(parse_truthy_env)
98        .unwrap_or(false)
99}

101/// Parse a truthy environment-variable value (`1`/`true`/`yes`/`on`,
102/// case-insensitive). Split out so it can be unit-tested without mutating the
103/// process-global environment (which would race other `open()` tests).
104fn parse_truthy_env(value: &str) -> bool {
105    matches!(
106        value.trim().to_ascii_lowercase().as_str(),
107        "1" | "true" | "yes" | "on"
108    )
109}

111/// Parse a [`DiskAccessMode`] from a string (`auto`/`buffered`/`mmap`/`direct`,
112/// case-insensitive). Returns `None` for unrecognized values so callers can
113/// keep the configured default. Pure for unit-testing without env mutation.
114fn parse_disk_access_mode(value: &str) -> Option<DiskAccessMode> {
115    match value.trim().to_ascii_lowercase().as_str() {
116        "auto" => Some(DiskAccessMode::Auto),
117        "buffered" | "buffer" => Some(DiskAccessMode::Buffered),
118        "mmap" | "mapped" => Some(DiskAccessMode::Mmap),
119        "direct" | "directio" | "direct_io" | "o_direct" => Some(DiskAccessMode::Direct),
120        _ => None,
121    }
122}

124/// Parse a [`PrefetchMode`] from a string (`off`/`sequential`/`willneed`/`auto`).
125/// Returns `None` for unrecognized values. Pure for unit-testing.
126fn parse_prefetch_mode(value: &str) -> Option<PrefetchMode> {
127    match value.trim().to_ascii_lowercase().as_str() {
128        "off" | "none" | "no" => Some(PrefetchMode::Off),
129        "sequential" | "seq" => Some(PrefetchMode::Sequential),
130        "willneed" | "will_need" | "will-need" => Some(PrefetchMode::WillNeed),
131        "auto" => Some(PrefetchMode::Auto),
132        _ => None,
133    }
134}

152/// Best-effort total physical RAM in bytes, or `None` when it cannot be
153/// determined on this platform. Used by [`DiskAccessMode::Auto`] to decide when
154/// a file is large enough to warrant page-cache-bypassing direct I/O.
155fn system_memory_bytes() -> Option<u64> {
156    #[cfg(unix)]
157    {
158        // SAFETY: `sysconf` is a pure query with no pointer arguments.
159        let pages = unsafe { libc::sysconf(libc::_SC_PHYS_PAGES) };
160        let page_size = unsafe { libc::sysconf(libc::_SC_PAGESIZE) };
161        if pages > 0 && page_size > 0 {
162            return Some((pages as u64).saturating_mul(page_size as u64));
163        }
164        None
165    }
166    #[cfg(not(unix))]
167    {
168        None
169    }
170}

172/// Decide which disk-access backend to use for a Data.db file.
173///
174/// Pure function (system memory injected) so the [`DiskAccessMode::Auto`]
175/// heuristic can be unit-tested deterministically. Resolution rules:
176/// - explicit `Buffered`/`Mmap`/`Direct` are returned unchanged (the caller
177///   applies graceful fallback if the OS refuses the backend);
178/// - `Auto` returns `Buffered` for files below `mmap_min_size_bytes`, `Direct`
179///   when the file exceeds `memory_fraction` of `system_memory` (and memory is
180///   known and direct I/O is available on this platform), otherwise `Mmap`.
181///
182/// The deprecated `use_mmap` flag / `CQLITE_USE_MMAP` env is folded in by the
183/// caller (it promotes a `Buffered` request to `Mmap`), so it is not an input
184/// here; an explicit mode always takes precedence.
185fn resolve_disk_access_mode(
186    configured: DiskAccessMode,
187    file_size: u64,
188    mmap_min_size_bytes: u64,
189    memory_fraction: f64,
190    system_memory: Option<u64>,
191    direct_io_available: bool,
192) -> DiskAccessMode {
193    // Zero-length files cannot be mapped and have nothing to read directly;
194    // always use buffered I/O for them regardless of the requested mode.
195    if file_size == 0 {
196        return DiskAccessMode::Buffered;
197    }
198    match configured {
199        DiskAccessMode::Buffered => DiskAccessMode::Buffered,
200        DiskAccessMode::Mmap => DiskAccessMode::Mmap,
201        DiskAccessMode::Direct => DiskAccessMode::Direct,
202        DiskAccessMode::Auto => {
203            if file_size < mmap_min_size_bytes {
204                return DiskAccessMode::Buffered;
205            }
206            let fraction = if memory_fraction.is_finite() && memory_fraction > 0.0 {
207                memory_fraction.min(1.0)
208            } else {
209                0.5
210            };
211            if direct_io_available {
212                if let Some(mem) = system_memory {
213                    let threshold = (mem as f64 * fraction) as u64;
214                    if file_size > threshold {
215                        return DiskAccessMode::Direct;
216                    }
217                }
218            }
219            DiskAccessMode::Mmap
220        }
221    }
222}

236/// Resolve [`PrefetchMode::Auto`] into the concrete advice the mmap backend
237/// should issue, or `None` for "no advice".
238///
239/// `memmap2::Advice` / `Mmap::advise` (madvise) are Unix-only, so this and its
240/// single call site are gated to `#[cfg(unix)]`. On non-Unix targets the mmap
241/// backend simply issues no read-ahead advice.
242#[cfg(unix)]
243fn mmap_advice_for(prefetch: PrefetchMode) -> Option<memmap2::Advice> {
244    match prefetch {
245        PrefetchMode::Off => None,
246        // Sequential is the right default for full-file scans: aggressive
247        // read-ahead plus drop-behind so a scan does not pin the whole file.
248        PrefetchMode::Sequential | PrefetchMode::Auto => Some(memmap2::Advice::Sequential),
249        PrefetchMode::WillNeed => Some(memmap2::Advice::WillNeed),
250    }
251}

253impl SSTableReader {
254    /// Open an SSTable file for reading
255    pub async fn open(path: &Path, config: &Config, platform: Arc<Platform>) -> Result<Self> {
256        // Resolve the disk-access backend (buffered / mmap / direct, or auto)
257        // from config + environment overrides. See `Config::storage.disk_access_mode`.
258        let mut reader_config = SSTableReaderConfig::default();
259        reader_config.use_mmap = config.storage.use_mmap || mmap_enabled_via_env();
260        reader_config.mmap_min_size_bytes = config.storage.mmap_min_size_bytes;
261
262        let file_size = tokio::fs::metadata(path).await?.len();
263
264        // The explicit mode comes from env > config. The legacy `use_mmap` flag
265        // is folded in by promoting an otherwise-`Buffered` request to `Mmap` so
266        // the old opt-in keeps working (`Auto` and explicit non-buffered modes
267        // are left untouched, so a real backend decision always wins).
268        let configured_mode = disk_access_mode_via_env().unwrap_or(config.storage.disk_access_mode);
269        let configured_mode =
270            if reader_config.use_mmap && matches!(configured_mode, DiskAccessMode::Buffered) {
271                DiskAccessMode::Mmap
272            } else {
273                configured_mode
274            };
275
276        let resolved_mode = resolve_disk_access_mode(
277            configured_mode,
278            file_size,
279            reader_config.mmap_min_size_bytes as u64,
280            config.storage.direct_io_memory_fraction,
281            system_memory_bytes(),
282            direct_io_available(),
283        );
284
285        // Build both the shared point-read source and the per-scan factory from
286        // the same backend decision so concurrent scans get independent cursors
287        // (issue #815). `ScanSource::Mapped` shares the same `Arc<Mmap>` so no
288        // extra mapping is created per scan. Every non-buffered backend degrades
289        // gracefully to buffered I/O if the OS/filesystem refuses it.
290        let prefetch = prefetch_mode_via_env().unwrap_or(config.storage.prefetch);
291        let (source, scan_source) = Self::build_block_sources(
292            path,
293            file_size,
294            resolved_mode,
295            prefetch,
296            config.storage.direct_io_prefetch_bytes,
297        )
298        .await?;
299        let file = Arc::new(Mutex::new(source));
300
301        // Parse header - read available bytes, not a fixed size
302        // NOTE: For NB format files (Cassandra 4.x+), Data.db often contains compressed row data
303        // with no embedded header. The header.rs module detects this via filename pattern and
304        // returns a minimal header loaded from Statistics.db instead.
305        let header_size = std::cmp::min(4096, file_size as usize);
306        let mut header_buffer = vec![0u8; header_size];
307        {
308            let mut file_guard = file.lock().await;
309            let bytes_read = file_guard.read(&mut header_buffer).await?;
310            header_buffer.truncate(bytes_read);
311        }
312
313        // Derive VersionGates from the SSTable filename BEFORE header parsing so
314        // parse_header_with_version_detection can receive them.  Gates are derived
315        // solely from the filename and need no file I/O, so this is safe to do here.
316        //
317        // Falls back to nb-compatible BIG gates when the filename is not a valid
318        // SSTable descriptor (e.g. paths used in unit tests).  Using nb-fallback
319        // maintains existing behaviour — the gates will not change parsing
320        // decisions until VG3 actually flips behaviour.
321        let version_gates = Arc::new(match VersionGates::from_path(path) {
322            Ok(gates) => gates,
323            Err(e) => {
324                log::debug!(
325                    "SSTableReader::open: could not derive VersionGates from {:?} ({}); \
326                     defaulting to nb-compatible BIG gates",
327                    path,
328                    e
329                );
330                VersionGates::Big(BigVersionGates::nb_fallback())
331            }
332        });
333
334        // VG5 / Issue #831 / #909: BTI ("da") read support.
335        //
336        // BTI SSTables use a Partitions.db trie + Rows.db row index instead of
337        // Index.db/Summary.db. We load BOTH (tiny) tries fully into memory here so
338        // the point-lookup path (`lookup_partition_via_bti_trie` /
339        // `bti_point_lookup`) can walk them for O(log n) partition resolution:
340        //   - Partitions.db resolves a partition key to either a direct Data.db
341        //     offset (NARROW partition) or a positive RowsOffset (WIDE partition).
342        //   - Rows.db, indexed by that RowsOffset, recovers the wide partition's
343        //     Data.db position (issue #909/#910).
344        //
345        // Loading Partitions.db + Rows.db is the ONLY BTI-specific step in open():
346        // the rest of the flow (header / compression / Statistics-driven schema)
347        // tolerates the absent Index.db/Summary.db gracefully (those loaders
348        // return None).
349        let (bti_partitions_db, bti_rows_db) = if matches!(*version_gates, VersionGates::Bti(_)) {
350            let base = extract_sstable_base_name(path).ok_or_else(|| {
351                Error::unsupported_format(format!(
352                    "BTI (da) SSTable '{}' has a non-standard filename; cannot derive the \
353                         sibling Partitions.db name required for trie point lookup (#831).",
354                    path.display()
355                ))
356            })?;
357            let parent = path.parent().unwrap_or_else(|| Path::new("."));
358            let partitions_path = parent.join(format!("{}-Partitions.db", base));
359            let partitions_bytes = tokio::fs::read(&partitions_path).await.map_err(|e| {
360                Error::unsupported_format(format!(
361                    "BTI (da) SSTable '{}' is missing its sibling Partitions.db trie \
362                         (expected '{}'): {}. BTI read support requires Partitions.db for \
363                         partition-key point lookup (#831).",
364                    path.display(),
365                    partitions_path.display(),
366                    e
367                ))
368            })?;
369
370            // Rows.db carries the within-partition row index for WIDE
371            // partitions (issue #909/#910). It is ALWAYS emitted for a BTI
372            // SSTable (possibly 0 bytes for a narrow-only table), so a missing
373            // Rows.db is a structural error. A 0-byte file is valid: no
374            // partition resolved to a positive RowsOffset, so the point-lookup
375            // path never indexes into it.
376            let rows_path = parent.join(format!("{}-Rows.db", base));
377            let rows_bytes = tokio::fs::read(&rows_path).await.map_err(|e| {
378                Error::unsupported_format(format!(
379                    "BTI (da) SSTable '{}' is missing its sibling Rows.db row-index trie \
380                         (expected '{}'): {}. BTI read support requires Rows.db to resolve \
381                         wide-partition point lookups (#909/#910).",
382                    path.display(),
383                    rows_path.display(),
384                    e
385                ))
386            })?;
387
388            (Some(Arc::new(partitions_bytes)), Some(Arc::new(rows_bytes)))
389        } else {
390            let none: Option<Arc<Vec<u8>>> = None;
391            (none.clone(), none)
392        };
393
394        let config = crate::cql::config::ParserConfig::default();
395        let parser = SSTableParser::new(config)?;
396        // Parse the header using enhanced version detection - strict error propagation.
397        // VersionGates are passed so VG3 can flip version-sensitive parsing decisions
398        // inside header parsing without re-deriving gates from the filename.
399        let header = parse_header_with_version_detection(&header_buffer, path, &version_gates)
400            .await
401            .map_err(|e| {
402                Error::corruption(format!(
403                    "Failed to parse SSTable header for file '{}': {}. This indicates either \
404                     file corruption or an unsupported SSTable format. File size: {} bytes, \
405                     header buffer size: {} bytes.",
406                    path.display(),
407                    e,
408                    file_size,
409                    header_buffer.len()
410                ))
411            })?;
412        let header_size = calculate_actual_header_size(&header, &header_buffer)?;
413
414        // Schema extraction deferred until after Statistics.db columns are loaded (Issue #163)
415        // See schema extraction code after statistics_reader loading below
416
417        // Seek to start of data section
418        {
419            let mut file_guard = file.lock().await;
420            file_guard
421                .seek(std::io::SeekFrom::Start(header_size as u64))
422                .await?;
423        }
424
425        // Initialize compression reader with improved format detection
426        let compression_reader = detect_and_initialize_compression(&header, path).await?;
427
428        // Load CompressionInfo.db for chunked decompression (if it exists)
429        let compression_info = Self::load_compression_info_metadata(path, &platform).await?;
430
431        // Pre-validate component architecture for better error handling
432        let components = Self::detect_component_files(path).await?;
433        if !components.is_empty() {
434            let integrity_issues = Self::validate_component_integrity(path, &components).await?;
435            if !integrity_issues.is_empty() {
436                log::warn!(
437                    "Component integrity issues detected but proceeding with loading: {:?}",
438                    integrity_issues
439                );
440            }
441        }
442
443        // Load index if available (supports both integrated and component-based)
444        let index = Self::load_index(&file, &header, &platform, path).await?;
445
446        // Load bloom filter if available (supports both integrated and component-based)
447        let bloom_filter = Self::load_bloom_filter(&file, &header, &platform, path).await?;
448
449        // Load spec readers for enhanced metadata and lookups
450        let index_reader = Self::load_index_reader(path, &platform).await;
451        let summary_reader = Self::load_summary_reader(path, &platform).await;
452        let statistics_reader = Self::load_statistics_reader(path, &platform).await;
453
454        // Extract SerializationHeader columns from Statistics.db (Issue #163)
455        // This enables schema extraction for V5CompressedLegacy format
456        let mut header = header; // Make mutable to populate columns
457        if let Some(ref stats_reader) = statistics_reader {
458            let statistics = stats_reader.statistics();
459            let partition_columns = &statistics.serialization_header_partition_keys;
460            let clustering_columns = &statistics.serialization_header_clustering_keys;
461            let regular_columns = &statistics.serialization_header_columns;
462
463            if !partition_columns.is_empty()
464                || !clustering_columns.is_empty()
465                || !regular_columns.is_empty()
466            {
467                log::debug!(
468                    "Populating header columns from Statistics.db SerializationHeader: {} partition keys, {} clustering keys, {} regular columns",
469                    partition_columns.len(),
470                    clustering_columns.len(),
471                    regular_columns.len()
472                );
473
474                let mut merged_columns = Vec::with_capacity(
475                    partition_columns.len() + clustering_columns.len() + regular_columns.len(),
476                );
477                merged_columns.extend_from_slice(partition_columns);
478                merged_columns.extend_from_slice(clustering_columns);
479                merged_columns.extend_from_slice(regular_columns);
480
481                header.columns = merged_columns;
482            }
483        }
484
485        // Extract schema from header for V5.0+ formats (after columns are populated)
486        let schema = if matches!(
487            header.cassandra_version,
488            CassandraVersion::V5_0NewBig
489                | CassandraVersion::V5_0Bti
490                | CassandraVersion::V5_0DataFormat
491                | CassandraVersion::V5_0FormatC
492                | CassandraVersion::V5_0FormatD
493                | CassandraVersion::V5_0FormatE
494                | CassandraVersion::V5_0FormatF
495                | CassandraVersion::V5_0FormatG
496        ) {
497            match TableSchema::from_sstable_header(&header) {
498                Ok(s) => {
499                    log::debug!(
500                        "Extracted schema from SSTable header: {}.{} ({} columns, {} partition keys, {} clustering keys)",
501                        s.keyspace,
502                        s.table,
503                        s.columns.len(),
504                        s.partition_keys.len(),
505                        s.clustering_keys.len()
506                    );
507                    Some(Arc::new(s))
508                }
509                Err(e) => {
510                    log::warn!(
511                        "Failed to extract schema from SSTable header for {}: {}. Schema-aware parsing will not be available.",
512                        path.display(),
513                        e
514                    );
515                    None
516                }
517            }
518        } else {
519            // Legacy formats don't have schema in header
520            None
521        };
522
523        // Derive block_count from CompressionInfo.db when available — this is the
524        // authoritative source for compressed SSTables (no-heuristics mandate #28).
525        // Each entry in chunk_offsets corresponds to one compressed block in Data.db.
526        let block_count = compression_info
527            .as_ref()
528            .map(|ci| ci.chunk_offsets.len() as u64)
529            .unwrap_or(0);
530
531        let stats = SSTableReaderStats {
532            file_size,
533            entry_count: header.stats.row_count,
534            table_count: 1, // Will be updated as we discover tables
535            block_count,
536            index_size: 0,        // Will be updated if index is loaded
537            bloom_filter_size: 0, // Will be updated if bloom filter is loaded
538            compression_ratio: header.stats.compression_ratio,
539            cache_hit_rate: 0.0,
540        };
541
542        // Extract generation from filename or use default
543        let generation = extract_generation_from_path(path);
544
545        Ok(Self {
546            file_path: path.to_path_buf(),
547            file,
548            scan_source,
549            header,
550            parser,
551            index,
552            bloom_filter,
553            compression_reader,
554            block_meta_cache: HashMap::new(),
555            block_cache: HashMap::new(),
556            config: reader_config,
557            platform,
558            stats,
559            cache_hits: AtomicU64::new(0),
560            cache_misses: AtomicU64::new(0),
561            #[cfg(feature = "tombstones")]
562            tombstone_merger: TombstoneMerger::new(),
563            generation,
564            actual_header_size: header_size,
565            index_reader,
566            summary_reader,
567            statistics_reader,
568            schema_registry: None, // Will be set by set_schema_registry() after construction
569            schema,
570            udt_registry: None, // Will be set when available for UDT-aware parsing
571            compression_info: compression_info.map(Arc::new),
572            version_gates,
573            bti_partitions_db,
574            bti_rows_db,
575        })
576    }

596    /// Open a buffered point-read source and its per-scan factory. Shared by the
597    /// buffered backend and as the graceful fallback for mmap/direct.
598    async fn open_buffered_sources(path: &Path) -> Result<(BlockSource, ScanSource)> {
599        Ok((
600            BlockSource::buffered(File::open(path).await?),
601            ScanSource::Buffered,
602        ))
603    }

605    /// Build the shared point-read [`BlockSource`] and the per-scan
606    /// [`ScanSource`] for a resolved [`DiskAccessMode`].
607    ///
608    /// `prefetch` selects read-ahead advice for mmap and (together with
609    /// `prefetch_bytes`) the direct-I/O read-ahead window. Each non-buffered
610    /// backend degrades gracefully to buffered I/O if the OS or filesystem
611    /// refuses it (mmap can fail on network mounts; direct I/O is unsupported on
612    /// some platforms/filesystems), so opening never fails purely because a
613    /// faster backend is unavailable.
614    async fn build_block_sources(
615        path: &Path,
616        file_size: u64,
617        mode: DiskAccessMode,
618        prefetch: PrefetchMode,
619        prefetch_bytes: usize,
620    ) -> Result<(BlockSource, ScanSource)> {
621        // With prefetch off, use the minimal aligned read-ahead the direct
622        // backend still requires (a single block, rounded up inside the cursor).
623        let direct_window = if matches!(prefetch, PrefetchMode::Off) {
624            1
625        } else {
626            prefetch_bytes
627        };
628        match mode {
629            DiskAccessMode::Buffered => Self::open_buffered_sources(path).await,
630            DiskAccessMode::Mmap => match Self::map_file(path) {
631                Ok(mmap) => {
632                    log::debug!(
633                        "Opened {} via memory map ({} bytes)",
634                        path.display(),
635                        file_size
636                    );
637                    // Best-effort read-ahead advice (Unix-only; madvise has no
638                    // Windows equivalent here). Failure is non-fatal.
639                    #[cfg(unix)]
640                    if let Some(advice) = mmap_advice_for(prefetch) {
641                        if let Err(e) = mmap.advise(advice) {
642                            log::debug!(
643                                "madvise({:?}) on {} failed: {}",
644                                advice,
645                                path.display(),
646                                e
647                            );
648                        }
649                    }
650                    let mmap = Arc::new(mmap);
651                    Ok((BlockSource::mapped(mmap.clone()), ScanSource::Mapped(mmap)))
652                }
653                Err(e) => {
654                    log::warn!(
655                        "Memory-mapping {} failed ({}); falling back to buffered I/O",
656                        path.display(),
657                        e
658                    );
659                    Self::open_buffered_sources(path).await
660                }
661            },
662            DiskAccessMode::Direct => {
663                #[cfg(unix)]
664                {
665                    match source::DirectCursor::open(path, direct_window) {
666                        Ok(cursor) => {
667                            log::debug!(
668                                "Opened {} via direct I/O ({} bytes, {}-byte window)",
669                                path.display(),
670                                file_size,
671                                direct_window
672                            );
673                            Ok((
674                                BlockSource::direct(cursor),
675                                ScanSource::Direct {
676                                    window: direct_window,
677                                },
678                            ))
679                        }
680                        Err(e) => {
681                            log::warn!(
682                                "Direct I/O on {} failed ({}); falling back to buffered I/O",
683                                path.display(),
684                                e
685                            );
686                            Self::open_buffered_sources(path).await
687                        }
688                    }
689                }
690                #[cfg(not(unix))]
691                {
692                    let _ = direct_window;
693                    log::warn!(
694                        "Direct I/O is unavailable on this platform; using buffered I/O for {}",
695                        path.display()
696                    );
697                    Self::open_buffered_sources(path).await
698                }
699            }
700            // `resolve_disk_access_mode` never yields `Auto`; handle defensively.
701            DiskAccessMode::Auto => Self::open_buffered_sources(path).await,
702        }
703    }

705    /// Memory-map an SSTable file read-only.
706    ///
707    /// # Safety / correctness
708    ///
709    /// The returned [`Mmap`](memmap2::Mmap) aliases the file's bytes for its
710    /// entire lifetime. SSTables are immutable once written, and CQLite treats
711    /// them as read-only inputs, so this matches Cassandra's own mmap read
712    /// strategy. Mutating the underlying file while a reader is open is
713    /// undefined behaviour — callers must not do so.
714    ///
715    /// Note that only the initial mapping is fallible here. Once mapped, a later
716    /// page fault — caused by truncation, deletion, or a network/overlay
717    /// filesystem hiccup — raises `SIGBUS` and **cannot** be recovered as an
718    /// `io::Error`. This is why mmap is opt-in and gated on immutable local
719    /// files; see [`Config`]'s `storage.use_mmap` for the full constraints.
720    fn map_file(path: &Path) -> Result<memmap2::Mmap> {
721        let std_file = std::fs::File::open(path)?;
722        // SAFETY: read-only mapping of a file assumed immutable for the
723        // reader's lifetime; see the function-level note above.
724        let mmap = unsafe { memmap2::MmapOptions::new().map(&std_file)? };
725        Ok(mmap)
726    }

728    /// Load CompressionInfo.db metadata for chunked reading
729    async fn load_compression_info_metadata(
730        path: &Path,
731        _platform: &Arc<Platform>,
732    ) -> Result<Option<CompressionInfo>> {
733        use tokio::fs::File;
734        use tokio::io::AsyncReadExt;
735
736        // Try to find CompressionInfo.db in same directory
737        let parent_dir = path.parent().unwrap_or(Path::new("."));
738        let base_name = path.file_stem().and_then(|s| s.to_str()).and_then(|s| {
739            // Extract base name: "nb-1-big-Data.db" -> "nb-1-big"
740            let parts: Vec<&str> = s.split('-').collect();
741            if parts.len() >= 4 {
742                Some(parts[0..3].join("-"))
743            } else {
744                None
745            }
746        });
747
748        if let Some(base) = base_name {
749            let compression_info_path = parent_dir.join(format!("{}-CompressionInfo.db", base));
750            if compression_info_path.exists() {
751                let mut file = File::open(&compression_info_path).await?;
752                let mut buffer = Vec::new();
753                file.read_to_end(&mut buffer).await?;
754
755                match CompressionInfo::parse(&buffer) {
756                    Ok(info) => {
757                        log::debug!(
758                            "Loaded CompressionInfo: algorithm={}, chunk_length={}, chunks={}",
759                            info.algorithm,
760                            info.chunk_length,
761                            info.chunk_offsets.len()
762                        );
763                        return Ok(Some(info));
764                    }
765                    Err(e) => {
766                        log::warn!("Failed to parse CompressionInfo.db: {}", e);
767                    }
768                }
769            }
770        }
771
772        Ok(None)
773    }

848    /// Get the SSTable format version string
849    pub fn format_version(&self) -> Result<String> {
850        let filename = self
851            .file_path
852            .file_name()
853            .and_then(|f| f.to_str())
854            .ok_or_else(|| {
855                Error::InvalidPath(format!("Invalid SSTable filename: {:?}", self.file_path))
856            })?;
857
858        let parts: Vec<&str> = filename.split('-').collect();
859        if parts.is_empty() {
860            return Err(Error::InvalidFormat(format!(
861                "Cannot extract format version from filename: {}",
862                filename
863            )));
864        }
865
866        Ok(parts[0].to_string())
867    }

881    /// Extract write time from entry metadata
882    pub fn extract_write_time_from_entry(&self, _key: &RowKey, value: &Value) -> i64 {
883        use log::warn;
884
885        match value {
886            Value::Tombstone(info) => info.deletion_time,
887            _ => std::time::SystemTime::now()
888                .duration_since(std::time::UNIX_EPOCH)
889                .map(|d| d.as_micros() as i64)
890                .unwrap_or_else(|e| {
891                    warn!("Failed to get system time: {}; using fallback value 0", e);
892                    0
893                }),
894        }
895    }
