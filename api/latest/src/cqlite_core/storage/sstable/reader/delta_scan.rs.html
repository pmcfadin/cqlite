1//! Delta-scan record model and streaming API for CDC-style Parquet projections
2//! (Epic #696, Issue #697 types, Issue #698 `scan_delta` implementation).
3//!
4//! A flushed SSTable is a delta, not a snapshot.  Projecting it to Parquet
5//! correctly requires carrying per-cell write-timestamps and representing every
6//! delete shape Cassandra produces — otherwise a downstream union of per-flush
7//! files resurrects deleted data and merges stale cells.
8//!
9//! ## Design contract
10//!
11//! One SSTable generation in, faithful change events out.  Reconciliation
12//! (LWW merge, tombstone application, TTL filtering) is deliberately the
13//! downstream consumer's responsibility.
14//!
15//! ## Types
16//!
17//! | Type | Purpose |
18//! |------|---------|
19//! | [`DeltaRecord`] | Discriminated union of every change shape |
20//! | [`CellDelta`] | Per-cell value + writetime + TTL + collection flag |
21//! | [`RangeBound`] | Typed, possibly-prefix clustering-key bound |
22//! | [`RowKeys`] | Partition key + optional clustering columns |
23//! | [`CellMeta`] | Row liveness metadata (timestamp, TTL) |
24//!
25//! ## Streaming API
26//!
27//! [`scan_delta`] streams [`DeltaRecord`]s from a single SSTable generation
28//! (one `Data.db` file) via a [`tokio::sync::mpsc`] channel.  It lives on the
29//! reader layer and does **not** route through the query engine (which merges
30//! generations and suppresses tombstones — the opposite contract).
31//!
32//! Row/range/partition tombstone emission is Issue #699 scope.  This version
33//! errors or returns early on delete-bearing input; upsert and static-upsert
34//! paths are complete and correct.
35//!
36//! ## Feature gate
37//!
38//! Everything in this module is behind `feature = "delta-scan"` and will not
39//! compile into the default crate build.

49// ScanSummary — scan-level aggregate statistics (Issue #700, DS4)
50// ---------------------------------------------------------------------------
51
52/// Summary statistics produced by a [`scan_delta`] run.
53///
54/// The caller receives a [`ScanSummaryHandle`] alongside the record stream;
55/// after the stream is drained the handle's counters reflect the full scan.
56///
57/// ## Element tombstones (Issue #493 / DS4)
58///
59/// Non-frozen collection cells may contain **element-level removals**
60/// (`s = s - {x}` for sets, individual map-key deletions).  V1 of the
61/// delta-scan layer cannot represent these at element granularity; they are
62/// detected, counted here, and reported as a warning — but not silently
63/// dropped.  Consumers that need element-level fidelity must wait for full
64/// Issue #493 implementation.
65#[derive(Debug, Clone)]
66pub struct ScanSummary {
67    /// Total number of element-level collection tombstones detected across
68    /// all collection columns in this scan.
69    ///
70    /// A non-zero value means the delta contains `s = s - {x}` style
71    /// removals that are not represented in the emitted [`DeltaRecord`]s
72    /// (Issue #493 follow-up).
73    pub element_tombstones_detected: u64,
74}
75
76/// Handle to in-progress scan summary counters.
77///
78/// Returned by [`scan_delta`] alongside the record receiver.
79/// After the receiver is drained (stream complete), call [`ScanSummaryHandle::read`]
80/// to obtain the final [`ScanSummary`].
81///
82/// The handle is cheaply clonable — all clones share the same counters.
83#[derive(Debug, Clone)]
84pub struct ScanSummaryHandle {
85    element_tombstones: Arc<AtomicU64>,
86}
87
88impl ScanSummaryHandle {
89    fn new() -> Self {
90        Self {
91            element_tombstones: Arc::new(AtomicU64::new(0)),
92        }
93    }
94
95    /// Increment the element-tombstone counter by `n`.
96    pub(super) fn add_element_tombstones(&self, n: u64) {
97        self.element_tombstones.fetch_add(n, Ordering::Relaxed);
98    }
99
100    /// Read the current scan summary.  Call after draining the record receiver
101    /// for a complete picture.
102    pub fn read(&self) -> ScanSummary {
103        ScanSummary {
104            element_tombstones_detected: self.element_tombstones.load(Ordering::Relaxed),
105        }
106    }
107}

113/// Combined partition key + clustering columns for a single row (or partition).
114///
115/// For records that address a full partition (`PartitionDelete`,
116/// `StaticUpsert`, `RangeDelete`) the `clustering` vec is empty.  For
117/// row-addressed records it holds the decoded clustering column values in
118/// primary-key definition order.
119///
120/// Values are decoded according to the `TableSchema` in effect at scan time;
121/// the no-heuristics mandate applies — every component uses schema-authorised
122/// types.
123#[derive(Debug, Clone, PartialEq)]
124pub struct RowKeys {
125    /// Decoded partition-key components in definition order.
126    ///
127    /// Multi-component partition keys (composite) appear as multiple elements.
128    pub partition: Vec<Value>,
129
130    /// Decoded clustering-key components in definition order.
131    ///
132    /// Empty when the record is partition-scoped (partition delete, static
133    /// upsert, or range-delete — in which case bounds carry the clustering
134    /// information instead).
135    pub clustering: Vec<Value>,
136}
137
138impl RowKeys {
139    /// Create a partition-scoped key (no clustering components).
140    pub fn partition_only(partition: Vec<Value>) -> Self {
141        Self {
142            partition,
143            clustering: Vec::new(),
144        }
145    }
146
147    /// Create a fully-specified row key.
148    pub fn new(partition: Vec<Value>, clustering: Vec<Value>) -> Self {
149        Self {
150            partition,
151            clustering,
152        }
153    }
154}

159
160/// Liveness metadata for a row's primary-key cell.
161///
162/// Cassandra writes a liveness-info record alongside a row when the row was
163/// created with `INSERT` (not `UPDATE`).  `UPDATE` statements produce rows
164/// with no liveness info; those rows have `liveness: None` in
165/// [`DeltaRecord::Upsert`].
166///
167/// The `writetime` here is the row-level primary-key liveness timestamp,
168/// equivalent to `__ts` in the Parquet envelope.
169#[derive(Debug, Clone, PartialEq)]
170pub struct CellMeta {
171    /// Writetime in microseconds since the Unix epoch.
172    pub writetime: i64,
173
174    /// Expiry time in microseconds since the Unix epoch, if a TTL was set on
175    /// the `INSERT`.  `None` means no TTL.
176    pub expires_at: Option<i64>,
177}
178
179impl CellMeta {
180    /// Create a liveness record with no TTL.
181    pub fn new(writetime: i64) -> Self {
182        Self {
183            writetime,
184            expires_at: None,
185        }
186    }
187
188    /// Create a liveness record with a TTL expiry time.
189    pub fn with_ttl(writetime: i64, expires_at: i64) -> Self {
190        Self {
191            writetime,
192            expires_at: Some(expires_at),
193        }
194    }
195}

197// ---------------------------------------------------------------------------
198// CellDelta
199// ---------------------------------------------------------------------------
200
201/// Per-cell change record carried inside [`DeltaRecord::Upsert`] and
202/// [`DeltaRecord::StaticUpsert`].
203///
204/// A `null` [`value`][Self::value] represents a cell tombstone
205/// (`DELETE col FROM t WHERE …`).  A non-null value is the cell content as
206/// decoded by the schema-aware reader — no heuristics.
207///
208/// ## Collection columns (v1 limitation)
209///
210/// Non-frozen collection columns carry per-element writetimes and element
211/// tombstones that v1 cannot faithfully represent at element granularity.
212/// For those columns:
213///
214/// - `value` holds the elements present in this generation (for an append
215///   `s = s + {…}` that is only the appended elements — correct delta
216///   semantics).
217/// - `writetime` is the maximum element writetime.
218/// - `replaced` is `true` when the generation carries a collection tombstone
219///   (i.e. an overwrite `s = {…}`), signalling consumers to replace rather
220///   than merge.
221///
222/// Full element-level fidelity is a tracked follow-up (Issue #493).
223#[derive(Debug, Clone, PartialEq)]
224pub struct CellDelta {
225    /// The decoded cell value, or `None` for a cell tombstone.
226    pub value: Option<Value>,
227
228    /// Writetime in microseconds since the Unix epoch.
229    pub writetime: i64,
230
231    /// TTL expiry time in microseconds since the Unix epoch.
232    ///
233    /// `None` means no TTL was set on this cell.  The delta-scan layer never
234    /// resolves TTLs at scan time — whether a cell is expired is left to the
235    /// downstream consumer (idempotent output guarantee).
236    pub expires_at: Option<i64>,
237
238    /// `true` when this collection cell carries a collection-level tombstone,
239    /// meaning the downstream consumer must **replace** rather than merge the
240    /// prior collection state.
241    ///
242    /// Always `false` for scalar (non-collection) columns.
243    pub replaced: bool,
244}
245
246impl CellDelta {
247    /// Create a simple value cell with no TTL and no collection-replace flag.
248    pub fn value(value: Value, writetime: i64) -> Self {
249        Self {
250            value: Some(value),
251            writetime,
252            expires_at: None,
253            replaced: false,
254        }
255    }
256
257    /// Create a cell tombstone (no value, just a deletion timestamp).
258    pub fn tombstone(writetime: i64) -> Self {
259        Self {
260            value: None,
261            writetime,
262            expires_at: None,
263            replaced: false,
264        }
265    }
266
267    /// Create a value cell with a TTL expiry time.
268    pub fn value_with_ttl(value: Value, writetime: i64, expires_at: i64) -> Self {
269        Self {
270            value: Some(value),
271            writetime,
272            expires_at: Some(expires_at),
273            replaced: false,
274        }
275    }
276
277    /// Create a collection cell that replaces (not merges) prior state.
278    pub fn collection_replace(value: Value, writetime: i64) -> Self {
279        Self {
280            value: Some(value),
281            writetime,
282            expires_at: None,
283            replaced: true,
284        }
285    }
286}

288// ---------------------------------------------------------------------------
289// RangeBound
290// ---------------------------------------------------------------------------
291
292/// One bound of a range-tombstone clustering-key range.
293///
294/// ## Prefix bounds
295///
296/// Cassandra range tombstones may specify bounds that are **prefixes** of the
297/// full clustering key — for example, a table with clustering columns
298/// `(year INT, month INT, day INT)` can have a range tombstone covering all
299/// rows in `year = 2024`.  In that case:
300///
301/// ```text
302/// start = RangeBound { values: [Value::Integer(2024)], inclusive: true  }
303/// end   = RangeBound { values: [Value::Integer(2024)], inclusive: true  }
304/// ```
305///
306/// Trailing clustering components that are absent from the bound are simply
307/// not present in [`values`][Self::values].  Consumers must treat a prefix
308/// bound as matching all rows whose clustering key begins with the given
309/// prefix (within the inclusive/exclusive constraint).
310///
311/// ## Empty bound
312///
313/// An empty `values` vec with `inclusive: false` represents an open
314/// (unbounded) end — i.e. from/to the beginning or end of the partition.
315#[derive(Debug, Clone, PartialEq)]
316pub struct RangeBound {
317    /// The clustering-key prefix values for this bound, in primary-key order.
318    ///
319    /// May be shorter than the full clustering-key arity (prefix bound).
320    /// An empty vec means an open (unbounded) end.
321    pub values: Vec<Value>,
322
323    /// Whether the bound is inclusive (`>=` / `<=`) or exclusive (`>` / `<`).
324    pub inclusive: bool,
325}
326
327impl RangeBound {
328    /// Create a bound from a full or partial set of clustering values.
329    pub fn new(values: Vec<Value>, inclusive: bool) -> Self {
330        Self { values, inclusive }
331    }
332
333    /// Convenience: create an inclusive bound.
334    pub fn inclusive(values: Vec<Value>) -> Self {
335        Self {
336            values,
337            inclusive: true,
338        }
339    }
340
341    /// Convenience: create an exclusive bound.
342    pub fn exclusive(values: Vec<Value>) -> Self {
343        Self {
344            values,
345            inclusive: false,
346        }
347    }
348
349    /// An open (unbounded) end — matches from/to the start or end of the partition.
350    pub fn open() -> Self {
351        Self {
352            values: Vec::new(),
353            inclusive: false,
354        }
355    }
356
357    /// Returns `true` if this is a prefix bound (fewer values than the full
358    /// clustering-key arity of the table, or the full arity is not known).
359    ///
360    /// Note: callers that know the table's clustering-key arity should compare
361    /// `self.values.len()` against that arity directly rather than relying on
362    /// this method.
363    pub fn is_prefix(&self) -> bool {
364        // An empty bound is not a prefix — it is an open bound.
365        // Any non-empty bound with at least one value is potentially a prefix;
366        // the caller is responsible for comparing against the full arity.
367        !self.values.is_empty()
368    }
369}

372// DeltaRecord

374
375/// A single change record emitted by the delta-scan API.
376///
377/// Records stream in SSTable order (partition, then clustering).  The scan
378/// makes no cross-SSTable decisions — no merge, no GC-grace filtering.
379///
380/// ## Variant reference
381///
382/// | Variant | CQL operation | Key scope |
383/// |---------|---------------|-----------|
384/// | `Upsert` | `INSERT`/`UPDATE` on regular columns | `(pk, ck)` |
385/// | `StaticUpsert` | `UPDATE` on static columns | `pk` |
386/// | `RowDelete` | `DELETE FROM t WHERE pk=? AND ck=?` | `(pk, ck)` |
387/// | `RangeDelete` | `DELETE FROM t WHERE pk=? AND ck>=? AND ck<?` | `pk` + bounds |
388/// | `PartitionDelete` | `DELETE FROM t WHERE pk=?` | `pk` |
389///
390/// ## Downstream merge keys
391///
392/// - `Upsert` / `RowDelete` — reconcile on `(partition, clustering)`.
393/// - `StaticUpsert` — reconcile on `partition`.
394/// - `PartitionDelete` / `RangeDelete` — apply as predicates using `__ts`
395///   (i.e. `deleted_at`) to decide last-write-wins per cell.
396#[derive(Debug, Clone, PartialEq)]
397pub enum DeltaRecord {
398    /// A row-level insert or update of regular (non-static) columns.
399    ///
400    /// `liveness` is `Some` when the row was created with `INSERT` and carries
401    /// a primary-key liveness timestamp.  `UPDATE` statements produce rows
402    /// with `liveness: None`.
403    ///
404    /// Each element of `cells` is a `(column_id, delta)` pair.  Only the
405    /// columns actually written in this generation appear; absent columns are
406    /// not present (they are null in the Parquet envelope).
407    Upsert {
408        /// Partition key + clustering columns for this row.
409        keys: RowKeys,
410        /// Row liveness info, present only for `INSERT` operations.
411        liveness: Option<CellMeta>,
412        /// Per-column cell deltas for non-static columns modified in this row.
413        cells: Vec<(ColumnId, CellDelta)>,
414    },
415
416    /// An update of one or more static columns for a partition.
417    ///
418    /// Static columns belong to the partition, not individual rows.  The
419    /// clustering key is empty in `partition_key`.
420    StaticUpsert {
421        /// Partition key (no clustering columns).
422        partition_key: RowKeys,
423        /// Per-column cell deltas for static columns modified.
424        cells: Vec<(ColumnId, CellDelta)>,
425    },
426
427    /// A row-level tombstone (`DELETE FROM t WHERE pk=? AND ck=?`).
428    RowDelete {
429        /// Partition key + clustering columns identifying the deleted row.
430        keys: RowKeys,
431        /// Deletion timestamp in microseconds since the Unix epoch
432        /// (`markedForDeleteAt` in Cassandra internals).
433        deleted_at: i64,
434    },
435
436    /// A range tombstone covering a contiguous clustering-key range within a
437    /// single partition.
438    ///
439    /// `start` and `end` may be prefix bounds (see [`RangeBound`]).
440    ///
441    /// ```text
442    /// DELETE FROM t WHERE pk=1 AND ck >= 'a' AND ck < 'm'
443    /// ```
444    RangeDelete {
445        /// Partition key (no clustering columns — bounds carry the range).
446        partition_key: RowKeys,
447        /// Inclusive or exclusive lower clustering-key bound.
448        start: RangeBound,
449        /// Inclusive or exclusive upper clustering-key bound.
450        end: RangeBound,
451        /// Deletion timestamp in microseconds since the Unix epoch.
452        deleted_at: i64,
453    },
454
455    /// A partition-level tombstone (`DELETE FROM t WHERE pk=?`).
456    ///
457    /// Supersedes every row and cell in the partition whose writetime is older
458    /// than `deleted_at`.
459    PartitionDelete {
460        /// Partition key (no clustering columns).
461        partition_key: RowKeys,
462        /// Deletion timestamp in microseconds since the Unix epoch.
463        deleted_at: i64,
464    },
465}
466
467impl DeltaRecord {
468    /// Return the partition-key portion of any record type.
469    pub fn partition_key(&self) -> &[Value] {
470        match self {
471            DeltaRecord::Upsert { keys, .. } => &keys.partition,
472            DeltaRecord::StaticUpsert { partition_key, .. } => &partition_key.partition,
473            DeltaRecord::RowDelete { keys, .. } => &keys.partition,
474            DeltaRecord::RangeDelete { partition_key, .. } => &partition_key.partition,
475            DeltaRecord::PartitionDelete { partition_key, .. } => &partition_key.partition,
476        }
477    }
478
479    /// Return the `__op` discriminator string used in the Parquet envelope.
480    pub fn op_name(&self) -> &'static str {
481        match self {
482            DeltaRecord::Upsert { .. } => "upsert",
483            DeltaRecord::StaticUpsert { .. } => "static_upsert",
484            DeltaRecord::RowDelete { .. } => "row_delete",
485            DeltaRecord::RangeDelete { .. } => "range_delete",
486            DeltaRecord::PartitionDelete { .. } => "partition_delete",
487        }
488    }
489}

491// ---------------------------------------------------------------------------
492// scan_delta — streaming API (Issue #698)
493// ---------------------------------------------------------------------------
494
495/// Stream [`DeltaRecord`]s from a single SSTable generation directory.
496///
497/// Opens the first `Data.db` file found under `sstable_dir` and streams
498/// [`DeltaRecord::Upsert`] and [`DeltaRecord::StaticUpsert`] records in
499/// on-disk partition/clustering order via the returned channel receiver.
500///
501/// ## Scope (Issue #698)
502///
503/// This implementation handles **upserts and static upserts only**.
504/// Row, range, and partition tombstones are not yet emitted (Issue #699);
505/// if a row tombstone is encountered the scan continues past it (the row is
506/// silently dropped with a `log::debug!` trace — not a hard error, since most
507/// real SSTables have no tombstones and the issue is out of scope here).
508///
509/// ## Contract
510///
511/// - Records stream in SSTable order (partition, then clustering).
512/// - No cross-SSTable merge, no GC-grace filtering.
513/// - Every live cell carries `writetime` and `expires_at`; TTL is never
514///   resolved at scan time (idempotent output).
515/// - Columns absent from a row are absent from `cells` (not null).
516/// - A cell tombstone (`DELETE col FROM …`) appears as
517///   `CellDelta { value: None, writetime: t, … }`.
518/// - INSERT rows carry `liveness: Some(CellMeta { writetime, expires_at })`;
519///   UPDATE rows carry `liveness: None`.
520///
521/// ## Memory
522///
523/// The channel is bounded by `buffer_size` records.  The parse task pauses
524/// when the consumer falls behind.  Individual [`DeltaRecord`]s are bounded in
525/// size and never accumulated into a full-table collection.
526///
527/// **Caveat**: `prepare_delta_scan` → `stitch_all_chunks` fully materialises the
528/// decompressed data section of the SSTable into a single `Vec<u8>` before
529/// streaming begins.  For very large SSTables this may approach the 128 MB
530/// memory budget; callers should be aware that the stitched buffer is resident
531/// for the duration of the scan.
532///
533/// ## Errors
534///
535/// A hard parse error (corrupt SSTable, missing schema, etc.) is forwarded
536/// as an `Err(…)` item in the channel stream and terminates the scan.
537///
538/// [`scan_delta`] returns immediately with an error if the SSTable directory
539/// does not exist or contains no `Data.db` file.
540/// Return type of [`scan_delta`]: a channel receiver for [`DeltaRecord`]s and
541/// a [`ScanSummaryHandle`] for collecting aggregate scan statistics.
542pub type ScanDeltaOutput = (
543    tokio::sync::mpsc::Receiver<crate::Result<DeltaRecord>>,
544    ScanSummaryHandle,
545);
546
547pub fn scan_delta(
548    sstable_dir: std::path::PathBuf,
549    schema: crate::schema::TableSchema,
550    buffer_size: usize,
551) -> ScanDeltaOutput {
552    let (tx, rx) = tokio::sync::mpsc::channel(buffer_size.max(1));
553    let summary = ScanSummaryHandle::new();
554    let summary_for_task = summary.clone();
555    tokio::spawn(async move {
556        if let Err(e) = run_scan_delta(sstable_dir, schema, tx.clone(), summary_for_task).await {
557            let _ = tx.send(Err(e)).await;
558        }
559    });
560    (rx, summary)
561}
562
563/// Internal async driver for [`scan_delta`].
564async fn run_scan_delta(
565    sstable_dir: std::path::PathBuf,
566    schema: crate::schema::TableSchema,
567    tx: tokio::sync::mpsc::Sender<crate::Result<DeltaRecord>>,
568    summary: ScanSummaryHandle,
569) -> crate::Result<()> {
570    use crate::storage::sstable::reader::SSTableReader;
571
572    // Find the Data.db file in this directory.
573    let data_db = find_data_db(&sstable_dir)?;
574
575    let config = crate::Config::default();
576    let platform =
577        std::sync::Arc::new(crate::Platform::new(&config).await.map_err(|e| {
578            crate::Error::corruption(format!("scan_delta: platform init failed: {e}"))
579        })?);
580
581    let reader = std::sync::Arc::new(
582        SSTableReader::open(&data_db, &config, platform)
583            .await
584            .map_err(|e| {
585                crate::Error::corruption(format!("scan_delta: failed to open {:?}: {e}", data_db))
586            })?,
587    );
588
589    // Wrap schema in Arc once — both the emit closure and parse call share the
590    // same allocation rather than cloning the struct twice.
591    let schema_arc = std::sync::Arc::new(schema);
592
593    // Stitch + parse using a private per-scan cursor (issue #815): no scan-wide
594    // mutex needed because the cursor owns its own file position / chunk index.
595    // The parsing itself is synchronous and moves into spawn_blocking.
596    let (stitched, parser) = reader.prepare_delta_scan().await?;
597
598    let schema_for_parse = std::sync::Arc::clone(&schema_arc);
599    let reader_arc = std::sync::Arc::clone(&reader);
600
601    // The parse closure is synchronous; run it on a blocking thread so it can
602    // use `blocking_send` without stalling the async runtime (mirrors
603    // `parse_stitched_stream` in data_access.rs, issue #790).
604    let parse_result = tokio::task::spawn_blocking(move || -> crate::Result<()> {
605        parser.parse_block_emit_delta(
606            &stitched,
607            Some(&schema_for_parse),
608            &reader_arc,
609            |(
610                partition_key_raw,
611                cells,
612                cell_meta,
613                row_liveness_ts,
614                is_static,
615                is_row_tombstone,
616                marked_for_delete_at,
617                range_info,       // Issue #699: Some((start_vals,start_incl,end_vals,end_incl,del_at)) for range tombstone
618                is_partition_tombstone, // Issue #699: true for partition-level tombstone
619                col_complex_meta, // Issue #700 DS4: per-column ComplexColumnMeta
620                liveness_expires_at_micros, // Issue #702: TTL liveness expiry in µs (epoch-s * 1_000_000)
621            )| {
622                // ----------------------------------------------------------------
623                // Decode partition key from raw bytes.
624                // (Needed for all record types, so decode upfront.)
625                // ----------------------------------------------------------------
626                let pk_columns = crate::storage::partition_key_codec::decode_partition_key_columns(
627                    &partition_key_raw.0,
628                    &schema_arc,
629                )
630                .map_err(|e| crate::Error::corruption(format!(
631                    "scan_delta: failed to decode partition key (raw bytes {:?}): {e}",
632                    partition_key_raw.0
633                )))?;
634                let partition_values: Vec<Value> = pk_columns.into_iter().map(|(_, v)| v).collect();
635
636                // ----------------------------------------------------------------
637                // Issue #699: Partition tombstone
638                // ----------------------------------------------------------------
639                if is_partition_tombstone {
640                    let deleted_at = marked_for_delete_at.ok_or_else(|| {
641                        crate::Error::corruption(format!(
642                            "scan_delta: partition tombstone for pk={:?} has no markedForDeleteAt \
643                             — cannot represent faithfully (no-heuristics, issue #28)",
644                            partition_values
645                        ))
646                    })?;
647                    let record = DeltaRecord::PartitionDelete {
648                        partition_key: RowKeys::partition_only(partition_values),
649                        deleted_at,
650                    };
651                    return match tx.blocking_send(Ok(record)) {
652                        Ok(()) => Ok(std::ops::ControlFlow::Continue(())),
653                        Err(_) => Ok(std::ops::ControlFlow::Break(())),
654                    };
655                }
656
657                // ----------------------------------------------------------------
658                // Issue #699: Range tombstone
659                // ----------------------------------------------------------------
660                if let Some((start_vals, start_incl, end_vals, end_incl, del_at)) = range_info {
661                    let record = DeltaRecord::RangeDelete {
662                        partition_key: RowKeys::partition_only(partition_values),
663                        start: RangeBound::new(start_vals, start_incl),
664                        end: RangeBound::new(end_vals, end_incl),
665                        deleted_at: del_at,
666                    };
667                    return match tx.blocking_send(Ok(record)) {
668                        Ok(()) => Ok(std::ops::ControlFlow::Continue(())),
669                        Err(_) => Ok(std::ops::ControlFlow::Break(())),
670                    };
671                }
672
673                // Build key-column name sets for filtering.
674                let pk_col_names: std::collections::HashSet<&str> = schema_arc
675                    .partition_keys.iter().map(|k| k.name.as_str()).collect();
676                let clustering_col_names: std::collections::HashSet<&str> = schema_arc
677                    .clustering_keys.iter().map(|ck| ck.name.as_str()).collect();
678                let static_col_names: std::collections::HashSet<&str> = schema_arc
679                    .columns.iter().filter(|c| c.is_static).map(|c| c.name.as_str()).collect();
680
681                // Extract clustering values in declaration order.
682                let clustering_values: Vec<Value> = schema_arc
683                    .clustering_keys.iter()
684                    .filter_map(|ck| cells.get(&ck.name).cloned())
685                    .collect();
686
687                // ----------------------------------------------------------------
688                // Issue #699: Row tombstone
689                // ----------------------------------------------------------------
690                if is_row_tombstone {
691                    let deleted_at = marked_for_delete_at.ok_or_else(|| {
692                        crate::Error::corruption(format!(
693                            "scan_delta: row tombstone for pk={:?} ck={:?} has no markedForDeleteAt \
694                             — cannot represent faithfully (no-heuristics, issue #28)",
695                            partition_values, clustering_values
696                        ))
697                    })?;
698                    let record = DeltaRecord::RowDelete {
699                        keys: RowKeys::new(partition_values, clustering_values),
700                        deleted_at,
701                    };
702                    return match tx.blocking_send(Ok(record)) {
703                        Ok(()) => Ok(std::ops::ControlFlow::Continue(())),
704                        Err(_) => Ok(std::ops::ControlFlow::Break(())),
705                    };
706                }
707
708                // ----------------------------------------------------------------
709                // DS4 (Issue #700): Process element tombstone counts from this row's
710                // collection columns and update the scan summary.
711                // ----------------------------------------------------------------
712                {
713                    let mut row_element_tombstones: u64 = 0;
714                    for (col_name, ccm) in &col_complex_meta {
715                        if ccm.element_tombstone_count > 0 {
716                            row_element_tombstones += ccm.element_tombstone_count;
717                            log::warn!(
718                                "scan_delta DS4: collection column '{}' has {} element-level tombstone(s) \
719                                 that cannot be represented in v1 delta semantics (Issue #493 follow-up). \
720                                 These removals are counted in the scan summary but not in the emitted records.",
721                                col_name, ccm.element_tombstone_count
722                            );
723                        }
724                    }
725                    if row_element_tombstones > 0 {
726                        summary.add_element_tombstones(row_element_tombstones);
727                    }
728                }
729
730                // ----------------------------------------------------------------
731                // Build CellDelta entries for non-key columns only.
732                // ----------------------------------------------------------------
733                let mut cell_deltas: Vec<(ColumnId, CellDelta)> = Vec::new();
734
735                for (col_name, value) in &cells {
736                    // Skip key columns — they are part of RowKeys, not cell payload.
737                    if pk_col_names.contains(col_name.as_str())
738                        || clustering_col_names.contains(col_name.as_str())
739                    {
740                        continue;
741                    }
742                    // Static upsert rows: only static columns.
743                    // Regular upsert rows: only non-static columns.
744                    if is_static && !static_col_names.contains(col_name.as_str()) {
745                        continue;
746                    }
747                    if !is_static && static_col_names.contains(col_name.as_str()) {
748                        continue;
749                    }
750
751                    let meta = cell_meta.get(col_name.as_str());
752                    let (writetime, expires_at) = match meta {
753                        Some(m) => {
754                            let exp = m.expiration.as_ref().map(|e| {
755                                // expires_at_seconds is epoch-seconds; delta-scan
756                                // contract requires epoch-microseconds.
757                                e.expires_at_seconds.saturating_mul(1_000_000)
758                            });
759                            (m.write_timestamp_micros, exp)
760                        }
761                        // Fallback: row-level liveness timestamp (should not happen
762                        // when want_cell_metadata=true, but be defensive).
763                        None => (row_liveness_ts.unwrap_or(0), None),
764                    };
765
766                    // DS4 (Issue #700): For collection columns, check whether this
767                    // generation carries a collection-level tombstone (overwrite semantics).
768                    // `replaced = true` signals downstream consumers to replace rather than
769                    // merge the prior collection state.  Always `false` for scalar columns.
770                    //
771                    // Also: the normal read-path `cell_meta` stores `row_ts` for collection
772                    // columns (not per-element max) to keep WRITETIME(col) semantics correct.
773                    // For the delta-scan path we override writetime with the max element
774                    // writetime from ComplexColumnMeta when it is non-zero (roborev Finding 1).
775                    let (replaced, writetime) = match col_complex_meta.get(col_name.as_str()) {
776                        Some(ccm) => {
777                            let effective_wt = if ccm.max_element_writetime != 0 {
778                                ccm.max_element_writetime
779                            } else {
780                                writetime
781                            };
782                            (ccm.has_collection_tombstone, effective_wt)
783                        }
784                        None => (false, writetime),
785                    };
786
787                    let cell = match value {
788                        // Cell tombstone: IS_DELETED flag was set on the cell.
789                        Value::Tombstone(info)
790                            if info.tombstone_type == TombstoneType::CellTombstone =>
791                        {
792                            CellDelta {
793                                value: None,
794                                // Use the tombstone's own deletion_time (authoritative),
795                                // NOT the cell_meta write_timestamp (which inherits row ts).
796                                writetime: info.deletion_time,
797                                expires_at: None,
798                                replaced: false,
799                            }
800                        }
801                        _ => CellDelta {
802                            value: Some(value.clone()),
803                            writetime,
804                            expires_at,
805                            replaced,
806                        },
807                    };
808
809                    cell_deltas.push((ColumnId::new(col_name), cell));
810                }
811
812                // ----------------------------------------------------------------
813                // Emit the appropriate DeltaRecord variant (Upsert / StaticUpsert).
814                // ----------------------------------------------------------------
815                let record = if is_static {
816                    DeltaRecord::StaticUpsert {
817                        partition_key: RowKeys::partition_only(partition_values),
818                        cells: cell_deltas,
819                    }
820                } else {
821                    // Build liveness CellMeta, carrying the TTL expiry when present (Issue #702).
822                    let liveness = row_liveness_ts.map(|ts| {
823                        match liveness_expires_at_micros {
824                            Some(exp) => CellMeta::with_ttl(ts, exp),
825                            None => CellMeta::new(ts),
826                        }
827                    });
828                    DeltaRecord::Upsert {
829                        keys: RowKeys::new(partition_values, clustering_values),
830                        liveness,
831                        cells: cell_deltas,
832                    }
833                };
834
835                // Forward to the channel.  `blocking_send` is correct here
836                // because we are inside spawn_blocking (not an async context).
837                match tx.blocking_send(Ok(record)) {
838                    Ok(()) => Ok(std::ops::ControlFlow::Continue(())),
839                    Err(_) => {
840                        // Consumer dropped: stop streaming.
841                        Ok(std::ops::ControlFlow::Break(()))
842                    }
843                }
844            },
845        )
846    })
847    .await;
848
849    match parse_result {
850        Ok(result) => result,
851        Err(join_err) => Err(crate::Error::corruption(format!(
852            "scan_delta: parse task panicked: {join_err}"
853        ))),
854    }
855}

857/// Find the `Data.db` file inside an SSTable directory.
858///
859/// Cassandra names Data.db files with a generation prefix, e.g.:
860/// `nb-1-big-Data.db` or `na-1-big-Data.db`.
861///
862/// Returns an error if no matching file is found.  If more than one `*-Data.db`
863/// file is present (which violates the single-generation contract), a warning is
864/// logged and the lexicographically smallest file name is returned so behaviour
865/// is at least deterministic rather than OS-dependent.
866fn find_data_db(dir: &std::path::Path) -> crate::Result<std::path::PathBuf> {
867    if !dir.exists() {
868        return Err(crate::Error::corruption(format!(
869            "scan_delta: SSTable directory does not exist: {:?}",
870            dir
871        )));
872    }
873
874    let entries = std::fs::read_dir(dir).map_err(|e| {
875        crate::Error::corruption(format!("scan_delta: cannot read directory {:?}: {e}", dir))
876    })?;
877
878    let mut candidates: Vec<std::path::PathBuf> = entries
879        .filter_map(|e| e.ok())
880        .filter(|e| {
881            e.file_name()
882                .to_str()
883                .map(|n| n.ends_with("-Data.db"))
884                .unwrap_or(false)
885        })
886        .map(|e| e.path())
887        .collect();
888
889    if candidates.is_empty() {
890        return Err(crate::Error::corruption(format!(
891            "scan_delta: no Data.db file found in {:?}",
892            dir
893        )));
894    }
895
896    if candidates.len() > 1 {
897        candidates.sort();
898        log::warn!(
899            "scan_delta: {:?} contains {} Data.db files (expected 1 per generation); \
900             using lexicographically first: {:?}. Consider compacting before scanning.",
901            dir,
902            candidates.len(),
903            candidates[0]
904        );
905    }
906
907    Ok(candidates.remove(0))
908}
