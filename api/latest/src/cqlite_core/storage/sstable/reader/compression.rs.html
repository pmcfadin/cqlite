1//! SSTable component-name derivation for the reader open path.
2//!
3//! Since issue #1597 (Epic G / G1) the reader no longer performs speculative
4//! compression detection here. The open path parses `CompressionInfo.db` exactly
5//! once via `compression_info::CompressionInfo::parse` and derives the compression
6//! algorithm from that single result — see `reader::SSTableReader::open` and
7//! `load_compression_info_metadata`. The former `detect_and_initialize_compression`
8//! machinery (a second `parse_binary`, a ~25-generation `exists()` probe loop, and
9//! filename/entropy heuristics) is deleted.
10//!
11//! What remains is [`extract_sstable_base_name`], the deterministic
12//! `SsTableDescriptor`-derived base-name helper still used to locate the `CRC.db`
13//! sidecar for uncompressed BIG tables (`load_crc_reader`).
14
15use std::path::Path;
16
17/// Extract SSTable base name from path (e.g., "nb-1-big-Data.db" -> "nb-1-big")
18pub fn extract_sstable_base_name(path: &Path) -> Option<String> {
19    let filename = path.file_name()?.to_str()?;
20
21    // Require a .db component (preserves the legacy contract: no-extension and
22    // wrong-extension names yield None).
23    let filename_no_ext = filename.strip_suffix(".db")?;
24
25    // Prefer the descriptor parser, which finds the big/bti format segment even
26    // when the SSTable id is a hyphenated UUID (e.g.
27    // "da-00000000-0000-0000-0000-000000000001-bti-Data.db") that a fixed
28    // parts[0..3] split would mangle into the wrong base name, so the
29    // "*-CompressionInfo.db" sidecar lookup failed and compressed data was read
30    // as uncompressed (roborev #970).
31    if let Ok(d) = crate::storage::sstable::version_gate::SsTableDescriptor::parse(path) {
32        return Some(format!(
33            "{}-{}-{}",
34            d.version,
35            d.sstable_id,
36            d.format.as_str()
37        ));
38    }
39
40    // Fallback for non-standard names the descriptor rejects: keep the legacy
41    // {prefix}-{generation}-{format} heuristic so existing callers/tests are
42    // unchanged (e.g. "nb-1-big.db" with only 3 parts still yields None).
43    let parts: Vec<&str> = filename_no_ext.split('-').collect();
44    if parts.len() >= 4 {
45        Some(parts[0..3].join("-"))
46    } else {
47        tracing::warn!("Non-standard SSTable filename pattern: {}", filename);
48        None
49    }
50}
