1//! Per-element / per-cell compaction read contract (epic #899, Phase A).
2//!
3//! The compaction read path historically emitted `(RowKey, ScanRow, i64)` per row,
4//! collapsing every non-frozen collection / UDT into a single nested [`Value`]
5//! with one row-level timestamp. That representation cannot reconcile two
6//! SSTables that wrote DISJOINT elements of the same multi-cell column (each
7//! element has its own timestamp / ttl / local-deletion-time / cell-path), nor
8//! can it carry a real per-column complex deletion (Cassandra's
9//! `markedForDeleteAt` + `localDeletionTime` marker written ahead of a
10//! multi-cell column's elements).
11//!
12//! [`CompactionRow`] replaces that tuple on the **compaction-only** read path. It
13//! preserves the on-disk per-element granularity so the k-way merge can perform
14//! byte-faithful per-`(column, cell_path)` reconciliation. The user-facing read
15//! path (`scan` / `get` / `iterate_all_partitions` / `WRITETIME(collection)`) is
16//! UNCHANGED — it still uses the collapsed [`Value`] representation.
17//!
18//! Byte-format invariants this representation must preserve (see
19//! `docs/sstables-definitive-guide/` Ch.5 + Appendix B):
20//! - Element order is the on-disk order (SET by serialized bytes, MAP by key
21//!   bytes, LIST insertion order with 16-byte TimeUUID paths). Per-element
22//!   timestamps must NOT reorder elements.
23//! - A complex deletion's `LIVE` sentinel is `(i64::MIN, i32::MAX)`; a real
24//!   deletion carries `(markedForDeleteAt µs, localDeletionTime s)`.
25//! - Far-future local deletion times in `[2^31, 2^32)` are preserved as the
26//!   wrapping `as u32 as i32` value — never widened to i64.
27

31/// One row surfaced by the compaction read path, carrying per-element complex
32/// cells and the real per-column complex deletion (epic #899, Phase A).
33///
34/// This is the compaction-only counterpart of the old `(RowKey, ScanRow, i64)`
35/// tuple. `row_timestamp` is the row-level write timestamp (for a tombstone it
36/// is `markedForDeleteAt`); `row_data` holds either a row tombstone or the live
37/// simple + complex cells.
38#[derive(Debug, Clone, PartialEq)]
39pub struct CompactionRow {
40    /// Partition key bytes (token derived downstream).
41    pub key: RowKey,
42    /// Row-level write timestamp in microseconds (for a tombstone:
43    /// `markedForDeleteAt`).
44    pub row_timestamp: i64,
45    /// Row payload: tombstone or live cells.
46    pub row_data: CompactionRowData,
47}
48
49impl CompactionRow {
50    /// Build a [`CompactionRow`] from the legacy collapsed `(RowKey, ScanRow,
51    /// timestamp)` representation (the non-V5 compaction fallback path).
52    ///
53    /// This loses per-element complex granularity (the legacy fallback has none
54    /// to begin with): a live `ScanRow::Row` becomes simple cells, a
55    /// `ScanRow::Marker(Value::Tombstone(RowTombstone))` becomes a row tombstone,
56    /// any other marker becomes a single `value` cell. The V5CompressedLegacy
57    /// path bypasses this and builds per-element rows directly.
58    ///
59    /// Issue #1334: the reader carries every row through the single [`ScanRow`]
60    /// carrier — this consumer disassembles that same carrier (no `Value::Map`
61    /// bifurcation).
62    pub fn from_legacy_value(key: RowKey, row: ScanRow, row_timestamp: i64) -> Self {
63        let row_data = match row {
64            // A live row's interned cells become simple cells.
65            ScanRow::Row(entries) => {
66                let simple = entries
67                    .into_iter()
68                    .map(|(k, v)| {
69                        let timestamp = match &v {
70                            Value::Tombstone(info) => info.deletion_time,
71                            _ => row_timestamp,
72                        };
73                        SimpleCell {
74                            column: k.to_string(),
75                            value: v,
76                            timestamp,
77                            ttl: None,
78                            local_deletion_time: None,
79                        }
80                    })
81                    .collect();
82                CompactionRowData::Live {
83                    simple,
84                    complex: Vec::new(),
85                    // A live `ScanRow::Row` never carries a coexisting row deletion
86                    // (a row tombstone arrives as a `ScanRow::Marker`, handled below).
87                    row_deletion: None,
88                    row_liveness: RowLiveness::default(),
89                }
90            }
91            // A row tombstone marker becomes a row tombstone.
92            ScanRow::Marker(Value::Tombstone(info))
93                if info.tombstone_type == TombstoneType::RowTombstone =>
94            {
95                CompactionRowData::Tombstone {
96                    deletion_time: info.deletion_time,
97                    local_deletion_time: 0,
98                    // The legacy collapsed-value fallback has no clustering capture
99                    // (the clustering prefix is not surfaced on this path), so the
100                    // tombstone lands in the partition's `None` clustering bucket
101                    // exactly as before (#912 carries clustering only on the V5
102                    // per-element path).
103                    clustering: Vec::new(),
104                }
105            }
106            // A raw undecoded fallback row collapses to a single `value` cell
107            // carrying the raw bytes as a blob — the exact pre-#1334 shape a bare
108            // `Value::Blob` produced in this legacy collapsed-value fallback.
109            ScanRow::RawRow(bytes) => CompactionRowData::Live {
110                simple: vec![SimpleCell {
111                    column: "value".to_string(),
112                    value: Value::Blob(bytes.into()),
113                    timestamp: row_timestamp,
114                    ttl: None,
115                    local_deletion_time: None,
116                }],
117                complex: Vec::new(),
118                row_deletion: None,
119                row_liveness: RowLiveness::default(),
120            },
121            // Any other marker (null row, cell tombstone, …) collapses to a single
122            // `value` cell, exactly as the pre-#1334 fallback did.
123            ScanRow::Marker(other) => CompactionRowData::Live {
124                simple: vec![SimpleCell {
125                    column: "value".to_string(),
126                    value: other,
127                    timestamp: row_timestamp,
128                    ttl: None,
129                    local_deletion_time: None,
130                }],
131                complex: Vec::new(),
132                row_deletion: None,
133                row_liveness: RowLiveness::default(),
134            },
135        };
136        CompactionRow {
137            key,
138            row_timestamp,
139            row_data,
140        }
141    }
142}

144/// A clustering bound of a range-tombstone marker surfaced on the compaction
145/// read path (issue #933).
146///
147/// Reader-native counterpart of
148/// [`crate::storage::write_engine::mutation::ClusteringBound`]; kept here so the
149/// compaction read contract does not depend on the write-engine types. Each
150/// bound carries its clustering-prefix `(name, value)` pairs (possibly a PREFIX
151/// shorter than the full clustering arity). An open bound (the writer emits these
152/// as an inclusive bound with zero clustering values) is [`Self::Bottom`] /
153/// [`Self::Top`].
154#[derive(Debug, Clone, PartialEq)]
155pub enum CompactionBound {
156    /// Inclusive bound (the clustering prefix is part of the deletion range).
157    Inclusive(Vec<(String, Value)>),
158    /// Exclusive bound (the clustering prefix is NOT part of the deletion range).
159    Exclusive(Vec<(String, Value)>),
160    /// Before all clustering keys (start of partition).
161    Bottom,
162    /// After all clustering keys (end of partition).
163    Top,
164}

166/// Primary-key (row-marker) liveness surfaced on a live compaction row so a
167/// READ consumer can apply Cassandra's row-visibility rule (issue #2374/#2789).
168///
169/// A row is visible to a `SELECT` iff it has at least one live data cell OR a
170/// LIVE primary-key liveness marker (`HAS_TIMESTAMP`, whose TTL — if any — has
171/// not expired). Compaction (the WRITE path) never SERIALIZES these fields — it
172/// retains an expired marker within gc_grace for byte-parity, and
173/// `merge_entry_to_mutation` does not read `row_liveness` at all (the emitted row
174/// marker is derived from `MergeEntry::timestamp` and the surviving cells) — so the
175/// consumer of the values is the read side (Flight `do_get` / cross-generation read
176/// merge).
177///
178/// Since #3094 the write path does gate one BRANCH on
179/// [`Self::marker_survives_floor`]: `apply_partition_shadowing`'s
180/// `!has_data && !marker_live` arm. That branch cannot move an emitted byte. It is
181/// reachable only when no data cell survives the partition floor, and in that case
182/// the reconciled `MergeEntry::timestamp` — `merge::reconcile`'s max over the
183/// surviving cells, INCLUDING the clustering pseudo-cells, whose timestamp is the
184/// marker's own — is itself at/below the floor, so the arm concludes exactly what the
185/// pre-#3094 row-timestamp test concluded. The #3094 shape (a cell tombstone written
186/// after the deletion) never reaches it at all: `apply_partition_shadowing`'s
187/// `is_data` test is BY NAME, so that tombstone survives the floor as data and
188/// `has_data` stays `true` — there `marker_survives_floor` only selects the carried,
189/// never-serialized `row_liveness`.
190///
191/// No-heuristics (#28): every field comes from the authoritative on-disk row header,
192/// never inferred.
193#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
194pub struct RowLiveness {
195    /// Whether the row carried a primary-key liveness marker (`HAS_TIMESTAMP` —
196    /// i.e. it was INSERTed, not created implicitly by a data-cell UPDATE).
197    pub has_marker: bool,
198    /// Marker expiry (epoch seconds) when the liveness marker carries a TTL
199    /// (`HAS_TTL`); `None` when the marker is live-forever (no TTL). Only
200    /// meaningful when `has_marker` is `true`. Reinterpreted UNSIGNED upstream
201    /// so a post-2038 expiry is not wrapped negative.
202    pub expires_at_seconds: Option<i64>,
203    /// The marker's authoritative WRITE timestamp in microseconds (the row
204    /// header liveness `timestamp`), when present. Populated from the on-disk
205    /// row header only (no-heuristics, #28) — never inferred. Carries the
206    /// last-write-wins key for [`Self::merge`] so a cross-generation fold keeps
207    /// the NEWER liveness marker outright (a newer TTL'd marker can supersede an
208    /// older live-forever one, and vice-versa) instead of a most-permissive
209    /// union. `None` when `has_marker` is `false`.
210    pub marker_timestamp: Option<i64>,
211}
212
213impl RowLiveness {
214    /// `true` when the primary-key liveness marker is present AND still live at
215    /// `now_secs` (no TTL, or its expiry is strictly in the future). Matches
216    /// `Cell.isLive`: live iff `now < localExpirationTime`.
217    pub fn marker_live_at(&self, now_secs: i64) -> bool {
218        self.has_marker && self.expires_at_seconds.is_none_or(|s| s > now_secs)
219    }
220
221    /// Whether a reconciled row's primary-key liveness marker survives a DELETION
222    /// FLOOR — the `markedForDeleteAt` (µs) of a covering partition (or range)
223    /// deletion — and may therefore still be carried forward.
224    ///
225    /// Cassandra judges this on the MARKER's OWN write timestamp: `BTreeRow.filter`
226    /// (`cassandra-5.0.8`) computes
227    /// `if (activeDeletion.deletes(newInfo.timestamp())) newInfo = LivenessInfo.EMPTY;`
228    /// and `DeletionTime.deletes(long ts)` is `ts <= markedForDeleteAt`. So the
229    /// marker survives iff [`marker_timestamp`](Self::marker_timestamp) is strictly
230    /// greater than `floor`.
231    ///
232    /// `row_timestamp` — the RECONCILED ROW timestamp (the max over the row's
233    /// surviving cells) — is therefore consulted ONLY on the marker-LESS arm
234    /// (`marker_timestamp == None`, i.e. `has_marker == false`): there is no marker
235    /// timestamp to judge, so the row-level test stays the sole condition and
236    /// marker-less behaviour is unchanged. (Purging a marker-less tombstone-only row
237    /// is tracked separately as issue #3121.)
238    ///
239    /// Deliberately NOT an `&&` of both tests. `row_timestamp` ALONE is the #3094
240    /// defect: a cell TOMBSTONE written AFTER the deletion raises the reconciled row
241    /// timestamp above `floor` even when the marker itself is covered, so a deleted
242    /// marker was carried forward and resurrected an all-null phantom row out of a
243    /// deleted partition — reaching every consumer that decides row visibility from
244    /// the merged entry alone (Flight's `producer.rs::entry_to_row`, shared by its
245    /// row-stream and pushed-down-aggregate producers). But keeping `row_timestamp`
246    /// as an ADDITIONAL necessary condition on the marker-present arm is equally
247    /// non-Cassandra in the other direction: `BTreeRow.filter` never consults the
248    /// row's cell timestamps when deciding the liveness marker's fate, so a marker
249    /// strictly newer than `floor` whose data cells all predate it must still keep
250    /// the key-only row VISIBLE. Conjoining the two tests would HIDE that row, and
251    /// is unobservable today only because of an UNASSERTED cross-module invariant
252    /// (`merge::reconcile` folds `row_timestamp` over the surviving clustering
253    /// pseudo-cells too, whose timestamp is the marker's own, so
254    /// `row_timestamp >= marker_timestamp` happens to hold). Splitting the arms
255    /// removes the dependency on that invariant instead of documenting it: a future
256    /// cleanup that excludes pseudo-cells from that fold can no longer silently
257    /// hide a row Cassandra returns.
258    pub fn marker_survives_floor(&self, row_timestamp: i64, floor: i64) -> bool {
259        match self.marker_timestamp {
260            // Marker present: judged on its OWN write timestamp, exactly as
261            // `BTreeRow.filter` does — nothing about the row's cells participates.
262            Some(marker_ts) => marker_ts > floor,
263            // No marker timestamp: the row-level test is the only evidence there is.
264            None => row_timestamp > floor,
265        }
266    }
267
268    /// Fold two liveness markers across generations by Cassandra
269    /// last-write-wins on the marker WRITE timestamp (issue #2374/#2789): the
270    /// marker with the higher [`marker_timestamp`](Self::marker_timestamp) wins
271    /// OUTRIGHT — its expiry / live-forever state is taken as-is, so a newer
272    /// TTL'd marker supersedes an older live-forever one (and vice-versa). Later
273    /// expiry is only a TIE-BREAK when the timestamps compare equal (or both are
274    /// absent). A generation with no marker contributes nothing, so the surviving
275    /// marker (if any) is carried through unchanged.
276    ///
277    /// This replaces the former most-permissive union, which unconditionally let
278    /// a live-forever marker win regardless of write order and diverged from
279    /// Cassandra for a reverse-timestamp reinsertion (older live-forever, newer
280    /// expired TTL): the union reported the row VISIBLE where Cassandra keeps the
281    /// newer expired liveness and HIDES it.
282    pub fn merge(self, other: RowLiveness) -> RowLiveness {
283        match (self.has_marker, other.has_marker) {
284            (false, false) => RowLiveness::default(),
285            (true, false) => self,
286            (false, true) => other,
287            (true, true) => {
288                // Both generations carry a marker: last-write-wins on the
289                // authoritative write timestamp; later expiry breaks a tie.
290                let self_wins = match (self.marker_timestamp, other.marker_timestamp) {
291                    (Some(a), Some(b)) if a != b => a > b,
292                    // Equal (or one/both absent) timestamps → tie-break on the
293                    // later expiry, treating live-forever (`None`) as the latest.
294                    _ => match (self.expires_at_seconds, other.expires_at_seconds) {
295                        (None, _) => true,
296                        (_, None) => false,
297                        (Some(a), Some(b)) => a >= b,
298                    },
299                };
300                if self_wins {
301                    self
302                } else {
303                    other
304                }
305            }
306        }
307    }
308}

310/// Live-or-tombstone payload of a [`CompactionRow`].
311#[derive(Debug, Clone, PartialEq)]
312pub enum CompactionRowData {
313    /// A complete range tombstone (issue #933): the paired start + end bounds of
314    /// a clustering-range delete, with the authoritative deletion timestamps.
315    ///
316    /// The reader pairs the on-disk start/end bound markers (or boundary markers)
317    /// into one self-contained range so the compaction merge can shadow covered
318    /// cells AND re-emit the surviving marker to the output SSTable. `deletion_time`
319    /// is `markedForDeleteAt` (microseconds); `local_deletion_time` is the GC-grace
320    /// clock (seconds, carried as the wrapping `as u32 as i32` for far-future LDTs).
321    RangeMarker {
322        /// Start bound of the deleted clustering range.
323        start: CompactionBound,
324        /// End bound of the deleted clustering range.
325        end: CompactionBound,
326        /// `markedForDeleteAt` in microseconds.
327        deletion_time: i64,
328        /// `localDeletionTime` in seconds (GC-grace clock).
329        local_deletion_time: i32,
330    },
331    /// Partition-level tombstone (whole-partition delete) carrying its
332    /// authoritative timestamps (issue #1072).
333    ///
334    /// Surfaced by the compaction read path as a synthetic carrier row (no
335    /// clustering) so the cross-generation merge can apply the partition deletion
336    /// as the OUTERMOST floor — shadowing every older cell/row/range/complex
337    /// marker across ALL merge sources — and re-emit the surviving partition
338    /// tombstone to the output SSTable. Without this carrier a newer partition
339    /// tombstone in one SSTable failed to shadow older live rows in another,
340    /// resurrecting deleted partitions. `deletion_time` is `markedForDeleteAt`
341    /// (microseconds); `local_deletion_time` is the GC-grace clock (seconds,
342    /// carried as the wrapping `as u32 as i32` for far-future LDTs).
343    PartitionDelete {
344        /// `markedForDeleteAt` in microseconds.
345        deletion_time: i64,
346        /// `localDeletionTime` in seconds (GC-grace clock).
347        local_deletion_time: i32,
348    },
349    /// Row tombstone (whole-row delete) carrying its authoritative timestamps.
350    Tombstone {
351        /// `markedForDeleteAt` in microseconds.
352        deletion_time: i64,
353        /// `localDeletionTime` in seconds (GC-grace clock).
354        local_deletion_time: i32,
355        /// Clustering columns `(name, value)` in schema order identifying which
356        /// clustering row this tombstone deletes (#912). On disk a row tombstone
357        /// still carries its clustering prefix; capturing it here lets the merge
358        /// route the tombstone into its own clustering bucket instead of
359        /// collapsing every row tombstone (and the static row) into the single
360        /// `None` bucket. Empty for an unclustered table (the partition's single
361        /// row) and for the legacy collapsed-value fallback.
362        clustering: Vec<(String, Value)>,
363    },
364    /// Live row: simple (single-cell) columns plus complex (multi-cell)
365    /// columns with their per-element cells and optional complex deletion.
366    Live {
367        /// Simple, single-cell columns (incl. clustering columns surfaced as
368        /// cells, and cell tombstones for deleted simple columns).
369        simple: Vec<SimpleCell>,
370        /// Complex (non-frozen collection / UDT) columns, each with its
371        /// per-element cells and optional complex deletion.
372        complex: Vec<ComplexColumn>,
373        /// Row-level deletion that COEXISTS with the surviving live cells
374        /// (issue #932). `Some((markedForDeleteAt µs, localDeletionTime s))`
375        /// when this row carried `HAS_DELETION` AND still has surviving cells
376        /// (the cells the merge kept are strictly newer than the deletion). The
377        /// deletion is preserved so it keeps shadowing older cells of OTHER
378        /// columns in SSTables not part of a partial compaction. `None` for a
379        /// plain live row with no row deletion. A row whose ONLY payload is the
380        /// deletion (no surviving cells) is a [`Self::Tombstone`], not a `Live`
381        /// with this field set.
382        row_deletion: Option<(i64, i32)>,
383        /// Primary-key (row-marker) liveness of this row (issue #2374/#2789),
384        /// carried so a READ consumer can hide a row whose only content is an
385        /// EXPIRED liveness marker plus already-tombstoned cells. Carry-only for
386        /// reads — the compaction WRITE path ignores it (byte-parity preserved).
387        row_liveness: RowLiveness,
388    },
389}
390
391impl CompactionRowData {
392    /// Issue #3809 (Finding 1): the identity invariant of a ROW DELETION — refuse
393    /// one that has LOST the clustering of the row it deletes.
394    ///
395    /// The subject is the row deletion, NOT one enum variant: a row deletion is
396    /// carried by [`Self::Tombstone`] (nothing survived it) AND by
397    /// `Self::Live { row_deletion: Some(..) }` (issue #932 — cells written after it
398    /// survive), and both hand the same `deletion_time` to the merge, so both are
399    /// checked. The builder in `row_decoder/compaction.rs` calls this for EVERY
400    /// non-static row carrying a row deletion, before it decides which of the two
401    /// variants to emit. A row with NO row deletion is deliberately NOT checked: it
402    /// has no `deletion_time`, so it cannot become the `None` bucket's row deletion
403    /// and can shadow nothing (the boundary is stated at the call site).
404    ///
405    /// `declared_clustering_columns` is the table's clustering arity;
406    /// `recovered_clustering_values` is how many of those the row actually yielded.
407    /// Returns `Err(Error::Corruption)` when a NON-static row on a CLUSTERED table
408    /// recovered fewer than it declares, and `Ok(())` otherwise.
409    ///
410    /// `pub(crate)` deliberately (issue #3366): the ONE production caller is the
411    /// private per-element builder in `row_decoder/compaction.rs`, and nothing in
412    /// this repo detects public-API drift, so this is not exported. The scalar
413    /// cases are pinned by the unit tests at the foot of this file; the byte-level
414    /// case is `cqlite-core/tests/issue_3809_tombstone_clustering_identity.rs`.
415    ///
416    /// # Why this cannot be a silent `clustering: []`
417    ///
418    /// An empty clustering does not merely lose precision, it loses IDENTITY.
419    /// `SSTableRowIteratorAdapter::extract_clustering_key_from_compaction` maps it
420    /// to `None` on BOTH arms — from `Tombstone.clustering`, and for a `Live` row
421    /// from its `simple` cells — so the row joins the `None` reconcile bucket, the
422    /// bucket that also holds the partition's STATIC row, where `merge::reconcile`
423    /// adopts its `deletion_time` as the WHOLE group's row deletion and
424    /// `shadow_by_row_deletion` drops every cell in that bucket at or below it. The
425    /// row deletion is then either dropped outright (resurrecting the very row it
426    /// was written to delete) or, with no static column present, written by
427    /// `data_writer::rows` with no clustering prefix and no `IS_STATIC` flag: a
428    /// structurally invalid row on a clustered table. All of that on the WRITE path,
429    /// with no diagnostic.
430    ///
431    /// # Cassandra authority (pinned `cassandra-5.0.8` — never a working tree)
432    ///
433    /// A clustering prefix is ARITY-TOTAL, so a partial one is not a shape
434    /// Cassandra can write:
435    ///
436    /// * `db/Clustering.java` — `Serializer.serialize` asserts
437    ///   `clustering.size() == types.size()`, and `deserialize` reads exactly
438    ///   `types.size()` values.
439    /// * `db/rows/UnfilteredSerializer.java` — `deserializeTombstonesOnly` builds
440    ///   `BTreeRow.emptyDeletedRow(clustering, deletion)`: a row tombstone always
441    ///   carries a full clustering. The clustering prefix is written from the row's
442    ///   flags before any deletion or cell body, so a row whose deletion coexists
443    ///   with surviving cells carries the same full clustering.
444    ///
445    /// # The two EXEMPT cases, which are correct input and must never red
446    ///
447    /// * `is_static` — Cassandra distinguishes `Clustering.EMPTY` from
448    ///   `Clustering.STATIC_CLUSTERING` by `kind()` (`Clustering.java:102,124`), a
449    ///   distinction this `Vec<(String, Value)>` cannot express, so `[]` is how a
450    ///   static row's clustering is represented here. A static row also carries no
451    ///   clustering prefix on disk, so zero recovered values is EXPECTED.
452    /// * `declared_clustering_columns == 0` — `[]` is the complete and only
453    ///   clustering a table with no clustering columns has.
454    ///
455    /// # Why ARITY only, and never the VALUE (roborev #3809 job 93, REFUTED)
456    ///
457    /// `recovered_clustering_values` is a COUNT of present components; the values
458    /// themselves are deliberately not judged, `Value::Null` included. Refusing a
459    /// null component was proposed and refuted on two independent grounds, both
460    /// from the pinned tag rather than from CQLite's own behaviour:
461    ///
462    /// * The harm is UNREACHABLE for a row. The only writer of the null header
463    ///   state is `ClusteringPrefix.Serializer.makeHeader` on a `null` component
464    ///   (`db/ClusteringPrefix.java:548`), and the CQL write path rejects a null
465    ///   clustering value before any `Clustering` is built
466    ///   (`cql3/restrictions/SingleColumnRestriction.java:195`,
467    ///   `MultiColumnRestriction.java:242`). The one derived-row path that
468    ///   contemplates a null row clustering is compact-table-gated
469    ///   (`db/view/ViewUpdateGenerator.java:179`) and `isCompactTable()` is hard
470    ///   `false` in 5.0 (`schema/TableMetadata.java:291`), with startup refused for
471    ///   any table still carrying the pre-4.0 compact flags (`:101`).
472    /// * A value test could not tell that case apart, and would REFUSE correct
473    ///   data. `parse_clustering_prefix` maps the IS_EMPTY header state to
474    ///   `Value::Null` for every clustering type outside a four-spelling allowlist
475    ///   (`row_decoder/row_framing.rs`), whereas Cassandra materialises an empty
476    ///   clustering value as a zero-length value EXPLICITLY distinct from null
477    ///   (`ClusteringPrefix.java:514`) and declares empty valid input for 17 type
478    ///   families (`allowsEmpty()`, `db/marshal/AbstractType.java:513`) — of which
479    ///   `StringType`, `BytesType`, `TupleType` and `AbstractCompositeType` treat
480    ///   it as MEANINGFUL rather than null-equivalent (`isEmptyValueMeaningless()`,
481    ///   `AbstractType.java:458`, overridden true by `Int32Type.java:56`). So a
482    ///   null test here reds a legal empty-valued clustering, and a guard that reds
483    ///   on correct input is worse than no guard.
484    ///
485    /// Header state `0b11` needs no arm either: `makeHeader`'s branches are
486    /// exclusive so no writer emits it, and Cassandra's reader tests `isNull` first
487    /// (`ClusteringPrefix.java:565`) and so consumes no value bytes — byte-for-byte
488    /// what CQLite does. `ClusteringPrefix.Kind` is likewise out of scope here: a
489    /// row's prefix carries NO kind byte (`UnfilteredSerializer.java:191`, `:481`
490    /// use `Clustering.serializer`), the ordinal being written only for markers
491    /// (`:287`, `ClusteringBoundOrBoundary.java:105`), which never reach this
492    /// builder.
493    ///
494    /// The IS_EMPTY/IS_NULL conflation named above is a real upstream read-path
495    /// divergence, tracked as issue #3988, but it is not this invariant's to fix:
496    /// refusing a lost identity BY VALUE requires the two states to be
497    /// distinguishable here first.
498    ///
499    /// # Why `Error::Corruption`, and what it does NOT claim
500    ///
501    /// The variant is kept for its CONTRACT: `Error::is_recoverable` returns
502    /// `false` for it, so compaction stops instead of retrying a defect that
503    /// re-deriving the same input reproduces exactly. But `Corruption` renders as
504    /// `Data corruption: ...`, and with today's decoder that attribution would be
505    /// wrong: `parse_clustering_prefix` is itself arity-total (it pushes exactly
506    /// `schema.clustering_keys.len()` values or returns `Err`), so a short
507    /// clustering cannot reach here from Cassandra-written bytes — a fire points
508    /// at a CQLite decoder / column-resolution regression, not at damaged data.
509    /// The message therefore SAYS SO, rather than leaving an operator hunting a
510    /// disk fault. (`Error::Internal` would carry the same `is_recoverable ==
511    /// false` but retitles the telemetry bucket from `Corruption` to `Other`
512    /// (`observability/error_schema.rs:162,209`) and drops the `Data` category;
513    /// the guard cannot itself tell a reader defect from a hand-written or
514    /// otherwise non-Cassandra file, so the data-shaped bucket is kept and the
515    /// uncertainty is stated in the text.)
516    pub(crate) fn require_tombstone_clustering_identity(
517        keyspace: &str,
518        table: &str,
519        is_static: bool,
520        declared_clustering_columns: usize,
521        recovered_clustering_values: usize,
522    ) -> Result<()> {
523        // `declared_clustering_columns == 0` is STRUCTURALLY REDUNDANT under the
524        // `<` predicate below (`recovered < 0` cannot hold) and is retained on
525        // purpose: it is the exemption this function's contract states, and it is
526        // what keeps a table with no clustering columns correct if the predicate is
527        // ever tightened to `!=` or `>` (an arity-TOTAL comparison, which is what
528        // `Clustering.java`'s own assert is). Both directions are pinned by the
529        // unit tests at the foot of this file.
530        if is_static || declared_clustering_columns == 0 {
531            return Ok(());
532        }
533        if recovered_clustering_values < declared_clustering_columns {
534            return Err(Error::corruption(format!(
535                "row deletion in {keyspace}.{table} lost its clustering identity: \
536                 the schema declares {declared_clustering_columns} clustering \
537                 column(s) but only {recovered_clustering_values} clustering \
538                 value(s) were recovered from the row's clustering prefix (#912). \
539                 This covers a row deletion with or without surviving cells (#932), \
540                 since both carry the same deletion timestamp into the merge. \
541                 Cassandra never writes a partial clustering (Clustering.java's \
542                 clustering.size() == types.size() assert), and emitting an empty \
543                 clustering would make this row deletion identify no row (issue \
544                 #3809). The ON-DISK BYTES MAY BE SOUND: the clustering-prefix \
545                 decoder is arity-total, so this shape is unreachable from \
546                 Cassandra-written input and most likely indicates a CQLite \
547                 decoder or column-resolution regression rather than damaged data"
548            )));
549        }
550        Ok(())
551    }
552}

554/// A single-cell (simple) column value with its write metadata.
555///
556/// Cell tombstones for simple columns are represented by `value` holding a
557/// `Value::Tombstone(CellTombstone)` (matching the legacy compaction stream).
558#[derive(Debug, Clone, PartialEq)]
559pub struct SimpleCell {
560    /// Column name.
561    pub column: String,
562    /// Decoded cell value (or `Value::Tombstone` for a cell delete).
563    pub value: Value,
564    /// Effective cell write timestamp in microseconds (cell-own timestamp when
565    /// present, else the row liveness timestamp).
566    pub timestamp: i64,
567    /// TTL in seconds when the cell is expiring (`None` otherwise).
568    pub ttl: Option<u32>,
569    /// `localDeletionTime` in seconds for an expiring / tombstone cell
570    /// (`None` when not applicable).
571    pub local_deletion_time: Option<i32>,
572}

574/// A complex (non-frozen collection / UDT) column: its per-element cells plus
575/// an optional complex deletion marker covering elements written at or before
576/// `marked_for_delete_at`.
577#[derive(Debug, Clone, PartialEq)]
578pub struct ComplexColumn {
579    /// Column name.
580    pub column: String,
581    /// `Some((markedForDeleteAt µs, localDeletionTime s))` when a real complex
582    /// deletion is present; `None` for the `LIVE` sentinel (no overwrite).
583    pub complex_deletion: Option<(i64, i32)>,
584    /// Per-element cells in on-disk order (epic #899 substrate / contract).
585    pub elements: Vec<ComplexElement>,
586    /// The whole-collection `Value` the reader collapses this column into
587    /// (`Value::List` / `Value::Set` / `Value::Map`), EXACTLY as the
588    /// pre-Phase-A read path produced it (SET/LIST element tombstones dropped,
589    /// MAP null/tombstoned entries kept as `(key, Null)`, empty/overwritten
590    /// collections kept as the empty collection).
591    ///
592    /// PHASE A NEUTRALITY (roborev #863, Finding 3): the merge OUTPUT path uses
593    /// this collapsed value so the (untouched) writer emits byte-identical bytes
594    /// to pre-Phase-A. The per-element `elements` ride alongside as the Phase-C
595    /// foundation and are asserted by the reader-contract tests; per-element
596    /// writer emit is Phase C.
597    pub collapsed_value: Value,
598}
599
600/// A single element of a complex column (a list/set member, or a map entry).
601#[derive(Debug, Clone, PartialEq)]
602pub struct ComplexElement {
603    /// Raw cell-path bytes that identify this element (the serialized element
604    /// for a SET, the key for a MAP, the 16-byte TimeUUID for a LIST). Must be
605    /// preserved byte-for-byte so the writer can round-trip it.
606    pub cell_path: Vec<u8>,
607    /// Decoded element value (`None` for a tombstoned or empty-value element,
608    /// e.g. SET members which store the element in the path with an empty
609    /// value).
610    pub value: Option<Value>,
611    /// Decoded element key for a MAP entry (the map key parsed from the
612    /// `cell_path`). `None` for LIST / SET / UDT elements. Used to reconstruct a
613    /// whole `Value::Map` for the writer-facing mutation while reconcile still
614    /// keys on the raw `cell_path` bytes (epic #899, Phase A bridge).
615    pub decoded_key: Option<Value>,
616    /// Per-element write timestamp in microseconds (element-own when present,
617    /// else the row liveness timestamp).
618    pub timestamp: i64,
619    /// TTL in seconds when the element is expiring (`None` otherwise).
620    pub ttl: Option<u32>,
621    /// `localDeletionTime` in seconds for an expiring / deleted element
622    /// (`None` when not applicable). Far-future values in `[2^31, 2^32)` are
623    /// kept as the wrapping `as u32 as i32` representation.
624    pub local_deletion_time: Option<i32>,
625    /// Whether this element carries the IS_DELETED (0x01) flag (an
626    /// element-level tombstone).
627    pub is_deleted: bool,
628    /// Whether the on-disk cell carried the HAS_EMPTY_VALUE (0x04) flag.
629    ///
630    /// `true` for a SET member (whose value lives in the `cell_path`, not the
631    /// cell value) and for any genuinely empty-value element. The compaction
632    /// writer uses THIS flag — not the decoded [`value`](Self::value) — to decide
633    /// whether to emit an on-disk value, so a SET element round-trips byte-for-
634    /// byte (its decoded member is reconstructed from `cell_path`, never written
635    /// as a cell value). Distinct from `is_deleted`: an empty-value live element
636    /// is not a tombstone.
637    pub has_empty_value: bool,
638}

640#[cfg(test)]
641mod row_liveness_tests {
642    use super::RowLiveness;
643
644    fn marker(ts: i64, expires_at_seconds: Option<i64>) -> RowLiveness {
645        RowLiveness {
646            has_marker: true,
647            expires_at_seconds,
648            marker_timestamp: Some(ts),
649        }
650    }
651
652    /// Issue #2374/#2789: the cross-generation fold is Cassandra last-write-wins
653    /// on the marker WRITE timestamp, NOT a most-permissive union. A newer
654    /// EXPIRED-TTL marker supersedes an older live-forever marker for a key-only
655    /// row → the row is HIDDEN once `now` passes the newer marker's expiry.
656    ///
657    /// Pre-fix (union) this returned live-forever → `marker_live_at` true → the
658    /// read path wrongly reported the row VISIBLE.
659    #[test]
660    fn newer_expired_ttl_supersedes_older_live_forever() {
661        // gen A: INSERT live-forever @ts=200; gen B: INSERT ... USING TTL @ts=300
662        // whose marker has since EXPIRED (expiry at epoch second 1_000).
663        let gen_a = marker(200, None);
664        let gen_b = marker(300, Some(1_000));
665
666        let merged = gen_a.merge(gen_b);
667        // The NEWER (ts=300) expired marker wins outright.
668        assert_eq!(merged.marker_timestamp, Some(300));
669        assert_eq!(merged.expires_at_seconds, Some(1_000));
670        // Row is HIDDEN once now > expiry.
671        assert!(
672            !merged.marker_live_at(2_000),
673            "newer expired-TTL marker must hide the key-only row (timestamp-LWW)"
674        );
675        // Fold order must not matter.
676        let merged_rev = gen_b.merge(gen_a);
677        assert_eq!(merged_rev.marker_timestamp, Some(300));
678        assert!(!merged_rev.marker_live_at(2_000));
679    }
680
681    /// Reverse of the above: a newer live-forever marker supersedes an older
682    /// expired-TTL one → the key-only row is VISIBLE.
683    #[test]
684    fn newer_live_forever_supersedes_older_expired_ttl() {
685        // gen A: expired-TTL @ts=200 (expiry epoch second 500); gen B: live-forever @ts=300.
686        let gen_a = marker(200, Some(500));
687        let gen_b = marker(300, None);
688
689        let merged = gen_a.merge(gen_b);
690        assert_eq!(merged.marker_timestamp, Some(300));
691        assert_eq!(merged.expires_at_seconds, None);
692        assert!(
693            merged.marker_live_at(2_000),
694            "newer live-forever marker must keep the key-only row visible (timestamp-LWW)"
695        );
696        // Fold order must not matter.
697        assert!(gen_b.merge(gen_a).marker_live_at(2_000));
698    }
699
700    /// Equal timestamps → tie-break on the later expiry (live-forever latest).
701    #[test]
702    fn equal_timestamps_tie_break_on_later_expiry() {
703        let live_forever = marker(300, None);
704        let ttl = marker(300, Some(500));
705        assert_eq!(live_forever.merge(ttl).expires_at_seconds, None);
706        assert_eq!(ttl.merge(live_forever).expires_at_seconds, None);
707
708        let earlier = marker(300, Some(400));
709        let later = marker(300, Some(900));
710        assert_eq!(earlier.merge(later).expires_at_seconds, Some(900));
711        assert_eq!(later.merge(earlier).expires_at_seconds, Some(900));
712    }
713
714    /// Issue #3094: `marker_survives_floor` judges the marker on its OWN write
715    /// timestamp, so a cell tombstone that raises the RECONCILED ROW timestamp
716    /// above a covering deletion cannot resurrect a marker the deletion covers.
717    ///
718    /// Cassandra authority (`cassandra-5.0.8`): `BTreeRow.filter` does
719    /// `if (activeDeletion.deletes(newInfo.timestamp())) newInfo =
720    /// LivenessInfo.EMPTY;`, and `DeletionTime.deletes(long ts)` is
721    /// `ts <= markedForDeleteAt` — hence the equal-timestamp case below is DELETED.
722    #[test]
723    fn marker_survives_floor_uses_the_markers_own_timestamp() {
724        // The #3094 shape: marker @100 <= floor 500 < reconciled row ts 1_000 (raised
725        // by a cell tombstone written AFTER the deletion). The marker is DELETED.
726        assert!(
727            !marker(100, None).marker_survives_floor(1_000, 500),
728            "a marker covered by the floor must not be resurrected by a newer cell \
729             tombstone raising the reconciled row timestamp (#3094)"
730        );
731        // A marker strictly NEWER than the floor survives (the #2374/#2789 key-only
732        // row that coexists with an older partition deletion stays VISIBLE).
733        assert!(marker(900, None).marker_survives_floor(1_000, 500));
734        // Equal timestamps: the deletion wins (`ts <= markedForDeleteAt`).
735        assert!(!marker(500, None).marker_survives_floor(1_000, 500));
736        // The row timestamp is NOT a second necessary condition once a marker is
737        // present: `BTreeRow.filter` decides the marker's fate from `newInfo
738        // .timestamp()` alone and never consults the row's cell timestamps, so a
739        // marker strictly newer than the floor keeps the key-only row VISIBLE even
740        // when every data cell it coexists with is at/below the floor. (Conjoining
741        // the two tests would hide a row Cassandra returns; that this shape cannot
742        // currently reach here is an unasserted invariant of `merge::reconcile`'s
743        // row-ts fold, not a property of this rule.)
744        assert!(marker(900, None).marker_survives_floor(500, 500));
745        // MARKER-LESS row (`marker_timestamp == None`): unchanged behaviour — the
746        // row-level test is the sole condition (#3121 tracks purging such a row).
747        assert!(RowLiveness::default().marker_survives_floor(1_000, 500));
748        assert!(!RowLiveness::default().marker_survives_floor(500, 500));
749    }
750
751    /// A generation with no marker contributes nothing.
752    #[test]
753    fn absent_marker_carries_the_present_one_through() {
754        let present = marker(300, Some(500));
755        assert_eq!(present.merge(RowLiveness::default()), present);
756        assert_eq!(RowLiveness::default().merge(present), present);
757        assert_eq!(
758            RowLiveness::default().merge(RowLiveness::default()),
759            RowLiveness::default()
760        );
761    }
762}
