1//! Per-element / per-cell compaction read contract (epic #899, Phase A).
2//!
3//! The compaction read path historically emitted `(RowKey, Value, i64)` per row,
4//! collapsing every non-frozen collection / UDT into a single nested [`Value`]
5//! with one row-level timestamp. That representation cannot reconcile two
6//! SSTables that wrote DISJOINT elements of the same multi-cell column (each
7//! element has its own timestamp / ttl / local-deletion-time / cell-path), nor
8//! can it carry a real per-column complex deletion (Cassandra's
9//! `markedForDeleteAt` + `localDeletionTime` marker written ahead of a
10//! multi-cell column's elements).
11//!
12//! [`CompactionRow`] replaces that tuple on the **compaction-only** read path. It
13//! preserves the on-disk per-element granularity so the k-way merge can perform
14//! byte-faithful per-`(column, cell_path)` reconciliation. The user-facing read
15//! path (`scan` / `get` / `iterate_all_partitions` / `WRITETIME(collection)`) is
16//! UNCHANGED — it still uses the collapsed [`Value`] representation.
17//!
18//! Byte-format invariants this representation must preserve (see
19//! `docs/sstables-definitive-guide/` Ch.5 + Appendix B):
20//! - Element order is the on-disk order (SET by serialized bytes, MAP by key
21//!   bytes, LIST insertion order with 16-byte TimeUUID paths). Per-element
22//!   timestamps must NOT reorder elements.
23//! - A complex deletion's `LIVE` sentinel is `(i64::MIN, i32::MAX)`; a real
24//!   deletion carries `(markedForDeleteAt µs, localDeletionTime s)`.
25//! - Far-future local deletion times in `[2^31, 2^32)` are preserved as the
26//!   wrapping `as u32 as i32` value — never widened to i64.
27
28use crate::types::{RowKey, TombstoneType, Value};

30/// One row surfaced by the compaction read path, carrying per-element complex
31/// cells and the real per-column complex deletion (epic #899, Phase A).
32///
33/// This is the compaction-only counterpart of the old `(RowKey, Value, i64)`
34/// tuple. `row_timestamp` is the row-level write timestamp (for a tombstone it
35/// is `markedForDeleteAt`); `row_data` holds either a row tombstone or the live
36/// simple + complex cells.
37#[derive(Debug, Clone, PartialEq)]
38pub struct CompactionRow {
39    /// Partition key bytes (token derived downstream).
40    pub key: RowKey,
41    /// Row-level write timestamp in microseconds (for a tombstone:
42    /// `markedForDeleteAt`).
43    pub row_timestamp: i64,
44    /// Row payload: tombstone or live cells.
45    pub row_data: CompactionRowData,
46}
47
48impl CompactionRow {
49    /// Build a [`CompactionRow`] from the legacy collapsed `(RowKey, Value,
50    /// timestamp)` representation (the non-V5 compaction fallback path).
51    ///
52    /// This loses per-element complex granularity (the legacy fallback has none
53    /// to begin with): a `Value::Map` row becomes simple cells, a
54    /// `Value::Tombstone(RowTombstone)` becomes a row tombstone, any other value
55    /// becomes a single `value` cell. The V5CompressedLegacy path bypasses this
56    /// and builds per-element rows directly.
57    pub fn from_legacy_value(key: RowKey, value: Value, row_timestamp: i64) -> Self {
58        let row_data = match value {
59            Value::Tombstone(info) if info.tombstone_type == TombstoneType::RowTombstone => {
60                CompactionRowData::Tombstone {
61                    deletion_time: info.deletion_time,
62                    local_deletion_time: 0,
63                    // The legacy collapsed-value fallback has no clustering capture
64                    // (the clustering prefix is not surfaced on this path), so the
65                    // tombstone lands in the partition's `None` clustering bucket
66                    // exactly as before (#912 carries clustering only on the V5
67                    // per-element path).
68                    clustering: Vec::new(),
69                }
70            }
71            Value::Map(entries) => {
72                let simple = entries
73                    .into_iter()
74                    .map(|(k, v)| {
75                        let column = match k {
76                            Value::Text(s) => s,
77                            other => format!("{:?}", other),
78                        };
79                        let timestamp = match &v {
80                            Value::Tombstone(info) => info.deletion_time,
81                            _ => row_timestamp,
82                        };
83                        SimpleCell {
84                            column,
85                            value: v,
86                            timestamp,
87                            ttl: None,
88                            local_deletion_time: None,
89                        }
90                    })
91                    .collect();
92                CompactionRowData::Live {
93                    simple,
94                    complex: Vec::new(),
95                    // The legacy collapsed-value fallback never carries a
96                    // coexisting row deletion (a `Value::Tombstone` row maps to
97                    // `Tombstone` above; a `Value::Map` row is purely live).
98                    row_deletion: None,
99                }
100            }
101            other => CompactionRowData::Live {
102                simple: vec![SimpleCell {
103                    column: "value".to_string(),
104                    value: other,
105                    timestamp: row_timestamp,
106                    ttl: None,
107                    local_deletion_time: None,
108                }],
109                complex: Vec::new(),
110                row_deletion: None,
111            },
112        };
113        CompactionRow {
114            key,
115            row_timestamp,
116            row_data,
117        }
118    }
119}

121/// A clustering bound of a range-tombstone marker surfaced on the compaction
122/// read path (issue #933).
123///
124/// Reader-native counterpart of
125/// [`crate::storage::write_engine::mutation::ClusteringBound`]; kept here so the
126/// compaction read contract does not depend on the write-engine types. Each
127/// bound carries its clustering-prefix `(name, value)` pairs (possibly a PREFIX
128/// shorter than the full clustering arity). An open bound (the writer emits these
129/// as an inclusive bound with zero clustering values) is [`Self::Bottom`] /
130/// [`Self::Top`].
131#[derive(Debug, Clone, PartialEq)]
132pub enum CompactionBound {
133    /// Inclusive bound (the clustering prefix is part of the deletion range).
134    Inclusive(Vec<(String, Value)>),
135    /// Exclusive bound (the clustering prefix is NOT part of the deletion range).
136    Exclusive(Vec<(String, Value)>),
137    /// Before all clustering keys (start of partition).
138    Bottom,
139    /// After all clustering keys (end of partition).
140    Top,
141}
142
143/// Live-or-tombstone payload of a [`CompactionRow`].
144#[derive(Debug, Clone, PartialEq)]
145pub enum CompactionRowData {
146    /// A complete range tombstone (issue #933): the paired start + end bounds of
147    /// a clustering-range delete, with the authoritative deletion timestamps.
148    ///
149    /// The reader pairs the on-disk start/end bound markers (or boundary markers)
150    /// into one self-contained range so the compaction merge can shadow covered
151    /// cells AND re-emit the surviving marker to the output SSTable. `deletion_time`
152    /// is `markedForDeleteAt` (microseconds); `local_deletion_time` is the GC-grace
153    /// clock (seconds, carried as the wrapping `as u32 as i32` for far-future LDTs).
154    RangeMarker {
155        /// Start bound of the deleted clustering range.
156        start: CompactionBound,
157        /// End bound of the deleted clustering range.
158        end: CompactionBound,
159        /// `markedForDeleteAt` in microseconds.
160        deletion_time: i64,
161        /// `localDeletionTime` in seconds (GC-grace clock).
162        local_deletion_time: i32,
163    },
164    /// Row tombstone (whole-row delete) carrying its authoritative timestamps.
165    Tombstone {
166        /// `markedForDeleteAt` in microseconds.
167        deletion_time: i64,
168        /// `localDeletionTime` in seconds (GC-grace clock).
169        local_deletion_time: i32,
170        /// Clustering columns `(name, value)` in schema order identifying which
171        /// clustering row this tombstone deletes (#912). On disk a row tombstone
172        /// still carries its clustering prefix; capturing it here lets the merge
173        /// route the tombstone into its own clustering bucket instead of
174        /// collapsing every row tombstone (and the static row) into the single
175        /// `None` bucket. Empty for an unclustered table (the partition's single
176        /// row) and for the legacy collapsed-value fallback.
177        clustering: Vec<(String, Value)>,
178    },
179    /// Live row: simple (single-cell) columns plus complex (multi-cell)
180    /// columns with their per-element cells and optional complex deletion.
181    Live {
182        /// Simple, single-cell columns (incl. clustering columns surfaced as
183        /// cells, and cell tombstones for deleted simple columns).
184        simple: Vec<SimpleCell>,
185        /// Complex (non-frozen collection / UDT) columns, each with its
186        /// per-element cells and optional complex deletion.
187        complex: Vec<ComplexColumn>,
188        /// Row-level deletion that COEXISTS with the surviving live cells
189        /// (issue #932). `Some((markedForDeleteAt µs, localDeletionTime s))`
190        /// when this row carried `HAS_DELETION` AND still has surviving cells
191        /// (the cells the merge kept are strictly newer than the deletion). The
192        /// deletion is preserved so it keeps shadowing older cells of OTHER
193        /// columns in SSTables not part of a partial compaction. `None` for a
194        /// plain live row with no row deletion. A row whose ONLY payload is the
195        /// deletion (no surviving cells) is a [`Self::Tombstone`], not a `Live`
196        /// with this field set.
197        row_deletion: Option<(i64, i32)>,
198    },
199}

201/// A single-cell (simple) column value with its write metadata.
202///
203/// Cell tombstones for simple columns are represented by `value` holding a
204/// `Value::Tombstone(CellTombstone)` (matching the legacy compaction stream).
205#[derive(Debug, Clone, PartialEq)]
206pub struct SimpleCell {
207    /// Column name.
208    pub column: String,
209    /// Decoded cell value (or `Value::Tombstone` for a cell delete).
210    pub value: Value,
211    /// Effective cell write timestamp in microseconds (cell-own timestamp when
212    /// present, else the row liveness timestamp).
213    pub timestamp: i64,
214    /// TTL in seconds when the cell is expiring (`None` otherwise).
215    pub ttl: Option<u32>,
216    /// `localDeletionTime` in seconds for an expiring / tombstone cell
217    /// (`None` when not applicable).
218    pub local_deletion_time: Option<i32>,
219}
220
221/// A complex (non-frozen collection / UDT) column: its per-element cells plus
222/// an optional complex deletion marker covering elements written at or before
223/// `marked_for_delete_at`.
224#[derive(Debug, Clone, PartialEq)]
225pub struct ComplexColumn {
226    /// Column name.
227    pub column: String,
228    /// `Some((markedForDeleteAt µs, localDeletionTime s))` when a real complex
229    /// deletion is present; `None` for the `LIVE` sentinel (no overwrite).
230    pub complex_deletion: Option<(i64, i32)>,
231    /// Per-element cells in on-disk order (epic #899 substrate / contract).
232    pub elements: Vec<ComplexElement>,
233    /// The whole-collection `Value` the reader collapses this column into
234    /// (`Value::List` / `Value::Set` / `Value::Map`), EXACTLY as the
235    /// pre-Phase-A read path produced it (SET/LIST element tombstones dropped,
236    /// MAP null/tombstoned entries kept as `(key, Null)`, empty/overwritten
237    /// collections kept as the empty collection).
238    ///
239    /// PHASE A NEUTRALITY (roborev #863, Finding 3): the merge OUTPUT path uses
240    /// this collapsed value so the (untouched) writer emits byte-identical bytes
241    /// to pre-Phase-A. The per-element `elements` ride alongside as the Phase-C
242    /// foundation and are asserted by the reader-contract tests; per-element
243    /// writer emit is Phase C.
244    pub collapsed_value: Value,
245}
246
247/// A single element of a complex column (a list/set member, or a map entry).
248#[derive(Debug, Clone, PartialEq)]
249pub struct ComplexElement {
250    /// Raw cell-path bytes that identify this element (the serialized element
251    /// for a SET, the key for a MAP, the 16-byte TimeUUID for a LIST). Must be
252    /// preserved byte-for-byte so the writer can round-trip it.
253    pub cell_path: Vec<u8>,
254    /// Decoded element value (`None` for a tombstoned or empty-value element,
255    /// e.g. SET members which store the element in the path with an empty
256    /// value).
257    pub value: Option<Value>,
258    /// Decoded element key for a MAP entry (the map key parsed from the
259    /// `cell_path`). `None` for LIST / SET / UDT elements. Used to reconstruct a
260    /// whole `Value::Map` for the writer-facing mutation while reconcile still
261    /// keys on the raw `cell_path` bytes (epic #899, Phase A bridge).
262    pub decoded_key: Option<Value>,
263    /// Per-element write timestamp in microseconds (element-own when present,
264    /// else the row liveness timestamp).
265    pub timestamp: i64,
266    /// TTL in seconds when the element is expiring (`None` otherwise).
267    pub ttl: Option<u32>,
268    /// `localDeletionTime` in seconds for an expiring / deleted element
269    /// (`None` when not applicable). Far-future values in `[2^31, 2^32)` are
270    /// kept as the wrapping `as u32 as i32` representation.
271    pub local_deletion_time: Option<i32>,
272    /// Whether this element carries the IS_DELETED (0x01) flag (an
273    /// element-level tombstone).
274    pub is_deleted: bool,
275    /// Whether the on-disk cell carried the HAS_EMPTY_VALUE (0x04) flag.
276    ///
277    /// `true` for a SET member (whose value lives in the `cell_path`, not the
278    /// cell value) and for any genuinely empty-value element. The compaction
279    /// writer uses THIS flag — not the decoded [`value`](Self::value) — to decide
280    /// whether to emit an on-disk value, so a SET element round-trips byte-for-
281    /// byte (its decoded member is reconstructed from `cell_path`, never written
282    /// as a cell value). Distinct from `is_deleted`: an empty-value live element
283    /// is not a tombstone.
284    pub has_empty_value: bool,
285}
