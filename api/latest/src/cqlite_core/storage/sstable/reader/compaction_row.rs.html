1//! Per-element / per-cell compaction read contract (epic #899, Phase A).
2//!
3//! The compaction read path historically emitted `(RowKey, ScanRow, i64)` per row,
4//! collapsing every non-frozen collection / UDT into a single nested [`Value`]
5//! with one row-level timestamp. That representation cannot reconcile two
6//! SSTables that wrote DISJOINT elements of the same multi-cell column (each
7//! element has its own timestamp / ttl / local-deletion-time / cell-path), nor
8//! can it carry a real per-column complex deletion (Cassandra's
9//! `markedForDeleteAt` + `localDeletionTime` marker written ahead of a
10//! multi-cell column's elements).
11//!
12//! [`CompactionRow`] replaces that tuple on the **compaction-only** read path. It
13//! preserves the on-disk per-element granularity so the k-way merge can perform
14//! byte-faithful per-`(column, cell_path)` reconciliation. The user-facing read
15//! path (`scan` / `get` / `iterate_all_partitions` / `WRITETIME(collection)`) is
16//! UNCHANGED — it still uses the collapsed [`Value`] representation.
17//!
18//! Byte-format invariants this representation must preserve (see
19//! `docs/sstables-definitive-guide/` Ch.5 + Appendix B):
20//! - Element order is the on-disk order (SET by serialized bytes, MAP by key
21//!   bytes, LIST insertion order with 16-byte TimeUUID paths). Per-element
22//!   timestamps must NOT reorder elements.
23//! - A complex deletion's `LIVE` sentinel is `(i64::MIN, i32::MAX)`; a real
24//!   deletion carries `(markedForDeleteAt µs, localDeletionTime s)`.
25//! - Far-future local deletion times in `[2^31, 2^32)` are preserved as the
26//!   wrapping `as u32 as i32` value — never widened to i64.
27
28use crate::types::{RowKey, ScanRow, TombstoneType, Value};

30/// One row surfaced by the compaction read path, carrying per-element complex
31/// cells and the real per-column complex deletion (epic #899, Phase A).
32///
33/// This is the compaction-only counterpart of the old `(RowKey, ScanRow, i64)`
34/// tuple. `row_timestamp` is the row-level write timestamp (for a tombstone it
35/// is `markedForDeleteAt`); `row_data` holds either a row tombstone or the live
36/// simple + complex cells.
37#[derive(Debug, Clone, PartialEq)]
38pub struct CompactionRow {
39    /// Partition key bytes (token derived downstream).
40    pub key: RowKey,
41    /// Row-level write timestamp in microseconds (for a tombstone:
42    /// `markedForDeleteAt`).
43    pub row_timestamp: i64,
44    /// Row payload: tombstone or live cells.
45    pub row_data: CompactionRowData,
46}
47
48impl CompactionRow {
49    /// Build a [`CompactionRow`] from the legacy collapsed `(RowKey, ScanRow,
50    /// timestamp)` representation (the non-V5 compaction fallback path).
51    ///
52    /// This loses per-element complex granularity (the legacy fallback has none
53    /// to begin with): a live `ScanRow::Row` becomes simple cells, a
54    /// `ScanRow::Marker(Value::Tombstone(RowTombstone))` becomes a row tombstone,
55    /// any other marker becomes a single `value` cell. The V5CompressedLegacy
56    /// path bypasses this and builds per-element rows directly.
57    ///
58    /// Issue #1334: the reader carries every row through the single [`ScanRow`]
59    /// carrier — this consumer disassembles that same carrier (no `Value::Map`
60    /// bifurcation).
61    pub fn from_legacy_value(key: RowKey, row: ScanRow, row_timestamp: i64) -> Self {
62        let row_data = match row {
63            // A live row's interned cells become simple cells.
64            ScanRow::Row(entries) => {
65                let simple = entries
66                    .into_iter()
67                    .map(|(k, v)| {
68                        let timestamp = match &v {
69                            Value::Tombstone(info) => info.deletion_time,
70                            _ => row_timestamp,
71                        };
72                        SimpleCell {
73                            column: k.to_string(),
74                            value: v,
75                            timestamp,
76                            ttl: None,
77                            local_deletion_time: None,
78                        }
79                    })
80                    .collect();
81                CompactionRowData::Live {
82                    simple,
83                    complex: Vec::new(),
84                    // A live `ScanRow::Row` never carries a coexisting row deletion
85                    // (a row tombstone arrives as a `ScanRow::Marker`, handled below).
86                    row_deletion: None,
87                    row_liveness: RowLiveness::default(),
88                }
89            }
90            // A row tombstone marker becomes a row tombstone.
91            ScanRow::Marker(Value::Tombstone(info))
92                if info.tombstone_type == TombstoneType::RowTombstone =>
93            {
94                CompactionRowData::Tombstone {
95                    deletion_time: info.deletion_time,
96                    local_deletion_time: 0,
97                    // The legacy collapsed-value fallback has no clustering capture
98                    // (the clustering prefix is not surfaced on this path), so the
99                    // tombstone lands in the partition's `None` clustering bucket
100                    // exactly as before (#912 carries clustering only on the V5
101                    // per-element path).
102                    clustering: Vec::new(),
103                }
104            }
105            // A raw undecoded fallback row collapses to a single `value` cell
106            // carrying the raw bytes as a blob — the exact pre-#1334 shape a bare
107            // `Value::Blob` produced in this legacy collapsed-value fallback.
108            ScanRow::RawRow(bytes) => CompactionRowData::Live {
109                simple: vec![SimpleCell {
110                    column: "value".to_string(),
111                    value: Value::Blob(bytes.into()),
112                    timestamp: row_timestamp,
113                    ttl: None,
114                    local_deletion_time: None,
115                }],
116                complex: Vec::new(),
117                row_deletion: None,
118                row_liveness: RowLiveness::default(),
119            },
120            // Any other marker (null row, cell tombstone, …) collapses to a single
121            // `value` cell, exactly as the pre-#1334 fallback did.
122            ScanRow::Marker(other) => CompactionRowData::Live {
123                simple: vec![SimpleCell {
124                    column: "value".to_string(),
125                    value: other,
126                    timestamp: row_timestamp,
127                    ttl: None,
128                    local_deletion_time: None,
129                }],
130                complex: Vec::new(),
131                row_deletion: None,
132                row_liveness: RowLiveness::default(),
133            },
134        };
135        CompactionRow {
136            key,
137            row_timestamp,
138            row_data,
139        }
140    }
141}

143/// A clustering bound of a range-tombstone marker surfaced on the compaction
144/// read path (issue #933).
145///
146/// Reader-native counterpart of
147/// [`crate::storage::write_engine::mutation::ClusteringBound`]; kept here so the
148/// compaction read contract does not depend on the write-engine types. Each
149/// bound carries its clustering-prefix `(name, value)` pairs (possibly a PREFIX
150/// shorter than the full clustering arity). An open bound (the writer emits these
151/// as an inclusive bound with zero clustering values) is [`Self::Bottom`] /
152/// [`Self::Top`].
153#[derive(Debug, Clone, PartialEq)]
154pub enum CompactionBound {
155    /// Inclusive bound (the clustering prefix is part of the deletion range).
156    Inclusive(Vec<(String, Value)>),
157    /// Exclusive bound (the clustering prefix is NOT part of the deletion range).
158    Exclusive(Vec<(String, Value)>),
159    /// Before all clustering keys (start of partition).
160    Bottom,
161    /// After all clustering keys (end of partition).
162    Top,
163}

165/// Primary-key (row-marker) liveness surfaced on a live compaction row so a
166/// READ consumer can apply Cassandra's row-visibility rule (issue #2374/#2789).
167///
168/// A row is visible to a `SELECT` iff it has at least one live data cell OR a
169/// LIVE primary-key liveness marker (`HAS_TIMESTAMP`, whose TTL — if any — has
170/// not expired). Compaction (the WRITE path) does NOT consult this — it retains
171/// an expired marker within gc_grace for byte-parity — so this is carry-only for
172/// the read side (Flight `do_get` / cross-generation read merge), never a write
173/// decision. No-heuristics (#28): both fields come from the authoritative
174/// on-disk row header, never inferred.
175#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
176pub struct RowLiveness {
177    /// Whether the row carried a primary-key liveness marker (`HAS_TIMESTAMP` —
178    /// i.e. it was INSERTed, not created implicitly by a data-cell UPDATE).
179    pub has_marker: bool,
180    /// Marker expiry (epoch seconds) when the liveness marker carries a TTL
181    /// (`HAS_TTL`); `None` when the marker is live-forever (no TTL). Only
182    /// meaningful when `has_marker` is `true`. Reinterpreted UNSIGNED upstream
183    /// so a post-2038 expiry is not wrapped negative.
184    pub expires_at_seconds: Option<i64>,
185    /// The marker's authoritative WRITE timestamp in microseconds (the row
186    /// header liveness `timestamp`), when present. Populated from the on-disk
187    /// row header only (no-heuristics, #28) — never inferred. Carries the
188    /// last-write-wins key for [`Self::merge`] so a cross-generation fold keeps
189    /// the NEWER liveness marker outright (a newer TTL'd marker can supersede an
190    /// older live-forever one, and vice-versa) instead of a most-permissive
191    /// union. `None` when `has_marker` is `false`.
192    pub marker_timestamp: Option<i64>,
193}
194
195impl RowLiveness {
196    /// `true` when the primary-key liveness marker is present AND still live at
197    /// `now_secs` (no TTL, or its expiry is strictly in the future). Matches
198    /// `Cell.isLive`: live iff `now < localExpirationTime`.
199    pub fn marker_live_at(&self, now_secs: i64) -> bool {
200        self.has_marker && self.expires_at_seconds.is_none_or(|s| s > now_secs)
201    }
202
203    /// Fold two liveness markers across generations by Cassandra
204    /// last-write-wins on the marker WRITE timestamp (issue #2374/#2789): the
205    /// marker with the higher [`marker_timestamp`](Self::marker_timestamp) wins
206    /// OUTRIGHT — its expiry / live-forever state is taken as-is, so a newer
207    /// TTL'd marker supersedes an older live-forever one (and vice-versa). Later
208    /// expiry is only a TIE-BREAK when the timestamps compare equal (or both are
209    /// absent). A generation with no marker contributes nothing, so the surviving
210    /// marker (if any) is carried through unchanged.
211    ///
212    /// This replaces the former most-permissive union, which unconditionally let
213    /// a live-forever marker win regardless of write order and diverged from
214    /// Cassandra for a reverse-timestamp reinsertion (older live-forever, newer
215    /// expired TTL): the union reported the row VISIBLE where Cassandra keeps the
216    /// newer expired liveness and HIDES it.
217    pub fn merge(self, other: RowLiveness) -> RowLiveness {
218        match (self.has_marker, other.has_marker) {
219            (false, false) => RowLiveness::default(),
220            (true, false) => self,
221            (false, true) => other,
222            (true, true) => {
223                // Both generations carry a marker: last-write-wins on the
224                // authoritative write timestamp; later expiry breaks a tie.
225                let self_wins = match (self.marker_timestamp, other.marker_timestamp) {
226                    (Some(a), Some(b)) if a != b => a > b,
227                    // Equal (or one/both absent) timestamps → tie-break on the
228                    // later expiry, treating live-forever (`None`) as the latest.
229                    _ => match (self.expires_at_seconds, other.expires_at_seconds) {
230                        (None, _) => true,
231                        (_, None) => false,
232                        (Some(a), Some(b)) => a >= b,
233                    },
234                };
235                if self_wins {
236                    self
237                } else {
238                    other
239                }
240            }
241        }
242    }
243}

245/// Live-or-tombstone payload of a [`CompactionRow`].
246#[derive(Debug, Clone, PartialEq)]
247pub enum CompactionRowData {
248    /// A complete range tombstone (issue #933): the paired start + end bounds of
249    /// a clustering-range delete, with the authoritative deletion timestamps.
250    ///
251    /// The reader pairs the on-disk start/end bound markers (or boundary markers)
252    /// into one self-contained range so the compaction merge can shadow covered
253    /// cells AND re-emit the surviving marker to the output SSTable. `deletion_time`
254    /// is `markedForDeleteAt` (microseconds); `local_deletion_time` is the GC-grace
255    /// clock (seconds, carried as the wrapping `as u32 as i32` for far-future LDTs).
256    RangeMarker {
257        /// Start bound of the deleted clustering range.
258        start: CompactionBound,
259        /// End bound of the deleted clustering range.
260        end: CompactionBound,
261        /// `markedForDeleteAt` in microseconds.
262        deletion_time: i64,
263        /// `localDeletionTime` in seconds (GC-grace clock).
264        local_deletion_time: i32,
265    },
266    /// Partition-level tombstone (whole-partition delete) carrying its
267    /// authoritative timestamps (issue #1072).
268    ///
269    /// Surfaced by the compaction read path as a synthetic carrier row (no
270    /// clustering) so the cross-generation merge can apply the partition deletion
271    /// as the OUTERMOST floor — shadowing every older cell/row/range/complex
272    /// marker across ALL merge sources — and re-emit the surviving partition
273    /// tombstone to the output SSTable. Without this carrier a newer partition
274    /// tombstone in one SSTable failed to shadow older live rows in another,
275    /// resurrecting deleted partitions. `deletion_time` is `markedForDeleteAt`
276    /// (microseconds); `local_deletion_time` is the GC-grace clock (seconds,
277    /// carried as the wrapping `as u32 as i32` for far-future LDTs).
278    PartitionDelete {
279        /// `markedForDeleteAt` in microseconds.
280        deletion_time: i64,
281        /// `localDeletionTime` in seconds (GC-grace clock).
282        local_deletion_time: i32,
283    },
284    /// Row tombstone (whole-row delete) carrying its authoritative timestamps.
285    Tombstone {
286        /// `markedForDeleteAt` in microseconds.
287        deletion_time: i64,
288        /// `localDeletionTime` in seconds (GC-grace clock).
289        local_deletion_time: i32,
290        /// Clustering columns `(name, value)` in schema order identifying which
291        /// clustering row this tombstone deletes (#912). On disk a row tombstone
292        /// still carries its clustering prefix; capturing it here lets the merge
293        /// route the tombstone into its own clustering bucket instead of
294        /// collapsing every row tombstone (and the static row) into the single
295        /// `None` bucket. Empty for an unclustered table (the partition's single
296        /// row) and for the legacy collapsed-value fallback.
297        clustering: Vec<(String, Value)>,
298    },
299    /// Live row: simple (single-cell) columns plus complex (multi-cell)
300    /// columns with their per-element cells and optional complex deletion.
301    Live {
302        /// Simple, single-cell columns (incl. clustering columns surfaced as
303        /// cells, and cell tombstones for deleted simple columns).
304        simple: Vec<SimpleCell>,
305        /// Complex (non-frozen collection / UDT) columns, each with its
306        /// per-element cells and optional complex deletion.
307        complex: Vec<ComplexColumn>,
308        /// Row-level deletion that COEXISTS with the surviving live cells
309        /// (issue #932). `Some((markedForDeleteAt µs, localDeletionTime s))`
310        /// when this row carried `HAS_DELETION` AND still has surviving cells
311        /// (the cells the merge kept are strictly newer than the deletion). The
312        /// deletion is preserved so it keeps shadowing older cells of OTHER
313        /// columns in SSTables not part of a partial compaction. `None` for a
314        /// plain live row with no row deletion. A row whose ONLY payload is the
315        /// deletion (no surviving cells) is a [`Self::Tombstone`], not a `Live`
316        /// with this field set.
317        row_deletion: Option<(i64, i32)>,
318        /// Primary-key (row-marker) liveness of this row (issue #2374/#2789),
319        /// carried so a READ consumer can hide a row whose only content is an
320        /// EXPIRED liveness marker plus already-tombstoned cells. Carry-only for
321        /// reads — the compaction WRITE path ignores it (byte-parity preserved).
322        row_liveness: RowLiveness,
323    },
324}

326/// A single-cell (simple) column value with its write metadata.
327///
328/// Cell tombstones for simple columns are represented by `value` holding a
329/// `Value::Tombstone(CellTombstone)` (matching the legacy compaction stream).
330#[derive(Debug, Clone, PartialEq)]
331pub struct SimpleCell {
332    /// Column name.
333    pub column: String,
334    /// Decoded cell value (or `Value::Tombstone` for a cell delete).
335    pub value: Value,
336    /// Effective cell write timestamp in microseconds (cell-own timestamp when
337    /// present, else the row liveness timestamp).
338    pub timestamp: i64,
339    /// TTL in seconds when the cell is expiring (`None` otherwise).
340    pub ttl: Option<u32>,
341    /// `localDeletionTime` in seconds for an expiring / tombstone cell
342    /// (`None` when not applicable).
343    pub local_deletion_time: Option<i32>,
344}

346/// A complex (non-frozen collection / UDT) column: its per-element cells plus
347/// an optional complex deletion marker covering elements written at or before
348/// `marked_for_delete_at`.
349#[derive(Debug, Clone, PartialEq)]
350pub struct ComplexColumn {
351    /// Column name.
352    pub column: String,
353    /// `Some((markedForDeleteAt µs, localDeletionTime s))` when a real complex
354    /// deletion is present; `None` for the `LIVE` sentinel (no overwrite).
355    pub complex_deletion: Option<(i64, i32)>,
356    /// Per-element cells in on-disk order (epic #899 substrate / contract).
357    pub elements: Vec<ComplexElement>,
358    /// The whole-collection `Value` the reader collapses this column into
359    /// (`Value::List` / `Value::Set` / `Value::Map`), EXACTLY as the
360    /// pre-Phase-A read path produced it (SET/LIST element tombstones dropped,
361    /// MAP null/tombstoned entries kept as `(key, Null)`, empty/overwritten
362    /// collections kept as the empty collection).
363    ///
364    /// PHASE A NEUTRALITY (roborev #863, Finding 3): the merge OUTPUT path uses
365    /// this collapsed value so the (untouched) writer emits byte-identical bytes
366    /// to pre-Phase-A. The per-element `elements` ride alongside as the Phase-C
367    /// foundation and are asserted by the reader-contract tests; per-element
368    /// writer emit is Phase C.
369    pub collapsed_value: Value,
370}

372/// A single element of a complex column (a list/set member, or a map entry).
373#[derive(Debug, Clone, PartialEq)]
374pub struct ComplexElement {
375    /// Raw cell-path bytes that identify this element (the serialized element
376    /// for a SET, the key for a MAP, the 16-byte TimeUUID for a LIST). Must be
377    /// preserved byte-for-byte so the writer can round-trip it.
378    pub cell_path: Vec<u8>,
379    /// Decoded element value (`None` for a tombstoned or empty-value element,
380    /// e.g. SET members which store the element in the path with an empty
381    /// value).
382    pub value: Option<Value>,
383    /// Decoded element key for a MAP entry (the map key parsed from the
384    /// `cell_path`). `None` for LIST / SET / UDT elements. Used to reconstruct a
385    /// whole `Value::Map` for the writer-facing mutation while reconcile still
386    /// keys on the raw `cell_path` bytes (epic #899, Phase A bridge).
387    pub decoded_key: Option<Value>,
388    /// Per-element write timestamp in microseconds (element-own when present,
389    /// else the row liveness timestamp).
390    pub timestamp: i64,
391    /// TTL in seconds when the element is expiring (`None` otherwise).
392    pub ttl: Option<u32>,
393    /// `localDeletionTime` in seconds for an expiring / deleted element
394    /// (`None` when not applicable). Far-future values in `[2^31, 2^32)` are
395    /// kept as the wrapping `as u32 as i32` representation.
396    pub local_deletion_time: Option<i32>,
397    /// Whether this element carries the IS_DELETED (0x01) flag (an
398    /// element-level tombstone).
399    pub is_deleted: bool,
400    /// Whether the on-disk cell carried the HAS_EMPTY_VALUE (0x04) flag.
401    ///
402    /// `true` for a SET member (whose value lives in the `cell_path`, not the
403    /// cell value) and for any genuinely empty-value element. The compaction
404    /// writer uses THIS flag — not the decoded [`value`](Self::value) — to decide
405    /// whether to emit an on-disk value, so a SET element round-trips byte-for-
406    /// byte (its decoded member is reconstructed from `cell_path`, never written
407    /// as a cell value). Distinct from `is_deleted`: an empty-value live element
408    /// is not a tombstone.
409    pub has_empty_value: bool,
410}

412#[cfg(test)]
413mod row_liveness_tests {
414    use super::RowLiveness;
415
416    fn marker(ts: i64, expires_at_seconds: Option<i64>) -> RowLiveness {
417        RowLiveness {
418            has_marker: true,
419            expires_at_seconds,
420            marker_timestamp: Some(ts),
421        }
422    }
423
424    /// Issue #2374/#2789: the cross-generation fold is Cassandra last-write-wins
425    /// on the marker WRITE timestamp, NOT a most-permissive union. A newer
426    /// EXPIRED-TTL marker supersedes an older live-forever marker for a key-only
427    /// row → the row is HIDDEN once `now` passes the newer marker's expiry.
428    ///
429    /// Pre-fix (union) this returned live-forever → `marker_live_at` true → the
430    /// read path wrongly reported the row VISIBLE.
431    #[test]
432    fn newer_expired_ttl_supersedes_older_live_forever() {
433        // gen A: INSERT live-forever @ts=200; gen B: INSERT ... USING TTL @ts=300
434        // whose marker has since EXPIRED (expiry at epoch second 1_000).
435        let gen_a = marker(200, None);
436        let gen_b = marker(300, Some(1_000));
437
438        let merged = gen_a.merge(gen_b);
439        // The NEWER (ts=300) expired marker wins outright.
440        assert_eq!(merged.marker_timestamp, Some(300));
441        assert_eq!(merged.expires_at_seconds, Some(1_000));
442        // Row is HIDDEN once now > expiry.
443        assert!(
444            !merged.marker_live_at(2_000),
445            "newer expired-TTL marker must hide the key-only row (timestamp-LWW)"
446        );
447        // Fold order must not matter.
448        let merged_rev = gen_b.merge(gen_a);
449        assert_eq!(merged_rev.marker_timestamp, Some(300));
450        assert!(!merged_rev.marker_live_at(2_000));
451    }
452
453    /// Reverse of the above: a newer live-forever marker supersedes an older
454    /// expired-TTL one → the key-only row is VISIBLE.
455    #[test]
456    fn newer_live_forever_supersedes_older_expired_ttl() {
457        // gen A: expired-TTL @ts=200 (expiry epoch second 500); gen B: live-forever @ts=300.
458        let gen_a = marker(200, Some(500));
459        let gen_b = marker(300, None);
460
461        let merged = gen_a.merge(gen_b);
462        assert_eq!(merged.marker_timestamp, Some(300));
463        assert_eq!(merged.expires_at_seconds, None);
464        assert!(
465            merged.marker_live_at(2_000),
466            "newer live-forever marker must keep the key-only row visible (timestamp-LWW)"
467        );
468        // Fold order must not matter.
469        assert!(gen_b.merge(gen_a).marker_live_at(2_000));
470    }
471
472    /// Equal timestamps → tie-break on the later expiry (live-forever latest).
473    #[test]
474    fn equal_timestamps_tie_break_on_later_expiry() {
475        let live_forever = marker(300, None);
476        let ttl = marker(300, Some(500));
477        assert_eq!(live_forever.merge(ttl).expires_at_seconds, None);
478        assert_eq!(ttl.merge(live_forever).expires_at_seconds, None);
479
480        let earlier = marker(300, Some(400));
481        let later = marker(300, Some(900));
482        assert_eq!(earlier.merge(later).expires_at_seconds, Some(900));
483        assert_eq!(later.merge(earlier).expires_at_seconds, Some(900));
484    }
485
486    /// A generation with no marker contributes nothing.
487    #[test]
488    fn absent_marker_carries_the_present_one_through() {
489        let present = marker(300, Some(500));
490        assert_eq!(present.merge(RowLiveness::default()), present);
491        assert_eq!(RowLiveness::default().merge(present), present);
492        assert_eq!(
493            RowLiveness::default().merge(RowLiveness::default()),
494            RowLiveness::default()
495        );
496    }
