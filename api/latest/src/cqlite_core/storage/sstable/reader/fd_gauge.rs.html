1//! Reader-reported open-descriptor accounting for [`catalog::READER_FDS_OPEN`]
2//! (issue #1707, AI7 of epic #1686).
3//!
4//! # Why the readers report it, and NOT `/proc`
5//!
6//! `cqlite.proc.fds` already samples `/proc/self/fd` every ~2s (issue #2419) — the
7//! whole process, Linux only, sockets and WAL included. This gauge answers the
8//! narrower question an operator actually needs when a read path is close to
9//! `EMFILE`: how many descriptors are the SSTable READERS holding right now? The
10//! readers know that exactly, at the moment it changes, on every platform — so it is
11//! reported, never sampled and never inferred from a byte pattern or a directory
12//! listing.
13//!
14//! # It counts DESCRIPTORS, so several honest zeros exist
15//!
16//! An mmap-backed source holds a MAPPING, not a descriptor (the fd is closed right
17//! after `mmap`), and an `Arc` clone of an existing handle is not a new descriptor.
18//! Both contribute NOTHING here rather than a plausible-looking number: the #2314
19//! authoritative-data rule forbids inventing a value nobody measured, and an
20//! inflated fd count would send an operator hunting a limit they are nowhere near.
21//!
22//! # One atomic op per change, and the increment cannot be skipped
23//!
24//! [`OpenFdGauge::minted`] is the only PRODUCTION constructor and it increments;
25//! `Drop` decrements. The type's single field is private, so it cannot be built by a
26//! struct literal that forgets the increment (the same structural guarantee
27//! `read_path_probe::BlockingScanTaskGuard` uses), and it is deliberately NOT
28//! `Clone`/`Copy` — cloning it would decrement twice for one descriptor.
29//!
30//! Ordering is `AcqRel` on both sides so a reader thread that observes the
31//! decrement also observes everything the closing thread did before releasing the
32//! handle; the gauge emission then reports the value the same atomic op returned,
33//! never a separate `load` (a load-then-record pair can report a value that was
34//! never current under concurrency).
35//!
36//! # What that does and does NOT promise (issue #1707)
37//!
38//! It promises that every value REPORTED was the true level at some instant — no
39//! reading is fabricated, and none is a stale re-read of a counter that has since
40//! moved. It does NOT promise the last reported value is the CURRENT level: the
41//! atomic op and the gauge emission are two steps, so two threads transitioning at
42//! once (A `fetch_add`→1, B `fetch_sub`→0) can emit in the opposite order to the one
43//! they transitioned in, leaving `1` as the last reported value until the NEXT
44//! transition corrects it. The window is one emission and it is self-healing, which
45//! is why there is deliberately no `Mutex` here: serialising every open and close of
46//! a descriptor to tighten a gauge would be paying real contention on the read path
47//! for a reading that is already eventually right, and the pre-existing
48//! `SSTABLES_OPEN` counter has the byte-for-byte identical shape — one lock here
49//! would leave two patterns for one problem. Read the gauge as a level with a
50//! transition-latency of one event, not as a serialised ledger.
51
52use std::sync::atomic::{AtomicI64, Ordering};
53
54use crate::observability::catalog;
55
56/// Descriptors the SSTable readers currently hold open, process-wide.
57///
58/// `i64` because the gauge is `i64`, and because a signed type makes an unpaired
59/// decrement show up as a NEGATIVE reading — visibly wrong — instead of wrapping to
60/// a huge positive one that reads like genuine fd pressure.
61static READER_FDS_OPEN: AtomicI64 = AtomicI64::new(0);
62
63/// One open file descriptor's presence in [`catalog::READER_FDS_OPEN`].
64///
65/// Stored BESIDE the handle it accounts for, so its lifetime is the descriptor's:
66/// every close path — a clean drop, an early return, an unwind — decrements exactly
67/// once, with no `close`-site bookkeeping to forget.
68///
69/// # Declare it AFTER the handle — the field order is load-bearing
70///
71/// Rust drops struct (and enum-variant) fields in DECLARATION order, so a guard
72/// declared BEFORE the handle decrements while the process still holds the
73/// descriptor: the gauge briefly reports FEWER open fds than are really open, which
74/// is the wrong direction for a metric whose whole purpose is spotting `EMFILE`
75/// pressure. Every holder in this module tree therefore declares
76/// `field-holding-the-handle` first and the `OpenFdGauge` last, with a note at the
77/// site so a later tidy-up does not silently reorder them.
78///
79/// The single field is PRIVATE, so this cannot be built by a struct literal that
80/// forgets the increment (the same structural guarantee
81/// `read_path_probe::BlockingScanTaskGuard` uses), and the type is deliberately not
82/// `Clone`/`Copy` — cloning it would decrement twice for one descriptor.
83#[derive(Debug)]
84pub(crate) struct OpenFdGauge {
85    /// The counter this guard accounts into. Always [`READER_FDS_OPEN`] in a
86    /// production build, where [`Self::minted`] is the only constructor; a
87    /// TEST-LOCAL counter under `cfg(test)` via [`Self::minted_into`].
88    counter: &'static AtomicI64,
89}
90
91impl OpenFdGauge {
92    /// Account one descriptor that was JUST opened, and report the new level.
93    ///
94    /// Call this only where an `open(2)` really happened — never for an `Arc` clone,
95    /// a memory mapping, or a handle another `OpenFdGauge` already accounts for.
96    pub(crate) fn minted() -> Self {
97        // The value the increment itself produced (+1 on the previous), so the
98        // reported level is one that was genuinely current.
99        let now = READER_FDS_OPEN.fetch_add(1, Ordering::AcqRel) + 1;
100        record(now);
101        Self {
102            counter: &READER_FDS_OPEN,
103        }
104    }
105
106    /// TEST-ONLY: account into `counter` instead of the process-global one, and emit
107    /// nothing (a test-local counter has no metric series behind it).
108    ///
109    /// # Why the counter has to be injectable to test this at all
110    ///
111    /// The pairing properties — mint increments, `Drop` decrements, an UNWIND still
112    /// decrements — are statements about ONE guard, but [`READER_FDS_OPEN`] is
113    /// process-global and sibling unit tests in this binary open and close their own
114    /// readers CONCURRENTLY. Sampling that global before and after does not isolate
115    /// this test's delta; it just reads a foreign thread's transition as this test's
116    /// own, which is a race that really fired (a gate run observed `before = 24`,
117    /// `after = 23`). Widening the assertion to a range or a direction would not fix
118    /// it — the reading is still one a sibling can move. Giving the guard its own
119    /// counter removes the shared channel instead of trying to filter it, and still
120    /// exercises the same increment / `Drop` / unwind logic.
121    #[cfg(test)]
122    pub(crate) fn minted_into(counter: &'static AtomicI64) -> Self {
123        counter.fetch_add(1, Ordering::AcqRel);
124        Self { counter }
125    }
126}
127
128impl Drop for OpenFdGauge {
129    fn drop(&mut self) {
130        let now = self.counter.fetch_sub(1, Ordering::AcqRel) - 1;
131        // Emit only for the process-global counter — a test-local one has no series.
132        // In a production build `minted` is the only constructor, so this compares a
133        // `&'static` against the very address it was initialised with.
134        if std::ptr::eq(self.counter, &READER_FDS_OPEN) {
135            record(now);
136        }
137    }
138}
139
140/// Emit the gauge. No-op (and zero-cost) when the `observability` feature is off or
141/// no meter provider is installed.
142fn record(level: i64) {
143    crate::observability::record_gauge(catalog::READER_FDS_OPEN, level, &[]);
144}
145
146/// The current level, for tests and for a caller that wants the reading without
147/// waiting for the next change.
148#[cfg(test)]
149pub(crate) fn open_fds() -> i64 {
150    READER_FDS_OPEN.load(Ordering::Acquire)
151}
152
153#[cfg(test)]
154mod tests {
155    use super::*;
156
157    /// A counter of this test's OWN, so the assertions can be exact equalities. See
158    /// [`OpenFdGauge::minted_into`] for why the process-global counter cannot carry
159    /// these properties.
160    static PAIRING_FDS: AtomicI64 = AtomicI64::new(0);
161    /// A second private counter, so the two tests cannot perturb each other even
162    /// when the harness runs them on different threads at the same time.
163    static UNWIND_FDS: AtomicI64 = AtomicI64::new(0);
164
165    #[test]
166    fn a_guard_increments_on_mint_and_decrements_on_drop() {
167        assert_eq!(PAIRING_FDS.load(Ordering::Acquire), 0);
168        {
169            let _a = OpenFdGauge::minted_into(&PAIRING_FDS);
170            assert_eq!(PAIRING_FDS.load(Ordering::Acquire), 1);
171            let _b = OpenFdGauge::minted_into(&PAIRING_FDS);
172            assert_eq!(PAIRING_FDS.load(Ordering::Acquire), 2);
173        }
174        assert_eq!(
175            PAIRING_FDS.load(Ordering::Acquire),
176            0,
177            "both guards decremented — an unpaired increment would leak the level \
178             upward for the rest of the process"
179        );
180    }
181
182    #[test]
183    fn an_unwind_still_decrements() {
184        assert_eq!(UNWIND_FDS.load(Ordering::Acquire), 0);
185        let unwound = std::panic::catch_unwind(|| {
186            let _g = OpenFdGauge::minted_into(&UNWIND_FDS);
187            assert_eq!(UNWIND_FDS.load(Ordering::Acquire), 1);
188            panic!("simulated failure while a descriptor is held");
189        });
190        assert!(unwound.is_err());
191        assert_eq!(
192            UNWIND_FDS.load(Ordering::Acquire),
193            0,
194            "an unwinding open path must not leave the descriptor counted forever"
195        );
196    }
197
198    #[test]
199    fn the_production_constructor_accounts_into_the_process_global_counter() {
200        // The injected-counter tests above would still pass if `minted()` accounted
201        // somewhere else entirely, so pin the wiring itself: hold a guard from the
202        // PRODUCTION constructor and require the global level to be at least 1 while
203        // it lives. A lower bound, because concurrent siblings hold descriptors too —
204        // this asserts the wiring, never the level (the pairing properties are the
205        // deterministic tests above).
206        let _g = OpenFdGauge::minted();
207        assert!(
208            open_fds() >= 1,
209            "minted() must account into READER_FDS_OPEN, the counter the \
210             cqlite.reader.fds.open gauge reports"
211        );
212    }
