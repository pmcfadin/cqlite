1//! Corruption-location resolution for [`VerifyFinding`] (issue #4194).
2//!
3//! This module owns exactly one piece of logic: given a `Data.db` byte range a
4//! finding is anchored to, intersect it against a boundary source's own decoded
5//! partition extents to name which partitions the damage touches. It is a pure,
6//! closed-interval computation — **no byte-pattern scanning of `Data.db` is ever
7//! performed here** (no-heuristics mandate, issue #28; mechanically enforced by
8//! `scripts/tests/test_verify_location_no_resync_scan.sh`, which runs in the
9//! gate's UNSCOPED `roborev-lints` component — `tooling-tests` is diff-scoped
10//! and does not declare `cqlite-core/**` (#4266), so a core-only diff would
11//! skip it there).
12//!
13//! # Coordinate spaces (read this before touching call sites)
14//!
15//! `Index.db`'s `data_offset` and the BTI trie's resolved partition positions
16//! both address the **decompressed (logical)** `Data.db` byte stream — Cassandra
17//! opens a compressed `Data.db` through a reader that presents a virtual
18//! uncompressed view, and both boundary sources were written against that view.
19//! For an uncompressed table logical and physical coordinates coincide.
20//! [`resolve_partitions`] therefore always operates on a *logical* damaged
21//! range; callers computing a chunk's *physical* on-disk byte range (for
22//! [`Location::byte_offset`]/[`Location::byte_len`], which are reported for a
23//! human to go find the bytes) additionally derive the matching logical range
24//! before calling in — see `verify.rs`'s call sites for the per-check
25//! derivation (compressed: `chunk_index * chunk_length`; uncompressed: the
26//! physical offset, unchanged).
27//!
28//! [`VerifyFinding`]: super::verify::VerifyFinding
29//! [`Location::byte_offset`]: Location::byte_offset
30//! [`Location::byte_len`]: Location::byte_len

41/// What a [`Location`]'s physical `byte_offset`/`byte_len` actually describe.
42///
43/// The two readings are not interchangeable: rendering a declared range
44/// identically to a damaged one told the operator that 4 bytes were damaged
45/// when the real damage is the whole logical tail `partitions` enumerates.
46#[derive(Debug, Clone, Copy, PartialEq, Eq)]
47pub enum PhysicalAnchor {
48    /// `byte_offset`/`byte_len` cover bytes that are PRESENT in the file and
49    /// damaged — a failed chunk CRC, a decompression error. Reading that range
50    /// off disk yields the damaged bytes themselves.
51    DamagedExtent,
52    /// `byte_offset`/`byte_len` are a location the SSTable's OWN metadata
53    /// declares that the file does not satisfy: for `ChunkOffsetOutOfBounds`
54    /// the declared chunk offset plus a chunk record's MINIMUM SIZE of 4 bytes
55    /// does not FIT inside `Data.db`. The 4 is the minimum because a record is
56    /// `[compressed_bytes][4-byte trailing CRC32]`
57    /// (`compression_info.rs`), so even a zero-length payload occupies 4
58    /// bytes. The reported range `[byte_offset, byte_offset + 4)` is therefore
59    /// the record's HEAD at the declared offset — NOT the trailing CRC32,
60    /// which lies at the record's far end and whose position is unknown here
61    /// (the payload length is not known for an out-of-bounds chunk). Roborev
62    /// nit N1 (#4194): this doc called the reported range "the 4-byte TRAILING
63    /// inline CRC32", which names the wrong end of the record.
64    ///
65    /// The test is `offset + 4 > data_len` (verify.rs), so it ALSO fires when
66    /// the offset is itself readable and only the record's minimum extent runs
67    /// past EOF — hence "does not fit", never "absent"/"cannot be read",
68    /// matching `format_location`.
69    /// The damaged extent here is the LOGICAL range, which `partitions` names.
70    DeclaredRecord,
71}

73/// Where a [`VerifyFinding`] anchored to a `Data.db` byte range is located, and
74/// which partitions that range intersects.
75///
76/// [`VerifyFinding`]: super::verify::VerifyFinding
77#[derive(Debug, Clone, PartialEq, Eq)]
78pub struct Location {
79    /// The component the byte range belongs to — always `"Data.db"` today (the
80    /// only component with a chunk/offset grid a finding can anchor to).
81    pub component: String,
82    /// Physical (on-disk) start of the range this finding is ANCHORED to in
83    /// `component`. Whether those bytes are damaged, or merely DECLARED by the
84    /// SSTable's own metadata and absent from the file, is stated by
85    /// [`Location::anchor`] — read it before describing this range to a human
86    /// (roborev job 92 MEDIUM: these two fields previously claimed to be "the
87    /// damaged range" unconditionally, which for `ChunkOffsetOutOfBounds` named
88    /// 4 bytes at an offset PAST EOF beside a 900-partition damage list).
89    pub byte_offset: u64,
90    /// Physical (on-disk) length of the anchored range in `component`. See
91    /// [`Location::byte_offset`] and [`Location::anchor`].
92    pub byte_len: u64,
93    /// Whether `byte_offset`/`byte_len` describe damaged bytes or a declared
94    /// record location that the file does not satisfy.
95    pub anchor: PhysicalAnchor,
96    /// The chunk index the damage falls in, when `component` has a chunk grid
97    /// (compressed `CompressionInfo.db` chunks, or the fixed-size `CRC.db`
98    /// grid). `None` for a finding with no chunk grid.
99    pub chunk_index: Option<usize>,
100    /// The partitions whose `Data.db` extent intersects the damaged range.
101    pub partitions: PartitionResolution,
102}

104/// The largest number of partition keys [`resolve_partitions`] will
105/// materialize into `Resolved.keys` (roborev round-2 MEDIUM finding): a
106/// truncation's damaged range is `[first_bad_chunk_start, data_length)` —
107/// for a badly truncated file that can intersect essentially every partition
108/// in the SSTable, and this module's job is precisely to enumerate them by
109/// hex key, materializing one `String` (2x key length) per hit. A `Location`
110/// is a per-FINDING, not per-file, structure, so nothing else in this module
111/// bounds that count. `Resolved.truncated` names how many more intersected
112/// but were not materialized, so the report stays affirmative about what it
113/// omitted rather than either silently truncating or growing unbounded.
114pub const MAX_RESOLVED_KEYS: usize = 100;

116/// The outcome of resolving [`Location::partitions`] against a boundary source.
117#[derive(Debug, Clone, PartialEq, Eq)]
118pub enum PartitionResolution {
119    /// The boundary source was healthy and every intersecting partition's raw
120    /// key was recoverable. `keys` is empty when no boundary entry intersects
121    /// the damaged range (legitimate — e.g. the range falls entirely past the
122    /// last known partition); it is capped at [`MAX_RESOLVED_KEYS`], with
123    /// `truncated` naming how many additional intersecting partitions exist
124    /// beyond the cap (`0` when nothing was omitted).
125    Resolved { keys: Vec<KeyRef>, truncated: usize },
126    /// The boundary source could not be trusted (or a needed partition key was
127    /// unavailable) — the cause is named, never a guess and never a silent
128    /// empty list standing in for "unknown".
129    Unresolved(String),
130}

156/// A chunk/offset-anchored finding awaiting location resolution (issue #4194).
157///
158/// Pushed alongside its finding at the check site — where the physical byte
159/// range and the LOGICAL (decompressed) damaged range are cheaply computable
160/// from local context (`CompressionInfo`, the CRC.db chunk grid) — because
161/// whether the boundary source (`Index.db` / `Partitions.db`) can be TRUSTED
162/// is only knowable once every check has run (design.md §D2). Resolved in one
163/// pass at the end of `verify_components` by [`finalize_locations`].
164///
165/// `pub(crate)` fields (#1116/#1135 file-size relocation): constructed at each
166/// check site in `verify.rs`, resolved here, so both must cross that boundary.
167pub(crate) struct PendingLocation {
168    /// Index into `findings` of the finding this location belongs to.
169    pub(crate) finding_index: usize,
170    /// Component the byte range belongs to — always `"Data.db"` today.
171    pub(crate) component: String,
172    /// Physical (on-disk) start of the damaged range.
173    pub(crate) byte_offset: u64,
174    /// Physical (on-disk) length of the damaged range.
175    pub(crate) byte_len: u64,
176    /// The chunk index the damage falls in, when the component has a chunk grid.
177    pub(crate) chunk_index: Option<usize>,
178    /// The damaged range in `Data.db` LOGICAL (decompressed) offset space —
179    /// the space `Index.db`/the BTI trie address (see this module's doc for
180    /// why this differs from the physical range above).
181    pub(crate) damaged_logical: (u64, u64),
182    /// The total LOGICAL length bounding the last boundary entry's extent.
183    pub(crate) logical_len: u64,
184    /// Where `logical_len` came from. Carried per-check-site because only the
185    /// check site knows (see [`LogicalLenSource`]).
186    pub(crate) logical_len_source: LogicalLenSource,
187    /// Whether the physical range above is a damaged extent or a declared
188    /// record location — carried per-finding because it is a property of the
189    /// CHECK that produced it, not of the component.
190    pub(crate) anchor: PhysicalAnchor,
191}

200/// COMPACT one-line rendering of a [`Location`]: component, chunk and physical
201/// range plus the partition COUNT — never the key list (issue #4194, roborev

203///
204/// `VerifyFinding`'s `Display` renders THIS, not [`format_location`].

208/// in `Display` therefore made the "summary line" run to kilobytes (up to
209/// [`MAX_RESOLVED_KEYS`] hex keys per finding) and duplicated, verbatim, the

217/// Shared renderer behind [`format_location`] and [`format_location_compact`].
218/// ONE implementation on purpose: the CLI previously carried a second,
219/// near-identical renderer that drifted from this one, so the compact variant
220/// is a flag here rather than a separate function body.
221fn format_location_impl(loc: &Location, full: bool) -> String {
222    let chunk = loc
223        .chunk_index
224        .map(|c| format!("chunk {c}, "))
225        .unwrap_or_default();
226    let partitions = match &loc.partitions {
227        PartitionResolution::Resolved { keys, .. } if keys.is_empty() => "0 partitions".to_string(),
228        PartitionResolution::Resolved { keys, truncated } => {
229            // Issue #4194, roborev round-2 MEDIUM finding: `truncated` names
230            // how many more intersecting partitions exist beyond the
231            // `MAX_RESOLVED_KEYS` cap, so a badly truncated Data.db's report
232            // stays affirmative about what it omitted rather than silently
233            // showing a partial list as if it were complete.
234            let more = if *truncated > 0 {
235                format!(" (+{truncated} more, capped)")
236            } else {
237                String::new()
238            };
239            if full {
240                format!(
241                    "{} partition(s){}: {}",
242                    keys.len(),
243                    more,
244                    keys.iter()
245                        .map(|k| k.key_hex.as_str())
246                        .collect::<Vec<_>>()
247                        .join(", ")
248                )
249            } else {
250                format!("{} partition(s){}", keys.len(), more)
251            }
252        }
253        PartitionResolution::Unresolved(cause) => format!("partitions unresolved ({cause})"),
254    };
255    // The physical range is rendered according to what it IS (roborev job 92
256    // MEDIUM). A `DeclaredRecord` anchor names an offset the file does not
257    // reach, so it is labelled `declared` and explicitly disclosed as absent —
258    // rendering it identically to a damaged extent told the operator that 4
259    // bytes were damaged when the real damage was the whole logical tail the
260    // partition list enumerates.
261    let physical = match loc.anchor {
262        PhysicalAnchor::DamagedExtent => {
263            format!("offset 0x{:x} len {}", loc.byte_offset, loc.byte_len)
264        }
265        // "does not fit within the file", NOT "not present" (roborev job
266        // 108). The out-of-bounds test is `offset + 4 > data_len`, which also
267        // fires when the offset itself is INSIDE the file and only the record's
268        // tail runs past the end (data_len 1000, offset 998: two of those bytes
269        // are readable). Claiming the range is absent would over-state the
270        // evidence in exactly the way this anchor exists to prevent.
271        // The compact form keeps the `declared` LABEL — the distinction it
272        // draws (an offset the file does not reach, not a damaged extent) is
273        // the point of the anchor and survives dropping the prose.
274        PhysicalAnchor::DeclaredRecord if !full => {
275            format!(
276                "declared offset 0x{:x} len {}",
277                loc.byte_offset, loc.byte_len
278            )
279        }
280        PhysicalAnchor::DeclaredRecord => format!(
281            "declared offset 0x{:x} len {} (declared by metadata; the record does not fit \
282             within the file)",
283            loc.byte_offset, loc.byte_len
284        ),
285    };
286    format!("{}: {}{} — {}", loc.component, chunk, physical, partitions)
287}

289/// Lower-case hex encode, one `write!` per byte rather than one `String`
290/// allocation per byte (roborev round-2 LOW: `format!` in the loop is on the
291/// path [`MAX_RESOLVED_KEYS`] makes hot — up to 100 calls per `Location`).
292fn hex_encode(bytes: &[u8]) -> String {
293    use std::fmt::Write;
294    let mut out = String::with_capacity(bytes.len() * 2);
295    for b in bytes {
296        let _ = write!(out, "{:02x}", b);
297    }
298    out
299}

358/// The cause named when a BTI boundary source's leaves were never
359/// CORROBORATED against `Data.db` (issue #4194; owner ruling 2026-10-02:
360/// option (a), fail closed).
361///
362/// # Why a BTI `Resolved` needs corroboration and a BIG one does not
363///
364/// A BIG `Index.db` entry is `(raw key, data position)` — one declaration,
365/// parsed together, whose position is now checked against the file's own
366/// declared logical length ([`BOUNDARY_ENTRY_OFFSET_OUT_OF_BOUNDS`],
367/// [`first_order_violation`]). A BTI leaf's identity is NOT one declaration:
368/// the trie emits a byte-comparable PREFIX, and the raw key is recovered
369/// separately through the leaf's PAYLOAD (`Rows.db` for a `RowsOffset` leaf, a
370/// `Data.db` scan for a `DataOffset` leaf). A corruption that keeps a leaf's
371/// prefix while rewriting its payload to point at a DIFFERENT partition thus
372/// yields a trie that parses cleanly and resolves confidently to the WRONG
373/// key — and the only thing that can see it is the FULL-mode identity
374/// cross-check (`bti_partition_identity_mismatch`), which compares every
375/// leaf's resolved payload against the keys decoded from `Data.db`.
376///
377/// # Why "no mismatch reported" is not evidence the leaves agree
378///
379/// The cross-check sits two levels deep inside conditions a corrupt file
380/// routinely trips, so its silence usually means nothing looked:
381///
382///   1. QUICK mode never scans at all.
383///   2. It is nested in `if !compression_metadata_corrupt`, which ANY
384///      `CompressionInfoCorrupt` or `ChunkOffsetOutOfBounds` finding sets — so
385///      a footer-flip plus a corrupt `CompressionInfo.db` evades it with no
386///      `Data.db` damage at all.
387///   3. Most sharply, `ChunkOffsetOutOfBounds` is pushed in
388///      `check_compression_info` ALONGSIDE the very `PendingLocation` that
389///      needs resolving: ONE corruption event both creates the location and
390///      disables the only guard on it.
391///   4. A `Data.db` corrupt enough to produce a location at all commonly
392///      fails the scan outright, which skips the cross-check too.
393///
394/// Resolving against an uncorroborated trie in any of those states is the
395/// confident wrong answer §D2 exists to refuse. A CORROBORATED `Resolved` is
396/// unaffected; restoring an UNCORROBORATED one as a distinct state is option
397/// (b) (milestone 0.19).
398pub const BTI_IDENTITY_UNCORROBORATED: &str = concat!(
399    "BTI partition-index leaves were not corroborated against Data.db ",
400    "(the FULL-mode identity cross-check did not run to completion)"
401);

416/// `Data.db` is SHORTER than the boundary source describes — an OBSERVATION,
417/// deliberately naming no culprit (roborev job 133 MEDIUM, issue #4194).
418///
419/// Distinct from [`BOUNDARY_ENTRY_OFFSET_OUT_OF_BOUNDS`], which does accuse the
420/// boundary source and is correct ONLY against a DECLARED length. Against a
421/// MEASURED length the identical arithmetic has TWO indistinguishable
422/// explanations:
423///
424/// * `Data.db` was truncated and the boundary source is intact, or
425/// * `Data.db` is intact and a boundary entry's position is corrupt.
426///
427/// Nothing available on the uncompressed path separates them — `CRC.db` cannot
428/// (a truncation inside the final chunk leaves the chunk count unchanged, which
429/// is exactly the reachable window), and there is no declared original length
430/// to compare against. Both live cases are covered:
431/// `l1_4_truncated_uncompressed_data_db_names_data_db_not_the_boundary_source`
432/// truncates an intact table, and
433/// `blocker2_index_position_past_the_logical_length_is_refused_by_name`
434/// corrupts an index position on an intact one — and they are
435/// INDISTINGUISHABLE from this evidence, so naming either as the culprit would
436/// be a guess (issue #28). Stating the observation and the ambiguity is the
437/// fail-closed answer; resolving it needs a declared length (follow-up).
438pub const DATA_DB_SHORTER_THAN_BOUNDARY_SOURCE: &str =
439    "Data.db is shorter than the boundary source describes";

441/// Where a `logical_len` came from — carried, never guessed (issue #28).
442///
443/// This distinction is load-bearing, not bookkeeping. The bounds check in
444/// [`resolve_partitions`] asks "does the last boundary entry start beyond the
445/// end of `Data.db`'s logical stream?", and the answer means OPPOSITE things
446/// depending on where the end came from:
447///
448/// * A DECLARED length is independent of the file's current size, so an entry
449///   past it can only mean the boundary source disagrees with itself.
450/// * A MEASURED length SHRINKS when `Data.db` is truncated, so an entry past
451///   it is the expected, correct reading of an INTACT boundary source
452///   describing bytes that no longer exist.
453///
454/// Conflating them made every uncompressed truncation — the headline class
455/// this verb exists to locate — refuse resolution while blaming an intact
456/// `Index.db`.
457#[derive(Debug, Clone, Copy, PartialEq, Eq)]
458pub enum LogicalLenSource {
459    /// The boundary source's own DECLARED total logical length
460    /// (`CompressionInfo::data_length`). Unaffected by truncation.
461    Declared,
462    /// The CURRENT physical `Data.db` length, measured by `stat` at verify
463    /// time (the uncompressed path, where physical == logical). A truncation
464    /// shrinks this value.
465    MeasuredDataDbLength,
466}

468/// The boundary source's trust decision for a WHOLE report (design.md §D2:
469/// fully trusted or not trusted at all, never partially).
470///
471/// Replaces the former `boundary_source_healthy: bool` + `Option<&[_]>` pair
472/// (roborev #4194): two parameters could express four states, two of which
473/// ("healthy but absent", "unhealthy but present") had to be reconciled by
474/// convention at every call site, and the `None` arm could only ever render
475/// ONE generic cause. A rejection now CARRIES its cause, which is what lets
476/// `Index.db`'s open error, its order violation and its out-of-bounds offset
477/// each reach the operator under their own name.
478pub enum BoundarySource<'a> {
479    /// Trusted, pre-sorted ascending by logical `data_offset`.
480    Trusted(&'a [BoundaryEntry]),
481    /// Not trusted — the named cause, never a silent empty list.
482    Rejected(String),
483}

485/// The index of the first boundary entry whose declared `Data.db` position
486/// does not STRICTLY exceed its predecessor's, in ON-DISK PARSE ORDER, or
487/// `None` when the sequence ascends strictly.
488///
489/// # Why non-ascending parse order is corruption, not a quirk
490///
491/// `Index.db` entries are written in decorated-key order, which the format
492/// guarantees is also ascending `Data.db` offset order — the definitive
493/// guide states it outright: "the ordering guarantee that `Index.db` entries
494/// ascend in token order and therefore in `Data.db` offset order"
495/// (`docs/sstables-definitive-guide/chapters/06-index-and-summary.md`,
496/// §"Token Ordering Requirement" and the sequential-windowing note). So a
497/// descent in parse order is a declaration this format cannot produce.
498///
499/// # Why STRICT, not merely non-decreasing
500///
501/// Two partitions cannot share a start offset — every partition occupies at
502/// least a header's worth of bytes. Two EQUAL positions give the earlier
503/// entry the extent `[s, s)`, which intersects nothing, so that partition
504/// would be silently dropped from every location in the report: the same
505/// silent-drop shape as the out-of-bounds offset below, reached by a
506/// different corruption.
507///
508/// This is a consistency check over AUTHORITATIVE metadata the file declares
509/// about itself, not a byte-pattern guess (issue #28) — the same pattern
510/// `verify.rs`'s `check_compression_info` already applies to
511/// `CompressionInfo.db`'s chunk offsets, whose own rationale ("`validate()`
512/// only enforces ascending order; a single corrupted offset (e.g. an MSB set)
513/// is ascending yet points past EOF") transfers here verbatim.
514pub fn first_order_violation(entries: &[BoundaryEntry]) -> Option<usize> {
515    // A running `prev`, deliberately: the slice-pairs and closure-predicate
516    // search primitives `test_verify_location_no_resync_scan.sh` refuses in
517    // this file are the idiomatic alternatives, and this form needs no waiver
518    // for any of them — while the obvious bare indexed `for` loop tripped
519    // clippy's `manual_find`, which is what pushed the first two attempts
520    // toward a waiver. (Naming those primitives literally here would itself
521    // trip the guard, which is purely lexical.)
522    let mut prev: Option<u64> = None;
523    for (i, (offset, _)) in entries.iter().enumerate() {
524        if let Some(previous) = prev {
525            if *offset <= previous {
526                return Some(i);
527            }
528        }
529        prev = Some(*offset);
530    }
531    None
532}

534/// Intersect `damaged` (a closed-open `[start, end)` range in `Data.db`
535/// LOGICAL/decompressed offset space) against `sorted_boundary_entries` — one
536/// `(data_offset, raw_key)` pair per partition the boundary source (`Index.db`
537/// or the BTI `Partitions.db` trie) names, **already sorted ascending by
538/// `data_offset`** (a precondition, not re-sorted here — roborev round-1
539/// MEDIUM finding: this function used to sort its input on EVERY call, i.e.
540/// once per pending location, even though the boundary source is read once
541/// per report and its natural on-disk order is already ascending offset; the
542/// caller sorts once — see `verify.rs`'s `finalize_locations`). `raw_key` is
543/// `None` when the entry's identity is not yet known (see
544/// [`PARTITION_KEY_UNAVAILABLE`]).
545///
546/// `logical_len` bounds the LAST entry's extent (there is no "next entry" to
547/// derive it from): `CompressionInfo::data_length` when compressed, the
548/// physical `Data.db` length when not. `logical_len_source` says WHICH of
549/// those it is — see [`LogicalLenSource`] for why that changes the meaning of
550/// a failed bounds check, and therefore which component gets named.
551///
552/// This is a closed-interval test only — no `Data.db` bytes are read or
553/// scanned here (design.md §D1/§D6, issue #28).
554pub fn resolve_partitions(
555    damaged: (u64, u64),
556    sorted_boundary_entries: &[BoundaryEntry],
557    logical_len: u64,
558    logical_len_source: LogicalLenSource,
559) -> PartitionResolution {
560    // The clear form, restored (roborev job 102 LOW). This was written as a
561    // manual `zip(iter().skip(1))` purely to dodge the no-resync-scan guard's
562    // lexical slice-pairs pattern — contorting production code to satisfy a
563    // grep, over a slice of `(u64, Option<Arc<[u8]>>)` tuples that has nothing
564    // to do with scanning `Data.db` bytes for a header. The guard now has a
565    // named line-level opt-out, which is the right place for the exception, so
566    // the code says what it means and the marker records WHY it is allowed.
567    debug_assert!(
568        sorted_boundary_entries
569            .windows(2) // no-resync-scan-allow:windows — sortedness over boundary tuples, not a byte scan
570            .all(|w| w[0].0 <= w[1].0),
571        "resolve_partitions requires its input sorted ascending by data_offset"
572    );
573
574    // BOUNDS CHECK AGAINST THE DECLARED LOGICAL LENGTH — roborev blocker #2
575    // (#4194). Both halves of the extent derivation below come from
576    // AUTHORITATIVE metadata, but nothing checked that the two AGREE, and
577    // `Index.db`'s own parse cannot: flip one bit of a NON-LEADING byte of a
578    // multi-byte `position` vint and the length prefix is unchanged, so every
579    // entry still parses, `is_fully_parsed()` is true, and no
580    // `IndexEntryCorrupt` fires. The bogus position then sorts LAST, where
581    // `end` is `logical_len` — below its own `start` — and the `end.max(*start)`
582    // clamp below made it the empty extent `[huge, huge)`. That dropped the
583    // partition from the report silently (it intersects nothing, so it is
584    // never reported, never counted in `truncated`, never `Unresolved`) AND
585    // widened its former left neighbour's extent over its real bytes, so a
586    // clean key was printed as damaged with full confidence.
587    //
588    // A consistency check over metadata the file declares about itself, NOT a
589    // byte-pattern guess (issue #28) — the same check `check_compression_info`
590    // already applies to `CompressionInfo.db`'s chunk offsets, for the reason
591    // stated there. `>= logical_len`, not `>`: an entry starting exactly AT
592    // the declared end has the same degenerate `[len, len)` extent, so `>`
593    // would leave that one value silently dropped. O(1), not O(n): the input
594    // is sorted (asserted above), so the last entry is the only candidate.
595    //
596    // WHICH COMPONENT THIS ACCUSES DEPENDS ON `logical_len_source` (roborev job
597    // 133 MEDIUM). The test above is sound only against a length the boundary
598    // source DECLARED. Against a MEASURED `Data.db` length it fires on the
599    // normal, correct reading of a TRUNCATED file — an intact `Index.db`
600    // faithfully describing bytes that no longer exist — so reporting
601    // `BOUNDARY_ENTRY_OFFSET_OUT_OF_BOUNDS` there names the wrong suspect and
602    // suppresses exactly the partition list this verb exists to produce.
603    if let Some((last_start, _)) = sorted_boundary_entries.last() {
604        if *last_start >= logical_len {
605            return PartitionResolution::Unresolved(match logical_len_source {
606                LogicalLenSource::Declared => format!(
607                    "{BOUNDARY_ENTRY_OFFSET_OUT_OF_BOUNDS}: declared position {last_start} \
608                     (0x{last_start:x}) is not below the declared logical length {logical_len}"
609                ),
610                LogicalLenSource::MeasuredDataDbLength => format!(
611                    "{DATA_DB_SHORTER_THAN_BOUNDARY_SOURCE}: boundary source declares a \
612                     partition at position {last_start} (0x{last_start:x}), but Data.db is \
613                     only {logical_len} bytes — from this evidence a Data.db truncation and \
614                     a corrupt boundary-source position are INDISTINGUISHABLE (no declared \
615                     original length exists on the uncompressed path)"
616                ),
617            });
618        }
619    }
620
621    // Bounded DURING accumulation (`hits.len() < MAX_RESOLVED_KEYS`), not just
622    // at the end: a truncation's damaged range `[first_bad_chunk_start,
623    // logical_len)` can intersect essentially every partition in the file, so
624    // materializing them all before capping would defeat the cap.
625    //
626    // `seen` dedups by raw key DURING accumulation too. A post-hoc dedup after
627    // the cap would let a boundary source naming one raw key twice
628    // (defensive-only; not an expected shape) consume TWO cap slots and inflate
629    // `truncated` with pre-dedup entries — rendering e.g. "40 partition(s)
630    // (+60 more, capped)" when only 100 DISTINCT partitions intersect.
631    //
632    // `seen` is itself bounded to MAX_RESOLVED_KEYS: growing it to O(distinct
633    // intersecting partitions) would reintroduce exactly the O(partitions)
634    // growth the cap exists to eliminate. Once `hits` is full, dedup stops — a
635    // duplicate past the cap just counts as one more `truncated` entry, a minor
636    // over-count traded for a hard memory bound.
637    let mut hits: Vec<KeyRef> = Vec::new();
638    let mut seen: std::collections::HashSet<&[u8]> = std::collections::HashSet::new();
639    let mut unknown = false;
640    let mut truncated = 0usize;
641    for (i, (start, key)) in sorted_boundary_entries.iter().enumerate() {
642        let end = sorted_boundary_entries
643            .get(i + 1)
644            .map(|(next_start, _)| *next_start)
645            .unwrap_or(logical_len);
646        // `end >= *start` for EVERY entry, on BOTH arms — so `.max(*start)` is
647        // UNREACHABLE and no longer load-bearing (it was what silently turned a
648        // corrupt BIG entry into an empty extent, blocker #2). BIG: strict
649        // ascent is enforced by `first_order_violation` and the last entry by
650        // the bounds check above. BTI: a `Trusted` source is now CORROBORATED
651        // (#4194 option (a)), and corroboration requires every leaf's
652        // `data_position` to be a DECODED PARTITION START carrying the matching
653        // key, so positions are real and distinct. Kept purely as a
654        // non-panicking release fallback — dropping it would let a future
655        // invariant break compute a reversed range silently instead.
656        debug_assert!(
657            end >= *start,
658            "boundary entry {i} has extent end {end} below its start {start}; the bounds \
659             check in resolve_partitions should have refused this source"
660        );
661        let extent = (*start, end.max(*start));
662        if ranges_intersect(damaged, extent) {
663            match key {
664                Some(k) => {
665                    let raw: &[u8] = k;
666                    if hits.len() < MAX_RESOLVED_KEYS {
667                        if seen.insert(raw) {
668                            hits.push(KeyRef::from_raw(raw));
669                        }
670                    } else if !seen.contains(raw) {
671                        truncated += 1;
672                    }
673                }
674                None => unknown = true,
675            }
676        }
677    }
678
679    if unknown {
680        return PartitionResolution::Unresolved(PARTITION_KEY_UNAVAILABLE.to_string());
681    }
682    hits.sort_by(|a, b| a.key_hex.cmp(&b.key_hex));
683    PartitionResolution::Resolved {
684        keys: hits,
685        truncated,
686    }
687}

689/// Build a [`Location`] for a chunk/offset-anchored finding, fail-closed on a
690/// damaged boundary source (design.md §D2): a
691/// [`BoundarySource::Rejected`] always yields `Unresolved` carrying that
692/// rejection's OWN cause — the boundary source is either fully trusted or not
693/// trusted at all, never partially, and a refusal always names why.
694#[allow(clippy::too_many_arguments)]
695pub fn resolve_location(
696    component: &str,
697    byte_offset: u64,
698    byte_len: u64,
699    anchor: PhysicalAnchor,
700    chunk_index: Option<usize>,
701    boundary: &BoundarySource<'_>,
702    damaged_logical: (u64, u64),
703    logical_len: u64,
704    logical_len_source: LogicalLenSource,
705) -> Location {
706    let partitions = match boundary {
707        BoundarySource::Trusted(entries) => {
708            resolve_partitions(damaged_logical, entries, logical_len, logical_len_source)
709        }
710        BoundarySource::Rejected(cause) => PartitionResolution::Unresolved(cause.clone()),
711    };
712    Location {
713        component: component.to_string(),
714        byte_offset,
715        byte_len,
716        anchor,
717        chunk_index,
718        partitions,
719    }
720}

722/// Resolve every [`PendingLocation`] in one pass and write the result back
723/// onto its owning finding (issue #4194, design.md §D1/§D2).
724///
725/// The boundary source is either fully trusted for this whole report or not
726/// trusted at all — never partially: `boundary_healthy` is a single decision
727/// (no finding against the boundary component, and no index-structure-corrupt
728/// finding class) applied uniformly to every pending location, so a damaged
729/// boundary source poisons ALL of them with
730/// [`BOUNDARY_SOURCE_UNREADABLE`], never just the finding that happens to be
731/// nearest the damage.
732///
733/// `pub(crate)` (#1116 relocation): called from `verify.rs`'s
734/// `verify_components` once every check has run.
735pub(crate) async fn finalize_locations(
736    dir: &Path,
737    components: &ComponentSet,
738    findings: &mut [VerifyFinding],
739    pending: Vec<PendingLocation>,
740    bti_leaves: Option<&[BtiResolvedLeaf]>,
741    scan_position_map: Option<&std::collections::HashMap<u64, Vec<u8>>>,
742    // Whether the FULL-mode BTI identity cross-check ran to completion AND
743    // agreed — see `BTI_IDENTITY_UNCORROBORATED`. Never consulted for BIG.
744    bti_identity_corroborated: bool,
745    platform: Arc<Platform>,
746) {
747    // Distrust on EITHER signal (a union, never component-only): a finding
748    // whose CLASS says the index structure is corrupt, OR any finding at all
749    // against the boundary component itself. The class test alone missed
750    // `MissingComponent`/`UnexpectedEof` against `Index.db`/`Partitions.db`,
751    // and — the genuine wrong-answer case — a missing `Rows.db`, where
752    // `check_bti_structure` still returns the `DataOffset` leaves it could
753    // read: a PARTIAL boundary list that would then be presented as
754    // `Resolved`. The component test cannot replace the class test either,
755    // since `BtiTrieCorrupt` is raised on `Rows.db`, which spec L2 does not
756    // name as a boundary source. Deliberately over-conservative where the two
757    // overlap (a boundary component present on disk but unlisted in TOC.txt
758    // poisons every location): §D2 prefers a refused answer to a confident
759    // wrong one.
760    //
761    // Expressed as an `Option<String>` CAUSE rather than a bool: every
762    // rejection path below (this one, a partial `IndexReader` parse, an open
763    // error, an order violation) carries its own cause through to the
764    // operator (roborev I1/#blocker 2, #4194).
765    let boundary_components: &[&str] = match components.format {
766        SsTableFormat::Big => &["Index.db"],
767        SsTableFormat::Bti => &["Partitions.db", "Rows.db"],
768    };
769    let component_or_class_distrust = findings.iter().any(|f| {
770        boundary_components.contains(&f.component.as_str())
771            || match components.format {
772                SsTableFormat::Big => f.class == VerifyErrorClass::IndexEntryCorrupt,
773                SsTableFormat::Bti => matches!(
774                    f.class,
775                    VerifyErrorClass::BtiRootPointerCorrupt | VerifyErrorClass::BtiTrieCorrupt
776                ),
777            }
778    });
779
780    // Boundary entries: `(logical Data.db position, raw key when known)`, one
781    // per partition the boundary source names — built only when the boundary
782    // source is healthy (an unhealthy one is never read for this purpose,
783    // matching the fail-closed contract regardless of what re-reading it
784    // might yield). Keys are `Arc<[u8]>` (roborev round-1 MEDIUM finding): BIG
785    // reuses `PartitionIndexEntry::raw_key`/`key_digest`'s ALREADY-`Arc`
786    // storage via a refcount bump, never an `O(key_len)` byte copy that would
787    // double the resident partition-index memory for a large table.
788    // BTI CORROBORATION GATE (issue #4194, owner ruling: option (a), fail
789    // closed). Checked AFTER `component_or_class_distrust` so a direct finding
790    // against the boundary source still reports its own, more specific cause.
791    let bti_uncorroborated = components.format == SsTableFormat::Bti && !bti_identity_corroborated;
792    let mut built: Result<Vec<BoundaryEntry>, String> = if component_or_class_distrust {
793        Err(BOUNDARY_SOURCE_UNREADABLE.to_string())
794    } else if bti_uncorroborated {
795        Err(BTI_IDENTITY_UNCORROBORATED.to_string())
796    } else {
797        match components.format {
798            SsTableFormat::Big => {
799                use crate::storage::sstable::index_reader::IndexReader;
800                let index_path = components.path(dir, "Index.db");
801                match IndexReader::open(&index_path, platform).await {
802                    // `IndexReader` uses a DIFFERENT parser from
803                    // `check_big_index`'s structural walk above, and per Check
804                    // 4's own doc it "silently TRUNCATES the partition list on
805                    // the first malformed Index.db entry" (#2302) — exposed via
806                    // `is_fully_parsed()`. So no `IndexEntryCorrupt` does NOT
807                    // mean `IndexReader` parsed the whole file; if the two
808                    // parsers disagree, presenting a partial prefix as
809                    // `Resolved` is a confident WRONG answer, exactly what the
810                    // fail-closed contract (§D2) exists to
811                    // prevent. Downgrade `boundary_healthy` itself (not just
812                    // this arm's `None`) so every OTHER pending location in
813                    // this report is poisoned too, matching "fully trusted or
814                    // not at all".
815                    Ok(reader) if !reader.is_fully_parsed() => {
816                        Err(BOUNDARY_SOURCE_UNREADABLE.to_string())
817                    }
818                    Ok(reader) => {
819                        let entries: Vec<BoundaryEntry> = reader
820                            .get_partition_entries()
821                            .iter()
822                            .map(|e| {
823                                let raw = e.raw_key.clone().unwrap_or_else(|| e.key_digest.clone());
824                                (e.data_offset, Some(raw))
825                            })
826                            .collect();
827                        // PARSE-ORDER VALIDATION (roborev blocker #2, #4194).
828                        // Checked BEFORE the sort below, because the sort is
829                        // exactly what HIDES this corruption: a bit flip in a
830                        // non-leading byte of a `position` vint leaves every
831                        // entry parsing cleanly, and sorting then quietly
832                        // moves the bogus entry into a position where its
833                        // extent collapses and its neighbour's widens. The
834                        // format guarantees strict ascent in parse order
835                        // (definitive guide Ch.6, cited on
836                        // `first_order_violation`), so a violation is
837                        // corruption and the source is refused by name.
838                        match first_order_violation(&entries) {
839                            Some(i) => Err(format!(
840                                "{BOUNDARY_ENTRY_ORDER_VIOLATION}: Index.db entry {i} declares \
841                                 Data.db position {} (0x{:x}) after entry {} declared {} \
842                                 (0x{:x})",
843                                entries[i].0,
844                                entries[i].0,
845                                i - 1,
846                                entries[i - 1].0,
847                                entries[i - 1].0
848                            )),
849                            None => Ok(entries),
850                        }
851                    }
852                    // The REAL error, named (roborev I1, #4194): this arm used
853                    // to discard it and fall through to the generic
854                    // "unavailable", so `Permission denied` on `Index.db`
855                    // reached the operator as no cause at all.
856                    Err(e) => Err(format!(
857                        "{BOUNDARY_SOURCE_OPEN_FAILED}: {}: {e}",
858                        index_path.display()
859                    )),
860                }
861            }
862            SsTableFormat::Bti => match bti_leaves {
863                Some(leaves) => Ok(leaves
864                    .iter()
865                    .map(|leaf| {
866                        let key = leaf.inline_raw_key.as_deref().map(Arc::from).or_else(|| {
867                            scan_position_map
868                                .and_then(|m| m.get(&leaf.data_position))
869                                .map(|k| Arc::from(k.as_slice()))
870                        });
871                        (leaf.data_position, key)
872                    })
873                    .collect()),
874                None => Err(BOUNDARY_SOURCE_UNAVAILABLE.to_string()),
875            },
876        }
877    };
878    // `resolve_partitions` requires its input pre-sorted ascending by
879    // `data_offset` (roborev round-1 MEDIUM — sort ONCE here, not per pending
880    // location). For BIG it is a NO-OP by the time it runs: the format
881    // guarantees strict ascent in parse order (guide Ch.6) and
882    // `first_order_violation` has just refused the source if it did not hold.
883    // Retained because BTI leaves come from a byte-comparable-KEY-order DFS
884    // trie walk, which is NOT Data.db offset order, and so the precondition is
885    // established for BOTH arms by one statement.
886    if let Ok(entries) = built.as_mut() {
887        entries.sort_by_key(|(offset, _)| *offset);
888    }
889    let boundary = match &built {
890        Ok(entries) => BoundarySource::Trusted(entries),
891        Err(cause) => BoundarySource::Rejected(cause.clone()),
892    };
893
894    for p in pending {
895        let location = resolve_location(
896            &p.component,
897            p.byte_offset,
898            p.byte_len,
899            p.anchor,
900            p.chunk_index,
901            &boundary,
902            p.damaged_logical,
903            p.logical_len,
904            p.logical_len_source,
905        );
906        if let Some(f) = findings.get_mut(p.finding_index) {
907            f.location = Some(location);
908        }
909    }
910}
