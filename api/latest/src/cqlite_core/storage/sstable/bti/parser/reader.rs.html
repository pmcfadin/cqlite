25/// BTI header structure for index files
26#[derive(Debug, Clone)]
27pub struct BtiHeader {
28    /// BTI format magic number
29    pub magic: u32,
30    /// Format version
31    pub version: u16,
32    /// Format flags
33    pub flags: u16,
34    /// Offset to root node
35    pub root_offset: u64,
36    /// Number of entries in the index
37    pub entry_count: u64,
38    /// Additional metadata size
39    pub metadata_size: u32,
40}
41
42impl BtiHeader {
43    /// BTI magic number
44    pub const MAGIC: u32 = 0x6461_0000; // 'da\0\0'
45
46    /// Current BTI version
47    pub const VERSION: u16 = 0x0001;
48
49    /// A benign placeholder header for files that carry no fictional
50    /// [`BtiHeader`] (i.e. every real Cassandra `Rows.db`/`Partitions.db`, which
51    /// are footer-rooted).  Used by [`RowsParser::new`] so the trie-based,
52    /// whole-file entry points work on real files that legitimately lack the
53    /// header; the fields are never consulted by those entry points.
54    pub fn placeholder() -> Self {
55        BtiHeader {
56            magic: Self::MAGIC,
57            version: Self::VERSION,
58            flags: 0,
59            root_offset: 0,
60            entry_count: 0,
61            metadata_size: 0,
62        }
63    }
64
65    /// Parse BTI header from bytes
66    pub fn parse(data: &[u8]) -> BtiResult<(Self, usize)> {
67        if data.len() < 24 {
68            return Err(Error::Parse("BTI header too short".to_string()));
69        }
70
71        let magic = u32::from_be_bytes([data[0], data[1], data[2], data[3]]);
72        if magic != Self::MAGIC {
73            return Err(Error::Parse(format!(
74                "Invalid BTI magic: 0x{:08x}, expected 0x{:08x}",
75                magic,
76                Self::MAGIC
77            )));
78        }
79
80        let version = u16::from_be_bytes([data[4], data[5]]);
81        if version != Self::VERSION {
82            return Err(Error::Parse(format!(
83                "Unsupported BTI version: 0x{:04x}, expected 0x{:04x}",
84                version,
85                Self::VERSION
86            )));
87        }
88
89        let flags = u16::from_be_bytes([data[6], data[7]]);
90        let root_offset = u64::from_be_bytes([
91            data[8], data[9], data[10], data[11], data[12], data[13], data[14], data[15],
92        ]);
93        let entry_count = u64::from_be_bytes([
94            data[16], data[17], data[18], data[19], data[20], data[21], data[22], data[23],
95        ]);
96
97        let metadata_size = if data.len() >= 28 {
98            u32::from_be_bytes([data[24], data[25], data[26], data[27]])
99        } else {
100            0
101        };
102
103        let header = BtiHeader {
104            magic,
105            version,
106            flags,
107            root_offset,
108            entry_count,
109            metadata_size,
110        };
111
112        let header_size = if metadata_size > 0 { 28 } else { 24 };
113        Ok((header, header_size))
114    }
115
116    /// Serialize header to bytes
117    pub fn to_bytes(&self) -> Vec<u8> {
118        let mut bytes = Vec::with_capacity(28);
119
120        bytes.extend_from_slice(&self.magic.to_be_bytes());
121        bytes.extend_from_slice(&self.version.to_be_bytes());
122        bytes.extend_from_slice(&self.flags.to_be_bytes());
123        bytes.extend_from_slice(&self.root_offset.to_be_bytes());
124        bytes.extend_from_slice(&self.entry_count.to_be_bytes());
125
126        if self.metadata_size > 0 {
127            bytes.extend_from_slice(&self.metadata_size.to_be_bytes());
128        }
129
130        bytes
131    }
132}

134/// Parser for Partitions.db BTI index
135pub struct PartitionsParser<R: Read + Seek> {
136    /// Input reader
137    reader: R,
138    /// BTI header
139    header: BtiHeader,
140    /// Byte-comparable encoder for key encoding
141    encoder: ByteComparableEncoder,
142    /// Node cache for performance
143    node_cache: FxHashMap<u64, BtiNode>,
144}
145
146impl<R: Read + Seek> PartitionsParser<R> {
147    /// Create new partitions parser
148    pub fn new(mut reader: R) -> BtiResult<Self> {
149        // Read and parse header
150        reader.seek(SeekFrom::Start(0))?;
151        let mut header_data = vec![0u8; 28];
152        reader.read_exact(&mut header_data)?;
153
154        let (header, _) = BtiHeader::parse(&header_data)?;
155
156        Ok(Self {
157            reader,
158            header,
159            encoder: ByteComparableEncoder::new(),
160            node_cache: FxHashMap::default(),
161        })
162    }
163
164    /// Lookup partition by key
165    pub fn lookup_partition(&mut self, partition_key: &[Value]) -> BtiResult<Option<PayloadRef>> {
166        // Encode partition key for lookup
167        let encoded_key = self.encoder.encode_composite_key(partition_key)?;
168
169        // Navigate trie to find the partition
170        let mut navigator = TrieNavigator::new(self.header.root_offset);
171
172        self.lookup_in_trie(&mut navigator, &encoded_key)
173    }
174
175    /// Navigate trie to find encoded key
176    fn lookup_in_trie(
177        &mut self,
178        navigator: &mut TrieNavigator,
179        encoded_key: &[u8],
180    ) -> BtiResult<Option<PayloadRef>> {
181        let mut key_pos = 0;
182
183        loop {
184            // Load current node
185            let current_node = self.load_node(navigator.current_offset)?;
186
187            // If this is a payload-only node (leaf), return its payload
188            if current_node.is_leaf() {
189                return Ok(current_node.get_payload().cloned());
190            }
191
192            // Check if we have a payload at this level (for prefix matches)
193            if let Some(payload) = current_node.get_payload() {
194                if key_pos >= encoded_key.len() {
195                    return Ok(Some(payload.clone()));
196                }
197            }
198
199            // If we've consumed all key bytes, return any payload we have
200            if key_pos >= encoded_key.len() {
201                return Ok(current_node.get_payload().cloned());
202            }
203
204            // Find transition for next byte
205            let next_byte = encoded_key[key_pos];
206            if let Some(child_pointer) = current_node.find_child(next_byte) {
207                navigator.navigate_to_child(next_byte, child_pointer)?;
208                key_pos += 1;
209            } else {
210                // No transition found - key doesn't exist
211                return Ok(None);
212            }
213        }
214    }
215
216    /// Load node from file
217    fn load_node(&mut self, offset: u64) -> BtiResult<BtiNode> {
218        if let Some(cached_node) = self.node_cache.get(&offset) {
219            return Ok(cached_node.clone());
220        }
221
222        // Read node from file
223        self.reader.seek(SeekFrom::Start(offset))?;
224        let mut node_data = vec![0u8; 4096]; // Read up to 4KB for node
225        let bytes_read = self.reader.read(&mut node_data)?;
226        node_data.truncate(bytes_read);
227
228        // Parse node
229        let node = self.parse_node_data(&node_data, offset)?;
230
231        // Cache the node
232        self.node_cache.insert(offset, node.clone());
233        Ok(node)
234    }
235
236    /// Parse node data from bytes.
237    ///
238    /// Delegates to the module-level [`parse_bti_node`] helper which handles
239    /// all 16 BTI node-type ordinals defined in `TrieNode.java`.
240    fn parse_node_data(&self, data: &[u8], offset: u64) -> BtiResult<BtiNode> {
241        parse_bti_node(data, offset)
242    }
243
244    /// Iterator over all partitions in the index
245    pub fn iterate_partitions(&mut self) -> BtiResult<PartitionIterator<'_, R>> {
246        PartitionIterator::new(self)
247    }
248
249    /// Get header information
250    pub fn header(&self) -> &BtiHeader {
251        &self.header
252    }
253
254    /// Get statistics about the index
255    pub fn get_stats(&self) -> BtiIndexStats {
256        BtiIndexStats {
257            entry_count: self.header.entry_count,
258            root_offset: self.header.root_offset,
259            cached_nodes: self.node_cache.len(),
260        }
261    }
262}

264/// Parser for Rows.db BTI index (clustering keys within a partition)
265pub struct RowsParser<R: Read + Seek> {
266    /// Input reader
267    reader: R,
268    /// BTI header
269    header: BtiHeader,
270    /// Byte-comparable encoder for key encoding
271    encoder: ByteComparableEncoder,
272    /// Node cache for performance
273    node_cache: FxHashMap<u64, BtiNode>,
274}
275
276impl<R: Read + Seek> RowsParser<R> {
277    /// Create new rows parser.
278    ///
279    /// A real Cassandra 5.0 `Rows.db` has **no** whole-file [`BtiHeader`] (that
280    /// fictional 28-byte header is only emitted by CQLite's own test/index
281    /// writers); its leading bytes are trie node data and its layout is described
282    /// per-partition via the `RowsOffset` from `Partitions.db`.  The trie-based
283    /// entry points used here ([`range_query`](Self::range_query),
284    /// [`range_query_encoded`](Self::range_query_encoded),
285    /// [`iterate_rows`](Self::iterate_rows)) read the whole file and root from a
286    /// resolved per-partition entry, so they do not consult `self.header`.
287    ///
288    /// We therefore *attempt* to parse the fictional header (for files that do
289    /// carry it) but fall back to a benign default when it is absent — rather
290    /// than rejecting every real `Rows.db` with an "invalid BTI magic" error.
291    pub fn new(mut reader: R) -> BtiResult<Self> {
292        // Attempt to read the (optional, CQLite-only) fictional header.  Real
293        // Rows.db files have no such header; a parse failure there is expected
294        // and must not block construction.
295        reader.seek(SeekFrom::Start(0))?;
296        let mut header_data = vec![0u8; 28];
297        let header = match reader.read_exact(&mut header_data) {
298            Ok(()) => BtiHeader::parse(&header_data)
299                .map(|(h, _)| h)
300                .unwrap_or_else(|_| BtiHeader::placeholder()),
301            // File shorter than 28 bytes (e.g. tiny/empty Rows.db): also fine.
302            Err(_) => BtiHeader::placeholder(),
303        };
304        // Leave the reader positioned at the start for whole-file reads.
305        reader.seek(SeekFrom::Start(0))?;
306
307        Ok(Self {
308            reader,
309            header,
310            encoder: ByteComparableEncoder::new(),
311            node_cache: FxHashMap::default(),
312        })
313    }

393    /// Clustering-key range/slice traversal of a single partition's `Rows.db`
394    /// row-index trie (issue #832), taking **pre-encoded byte-comparable
395    /// clustering bounds** and applying row-index **separator** semantics.
396    ///
397    /// ## Rooting (Finding A)
398    ///
399    /// `rows_offset` is the partition's `RowsOffset` from the `Partitions.db`
400    /// lookup ([`BtiPartitionLocation::RowsOffset`]).  It points at the
401    /// per-partition `TrieIndexEntry`, NOT the trie root, so this resolves the
402    /// real trie root via [`resolve_rows_db_entry`] before traversing.
403    ///
404    /// ## Separator semantics (Finding B)
405    ///
406    /// `encoded_start`/`encoded_end` are inclusive byte-comparable clustering
407    /// bounds in the **same encoding as the trie keys** (Cassandra OSS50
408    /// clustering byte-comparable form).  Because the trie stores separators
409    /// (block boundaries), block selection uses
410    /// [`select_row_index_blocks_for_range`].
411    ///
412    /// Reversed bounds yield an empty result.
413    ///
414    /// Returns the resolved [`BtiRowIndexHeader`] together with the selected
415    /// blocks; each block's `data_offset` is relative to `header.data_position`.
416    pub fn range_query_encoded(
417        &mut self,
418        rows_offset: usize,
419        encoded_start: &[u8],
420        encoded_end: &[u8],
421    ) -> BtiResult<(BtiRowIndexHeader, Vec<BtiRowIndexEntry>)> {
422        let trie_data = self.read_full_rows_db()?;
423        let header = resolve_rows_db_entry(&trie_data, rows_offset)?;
424        let all = iterate_rows_in_bti_trie(&trie_data, header.trie_root)?;
425        let blocks = select_row_index_blocks_for_range(&all, encoded_start, encoded_end);
426        Ok((header, blocks))
427    }

429    /// Typed clustering-key range/slice traversal (issue #832 Finding 1).
430    ///
431    /// Encodes the `Value` clustering bounds in Cassandra **OSS50
432    /// byte-comparable** form — the SAME encoding the `Rows.db` row-index trie
433    /// stores its separators in — via [`encode_clustering_bound_oss50`], then
434    /// delegates to the separator-aware [`range_query_encoded`](Self::range_query_encoded).
435    ///
436    /// Supported clustering types: `int`, `bigint`/`counter`, `smallint`,
437    /// `tinyint`, `boolean`, `timestamp`, `uuid`/`timeuuid`, `text`/`ascii`,
438    /// `blob`/`inet`.  Any other clustering type returns an explicit
439    /// `Error::Parse` (no silent wrong results — issue #28).
440    ///
441    /// `rows_offset` is the partition's `RowsOffset` (resolved to the real trie
442    /// root via [`resolve_rows_db_entry`], Finding A).  Reversed bounds yield an
443    /// empty result.
444    pub fn range_query(
445        &mut self,
446        rows_offset: usize,
447        start_key: &[Value],
448        end_key: &[Value],
449    ) -> BtiResult<Vec<BtiRowIndexEntry>> {
450        let encoded_start = encode_clustering_bound_oss50(start_key)?;
451        let encoded_end = encode_clustering_bound_oss50(end_key)?;
452        if encoded_start > encoded_end {
453            return Ok(Vec::new());
454        }
455        let (_, blocks) = self.range_query_encoded(rows_offset, &encoded_start, &encoded_end)?;
456        Ok(blocks)
457    }

459    /// Order-aware typed clustering range query: the read-side counterpart to the
460    /// writer's [`encode_clustering_bound_oss50_with_order`]. The trie stores
461    /// DESC columns' separators in REVERSED byte-comparable form (complemented
462    /// bytes), so a lookup MUST encode its bounds with the SAME per-column order
463    /// and then compare in the trie's ascending BYTE order.
464    ///
465    /// `is_reversed[i]` matches `start_key[i]`/`end_key[i]` positionally (schema
466    /// clustering order). Both bounds are encoded WITH the per-column order.
467    ///
468    /// ## Reversed-bounds contract (matches [`range_query`](Self::range_query))
469    ///
470    /// Like [`range_query`](Self::range_query), a reversed range yields an empty
471    /// result rather than silently re-ordering the bounds. The bounds are NOT
472    /// swapped.
473    pub fn range_query_with_order(
474        &mut self,
475        rows_offset: usize,
476        start_key: &[Value],
477        end_key: &[Value],
478        is_reversed: &[bool],
479    ) -> BtiResult<Vec<BtiRowIndexEntry>> {
480        let encoded_start = encode_clustering_bound_oss50_with_order(start_key, is_reversed)?;
481        let encoded_end = encode_clustering_bound_oss50_with_order(end_key, is_reversed)?;
482        // Reversed bounds in the trie's ascending byte space yield empty, exactly
483        // like `range_query`. Do NOT swap — pass the encoded bounds through as-is.
484        if encoded_start > encoded_end {
485            return Ok(Vec::new());
486        }
487        let (_, blocks) = self.range_query_encoded(rows_offset, &encoded_start, &encoded_end)?;
488        Ok(blocks)
489    }

521/// Iterator over **all** partitions in a `Partitions.db` BTI index, in
522/// byte-comparable order (issue #832).
523///
524/// The full trie is loaded via the footer-based loader (NOT the fictional
525/// [`BtiHeader`]) and traversed in-order during [`PartitionIterator::new`]; the
526/// materialized entries are then yielded one at a time.
527///
528/// The yielded `Vec<u8>` key is the reconstructed *byte-comparable token* key
529/// (concatenated transition bytes), NOT the original partition key.  The
530/// [`BtiPartitionLocation`] offset is definitive.
531pub struct PartitionIterator<'a, R: Read + Seek> {
532    #[allow(dead_code)]
533    parser: &'a mut PartitionsParser<R>,
534    /// Materialized entries in byte-comparable order.
535    entries: std::vec::IntoIter<(Vec<u8>, BtiPartitionLocation)>,
536    /// A deferred error to surface on the first `next()` call.
537    pending_error: Option<Error>,
538}
539
540impl<'a, R: Read + Seek> PartitionIterator<'a, R> {
541    fn new(parser: &'a mut PartitionsParser<R>) -> BtiResult<Self> {
542        // Load and traverse via the footer-based loader; surface any error on
543        // the first `next()` call rather than failing construction (so callers
544        // that ignore errors still see a non-silent failure).
545        let (entries, pending_error) = match load_bti_trie_via_footer(&mut parser.reader)
546            .and_then(|(trie, root)| dfs_collect_partition_entries(&trie, root))
547        {
548            Ok(v) => (v, None),
549            Err(e) => (Vec::new(), Some(e)),
550        };
551        Ok(Self {
552            parser,
553            entries: entries.into_iter(),
554            pending_error,
555        })
556    }
557}
558
559impl<'a, R: Read + Seek> Iterator for PartitionIterator<'a, R> {
560    type Item = BtiResult<(Vec<u8>, BtiPartitionLocation)>;
561
562    fn next(&mut self) -> Option<Self::Item> {
563        if let Some(err) = self.pending_error.take() {
564            return Some(Err(err));
565        }
566        self.entries.next().map(Ok)
567    }
568}

570/// Iterator over the row-index block entries of **one partition's** `Rows.db`
571/// row-index trie (issue #832), in ascending byte-comparable clustering order.
572///
573/// A real `Rows.db` concatenates one per-partition `TrieIndexEntry` + row-index
574/// trie per wide partition.  The iterator is created from the partition's
575/// `RowsOffset` (from `Partitions.db`); that offset points at the per-partition
576/// entry, so the real trie root is resolved via [`resolve_rows_db_entry`]
577/// (Finding A) before traversal.
578///
579/// An empty (e.g. 0-byte) `Rows.db` yields nothing without panicking.
580pub struct RowIterator<'a, R: Read + Seek> {
581    #[allow(dead_code)]
582    parser: &'a mut RowsParser<R>,
583    entries: std::vec::IntoIter<(Vec<u8>, BtiRowIndexEntry)>,
584    pending_error: Option<Error>,
585}
586
587impl<'a, R: Read + Seek> RowIterator<'a, R> {
588    fn new(parser: &'a mut RowsParser<R>, rows_offset: usize) -> BtiResult<Self> {
589        // An empty Rows.db (e.g. a 0-byte file for SSTables with no row index)
590        // yields nothing rather than erroring.
591        let file_size = parser.reader.seek(SeekFrom::End(0))?;
592        if file_size == 0 {
593            return Ok(Self {
594                parser,
595                entries: Vec::new().into_iter(),
596                pending_error: None,
597            });
598        }
599
600        // Finding A: resolve the per-partition TrieIndexEntry at `rows_offset`
601        // to recover the actual trie root, then traverse from THAT root.
602        let (entries, pending_error) = match parser.read_full_rows_db().and_then(|trie| {
603            let header = resolve_rows_db_entry(&trie, rows_offset)?;
604            iterate_rows_in_bti_trie(&trie, header.trie_root)
605        }) {
606            Ok(v) => (v, None),
607            Err(e) => (Vec::new(), Some(e)),
608        };
609        Ok(Self {
610            parser,
611            entries: entries.into_iter(),
612            pending_error,
613        })
614    }
615}

617impl<'a, R: Read + Seek> Iterator for RowIterator<'a, R> {
618    type Item = BtiResult<(Vec<u8>, BtiRowIndexEntry)>;
619
620    fn next(&mut self) -> Option<Self::Item> {
621        if let Some(err) = self.pending_error.take() {
622            return Some(Err(err));
623        }
624        self.entries.next().map(Ok)
625    }
626}

649    fn make_bti_file(root_node_bytes: Vec<u8>) -> Vec<u8> {
650        let root_offset: u64 = 64; // place root after header + padding
651        let mut data = Vec::new();
652        data.extend_from_slice(&BtiHeader::MAGIC.to_be_bytes());
653        data.extend_from_slice(&BtiHeader::VERSION.to_be_bytes());
654        data.extend_from_slice(&0u16.to_be_bytes()); // flags
655        data.extend_from_slice(&root_offset.to_be_bytes());
656        data.extend_from_slice(&1u64.to_be_bytes()); // entry_count
657        data.extend_from_slice(&0u32.to_be_bytes()); // metadata_size
658        while data.len() < root_offset as usize {
659            data.push(0);
660        }
661        data.extend(root_node_bytes);
662        data
663    }
664
665    /// PayloadOnly (ordinal 0) with a non-zero payload flag (nibble = 1).
666    fn payload_only_node(data_offset: u64, length: u32) -> Vec<u8> {
667        let mut v = vec![0x01u8]; // ordinal=0, payload_flags=1
668        v.extend_from_slice(&data_offset.to_be_bytes());
669        v.extend_from_slice(&length.to_be_bytes());
670        v
671    }
672
673    /// Sparse8 (ordinal 5): [0x50|pf] [count] [count transition bytes] [count 1-byte deltas]
674    fn sparse8_node(payload_flags: u8, pairs: &[(u8, u8)]) -> Vec<u8> {
675        let mut v = vec![0x50 | (payload_flags & 0x0F), pairs.len() as u8];
676        for &(t, _) in pairs {
677            v.push(t);
678        }
679        for &(_, d) in pairs {
680            v.push(d);
681        }
682        v
683    }
684
685    /// Dense16 (ordinal 11): [0xB0|pf] [start] [len-1] [range * 2-byte deltas]
686    fn dense16_node(payload_flags: u8, start: u8, deltas: &[u16]) -> Vec<u8> {
687        let len = deltas.len() as u8;
688        let mut v = vec![0xB0 | (payload_flags & 0x0F), start, len - 1];
689        for &d in deltas {
690            v.extend_from_slice(&d.to_be_bytes());
691        }
692        v
693    }
694
695    /// SingleNoPayload4 (ordinal 1): delta in low 4 bits of first byte, no payload.
696    fn single_nopayload4_node(delta4: u8, transition: u8) -> Vec<u8> {
697        vec![0x10 | (delta4 & 0x0F), transition]
698    }

844    /// Build a 3-leaf Rows.db-style trie via a Sparse8 root (single-byte keys).
845    fn make_rows_trie_three(
846        (k1, p1): (u8, u8),
847        (k2, p2): (u8, u8),
848        (k3, p3): (u8, u8),
849    ) -> (Vec<u8>, usize) {
850        let mut trie = Vec::new();
851        let o1 = trie.len() as u64; // 0
852        trie.extend_from_slice(&[0x01, p1]); // row leaf no marker
853        let o2 = trie.len() as u64; // 2
854        trie.extend_from_slice(&[0x01, p2]);
855        let o3 = trie.len() as u64; // 4
856        trie.extend_from_slice(&[0x01, p3]);
857        let root = trie.len() as u64; // 6
858        trie.push(0x50); // Sparse8
859        trie.push(0x03); // count=3
860        trie.push(k1);
861        trie.push(k2);
862        trie.push(k3);
863        trie.push((root - o1) as u8);
864        trie.push((root - o2) as u8);
865        trie.push((root - o3) as u8);
866        (trie, root as usize)
867    }

869    /// Wrap a single-byte-keyed 3-leaf Rows.db trie behind a synthetic
870    /// per-partition `TrieIndexEntry`, returning `(buffer, rows_offset)`.
871    fn make_rows_db_with_three(
872        (k1, p1): (u8, u8),
873        (k2, p2): (u8, u8),
874        (k3, p3): (u8, u8),
875    ) -> (Vec<u8>, usize) {
876        let (trie, root) = make_rows_trie_three((k1, p1), (k2, p2), (k3, p3));
877        let mut buf = trie; // bytes [0, root..] are the trie nodes
878        let rows_offset = buf.len(); // TrieIndexEntry starts right after the trie
879
880        // key: length 4, value 0x00000007 (content is irrelevant to range tests).
881        buf.extend_from_slice(&4u16.to_be_bytes());
882        buf.extend_from_slice(&[0x00, 0x00, 0x00, 0x07]);
883        let base = rows_offset + 4; // RowsOffset + key_length
884
885        // dataPos = 0 (unsigned vint); block offsets are relative to it.
886        buf.push(0);
887        // rootΔ such that trie_root = `root`: Δ = root - base (zigzag + uvint).
888        let root_delta: i64 = root as i64 - base as i64;
889        let zig = ((root_delta << 1) ^ (root_delta >> 63)) as u64;
890        write_uvint(&mut buf, zig);
891        // blockCount = 3 (unsigned vint).
892        buf.push(3);
893        // partition DeletionTime: MODERN DA live sentinel (no deletion).
894        buf.push(0x80);
895
896        (buf, rows_offset)
897    }

899    /// Minimal unsigned-vint writer (count-leading-ones form) for test fixtures.
900    fn write_uvint(out: &mut Vec<u8>, v: u64) {
901        if v < 0x80 {
902            // 1-byte form: high bit 0, 7 data bits.
903            out.push(v as u8);
904            return;
905        }
906        // 2-byte form: leading byte `10xxxxxx` + 1 trailing byte = 14 data bits.
907        assert!(v < 0x4000, "test vint fixture expects <= 14-bit values");
908        out.push(0x80 | (v >> 8) as u8);
909        out.push((v & 0xFF) as u8);
910    }
