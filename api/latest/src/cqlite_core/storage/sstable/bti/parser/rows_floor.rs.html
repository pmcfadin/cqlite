1//! O(key-length) `Rows.db` separator-floor / strict-ceiling trie walks (issue
2//! #1647 / L1).
3//!
4//! A `Rows.db` row-index trie stores **separators**, not block start keys: the
5//! separator `s_i` labels the boundary at the START of block `i`, and block `i`
6//! covers the half-open clustering interval `[s_i, s_{i+1})`.  Cassandra locates
7//! the block for a clustering key `K` with a downward `separatorFloor(K)` walk
8//! (`org.apache.cassandra.io.sstable.format.bti.RowIndexReader#separatorFloor`,
9//! built on `Walker#prefixAndNeighbours`) that visits O(len(K)) nodes — NOT by
10//! enumerating every block.  Before L1, CQLite's clustering read path materialized
11//! ALL row-index blocks (a full DFS) and then linearly filtered them; on the
12//! acceptance fixture that is ~42 visited nodes per read regardless of how few
13//! blocks a slice touches.
14//!
15//! This module provides the two authoritative walks the clustering-window path
16//! needs, each a byte-by-byte descent that tracks only the closest lesser/greater
17//! branch (never a collection of all blocks):
18//!
19//! - [`rows_floor_block`] — `separatorFloor(K)`: the block whose interval contains
20//!   `K` (the largest separator `<= K`), or `None` when `K` sorts below the first
21//!   separator (the "implicit first block" at the partition body start).
22//! - [`rows_strict_ceiling_block`] — the block with the smallest separator
23//!   **strictly greater** than `K` (`K`'s successor block), or `None` when `K`
24//!   sorts at/after the last separator (the window then runs to the partition end).
25//!
26//! Both are byte-for-byte equivalent to selecting from the full ascending
27//! `(separator, block)` list ([`super::rows::iterate_rows_in_bti_trie`] +
28//! [`super::rows::select_row_index_blocks_for_range`]); that enumerate-then-filter
29//! path is retained for genuine full-partition enumeration.
30//!
31//! Reference: cassandra-5.0.0 `Walker.java` (`follow` / `prefixAndNeighbours` /
32//! `goMin` / `goMax`), `RowIndexReader.java` (`separatorFloor`);
33//! docs/sstables-definitive-guide chapter 17.

53/// Read the `BtiRowIndexEntry` from the payload attached to a `Rows.db` node at
54/// `node_offset` (`None` if it carries no payload).  Pass `parsed = Some(&node)` when
55/// the caller already parsed the node (the DFS single parse, issue #1650 / L3) so a
56/// payload-bearing internal node is not re-parsed; `None` parses it lazily here.
57///
58/// Structurally parallels [`read_node_payload`](super::partitions::read_node_payload)
59/// but decodes the Rows.db payload format via [`decode_bti_row_payload`].
60pub(super) fn read_row_node_payload(
61    trie_data: &[u8],
62    node_offset: usize,
63    node: Option<&BtiNode>,
64) -> BtiResult<Option<BtiRowIndexEntry>> {
65    if node_offset >= trie_data.len() {
66        return Err(Error::Parse(format!(
67            "Rows.db payload read: node_offset {node_offset} out of bounds"
68        )));
69    }
70    let header_byte = trie_data[node_offset];
71    let ordinal = (header_byte >> 4) & 0x0F;
72    let payload_flags = header_byte & 0x0F;
73
74    // SingleNoPayload variants (ordinals 1, 3) carry their delta in the low
75    // nibble and never have a payload.  See `read_node_payload`.
76    if ordinal == 1 || ordinal == 3 {
77        return Ok(None);
78    }
79
80    if ordinal == 0 {
81        if payload_flags == 0 {
82            return Err(Error::Parse(
83                "Rows.db PayloadOnly node has zero payload_flags".to_string(),
84            ));
85        }
86        let payload_start = node_offset + 1;
87        Ok(Some(decode_bti_row_payload(
88            trie_data,
89            payload_start,
90            payload_flags,
91        )?))
92    } else if payload_flags != 0 {
93        // Use the caller's pre-parsed node when supplied (the DFS single parse,
94        // issue #1650), else parse here.  Slice must start at the node.
95        let owned;
96        let node = match node {
97            Some(n) => n,
98            None => {
99                owned = parse_bti_node(&trie_data[node_offset..], node_offset as u64)?;
100                &owned
101            }
102        };
103        let payload_start = payload_start_in_node(node, trie_data, node_offset)?;
104        Ok(Some(decode_bti_row_payload(
105            trie_data,
106            payload_start,
107            payload_flags,
108        )?))
109    } else {
110        Ok(None)
111    }
112}

115/// Result of searching a node's ascending child list for the next key byte
116/// (mirrors `TrieNode.search`).  `insertion` is the number of children whose
117/// transition byte is `< b` (the lower-bound index); `exact` is the child index
118/// when some child's byte equals `b`.
119///
120/// For `b == END_OF_STREAM` (key exhausted, which sorts below every real 0..=255
121/// transition byte) the result is `{ insertion: 0, exact: None }`, exactly as
122/// Cassandra's `search` returns `-1` for the end-of-stream sentinel.
123struct ChildSearch {
124    /// Number of children with transition byte strictly `< b` (lower bound).
125    insertion: usize,
126    /// The child index when `children[i].byte == b`, else `None`.
127    exact: Option<usize>,
128}
129
130#[cfg(not(feature = "tombstones"))]
131/// Locate `b` in the node's ascending `(byte, child)` children.
132fn search_children(children: &[(u8, usize)], b: Option<u8>) -> ChildSearch {
133    let Some(byte) = b else {
134        // END_OF_STREAM sorts below every real transition byte.
135        return ChildSearch {
136            insertion: 0,
137            exact: None,
138        };
139    };
140    // Children are ascending by byte; find the first index with byte >= `byte`.
141    let insertion = children.partition_point(|&(cb, _)| cb < byte);
142    let exact = match children.get(insertion) {
143        Some(&(cb, _)) if cb == byte => Some(insertion),
144        _ => None,
145    };
146    ChildSearch { insertion, exact }
147}

150/// Descend to the MAXIMUM (rightmost) leaf under `node_offset` and return its
151/// row-index payload (mirrors `Walker#goMax`, always following the last child
152/// until a leaf, then reading the payload).
153fn go_max_payload(trie_data: &[u8], mut node_offset: usize) -> BtiResult<BtiRowIndexEntry> {
154    let mut steps = 0usize;
155    loop {
156        record_visit(trie_data, &mut steps)?;
157        let node = parse_bti_node_for_traversal(trie_data, node_offset)?;
158        match ordered_children(&node).last() {
159            Some(&(_, child)) => node_offset = child,
160            None => break,
161        }
162    }
163    read_row_node_payload(trie_data, node_offset, None)?.ok_or_else(|| {
164        Error::Parse(format!(
165            "Rows.db floor: max leaf at offset {node_offset} carries no row-index payload \
166             (corrupt trie)"
167        ))
168    })
169}

171#[cfg(not(feature = "tombstones"))]
172/// Descend to the MINIMUM (leftmost) key under `node_offset` and return its
173/// row-index payload (mirrors `Walker#goMin`, stopping at the first node that
174/// carries a payload, else following the first child to a leaf).
175fn go_min_payload(trie_data: &[u8], mut node_offset: usize) -> BtiResult<BtiRowIndexEntry> {
176    let mut steps = 0usize;
177    loop {
178        record_visit(trie_data, &mut steps)?;
179        if let Some(payload) = read_row_node_payload(trie_data, node_offset, None)? {
180            return Ok(payload);
181        }
182        let node = parse_bti_node_for_traversal(trie_data, node_offset)?;
183        match ordered_children(&node).first() {
184            Some(&(_, child)) => node_offset = child,
185            None => {
186                return Err(Error::Parse(format!(
187                    "Rows.db ceiling: min leaf at offset {node_offset} carries no row-index \
188                     payload (corrupt trie)"
189                )))
190            }
191        }
192    }
193}

195#[cfg(not(feature = "tombstones"))]
196/// Count one visited node and enforce the acyclic total-work bound: an acyclic
197/// trie has each node on a downward path occupy `>= 1` byte, so a walk that visits
198/// more nodes than the trie has bytes is proof of a cycle/corruption. Mirrors the
199/// DFS bound in [`super::traversal`].
200#[inline]
201fn record_visit(trie_data: &[u8], steps: &mut usize) -> BtiResult<()> {
202    crate::storage::sstable::read_work_counters::record_bti_node_visited();
203    *steps = steps.saturating_add(1);
204    if *steps > trie_data.len().saturating_add(1) {
205        return Err(Error::Parse(format!(
206            "Rows.db walk exceeded total work bound ({} nodes > trie size {}; corrupt or \
207             cyclic trie)",
208            *steps,
209            trie_data.len()
210        )));
211    }
212    Ok(())
213}

215#[cfg(not(feature = "tombstones"))]
216/// Compute `separatorFloor(target_key)`: the row-index block whose half-open
217/// clustering interval `[s_i, s_{i+1})` contains `target_key` — i.e. the block for
218/// the largest separator `<= target_key`.
219///
220/// Returns `Ok(None)` when `target_key` sorts strictly below the first stored
221/// separator: there is no stored block for it, only the trie-implicit first block
222/// at the partition body start (the caller decodes from rel 0 in that case; see
223/// `select_row_index_blocks_for_range`'s "implicit first block" note and
224/// `reader/data_access/bti.rs`).
225///
226/// This is byte-for-byte equivalent to taking the block with the maximum
227/// separator `<= target_key` from the full ascending `(separator, block)` list,
228/// but visits only O(len(target_key)) nodes.  Mirrors
229/// `RowIndexReader#separatorFloor` (`prefixAndNeighbours` + `goMax(lesserBranch)`).
230pub(crate) fn rows_floor_block(
231    trie_data: &[u8],
232    root_offset: usize,
233    target_key: &[u8],
234) -> BtiResult<Option<BtiRowIndexEntry>> {
235    // `payload` holds the closest prefix separator payload; `lesser_branch` holds
236    // the closest strictly-lesser subtree root (mirrors `prefixAndNeighbours`).
237    let mut payload: Option<BtiRowIndexEntry> = None;
238    let mut lesser_branch: Option<usize> = None;
239    let mut node_offset = root_offset;
240    let mut key_idx = 0usize;
241    let mut steps = 0usize;
242
243    loop {
244        record_visit(trie_data, &mut steps)?;
245        let node = parse_bti_node_for_traversal(trie_data, node_offset)?;
246        let children = ordered_children(&node);
247
248        let b = target_key.get(key_idx).copied();
249        key_idx += 1;
250        let ChildSearch { insertion, exact } = search_children(&children, b);
251
252        if insertion == 0 {
253            // `search` in {0 (exact-first), -1 (below all)}: the current node's
254            // separator is still a valid prefix floor candidate — keep it.
255            if let Some(p) = read_row_node_payload(trie_data, node_offset, None)? {
256                payload = Some(p);
257            }
258        } else {
259            // A strictly-lesser child exists: it (its max) is a closer floor than
260            // the current prefix, so record it and drop the prefix candidate.
261            lesser_branch = Some(children[insertion - 1].1);
262            payload = None;
263        }
264
265        match exact {
266            // Exact transition: descend and keep matching the key.
267            Some(i) => node_offset = children[i].1,
268            // No exact transition: the walk ends here.
269            None => break,
270        }
271    }
272
273    if let Some(p) = payload {
274        return Ok(Some(p));
275    }
276    // No prefix separator matched: the floor is the maximum of the closest lesser
277    // branch (or `None` — `target_key` sorts below the first separator).
278    match lesser_branch {
279        Some(lb) => Ok(Some(go_max_payload(trie_data, lb)?)),
280        None => Ok(None),
281    }
282}

284#[cfg(not(feature = "tombstones"))]
285/// Compute the block with the smallest separator **strictly greater** than
286/// `target_key` (`target_key`'s successor block).
287///
288/// Returns `Ok(None)` when `target_key` sorts at or after the last separator: no
289/// stored block follows it, so the caller's row-body window runs to the partition
290/// end. Used to bound the END of a clustering slice's decode window: the exclusive
291/// window end is the successor of `floor(end)`, which equals the smallest separator
292/// strictly greater than `end`.
293///
294/// Byte-for-byte equivalent to the minimum separator `> target_key` in the full
295/// ascending list, but visits only O(len(target_key)) nodes.  Mirrors
296/// `Walker#followWithGreater` + `goMin(greaterBranch)`.
297pub(crate) fn rows_strict_ceiling_block(
298    trie_data: &[u8],
299    root_offset: usize,
300    target_key: &[u8],
301) -> BtiResult<Option<BtiRowIndexEntry>> {
302    // `greater_branch` holds the closest subtree whose keys are all > target_key.
303    let mut greater_branch: Option<usize> = None;
304    let mut node_offset = root_offset;
305    let mut key_idx = 0usize;
306    let mut steps = 0usize;
307
308    loop {
309        record_visit(trie_data, &mut steps)?;
310        let node = parse_bti_node_for_traversal(trie_data, node_offset)?;
311        let children = ordered_children(&node);
312
313        let b = target_key.get(key_idx).copied();
314        key_idx += 1;
315        let ChildSearch { insertion, exact } = search_children(&children, b);
316
317        // The child immediately greater than the search position: exact match at
318        // `i` -> child `i+1`; no exact match -> child at `insertion`. A deeper
319        // greater branch shares a longer prefix with the key, so its minimum is a
320        // closer successor — overwrite when the current node has one.
321        let greater_idx = match exact {
322            Some(i) => i + 1,
323            None => insertion,
324        };
325        if greater_idx < children.len() {
326            greater_branch = Some(children[greater_idx].1);
327        }
328
329        match exact {
330            Some(i) => node_offset = children[i].1,
331            None => break,
332        }
333    }
334
335    match greater_branch {
336        Some(gb) => Ok(Some(go_min_payload(trie_data, gb)?)),
337        None => Ok(None),
338    }
339}

347    /// A `Rows.db` PayloadOnly leaf with no open marker (payloadBits=1): a
348    /// single-byte SizedInts Data.db position (value 0..=127).
349    fn row_leaf(pos: u8) -> Vec<u8> {
350        assert!(pos <= 127, "use a 1-byte SizedInts position");
351        vec![0x01, pos] // ordinal=0, payloadBits=1 (no FLAG_OPEN_MARKER)
352    }
353
354    /// Build a Rows.db-style trie whose separators are the single bytes `seps`
355    /// (ascending), each mapping to a distinct 1-byte payload position. Returns
356    /// `(trie_bytes, root_offset)`. Uses a Sparse8 root over single-byte leaves,
357    /// so every separator is exactly one byte — the canonical single-column
358    /// clustering shape.
359    fn make_trie(seps: &[u8]) -> (Vec<u8>, usize) {
360        let mut trie = Vec::new();
361        let mut leaf_offsets = Vec::new();
362        for (i, _) in seps.iter().enumerate() {
363            leaf_offsets.push(trie.len() as u64);
364            // distinct positions 1,2,3,... so payloads are identifiable.
365            trie.extend_from_slice(&row_leaf((i + 1) as u8));
366        }
367        let root = trie.len() as u64;
368        trie.push(0x50); // Sparse8
369        trie.push(seps.len() as u8); // count
370        for &s in seps {
371            trie.push(s);
372        }
373        for &off in &leaf_offsets {
374            trie.push((root - off) as u8);
375        }
376        (trie, root as usize)
377    }
378
379    /// A `Rows.db` PayloadOnly leaf carrying an OPEN range-tombstone marker
380    /// (`FLAG_OPEN_MARKER` set, payloadBits = `1 | 8 = 9`): a 1-byte SizedInts
381    /// Data.db position followed by a NON-live MODERN DA `DeletionTime`
382    /// (`markedForDeleteAt` i64 BE, `localDeletionTime` u32 BE — see
383    /// `decode_da_deletion_time`). `markedForDeleteAt`'s leading byte is `0x00`
384    /// (not the `0x80` LIVE sentinel), so this always decodes to
385    /// `open_marker.is_some()`.
386    fn row_leaf_with_open_marker(pos: u8) -> Vec<u8> {
387        assert!(pos <= 127, "use a 1-byte SizedInts position");
388        let mut v = vec![0x09u8, pos]; // ordinal=0, payloadBits=9 (FLAG_OPEN_MARKER set)
389        v.extend_from_slice(&1_000i64.to_be_bytes()); // markedForDeleteAt (non-live)
390        v.extend_from_slice(&500u32.to_be_bytes()); // localDeletionTime
391        v
392    }

394    /// Like [`make_trie`] but the leaf at `marker_index` carries an OPEN
395    /// range-tombstone marker ([`row_leaf_with_open_marker`]); every other leaf
396    /// is marker-free ([`row_leaf`]). Used to pin the floor-block-ONLY
397    /// narrowing intent (rust-reviewer follow-up, issue #1647): the caller's
398    /// open-marker correctness guard (`data_access::bti::bti_clustering_row_window`)
399    /// inspects ONLY the returned floor block's `open_marker`, never every block
400    /// in the partition.
401    fn make_trie_with_marker(seps: &[u8], marker_index: usize) -> (Vec<u8>, usize) {
402        let mut trie = Vec::new();
403        let mut leaf_offsets = Vec::new();
404        for (i, _) in seps.iter().enumerate() {
405            leaf_offsets.push(trie.len() as u64);
406            if i == marker_index {
407                trie.extend_from_slice(&row_leaf_with_open_marker((i + 1) as u8));
408            } else {
409                trie.extend_from_slice(&row_leaf((i + 1) as u8));
410            }
411        }
412        let root = trie.len() as u64;
413        trie.push(0x50); // Sparse8
414        trie.push(seps.len() as u8); // count
415        for &s in seps {
416            trie.push(s);
417        }
418        for &off in &leaf_offsets {
419            trie.push((root - off) as u8);
420        }
421        (trie, root as usize)
422    }

424    /// Build a Rows.db-style trie with MULTI-BYTE (composite) separators
425    /// sharing a common first byte, spanning TWO trie levels: `sep0=[0x10,0x05]`
426    /// -> pos 1, `sep1=[0x10,0x0A]` -> pos 2 (both under first byte `0x10`, via
427    /// an intermediate Sparse8 node), `sep2=[0x20,0x03]` -> pos 3 (under first
428    /// byte `0x20`, via an intermediate Single8 node). Exercises the prefix-floor
429    /// walk's multi-byte descent plus a genuine 2-hop `go_max`/`go_min` traversal
430    /// through an intermediate (non-leaf) node — the single-byte [`make_trie`]
431    /// above only ever has ONE level, so `go_max`/`go_min` land on a leaf in one
432    /// hop. Returns `(trie_bytes, root_offset)`.
433    fn make_composite_trie() -> (Vec<u8>, usize) {
434        let mut trie = Vec::new();
435        let leaf1 = trie.len() as u64; // sep0=[0x10,0x05] -> pos 1
436        trie.extend_from_slice(&row_leaf(1));
437        let leaf2 = trie.len() as u64; // sep1=[0x10,0x0A] -> pos 2
438        trie.extend_from_slice(&row_leaf(2));
439        let leaf3 = trie.len() as u64; // sep2=[0x20,0x03] -> pos 3
440        trie.extend_from_slice(&row_leaf(3));
441
442        // Intermediate A: Sparse8 (no payload) routing second byte {0x05, 0x0A}
443        // under first byte 0x10.
444        let node_a = trie.len() as u64;
445        trie.push(0x50); // Sparse8, payload_flags=0
446        trie.push(0x02); // count=2
447        trie.push(0x05);
448        trie.push(0x0A);
449        trie.push((node_a - leaf1) as u8);
450        trie.push((node_a - leaf2) as u8);
451
452        // Intermediate B: Single8 (no payload) routing second byte 0x03 under
453        // first byte 0x20.
454        let node_b = trie.len() as u64;
455        trie.push(0x20); // Single8, payload_flags=0
456        trie.push(0x03); // transition byte
457        trie.push((node_b - leaf3) as u8);
458
459        // Root: Sparse8 (no payload) routing first byte {0x10, 0x20}.
460        let root = trie.len() as u64;
461        trie.push(0x50);
462        trie.push(0x02);
463        trie.push(0x10);
464        trie.push(0x20);
465        trie.push((root - node_a) as u8);
466        trie.push((root - node_b) as u8);
467
468        (trie, root as usize)
469    }

471    /// The floor walk equals the maximum stored separator `<= key` (or `None`
472    /// below the first separator), for EVERY boundary class.
473    #[test]
474    fn floor_matches_enumerate_and_filter() {
475        let seps = [0x10u8, 0x20, 0x30, 0x40];
476        let (trie, root) = make_trie(&seps);
477        let all = iterate_rows_in_bti_trie(&trie, root).unwrap();
478        assert_eq!(all.len(), 4);
479
480        // Oracle: the block for the largest separator <= key, else None.
481        let oracle = |key: &[u8]| -> Option<u64> {
482            all.iter()
483                .filter(|(sep, _)| sep.as_slice() <= key)
484                .next_back()
485                .map(|(_, e)| e.data_offset)
486        };
487
488        // exact matches, between-block keys, before-first, after-last, multi-byte.
489        let probes: &[&[u8]] = &[
490            &[0x00],
491            &[0x0F],
492            &[0x10],
493            &[0x18],
494            &[0x20],
495            &[0x2F],
496            &[0x30],
497            &[0x40],
498            &[0x41],
499            &[0xFF],

501            &[0x30, 0x99],
502            &[],
503        ];
504        for key in probes {
505            let got = rows_floor_block(&trie, root, key)

534            &[0x41],
535            &[0xFF],
536            &[0x10, 0x00],
537            &[],
538        ];
539        for key in probes {
540            let got = rows_strict_ceiling_block(&trie, root, key)
541                .unwrap()
542                .map(|e| e.data_offset);
543            assert_eq!(got, oracle(key), "ceiling mismatch for key {key:02x?}");
544        }

547    /// End-to-end `[body_start, body_end)` window equivalence: for every `(start,
548    /// end)` the new floor+ceiling window equals the pre-L1 enumerate-then-select
549    /// window, byte-for-byte. This is the exact oracle `bti_clustering_row_window`
550    /// relies on — modelled here with `has_static == false` (the synthetic trie has
551    /// no static row), so `body_start` narrows whenever the start is not below the
552    /// first separator.
553    #[test]
554    fn window_bounds_match_select_row_index_blocks() {
555        const END_INF: usize = usize::MAX;
556        let seps = [0x10u8, 0x20, 0x30, 0x40];
557        let (trie, root) = make_trie(&seps);
558        let all = iterate_rows_in_bti_trie(&trie, root).unwrap();
559        let first_sep = all.first().map(|(sep, _)| sep.clone()).unwrap();
560
561        let probes: &[u8] = &[0x00, 0x0F, 0x10, 0x18, 0x20, 0x30, 0x40, 0x41, 0xFF];
562        for &s in probes {
563            for &e in probes {
564                if s > e {
565                    continue;
566                }
567                let start = [s];
568                let end = [e];
569
570                // Pre-L1 window (transcribed from the old bti_clustering_row_window
571                // block-selection math, has_static == false).
572                let blocks = select_row_index_blocks_for_range(&all, &start, &end);
573                let includes_implicit = start.as_slice() < first_sep.as_slice();
574                let pre: Option<(usize, usize)> = if blocks.is_empty() {
575                    // blocks empty <=> range below the first separator (implicit).
576                    assert!(includes_implicit, "empty selection must be implicit-first");
577                    let first_off = all.first().map(|(_, b)| b.data_offset as usize).unwrap();
578                    Some((0, first_off))
579                } else {
580                    let body_start = if includes_implicit {
581                        0
582                    } else {
583                        blocks.iter().map(|b| b.data_offset as usize).min().unwrap()
584                    };
585                    let last = blocks.iter().map(|b| b.data_offset).max().unwrap();
586                    let body_end = all
587                        .iter()
588                        .map(|(_, b)| b.data_offset)
589                        .filter(|&o| o > last)
590                        .min()
591                        .map(|o| o as usize)
592                        .unwrap_or(END_INF);
593                    Some((body_start, body_end))
594                };
595
596                // New window (floor + strict-ceiling walks, has_static == false).
597                let floor = rows_floor_block(&trie, root, &start).unwrap();
598                let ceil = rows_strict_ceiling_block(&trie, root, &end).unwrap();
599                let body_start = match &floor {
600                    Some(b) => b.data_offset as usize, // not implicit => floor narrows
601                    None => 0,                         // implicit-first => partition body start
602                };
603                let body_end = ceil.map(|b| b.data_offset as usize).unwrap_or(END_INF);
604                let new = Some((body_start, body_end));
605
606                assert_eq!(new, pre, "window mismatch for start {s:02x} end {e:02x}");
607            }
608        }
609    }

628    /// Pin the floor-block-ONLY narrowing intent (rust-reviewer follow-up, issue
629    /// #1647): the clustering-window correctness guard in
630    /// `data_access::bti::bti_clustering_row_window` inspects ONLY the returned
631    /// floor block's `open_marker` — it never scans every block in the
632    /// partition. Confirmed against Cassandra
633    /// `SSTableIterator.ForwardIndexedReader.setForSlice` (which reads
634    /// `separatorFloor(...).openDeletion` exclusively) and
635    /// `BtiFormatPartitionWriter` (which stores a CUMULATIVE `startOpenMarker`
636    /// per block, so the floor block's own marker already reflects any deletion
637    /// open at the window start — a marker recorded on an EARLIER block is
638    /// carried forward onto every later block it still covers, so there is
639    /// nothing left to observe on a block that is not the floor).
640    #[test]
641    fn floor_block_open_marker_is_the_only_signal_the_guard_needs() {
642        let seps = [0x10u8, 0x20, 0x30];
643        // The MIDDLE block (sep=0x20) carries the open marker; sep=0x10/0x30
644        // do not.
645        let (trie, root) = make_trie_with_marker(&seps, 1);
646
647        // (a) A query whose floor IS the marker-carrying block observes
648        // `open_marker.is_some()` — exactly the signal the guard's
649        // fallback-to-`Ok(None)` branch keys off.
650        let floor_on_marker_block = rows_floor_block(&trie, root, &[0x25]).unwrap().unwrap();
651        assert_eq!(
652            floor_on_marker_block.data_offset, 2,
653            "floor(0x25) must be the sep=0x20 block"
654        );
655        assert!(
656            floor_on_marker_block.open_marker.is_some(),
657            "the floor-block-only guard needs open_marker to survive on the \
658             block it actually returns"
659        );
660
661        // (b) A query whose floor is a DIFFERENT (non-floor-for-this-query)
662        // block — even though the marker-carrying block still exists later in
663        // the same partition — observes `open_marker.is_none()`: the marker on
664        // a block that ISN'T the floor never leaks into the returned entry, so
665        // the guard does NOT fall back and the window still narrows.
666        let floor_before_marker_block = rows_floor_block(&trie, root, &[0x15]).unwrap().unwrap();
667        assert_eq!(
668            floor_before_marker_block.data_offset, 1,
669            "floor(0x15) must be the sep=0x10 block"
670        );
671        assert!(
672            floor_before_marker_block.open_marker.is_none(),
673            "a marker on a NON-floor block must not leak into the returned \
674             floor entry (the guard must not spuriously fall back)"
675        );
676    }

678    /// Multi-byte (composite) separator coverage (rust-reviewer follow-up,
679    /// issue #1647): [`make_trie`] above only ever emits single-byte
680    /// separators via a one-level Sparse8 root, so `go_max`/`go_min` always
681    /// land on a leaf in a single hop. This pins the SAME floor/ceiling
682    /// equivalence oracle over [`make_composite_trie`]'s two-level, multi-byte
683    /// separators, which forces a genuine multi-byte prefix-follow PLUS a
684    /// real 2-hop `go_max_payload`/`go_min_payload` descent through an
685    /// intermediate (non-leaf) node for the `[0x18]` probe.
686    #[test]
687    fn floor_and_ceiling_multi_byte_separators() {
688        let (trie, root) = make_composite_trie();
689        let all = iterate_rows_in_bti_trie(&trie, root).unwrap();
690        assert_eq!(
691            all.iter().map(|(k, _)| k.clone()).collect::<Vec<_>>(),
692            vec![vec![0x10, 0x05], vec![0x10, 0x0A], vec![0x20, 0x03]],
693            "composite trie must yield the 3 multi-byte separators in byte order"
694        );
695
696        let floor_oracle = |key: &[u8]| -> Option<u64> {
697            all.iter()
698                .filter(|(sep, _)| sep.as_slice() <= key)
699                .next_back()
700                .map(|(_, e)| e.data_offset)
701        };
702        let ceil_oracle = |key: &[u8]| -> Option<u64> {
703            all.iter()
704                .find(|(sep, _)| sep.as_slice() > key)
705                .map(|(_, e)| e.data_offset)
706        };
707
708        // Exact matches on both bytes, between-separator probes (including one
709        // sharing only the first byte, `0x18`, which forces a multi-hop
710        // go_max/go_min descent through the intermediate node), before-first,
711        // after-last, and deeper-than-any-leaf-key probes.
712        let probes: &[&[u8]] = &[
713            &[0x00],
714            &[0x10],
715            &[0x10, 0x00],
716            &[0x10, 0x05],
717            &[0x10, 0x07],
718            &[0x10, 0x0A],
719            &[0x10, 0xFF],
720            &[0x18],
721            &[0x20],
722            &[0x20, 0x00],
723            &[0x20, 0x03],
724            &[0x20, 0x03, 0x00],
725            &[0x20, 0xFF],
726            &[0xFF],
727            &[],
728        ];
729        for key in probes {
730            let floor = rows_floor_block(&trie, root, key)
731                .unwrap()
732                .map(|e| e.data_offset);
733            assert_eq!(
734                floor,
735                floor_oracle(key),
736                "floor mismatch for composite key {key:02x?}"
737            );
738            let ceil = rows_strict_ceiling_block(&trie, root, key)
739                .unwrap()
740                .map(|e| e.data_offset);
741            assert_eq!(
742                ceil,
743                ceil_oracle(key),
744                "ceiling mismatch for composite key {key:02x?}"
745            );
746        }
747    }
