1//! `Rows.db` in-trie payload decoding (`RowIndexReader` / `TrieIndexEntry`).
2//!
3//! IMPORTANT: `Rows.db` in-trie payloads are NOT the `Partitions.db` payload
4//! format (which is a hash byte + SizedInts *signed* position).  A `Rows.db`
5//! trie leaf carries a `RowIndexReader.IndexInfo` whose byte layout is defined
6//! authoritatively by `RowIndexReader.readPayload`
7//! (cassandra-5.0.0 `RowIndexReader.java:111-125`):
8//!
9//!   static IndexInfo readPayload(ByteBuffer buf, int ppos, int bits, Version v) {
10//!       if (bits == 0) return null;
11//!       int bytes = bits & ~FLAG_OPEN_MARKER;            // FLAG_OPEN_MARKER = 8
12//!       long offset = SizedInts.read(buf, ppos, bytes);  // SizedInts, NOT a vint
13//!       ppos += bytes;
14//!       DeletionTime del = (bits & FLAG_OPEN_MARKER) != 0
15//!                          ? DeletionTime.deserialize(buf, ppos) : null;
16//!       return new IndexInfo(offset, del);
17//!   }
18//!
19//! So the low nibble of the node header byte (`payloadBits`) splits as:
20//!   - low 3 bits  → the number of `SizedInts` bytes encoding the block offset
21//!   - bit 0x8     → FLAG_OPEN_MARKER: an open-deletion `DeletionTime` follows
22//!
23//! The `offset` field is the block's offset **relative to the partition start**
24//! in `Data.db`, so absolute Data.db position = `entry.data_position + offset`.
25//!
26//! Reference: docs/sstables-definitive-guide chapter 17 (Rows.db footer);
27//!            cassandra-5.0.0 `RowIndexReader.java`, `RowIndexWriter.java`,
28//!            `TrieIndexEntry.java`, `SizedInts.java`.
29

39/// The `FLAG_OPEN_MARKER` bit in a `Rows.db` trie node's `payloadBits`
40/// (low nibble of the header byte).  When set, an open-deletion `DeletionTime`
41/// follows the `SizedInts` block offset.  Mirrors
42/// `RowIndexReader.FLAG_OPEN_MARKER`.
43pub const FLAG_OPEN_MARKER: u8 = 0x8;

45/// A decoded `Rows.db` in-trie row-index block entry (`RowIndexReader.IndexInfo`).
46///
47/// The headline field is [`data_offset`](Self::data_offset): the block's offset
48/// **relative to the partition start** in `Data.db`.  To obtain the absolute
49/// `Data.db` byte position, add the partition's data position (see
50/// [`BtiRowIndexHeader::data_position`]).
51#[derive(Debug, Clone, PartialEq, Eq)]
52pub struct BtiRowIndexEntry {
53    /// Block offset **relative to the partition start**, decoded via
54    /// `SizedInts.read(buf, ppos, payloadBits & ~FLAG_OPEN_MARKER)`.
55    pub data_offset: u64,
56    /// Open-deletion time `(local_deletion_time, marked_for_delete_at)`,
57    /// present only when the `FLAG_OPEN_MARKER` payload bit is set.
58    pub open_marker: Option<(i32, i64)>,
59}

61/// Read an unsigned VInt (Cassandra count-leading-ones encoding, **not** ZigZag)
62/// from `data`, returning `(value, bytes_consumed)`.
63///
64/// This is the encoding Cassandra uses for Data.db positions in the row index
65/// (`DataOutputPlus.writeUnsignedVInt`).  The number of extra bytes equals the
66/// number of leading 1-bits in the first byte; the value is big-endian across
67/// the remaining bits.
68fn read_unsigned_vint_from_slice(data: &[u8]) -> BtiResult<(u64, usize)> {
69    if data.is_empty() {
70        return Err(Error::Parse(
71            "Rows.db payload: unexpected end of data reading unsigned vint".to_string(),
72        ));
73    }
74    let first = data[0];
75    let extra_bytes = first.leading_ones() as usize;
76    if extra_bytes > 8 {
77        return Err(Error::Parse(format!(
78            "Rows.db payload: invalid unsigned vint first byte 0x{first:02x}"
79        )));
80    }
81    let total = extra_bytes + 1;
82    if data.len() < total {
83        return Err(Error::Parse(format!(
84            "Rows.db payload: unsigned vint needs {total} bytes, have {}",
85            data.len()
86        )));
87    }
88
89    // Data bits in the first byte: 8 - extra_bytes - 1 (the separator 0 bit),
90    // except when extra_bytes == 8 (first byte is all ones, no data bits).
91    let mut value: u64 = if extra_bytes >= 8 {
92        0
93    } else {
94        let data_bits = 8 - extra_bytes - 1;
95        let mask = if data_bits == 0 {
96            0
97        } else {
98            (1u16 << data_bits) - 1
99        };
100        (first as u16 & mask) as u64
101    };
102    for &b in &data[1..total] {
103        value = (value << 8) | (b as u64);
104    }
105    Ok((value, total))
106}

108/// Read a signed VInt (Cassandra zig-zag, `DataInputPlus.readVInt`) from `data`,
109/// returning `(value, bytes_consumed)`.
110fn read_signed_vint_from_slice(data: &[u8]) -> BtiResult<(i64, usize)> {
111    let (u, n) = read_unsigned_vint_from_slice(data)?;
112    // ZigZag decode: (u >>> 1) ^ -(u & 1)
113    let value = ((u >> 1) as i64) ^ -((u & 1) as i64);
114    Ok((value, n))
115}

132/// The modern (DA/BTI) `DeletionTime` "live" sentinel byte.
133///
134/// In the `da`-family on-disk serializer, a `DeletionTime` written by the BTI
135/// row-index / trie-index path is encoded as a single `0x80` byte when it is
136/// `DeletionTime.LIVE` (no deletion), and otherwise as the full value (see
137/// [`decode_da_deletion_time`]).
138const DA_DELETION_TIME_LIVE_SENTINEL: u8 = 0x80;
139
140/// Width of a non-live modern (DA) `DeletionTime` body: `i64 markedForDeleteAt`
141/// followed by `u32 localDeletionTime`.
142const DA_DELETION_TIME_BODY_LEN: usize = 12;
143
144/// Decode a modern (DA/BTI) `DeletionTime` at `data[start..]`, returning
145/// `(deletion, bytes_consumed)` where `deletion` is `None` for the LIVE
146/// sentinel (issue #832 Finding 2).
147///
148/// Layout (mirrors `org.apache.cassandra.db.DeletionTime.Serializer` in the
149/// `da`/trie-index format, cassandra-5.0.0):
150///
151///   - a single `0x80` byte → `DeletionTime.LIVE` (no deletion); consumes 1 byte.
152///   - otherwise the body is `[markedForDeleteAt : i64 BE][localDeletionTime :
153///     u32 BE]` — `markedForDeleteAt` FIRST, then `localDeletionTime`; consumes
154///     12 bytes.  This differs from the LEGACY layout in BOTH field order and
155///     the width/signedness of `localDeletionTime` (modern: `u32`).
156///
157/// Returns the deletion as `(local_deletion_time, marked_for_delete_at)` to
158/// match [`BtiRowIndexEntry::open_marker`]'s existing tuple ordering, even
159/// though the modern wire order is the reverse.
160///
161/// # Errors
162/// Returns a parse error if `start` is out of bounds or a non-live value is
163/// truncated.
164fn decode_da_deletion_time(data: &[u8], start: usize) -> BtiResult<(Option<(i32, i64)>, usize)> {
165    if start >= data.len() {
166        return Err(Error::Parse(format!(
167            "DA DeletionTime: start {start} beyond buffer size {}",
168            data.len()
169        )));
170    }
171    if data[start] == DA_DELETION_TIME_LIVE_SENTINEL {
172        return Ok((None, 1));
173    }
174    if start + DA_DELETION_TIME_BODY_LEN > data.len() {
175        return Err(Error::Parse(format!(
176            "DA DeletionTime: non-live value needs {DA_DELETION_TIME_BODY_LEN} bytes, have {}",
177            data.len().saturating_sub(start)
178        )));
179    }
180    let b = &data[start..start + DA_DELETION_TIME_BODY_LEN];
181    // markedForDeleteAt FIRST (i64), then localDeletionTime (u32).
182    let marked_for_delete_at = i64::from_be_bytes([b[0], b[1], b[2], b[3], b[4], b[5], b[6], b[7]]);
183    let local_deletion_time = u32::from_be_bytes([b[8], b[9], b[10], b[11]]) as i32;
184    Ok((
185        Some((local_deletion_time, marked_for_delete_at)),
186        DA_DELETION_TIME_BODY_LEN,
187    ))
188}

190/// Decode a `Rows.db` in-trie payload (`RowIndexReader.IndexInfo`) at
191/// `payload_start` inside `trie_data`, given the node's `payload_bits` (low
192/// nibble of the header byte).
193///
194/// Layout (mirrors `RowIndexReader.readPayload`, cassandra-5.0.0
195/// `RowIndexReader.java:111-125`):
196///   - `bytes = payload_bits & !FLAG_OPEN_MARKER` → block offset is a
197///     `SizedInts` value of `bytes` bytes (the offset is relative to the
198///     partition's data position).
199///   - if `payload_bits & FLAG_OPEN_MARKER`, an open-deletion `DeletionTime`
200///     follows in the MODERN DA form ([`decode_da_deletion_time`]).
201///
202/// A `payload_bits` of `0` is not a valid leaf payload here (the caller filters
203/// such nodes out) and yields an error.
204pub fn decode_bti_row_payload(
205    trie_data: &[u8],
206    payload_start: usize,
207    payload_bits: u8,
208) -> BtiResult<BtiRowIndexEntry> {
209    if payload_start > trie_data.len() {
210        return Err(Error::Parse(format!(
211            "Rows.db payload start {payload_start} beyond trie size {}",
212            trie_data.len()
213        )));
214    }
215
216    // Low 3 bits = number of SizedInts bytes; bit 0x8 = open-marker flag.
217    let offset_bytes = (payload_bits & !FLAG_OPEN_MARKER) as usize;
218    if offset_bytes == 0 || offset_bytes > 7 {
219        // RowIndexWriter asserts `bytes < 8` ("rows larger than 32 PiB"); a
220        // 0-byte offset would mean an empty payload, which the trie does not
221        // emit for a real row-index block.
222        return Err(Error::Parse(format!(
223            "Rows.db payload: invalid SizedInts byte count {offset_bytes} \
224             (payload_bits=0x{payload_bits:02x}); expected 1..=7"
225        )));
226    }
227    if payload_start + offset_bytes > trie_data.len() {
228        return Err(Error::Parse(format!(
229            "Rows.db payload: SizedInts offset needs {offset_bytes} bytes, have {}",
230            trie_data.len().saturating_sub(payload_start)
231        )));
232    }
233
234    // SizedInts is a signed sign-extended big-endian read (SizedInts.read).
235    // Block offsets are non-negative in practice, but we decode faithfully.
236    let raw = sized_ints_read_from_slice(&trie_data[payload_start..payload_start + offset_bytes])?;
237    let data_offset = raw as u64;
238
239    let open_marker = if payload_bits & FLAG_OPEN_MARKER != 0 {
240        // DA/BTI modern DeletionTime (issue #832 Finding 2).
241        let dt_start = payload_start + offset_bytes;
242        let (deletion, _consumed) = decode_da_deletion_time(trie_data, dt_start)?;
243        deletion
244    } else {
245        None
246    };
247
248    Ok(BtiRowIndexEntry {
249        data_offset,
250        open_marker,
251    })
252}

268/// Enumerate every row-index entry in a `Rows.db` row-index trie **rooted at an
269/// explicit `root_offset`**, in byte-comparable order
270/// (`(reconstructed_clustering_key, BtiRowIndexEntry)`).
271///
272/// ## Why the root must be supplied by the caller
273///
274/// A real Cassandra 5.0 `Rows.db` is NOT a single whole-file trie: it holds
275/// **many independent per-partition row-index tries** concatenated together.
276/// There is one row-index trie per (wide) partition, and the root of a given
277/// partition's trie is the `RowsOffset` returned from the corresponding
278/// `Partitions.db` lookup ([`BtiPartitionLocation::RowsOffset`]) — it is NOT the
279/// 8-byte file footer, which spans the whole file and would misparse any
280/// multi-partition `Rows.db`.
281///
282/// This is therefore the correct general entry point: pass the full `Rows.db`
283/// bytes as `trie_data` and the partition's `RowsOffset` as `root_offset`.
284///
285/// An out-of-bounds `root_offset` (e.g. on empty `trie_data`) yields a clean
286/// parse error rather than a panic.
287pub fn iterate_rows_in_bti_trie(
288    trie_data: &[u8],
289    root_offset: usize,
290) -> BtiResult<Vec<(Vec<u8>, BtiRowIndexEntry)>> {
291    dfs_collect_row_entries(trie_data, root_offset)
292}

294// ─────────────────────────────────────────────────────────────────────────────
295// Per-partition Rows.db entry resolution — TrieIndexEntry (issue #832 Finding A)
296// ─────────────────────────────────────────────────────────────────────────────
297//
298// The positive `position` stored in a `Partitions.db` leaf payload
299// (`BtiPartitionLocation::RowsOffset`) does NOT point at a row-index trie root.
300// It points at this partition's **row-index entry** in `Rows.db`, which must be
301// deserialized to recover the actual trie root (plus the partition's Data.db
302// position, block count and partition-level deletion).
303//
304// On-disk layout at `RowsOffset`:
305//   [u16 key_length][partition key bytes]      ← short-length-prefixed key
306//   [data file position : unsigned vint]       ← partition start in Data.db
307//   [trie_root - base    : SIGNED vint]         ← base = RowsOffset + key_length
308//   [row index block count : unsigned vint32]
309//   [partition DeletionTime]                    ← delta/compact form; best-effort
310//
311// `TrieIndexEntry.deserialize` computes `indexTrieRoot = readVInt() + base`.
312// ─────────────────────────────────────────────────────────────────────────────
313
314/// A single `Rows.db` row-index entry paired with its reconstructed
315/// byte-comparable clustering separator key, as yielded by the in-order DFS.
316pub type BtiRowIndexEntryWithKey = (Vec<u8>, BtiRowIndexEntry);

318/// A deserialized per-partition `Rows.db` row-index entry (Cassandra
319/// `TrieIndexEntry`).  Produced by [`resolve_rows_db_entry`] from the
320/// `RowsOffset` returned by a `Partitions.db` lookup.
321#[derive(Debug, Clone, PartialEq, Eq)]
322pub struct BtiRowIndexHeader {
323    /// Absolute byte position of the partition's start in `Data.db`.  Block
324    /// offsets in [`BtiRowIndexEntry::data_offset`] are relative to this.
325    pub data_position: u64,
326    /// Byte offset, within the `Rows.db` file, of this partition's row-index
327    /// trie root node.  Feed this to [`iterate_rows_in_bti_trie`] for a full
328    /// in-order traversal. For range selection, call [`resolve_rows_db_entry`]
329    /// first (with the `RowsOffset`) to get this header, then pass
330    /// `iterate_rows_in_bti_trie` output to [`select_row_index_blocks_for_range`].
331    pub trie_root: usize,
332    /// Number of row-index blocks indexed by this partition's trie.
333    pub block_count: u32,
334    /// Partition-level deletion `(local_deletion_time, marked_for_delete_at)`,
335    /// decoded via the MODERN DA `DeletionTime` form
336    /// ([`decode_da_deletion_time`], issue #832 Finding 2); `None` for the `0x80`
337    /// LIVE sentinel or when too few trailing bytes remain.
338    pub partition_deletion: Option<(i32, i64)>,
339}

341/// Resolve a partition's row-index entry in `Rows.db`, given the `RowsOffset`
342/// from a `Partitions.db` lookup ([`BtiPartitionLocation::RowsOffset`]).
343///
344/// This is the fix for issue #832 Finding A: `RowsOffset` is the offset of the
345/// per-partition `TrieIndexEntry`, NOT a trie root.  This deserializes that
346/// entry — recovering the partition's Data.db position, the actual row-index
347/// trie root, the block count and the partition deletion — so traversal can be
348/// rooted correctly.
349///
350/// `rows_db` is the full `Rows.db` file contents; `rows_offset` is the
351/// `RowsOffset` value.  All reads are bounds-checked.
352///
353/// # Errors
354/// Returns a parse error if `rows_offset` is out of bounds, the key length is
355/// implausible, the vint fields are truncated, or the recovered trie root falls
356/// outside `rows_db`.
357pub fn resolve_rows_db_entry(rows_db: &[u8], rows_offset: usize) -> BtiResult<BtiRowIndexHeader> {
358    // Issue #1647 (L1): count every `TrieIndexEntry.deserialize` so the clustering
359    // read path can prove it resolves the per-partition entry EXACTLY once.
360    crate::storage::sstable::read_work_counters::record_rows_db_entry_resolve();
361    if rows_offset + 2 > rows_db.len() {
362        return Err(Error::Parse(format!(
363            "Rows.db entry: rows_offset {rows_offset} + 2 (key length) exceeds file size {}",
364            rows_db.len()
365        )));
366    }
367
368    // [u16 key_length][key bytes]
369    let key_length = u16::from_be_bytes([rows_db[rows_offset], rows_db[rows_offset + 1]]) as usize;
370    let entry_start = rows_offset + 2 + key_length;
371    if entry_start > rows_db.len() {
372        return Err(Error::Parse(format!(
373            "Rows.db entry: key length {key_length} at offset {rows_offset} overruns file size {}",
374            rows_db.len()
375        )));
376    }
377
378    // base for the SIGNED root delta = RowsOffset + key_length (see module note).
379    let base = rows_offset + key_length;
380
381    let mut cur = entry_start;
382    let (data_position, n) = read_unsigned_vint_from_slice(&rows_db[cur..])?;
383    cur += n;
384
385    let (root_delta, n) = read_signed_vint_from_slice(&rows_db[cur..])?;
386    cur += n;
387
388    // indexTrieRoot = readVInt() + base   (TrieIndexEntry.deserialize)
389    let trie_root_signed = root_delta + base as i64;
390    if trie_root_signed < 0 || (trie_root_signed as usize) >= rows_db.len() {
391        return Err(Error::Parse(format!(
392            "Rows.db entry: recovered trie root {trie_root_signed} out of bounds \
393             (base={base}, delta={root_delta}, file size={})",
394            rows_db.len()
395        )));
396    }
397    let trie_root = trie_root_signed as usize;
398
399    let (block_count_u64, n) = read_unsigned_vint_from_slice(&rows_db[cur..])?;
400    cur += n;
401    let block_count = u32::try_from(block_count_u64).map_err(|_| {
402        Error::Parse(format!(
403            "Rows.db entry: implausible block count {block_count_u64}"
404        ))
405    })?;
406
407    // Partition DeletionTime: decode the MODERN DA/BTI form (issue #832
408    // Finding 2) best-effort: if too few trailing bytes remain, leave it `None`
409    // rather than failing (it is not required for traversal correctness).
410    let partition_deletion = match decode_da_deletion_time(rows_db, cur) {
411        Ok((deletion, _consumed)) => deletion,
412        Err(_) => None,
413    };
414
415    Ok(BtiRowIndexHeader {
416        data_position,
417        trie_root,
418        block_count,
419        partition_deletion,
420    })
421}

423/// Select the row-index blocks that may contain clustering keys in the
424/// inclusive byte-comparable range `[start, end]`, applying row-index
425/// **separator** semantics (issue #832 Finding B).
426///
427/// ## Why naive `[start, end]` filtering is wrong
428///
429/// A `Rows.db` row-index trie stores **separators**, not block start keys.  For
430/// consecutive blocks the writer (`RowIndexWriter.add`) stores the shortest
431/// `sep` with `prevMax < sep <= nextBlockFirstKey`, and `complete()` appends a
432/// trailing separator after the last block.  Consequently the separator `s_i`
433/// labels the boundary at the START of block `i`'s key range, and block `i`
434/// covers the half-open key interval `[s_i, s_{i+1})` (the final block runs to
435/// the trailing separator).  A reader locates the block for a key `K` via the
436/// trie *floor* of `K` (`RowIndexReader.separatorFloor`).
437///
438/// Therefore a block `i` overlaps the requested clustering range `[start, end]`
439/// iff its key interval `[s_i, s_{i+1})` intersects `[start, end]`:
440///
441///   `s_i <= end`  AND  `s_{i+1} > start`
442///
443/// (For the last block, `s_{i+1}` is treated as +∞.)
444///
445/// `entries` MUST be the full ascending-order `(separator, block)` list for one
446/// partition (as produced by [`iterate_rows_in_bti_trie`]).  `start`/`end` are
447/// byte-comparable clustering bounds in the **same encoding as the trie keys**.
448/// Reversed bounds (`start > end`) yield an empty result.
449///
450/// ## Implicit first block (issue #1968)
451///
452/// The trie stores a separator per block EXCEPT the first: the block covering
453/// keys BELOW `entries[0]`'s separator lives at the partition body start and has
454/// NO entry here (mirroring `RowIndexReader.separatorFloor`, which returns the
455/// partition start for a key below the first separator).  This function therefore
456/// only ever returns STORED blocks — it CANNOT return that implicit first block.
457/// A caller whose `start` sorts below `entries[0]`'s separator (e.g. an OPEN lower
458/// bound, `start == b""`) MUST additionally decode from the partition body start
459/// so the earliest clustering rows are not dropped; see
460/// `resolve_bti_clustering_seek_window` in `reader/data_access/bti.rs`.
461pub fn select_row_index_blocks_for_range(
462    entries: &[(Vec<u8>, BtiRowIndexEntry)],
463    start: &[u8],
464    end: &[u8],
465) -> Vec<BtiRowIndexEntry> {
466    if start > end || entries.is_empty() {
467        return Vec::new();
468    }
469
470    let mut out = Vec::new();
471    for (i, (sep_i, block)) in entries.iter().enumerate() {
472        // s_{i+1}: the next separator, or +∞ for the last block.
473        let next_is_greater_than_start = match entries.get(i + 1) {
474            Some((sep_next, _)) => sep_next.as_slice() > start,
475            None => true, // +∞ > start
476        };
477        let overlaps = sep_i.as_slice() <= end && next_is_greater_than_start;
478        if overlaps {
479            out.push(block.clone());
480        }
481    }
482    out
483}

496pub fn iterate_rows_for_partition(

498    rows_offset: usize,
499) -> BtiResult<(BtiRowIndexHeader, Vec<BtiRowIndexEntryWithKey>)> {
500    let header = resolve_rows_db_entry(rows_db, rows_offset)?;
501    let entries = iterate_rows_in_bti_trie(rows_db, header.trie_root)?;
502    Ok((header, entries))
503}

505/// Enumerate row-index entries in a `Rows.db` file that is a **single-partition**
506/// trie rooted at its 8-byte footer, in byte-comparable order.
507///
508/// ## Precondition
509///
510/// This treats the WHOLE file as one trie whose root is named by the trailing
511/// 8-byte footer.  That is only correct when the `Rows.db` contains exactly one
512/// partition's row-index trie (or is empty).  For a real multi-partition
513/// `Rows.db` you MUST instead use [`iterate_rows_in_bti_trie`] with the
514/// per-partition `RowsOffset` obtained from `Partitions.db`.
515///
516/// A `< 8`-byte (e.g. 0-byte) `Rows.db` yields an empty Vec without erroring.
517pub fn iterate_rows_in_bti_file<R: Read + Seek>(
518    reader: &mut R,
519) -> BtiResult<Vec<(Vec<u8>, BtiRowIndexEntry)>> {
520    let file_size = reader.seek(SeekFrom::End(0))?;
521    if file_size < 8 {
522        return Ok(Vec::new());
523    }
524    let (trie_data, root_offset) = load_bti_trie_via_footer(reader)?;
525    iterate_rows_in_bti_trie(&trie_data, root_offset)
526}

532    /// Dense16 (ordinal 11): [0xB0|pf] [start] [len-1] [range * 2-byte deltas]
533    fn dense16_node(payload_flags: u8, start: u8, deltas: &[u16]) -> Vec<u8> {
534        let len = deltas.len() as u8;
535        let mut v = vec![0xB0 | (payload_flags & 0x0F), start, len - 1];
536        for &d in deltas {
537            v.extend_from_slice(&d.to_be_bytes());
538        }
539        v
540    }

550    /// `(trie_bytes, root_offset)`.
551    fn make_rows_trie_three(
552        (k1, p1): (u8, u8),
553        (k2, p2): (u8, u8),
554        (k3, p3): (u8, u8),
555    ) -> (Vec<u8>, usize) {
556        let mut trie = Vec::new();

559        let o2 = trie.len() as u64; // 2

563        let root = trie.len() as u64; // 6
564        trie.push(0x50); // Sparse8

569        trie.push((root - o1) as u8);
570        trie.push((root - o2) as u8);
571        trie.push((root - o3) as u8);
572        (trie, root as usize)
573    }

809    /// Finding A (issue #832): `resolve_rows_db_entry` deserializes a synthetic
810    /// per-partition `TrieIndexEntry` and recovers the trie root.
811    #[test]
812    fn resolve_rows_db_entry_recovers_root_and_metadata() {
813        // Place a 1-byte pad so the trie "root" lives at a non-zero offset.
814        let mut buf = vec![0xEEu8; 4]; // bytes 0..4: pretend trie nodes
815        let rows_offset = buf.len(); // entry starts here, e.g. 4
816
817        // key: length 4, value 0x00000007
818        buf.extend_from_slice(&4u16.to_be_bytes());
819        buf.extend_from_slice(&[0x00, 0x00, 0x00, 0x07]);
820        let base = rows_offset + 4; // RowsOffset + key_length
821
822        // dataPos = 123 (unsigned vint, 1 byte since < 128)
823        buf.push(123);
824        // rootΔ such that trie_root = 2 (a node inside the pad region): Δ = 2 - base
825        let root_delta: i64 = 2 - base as i64;
826        // zigzag-encode then unsigned-vint (1 byte for small magnitudes)
827        let zig = ((root_delta << 1) ^ (root_delta >> 63)) as u64;
828        assert!(zig < 128, "test setup expects a 1-byte vint");
829        buf.push(zig as u8);
830        // blockCount = 38 (unsigned vint)
831        buf.push(38);
832        // partition DeletionTime (MODERN DA non-live): [i64 mfda=17][u32 ldt=9].
833        buf.extend_from_slice(&17i64.to_be_bytes());
834        buf.extend_from_slice(&9u32.to_be_bytes());
835
836        let header = resolve_rows_db_entry(&buf, rows_offset).unwrap();
837        assert_eq!(header.data_position, 123);
838        assert_eq!(
839            header.trie_root, 2,
840            "trie root = rootΔ + (RowsOffset + keylen)"
841        );
842        assert_eq!(header.block_count, 38);
843        // (local_deletion_time, marked_for_delete_at) = (9, 17).
844        assert_eq!(header.partition_deletion, Some((9, 17)));
845
846        // Out-of-bounds RowsOffset → clean error, no panic.
847        assert!(resolve_rows_db_entry(&buf, buf.len() + 10).is_err());
848    }
