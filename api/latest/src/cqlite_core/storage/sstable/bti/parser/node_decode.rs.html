16/// Classify the high nibble of a BTI node's first byte into one of the four
17/// Rust-level node categories.
18///
19/// The mapping follows the 16-entry `Types.values[]` array in `TrieNode.java`:
20/// - nibble 0               → `PayloadOnly`
21/// - nibbles 1-4 (Singles)  → `Single`
22/// - nibbles 5-9 (Sparses)  → `Sparse`
23/// - nibbles 10-15 (Denses) → `Dense`
24pub(crate) fn classify_node_nibble(nibble: u8) -> BtiResult<BtiNodeType> {
25    match nibble {
26        0 => Ok(BtiNodeType::PayloadOnly),
27        1..=4 => Ok(BtiNodeType::Single),
28        5..=9 => Ok(BtiNodeType::Sparse),
29        10..=15 => Ok(BtiNodeType::Dense),
30        other => Err(Error::Parse(format!(
31            "Invalid BTI node type nibble: {}",
32            other
33        ))),
34    }
35}

37/// Return the number of bytes used to store each child pointer for this node,
38/// given the raw high nibble (`ordinal`) from the first byte.
39///
40/// Matches the `bytesPerPointer` field of each `TrieNode` subtype in
41/// `TrieNode.java`.  Fractional (12-bit) encodings return `0` as a sentinel;
42/// callers that need to handle those cases do so explicitly.
43pub(crate) fn pointer_bytes_for_ordinal(ordinal: u8) -> u8 {
44    match ordinal {
45        0 => 0,  // PayloadOnly — no pointers
46        1 => 0,  // SingleNoPayload4  — 4-bit delta in low nibble, handled specially
47        2 => 1,  // Single8
48        3 => 0,  // SingleNoPayload12 — 12-bit delta across first two bytes
49        4 => 2,  // Single16
50        5 => 1,  // Sparse8
51        6 => 0,  // Sparse12 — 12-bit packed
52        7 => 2,  // Sparse16
53        8 => 3,  // Sparse24
54        9 => 5,  // Sparse40
55        10 => 0, // Dense12  — 12-bit packed
56        11 => 2, // Dense16
57        12 => 3, // Dense24
58        13 => 4, // Dense32
59        14 => 5, // Dense40
60        15 => 8, // LongDense
61        _ => 0,
62    }
63}

65/// Parse a `PayloadOnly` (ordinal 0) node.  PayloadOnly MUST carry a payload
66/// (it is a leaf node), decoded via the legacy 12-byte [`parse_payload_ref`].
67fn parse_payload_only_node(data: &[u8], has_payload: bool) -> BtiResult<BtiNode> {
68    if !has_payload {
69        return Err(Error::Parse(
70            "PayloadOnly node has no payload flags set".to_string(),
71        ));
72    }
73    let payload = parse_payload_ref(&data[1..])?;
74    Ok(BtiNode {
75        node_type: BtiNodeType::PayloadOnly,
76        level: 0,
77        key_prefix: Vec::new(),
78        data: BtiNodeData::PayloadOnly { payload },
79    })
80}

82/// Parse a `Single` (ordinals 1-4) node.
83///
84/// Three concrete layouts depending on ordinal:
85///
86///   ordinal 1 (SingleNoPayload4):
87///     byte 0: [1|delta_high4]  (delta is low 4 bits of byte 0)
88///     byte 1: transition byte; NO payload
89///   ordinal 3 (SingleNoPayload12):
90///     byte 0: [3|delta_high4]
91///     byte 1: delta_low8 → delta = (low4(byte0) << 8) | byte1
92///     byte 2: transition byte; NO payload
93///   ordinals 2, 4 (Single8, Single16):
94///     byte 0: [ordinal|payload_flags]
95///     byte 1: transition byte
96///     bytes 2..(2+ptr_bytes): backward delta (unsigned big-endian)
97///     [payload if has_payload]
98fn parse_single_node(data: &[u8], offset: u64, ordinal: u8, header_byte: u8) -> BtiResult<BtiNode> {
99    crate::storage::sstable::read_work_counters::record_bti_pointer_decode(); // H5 (#1650): 1 child
100    let transition = match ordinal {
101        1 => {
102            // SingleNoPayload4
103            if data.len() < 2 {
104                return Err(Error::Parse(
105                    "SingleNoPayload4 node data too short".to_string(),
106                ));
107            }
108            let delta = (header_byte & 0x0F) as u64;
109            let child_offset = offset.saturating_sub(delta);
110            Transition::new(data[1], SizedPointer::new(child_offset))
111        }
112        3 => {
113            // SingleNoPayload12
114            if data.len() < 3 {
115                return Err(Error::Parse(
116                    "SingleNoPayload12 node data too short".to_string(),
117                ));
118            }
119            let delta = (((header_byte & 0x0F) as u64) << 8) | (data[1] as u64);
120            let child_offset = offset.saturating_sub(delta);
121            Transition::new(data[2], SizedPointer::new(child_offset))
122        }
123        _ => {
124            // Single8 (ordinal 2) or Single16 (ordinal 4)
125            let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
126            let needed = 2 + ptr_bytes;
127            if data.len() < needed {
128                return Err(Error::Parse(format!(
129                    "Single node (ordinal {}) data too short: need {} bytes, have {}",
130                    ordinal,
131                    needed,
132                    data.len()
133                )));
134            }
135            let delta = read_be_unsigned(&data[2..2 + ptr_bytes]);
136            let child_offset = offset.saturating_sub(delta);
137            Transition::new(data[1], SizedPointer::new(child_offset))
138        }
139    };
140    Ok(BtiNode {
141        node_type: BtiNodeType::Single,
142        level: 1,
143        key_prefix: Vec::new(),
144        data: BtiNodeData::Single { transition },
145    })
146}

148/// Parse a `Sparse` (ordinals 5-9) node.
149///
150/// Layout (ordinals 5, 7, 8, 9 — full-byte pointers):
151///   byte 0: [ordinal|payload_flags]
152///   byte 1: count (number of transitions, 1-255)
153///   bytes 2..(2+count): transition bytes (sorted)
154///   then count × ptr_bytes: backward deltas
155///   [payload if has_payload]
156///
157/// ordinal 6 (Sparse12) packs two 12-bit deltas into 3 bytes.
158fn parse_sparse_node(data: &[u8], offset: u64, ordinal: u8) -> BtiResult<BtiNode> {
159    if data.len() < 2 {
160        return Err(Error::Parse("Sparse node data too short".to_string()));
161    }
162    let count = data[1] as usize;
163    if count == 0 {
164        return Err(Error::Parse(
165            "Sparse node must have at least one transition".to_string(),
166        ));
167    }
168
169    let bytes_start = 2;
170    let pointers_start = bytes_start + count;
171
172    let mut transitions = Vec::with_capacity(count);
173    if ordinal == 6 {
174        // Sparse12: each pair of pointers packed into 3 bytes
175        let packed_len = (count * 3).div_ceil(2); // ceil(count * 12 / 8) = ceil(count * 3 / 2)
176        let needed = pointers_start + packed_len;
177        if data.len() < needed {
178            return Err(Error::Parse(format!(
179                "Sparse12 node data too short: need {}, have {}",
180                needed,
181                data.len()
182            )));
183        }
184        for i in 0..count {
185            crate::storage::sstable::read_work_counters::record_bti_pointer_decode(); // H5 (#1650)
186            let t_byte = data[bytes_start + i];
187            let delta = read_12bit_packed(&data[pointers_start..], i);
188            let child_offset = offset.saturating_sub(delta);
189            transitions.push(Transition::new(t_byte, SizedPointer::new(child_offset)));
190        }
191    } else {
192        let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
193        let needed = pointers_start + count * ptr_bytes;
194        if data.len() < needed {
195            return Err(Error::Parse(format!(
196                "Sparse node (ordinal {}) data too short: need {}, have {}",
197                ordinal,
198                needed,
199                data.len()
200            )));
201        }
202        for i in 0..count {
203            crate::storage::sstable::read_work_counters::record_bti_pointer_decode(); // H5 (#1650)
204            let t_byte = data[bytes_start + i];
205            let ptr_off = pointers_start + i * ptr_bytes;
206            let delta = read_be_unsigned(&data[ptr_off..ptr_off + ptr_bytes]);
207            let child_offset = offset.saturating_sub(delta);
208            transitions.push(Transition::new(t_byte, SizedPointer::new(child_offset)));
209        }
210    }
211    Ok(BtiNode {
212        node_type: BtiNodeType::Sparse,
213        level: 1,
214        key_prefix: Vec::new(),
215        data: BtiNodeData::Sparse { transitions },
216    })
217}

219/// Parse a `Dense` (ordinals 10-15) node.
220///
221/// Layout (ordinals 11-14 — full-byte pointers): byte 0 `[ordinal|payload_flags]`,
222/// byte 1 start byte, byte 2 `(range_len - 1)`, then `range_len × ptr_bytes` backward
223/// deltas (`0` means "no transition"), then `[payload if has_payload]`.
224/// ordinal 10 (Dense12): packed 12-bit deltas; ordinal 15 (LongDense): 8-byte.
225fn parse_dense_node(data: &[u8], offset: u64, ordinal: u8) -> BtiResult<BtiNode> {
226    if data.len() < 3 {
227        return Err(Error::Parse("Dense node data too short".to_string()));
228    }
229    let start_byte = data[1];
230    let range_len = data[2] as usize + 1; // byte2 is (len-1)
231
232    let mut children = Vec::with_capacity(range_len);
233    if ordinal == 10 {
234        // Dense12: ceil(range_len * 12 / 8) = (range_len * 3 + 1) / 2
235        let packed_len = (range_len * 3).div_ceil(2);
236        let needed = 3 + packed_len;
237        if data.len() < needed {
238            return Err(Error::Parse(format!(
239                "Dense12 node data too short: need {}, have {}",
240                needed,
241                data.len()
242            )));
243        }
244        for i in 0..range_len {
245            crate::storage::sstable::read_work_counters::record_bti_pointer_decode(); // H5 (#1650)
246            let delta = read_12bit_packed(&data[3..], i);
247            children.push(dense_child(offset, delta));
248        }
249    } else {
250        let ptr_bytes = pointer_bytes_for_ordinal(ordinal) as usize;
251        let needed = 3 + range_len * ptr_bytes;
252        if data.len() < needed {
253            return Err(Error::Parse(format!(
254                "Dense node (ordinal {}) data too short: need {}, have {}",
255                ordinal,
256                needed,
257                data.len()
258            )));
259        }
260        for i in 0..range_len {
261            crate::storage::sstable::read_work_counters::record_bti_pointer_decode(); // H5 (#1650)
262            let ptr_off = 3 + i * ptr_bytes;
263            let delta = read_be_unsigned(&data[ptr_off..ptr_off + ptr_bytes]);
264            children.push(dense_child(offset, delta));
265        }
266    }
267    Ok(BtiNode {
268        node_type: BtiNodeType::Dense,
269        level: 1,
270        key_prefix: Vec::new(),
271        data: BtiNodeData::Dense {
272            start_byte,
273            children,
274        },
275    })
276}

278/// Resolve one Dense child slot from its raw backward delta.
279///
280/// `delta == 0` is the "no transition" sentinel → `None`.  Any other delta is
281/// a real child at `offset - delta` (which may be absolute offset 0 — a
282/// legitimate child, distinct from the sentinel).
283fn dense_child(offset: u64, delta: u64) -> Option<SizedPointer> {
284    if delta == 0 {
285        None
286    } else {
287        Some(SizedPointer::new(offset.saturating_sub(delta)))
288    }
289}

291/// Parse a BTI trie node from a raw byte slice.
292///
293/// This is the **single shared implementation** used by both the partition and
294/// row parsers.  It correctly dispatches all 16 node-type ordinals defined in
295/// `TrieNode.java` and produces the appropriate [`BtiNode`] variant.
296///
297/// # Arguments
298/// * `data`   – raw bytes starting at the node's first byte
299/// * `offset` – absolute file offset of this node (used to compute child
300///   positions; see the module-level doc for the sign convention)
301///
302/// # Errors
303/// Returns a [`crate::error::Error::Parse`] if the data is too short, the
304/// node-type nibble is out of range, or any other structural invariant is
305/// violated.
306pub(crate) fn parse_bti_node(data: &[u8], offset: u64) -> BtiResult<BtiNode> {
307    if data.is_empty() {
308        return Err(Error::Parse("Empty BTI node data".to_string()));
309    }
310
311    let header_byte = data[0];
312    let ordinal = (header_byte >> 4) & 0x0F;
313    let payload_flags = header_byte & 0x0F;
314    let has_payload = payload_flags != 0;
315    let node_type = classify_node_nibble(ordinal)?;
316
317    match node_type {
318        BtiNodeType::PayloadOnly => parse_payload_only_node(data, has_payload),
319        BtiNodeType::Single => parse_single_node(data, offset, ordinal, header_byte),
320        BtiNodeType::Sparse => parse_sparse_node(data, offset, ordinal),
321        BtiNodeType::Dense => parse_dense_node(data, offset, ordinal),
322    }
323}

325/// Read a big-endian unsigned integer of 0-8 bytes from `data`.
326///
327/// Returns 0 for an empty slice; panics if `data.len() > 8` (caller is
328/// responsible for bounds-checking first).
329pub(crate) fn read_be_unsigned(data: &[u8]) -> u64 {
330    let mut result = 0u64;
331    for &byte in data {
332        result = (result << 8) | (byte as u64);
333    }
334    result
335}

337/// Read a 12-bit value stored in the packed format used by Sparse12 / Dense12.
338///
339/// The packing (from `TrieNode.java#read12Bits`):
340/// ```text
341/// word = getShort(base + (3 * index) / 2)
342/// if (index & 1) == 0:  value = word >> 4
343/// else:                  value = word & 0xFFF
344/// ```
345pub(crate) fn read_12bit_packed(data: &[u8], index: usize) -> u64 {
346    let byte_offset = (3 * index) / 2;
347    if byte_offset + 1 >= data.len() {
348        return 0;
349    }
350    let word = ((data[byte_offset] as u16) << 8) | (data[byte_offset + 1] as u16);
351    let value = if (index & 1) == 0 {
352        word >> 4
353    } else {
354        word & 0x0FFF
355    };
356    value as u64
357}

359/// Parse a [`PayloadRef`] from the bytes immediately following a node header.
360///
361/// Format (matches `RowIndexReader.readPayload` / `PartitionIndex`):
362///   8 bytes big-endian: data file offset
363///   4 bytes big-endian: payload byte length
364pub(crate) fn parse_payload_ref(data: &[u8]) -> BtiResult<PayloadRef> {
365    if data.len() < 12 {
366        return Err(Error::Parse(format!(
367            "PayloadRef data too short: need 12 bytes, have {}",
368            data.len()
369        )));
370    }
371    let offset = u64::from_be_bytes([
372        data[0], data[1], data[2], data[3], data[4], data[5], data[6], data[7],
373    ]);
374    let length = u32::from_be_bytes([data[8], data[9], data[10], data[11]]);
375    Ok(PayloadRef::new(offset, length))
376}

455    /// Sparse12 (ordinal 6): packed 12-bit pointers.
456    ///
457    /// Layout per TrieNode.java Sparse12.serialize():
458    ///   [0x60|pf] [count] [count transition bytes]
459    ///   [ceil(count*3/2) packed-pointer bytes]
460    ///
461    /// Total node bytes = 2 + count + ceil(count*3/2).
462    /// count=1 → 5 bytes; count=2 → 7 bytes.
463    fn sparse12_node(payload_flags: u8, pairs: &[(u8, u16)]) -> Vec<u8> {
464        let count = pairs.len();
465        let mut v = vec![0x60 | (payload_flags & 0x0F), count as u8];
466        for &(t, _) in pairs {
467            v.push(t);
468        }
469        // Pack pointers: process pairs, then trailing odd entry
470        let mut i = 0;
471        while i + 2 <= count {
472            let p0 = pairs[i].1 as u32;
473            let p1 = pairs[i + 1].1 as u32;
474            v.push((p0 >> 4) as u8);
475            v.push(((p0 << 4) | (p1 >> 8)) as u8);
476            v.push((p1 & 0xFF) as u8);
477            i += 2;
478        }
479        if i < count {
480            // Trailing odd pointer: writeShort((short)(pd << 4)) big-endian
481            let pd = pairs[i].1 as u32;
482            let s = (pd << 4) as u16;
483            v.extend_from_slice(&s.to_be_bytes());
484        }
485        v
486    }

488    /// Sparse24 (ordinal 8): [0x80|pf] [count] [count transition bytes] [count * 3-byte big-endian deltas]
489    fn sparse24_node(payload_flags: u8, pairs: &[(u8, u32)]) -> Vec<u8> {
490        let count = pairs.len();
491        let mut v = vec![0x80 | (payload_flags & 0x0F), count as u8];
492        for &(t, _) in pairs {
493            v.push(t);

504    /// Sparse40 (ordinal 9): [0x90|pf] [count] [count transition bytes] [count * 5-byte big-endian deltas]
505    fn sparse40_node(payload_flags: u8, pairs: &[(u8, u64)]) -> Vec<u8> {
506        let count = pairs.len();
507        let mut v = vec![0x90 | (payload_flags & 0x0F), count as u8];
508        for &(t, _) in pairs {
509            v.push(t);
510        }
511        for &(_, d) in pairs {
512            // 5-byte big-endian
513            v.push(((d >> 32) & 0xFF) as u8);
514            v.push(((d >> 24) & 0xFF) as u8);
515            v.push(((d >> 16) & 0xFF) as u8);
516            v.push(((d >> 8) & 0xFF) as u8);
517            v.push((d & 0xFF) as u8);
518        }

522    /// Dense12 (ordinal 10): packed 12-bit pointers for a contiguous byte range.
523    ///
524    /// Layout per TrieNode.java Dense12.serialize():
525    ///   [0xA0|pf] [start_byte] [range_len - 1] [ceil(range_len*3/2) packed bytes]
526    fn dense12_node(payload_flags: u8, start: u8, deltas: &[u16]) -> Vec<u8> {
527        let range_len = deltas.len();
528        let mut v = vec![0xA0 | (payload_flags & 0x0F), start, (range_len - 1) as u8];
529        let mut carry: u8 = 0;
530        for (i, &d) in deltas.iter().enumerate() {
531            let val = d as u32;
532            if (i & 1) == 0 {
533                v.push((val >> 4) as u8);
534                carry = (val << 4) as u8;
535            } else {
536                v.push(carry | (val >> 8) as u8);
537                v.push((val & 0xFF) as u8);
538                carry = 0;
539            }
540        }
541        // If odd number of entries, flush carry byte
542        if (range_len & 1) == 1 {
543            v.push(carry);
544        }
545        v
546    }
