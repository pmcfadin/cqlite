54/// BTI format detection result
55#[derive(Debug, Clone, PartialEq)]
56pub enum FormatType {
57    /// Legacy BIG format
58    Big,
59    /// BTI (Big Trie-Indexed) format
60    Bti,
61}

63/// BTI SSTable metadata
64#[derive(Debug, Clone)]
65pub struct BtiMetadata {
66    /// Format version
67    pub version: u16,
68    /// Cassandra version that created this BTI
69    pub cassandra_version: CassandraVersion,
70    /// Root trie node offset in Partitions.db
71    pub partition_trie_root: u64,
72    /// Root trie node offset in Rows.db (if present)
73    pub row_trie_root: Option<u64>,
74    /// Number of partitions in the SSTable
75    pub partition_count: u64,
76    /// Additional BTI-specific properties
77    pub properties: HashMap<String, String>,
78}

80/// Detect BTI format from magic number
81pub fn detect_format(magic_number: u32) -> FormatType {
82    match magic_number {
83        BTI_MAGIC_NUMBER => FormatType::Bti,
84        _ => FormatType::Big, // Default to BIG format for all other magic numbers
85    }
86}
87
88/// Check if a magic number indicates BTI format
89pub fn is_bti_format(magic_number: u32) -> bool {
90    magic_number == BTI_MAGIC_NUMBER
91}

127#[cfg(test)]
128mod tests {
129    use super::*;
130
131    #[test]
132    fn test_bti_magic_number_detection() {
133        assert_eq!(detect_format(BTI_MAGIC_NUMBER), FormatType::Bti);
134        assert_eq!(detect_format(0x6F61_0000), FormatType::Big); // Legacy 'oa' format
135        assert_eq!(detect_format(0x0040_0000), FormatType::Big); // Cassandra 5.0 'nb' format
136
137        assert!(is_bti_format(BTI_MAGIC_NUMBER));
138        assert!(!is_bti_format(0x6F61_0000));
139    }
140
141    #[test]
142    fn test_bti_error_display() {
143        let err = node::BtiError::InvalidNodeType(0xFF);
144        assert!(err.to_string().contains("Invalid BTI trie node type: 0xFF"));
145
146        let err = node::BtiError::MaxDepthExceeded(150);
147        assert!(err
148            .to_string()
149            .contains("BTI trie depth exceeded maximum: 150"));
150
151        let err = node::BtiError::InvalidByteComparableKey("bad_key".to_string());
152        assert!(err
153            .to_string()
154            .contains("Invalid byte-comparable key: bad_key"));
155    }
156
157    #[test]
158    fn test_bti_config_default() {
159        let config = BtiConfig::default();
