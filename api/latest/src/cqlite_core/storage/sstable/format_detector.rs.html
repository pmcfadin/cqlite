1//! SSTable format detection and version identification
2//!
3//! This module *classifies* SSTable version letters into format families for
4//! diagnostics and routing. Classification (recognition) is deliberately
5//! broader than the **supported** read floor: a pre-`na` (`ma`–`me`,
6//! Cassandra 3.x) or pre-3.0 (`ic`/`jb`) version is still *recognized* so we
7//! can name it in errors, but it is NOT *supported*.
8//!
9//! ## Supported floor (#1249)
10//!
11//! "Supported" means CQLite has a real read path for the version: `na`/`nb`
12//! BIG (Cassandra 4.0 / 5.0 compat mode) and `oa`/`da` (Cassandra 5.0 BIG /
13//! BTI). Anything below `na` is recognized-but-unsupported so this preflight
14//! API agrees with `SSTableReader::open` / `StatisticsReader::open`, which
15//! reject below-floor SSTables outright. "Known" never implies "supported":
16//! `is_supported("ma")` is `false`.
17
18use crate::{Error, Result};
19use std::collections::HashMap;
20use std::path::Path;

22/// SSTable format versions supported by CQLite
23#[derive(Debug, Clone, PartialEq, Eq)]
24pub enum SSTableFormat {
25    /// Cassandra 2.x formats (ic, jb, etc.)
26    V2x(String),
27    /// Cassandra 3.x formats (ma, mb, mc, md, me)
28    V3x(String),
29    /// Cassandra 4.x formats (na, nb)
30    V4x(String),
31    /// Cassandra 5.x formats (oa, etc.)
32    V5x(String),
33    /// Unknown or unsupported format
34    Unknown(String),
35}
36
37impl SSTableFormat {
38    /// Get the format version string
39    pub fn version(&self) -> &str {
40        match self {
41            SSTableFormat::V2x(v) => v,
42            SSTableFormat::V3x(v) => v,
43            SSTableFormat::V4x(v) => v,
44            SSTableFormat::V5x(v) => v,
45            SSTableFormat::Unknown(v) => v,
46        }
47    }
48
49    /// Check if format supports compression
50    pub fn supports_compression(&self) -> bool {
51        match self {
52            SSTableFormat::V2x(_) => true,
53            SSTableFormat::V3x(_) => true,
54            SSTableFormat::V4x(_) => true,
55            SSTableFormat::V5x(_) => true,
56            SSTableFormat::Unknown(_) => false,
57        }
58    }
59
60    /// Check if format uses chunk-based compression
61    pub fn uses_chunk_compression(&self) -> bool {
62        match self {
63            SSTableFormat::V2x(_) => true,
64            SSTableFormat::V3x(_) => true,
65            SSTableFormat::V4x(_) => true,
66            SSTableFormat::V5x(_) => true,
67            SSTableFormat::Unknown(_) => false,
68        }
69    }
70
71    /// Get expected compression algorithm
72    pub fn default_compression(&self) -> &'static str {
73        match self {
74            SSTableFormat::V2x(_) => "SnappyCompressor",
75            SSTableFormat::V3x(_) => "LZ4Compressor",
76            SSTableFormat::V4x(_) => "LZ4Compressor",
77            SSTableFormat::V5x(_) => "LZ4Compressor",
78            SSTableFormat::Unknown(_) => "LZ4Compressor",
79        }
80    }
81
82    /// Whether CQLite *supports reading* this classified format (#1249).
83    ///
84    /// Supported = the na+ version floor: `na`/`nb` BIG (V4x) and `oa`/`da`
85    /// (V5x). Pre-`na` BIG (`ma`–`me`, V3x) and Cassandra 2.x (V2x) are
86    /// *recognized* by classification but are NOT supported, matching the
87    /// reject-at-open behaviour of `SSTableReader`/`StatisticsReader`.
88    pub fn is_supported(&self) -> bool {
89        match self {
90            // Exactly the four versions with a real read path — matching
91            // `supported_versions()`. A future/typo'd letter in the same family
92            // (e.g. `V4x("nc")`, `V5x("ob")`) is NOT a supported version and
93            // must return false (#1249).
94            SSTableFormat::V4x(v) => matches!(v.as_str(), "na" | "nb"),
95            SSTableFormat::V5x(v) => matches!(v.as_str(), "oa" | "da"),
96            // Pre-`na` (Cassandra 3.x) and 2.x are below the floor.
97            SSTableFormat::V2x(_) | SSTableFormat::V3x(_) => false,
98            SSTableFormat::Unknown(_) => false,
99        }
100    }
101}

103/// SSTable format detector with comprehensive version support
104pub struct FormatDetector {
105    /// Known format version mappings
106    format_map: HashMap<String, SSTableFormat>,
107}
108
109impl FormatDetector {
110    /// Create a new format detector with all *known* (recognizable) versions.
111    ///
112    /// The map is the recognition set, NOT the supported set: it includes
113    /// pre-`na` (`ma`–`me`) and 2.x letters so they can be classified and named
114    /// in diagnostics. Use [`FormatDetector::is_supported`] /
115    /// [`FormatDetector::supported_versions`] for the na+ supported floor.
116    pub fn new() -> Self {
117        let mut format_map = HashMap::new();
118
119        // Cassandra 2.x formats (recognized for diagnostics; NOT supported)
120        format_map.insert("ic".to_string(), SSTableFormat::V2x("ic".to_string()));
121        format_map.insert("jb".to_string(), SSTableFormat::V2x("jb".to_string()));
122
123        // Cassandra 3.x formats (recognized for diagnostics; NOT supported — #1249)
124        format_map.insert("ma".to_string(), SSTableFormat::V3x("ma".to_string()));
125        format_map.insert("mb".to_string(), SSTableFormat::V3x("mb".to_string()));
126        format_map.insert("mc".to_string(), SSTableFormat::V3x("mc".to_string()));
127        format_map.insert("md".to_string(), SSTableFormat::V3x("md".to_string()));
128        format_map.insert("me".to_string(), SSTableFormat::V3x("me".to_string()));
129
130        // Cassandra 4.x formats
131        format_map.insert("na".to_string(), SSTableFormat::V4x("na".to_string()));
132        format_map.insert("nb".to_string(), SSTableFormat::V4x("nb".to_string()));
133
134        // Cassandra 5.x formats
135        format_map.insert("oa".to_string(), SSTableFormat::V5x("oa".to_string()));
136        // BTI format (BtiFormat.java:287-420): `da` is the only BTI version letter.
137        // Mapped to V5x rather than Unknown so callers can select the BTI read path.
138        // Full BTI routing is completed in VG5; here we ensure `da` is not Unknown.
139        format_map.insert("da".to_string(), SSTableFormat::V5x("da".to_string()));
140
141        Self { format_map }
142    }
143
144    /// Detect SSTable format from file path
145    ///
146    /// SSTable files follow pattern: {version}-{generation}-{size}-{component}.db
147    /// Example: nb-1-big-Data.db
148    pub fn detect_from_path(&self, path: &Path) -> Result<SSTableFormat> {
149        let filename = path
150            .file_name()
151            .and_then(|f| f.to_str())
152            .ok_or_else(|| Error::InvalidPath(format!("Invalid SSTable filename: {:?}", path)))?;
153
154        // Extract format version from filename
155        let parts: Vec<&str> = filename.split('-').collect();
156        if parts.len() < 4 {
157            return Err(Error::InvalidFormat(format!(
158                "Invalid SSTable filename format: {}",
159                filename
160            )));
161        }
162
163        let version = parts[0];
164        self.detect_from_version(version)
165    }
166
167    /// Detect format from version string
168    pub fn detect_from_version(&self, version: &str) -> Result<SSTableFormat> {
169        self.format_map
170            .get(version)
171            .cloned()
172            .or_else(|| Some(SSTableFormat::Unknown(version.to_string())))
173            .ok_or_else(|| {
174                Error::UnsupportedFormat(format!("Unknown SSTable version: {}", version))
175            })
176    }
177
178    /// Detect format from multiple SSTable files in a directory
179    pub fn detect_from_directory(&self, dir: &Path) -> Result<SSTableFormat> {
180        use std::fs;
181
182        let entries = fs::read_dir(dir).map_err(Error::Io)?;
183
184        for entry in entries {
185            let entry = entry.map_err(Error::Io)?;
186            let path = entry.path();
187
188            if path.extension().and_then(|s| s.to_str()) == Some("db") {
189                if let Ok(format) = self.detect_from_path(&path) {
190                    return Ok(format);
191                }
192            }
193        }
194
195        Err(Error::InvalidFormat(
196            "No valid SSTable files found in directory".to_string(),
197        ))
198    }
199
200    /// Get all *supported* format versions (na+ floor, #1249).
201    ///
202    /// Returns only versions CQLite can actually read: `na`/`nb` BIG and
203    /// `oa`/`da` (V4x/V5x). Pre-`na` (`ma`–`me`) and 2.x letters are
204    /// recognizable via [`FormatDetector::detect_from_version`] but are
205    /// deliberately absent here.
206    pub fn supported_versions(&self) -> Vec<String> {
207        self.format_map
208            .iter()
209            .filter(|(_, fmt)| fmt.is_supported())
210            .map(|(v, _)| v.clone())
211            .collect()
212    }
213
214    /// Check if a format version is *supported* (has a read path).
215    ///
216    /// This is the na+ floor (#1249), NOT mere recognition: a recognized but
217    /// below-floor version such as `ma` returns `false`, so preflighting with
218    /// this API agrees with `SSTableReader::open`, which rejects it.
219    pub fn is_supported(&self, version: &str) -> bool {
220        self.format_map
221            .get(version)
222            .map(SSTableFormat::is_supported)
223            .unwrap_or(false)
224    }
225}
226
227impl Default for FormatDetector {
228    fn default() -> Self {
229        Self::new()
230    }
231}

233/// SSTable file components
234#[derive(Debug, Clone, PartialEq, Eq)]
235#[allow(clippy::upper_case_acronyms)]
236pub enum SSTableComponent {
237    Data,
238    Index,
239    Summary,
240    Filter,
241    CompressionInfo,
242    Statistics,
243    Digest,
244    /// Per-chunk CRC32 file for uncompressed SSTables (Cassandra `CRC.db`).
245    ///
246    /// Optional: emitted for uncompressed tables only (mutually exclusive with
247    /// `CompressionInfo.db`) and shared by BIG and BTI formats. Recognized here
248    /// for consistency with `directory::types::SSTableComponent::Crc`.
249    Crc,
250    TOC,
251}
252
253impl SSTableComponent {
254    /// Parse component from filename
255    pub fn from_filename(filename: &str) -> Option<Self> {
256        if filename.ends_with("-Data.db") {
257            Some(SSTableComponent::Data)
258        } else if filename.ends_with("-Index.db") {
259            Some(SSTableComponent::Index)
260        } else if filename.ends_with("-Summary.db") {
261            Some(SSTableComponent::Summary)
262        } else if filename.ends_with("-Filter.db") {
263            Some(SSTableComponent::Filter)
264        } else if filename.ends_with("-CompressionInfo.db") {
265            Some(SSTableComponent::CompressionInfo)
266        } else if filename.ends_with("-Statistics.db") {
267            Some(SSTableComponent::Statistics)
268        } else if filename.ends_with("-Digest.crc32") {
269            Some(SSTableComponent::Digest)
270        } else if filename.ends_with("-CRC.db") {
271            Some(SSTableComponent::Crc)
272        } else if filename.ends_with("-TOC.txt") {
273            Some(SSTableComponent::TOC)
274        } else {
275            None
276        }
277    }
278
279    /// Get component file suffix
280    pub fn suffix(&self) -> &'static str {
281        match self {
282            SSTableComponent::Data => "Data.db",
283            SSTableComponent::Index => "Index.db",
284            SSTableComponent::Summary => "Summary.db",
285            SSTableComponent::Filter => "Filter.db",
286            SSTableComponent::CompressionInfo => "CompressionInfo.db",
287            SSTableComponent::Statistics => "Statistics.db",
288            SSTableComponent::Digest => "Digest.crc32",
289            SSTableComponent::Crc => "CRC.db",
290            SSTableComponent::TOC => "TOC.txt",
291        }
292    }
293}

295/// SSTable file info extracted from path
296#[derive(Debug, Clone)]
297pub struct SSTableInfo {
298    pub format: SSTableFormat,
299    /// Raw SSTable identifier as a string.
300    ///
301    /// Cassandra 5.0 supports two id forms (Descriptor.java:95):
302    /// - Sequential integer: `"1"`, `"2"`, …
303    /// - UUID-based hex string: `"6aa08200a25111f0a3fef1a551383fb9"`
304    ///
305    /// This field stores the raw string so both forms round-trip correctly.
306    /// Use `generation_numeric()` when you need an integer id.
307    pub sstable_id: String,
308    /// The `<format>` name segment (`"big"` or `"bti"`).
309    pub size: String,
310    pub component: SSTableComponent,
311    pub base_name: String,
312}
313
314impl SSTableInfo {
315    /// Return the sequential generation number if the id is a plain integer.
316    ///
317    /// Returns `None` for UUID-based ids.
318    pub fn generation_numeric(&self) -> Option<u64> {
319        self.sstable_id.parse::<u64>().ok()
320    }
321
322    /// Parse SSTable info from file path.
323    ///
324    /// Accepts filenames with both sequential and UUID-based SSTable ids:
325    /// - `nb-1-big-Data.db`
326    /// - `nb-6aa08200a25111f0a3fef1a551383fb9-big-Data.db`
327    ///
328    /// The format segment (`big` / `bti`) is located by scanning right-to-left
329    /// from the end of the parts list, so multi-segment UUID ids do not
330    /// interfere with detection.
331    pub fn from_path(path: &Path) -> Result<Self> {
332        let filename = path
333            .file_name()
334            .and_then(|f| f.to_str())
335            .ok_or_else(|| Error::InvalidPath(format!("Invalid SSTable filename: {:?}", path)))?;
336
337        // Strip extension for splitting.
338        let base = if let Some(b) = filename.strip_suffix(".db") {
339            b
340        } else if let Some(b) = filename.strip_suffix(".txt") {
341            b
342        } else {
343            filename
344        };
345
346        let parts: Vec<&str> = base.split('-').collect();
347        if parts.len() < 4 {
348            return Err(Error::InvalidFormat(format!(
349                "Invalid SSTable filename format: {}",
350                filename
351            )));
352        }
353
354        let version = parts[0];
355
356        // Locate the format segment ("big"/"bti") by scanning right-to-left.
357        // Start at index 2 so we never confuse the version with the format.
358        let format_idx = parts[2..]
359            .iter()
360            .enumerate()
361            .rev()
362            .find(|(_, p)| **p == "big" || **p == "bti")
363            .map(|(i, _)| i + 2);
364
365        let format_idx = format_idx.ok_or_else(|| {
366            Error::InvalidFormat(format!(
367                "No 'big' or 'bti' format segment found in {:?}",
368                filename
369            ))
370        })?;
371
372        let size = parts[format_idx].to_string(); // "big" or "bti"
373
374        // id is everything between version and format
375        let sstable_id = parts[1..format_idx].join("-");
376
377        // component is everything after format, re-joined with original extension
378        let component_base = parts[format_idx + 1..].join("-");
379        let extension = if filename.ends_with(".db") {
380            ".db"
381        } else {
382            ".txt"
383        };
384        let component_filename = format!("-{}{}", component_base, extension);
385        let component = SSTableComponent::from_filename(&component_filename).ok_or_else(|| {
386            Error::InvalidFormat(format!(
387                "Unknown SSTable component {:?} in {:?}",
388                component_base, filename
389            ))
390        })?;
391
392        // base_name is <version>-<id>-<format>  (used to build sibling filenames)
393        let base_name = format!("{}-{}-{}", version, sstable_id, size);
394
395        let detector = FormatDetector::new();
396        let format = detector.detect_from_version(version)?;
397
398        Ok(SSTableInfo {
399            format,
400            sstable_id,
401            size,
402            component,
403            base_name,
404        })
405    }
406
407    /// Get path to companion component file
408    pub fn companion_path(
409        &self,
410        component: SSTableComponent,
411        base_dir: &Path,
412    ) -> std::path::PathBuf {
413        base_dir.join(format!("{}-{}", self.base_name, component.suffix()))
414    }
415}

454    /// #1249 R3: `is_supported` reflects the na+ read floor, NOT recognition.
455    /// Pre-`na` (`ma`–`me`, Cassandra 3.x) and 2.x letters MUST be unsupported,
456    /// while `na`/`nb`/`oa`/`da` MUST be supported. This keeps the preflight
457    /// API consistent with `SSTableReader`/`StatisticsReader::open`.
458    #[test]
459    fn test_is_supported_respects_na_floor() {
460        let detector = FormatDetector::new();

463        for v in &["ma", "mb", "mc", "md", "me", "ic", "jb"] {

473                v

476
477        // na+ floor: BIG na/nb and BIG/BTI oa/da are supported.
478        for v in &["na", "nb", "oa", "da"] {
