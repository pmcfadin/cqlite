1//! Cfg-gated multi-generation streaming-merge resident-rows probe (issue #1579,
2//! Epic D / D3).
3//!
4//! # Why this exists
5//!
6//! D3 makes `scan_stream` over more than one SSTable generation feed each
7//! reconciled partition from the k-way merger STRAIGHT into the bounded output
8//! channel, instead of collecting the ENTIRE reconciled table into a `Vec`,
9//! sorting it, and only then dribbling it down the channel. A correct answer
10//! never proves *how much memory* the merge held resident — the same rows come
11//! out either way. This probe makes the peak resident set observable so the
12//! O(window) claim is pinned the no-heuristics way: observe the *work* (rows the
13//! streaming producer held resident at once), not just the result.
14//!
15//! # The counter
16//!
17//! - [`record_resident`] raises a process-global high-water mark to `n` — the
18//!   number of rows the streaming producer had materialized-but-not-yet-drained
19//!   at one instant. The streaming driver reports each stepped partition's row
20//!   count (its resident window is one partition at a time). The pre-D3 eager
21//!   path reported `merged.len()` — the whole reconciled table — so the peak was
22//!   proportional to the row count and the D3 memory guard
23//!   (`peak_resident` FLAT across fixture sizes) flips red without the streaming
24//!   rework.
25//!
26//! # Zero overhead in release (same pattern as [`crate::query::agg_stream_probe`])
27//!
28//! [`record_resident`] is an **unconditional** `#[inline(always)]` public
29//! function; its body is `#[cfg(any(test, feature = "work-counters"))]`. In a
30//! default/release build (no `work-counters`, no `cfg(test)`) the body is empty
31//! and optimizes away — no atomic is even linked. The getter + [`reset`] are
32//! `#[cfg(any(test, feature = "work-counters"))]`: in-crate unit tests reach them
33//! via the `test` cfg; integration tests in `tests/` enable the `work-counters`
34//! feature (they compile the lib WITHOUT its `test` cfg).
35
36#[cfg(any(test, feature = "work-counters"))]
37use std::sync::atomic::{AtomicU64, Ordering};
38
39/// Process-global high-water mark of rows the streaming multi-generation merge
40/// held resident at one instant since the last [`reset`] (test/feature builds
41/// only).
42#[cfg(any(test, feature = "work-counters"))]
43static PEAK_RESIDENT: AtomicU64 = AtomicU64::new(0);
44
45/// Record that the streaming multi-generation merge held `n` rows resident at
46/// one instant, raising the process-global high-water mark. Called
47/// unconditionally by the streaming merge driver; the body compiles to a no-op
48/// in a release build.
49#[inline(always)]
50pub fn record_resident(n: u64) {
51    #[cfg(any(test, feature = "work-counters"))]
52    PEAK_RESIDENT.fetch_max(n, Ordering::Relaxed);
53    #[cfg(not(any(test, feature = "work-counters")))]
54    let _ = n;
55}
56
57/// Peak resident rows the streaming multi-generation merge held at one instant
58/// since the last [`reset`]. Streaming holds one partition at a time, so this is
59/// FLAT across fixture sizes; the pre-D3 eager path reported the whole table.
60#[cfg(any(test, feature = "work-counters"))]
61pub fn peak_resident() -> u64 {
62    PEAK_RESIDENT.load(Ordering::Relaxed)
63}
64
65/// Clear the process-global high-water mark. Integration tests call this before a
66/// measured stream so a stale value cannot satisfy a later assertion; because the
67/// global is shared, callers serialize on the shared test mutex (the counter-test
68/// convention).
69#[cfg(any(test, feature = "work-counters"))]
70pub fn reset() {
71    PEAK_RESIDENT.store(0, Ordering::Relaxed);
72}
73
74#[cfg(test)]
75mod tests {
76    use super::*;
77    use serial_test::serial;
78
79    // Exercises the max/get/reset contract on the process-global. Serialized so it
80    // never overlaps another test mutating the same global.
81    #[test]
82    #[serial]
83    fn peak_resident_high_water_and_reset() {
84        reset();
85        assert_eq!(peak_resident(), 0);
86        record_resident(5);
87        record_resident(3); // lower value does not lower the high-water mark
88        assert_eq!(peak_resident(), 5);
89        record_resident(9);
90        assert_eq!(peak_resident(), 9);
91        reset();
92        assert_eq!(peak_resident(), 0);
93    }
94}
