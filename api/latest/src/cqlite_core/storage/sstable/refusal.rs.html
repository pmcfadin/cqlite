1//! Refused-SSTable ledger (issue #4159).
2//!
3//! The initial load paths open each `Data.db` best-effort: one file that cannot
4//! be opened (metadata/header refusal, bad `Statistics.db`, unsupported version,
5//! I/O error) must not make every OTHER table in the directory unreadable. But
6//! the old code logged the error and dropped it, so a table whose every
7//! generation was refused had ZERO readers and `scan` answered `Ok(vec![])` —
8//! indistinguishable from an empty table (the #3721 swallow class at SSTable
9//! granularity).
10//!
11//! The manager now RECORDS each refusal here, keyed by the table the file
12//! belongs to, and every read entry point consults the ledger before it
13//! answers: `Ok(empty)` means "no data", `Err` means "unreadable". Partial
14//! refusals (some generations opened, one did not) also error, because a
15//! partial answer presented as complete is the same silent data loss.
16//!
17//! The recorded cause keeps its error KIND (a `Schema` refusal stays `Schema`,
18//! #4139) rather than being flattened to text.
19
20use crate::Error;
21use std::collections::HashMap;
22use std::path::{Path, PathBuf};
23use std::sync::Mutex;

25/// One refused `Data.db`: where, and the cause (kind-preserving).
26#[derive(Debug)]
27struct Refusal {
28    path: PathBuf,
29    cause: Error,
30}
31
32/// Table key -> refused generations.
33#[derive(Debug, Default)]
34pub(crate) struct RefusalLedger {
35    by_table: Mutex<HashMap<String, Vec<Refusal>>>,
36}

38/// Rebuild `e` as an owned error of the SAME kind (`Error` is not `Clone`
39/// because it can carry an `io::Error`); kinds that carry no structure fold into
40/// `Storage`, which still names the full cause text.
41fn clone_kind(e: &Error) -> Error {
42    match e {
43        Error::Schema(m) => Error::Schema(m.clone()),
44        Error::Corruption(m) => Error::Corruption(m.clone()),
45        Error::InvalidFormat(m) => Error::InvalidFormat(m.clone()),
46        Error::UnsupportedFormat(m) => Error::UnsupportedFormat(m.clone()),
47        Error::UnsupportedVersion { version, floor } => Error::UnsupportedVersion {
48            version: version.clone(),
49            floor: floor.clone(),
50        },
51        other => Error::Storage(other.to_string()),
52    }
53}

55/// Unqualified table name of a (possibly `keyspace.table`) key.
56fn unqualified(name: &str) -> &str {
57    name.rfind('.').map_or(name, |dot| &name[dot + 1..])
58}

60/// Ledger key for a refusal whose table cannot be derived from its path
61/// (non-standard or non-UTF-8 directory names). It matches EVERY read: when the
62/// affected table is unknowable, no table can be proven unaffected (#4159).
63pub(crate) const UNRESOLVED_KEY: &str = "\u{0}unresolved";
64
65/// [`table_dir_table_key`](super::refresh::table_dir_table_key) or [`UNRESOLVED_KEY`].
66pub(crate) fn table_dir_key_or_unresolved(path: &Path) -> String {
67    super::refresh::table_dir_table_key(path).unwrap_or_else(|| UNRESOLVED_KEY.to_string())
68}

70/// Ledger key for a refused file whose reader never existed (no header table
71/// name): derived ONLY through `base_path_table_key`, the helper (with its base-directory
72/// exclusion) the manager keys base-path readers with;
73/// anything else is [`UNRESOLVED_KEY`]. An arbitrary parent-directory name is not
74/// authoritative for a nonstandard layout, so it is deliberately not a fallback.
75pub(crate) fn refusal_key(path: &Path, base_dir_name: &str) -> String {
76    super::refresh::base_path_table_key(path, base_dir_name, "")
77        .unwrap_or_else(|| UNRESOLVED_KEY.to_string())
78}

80/// `e` rebuilt as the SAME kind with `what` named in its message, so a failure that returns
81/// to a caller says which file or directory it concerns without changing how it matches.
82fn name_failure(e: Error, what: &str) -> Error {
83    let msg = format!("{what}: {e}");
84    match e {
85        Error::Schema(_) => Error::Schema(msg),
86        Error::Corruption(_) => Error::Corruption(msg),
87        Error::InvalidFormat(_) => Error::InvalidFormat(msg),
88        Error::UnsupportedFormat(_) => Error::UnsupportedFormat(msg),
89        Error::UnsupportedVersion { version, floor } => Error::UnsupportedVersion {
90            version: format!("{version} ({what})"),
91            floor,
92        },
93        Error::Io(io) => Error::Io(std::io::Error::new(io.kind(), msg)),
94        other => Error::Storage(format!("{what}: {other}")),
95    }
96}

98/// Is this directory entry a GENERATION? A `*-Data.db` name alone is not enough: only a REGULAR
99/// file is one (`DirEntry::file_type` does not follow symlinks), so a directory or symlink that
100/// merely carries the name is ignored -- not a generation and not a refusal (roborev #3234 r12).
101/// If the type cannot be observed the entry stays a candidate, so the failure surfaces as a
102/// refusal on open instead of the entry silently vanishing.
103pub(crate) async fn is_generation_entry(filename: &str, entry: &tokio::fs::DirEntry) -> bool {
104    filename.ends_with("-Data.db")
105        && !super::is_apple_double_sidecar(filename)
106        && entry.file_type().await.map(|t| t.is_file()).unwrap_or(true)
107}

109impl RefusalLedger {
110    /// Record that `path` (belonging to table `key`) was refused with `cause`.
111    pub(crate) fn record(&self, key: String, path: &Path, cause: &Error) {
112        tracing::warn!("SSTable {:?} refused ({}): {}", path, key, cause);
113        if let Ok(mut map) = self.by_table.lock() {
114            let list = map.entry(key).or_default();
115            list.retain(|r| r.path != path);
116            list.push(Refusal {
117                path: path.to_path_buf(),
118                cause: clone_kind(cause),
119            });
120        }
121    }
122
123    /// Forget any refusal recorded for `path` (it was re-opened successfully).
124    pub(crate) fn clear_path(&self, path: &Path) {
125        if let Ok(mut map) = self.by_table.lock() {
126            for list in map.values_mut() {
127                list.retain(|r| r.path != path);
128            }
129            map.retain(|_, l| !l.is_empty());
130        }
131    }
132
133    /// Forget the refusals of every path in `paths` (they opened, and the refresh
134    /// that opened them has been committed).
135    pub(crate) fn clear_paths(&self, paths: &[PathBuf]) {
136        for p in paths {
137            self.clear_path(p);
138        }
139    }
140
141    /// Reconcile after a SUCCESSFUL refresh: keep a refusal only for a `Data.db`
142    /// still in the discovered set. Directory-level refusals are all dropped:
143    /// discovery fails the refresh when an existing directory cannot be listed, so
144    /// reaching the commit means every configured directory was enumerated (or is
145    /// gone), and a still-unreadable one is re-recorded by the next load/refresh.
146    pub(crate) fn retain_discovered(&self, discovered: &[PathBuf]) {
147        if let Ok(mut map) = self.by_table.lock() {
148            for list in map.values_mut() {
149                list.retain(|r| {
150                    r.path.to_string_lossy().ends_with("-Data.db") && discovered.contains(&r.path)
151                });
152            }
153            map.retain(|_, l| !l.is_empty());
154        }
155    }
156
157    /// `Err` naming the cause when any generation of `table_name` was refused.
158    ///
159    /// Matching mirrors the reader resolution: exact key first; otherwise the
160    /// unqualified names must agree and at most one side may carry a keyspace
161    /// (two different keyspaces never bleed into one another).
162    pub(crate) fn ensure_readable(&self, table_name: &str) -> crate::Result<()> {
163        let Ok(map) = self.by_table.lock() else {
164            return Err(Error::Internal(
165                "refused-SSTable ledger lock poisoned; cannot prove the table readable".into(),
166            ));
167        };
168        // Deterministic report: exact-key refusals first, then unqualified matches,
169        // then UNRESOLVED; within a class, sorted by path. Never HashMap order.
170        let mut ranked: Vec<(u8, &Refusal)> = Vec::new();
171        for (key, list) in map.iter() {
172            let both_qualified = key.contains('.') && table_name.contains('.');
173            let class = if key == table_name {
174                Some(0)
175            } else if key == UNRESOLVED_KEY {
176                Some(2)
177            } else if !both_qualified && unqualified(key) == unqualified(table_name) {
178                Some(1)
179            } else {
180                None
181            };
182            if let Some(c) = class {
183                ranked.extend(list.iter().map(|r| (c, r)));
184            }
185        }
186        ranked.sort_by(|a, b| (a.0, &a.1.path).cmp(&(b.0, &b.1.path)));
187        let hits: Vec<&Refusal> = ranked.into_iter().map(|(_, r)| r).collect();
188        let Some(first) = hits.first() else {
189            return Ok(());
190        };
191        // Name EVERY refused file (capped) so no cause is hidden behind another.
192        let mut names: Vec<String> = hits
193            .iter()
194            .take(4)
195            .map(|r| r.path.display().to_string())
196            .collect();
197        if hits.len() > 4 {
198            names.push(format!("and {} more", hits.len() - 4));
199        }
200        let msg = format!(
201            "table '{}' is unreadable: SSTable(s) refused: {}; first cause: {}",
202            table_name,
203            names.join(", "),
204            first.cause
205        );
206        Err(match &first.cause {
207            Error::UnsupportedVersion { version, floor } => Error::UnsupportedVersion {
208                version: format!("{version} ({msg})"),
209                floor: floor.clone(),
210            },
211            Error::Schema(_) => Error::Schema(msg),
212            Error::Corruption(_) => Error::Corruption(msg),
213            Error::InvalidFormat(_) => Error::InvalidFormat(msg),
214            Error::UnsupportedFormat(_) => Error::UnsupportedFormat(msg),
215            _ => Error::Storage(msg),
216        })
217    }
218}

220impl super::SSTableManager {
221    /// Reader snapshot for a read that returns rows: `Err` naming the cause when
222    /// any SSTable of the table was refused, else
223    /// [`resolve_reader_snapshot`](Self::resolve_reader_snapshot). Probes and shape
224    /// hints keep the unchecked form (they are fail-safe: the real read errs).
225    pub(crate) async fn checked_snapshot(
226        &self,
227        table_id: &crate::types::TableId,
228    ) -> crate::Result<(Vec<std::sync::Arc<super::reader::SSTableReader>>, bool)> {
229        self.refusals.ensure_readable(table_id.name())?;
230        #[cfg(test)]
231        race_hook::pause_if_probe(table_id.name()).await;
232        let snapshot = self.resolve_reader_snapshot(table_id).await;
233        // The consistency point: a refusal recorded (by a concurrent refresh) while
234        // the snapshot was being taken is visible HERE, so a result is never
235        // returned after the refusal it should have reported.
236        self.refusals.ensure_readable(table_id.name())?;
237        Ok(snapshot)
238    }
239
240    /// [`checked_snapshot`](Self::checked_snapshot) for the streaming scans.
241    #[cfg(not(feature = "tombstones"))]
242    pub(super) async fn checked_table_readers(
243        &self,
244        table_id: &crate::types::TableId,
245    ) -> crate::Result<Vec<std::sync::Arc<super::reader::SSTableReader>>> {
246        self.refusals.ensure_readable(table_id.name())?;
247        #[cfg(test)]
248        race_hook::pause_if_probe(table_id.name()).await;
249        let readers = self.resolve_table_readers(table_id).await;
250        self.refusals.ensure_readable(table_id.name())?; // consistency point, as above
251        Ok(readers)
252    }
253
254    /// Record that an EXISTING table directory could not be listed, and hand the
255    /// error back for the caller to return (refresh) or drop (initial load).
256    pub(super) fn refuse_dir(&self, dir: &Path, e: crate::Error) -> crate::Error {
257        let probe = dir.join("unreadable-Data.db");
258        self.refusals
259            .record(table_dir_key_or_unresolved(&probe), dir, &e);
260        e
261    }
262
263    /// List the `Data.db` files of one existing table directory for a REFRESH. EVERY
264    /// failure -- opening the listing or advancing it -- is returned (naming the
265    /// directory), so a partial enumeration can never be mistaken for a complete one.
266    /// A failed refresh changes NOTHING (storage-freshness spec): neither the reader
267    /// set nor the refusal ledger; its `Err` is the loud signal.
268    pub(super) async fn list_data_files(&self, dir: &Path) -> crate::Result<Vec<PathBuf>> {
269        let mut entries = match self.platform.fs().read_dir(dir).await {
270            Ok(entries) => entries,
271            Err(e) => return Err(name_failure(e, &format!("directory {}", dir.display()))),
272        };
273        let mut out = Vec::new();
274        loop {
275            let entry = match entries.next_entry().await {
276                Ok(Some(entry)) => entry,
277                Ok(None) => break,
278                Err(e) => {
279                    return Err(name_failure(
280                        e.into(),
281                        &format!("directory {}", dir.display()),
282                    ))
283                }
284            };
285            let path = entry.path();
286            let fname = path.file_name().and_then(|n| n.to_str()).unwrap_or("");
287            if is_generation_entry(fname, &entry).await {
288                out.push(path);
289            }
290        }
291        Ok(out)
292    }
293
294    /// Hint-only form: a refused table yields no readers.
295    #[cfg(not(feature = "tombstones"))]
296    pub(super) async fn readers_if_readable(
297        &self,
298        table_id: &crate::types::TableId,
299    ) -> Vec<std::sync::Arc<super::reader::SSTableReader>> {
300        self.checked_table_readers(table_id)
301            .await
302            .unwrap_or_default()
303    }
304
305    /// Open a refresh candidate. A failure is returned NAMING the candidate (same error
306    /// kind) and records nothing: a failed refresh changes neither the reader set nor
307    /// the refusal ledger (storage-freshness spec, "Corrupt new generation rejects the
308    /// whole refresh"). A success does not clear anything here either: stale refusals
309    /// are cleared together with the reader-set swap, by
310    /// [`commit_refusals`](Self::commit_refusals).
311    pub(super) async fn open_candidate(
312        &self,
313        path: &Path,
314    ) -> crate::Result<std::sync::Arc<super::reader::SSTableReader>> {
315        self.open_reader_with_schema(path)
316            .await
317            .map_err(|e| name_failure(e, &format!("refresh candidate {}", path.display())))
318    }
319
320    /// Commit a SUCCESSFUL refresh's ledger changes with the reader-set swap:
321    /// the candidates that opened lose their refusals, and refusals of files no
322    /// longer on disk are reconciled away.
323    pub(super) fn commit_refusals(&self, opened: &[PathBuf], discovered: &[PathBuf]) {
324        self.refusals.clear_paths(opened);
325        self.refusals.retain_discovered(discovered);
326    }
327}

329/// Test-only deterministic pause between the first ledger check and the reader
330/// snapshot (no sleeps): active only for the reserved probe table name.
331#[cfg(test)]
332pub(crate) mod race_hook {
333    use std::sync::{Arc, Mutex};
334    use tokio::sync::Notify;
335
336    pub(crate) const PROBE: &str = "__race_probe__";
337    type Gate = (Arc<Notify>, Arc<Notify>); // (reached, release)
338    static GATE: Mutex<Option<Gate>> = Mutex::new(None);
339
340    pub(crate) fn arm() -> Gate {
341        let gate = (Arc::new(Notify::new()), Arc::new(Notify::new()));
342        if let Ok(mut g) = GATE.lock() {
343            *g = Some(gate.clone());
344        }
345        gate
346    }
347
348    pub(crate) async fn pause_if_probe(name: &str) {
349        if name != PROBE {
350            return;
351        }
352        let gate = GATE.lock().ok().and_then(|g| g.clone());
353        if let Some((reached, release)) = gate {
354            reached.notify_one();
355            release.notified().await;
356        }
357    }
358}

364    /// A refusal recorded between the ledger check and the snapshot (a concurrent
365    /// refresh) must turn the read into an `Err`, never an empty/partial `Ok`.
366    #[tokio::test]
367    async fn refusal_recorded_during_snapshot_fails_the_read() {
368        let tmp = tempfile::tempdir().expect("tmp");
369        let config = crate::Config::default();
370        let platform = std::sync::Arc::new(crate::Platform::new(&config).await.expect("platform"));
371        let manager = std::sync::Arc::new(
372            super::super::SSTableManager::new(
373                tmp.path(),
374                &config,
375                platform,
376                #[cfg(feature = "state_machine")]
377                None,
378            )
379            .await
380            .expect("manager"),
381        );
382        let (reached, release) = race_hook::arm();
383        let reader = {
384            let m = manager.clone();
385            tokio::spawn(async move {
386                m.checked_snapshot(&crate::types::TableId::from(race_hook::PROBE))
387                    .await
388                    .map(|_| ())
389            })
390        };
391        reached.notified().await; // the read has passed its first ledger check
392        manager.refusals.record(
393            race_hook::PROBE.to_string(),
394            Path::new("/d/ks/probe-1/nb-1-big-Data.db"),
395            &Error::Corruption("refused by a concurrent refresh".into()),
396        );
397        release.notify_one();
398        let result = reader.await.expect("join");
399        assert!(
400            result.is_err(),
401            "the read must observe the refusal recorded mid-read: {result:?}"
402        );
403    }

405    #[test]
406    fn coexisting_refusal_classes_report_deterministically() {
407        // Exact key (Schema) outranks unqualified (Corruption) outranks UNRESOLVED
408        // (InvalidFormat), whatever the insertion order.
409        let rec = |l: &RefusalLedger, k: &str, p: &str, e: Error| {
410            l.record(k.to_string(), Path::new(p), &e)
411        };
412        for order in [[0, 1, 2], [2, 1, 0], [1, 2, 0], [1, 0, 2]] {
413            let ledger = RefusalLedger::default();
414            for i in order {
415                match i {
416                    0 => rec(
417                        &ledger,
418                        "ks.t",
419                        "/d/ks/t-1/nb-1-big-Data.db",
420                        Error::Schema("a".into()),
421                    ),
422                    1 => rec(
423                        &ledger,
424                        "t",
425                        "/d/other/t-2/nb-2-big-Data.db",
426                        Error::Corruption("b".into()),
427                    ),
428                    _ => rec(
429                        &ledger,
430                        UNRESOLVED_KEY,
431                        "/d/\u{fffd}/nb-3-big-Data.db",
432                        Error::InvalidFormat("c".into()),
433                    ),
434                }
435            }
436            match ledger.ensure_readable("ks.t") {
437                Err(Error::Schema(m)) => assert!(
438                    m.find("t-1").unwrap_or(usize::MAX) < m.find("t-2").unwrap_or(0),
439                    "{m}"
440                ),
441                other => panic!("expected the exact-key Schema refusal first, got {other:?}"),
442            }
443        }
444    }
