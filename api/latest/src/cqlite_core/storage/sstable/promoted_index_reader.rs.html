1//! Decoder for the BIG ("nb") format promoted index payload (Issue #993).
2//!
3//! The promoted index lives inside each wide-partition Index.db entry. It lets a
4//! reader seek to a clustering-key range inside a partition larger than 64 KiB
5//! without scanning the whole partition. The CQLite **writer** is the authoritative
6//! oracle for this format (`writer/index_writer.rs::serialize_promoted_index` /
7//! `serialize_index_info`); this module mirrors that layout exactly on the read path.
8//!
9//! Previously the read path DECODED-AWAY (skipped) this payload. This module decodes
10//! it instead, so wide-partition boundary metadata round-trips.
11//!
12//! # On-disk layout (BIG "nb" — `RowIndexEntry.IndexedEntry.serialize()`)
13//!
14//! ```text
15//! [headerLength: unsigned VInt]    ← Data.db byte offset from partition start to first row
16//! [DeletionTime: 12 bytes]         ← NB legacy form: [localDeletionTime: i32 BE][markedForDeleteAt: i64 BE]
17//! [count: unsigned VInt]           ← number of IndexInfo blocks
18//! [IndexInfo[0]..IndexInfo[N-1]]   ← serialized blocks, in order
19//! [offset[0]: i32 BE] ...          ← relative offsets from first IndexInfo start, one per block
20//! ```
21//!
22//! Each `IndexInfo` block:
23//! ```text
24//! [firstName: ClusteringPrefix]    ← min clustering key in block (header VInt + value bytes)
25//! [lastName: ClusteringPrefix]     ← max clustering key in block
26//! [offset: unsigned VInt]          ← byte offset from partition start
27//! [width: signed VInt]             ← (actual_width - WIDTH_BASE), zigzag-encoded
28//! [endOpenMarker: bool byte]       ← 0x00 = none; 0x01 = a DeletionTime (12 bytes) follows
29//! ```
30//!
31//! # Format ambiguity resolved against the oracle (Issue #993)
32//!
33//! `firstName` / `lastName` are serialized `ClusteringPrefix` byte sequences with the
34//! shape `[kind 1 byte][header VInt][value bytes…]` (Cassandra
35//! `ClusteringPrefix.serializer.serialize` — the IndexInfo form prepends the
36//! `Kind.ordinal()` byte; a full clustering key is kind `CLUSTERING = 0x04`, Issue
37//! #1186). Fixed-width clustering values carry **no** per-value length prefix, so a
38//! `ClusteringPrefix` is only self-delimiting when the decoder knows the clustering
39//! column types (the table's serialization header). Cassandra's
40//! `IndexInfo.Serializer.deserialize()` likewise threads a `ClusteringComparator` +
41//! `SerializationHeader` to split the names. For a single `int` clustering the prefix
42//! is the Cassandra-exact 6 bytes `04 00 <4-byte int>`.
43//!
44//! Mirroring the writer, this decoder therefore takes a caller-supplied
45//! [`PrefixLen`] callback that returns the byte length of one serialized
46//! `ClusteringPrefix` at the start of a slice (authoritative, schema-driven — no
47//! heuristics, per Issue #28). The width delta is decoded as an unsigned VInt then
48//! zigzag-decoded, exactly inverting the writer's `encode_signed` (zigzag →
49//! `encode_unsigned`).
50//!
51//! Sources:
52//! - `writer/index_writer.rs` (the oracle: `serialize_promoted_index`, `serialize_index_info`)
53//! - Cassandra `RowIndexEntry.IndexedEntry.serialize()` / `IndexInfo.Serializer`
54//! - `DeletionTime.LegacySerializer` (NB form: ldt i32 BE, then mfda i64 BE)
55

59/// Byte size of an NB (legacy) `DeletionTime`: i32 BE localDeletionTime + i64 BE markedForDeleteAt.
60const NB_DELETION_TIME_SIZE: usize = 12;
61
62/// Delta base for `IndexInfo.width` (Cassandra `IndexInfo.WIDTH_BASE` = 64 KiB).
63const INDEX_INFO_WIDTH_BASE: i64 = 64 * 1024;
64
65/// Decoded partition-level `DeletionTime` (NB legacy encoding).
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67pub struct PromotedDeletionTime {
68    /// `localDeletionTime` (seconds). `i32::MAX` == LIVE (no deletion).
69    pub local_deletion_time: i32,
70    /// `markedForDeleteAt` (microseconds). `i64::MIN` == LIVE (no deletion).
71    pub marked_for_delete_at: i64,
72}
73
74impl PromotedDeletionTime {
75    /// True when this is the LIVE sentinel (`i32::MAX` / `i64::MIN`).
76    pub fn is_live(&self) -> bool {
77        self.local_deletion_time == i32::MAX && self.marked_for_delete_at == i64::MIN
78    }
79}

81/// One decoded `IndexInfo` block from a promoted index.
82#[derive(Debug, Clone, PartialEq, Eq)]
83pub struct DecodedIndexInfo {
84    /// Serialized `ClusteringPrefix` bytes for the first unfiltered in this block.
85    pub first_name: Vec<u8>,
86    /// Serialized `ClusteringPrefix` bytes for the last unfiltered in this block.
87    pub last_name: Vec<u8>,
88    /// Byte offset from the partition start to this block's first unfiltered.
89    pub offset: u64,
90    /// Total width (bytes) of this block's data (delta already added back to `WIDTH_BASE`).
91    pub width: u64,
92    /// Open range-tombstone marker at the block boundary, if present:
93    /// `(local_deletion_time, marked_for_delete_at)`.
94    pub end_open_marker: Option<(i32, i64)>,
95}

97/// A fully decoded promoted index payload.
98#[derive(Debug, Clone, PartialEq, Eq)]
99pub struct DecodedPromotedIndex {
100    /// Data.db byte offset from partition start to the first row (`headerLength`).
101    pub header_length: u64,
102    /// Partition-level deletion time.
103    pub deletion_time: PromotedDeletionTime,
104    /// Number of IndexInfo blocks (== `entries.len()`).
105    pub count: u32,
106    /// Decoded IndexInfo blocks, in order.
107    pub entries: Vec<DecodedIndexInfo>,
108    /// Trailing offsets array: relative offset (bytes) from the first IndexInfo start
109    /// to each block's start. One entry per block.
110    pub offsets: Vec<i32>,
111}

113/// Callback returning the byte length of one serialized `ClusteringPrefix` at the
114/// start of `slice`.
115///
116/// Supplied by the caller from authoritative schema (clustering column types) — there
117/// is deliberately no heuristic fallback (Issue #28). Returns `Err` if the slice does
118/// not contain a complete prefix.
119pub type PrefixLen<'a> = dyn Fn(&[u8]) -> Result<usize> + 'a;

121/// Helper: read a big-endian `i32` from the front of `input`, returning the rest.
122fn take_be_i32(input: &[u8]) -> Result<(&[u8], i32)> {
123    if input.len() < 4 {
124        return Err(Error::Corruption(
125            "promoted index: truncated i32 (need 4 bytes)".to_string(),
126        ));
127    }
128    let (head, rest) = input.split_at(4);
129    let v = i32::from_be_bytes([head[0], head[1], head[2], head[3]]);
130    Ok((rest, v))
131}
132
133/// Helper: read a big-endian `i64` from the front of `input`, returning the rest.
134fn take_be_i64(input: &[u8]) -> Result<(&[u8], i64)> {
135    if input.len() < 8 {
136        return Err(Error::Corruption(
137            "promoted index: truncated i64 (need 8 bytes)".to_string(),
138        ));
139    }
140    let (head, rest) = input.split_at(8);
141    let mut buf = [0u8; 8];
142    buf.copy_from_slice(head);
143    Ok((rest, i64::from_be_bytes(buf)))
144}

146/// Decode one NB legacy `DeletionTime` (12 bytes).
147fn decode_deletion_time(input: &[u8]) -> Result<(&[u8], PromotedDeletionTime)> {
148    let (input, ldt) = take_be_i32(input)?;
149    let (input, mfda) = take_be_i64(input)?;
150    Ok((
151        input,
152        PromotedDeletionTime {
153            local_deletion_time: ldt,
154            marked_for_delete_at: mfda,
155        },
156    ))
157}

159/// Decode an unsigned VInt, mapping nom errors to a defensive `Error::Corruption`.
160fn decode_vuint(input: &[u8], what: &str) -> Result<(usize, u64)> {
161    match parse_vuint(input) {
162        Ok((rest, value)) => Ok((input.len() - rest.len(), value)),
163        Err(_) => Err(Error::Corruption(format!(
164            "promoted index: truncated/invalid VInt while reading {what}"
165        ))),
166    }
167}

169/// Decode a single `IndexInfo` block from the front of `input`.
170///
171/// `prefix_len` splits `firstName`/`lastName` using authoritative schema.
172fn decode_index_info<'a>(
173    input: &'a [u8],
174    prefix_len: &PrefixLen<'_>,
175) -> Result<(&'a [u8], DecodedIndexInfo)> {
176    // firstName ClusteringPrefix
177    let first_len = prefix_len(input)?;
178    if input.len() < first_len {
179        return Err(Error::Corruption(
180            "promoted index: firstName clustering prefix exceeds block".to_string(),
181        ));
182    }
183    let (first_name, input) = input.split_at(first_len);
184
185    // lastName ClusteringPrefix
186    let last_len = prefix_len(input)?;
187    if input.len() < last_len {
188        return Err(Error::Corruption(
189            "promoted index: lastName clustering prefix exceeds block".to_string(),
190        ));
191    }
192    let (last_name, input) = input.split_at(last_len);
193
194    // offset (unsigned VInt)
195    let (consumed, offset) = decode_vuint(input, "IndexInfo.offset")?;
196    let input = &input[consumed..];
197
198    // width delta (signed VInt = zigzag) → actual width
199    let (consumed, width_delta_u) = decode_vuint(input, "IndexInfo.width")?;
200    let input = &input[consumed..];
201    let width = INDEX_INFO_WIDTH_BASE
202        .checked_add(zigzag_decode(width_delta_u))
203        .filter(|w| *w >= 0)
204        .ok_or_else(|| {
205            Error::Corruption("promoted index: IndexInfo.width out of range".to_string())
206        })? as u64;
207
208    // endOpenMarker presence byte, optionally followed by a DeletionTime.
209    // NOTE (oracle coverage): the `0x01` (open-marker-present) branch below is
210    // validated by hand-built byte vectors only — the CQLite writer currently
211    // emits `0x00` for every block, and no committed Cassandra fixture yet places
212    // an open range-tombstone marker exactly at a promoted-index block boundary.
213    // A real-fixture assertion of the `0x01` branch is tracked for follow-up.
214    let (&marker_byte, input) = input
215        .split_first()
216        .ok_or_else(|| Error::Corruption("promoted index: missing endOpenMarker".to_string()))?;
217    let (input, end_open_marker) = match marker_byte {
218        0 => (input, None),
219        1 => {
220            let (rest, dt) = decode_deletion_time(input)?;
221            (
222                rest,
223                Some((dt.local_deletion_time, dt.marked_for_delete_at)),
224            )
225        }
226        other => {
227            return Err(Error::Corruption(format!(
228                "promoted index: invalid endOpenMarker byte {other:#x} (expected 0 or 1)"
229            )))
230        }
231    };
232
233    Ok((
234        input,
235        DecodedIndexInfo {
236            first_name: first_name.to_vec(),
237            last_name: last_name.to_vec(),
238            offset,
239            width,
240            end_open_marker,
241        },
242    ))
243}

245/// Decode a complete promoted index payload (the bytes that follow the
246/// `promoted_index_size` VInt in a BIG Index.db entry).
247///
248/// `prefix_len` returns the byte length of one serialized `ClusteringPrefix`,
249/// derived from authoritative schema (clustering column types). It is required
250/// because `firstName`/`lastName` are not self-delimiting without type info.
251///
252/// Returns `Err` (never panics) on any truncation or inconsistency.
253pub fn decode_promoted_index(
254    payload: &[u8],
255    prefix_len: &PrefixLen<'_>,
256) -> Result<DecodedPromotedIndex> {
257    // headerLength (unsigned VInt)
258    let (consumed, header_length) = decode_vuint(payload, "headerLength")?;
259    let input = &payload[consumed..];
260
261    // DeletionTime (12 bytes)
262    if input.len() < NB_DELETION_TIME_SIZE {
263        return Err(Error::Corruption(
264            "promoted index: truncated partition DeletionTime".to_string(),
265        ));
266    }
267    let (input, deletion_time) = decode_deletion_time(input)?;
268
269    // count (unsigned VInt)
270    let (consumed, count_u64) = decode_vuint(input, "count")?;
271    let input = &input[consumed..];
272    let count = u32::try_from(count_u64)
273        .map_err(|_| Error::Corruption("promoted index: count too large".to_string()))?;
274
275    // Bound `count` against the remaining payload BEFORE allocating: `count` is
276    // read from untrusted on-disk bytes, so a corrupt Index.db could declare a
277    // huge value and trigger a multi-hundred-GB `Vec::with_capacity` that aborts
278    // the process via `handle_alloc_error` (it would never reach the loop that
279    // returns Err). Every IndexInfo block consumes at least 1 byte for its
280    // payload AND contributes a 4-byte trailing offset entry, so a payload with
281    // `remaining` bytes can hold at most `remaining` blocks. `count > remaining`
282    // is therefore provably corrupt — fail closed instead of allocating. This
283    // upholds the module's "Returns Err (never panics)" guarantee for untrusted
284    // input.
285    if count as usize > input.len() {
286        return Err(Error::Corruption(format!(
287            "promoted index: declared block count {count} exceeds remaining payload \
288             length {} bytes (corrupt: each block needs >= 1 byte plus a 4-byte \
289             trailing offset entry)",
290            input.len()
291        )));
292    }
293
294    // IndexInfo[count] blocks.
295    let mut entries = Vec::with_capacity(count as usize);
296    let mut info = input;
297    for i in 0..count {
298        let (rest, entry) = decode_index_info(info, prefix_len)
299            .map_err(|e| Error::Corruption(format!("promoted index: IndexInfo block {i}: {e}")))?;
300        debug_assert!(rest.len() < info.len(), "IndexInfo decode must advance");
301        entries.push(entry);
302        info = rest;
303    }
304
305    // Trailing offsets array: one i32 BE per block.
306    let mut offsets = Vec::with_capacity(count as usize);
307    let mut rest = info;
308    for i in 0..count {
309        let (next, off) = take_be_i32(rest)
310            .map_err(|e| Error::Corruption(format!("promoted index: offsets[{i}]: {e}")))?;
311        offsets.push(off);
312        rest = next;
313    }
314
315    if !rest.is_empty() {
316        return Err(Error::Corruption(format!(
317            "promoted index: {} trailing bytes after offsets array",
318            rest.len()
319        )));
320    }
321
322    Ok(DecodedPromotedIndex {
323        header_length,
324        deletion_time,
325        count,
326        entries,
327        offsets,
328    })
329}

331/// Read just the IndexInfo block `count` from a promoted-index payload without
332/// decoding the (schema-dependent) `firstName`/`lastName` prefixes.
333///
334/// The leading `headerLength` (VInt), `DeletionTime` (12 bytes) and `count` (VInt)
335/// are schema-free, so the count is always recoverable. Returns `Err` on a
336/// truncated/invalid header (never panics).
337pub fn peek_block_count(payload: &[u8]) -> Result<u32> {
338    let (consumed, _header_length) = decode_vuint(payload, "headerLength")?;
339    let input = &payload[consumed..];
340    if input.len() < NB_DELETION_TIME_SIZE {
341        return Err(Error::Corruption(
342            "promoted index: truncated DeletionTime in peek_block_count".to_string(),
343        ));
344    }
345    let input = &input[NB_DELETION_TIME_SIZE..];
346    let (consumed, count_u64) = decode_vuint(input, "count")?;
347    let count = u32::try_from(count_u64)
348        .map_err(|_| Error::Corruption("promoted index: count too large".to_string()))?;
349
350    // Apply the SAME untrusted-`count` sanity bound as `decode_promoted_index`: the
351    // count is read straight from on-disk Index.db bytes, so a corrupt payload could
352    // declare an implausible value. Every IndexInfo block consumes at least 1 byte of
353    // body plus a 4-byte trailing offset, so the remaining payload can hold at most
354    // `remaining` blocks. Surfacing a bogus `count` here would otherwise leak into
355    // `block_count()` callers (stats `total_promoted_entries`, the parity validator)
356    // even though `decode_promoted_index` would reject the same payload. Fail closed
357    // to keep the trust posture consistent (still never panics).
358    let remaining = &input[consumed..];
359    if count as usize > remaining.len() {
360        return Err(Error::Corruption(format!(
361            "promoted index: declared block count {count} exceeds remaining payload \
362             length {} bytes (corrupt: each block needs >= 1 byte plus a 4-byte \
363             trailing offset entry)",
364            remaining.len()
365        )));
366    }
367    Ok(count)
368}

375    /// Test-local unsigned VInt encoder (ungated; the production `encode_unsigned`
376    /// lives behind the `write-support` feature). `encode_vuint` is the inverse of
377    /// `parse_vuint`, which is exactly what the decoder uses.
378    fn encode_unsigned(value: u64, buf: &mut Vec<u8>) {
379        buf.extend_from_slice(&encode_vuint(value));
380    }
381
382    /// Test-local signed VInt encoder: zigzag, then unsigned VInt — byte-identical
383    /// to the writer's `encode_signed`, but without the write-support gate.
384    fn encode_signed(value: i64, buf: &mut Vec<u8>) {
385        let zz = ((value << 1) ^ (value >> 63)) as u64;
386        encode_unsigned(zz, buf);
387    }
388
389    /// A prefix-length callback that mirrors the test fixtures' clustering prefix
390    /// encoding: a 1-byte header VInt (`0x00` = single present column / no columns),
391    /// followed by a fixed number of value bytes.
392    ///
393    /// `value_bytes` is the count of value bytes following the header byte. This is
394    /// the authoritative, schema-derived length the production caller would compute.
395    fn fixed_prefix_len(value_bytes: usize) -> impl Fn(&[u8]) -> Result<usize> {
396        move |slice: &[u8]| {
397            let need = 1 + value_bytes;
398            if slice.len() < need {
399                return Err(Error::Corruption(
400                    "test prefix_len: slice shorter than prefix".to_string(),
401                ));
402            }
403            Ok(need)
404        }
405    }

407    fn ck(value_bytes: &[u8]) -> Vec<u8> {
408        let mut v = vec![0x00u8]; // header VInt: all present
409        v.extend_from_slice(value_bytes);
410        v
411    }

413    /// Hand-build a promoted-index payload (no writer dependency) so the decoder can
414    /// be exercised in the minimal feature set. Mirrors `serialize_promoted_index`.
415    /// Each block tuple = (first_name, last_name, offset, width, end_open_marker).
416    #[allow(clippy::type_complexity)]
417    fn build_payload(
418        header_length: u64,
419        blocks: &[(Vec<u8>, Vec<u8>, u64, u64, Option<(i32, i64)>)],
420    ) -> Vec<u8> {
421        let mut info: Vec<u8> = Vec::new();
422        let mut starts: Vec<u32> = Vec::new();
423        for (first, last, offset, width, marker) in blocks {
424            starts.push(info.len() as u32);
425            info.extend_from_slice(first);
426            info.extend_from_slice(last);
427            encode_unsigned(*offset, &mut info);
428            encode_signed((*width as i64) - (64 * 1024), &mut info);
429            match marker {
430                None => info.push(0x00),
431                Some((ldt, mfda)) => {
432                    info.push(0x01);
433                    info.extend_from_slice(&ldt.to_be_bytes());
434                    info.extend_from_slice(&mfda.to_be_bytes());
435                }
436            }
437        }
438        let mut payload = Vec::new();
439        encode_unsigned(header_length, &mut payload);
440        payload.extend_from_slice(&i32::MAX.to_be_bytes()); // LIVE ldt
441        payload.extend_from_slice(&i64::MIN.to_be_bytes()); // LIVE mfda
442        encode_unsigned(blocks.len() as u64, &mut payload);
443        payload.extend_from_slice(&info);
444        for s in &starts {
445            payload.extend_from_slice(&(*s as i32).to_be_bytes());
446        }
447        payload
448    }

659        /// Encode with the authoritative writer, decode, assert byte-exact recovery.
660        #[test]
661        fn test_writer_roundtrip_two_blocks_varied_widths() {
662            let block1 = PromotedIndexBlock {
663                first_name: ck(b"aa"),
664                last_name: ck(b"am"),
665                offset: 0,
666                width: 50_000, // < 64 KiB → negative width delta
667                oss50_separator: None,
668            };
669            let block2 = PromotedIndexBlock {
670                first_name: ck(b"an"),
671                last_name: ck(b"zz"),
672                offset: 70_000,
673                width: 100_000, // > 64 KiB → positive width delta
674                oss50_separator: None,
675            };
676            // raw_key_len = 4 → headerLength = 2 + 4 + 12 = 18
677            let payload = serialize_promoted_index_for_test(&[block1.clone(), block2.clone()], 4);
678            let decoded = decode_promoted_index(&payload, &fixed_prefix_len(2)).unwrap();
679
680            assert_eq!(decoded.header_length, 18);
681            assert!(decoded.deletion_time.is_live());
682            assert_eq!(decoded.count, 2);
683
684            assert_eq!(decoded.entries[0].first_name, block1.first_name);
685            assert_eq!(decoded.entries[0].last_name, block1.last_name);
686            assert_eq!(decoded.entries[0].offset, block1.offset);
687            assert_eq!(decoded.entries[0].width, block1.width);
688            assert_eq!(decoded.entries[0].end_open_marker, None);
689
690            assert_eq!(decoded.entries[1].first_name, block2.first_name);
691            assert_eq!(decoded.entries[1].last_name, block2.last_name);
692            assert_eq!(decoded.entries[1].offset, block2.offset);
693            assert_eq!(decoded.entries[1].width, block2.width);
694
695            // Offsets monotonic, first is 0.
696            assert_eq!(decoded.offsets[0], 0);
697            assert!(decoded.offsets[1] > decoded.offsets[0]);
698        }
699
700        /// Single-block input → writer emits no promoted index (None); modeled here
701        /// as an empty payload that decodes to a count-0 / no-blocks structure when
702        /// promoted_len == 0 is handled by the caller (parse_big_index_entry).
703        #[test]
704        fn test_writer_roundtrip_width_exactly_base() {
705            let block1 = PromotedIndexBlock {
706                first_name: ck(b""),
707                last_name: ck(b""),
708                offset: 0,
709                width: 64 * 1024,
710                oss50_separator: None,

715                offset: 64 * 1024,
716                width: 64 * 1024,
717                oss50_separator: None,
718            };
719            let payload = serialize_promoted_index_for_test(&[block1, block2], 4);
720            let decoded = decode_promoted_index(&payload, &fixed_prefix_len(0)).unwrap();
721            assert_eq!(decoded.entries[0].width, 64 * 1024);
722            assert_eq!(decoded.entries[1].width, 64 * 1024);
723        }
