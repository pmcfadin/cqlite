1//! Cassandra OSS50 **byte-comparable** clustering-bound encoding.
2//!
3//! These encoders reproduce the on-disk separator bytes the `Rows.db` row-index
4//! trie stores (the SAME encoding), NOT CQLite's custom
5//! [`ByteComparableEncoder`](crate::storage::sstable::bti::encoder::ByteComparableEncoder)
6//! (which prepends a 1-byte type discriminator and so is byte-incompatible with
7//! the on-disk trie).
8
9use crate::{error::Error, storage::sstable::bti::node::BtiResult, types::Value};
10
11/// Cassandra component separator used between clustering components in the
12/// OSS50 byte-comparable form.  Mirrors `ByteSource.NEXT_COMPONENT` (value
13/// `0x40`) emitted by `ClusteringComparator.asByteComparable` /
14/// `ByteSource.withTerminator`/`ByteSource.of(...)` in
15/// `org.apache.cassandra.utils.bytecomparable.ByteSource` (cassandra-5.0.0).
16const OSS50_NEXT_COMPONENT: u8 = 0x40;
17
18/// OSS50 variable-length byte-comparable encoding for `BytesType`/`UTF8Type`
19/// (Cassandra `ByteSource`): every literal `0x00` byte is escaped as
20/// `0x00 0xFE` (`ESCAPE` + `ESCAPED_0_CONT`) and the component is terminated
21/// with `0x00 0xFF` (`ESCAPE` + `ESCAPED_0_DONE`).  This makes the encoding
22/// weakly prefix-free, so a shorter value sorts before a longer value that
23/// shares its prefix — matching the separators stored in a real `Rows.db`
24/// trie.  (The project's `ByteComparableEncoder` uses a different,
25/// non-Cassandra escape scheme and must NOT be used for trie-compatible bounds.)
26fn encode_varlen_oss50(bytes: &[u8], out: &mut Vec<u8>) {
27    for &b in bytes {
28        out.push(b);
29        if b == 0x00 {
30            out.push(0xFE);
31        }
32    }
33    out.push(0x00);
34    out.push(0xFF);
35}

37/// Encode a single clustering-component [`Value`] in Cassandra OSS50
38/// **byte-comparable** form (the SAME encoding the `Rows.db` row-index trie
39/// stores its separators in), appending to `out`.
40///
41/// This reproduces the per-type `AbstractType.asComparableBytes(...)` production
42/// used by Cassandra to build the byte-comparable keys:
43///
44/// - `Int32Type` → `(v as u32 ^ 0x8000_0000)` big-endian (sign-flip, 4B).
45/// - `LongType` → `(v as u64 ^ 0x8000_0000_0000_0000)` big-endian (8B).
46/// - `ShortType`/`ByteType` (`smallint`/`tinyint`) → sign-flip big-endian.
47/// - `BooleanType` → single byte `0x00`/`0x01`.
48/// - `TimestampType` (`timestamp`) → `LongType`-style sign-flip 8B.
49/// - `UUIDType`/`TimeUUIDType` → the 16 raw bytes.
50/// - `UTF8Type`/`AsciiType` and `BytesType` (`blob`) → raw value bytes,
51///   terminated by the OSS50 variable-length component terminator.
52///
53/// Any clustering type not enumerated here returns an explicit parse error
54/// (NO silent wrong results — issue #28 no-heuristics mandate).
55fn encode_clustering_component_oss50(value: &Value, out: &mut Vec<u8>) -> BtiResult<()> {
56    match value {
57        // int — Int32Type: sign-flip, big-endian (matches `wide_table` separators,
58        // e.g. ck=8 → 80 00 00 08).
59        Value::Integer(v) => {
60            out.extend_from_slice(&((*v as u32) ^ 0x8000_0000).to_be_bytes());
61            Ok(())
62        }
63        // bigint — LongType: sign-flip, big-endian, 8 bytes.
64        Value::BigInt(v) | Value::Counter(v) => {
65            out.extend_from_slice(&((*v as u64) ^ 0x8000_0000_0000_0000).to_be_bytes());
66            Ok(())
67        }
68        // smallint — ShortType: sign-flip, big-endian, 2 bytes.
69        Value::SmallInt(v) => {
70            out.extend_from_slice(&((*v as u16) ^ 0x8000).to_be_bytes());
71            Ok(())
72        }
73        // tinyint — ByteType: sign-flip, 1 byte.
74        Value::TinyInt(v) => {
75            out.push((*v as u8) ^ 0x80);
76            Ok(())
77        }
78        // boolean — BooleanType: single 0x00/0x01 byte.
79        Value::Boolean(b) => {
80            out.push(if *b { 0x01 } else { 0x00 });
81            Ok(())
82        }
83        // timestamp — TimestampType shares LongType's comparable form (8-byte
84        // sign-flip big-endian of the millisecond value).
85        Value::Timestamp(v) => {
86            out.extend_from_slice(&((*v as u64) ^ 0x8000_0000_0000_0000).to_be_bytes());
87            Ok(())
88        }
89        // uuid / timeuuid — raw 16 bytes (already byte-comparable in the on-disk
90        // separator form for this fixed-length type).
91        Value::Uuid(bytes) => {
92            out.extend_from_slice(bytes);
93            Ok(())
94        }
95        // text / ascii — OSS50 variable-length byte-comparable encoding.
96        Value::Text(s) => {
97            encode_varlen_oss50(s.as_bytes(), out);
98            Ok(())
99        }
100        // blob / inet — OSS50 variable-length byte-comparable encoding.
101        Value::Blob(b) | Value::Inet(b) => {
102            encode_varlen_oss50(b, out);
103            Ok(())
104        }
105        other => Err(Error::Parse(format!(
106            "BTI range_query: byte-comparable encoding not implemented for {:?}",
107            other.data_type()
108        ))),
109    }
110}

112/// Encode a multi-component clustering bound (`&[Value]`) in Cassandra OSS50
113/// byte-comparable form — the SAME encoding the `Rows.db` trie stores.
114///
115/// A single-component clustering encodes to the bare component bytes (matching
116/// the `wide_table` fixture separators, e.g. ck=8 → `80 00 00 08`, NO framing).
117/// Multi-component clusterings concatenate per-component byte-comparable
118/// encodings separated by [`OSS50_NEXT_COMPONENT`] (`ByteSource.NEXT_COMPONENT`,
119/// per `ClusteringComparator.asByteComparable`), with no leading/trailing frame
120/// so a prefix bound sorts before any longer key sharing it.
121pub fn encode_clustering_bound_oss50(values: &[Value]) -> BtiResult<Vec<u8>> {
122    // All-ascending convenience: every component uses its base byte-comparable
123    // form. Equivalent to `encode_clustering_bound_oss50_with_order` with all
124    // `is_reversed = false`.
125    let mut out = Vec::new();
126    for (i, v) in values.iter().enumerate() {
127        if i > 0 {
128            out.push(OSS50_NEXT_COMPONENT);
129        }
130        encode_clustering_component_oss50(v, &mut out)?;
131    }
132    Ok(out)
133}

135/// Encode a multi-component clustering bound applying each column's clustering
136/// **ORDER** (ASC/DESC), producing Cassandra OSS50 byte-comparable separators
137/// that sort in the SAME order the rows are physically written.
138///
139/// ## Why DESC must invert the component bytes
140///
141/// Cassandra wraps a `CLUSTERING ORDER BY (c DESC)` column's type in
142/// `ReversedType`. `ReversedType.asComparableBytes` delegates to the base type's
143/// byte-comparable production and then inverts the resulting `ByteSource` via
144/// `ByteSource.invert(...)`, which complements every emitted data byte
145/// (`b -> 0xFF ^ b`). Complementing every byte of a (weakly prefix-free)
146/// byte-comparable encoding reverses its lexicographic order, so a DESCENDING
147/// value order maps to an ASCENDING byte order. This is exactly what a `Rows.db`
148/// row-index trie needs: separators are always stored in ascending *byte* order,
149/// while the underlying clustering values run descending for a DESC column.
150/// (cassandra-5.0.0 `org.apache.cassandra.db.marshal.ReversedType.asComparableBytes`
151/// + `org.apache.cassandra.utils.bytecomparable.ByteSource.invert`.)
152///
153/// The inter-component framing byte ([`OSS50_NEXT_COMPONENT`], `0x40`) is emitted
154/// by the *comparator* (`ClusteringComparator.asByteComparable`), NOT by a
155/// component's type, so it is **not** inverted even when neighbouring components
156/// are DESC — only the per-component byte-comparable bytes are complemented. This
157/// matches Cassandra, where each `subtype(i)` (possibly a `ReversedType`) emits
158/// its own (already inverted) byte source and the comparator separates them with
159/// the un-inverted `NEXT_COMPONENT` byte.
160///
161/// `is_reversed[i]` MUST correspond positionally to `values[i]` (i.e. the schema
162/// clustering-key order). A short `is_reversed` slice treats missing entries as
163/// ascending (`false`).
164pub fn encode_clustering_bound_oss50_with_order(
165    values: &[Value],
166    is_reversed: &[bool],
167) -> BtiResult<Vec<u8>> {
168    let mut out = Vec::new();
169    for (i, v) in values.iter().enumerate() {
170        if i > 0 {
171            out.push(OSS50_NEXT_COMPONENT);
172        }
173        if is_reversed.get(i).copied().unwrap_or(false) {
174            // Encode the base component into a scratch buffer, then complement
175            // every byte (ReversedType / ByteSource.invert).
176            let mut scratch = Vec::new();
177            encode_clustering_component_oss50(v, &mut scratch)?;
178            for b in &scratch {
179                out.push(0xFF ^ *b);
180            }
181        } else {
182            encode_clustering_component_oss50(v, &mut out)?;
183        }
184    }
185    Ok(out)
186}

188#[cfg(test)]
189mod tests {
190    use super::*;
191
192    /// The OSS50 byte-comparable clustering encoder (issue #832 Finding 1)
193    /// reproduces the on-disk trie separator bytes.
194    #[test]
195    fn oss50_clustering_encoder() {
196        // Single int component — bare, sign-flip big-endian (matches fixture).
197        assert_eq!(
198            encode_clustering_bound_oss50(&[Value::Integer(8)]).unwrap(),
199            vec![0x80, 0x00, 0x00, 0x08]
200        );
201        assert_eq!(
202            encode_clustering_bound_oss50(&[Value::Integer(-1)]).unwrap(),
203            vec![0x7F, 0xFF, 0xFF, 0xFF]
204        );
205        // Ordering is preserved: -1 < 0 < 100 byte-comparably.
206        let neg = encode_clustering_bound_oss50(&[Value::Integer(-1)]).unwrap();
207        let zero = encode_clustering_bound_oss50(&[Value::Integer(0)]).unwrap();
208        let pos = encode_clustering_bound_oss50(&[Value::Integer(100)]).unwrap();
209        assert!(neg < zero && zero < pos);
210
211        // bigint — 8-byte sign-flip BE.
212        assert_eq!(
213            encode_clustering_bound_oss50(&[Value::BigInt(1)]).unwrap(),
214            vec![0x80, 0, 0, 0, 0, 0, 0, 0x01]
215        );
216
217        // Multi-component: int(1) + text("ab") joined with 0x40 NEXT_COMPONENT,
218        // text terminated by the OSS50 end marker 0x00 0xFF.
219        assert_eq!(
220            encode_clustering_bound_oss50(&[Value::Integer(1), Value::Text("ab".to_string())])
221                .unwrap(),
222            vec![0x80, 0x00, 0x00, 0x01, 0x40, b'a', b'b', 0x00, 0xFF]
223        );
224
225        // Variable-length OSS50: text terminates with 0x00 0xFF; a literal 0x00
226        // byte is escaped as 0x00 0xFE so it never collides with the terminator.
227        assert_eq!(
228            encode_clustering_bound_oss50(&[Value::Text("a".to_string())]).unwrap(),
229            vec![b'a', 0x00, 0xFF]
230        );
231        assert_eq!(
232            encode_clustering_bound_oss50(&[Value::Blob(vec![0x01, 0x00, 0x02])]).unwrap(),
233            vec![0x01, 0x00, 0xFE, 0x02, 0x00, 0xFF]
234        );
235        // Prefix-free ordering: "a" sorts before "ab".
236        let a = encode_clustering_bound_oss50(&[Value::Text("a".to_string())]).unwrap();
237        let ab = encode_clustering_bound_oss50(&[Value::Text("ab".to_string())]).unwrap();
238        assert!(a < ab);
239
240        // Unsupported clustering type errors out explicitly.
241        assert!(encode_clustering_bound_oss50(&[Value::Float(1.0)]).is_err());
242    }
243
244    /// Order-aware OSS50 encoder (DESC clustering): a `CLUSTERING ORDER BY DESC`
245    /// column wraps in Cassandra `ReversedType`, whose `asComparableBytes`
246    /// complements every byte of the base byte-comparable form.
247    #[test]
248    fn oss50_clustering_encoder_reversed_order() {
249        // DESC int(8): base 80 00 00 08, complemented -> 7F FF FF F7.
250        assert_eq!(
251            encode_clustering_bound_oss50_with_order(&[Value::Integer(8)], &[true]).unwrap(),
252            vec![0x7F, 0xFF, 0xFF, 0xF7]
253        );
254        // ASC path is identical to the plain encoder.
255        assert_eq!(
256            encode_clustering_bound_oss50_with_order(&[Value::Integer(8)], &[false]).unwrap(),
257            encode_clustering_bound_oss50(&[Value::Integer(8)]).unwrap()
258        );
259
260        // The decisive property: for DESC, larger VALUE => smaller BYTES.
261        let enc = |v: i32| {
262            encode_clustering_bound_oss50_with_order(&[Value::Integer(v)], &[true]).unwrap()
263        };
264        // Physical write order for DESC: 5,4,3,2,1,0. Their separator bytes must
265        // be strictly ASCENDING in that same sequence.
266        let write_order = [5, 4, 3, 2, 1, 0];
267        for w in write_order.windows(2) {
268            assert!(
269                enc(w[0]) < enc(w[1]),
270                "DESC: value {} written before {} must yield smaller separator bytes",
271                w[0],
272                w[1]
273            );
274        }
275
276        // Mixed ASC/DESC: first column ASC int, second column DESC int.
277        let mixed = encode_clustering_bound_oss50_with_order(
278            &[Value::Integer(1), Value::Integer(8)],
279            &[false, true],
280        )
281        .unwrap();
282        assert_eq!(
283            mixed,
284            vec![0x80, 0x00, 0x00, 0x01, 0x40, 0x7F, 0xFF, 0xFF, 0xF7],
285            "ASC component bare, 0x40 framing un-inverted, DESC component complemented"
286        );
287
288        // Mixed-order monotonicity.
289        let m = |a: i32, b: i32| {
290            encode_clustering_bound_oss50_with_order(
291                &[Value::Integer(a), Value::Integer(b)],
292                &[false, true],
293            )
294            .unwrap()
295        };
296        // Same first key (1), second key descending 9,8,7 -> ascending bytes.
297        assert!(m(1, 9) < m(1, 8) && m(1, 8) < m(1, 7));
298        // Different first key dominates regardless of second (ASC leading).
299        assert!(m(1, 0) < m(2, 9));
300    }
