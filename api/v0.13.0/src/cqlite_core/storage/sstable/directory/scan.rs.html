11/// Extract table name from directory name (strips UUID suffix)
12pub(crate) fn extract_table_name(dir_name: &str) -> Result<String> {
13    // Directory format: "tablename-{32-char-uuid}"
14    // Find the last hyphen and take everything before it
15    if let Some(hyphen_pos) = dir_name.rfind('-') {
16        let table_name = &dir_name[..hyphen_pos];
17        if table_name.is_empty() {
18            return Err(Error::invalid_path(format!(
19                "Empty table name in directory: {}",
20                dir_name
21            )));
22        }
23        Ok(table_name.to_string())
24    } else {
25        // Fallback: use entire directory name if no UUID suffix
26        Ok(dir_name.to_string())
27    }
28}

30/// Scan directory for SSTable files and group by generation
31pub(crate) fn scan_sstable_files(path: &Path, table_name: &str) -> Result<Vec<SSTableGeneration>> {
32    let entries = fs::read_dir(path)
33        .map_err(|e| Error::storage(format!("Failed to read directory: {:?}: {}", path, e)))?;
34
35    let mut generations_map: HashMap<(u32, String), SSTableGeneration> = HashMap::new();
36    let mut found_files = 0;
37    let mut valid_sstable_files = 0;
38
39    for entry in entries {
40        let entry = entry?;
41        let file_path = entry.path();
42        found_files += 1;
43
44        if let Some(file_name) = file_path.file_name().and_then(|n| n.to_str()) {
45            // Enhanced validation: Check if file exists and is readable
46            if !file_path.is_file() {
47                continue; // Skip directories and non-files
48            }
49
50            // Check file accessibility
51            if let Err(e) = fs::metadata(&file_path) {
52                log::warn!("Cannot access file {:?}: {}", file_path, e);
53                continue;
54            }
55
56            if let Some((version, generation, format, component)) =
57                parse_sstable_filename(file_name)?
58            {
59                valid_sstable_files += 1;
60                let key = (generation, format.clone());
61
62                let generation_obj =
63                    generations_map
64                        .entry(key.clone())
65                        .or_insert_with(|| SSTableGeneration {
66                            version: version.clone(),
67                            generation,
68                            format,
69                            table_name: table_name.to_string(),
70                            components: HashMap::new(),
71                            base_path: path.to_path_buf(),
72                        });
73
74                generation_obj.components.insert(component, file_path);
75            }
76        }
77    }
78
79    // Enhanced validation and reporting
80    if found_files == 0 {
81        return Err(Error::not_found(format!(
82            "Directory appears to be empty: {:?}",
83            path
84        )));
85    }
86
87    if valid_sstable_files == 0 {
88        return Err(Error::invalid_format(format!(
89            "No valid SSTable files found in directory: {:?}. Found {} files total, but none match the expected SSTable naming pattern (e.g., nb-1-big-Data.db)",
90            path,
91            found_files
92        )));
93    }
94
95    // Sort generations by number (newest first)
96    let mut generations: Vec<SSTableGeneration> = generations_map.into_values().collect();
97    generations.sort_by(|a, b| b.generation.cmp(&a.generation));
98
99    // Log summary for debugging
100    log::debug!(
101        "Directory scan completed: {} total files, {} SSTable files, {} generations found",
102        found_files,
103        valid_sstable_files,
104        generations.len()
105    );
106
107    Ok(generations)
108}

110/// Parse SSTable filename to extract version, generation, format, and component.
111///
112/// Returns `Ok(Some((version, generation, format, component)))` for recognised
113/// SSTable filenames.  Returns `Ok(None)` for non-SSTable files (e.g. `.jmx`,
114/// unknown extensions) or files whose SSTable id is not a plain integer (UUID-
115/// based ids are not yet supported by the `u32` generation field; see the
116/// `SSTableGeneration.sstable_id` tracking item).
117///
118/// # Examples
119/// ```text
120/// "nb-1-big-Data.db"     -> Some(("nb", 1, "big",  Data))
121/// "da-2-bti-Rows.db"     -> Some(("da", 2, "bti",  Rows))
122/// "oa-1-big-Data.db"     -> Some(("oa", 1, "big",  Data))
123/// ".gitignore"           -> None
124/// ```
125///
126/// The version letter is extracted via [`SsTableDescriptor`] which also validates
127/// the filename structure and supports both sequential and UUID-based SSTable ids.
128/// UUID-based ids are silently skipped (returning `None`) because the current
129/// `SSTableGeneration.generation` field is `u32`.
130pub(crate) fn parse_sstable_filename(
131    filename: &str,
132) -> Result<Option<(String, u32, String, SSTableComponent)>> {
133    use std::path::Path;
134
135    // Use SsTableDescriptor for authoritative filename parsing (handles both
136    // sequential and UUID-based ids, validates version letter format).
137    let desc = match SsTableDescriptor::parse(Path::new(filename)) {
138        Ok(d) => d,
139        Err(_) => {
140            // Not a recognised SSTable filename (e.g. .gitignore, jmx files).
141            return Ok(None);
142        }
143    };
144
145    // The generation field is u32; UUID-based ids (e.g. "6aa08200a251…") cannot
146    // be coerced to u32.  Skip them silently — these files will still be picked
147    // up once SSTableGeneration is upgraded to carry a String sstable_id.
148    let generation: u32 = match desc.sstable_id.parse() {
149        Ok(n) => n,
150        Err(_) => {
151            log::debug!(
152                "parse_sstable_filename: skipping UUID-id file {} (not a plain integer generation)",
153                filename
154            );
155            return Ok(None);
156        }
157    };
158
159    let format = desc.format.as_str().to_string();
160    let version = desc.version.clone();
161
162    // Return None for unrecognized components instead of propagating error.
163    // This follows the same pattern as TOC parsing (toc.rs:53-73).
164    let component = match SSTableComponent::from_str(&desc.component) {
165        Ok(c) => c,
166        Err(_) => {
167            log::debug!(
168                "Ignoring file with unrecognized component extension: {}",
169                filename
170            );
171            return Ok(None);
172        }
173    };
174
175    Ok(Some((version, generation, format, component)))
176}

178/// Scan directory for secondary index subdirectories
179pub(crate) fn scan_secondary_indexes(path: &Path, table_name: &str) -> Result<Vec<SecondaryIndex>> {
180    let entries = fs::read_dir(path)
181        .map_err(|e| Error::storage(format!("Failed to read directory: {:?}: {}", path, e)))?;
182
183    let mut secondary_indexes = Vec::new();
184
185    for entry in entries {
186        let entry = entry?;
187        let entry_path = entry.path();
188
189        if entry_path.is_dir() {
190            if let Some(dir_name) = entry_path.file_name().and_then(|n| n.to_str()) {
191                // Check if this is a secondary index directory (starts with '.' and ends with '_idx')
192                if dir_name.starts_with('.') && dir_name.ends_with("_idx") {
193                    // Extract index name (e.g., ".users_metadata_idx" -> "metadata_idx")
194                    let index_name = dir_name[1..].to_string(); // Remove leading '.'
195
196                    // Validate that the index name matches the table
197                    let expected_prefix = format!("{}_", table_name);
198                    if index_name.starts_with(&expected_prefix) {
199                        // Scan SSTable files in the secondary index directory
200                        let index_generations = scan_sstable_files(&entry_path, table_name)?;
201
202                        if !index_generations.is_empty() {
203                            secondary_indexes.push(SecondaryIndex {
204                                index_name,
205                                index_path: entry_path,
206                                generations: index_generations,
207                            });
208                        }
209                    }
210                }
211            }
212        }
213    }
214
215    Ok(secondary_indexes)
