1//! Compressed Data.db writer - writes compressed partition data
2//!
3//! Generates compressed Data.db files with per-chunk compression.
4//! Supports LZ4, Snappy, Deflate, and Zstd compression algorithms.
5//!
6//! # Chunk Format
7//!
8//! CRITICAL: CRC32 is TRAILING (after chunk data), NOT leading!
9//!
10//! ```text
11//! [compressed_chunk_0_bytes][crc32: 4 bytes BE]
12//! [compressed_chunk_1_bytes][crc32: 4 bytes BE]
13//! ...
14//! ```
15//!
16//! # Compression Flow
17//!
18//! 1. Accumulate uncompressed data in buffer
19//! 2. When buffer reaches chunk_size, compress and write
20//! 3. Write CRC32 of compressed data (AFTER the compressed bytes)
21//! 4. Track chunk offsets for CompressionInfo.db
22//! 5. On finish(), flush remaining buffer
23//!
24//! # Claim boundary — BUILT-BUT-UNWIRED (issue #1406, posture b)
25//!
26//! This writer is NOT wired into CQLite's production write surface (flush +
27//! compaction via `SSTableWriter` emit uncompressed SSTables only). It exists to
28//! let read-path fixtures synthesize compressed SSTables for exercising the
29//! decompressing reader. Any attempt to emit a compressed SSTable as a real,
30//! parity-claimed artifact is fail-closed by
31//! [`super::compression_info_writer::CompressionInfoWriter::guard_unsupported_production_write`].
32//! See that module's docs and issue #1406.
33//!
34//! References:
35//! - Parser: `cqlite-core/src/storage/sstable/reader/compression.rs`
36//! - Format docs: `docs/sstables-definitive-guide/chapters/09-compression.md`
37

45/// Compressor trait for algorithm implementations
46pub trait Compressor: Send + Sync {
47    /// Compress input data
48    fn compress(&self, data: &[u8]) -> Result<Vec<u8>>;
49
50    /// Get the Cassandra algorithm name
51    fn algorithm_name(&self) -> &'static str;
52
53    /// Get the compression algorithm enum
54    fn algorithm(&self) -> CompressionAlgorithm;
55}

100/// Deflate compressor implementation
101#[cfg(feature = "deflate")]
102#[derive(Debug, Clone)]
103pub struct DeflateCompressor {
104    /// Compression level (0-9, higher = better compression, slower)
105    level: u32,
106}
107
108#[cfg(feature = "deflate")]
109impl Default for DeflateCompressor {
110    fn default() -> Self {
111        Self { level: 6 } // Default compression level
112    }
113}
114
115#[cfg(feature = "deflate")]
116impl DeflateCompressor {
117    /// Create with custom compression level
118    pub fn with_level(level: u32) -> Self {
119        Self {
120            level: level.min(9),
121        }
122    }
123}
124
125#[cfg(feature = "deflate")]
126impl Compressor for DeflateCompressor {
127    fn compress(&self, data: &[u8]) -> Result<Vec<u8>> {
128        use flate2::write::ZlibEncoder;
129        use flate2::Compression;
130
131        // Cassandra's DeflateCompressor uses java.util.zip.Deflater, which emits a
132        // ZLIB-wrapped stream (2-byte header + DEFLATE body + Adler-32 trailer),
133        // NOT raw DEFLATE. Match it with ZlibEncoder so a CQLite-written chunk reads
134        // back through the zlib-aware decode path (#1082).
135        let mut encoder = ZlibEncoder::new(Vec::new(), Compression::new(self.level));
136        encoder
137            .write_all(data)
138            .map_err(|e| Error::Storage(format!("Deflate compression write failed: {}", e)))?;
139        encoder
140            .finish()
141            .map_err(|e| Error::Storage(format!("Deflate compression finish failed: {}", e)))
142    }
143
144    fn algorithm_name(&self) -> &'static str {
145        "DeflateCompressor"
146    }
147
148    fn algorithm(&self) -> CompressionAlgorithm {
149        CompressionAlgorithm::Deflate
150    }
151}

212/// Create a compressor for the given algorithm
213pub fn create_compressor(algorithm: CompressionAlgorithm) -> Result<Box<dyn Compressor>> {
214    match algorithm {
215        #[cfg(feature = "lz4")]
216        CompressionAlgorithm::Lz4 => Ok(Box::new(Lz4Compressor)),
217        #[cfg(not(feature = "lz4"))]
218        CompressionAlgorithm::Lz4 => Err(Error::InvalidInput(
219            "LZ4 compression not enabled (feature 'lz4' required)".to_string(),
220        )),
221
222        #[cfg(feature = "snappy")]
223        CompressionAlgorithm::Snappy => Ok(Box::new(SnappyCompressor)),
224        #[cfg(not(feature = "snappy"))]
225        CompressionAlgorithm::Snappy => Err(Error::InvalidInput(
226            "Snappy compression not enabled (feature 'snappy' required)".to_string(),
227        )),
228
229        #[cfg(feature = "deflate")]
230        CompressionAlgorithm::Deflate => Ok(Box::new(DeflateCompressor::default())),
231        #[cfg(not(feature = "deflate"))]
232        CompressionAlgorithm::Deflate => Err(Error::InvalidInput(
233            "Deflate compression not enabled (feature 'deflate' required)".to_string(),
234        )),
235
236        #[cfg(feature = "zstd")]
237        CompressionAlgorithm::Zstd => Ok(Box::new(ZstdCompressor::default())),
238        #[cfg(not(feature = "zstd"))]
239        CompressionAlgorithm::Zstd => Err(Error::InvalidInput(
240            "Zstd compression not enabled (feature 'zstd' required)".to_string(),
241        )),
242
243        CompressionAlgorithm::None => Ok(Box::new(NoopCompressor)),
244    }
245}

247/// Compressed Data.db writer
248///
249/// Accumulates uncompressed data in chunks, compresses each chunk, and writes
250/// each chunk record to the output as: [compressed_bytes][4-byte inline CRC32].
251///
252/// The CRC32 is computed over the compressed bytes and stored inline in Data.db —
253/// it is NOT stored in CompressionInfo.db. This matches Cassandra's
254/// CompressedSequentialWriter.java:192.
255pub struct CompressedDataWriter {
256    /// Output buffer for compressed data
257    output: Vec<u8>,
258    /// Compressor implementation
259    compressor: Box<dyn Compressor>,
260    /// Uncompressed chunk size (default 65536)
261    chunk_size: usize,
262    /// Buffer for accumulating uncompressed data
263    buffer: Vec<u8>,
264    /// Byte offset of each compressed-chunk record in output (for CompressionInfo.db)
265    chunk_offsets: Vec<u64>,
266    /// Total uncompressed bytes written (for CompressionInfo.db data_length)
267    uncompressed_length: u64,
268    /// Current write position in output
269    position: u64,
270}
271
272impl CompressedDataWriter {
273    /// Default chunk size (64KB)
274    pub const DEFAULT_CHUNK_SIZE: usize = 65536;
275
276    /// Create a new compressed data writer
277    pub fn new(compressor: Box<dyn Compressor>) -> Self {
278        Self::with_chunk_size(compressor, Self::DEFAULT_CHUNK_SIZE)
279    }
280
281    /// Create with custom chunk size
282    pub fn with_chunk_size(compressor: Box<dyn Compressor>, chunk_size: usize) -> Self {
283        Self {
284            output: Vec::new(),
285            compressor,
286            chunk_size,
287            buffer: Vec::with_capacity(chunk_size),
288            chunk_offsets: Vec::new(),
289            uncompressed_length: 0,
290            position: 0,
291        }
292    }
293
294    /// Map the active compressor to a bounded [`catalog::attr::COMPRESSION`]
295    /// value (issue #1036). Bounded set: lz4/snappy/deflate/zstd/none.
296    fn compression_attr(&self) -> &'static str {
297        match self.compressor.algorithm() {
298            CompressionAlgorithm::Lz4 => "lz4",
299            CompressionAlgorithm::Snappy => "snappy",
300            CompressionAlgorithm::Deflate => "deflate",
301            CompressionAlgorithm::Zstd => "zstd",
302            CompressionAlgorithm::None => "none",
303        }
304    }
305
306    /// Write uncompressed data
307    ///
308    /// Data is buffered until chunk_size is reached, then compressed and written.
309    #[tracing::instrument(name = "compression.write_chunk", skip(self, data))]
310    pub fn write(&mut self, data: &[u8]) -> Result<()> {
311        let mut remaining = data;
312
313        while !remaining.is_empty() {
314            // Fill buffer up to chunk_size
315            let space = self.chunk_size - self.buffer.len();
316            let take = remaining.len().min(space);
317
318            self.buffer.extend_from_slice(&remaining[..take]);
319            remaining = &remaining[take..];
320
321            // If buffer is full, flush it
322            if self.buffer.len() >= self.chunk_size {
323                self.flush_chunk()?;
324            }
325        }
326
327        Ok(())
328    }
329
330    /// Flush the current buffer as a compressed chunk
331    fn flush_chunk(&mut self) -> Result<()> {
332        if self.buffer.is_empty() {
333            return Ok(());
334        }
335
336        // Record chunk offset BEFORE writing (points to start of chunk record)
337        self.chunk_offsets.push(self.position);
338        let uncompressed_len = self.buffer.len();
339        self.uncompressed_length += uncompressed_len as u64;
340
341        // Compress the buffer
342        let compressed = self.compressor.compress(&self.buffer)?;
343
344        // Per-chunk compression ratio (issue #1036): compressed/uncompressed,
345        // tagged with the bounded algorithm attribute. Guard against a zero
346        // divisor (flush_chunk early-returns on an empty buffer, so this is just
347        // defensive).
348        if uncompressed_len > 0 {
349            crate::observability::record_histogram(
350                crate::observability::catalog::COMPRESSION_RATIO,
351                compressed.len() as f64 / uncompressed_len as f64,
352                &[(
353                    crate::observability::catalog::attr::COMPRESSION,
354                    self.compression_attr().into(),
355                )],
356            );
357        }
358
359        // Compute CRC32 of compressed data — stored INLINE in Data.db after compressed bytes.
360        // Authority: CompressedSequentialWriter.java:192.
361        // This CRC is NOT written to CompressionInfo.db.
362        let mut hasher = Hasher::new();
363        hasher.update(&compressed);
364        let crc32 = hasher.finalize();
365
366        // Write compressed data
367        self.output.extend_from_slice(&compressed);
368        self.position += compressed.len() as u64;
369
370        // Write CRC32 inline (TRAILING - after compressed data)
371        // CompressedSequentialWriter.java:203: chunkOffset += compressedLength + 4
372        self.output.extend_from_slice(&crc32.to_be_bytes());
373        self.position += 4;
374
375        // Clear buffer for next chunk
376        self.buffer.clear();
377
378        Ok(())
379    }
380
381    /// Finish writing and return compression metadata
382    ///
383    /// Flushes any remaining data in the buffer and builds CompressionMetadata
384    /// for writing to CompressionInfo.db.
385    pub fn finish(mut self) -> Result<(Vec<u8>, CompressionMetadata)> {
386        // Flush any remaining data
387        self.flush_chunk()?;
388
389        // Build compression metadata
390        let mut metadata =
391            CompressionMetadata::new(self.compressor.algorithm(), self.chunk_size as u32);
392
393        metadata.set_data_length(self.uncompressed_length);
394
395        for offset in &self.chunk_offsets {
396            metadata.add_chunk(*offset);
397        }
398
399        Ok((self.output, metadata))
400    }
401
402    /// Get current output position
403    pub fn position(&self) -> u64 {
404        self.position
405    }
406
407    /// Get number of chunks written so far
408    pub fn chunk_count(&self) -> usize {
409        self.chunk_offsets.len()
410    }
411}
