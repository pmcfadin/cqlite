1//! Finalization + component-path helpers for [`SSTableWriter`].
2//!
3//! Houses [`SSTableWriter::finish`] — the orchestration that writes every
4//! component in the critical order and selects the BIG vs BTI component set —
5//! together with the shared component-naming, bloom-filter, and CRC32 helpers
6//! it relies on. The BTI-specific `Rows.db`/`Partitions.db` serialization is
7//! delegated to [`SSTableWriter::write_bti_components`] (see `bti_state`), so
8//! `finish` stays a thin format-aware orchestrator. All bytes are unchanged
9//! from the prior inline implementation.
10

19impl SSTableWriter {
20    /// Finish writing all components and return SSTable information
21    ///
22    /// This method:
23    /// 1. Finalizes statistics metadata
24    /// 2. Writes all component files in the correct order
25    /// 3. Computes checksums
26    /// 4. Writes TOC.txt (publication barrier)
27    /// 5. Returns SSTableInfo with file paths and metadata
28    ///
29    /// # Returns
30    ///
31    /// SSTableInfo containing paths to all written files and metadata.
32    ///
33    /// # Errors
34    ///
35    /// Returns error if any component write fails.
36    ///
37    /// # Example
38    ///
39    /// ```rust,ignore
40    /// let info = writer.finish().await?;
41    /// println!("SSTable written to {}", info.data_path.display());
42    /// ```
43    #[tracing::instrument(name = "writer.finish", skip(self))]
44    pub async fn finish(mut self) -> Result<SSTableInfo> {
45        // Create keyspace/table subdirectory structure so the reader can
46        // extract the table name from the parent directory path. Owned clone so
47        // `finish_streaming()` can move `self.data_writer` out below.
48        let sstable_dir = self.sstable_dir.clone();
49        let sstable_dir = sstable_dir.as_path();
50        // Ensure the keyspace/table directory tree exists (an empty flush never
51        // opened a streaming sink, so this may be the point of creation). The
52        // flush durability barrier later fsyncs the full leaf→data-root ancestor
53        // chain unconditionally before truncating the WAL, so directory
54        // *creation* no longer needs to record which ancestors it made (issue
55        // #1392).
56        crate::storage::write_engine::durability::create_dir_all(sstable_dir)?;
57
58        // Finalize statistics metadata (normalize sentinel values)
59        self.stats.finalize();
60
61        // Capture format/generation up front so component paths can be computed
62        // after `self`'s writers are partially moved out by their `finish()` calls.
63        let format = self.format;
64        let generation = self.generation;
65        let is_bti = matches!(format, SSTableFormat::Bti);
66        let cpath =
67            |component: &str| Self::component_path_for(sstable_dir, generation, format, component);
68
69        // 1. Write Statistics.db (FIRST - provides delta baseline).
70        // BTI (`da`) requires the BtiFormat `StatsMetadata` layout (covered
71        // clustering Slice, uint deletion times, key range, token-space coverage)
72        // — Cassandra's sstabledump/sstablemetadata reject the legacy `nb` layout
73        // on a `da` descriptor (issue #911). BIG keeps the legacy layout.
74        let stats_path = cpath("Statistics.db");
75        let stats_writer = if is_bti {
76            StatisticsWriter::new_bti(stats_path.clone())
77        } else {
78            StatisticsWriter::new(stats_path.clone())
79        };
80        stats_writer.write(&self.stats, Some(&self.schema))?;
81        // fsync Statistics.db contents (issue #1392): stats_writer uses
82        // std::fs::write, which does not fsync.
83        Self::fsync_component(&stats_path).await?;
84
85        // 2. Finalize Data.db (Issue #492)
86        // The DataWriter has been streaming each partition to disk as it was
87        // written, so there is no whole-file buffer to write here. `finish_streaming`
88        // flushes and fsyncs the sink and returns the total byte size. If no
89        // partitions were written, lazily ensure an (empty) Data.db file exists so
90        // the downstream Digest CRC re-read and TOC publication remain valid.
91        //
92        // The BTI `Data.db` row/partition serialization is identical to BIG in
93        // Cassandra 5 (issue #908); only the filename descriptor differs.
94        let data_path = cpath("Data.db");
95        // The streamed Data.db is fsynced inside `finish_streaming` (issue
96        // #1392). The empty-Data fallback below is written via tokio::fs::write
97        // (no fsync), so it is fsynced explicitly.
98        // `finish_streaming` returns the byte size plus the whole-file
99        // (`Digest.crc32`) and per-chunk (`CRC.db`) checksums accumulated during
100        // the streaming write, so finalization needs no full re-read of Data.db
101        // (issue #1663).
102        let stream = self.data_writer.finish_streaming()?;
103        let data_size = stream.data_size;
104        if data_size == 0 && !data_path.exists() {
105            tokio::fs::write(&data_path, b"").await?;
106            Self::fsync_component(&data_path).await?;
107        }
108
109        // 3. Finalize Index.db (Issue #753) — BIG only.
110        // For BIG, the IndexWriter has been streaming each entry to Index.db; we
111        // flush/sync it here. For BTI (issue #908) there is no Index.db: the
112        // IndexWriter ran in counting-only mode (no sink, no retained entry bytes)
113        // purely to compute offsets, so there is nothing to flush and we report no
114        // path. (Calling `finish_streaming` on a non-streaming writer is an error,
115        // so we skip it.)
116        let index_path = if is_bti {
117            None
118        } else {
119            let _index_size = self.index_writer.finish_streaming()?;
120            Some(cpath("Index.db"))
121        };
122
123        // 4. Write Filter.db (path already set in constructor using the
124        // format-aware component path).
125        // A disabled bloom filter (bloom_filter_fp_chance = 1.0, Cassandra's
126        // AlwaysPresentFilter) writes NO Filter.db component and must be omitted
127        // from the TOC to stay byte-faithful (Issue #852).
128        // `filter_path` is `Some` only when a concrete Filter.db is actually
129        // written. A disabled filter (AlwaysPresentFilter) yields `None`, so
130        // downstream consumers (compaction publish + byte accounting) skip the
131        // non-existent component instead of trying to rename a missing file.
132        let filter_path = match self.filter_writer {
133            Some(filter_writer) => {
134                let emitted = !filter_writer.is_disabled();
135                let path = filter_writer.path().to_path_buf();
136                filter_writer.finish().await?;
137                if emitted {
138                    Some(path)
139                } else {
140                    None
141                }
142            }
143            None => None,
144        };
145        let filter_emitted = filter_path.is_some();
146
147        // 5. Write Summary.db — BIG only.
148        // BTI (issue #908) has no Summary.db; partition sampling is replaced by
149        // the partition trie. We still drive `summary_writer.finish()` to keep its
150        // accounting consistent, but for BTI we discard the bytes and write no file.
151        let summary_bytes = self.summary_writer.finish()?;
152        let summary_path = if is_bti {
153            None
154        } else {
155            let path = cpath("Summary.db");
156            tokio::fs::write(&path, summary_bytes).await?;
157            // fsync Summary.db contents (issue #1392).
158            Self::fsync_component(&path).await?;
159            Some(path)
160        };
161
162        // 5.25. Write Rows.db + Partitions.db (BTI, issue #766 / #908 / #910).
163        // Only emitted for SSTableFormat::Bti; for BIG both are None and nothing
164        // is written, keeping the default path byte-for-byte unchanged. The
165        // ordering rationale (Rows.db first, then the partition trie) and the
166        // empty-BTI refusal live in `write_bti_components`.
167        let bti_pending = self.bti_pending.take();
168        let partitions_trie = self.partitions_trie.take();
169        let (partitions_path, rows_path) =
170            Self::write_bti_components(bti_pending, partitions_trie, &cpath).await?;
171
172        // 5.5. CompressionInfo.db is omitted for uncompressed data.
173        // Real Cassandra 5 SSTables do not include CompressionInfo.db when
174        // data is uncompressed. The compression_info_writer module is retained
175        // for future compressed SSTable support.
176
177        // 6. Write Digest.crc32 (whole-file CRC32 of Data.db).
178        // The streaming DataWriter accumulated this CRC as it wrote, so the
179        // finished Data.db is NOT re-read here (issue #1663). The empty-Data.db
180        // lazy path (no partitions streamed, an empty file written above) has no
181        // accumulated bytes; it recomputes over the just-written empty file via
182        // `compute_crc32` so the value stays correct (CRC32 of zero bytes = 0).
183        let digest_path = cpath("Digest.crc32");
184        let digest_writer = DigestWriter::new(digest_path.clone());
185        let crc32_value = if data_size == 0 {
186            Self::compute_crc32(&data_path).await?
187        } else {
188            stream.digest_crc32
189        };
190        digest_writer.write(crc32_value)?;
191
192        // 6.5. Write CRC.db — per-chunk CRC32 for uncompressed BIG (issue #1197).
193        // Cassandra 5.0 writes a CRC.db for every uncompressed BIG (`nb`)
194        // SSTable, alongside Digest.crc32 (ChecksummedSequentialWriter). The
195        // layout is a big-endian i32 chunk-size header (64 KiB) followed by one
196        // big-endian u32 CRC32 per raw-data chunk. BTI (`da`) tables emit no
197        // CRC.db (verified against the Cassandra-written `da` fixtures), and the
198        // compressed path carries per-chunk CRCs inline (CompressionInfo.db),
199        // so this is gated on the BIG + uncompressed write path only.
200        // Flush vs compaction CRC.db tail (issue #1222). The compaction write
201        // path appends one trailing empty-final-chunk CRC32 = 0 (Cassandra's
202        // close-time zero-length buffer flush); the flush path does not. Selected
203        // by `is_compaction_output` so the flush CRC.db stays byte-identical to
204        // the #1190 goldens while the compaction CRC.db matches the #1017 goldens.
205        let crc_path = if is_bti {
206            None
207        } else {
208            use crate::storage::sstable::writer::crc_writer::{self, CrcTrailer};
209            let trailer = if self.is_compaction_output {
210                CrcTrailer::EmptyFinalChunk
211            } else {
212                CrcTrailer::None
213            };
214            // Assemble CRC.db from the per-chunk CRCs accumulated during the
215            // streaming write — no re-read of the finished Data.db (issue #1663).
216            // The trailer selection (flush vs compaction, issue #1222) and the
217            // is_bti gate are unchanged; only the checksum INPUT SOURCE differs.
218            let crc_path = cpath("CRC.db");
219            let bytes = crc_writer::assemble_crc_bytes(&stream.chunk_crcs, trailer);
220            tokio::fs::write(&crc_path, bytes).await?;
221            // fsync CRC.db contents (issue #1392): tokio::fs::write does not fsync.
222            Self::fsync_component(&crc_path).await?;
223            Some(crc_path)
224        };
225
226        // 7. Write TOC.txt (LAST - publication barrier).
227        //
228        // The TOC lists exactly the component set actually written. BIG lists
229        // Index.db + Summary.db; BTI (issue #908 / #910) omits both and lists
230        // Partitions.db AND Rows.db instead (matching the real `da` fixtures,
231        // which list Rows.db even when it is 0 bytes). TocWriter self-references
232        // TOC.txt.
233        use crate::storage::sstable::directory::types::SSTableComponent;
234        let toc_path = cpath("TOC.txt");
235        let toc_writer = TocWriter::new(toc_path.clone());
236        let mut components = vec![ComponentEntry::new(SSTableComponent::Data)];
237        // Filter.db is omitted entirely for a disabled bloom filter
238        // (AlwaysPresentFilter), matching Cassandra (Issue #852).
239        if filter_emitted {
240            components.push(ComponentEntry::new(SSTableComponent::Filter));
241        }
242        components.extend([
243            ComponentEntry::new(SSTableComponent::Statistics),
244            ComponentEntry::new(SSTableComponent::Digest),
245        ]);
246        // CRC.db is listed for uncompressed BIG tables (issue #1197).
247        if crc_path.is_some() {
248            components.push(ComponentEntry::new(SSTableComponent::Crc));
249        }
250        // BIG lists Index.db + Summary.db; BTI (issue #908) omits both.
251        if index_path.is_some() {
252            components.push(ComponentEntry::new(SSTableComponent::Index));
253        }
254        if summary_path.is_some() {
255            components.push(ComponentEntry::new(SSTableComponent::Summary));
256        }
257        // BTI (issue #766 / #908): list Partitions.db when the trie was emitted.
258        if partitions_path.is_some() {
259            components.push(ComponentEntry::new(SSTableComponent::Partitions));
260        }
261        // BTI (issue #910): list Rows.db when emitted (always for BTI).
262        if rows_path.is_some() {
263            components.push(ComponentEntry::new(SSTableComponent::Rows));
264        }
265        toc_writer.write(&components)?;
266
267        // Data.db bytes written by this SSTable writer (issue #1036). Counts
268        // flush and compaction output alike; finalize sums all components.
269        crate::observability::add_counter(
270            crate::observability::catalog::WRITE_BYTES,
271            data_size,
272            &[],
273        );
274
275        Ok(SSTableInfo {
276            data_path,
277            index_path,
278            filter_path,
279            summary_path,
280            stats_path,
281            compression_info_path: None,
282            partitions_path,
283            rows_path,
284            toc_path,
285            digest_path,
286            crc_path,
287            partition_count: self.partition_count,
288            data_size,
289        })
290    }

292    /// Fsync a component file's *contents* to the storage device (issue #1392).
293    ///
294    /// Components written via `tokio::fs::write` / `std::fs::write` (Statistics,
295    /// Summary, CRC, Rows, Partitions, and the empty-Data fallback) only reach
296    /// the OS page cache; a crash after the WAL is truncated could lose their
297    /// contents even though the directory entry was persisted. This reopens the
298    /// finished file and calls `sync_all` so the bytes are durable before the
299    /// caller (`SSTableWriter::finish`) returns — and therefore before the
300    /// flush's directory fsync and WAL truncate run. (Data.db and Index.db are
301    /// fsynced at their streaming-writer finish; Filter/Digest/TOC fsync inside
302    /// their own writers — none are re-synced here.)
303    pub(super) async fn fsync_component(path: &Path) -> Result<()> {
304        // Reopen with write access, not read-only: `sync_all` on a read-only
305        // handle can fail on Windows (issue #1392, FINDING 2). `read(true)` is
306        // kept so behavior is identical on unix (the file is only fsynced, never
307        // written through this handle).
308        let file = tokio::fs::OpenOptions::new()
309            .read(true)
310            .write(true)
311            .open(path)
312            .await
313            .map_err(|e| {
314                crate::error::Error::Storage(format!(
315                    "Failed to open {} for fsync: {e}",
316                    path.display()
317                ))
318            })?;
319        file.sync_all().await.map_err(|e| {
320            crate::error::Error::Storage(format!("Failed to fsync {}: {e}", path.display()))
321        })
322    }

324    /// Build a component file path for a given on-disk `format`.
325    ///
326    /// The Cassandra filename pattern is `<version>-<id>-<format>-<component>`
327    /// (`SsTableDescriptor::parse`). BIG components use the `nb` version letter
328    /// and the `big` format segment (`nb-<gen>-big-<component>`); BTI components
329    /// use the `da` version letter and the `bti` format segment
330    /// (`da-<gen>-bti-<component>`). This is the single source of truth for
331    /// component naming so the version/format ordering stays consistent.
332    pub(super) fn component_path_for(
333        output_dir: &Path,
334        generation: u64,
335        format: SSTableFormat,
336        component: &str,
337    ) -> PathBuf {
338        let (version, fmt) = match format {
339            SSTableFormat::Big => ("nb", "big"),
340            SSTableFormat::Bti => ("da", "bti"),
341        };
342        let filename = format!("{}-{}-{}-{}", version, generation, fmt, component);
343        output_dir.join(filename)
344    }

346    /// Resolve the table's `bloom_filter_fp_chance` (Issue #852).
347    ///
348    /// The value is read from `schema.comments["bloom_filter_fp_chance"]` when
349    /// present (the table-options bag carried on [`TableSchema`]). A value of
350    /// exactly `1.0` disables the bloom filter (Cassandra's
351    /// `AlwaysPresentFilter`); the [`FilterWriter`] then emits no `Filter.db`.
352    ///
353    /// When the schema does not carry a value, or it cannot be parsed, this
354    /// falls back to Cassandra's default of `0.01`. Out-of-range values
355    /// (outside `(0.0, 1.0]`) also fall back to the default so the writer never
356    /// produces an invalid filter; `FilterWriter::new` still enforces the
357    /// `(0.0, 1.0]` contract for explicitly supplied values.
358    ///
359    /// [`FilterWriter`]: super::FilterWriter
360    pub(super) fn bloom_filter_fp_chance(schema: &TableSchema) -> f64 {
361        const DEFAULT_FP_CHANCE: f64 = 0.01;
362        match schema.comments.get("bloom_filter_fp_chance") {
363            Some(raw) => match raw.trim().parse::<f64>() {
364                Ok(v) if v > 0.0 && v <= 1.0 => v,
365                _ => DEFAULT_FP_CHANCE,
366            },
367            None => DEFAULT_FP_CHANCE,
368        }
369    }

371    /// Compute the CRC32 digest of a finished component by streaming it through
372    /// a fixed-size buffer.
373    ///
374    /// Reads the file in 64 KiB pieces rather than slurping the whole component
375    /// into one `Vec` (`tokio::fs::read`). For a multi-GB merge output the
376    /// digest pass would otherwise allocate a buffer the size of the entire
377    /// Data.db, making end-to-end compaction peak memory scale with the output
378    /// size — defeating the bounded compaction-read work of issue #827. The
379    /// CRC32 is order-sensitive but chunk-size-agnostic, so streaming yields the
380    /// identical digest.
381    pub(super) async fn compute_crc32(file_path: &PathBuf) -> Result<u32> {
382        use tokio::io::AsyncReadExt;
383
384        const DIGEST_READ_BUFFER_BYTES: usize = 64 * 1024;
385
386        let mut file = tokio::fs::File::open(file_path).await?;
387        // Full re-read of a finished component for checksums — count it so the
388        // issue-#1663 work guard catches a regression that reintroduces the
389        // finish-time Data.db re-read on the non-empty path.
390        crate::storage::sstable::work_counters::add_data_db_checksum_full_read();
391        let mut hasher = crc32fast::Hasher::new();
392        let mut buffer = vec![0u8; DIGEST_READ_BUFFER_BYTES];
393        loop {
394            let n = file.read(&mut buffer).await?;
395            if n == 0 {
396                break;
397            }
398            hasher.update(&buffer[..n]);
399        }
400        Ok(hasher.finalize())
401    }
