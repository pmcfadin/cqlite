1//! Type-aware SET-element / MAP-key ordering for the SSTable writer (issue #1275,
2//! recursive composites #1296).
3//!
4//! Cassandra `SetType`/`MapType` are SORTED collections: a frozen set serializes
5//! its elements, and a frozen/non-frozen map its entries, in the order defined by
6//! the element/key type's own `AbstractType.compare` — NOT raw unsigned
7//! serialized-byte order. This module is the SINGLE shared comparator both the
8//! non-frozen complex-cell path (`complex.rs`) and the frozen `serialize_value`
9//! path (`encoding.rs`) call, so the two cannot drift.
10//!
11//! Responsibility split (epic #1116):
12//!   * [`scalar`] — the #1275 SCALAR leaf comparators (signed integers, Java
13//!     `Float/Double.compare`, signed `varint`, scale-aware `decimal`, `UUIDType`).
14//!   * [`composite`] — the #1296 RECURSIVE comparators for `tuple`/UDT and frozen
15//!     nested `set`/`list`/`map`, which recurse back into [`compare_collection_elements`]
16//!     so their leaves reuse the scalar logic above.
17//!
18//! Part of the `data_writer` responsibility split (issue #1118). `use super::*`
19//! provides the crate imports and sibling helpers re-exported from
20//! `data_writer/mod.rs` (notably `serialize_value`). No emitted bytes change for
21//! types whose comparator is unsigned-lexicographic.
22

29/// Total-order comparator for two SET elements / MAP keys, matching the
30/// element/key type's Cassandra `AbstractType.compare` (issue #1275, composites
31/// #1296).
32///
33/// For most types (text/ascii/blob/boolean/inet/date) that comparator IS
34/// unsigned-lexicographic over the serialized bytes, so raw byte order is
35/// correct (see the per-type audit on the fallback arm below). The SCALAR types
36/// that order differently (signed integers, float/double, varint, decimal, uuid)
37/// are handled by [`scalar`]; the COMPOSITE types that compare field/element-wise
38/// (tuple, UDT, nested frozen set/list/map) are handled RECURSIVELY by
39/// [`composite`], whose leaves call back into this function.
40///
41/// The ordering oracle is Cassandra's per-type `AbstractType.compare`
42/// (`org.apache.cassandra.db.marshal`). The decision is driven ENTIRELY by the
43/// element/key `Value` variant (authoritative CQL type metadata carried by the
44/// value itself) — never by inspecting serialized byte patterns (no-heuristics,
45/// issue #28). Any pair whose variants are not a known signed/numeric/composite
46/// type (or are mixed/unsupported) falls back to comparing their serialized
47/// bytes, the historical unsigned-lexicographic behavior that is correct for
48/// those types — and the conservative no-guess choice when type metadata for an
49/// element/key is otherwise unavailable.
50pub(crate) fn compare_collection_elements(a: &Value, b: &Value) -> Ordering {
51    match (a, b) {
52        // Signed fixed-width integers — native signed order.
53        (Value::TinyInt(x), Value::TinyInt(y)) => x.cmp(y),
54        (Value::SmallInt(x), Value::SmallInt(y)) => x.cmp(y),
55        (Value::Integer(x), Value::Integer(y)) => x.cmp(y),
56        (Value::BigInt(x), Value::BigInt(y)) => x.cmp(y),
57        (Value::Counter(x), Value::Counter(y)) => x.cmp(y),
58        // Temporal longs (TimestampType/TimeType extend/share LongType).
59        (Value::Timestamp(x), Value::Timestamp(y)) => x.cmp(y),
60        (Value::Time(x), Value::Time(y)) => x.cmp(y),
61        // Floating point — Float.compare / Double.compare total order.
62        (Value::Float32(x), Value::Float32(y)) => scalar::compare_f32_java(*x, *y),
63        (Value::Float(x), Value::Float(y)) => scalar::compare_f64_java(*x, *y),
64        // Varint — signed big-integer compare of two's-complement bodies.
65        (Value::Varint(x), Value::Varint(y)) => scalar::compare_signed_varint(x, y),
66        // Decimal — scale-aware numeric compare.
67        (
68            Value::Decimal {
69                scale: sa,
70                unscaled: ua,
71            },
72            Value::Decimal {
73                scale: sb,
74                unscaled: ub,
75            },
76        ) => scalar::compare_decimal(*sa, ua, *sb, ub),
77        // UUID/TimeUUID — version-first + v1-timestamp + tail compare (UUIDType).
78        (Value::Uuid(x), Value::Uuid(y)) => scalar::compare_uuid_cassandra(x, y),
79        // Composite types — recurse field/element-wise per Cassandra's
80        // TupleType/UserType/CollectionType comparators (issue #1296). Each
81        // bottoms out at this same dispatcher, reusing the scalar leaves above.
82        (Value::Tuple(x), Value::Tuple(y)) => composite::compare_tuple(x, y),
83        (Value::Udt(x), Value::Udt(y)) => composite::compare_udt(x, y),
84        // frozen<set<T>> arrives as Set: compare the SORTED element sequences
85        // (Cassandra stores set elements sorted; sorting first canonicalizes nested
86        // inner sets BOTTOM-UP before the outer orders by them, #1296).
87        (Value::Set(x), Value::Set(y)) => composite::compare_set(x, y),
88        // frozen<list<T>> arrives as List: compare in INSERTION order (no sort),
89        // shorter-first tiebreak.
90        (Value::List(x), Value::List(y)) => composite::compare_list(x, y),
91        // frozen<map<K,V>> — per entry compare key then value.
92        (Value::Map(x), Value::Map(y)) => composite::compare_map(x, y),
93        // Frozen wrappers: compare the inner values by the same rule (a
94        // frozen<set<int>> element is Frozen(Set(..)) → unwraps to the Set arm).
95        (Value::Frozen(x), other) => compare_collection_elements(x, other),
96        (x, Value::Frozen(y)) => compare_collection_elements(x, y),
97        // Everything else falls back to unsigned-lexicographic order over the
98        // serialized bytes. Per-type audit vs Cassandra `AbstractType.compare`
99        // (issue #1275 convergence audit) — each of these IS unsigned-byte-
100        // lexicographic, so raw serialized bytes are the correct order:
101        //   * `text`/`varchar` (UTF8Type) and `ascii` (AsciiType): both use
102        //     `ComparisonType.BYTE_ORDER` — UTF-8 / 7-bit ASCII bytes sort
103        //     unsigned-lexicographically, which equals codepoint order.
104        //   * `blob` (BytesType): BYTE_ORDER, raw bytes.
105        //   * `boolean` (BooleanType): BYTE_ORDER over the single serialized byte
106        //     (0x00 false < 0x01 true), which is the natural order.
107        //   * `inet` (InetAddressType): BYTE_ORDER — IPv4 (4 bytes) sorts before
108        //     any IPv6 (16 bytes) by length-then-byte, matching Cassandra.
109        //   * `date` (SimpleDateType): stored as an UNSIGNED 32-bit day count
110        //     offset by 2^31, compared `ByteBufferUtil.compareUnsigned`, i.e.
111        //     unsigned big-endian bytes — exactly raw-byte order.
112        //   * `duration` (DurationType): Cassandra makes duration NON-comparable
113        //     (`isEmptyValueMeaningless`/no total order) and FORBIDS it as a set
114        //     element or map key, so a `Value::Duration` cannot legitimately reach
115        //     this comparator; the fallback keeps the sort total/panic-free if one
116        //     somehow does, but it is not an ordering Cassandra would ever emit.
117        // A mixed/unsupported variant pair also lands here (byte fallback keeps
118        // the sort total) — including a composite whose element/field type
119        // metadata is unavailable, where the no-heuristics mandate (#28) forbids
120        // guessing, so we keep the conservative byte fallback rather than infer.
121        _ => match (serialize_value(a), serialize_value(b)) {
122            (Ok(ba), Ok(bb)) => ba.cmp(&bb),
123            // Serialization should not fail for a non-null collection element; if
124            // it does, treat as equal so the sort stays total and panic-free.
125            _ => Ordering::Equal,
126        },
127    }
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133
134    /// Int32Type/SetType: signed order — -1 before 0 before 1, even though
135    /// `0xFFFF_FFFF` (-1) sorts LAST under raw unsigned bytes.
136    #[test]
137    fn int_elements_sort_signed() {
138        let mut v = vec![Value::Integer(1), Value::Integer(-1), Value::Integer(0)];
139        v.sort_by(compare_collection_elements);
140        assert_eq!(
141            v,
142            vec![Value::Integer(-1), Value::Integer(0), Value::Integer(1)]
143        );
144    }
145
146    /// LongType (bigint), ShortType (smallint), ByteType (tinyint): all signed.
147    #[test]
148    fn other_signed_integers_sort_signed() {
149        let mut big = vec![Value::BigInt(5), Value::BigInt(-9), Value::BigInt(0)];
150        big.sort_by(compare_collection_elements);
151        assert_eq!(
152            big,
153            vec![Value::BigInt(-9), Value::BigInt(0), Value::BigInt(5)]
154        );
155
156        let mut small = vec![Value::SmallInt(2), Value::SmallInt(-2)];
157        small.sort_by(compare_collection_elements);
158        assert_eq!(small, vec![Value::SmallInt(-2), Value::SmallInt(2)]);
159
160        let mut tiny = vec![Value::TinyInt(7), Value::TinyInt(-7)];
161        tiny.sort_by(compare_collection_elements);
162        assert_eq!(tiny, vec![Value::TinyInt(-7), Value::TinyInt(7)]);
163    }
164
165    /// DoubleType/FloatType: Double.compare total order — -1.5 < -0.0 < +0.0 < 2.0,
166    /// NaN sorts last; raw big-endian bytes would put negatives high.
167    #[test]
168    fn floats_sort_by_total_order() {
169        let mut v = vec![
170            Value::Float(2.0),
171            Value::Float(f64::NAN),
172            Value::Float(-1.5),
173            Value::Float(0.0),
174        ];
175        v.sort_by(compare_collection_elements);
176        assert_eq!(v[0], Value::Float(-1.5));
177        assert_eq!(v[1], Value::Float(0.0));
178        assert_eq!(v[2], Value::Float(2.0));
179        assert!(matches!(v[3], Value::Float(f) if f.is_nan()));
180    }
181
182    /// Unsigned-lexicographic types (text/blob/boolean) keep serialized-byte
183    /// order — raw byte comparison is the correct Cassandra comparator there.
184    #[test]
185    fn text_keeps_byte_order() {
186        let mut v = vec![
187            Value::Text("c".to_string()),
188            Value::Text("a".to_string()),
189            Value::Text("b".to_string()),
190        ];
191        v.sort_by(compare_collection_elements);
192        assert_eq!(
193            v,
194            vec![
195                Value::Text("a".to_string()),
196                Value::Text("b".to_string()),
197                Value::Text("c".to_string()),
198            ]
199        );
200    }
