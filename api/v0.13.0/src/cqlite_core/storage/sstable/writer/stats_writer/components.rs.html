14/// Serialise the STATS `pendingRepair` nullable field: a single presence byte
15/// (`0x01` present, `0x00` null) followed by the raw 16-byte UUID when present.
16///
17/// Mirrors cassandra-5.0.0 `StatsMetadata.StatsMetadataSerializer.serialize`:
18/// `out.writeBoolean(pendingRepair != null)` then, if present,
19/// `UUIDSerializer.serialize` (the two `long`s, MSB then LSB, big-endian — which
20/// is exactly the 16 raw bytes). Used by both the `nb`/`oa` and `da` builders so
21/// a preserved pending-repair UUID round-trips through the read-path walk.
22fn write_pending_repair(buffer: &mut Vec<u8>, pending_repair: Option<[u8; 16]>) -> Result<()> {
23    match pending_repair {
24        Some(uuid) => {
25            buffer.write_all(&[0x01])?;
26            buffer.write_all(&uuid)?;
27        }
28        None => buffer.write_all(&[0x00])?,
29    }
30    Ok(())
31}

33impl StatisticsWriter {
34    /// Build VALIDATION component (MetadataType ordinal 0)
35    ///
36    /// Format (ValidationMetadata.java):
37    /// - partitioner class name (Java writeUTF: u16 BE length + UTF-8 bytes)
38    /// - bloom filter FP chance (f64 BE)
39    pub(super) fn build_validation_component(&self) -> Result<Vec<u8>> {
40        let mut buffer = Vec::new();
41
42        // Partitioner class name (Java writeUTF format)
43        let partitioner = b"org.apache.cassandra.dht.Murmur3Partitioner";
44
45        // Java writeUTF: u16 BE length prefix + modified UTF-8 bytes
46        let len = partitioner.len() as u16;
47        buffer.write_all(&len.to_be_bytes())?;
48        buffer.write_all(partitioner)?;
49
50        // Bloom filter false positive chance (f64 BE)
51        let fp_chance = 0.01f64;
52        buffer.write_all(&fp_chance.to_be_bytes())?;
53
54        Ok(buffer)
55    }

57    /// Build COMPACTION component (MetadataType ordinal 1)
58    ///
59    /// Format (CompactionMetadata.java):
60    /// - cardinality estimator (i32 BE length + HyperLogLogPlus bytes)
61    ///
62    /// We write a minimal valid empty HyperLogLogPlus sketch.
63    pub(super) fn build_compaction_component(&self) -> Result<Vec<u8>> {
64        let mut buffer = Vec::new();
65
66        // Cardinality estimator: ByteArrayUtil.writeWithLength(bytes, out)
67        // Format: i32 BE length + data bytes
68        //
69        // Minimal valid HyperLogLogPlus(p=11, sp=25) in SPARSE format:
70        // - 4 bytes: version (-2 as i32 = 0xFFFFFFFE)
71        // - 1 byte: p = 11 (0x0B)
72        // - 1 byte: sp = 25 (0x19)
73        // - 1 byte: format type = SPARSE (0x01)
74        // - 4 bytes: tempSetSize = 0
75        // - 4 bytes: sparseSetSize = 0
76        // Total: 15 bytes
77
78        const HLL_DATA: [u8; 15] = [
79            0xFF, 0xFF, 0xFF, 0xFE, // version = -2 (HyperLogLogPlus marker)
80            0x0B, // p = 11 (precision)
81            0x19, // sp = 25 (sparse precision)
82            0x01, // format = SPARSE
83            0x00, 0x00, 0x00, 0x00, // tempSetSize = 0
84            0x00, 0x00, 0x00, 0x00, // sparseSetSize = 0
85        ];
86
87        // Write length prefix (i32 BE)
88        buffer.write_all(&(HLL_DATA.len() as i32).to_be_bytes())?;
89
90        // Write HLL data
91        buffer.write_all(&HLL_DATA)?;
92
93        Ok(buffer)
94    }

96    /// Build STATS component (MetadataType ordinal 2)
97    ///
98    /// Format for nb version (StatsMetadata.java lines 401-512):
99    /// This is a complete serialization of all required fields for Cassandra 5.0 nb format.
100    pub(super) fn build_stats_component(&self, metadata: &StatisticsMetadata) -> Result<Vec<u8>> {
101        let mut buffer = Vec::new();
102
103        // 1-2. EstimatedHistogram estimatedPartitionSize and
104        // estimatedCellPerPartitionCount. Populated with one observation per
105        // partition (issue #1327) so Σ estimatedPartitionSize bucket counts ==
106        // partition_count (the authoritative read-side decode, issue #944).
107        metadata.estimated_partition_size.write_to(&mut buffer);
108        metadata.estimated_cell_count.write_to(&mut buffer);
109
110        // 3. CommitLogPosition commitLogUpperBound (NONE = segmentId=-1, position=0)
111        buffer.write_all(&(-1i64).to_be_bytes())?; // segmentId
112        buffer.write_all(&0i32.to_be_bytes())?; // position
113
114        // 4. long minTimestamp
115        buffer.write_all(&metadata.min_timestamp.to_be_bytes())?;
116
117        // 5. long maxTimestamp
118        buffer.write_all(&metadata.max_timestamp.to_be_bytes())?;
119
120        // 6. int minLocalDeletionTime (use Integer.MAX_VALUE if no deletions)
121        let min_del_time = if metadata.min_local_deletion_time == 0 {
122            i32::MAX
123        } else {
124            metadata.min_local_deletion_time
125        };
126        buffer.write_all(&min_del_time.to_be_bytes())?;
127
128        // 7. int maxLocalDeletionTime
129        let max_del_time = if metadata.max_local_deletion_time == 0 {
130            i32::MAX
131        } else {
132            metadata.max_local_deletion_time
133        };
134        buffer.write_all(&max_del_time.to_be_bytes())?;
135
136        // 8. int minTTL
137        buffer.write_all(&metadata.min_ttl.to_be_bytes())?;
138
139        // 9. int maxTTL
140        buffer.write_all(&metadata.max_ttl.to_be_bytes())?;
141
142        // 10. double compressionRatio (use -1.0 for unknown)
143        buffer.write_all(&(-1.0f64).to_be_bytes())?;
144
145        // 11. TombstoneHistogram estimatedTombstoneDropTime
146        // Populated from tombstone local-deletion-times accumulated during the write path.
147        // Legacy serializer (nb format): maxBinSize (i32), size (i32),
148        // then size × (f64 point, i64 value).
149        self.write_tombstone_histogram(&mut buffer, &metadata.tombstone_histogram)?;
150
151        // 12. int sstableLevel
152        buffer.write_all(&0i32.to_be_bytes())?;
153
154        // 13. long repairedAt — preserved from the (compatible) compaction
155        // inputs (issue #1021); 0 for an unrepaired SSTable / fresh flush.
156        buffer.write_all(&metadata.repaired_at.to_be_bytes())?;
157
158        // 14. int minClusteringCount (no clustering = 0)
159        buffer.write_all(&0i32.to_be_bytes())?;
160
161        // 15. [clustering values] - count=0 means no values to write
162
163        // 16. int maxClusteringCount
164        buffer.write_all(&0i32.to_be_bytes())?;
165
166        // 17. [clustering values] - count=0 means no values to write
167
168        // 18. boolean hasLegacyCounterShards
169        buffer.write_all(&[0x00])?; // false
170
171        // 19. long totalColumnsSet
172        buffer.write_all(&metadata.column_count.to_be_bytes())?;
173
174        // 20. long totalRows
175        buffer.write_all(&metadata.row_count.to_be_bytes())?;
176
177        // 21. CommitLogPosition commitLogLowerBound (NONE)
178        buffer.write_all(&(-1i64).to_be_bytes())?; // segmentId
179        buffer.write_all(&0i32.to_be_bytes())?; // position
180
181        // 22. IntervalSet<CommitLogPosition> commitLogIntervals (empty set: size=0)
182        buffer.write_all(&0i32.to_be_bytes())?;
183
184        // 23. pendingRepair (nullable): presence byte, then the 16-byte UUID
185        // when present. Preserved from compatible compaction inputs (#1021).
186        write_pending_repair(&mut buffer, metadata.pending_repair)?;
187
188        // 24. boolean isTransient — preserved from compatible inputs (#1021).
189        buffer.write_all(&[if metadata.is_transient { 0x01 } else { 0x00 }])?;
190
191        // 25. byte originatingHostId (0 = null)
192        buffer.write_all(&[0x00])?;
193
194        Ok(buffer)
195    }

197    /// Build the STATS component for the Cassandra-canonical `da` (BtiFormat)
198    /// layout.
199    ///
200    /// Field order and gating follow cassandra-5.0.0
201    /// `StatsMetadata.StatsMetadataSerializer.serialize` evaluated for
202    /// `BtiFormat.BtiVersion` (all version flags `true` except `hasLegacyMinMax`,
203    /// which is `false`):
204    ///
205    /// 1.  estimatedPartitionSize (EstimatedHistogram)
206    /// 2.  estimatedCellPerPartitionCount (EstimatedHistogram)
207    /// 3.  commitLogUpperBound (CommitLogPosition)
208    /// 4.  minTimestamp, maxTimestamp (long)
209    /// 5.  min/maxLocalDeletionTime as **unsigned int** (`hasUIntDeletionTime`);
210    ///     `NO_DELETION_TIME` (Long.MAX) maps to `0xFFFFFFFF`.
211    /// 6.  minTTL, maxTTL (int)
212    /// 7.  compressionRatio (double)
213    /// 8.  estimatedTombstoneDropTime (TombstoneHistogram)
214    /// 9.  sstableLevel (int), repairedAt (long)
215    /// 10. improvedMinMax (`!hasLegacyMinMax && hasImprovedMinMax`):
216    ///     clusteringTypes list + coveredClustering `Slice`. We emit
217    ///     `Slice.ALL` (BOTTOM..TOP) — a valid, conservative covering slice that
218    ///     Cassandra accepts and that matches the `Covered clusterings: [, ]`
219    ///     shown by `sstablemetadata` on the real `da` fixtures.
220    /// 11. hasLegacyCounterShards (bool)
221    /// 12. totalColumnsSet, totalRows (long)
222    /// 13. commitLogLowerBound (CommitLogPosition), commitLogIntervals (IntervalSet)
223    /// 14. pendingRepair (byte 0 = null)
224    /// 15. isTransient (bool)
225    /// 16. originatingHostId (byte 0 = null)
226    /// 17. hasPartitionLevelDeletions (bool)
227    /// 18. firstKey, lastKey (vint-length ByteBuffer) — `hasKeyRange`
228    /// 19. tokenSpaceCoverage (double) — `hasTokenSpaceCoverage`
229    pub(super) fn build_stats_component_da(
230        &self,
231        metadata: &StatisticsMetadata,
232        schema: Option<&TableSchema>,
233    ) -> Result<Vec<u8>> {
234        let mut buffer = Vec::new();
235
236        // 1-2. EstimatedHistogram estimatedPartitionSize / estimatedCellPerPartitionCount
237        // Populated per partition (issue #1327): Σ estimatedPartitionSize bucket
238        // counts == partition_count for the authoritative read-side decode.
239        metadata.estimated_partition_size.write_to(&mut buffer);
240        metadata.estimated_cell_count.write_to(&mut buffer);
241
242        // 3. CommitLogPosition commitLogUpperBound (NONE)
243        buffer.write_all(&(-1i64).to_be_bytes())?; // segmentId
244        buffer.write_all(&0i32.to_be_bytes())?; // position
245
246        // 4. minTimestamp, maxTimestamp
247        buffer.write_all(&metadata.min_timestamp.to_be_bytes())?;
248        buffer.write_all(&metadata.max_timestamp.to_be_bytes())?;
249
250        // 5. min/maxLocalDeletionTime — unsigned int (hasUIntDeletionTime).
251        // No tombstones (sentinel) → NO_DELETION_TIME_UNSIGNED_INTEGER = 0xFFFFFFFF;
252        // otherwise the low 32 bits of the (seconds-since-epoch) deletion time
253        // (Cassandra `CassandraUInt.fromLong` == `(int) value`).
254        let min_ldt_uint: u32 = if metadata.min_local_deletion_time == 0 {
255            0xFFFF_FFFF
256        } else {
257            metadata.min_local_deletion_time as u32
258        };
259        let max_ldt_uint: u32 = if metadata.max_local_deletion_time == 0 {
260            0xFFFF_FFFF
261        } else {
262            metadata.max_local_deletion_time as u32
263        };
264        buffer.write_all(&min_ldt_uint.to_be_bytes())?;
265        buffer.write_all(&max_ldt_uint.to_be_bytes())?;
266
267        // 6. minTTL, maxTTL
268        buffer.write_all(&metadata.min_ttl.to_be_bytes())?;
269        buffer.write_all(&metadata.max_ttl.to_be_bytes())?;
270
271        // 7. compressionRatio (-1.0 = unknown)
272        buffer.write_all(&(-1.0f64).to_be_bytes())?;
273
274        // 8. TombstoneHistogram estimatedTombstoneDropTime.
275        //
276        // The `da` (BtiFormat) version resolves `TombstoneHistogram.getSerializer`
277        // to the modern `HistogramSerializer` (long point + int value per bin),
278        // NOT the legacy serializer (double + long) used by older versions. Using
279        // the legacy entry encoding here mis-sizes the body and derails the
280        // subsequent `coveredClustering` Slice deserialization in Cassandra.
281        self.write_tombstone_histogram_modern(&mut buffer, &metadata.tombstone_histogram)?;
282
283        // 9. sstableLevel, repairedAt (repairedAt preserved from compatible
284        // compaction inputs, issue #1021).
285        buffer.write_all(&0i32.to_be_bytes())?;
286        buffer.write_all(&metadata.repaired_at.to_be_bytes())?;
287
288        // 10. improvedMinMax: clusteringTypes list + coveredClustering Slice.
289        //
290        // AbstractTypeSerializer.serializeList: unsigned-VInt count, then each
291        // type as an unsigned-VInt-length-prefixed UTF-8 marshal-class string —
292        // the exact encoding used for the SERIALIZATION_HEADER clusteringTypes.
293        let clustering_types: Vec<String> = schema
294            .map(|s| {
295                s.clustering_keys
296                    .iter()
297                    .map(|ck| cql_type_to_marshal_type(&ck.data_type))
298                    .collect()
299            })
300            .unwrap_or_default();
301        buffer.write_all(&encode_vuint(clustering_types.len() as u64))?;
302        for ty in &clustering_types {
303            buffer.write_all(&encode_vuint(ty.len() as u64))?;
304            buffer.write_all(ty.as_bytes())?;
305        }
306        // coveredClustering = Slice.ALL = (BOTTOM, TOP). A ClusteringBound
307        // serialises as `[byte kind ordinal][short size][values...]`; BOTTOM and
308        // TOP are empty bounds (size 0). Kind ordinals (ClusteringPrefix.Kind):
309        // INCL_START_BOUND = 1 (BOTTOM), INCL_END_BOUND = 6 (TOP).
310        const KIND_INCL_START_BOUND: u8 = 1;
311        const KIND_INCL_END_BOUND: u8 = 6;
312        buffer.write_all(&[KIND_INCL_START_BOUND])?;
313        buffer.write_all(&0u16.to_be_bytes())?; // start size = 0
314        buffer.write_all(&[KIND_INCL_END_BOUND])?;
315        buffer.write_all(&0u16.to_be_bytes())?; // end size = 0
316
317        // 11. hasLegacyCounterShards
318        buffer.write_all(&[0x00])?;
319
320        // 12. totalColumnsSet, totalRows
321        buffer.write_all(&metadata.column_count.to_be_bytes())?;
322        buffer.write_all(&metadata.row_count.to_be_bytes())?;
323
324        // 13. commitLogLowerBound (NONE) + commitLogIntervals (empty IntervalSet)
325        buffer.write_all(&(-1i64).to_be_bytes())?; // segmentId
326        buffer.write_all(&0i32.to_be_bytes())?; // position
327        buffer.write_all(&0i32.to_be_bytes())?; // interval set size = 0
328
329        // 14. pendingRepair (nullable): presence byte + optional UUID, preserved
330        // from compatible compaction inputs (#1021).
331        write_pending_repair(&mut buffer, metadata.pending_repair)?;
332
333        // 15. isTransient — preserved from compatible inputs (#1021).
334        buffer.write_all(&[if metadata.is_transient { 0x01 } else { 0x00 }])?;
335
336        // 16. originatingHostId (null)
337        buffer.write_all(&[0x00])?;
338
339        // 17. hasPartitionLevelDeletions
340        //
341        // cassandra-5.0.0 StatsMetadata.StatsMetadataSerializer.serialize line 495:
342        //   out.writeBoolean(component.hasPartitionLevelDeletions)
343        // DataOutput.writeBoolean emits a single byte: 0x01 for true, 0x00 false.
344        // Set when any partition written carries a partition-level tombstone.
345        let has_partition_deletions = if metadata.has_partition_level_deletions {
346            0x01u8
347        } else {
348            0x00u8
349        };
350        buffer.write_all(&[has_partition_deletions])?;
351
352        // 18. firstKey, lastKey (ByteBufferUtil.writeWithVIntLength:
353        // unsigned-VInt length then the raw key bytes). Empty when no partitions.
354        let first = metadata.first_key.as_deref().unwrap_or(&[]);
355        let last = metadata.last_key.as_deref().unwrap_or(&[]);
356        buffer.write_all(&encode_vuint(first.len() as u64))?;
357        buffer.write_all(first)?;
358        buffer.write_all(&encode_vuint(last.len() as u64))?;
359        buffer.write_all(last)?;
360
361        // 19. tokenSpaceCoverage (double). Cassandra writes NaN when not computed
362        // (sstablemetadata renders this as "Local token space coverage: NaN").
363        buffer.write_all(&f64::NAN.to_be_bytes())?;
364
365        Ok(buffer)
366    }

368    /// Serialise a `TombstoneHistogram` using Cassandra's legacy format (nb/mc).
369    ///
370    /// Binary layout (matches `TombstoneHistogram.LegacyHistogramSerializer`):
371    /// ```text
372    /// i32 BE  maxBinSize  — capacity: 100 when non-empty, 0 when empty
373    /// i32 BE  size        — actual number of populated bins
374    /// for each of the `size` bins (ascending point order):
375    ///   f64 BE  point  — local-deletion-time bucket centre (as double)
376    ///   i64 BE  value  — tombstone count for this bucket
377    /// ```
378    ///
379    /// An empty histogram writes 8 bytes (`maxBinSize=0, size=0`), matching
380    /// what Cassandra writes for SSTables with no tombstones.
381    pub(super) fn write_tombstone_histogram(
382        &self,
383        buffer: &mut Vec<u8>,
384        histogram: &TombstoneHistogram,
385    ) -> Result<()> {
386        if histogram.is_empty() {
387            // Empty: maxBinSize=0, size=0 — no entry pairs follow
388            buffer.write_all(&0i32.to_be_bytes())?; // maxBinSize
389            buffer.write_all(&0i32.to_be_bytes())?; // size
390        } else {
391            // Non-empty: maxBinSize=100 (the capacity constant), then the bins
392            buffer.write_all(&TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE.to_be_bytes())?; // maxBinSize
393            buffer.write_all(&histogram.size().to_be_bytes())?; // size
394            for (point, value) in histogram.entries() {
395                buffer.write_all(&point.to_be_bytes())?; // f64 BE point
396                buffer.write_all(&value.to_be_bytes())?; // i64 BE value
397            }
398        }
399        Ok(())
400    }

402    /// Serialise a `TombstoneHistogram` using Cassandra's modern (`oa`/`da`)
403    /// `HistogramSerializer` format.
404    ///
405    /// Identical framing to the legacy serializer (`i32 maxBinSize`, `i32 size`)
406    /// but each bin entry is `i64 BE point` + `i32 BE value` (12 bytes), where
407    /// the legacy serializer used `f64 point` + `i64 value` (16 bytes). The
408    /// version split: `StatsMetadata.StatsMetadataSerializer` resolves
409    /// `TombstoneHistogram.getSerializer(version)` to the modern
410    /// `HistogramSerializer` for `oa`+ (incl. `BtiFormat.BtiVersion` `da`) and
411    /// the `LegacyHistogramSerializer` for older versions.
412    ///
413    /// Authority: repo `docs/sstables-definitive-guide/statistics-db-writer-spec.md`
414    /// (TombstoneHistogram legacy vs modern note) and
415    /// cassandra-5.0.0 `TombstoneHistogram.java` (`HistogramSerializer` ->
416    /// `long` point + `int` value).
417    ///
418    /// ```text
419    /// i32 BE  maxBinSize  — capacity: 100 when non-empty, 0 when empty
420    /// i32 BE  size        — actual number of populated bins
421    /// for each of the `size` bins (ascending point order):
422    ///   i64 BE  point  — local-deletion-time bucket centre (as long)
423    ///   i32 BE  value  — tombstone count for this bucket
424    /// ```
425    fn write_tombstone_histogram_modern(
426        &self,
427        buffer: &mut Vec<u8>,
428        histogram: &TombstoneHistogram,
429    ) -> Result<()> {
430        if histogram.is_empty() {
431            buffer.write_all(&0i32.to_be_bytes())?; // maxBinSize
432            buffer.write_all(&0i32.to_be_bytes())?; // size
433        } else {
434            buffer.write_all(&TOMBSTONE_HISTOGRAM_MAX_BIN_SIZE.to_be_bytes())?; // maxBinSize
435            buffer.write_all(&histogram.size().to_be_bytes())?; // size
436            for (point, value) in histogram.entries() {
437                // Modern: long point (truncate the double bucket centre) + int value.
438                buffer.write_all(&(point as i64).to_be_bytes())?; // i64 BE point
439                buffer.write_all(&(value as i32).to_be_bytes())?; // i32 BE value
440            }
441        }
442        Ok(())
443    }

470    #[test]
471    fn test_build_stats_component() {
472        let writer = StatisticsWriter::new(PathBuf::from("test.db"));
473
474        let mut meta = StatisticsMetadata::new();
475        meta.min_timestamp = 1000000;
476        meta.max_timestamp = 2000000;
477        meta.min_local_deletion_time = 0;
478        meta.max_local_deletion_time = 0;
479        meta.min_ttl = 0;
480        meta.max_ttl = 0;
481        meta.partition_count = 100;
482        meta.row_count = 100;
483        meta.column_count = 200;
484
485        let result = writer.build_stats_component(&meta);
486        assert!(result.is_ok());
487
488        let data = result.unwrap();
489        assert!(!data.is_empty());
490
491        // STATS component now has a complex binary format (nb version)
492        // It should contain:
493        // - estimatedPartitionSize EstimatedHistogram — 156 buckets (issue #1327):
494        //   4 + 156*16 = 2500 bytes
495        // - estimatedCellPerPartitionCount EstimatedHistogram — 119 buckets
496        //   (EH(118), distinct Cassandra shape, issue #1327): 4 + 119*16 = 1908 bytes
497        // - CommitLogPosition upper bound (12 bytes)
498        // - min/max timestamps (16 bytes)
499        // - min/max deletion times (8 bytes)
500        // - min/max TTL (8 bytes)
501        // - compression ratio (8 bytes)
502        // - TombstoneHistogram (8 bytes for empty)
503        // - sstableLevel (4 bytes)
504        // - repairedAt (8 bytes)
505        // - min/max clustering count (8 bytes)
506        // - hasLegacyCounterShards (1 byte)
507        // - totalColumnsSet (8 bytes)
508        // - totalRows (8 bytes)
509        // - CommitLogPosition lower bound (12 bytes)
510        // - commitLogIntervals empty set (4 bytes)
511        // - pendingRepair (1 byte)
512        // - isTransient (1 byte)
513        // - originatingHostId (1 byte)
514        let partition_size_bytes = 4 + 156 * 16; // 2500
515        let cell_count_bytes = 4 + 119 * 16; // 1908 (EH(118) — distinct shape)
516        let two_histograms = partition_size_bytes + cell_count_bytes; // 4408
517        let fixed_tail = 12 + 16 + 8 + 8 + 8 + 8 + 4 + 8 + 8 + 1 + 8 + 8 + 12 + 4 + 1 + 1 + 1;
518        let expected_len = two_histograms + fixed_tail;
519        assert_eq!(data.len(), expected_len);
520
521        // Verify totalRows: it follows the two histograms + the fixed prefix up to
522        // (but excluding) totalRows itself.
523        //   commitLogUpper(12) + min/maxTs(16) + min/maxLDT(8) + min/maxTTL(8) +
524        //   compressionRatio(8) + tombstoneHistogram empty(8) + sstableLevel(4) +
525        //   repairedAt(8) + min clustering(4) + max clustering(4) +
526        //   hasLegacyCounterShards(1) + totalColumnsSet(8)
527        let row_count_offset = two_histograms + 12 + 16 + 8 + 8 + 8 + 8 + 4 + 8 + 4 + 4 + 1 + 8;
528        let row_count_bytes = &data[row_count_offset..row_count_offset + 8];
529        let row_count = u64::from_be_bytes(row_count_bytes.try_into().unwrap());
530        assert_eq!(row_count, 100);
531    }

533    /// The `da` STATS body must serialise `hasPartitionLevelDeletions` as a
534    /// single `writeBoolean` byte: `0x01` when the SSTable contains a
535    /// partition-level deletion, `0x00` otherwise. Authority: cassandra-5.0.0
536    /// `StatsMetadata.StatsMetadataSerializer.serialize` line 495
537    /// (`out.writeBoolean(component.hasPartitionLevelDeletions)`), gated by
538    /// `version.hasPartitionLevelDeletionsPresenceMarker()` which is `true` for
539    /// `BtiFormat.BtiVersion`.
540    #[test]
541    fn test_da_stats_has_partition_level_deletions_marker_byte() {
542        let writer = StatisticsWriter::new_bti(PathBuf::from("da-1-bti-Statistics.db"));
543
544        // Identical metadata except for the partition-level-deletion flag, so
545        // the only differing byte is the hasPartitionLevelDeletions marker.
546        let mut base = StatisticsMetadata::new();
547        base.min_timestamp = 1_000_000;
548        base.max_timestamp = 2_000_000;
549        base.min_local_deletion_time = 0;
550        base.max_local_deletion_time = 0;
551        base.partition_count = 1;
552        base.row_count = 1;
553        base.column_count = 1;
554        base.finalize();
555
556        let mut with_del = base.clone();
557        with_del.mark_partition_level_deletion();
558        assert!(with_del.has_partition_level_deletions);
559        assert!(!base.has_partition_level_deletions);
560
561        let no_del_bytes = writer.build_stats_component_da(&base, None).unwrap();
562        let del_bytes = writer.build_stats_component_da(&with_del, None).unwrap();
563
564        // Same length: only the one marker byte changes value.
565        assert_eq!(
566            no_del_bytes.len(),
567            del_bytes.len(),
568            "the flag must not change the serialized length"
569        );
570
571        // Exactly one byte differs, and it flips 0x00 <-> 0x01.
572        let diffs: Vec<usize> = no_del_bytes
573            .iter()
574            .zip(del_bytes.iter())
575            .enumerate()
576            .filter(|(_, (a, b))| a != b)
577            .map(|(i, _)| i)
578            .collect();
579        assert_eq!(
580            diffs.len(),
581            1,
582            "only the hasPartitionLevelDeletions marker byte should differ"
583        );
584        let marker = diffs[0];
585        assert_eq!(
586            no_del_bytes[marker], 0x00,
587            "no partition deletion => marker byte 0x00"
588        );
589        assert_eq!(
590            del_bytes[marker], 0x01,
591            "partition deletion present => marker byte 0x01"
592        );
593    }

614    #[test]
615    fn test_write_tombstone_histogram_nonempty() {
616        let writer = StatisticsWriter::new(PathBuf::from("test.db"));
617        let mut h = TombstoneHistogram::new();
618        h.update(1_700_000_000);
619        h.update(1_700_000_100);
620        let mut buf = Vec::new();
621        writer.write_tombstone_histogram(&mut buf, &h).unwrap();
622
623        // Non-empty with 2 bins:
624        //   4 bytes maxBinSize=100
625        //   4 bytes size=2
626        //   2 × (8 f64 + 8 i64) = 32 bytes
627        // Total = 40 bytes
628        assert_eq!(buf.len(), 40, "2-bin histogram should be 40 bytes");
629
630        let max_bin_size = i32::from_be_bytes(buf[0..4].try_into().unwrap());
631        let size = i32::from_be_bytes(buf[4..8].try_into().unwrap());
632        assert_eq!(
633            max_bin_size, 100,
634            "non-empty histogram maxBinSize should be 100"
635        );
636        assert_eq!(size, 2, "histogram size should be 2");
637
638        // Verify first bin point (should be 1_700_000_000.0)
639        let point0 = f64::from_be_bytes(buf[8..16].try_into().unwrap());
640        assert_eq!(point0, 1_700_000_000.0f64);
641        let value0 = i64::from_be_bytes(buf[16..24].try_into().unwrap());
642        assert_eq!(value0, 1);
643
644        // Verify second bin point (should be 1_700_000_100.0)
645        let point1 = f64::from_be_bytes(buf[24..32].try_into().unwrap());
646        assert_eq!(point1, 1_700_000_100.0f64);
647        let value1 = i64::from_be_bytes(buf[32..40].try_into().unwrap());
648        assert_eq!(value1, 1);
649    }
