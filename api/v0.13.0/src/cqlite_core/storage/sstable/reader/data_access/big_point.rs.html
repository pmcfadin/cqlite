1//! BIG ("nb" / uncompressed) partition point lookup (issue #1572).
2//!
3//! Before this module, `SSTableReader::get()` on a BIG SSTable read and
4//! decompressed the ENTIRE `Data.db` on every lookup: the legacy `SSTableIndex`
5//! (`self.index`) is keyed on key *digests*, so `find_entry()` called with the
6//! raw partition-key bytes always missed and the miss fell through to
7//! `scan_for_key` → whole-file stitch + decompress.
8//!
9//! The fix resolves the partition's uncompressed `Data.db` offset via the raw-key
10//! `index_reader` (the O(1) `Index.db` map, keyed on raw partition-key bytes since
11//! issue #552), then seeks and decompresses ONLY the chunk(s) covering that offset
12//! — reusing the exact chunk-targeted decode the BTI point lookup already proves
13//! correct ([`SSTableReader::bti_decompress_and_parse_target`], bti.rs). The
14//! `Index.db` data offset and the BTI trie offset share the same uncompressed
15//! `Data.db` offset domain (headerless NB starts the data section at file offset
16//! 0), so the same chunk-targeting arithmetic applies verbatim.
17//!
18//! No-heuristics (issue #28): the offset comes from `Index.db` and the covering
19//! chunk from `CompressionInfo` — chunk boundaries are never guessed. CRC / decode
20//! errors surface (the shared decode path runs the same per-chunk CRC as the scan),
21//! never swallowed (guardrail from #1411).
22
23use super::super::SSTableReader;
24use crate::types::{ScanRow, TableId};
25use crate::{Result, RowKey};
26
27impl SSTableReader {
28    /// Point-lookup entry for BIG ("nb"/uncompressed) readers (issue #1572).
29    ///
30    /// Resolution order:
31    /// 1. Bloom pre-check (unchanged) — a definite miss short-circuits to `None`.
32    ///    This is the fast definitive-absent path for the common case: the
33    ///    bloom filter answers "not present" for an absent key WITHOUT a scan.
34    /// 2. Fast path: `index_reader` (raw-key `Index.db` map) resolves the
35    ///    uncompressed partition offset; only the covering chunk(s) are read,
36    ///    CRC-checked, decompressed, and decoded.
37    /// 3. `scan_for_key` is the whole-file oracle. It is used for genuinely
38    ///    index-less readers (no `index_reader`); for the rare exact index hit
39    ///    that decodes to no matching row (schema-unavailable soft-miss or a
40    ///    benign per-row table-guard rejection); AND for EVERY `index_reader`
41    ///    MISS (issue #1572 correctness). An `Index.db` map miss is NOT treated
42    ///    as a definitive absent, because `IndexReader::open` opens a truncated /
43    ///    partially-corrupt Index.db with a PARTIAL prefix map and — critically —
44    ///    truncation aligned to an exact entry boundary (whole trailing entries
45    ///    dropped) leaves the prefix parsing cleanly to EOF, so it is
46    ///    indistinguishable from a complete map at parse time. There is no
47    ///    cleanly-available authoritative partition count at this layer to close
48    ///    that gap, so a miss conservatively falls back to the scan oracle,
49    ///    keeping `get()`/`scan()` in agreement on degraded inputs. The fast
50    ///    definitive-absent path for the common (non-degraded) case is the bloom
51    ///    pre-check in step 1, so this fallback fires only on a bloom
52    ///    false-positive — rare and acceptable. `SCAN_FOR_KEY_CALLS` stays
53    ///    observable on the fallback.
54    ///
55    /// `fully_qualified_match` is threaded through to the shared decode's per-row
56    /// table-consistency guard exactly as the BTI path does (issue #1321 / #1284).
57    pub(super) async fn big_get_with_resolution(
58        &self,
59        table_id: &TableId,
60        key: &RowKey,
61        fully_qualified_match: bool,
62    ) -> Result<Option<ScanRow>> {
63        use crate::observability::{self as obs, catalog};
64
65        // 1. Bloom pre-check (unchanged behaviour): a definite miss short-circuits.
66        if let Some(bloom_filter) = &self.bloom_filter {
67            let present = bloom_filter.might_contain(key.as_bytes());
68            obs::add_counter(
69                catalog::READ_BLOOM_CHECKS,
70                1,
71                &[
72                    (
73                        catalog::attr::RESULT,
74                        if present { "hit" } else { "miss" }.into(),
75                    ),
76                    (
77                        catalog::attr::SSTABLE_FORMAT,
78                        self.sstable_format_label().into(),
79                    ),
80                ],
81            );
82            if !present {
83                return Ok(None);
84            }
85        }
86
87        // 2. Fast path: resolve the partition offset via the raw-key Index.db map
88        //    and decode ONLY the covering chunk(s). The Index.db map is the
89        //    complete, authoritative partition set for a BIG SSTable.
90        if self.index_reader.is_some() {
91            match self.lookup_partition_with_index(key.as_bytes()).await? {
92                Some((data_offset, _size)) => {
93                    // `_size` (Index.db does not store partition size) is unused:
94                    // the shared chunk-targeted decode parses forward from the
95                    // offset until the partition is complete, so no size is needed.
96                    let schema_opt = self.get_table_schema(None);
97                    let parser = self.build_v5_parser(true);
98                    // Pass `data_offset` raw (NO `actual_header_size` add): the
99                    // chunk-targeted decode operates in the uncompressed data-section
100                    // domain that begins at offset 0 (and for `nb`
101                    // `actual_header_size == 0` anyway). The legacy whole-section
102                    // fallback below seeks the file past the header itself, so it must
103                    // add `actual_header_size`; this path must not.
104                    let found = self
105                        .bti_decompress_and_parse_target(
106                            data_offset as usize,
107                            key,
108                            table_id,
109                            fully_qualified_match,
110                            schema_opt.as_ref(),
111                            &parser,
112                        )
113                        .await?;
114                    if let Some(value) = found {
115                        if !self.filter_tombstone(&value) {
116                            return Ok(None);
117                        }
118                        return Ok(Some(value));
119                    }
120                    // Exact index hit but no matching row decoded (schema-unavailable
121                    // soft-miss / benign table-guard rejection). Fall through to the
122                    // safe scan_for_key oracle to preserve the pre-#1572 result.
123                }
124                None => {
125                    // Index.db map miss. An index miss is NOT treated as a
126                    // definitive absent (issue #1572 correctness): `IndexReader::open`
127                    // opens a truncated / partially-corrupt Index.db with a PARTIAL
128                    // prefix map, and truncation aligned to an EXACT entry boundary
129                    // (whole trailing entries dropped) leaves the surviving prefix
130                    // parsing cleanly to EOF — indistinguishable from a complete map
131                    // at parse time, with no cleanly-available authoritative partition
132                    // count at this layer to detect the loss. A partition whose entry
133                    // was dropped would then miss the map yet still be present in
134                    // Data.db — a silent get/scan divergence. So conservatively fall
135                    // back to the whole-file `scan_for_key` oracle, which keeps
136                    // `get()`/`scan()` in agreement and keeps the SCAN_FOR_KEY_CALLS
137                    // counter observable. The fast definitive-absent path for the
138                    // common (non-degraded) case is the bloom pre-check above; this
139                    // fallback fires only on a bloom false-positive (rare, acceptable).
140                    return self.scan_for_key(table_id, key).await;
141                }
142            }
143        }
144
145        // 3. Legacy fallbacks for index-less readers (or the rare hit-but-soft-miss).
146        //    NOTE: `self.index` (SSTableIndex) is keyed on key digests, so
147        //    `find_entry()` with raw key bytes misses; retained for the size==0
148        //    uncompressed handling and the writer-produced-index cases (issue #517).
149        if let Some(index) = &self.index {
150            if let Some(entry) = index.find_entry(table_id, key).await? {
151                if entry.size == 0 {
152                    log::debug!(
153                        "Index reports size=0 for key {:?}, using sequential scan fallback",
154                        key
155                    );
156                    return self.scan_for_key(table_id, key).await;
157                }
158                // Index offsets are relative to data section start - adjust for header.
159                let file_offset = entry.offset + self.actual_header_size as u64;
160                return self.read_value_at_offset(file_offset, entry.size).await;
161            }
162        }
163        self.scan_for_key(table_id, key).await
164    }
165}
