1//! Decoding for CQL scalar types that map to `ComparatorType::Custom` (issue #1627).
2//!
3//! `CqlType::Time`, `CqlType::Inet`, and the `json` schema type are represented as
4//! `ComparatorType::Custom("time")` / `Custom("inet")` / `Custom("json")` (see
5//! `ComparatorType::from_cql_type` and `CqlType::parse("json")`). Because the
6//! `Custom(name)` string is derived authoritatively from the schema, dispatching
7//! on it here is schema-driven — NOT a byte-pattern heuristic (no-heuristics
8//! mandate, issue #28).
9//!
10//! Wire formats mirror the authoritative decoders (`raw_value.rs` and the free
11//! `parse_value_with_comparator` in `comparator_value_parsing.rs`):
12//! * `time` — 8-byte big-endian `i64` nanoseconds-since-midnight → [`Value::Time`].
13//! * `inet` — raw address bytes, 4 for IPv4 or 16 for IPv6 (any other length is
14//!   rejected as corruption) → [`Value::Inet`].
15//! * `json` — UTF-8 text parsed into a `serde_json::Value` (invalid UTF-8 or
16//!   invalid JSON is rejected as corruption) → [`Value::Json`].
17//!
18//! Any genuinely-unknown custom type is preserved verbatim as [`Value::Blob`] —
19//! the only legitimate blob fallback.
20
21use crate::{types::Value, Error, Result};
22
23/// Decode a `ComparatorType::Custom(name)` value body into its typed `Value`.
24pub(super) fn decode_custom_scalar(name: &str, value_data: &[u8]) -> Result<Value> {
25    match name {
26        "time" => {
27            if value_data.len() == 8 {
28                let nanos = i64::from_be_bytes([
29                    value_data[0],
30                    value_data[1],
31                    value_data[2],
32                    value_data[3],
33                    value_data[4],
34                    value_data[5],
35                    value_data[6],
36                    value_data[7],
37                ]);
38                Ok(Value::Time(nanos))
39            } else {
40                Err(Error::corruption("Invalid time value length"))
41            }
42        }
43        "inet" => {
44            // CQL `inet` is valid only as a 4-byte IPv4 or 16-byte IPv6 address
45            // (Cassandra `InetAddressType`); reject any other length as corruption
46            // rather than surfacing a malformed address (mirrors the `time` arm).
47            if value_data.len() == 4 || value_data.len() == 16 {
48                Ok(Value::Inet(value_data.to_vec()))
49            } else {
50                Err(Error::corruption("Invalid inet value length"))
51            }
52        }
53        "json" => {
54            // Mirrors the authoritative `ComparatorType::Json` arm in
55            // `comparator_value_parsing::parse_value_with_comparator`: parse the
56            // body as UTF-8, then as a JSON document.
57            let json_text = std::str::from_utf8(value_data)
58                .map_err(|_| Error::corruption("Invalid UTF-8 in JSON value"))?;
59            let json_value: serde_json::Value = serde_json::from_str(json_text)
60                .map_err(|_| Error::corruption("Invalid JSON value"))?;
61            Ok(Value::Json(json_value))
62        }
63        // Genuinely-unknown custom type: preserve raw bytes verbatim.
64        _ => Ok(Value::Blob(value_data.to_vec())),
65    }
66}

68#[cfg(test)]
69mod tests {
70    use super::*;
71
72    #[test]
73    fn time_decodes_8_byte_be_nanos() {
74        let t: i64 = 4_500_000_000_000;
75        assert_eq!(
76            decode_custom_scalar("time", &t.to_be_bytes()).unwrap(),
77            Value::Time(t)
78        );
79    }
80
81    #[test]
82    fn time_wrong_length_errors() {
83        assert!(decode_custom_scalar("time", &[0u8; 4]).is_err());
84    }
85
86    #[test]
87    fn inet_ipv4_and_ipv6_preserved() {
88        assert_eq!(
89            decode_custom_scalar("inet", &[10, 0, 0, 1]).unwrap(),
90            Value::Inet(vec![10, 0, 0, 1])
91        );
92        let v6 = vec![0u8; 16];
93        assert_eq!(
94            decode_custom_scalar("inet", &v6).unwrap(),
95            Value::Inet(v6.clone())
96        );
97    }
98
99    #[test]
100    fn inet_wrong_length_errors() {
101        // Neither 4 (IPv4) nor 16 (IPv6) bytes: must be rejected as corruption.
102        assert!(decode_custom_scalar("inet", &[0u8; 5]).is_err());
103        assert!(decode_custom_scalar("inet", &[]).is_err());
104        assert!(decode_custom_scalar("inet", &[0u8; 15]).is_err());
105    }
106
107    #[test]
108    fn json_decodes_to_typed_value() {
109        assert_eq!(
110            decode_custom_scalar("json", br#"{"a":1}"#).unwrap(),
111            Value::Json(serde_json::json!({"a": 1}))
112        );
113        assert_eq!(
114            decode_custom_scalar("json", b"[1,2,3]").unwrap(),
115            Value::Json(serde_json::json!([1, 2, 3]))
116        );
117    }
118
119    #[test]
120    fn json_invalid_errors() {
121        // Not valid JSON.
122        assert!(decode_custom_scalar("json", b"{not json").is_err());
123        // Valid UTF-8 but empty is not a JSON document.
124        assert!(decode_custom_scalar("json", b"").is_err());
125        // Invalid UTF-8 bytes.
126        assert!(decode_custom_scalar("json", &[0xff, 0xfe]).is_err());
127    }
128
129    #[test]
130    fn unknown_custom_falls_back_to_blob() {
131        assert_eq!(
132            decode_custom_scalar("some_udt_marshaller", &[1, 2, 3]).unwrap(),
133            Value::Blob(vec![1, 2, 3])
134        );
135    }
136}
