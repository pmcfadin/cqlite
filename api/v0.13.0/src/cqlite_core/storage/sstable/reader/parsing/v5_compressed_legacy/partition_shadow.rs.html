1//! Read-side tombstone/TTL shadowing for a single partition (issue #1741).
2//!
3//! Single-generation / full-scan reads historically bypassed reconciliation and
4//! returned partition-deleted, range-tombstoned, or TTL-expired data as live. This
5//! helper restores Cassandra `SELECT` semantics on the READ path without routing
6//! through the write-support `ReconcileState`: it captures the partition-level
7//! deletion, tracks the currently-open range tombstone as the row stream is walked
8//! in clustering order, and decides per row whether a user-facing scan must hide it.
9//!
10//! It is **un-gated** — NOT behind the `write-support` feature — because read
11//! correctness must not depend on a write feature (AC2). It is shared by every
12//! user-facing emit path so there is ONE read-side shadowing implementation.
13//!
14//! Timestamp comparisons follow Cassandra `DeletionTime.deletes(ts) = ts <=
15//! markedForDeleteAt`: a deletion at `d` shadows the whole row iff every piece of
16//! the row's data is older than (or equal to) `d`. A row that still carries a cell
17//! strictly newer than the deletion survives (the deletion shadows only the
18//! already-absent older cells), matching the row-tombstone coexistence rule (#932).
19

22/// Authoritative per-clustering-column reversal flags for `schema`, in schema
23/// clustering-key order: `true` for a column declared `CLUSTERING ORDER BY
24/// (... DESC)`, `false` for ASC (issue #1741). Threaded into [`PartitionShadow`]
25/// so range-tombstone coverage compares clustering prefixes in physical storage
26/// order. Same `is_reversed` derivation the BTI reader uses
27/// (`data_access/bti.rs`). No-heuristics: schema clustering order only.
28pub(super) fn clustering_reversed_flags(schema: &crate::schema::TableSchema) -> Vec<bool> {
29    schema
30        .clustering_keys
31        .iter()
32        .map(|c| matches!(c.order, crate::schema::ClusteringOrder::Desc))
33        .collect()
34}

36/// Per-partition read-side shadowing state (issue #1741).
37pub(super) struct PartitionShadow {
38    /// Partition-level `markedForDeleteAt` (µs), or `None` when the partition is live.
39    partition_deletion: Option<i64>,
40    /// The currently-open range tombstone: `(start_bound_values, start_inclusive,
41    /// deleted_at_µs)`. Cassandra writes non-overlapping range tombstones in
42    /// clustering order — a start marker opens a range, an end marker closes it, and
43    /// a boundary marker closes one and opens the next — so at most ONE range is open
44    /// at a time. Every row physically between the open start marker and its
45    /// (not-yet-seen) end marker falls inside the range, so a row is covered while
46    /// the range is open iff it is at/after the start bound.
47    open_range: Option<(Vec<Value>, bool, i64)>,
48    /// Current wall-clock, epoch seconds, for read-time TTL expiry.
49    now_secs: i64,
50    /// Per-clustering-column reversal flags in schema order: `true` for a column
51    /// declared `CLUSTERING ORDER BY (... DESC)`. Range-tombstone coverage must
52    /// compare a row's clustering prefix against the range bounds in PHYSICAL
53    /// storage order, which for a DESC column is the reverse of value order — so
54    /// this carries the authoritative schema clustering order into the coverage
55    /// FSM. A missing/short entry is treated as ASC (`false`), matching the
56    /// `is_reversed` idiom used by the sibling reader code (`data_access/bti.rs`).
57    /// No-heuristics: sourced only from `TableSchema::clustering_keys`.
58    clustering_reversed: Vec<bool>,
59}
60
61impl PartitionShadow {
62    /// Open shadowing for a partition given its decoded partition-level deletion
63    /// (`Some((markedForDeleteAt_µs, localDeletionTime_s))` when the partition
64    /// carries a tombstone, `None` when live), the current epoch-seconds clock, and
65    /// the per-clustering-column reversal flags (`true` = DESC) in schema order.
66    pub(super) fn open(
67        now_secs: i64,
68        partition_deletion: Option<(i64, i32)>,
69        clustering_reversed: Vec<bool>,
70    ) -> Self {
71        Self {
72            partition_deletion: partition_deletion.map(|(mfda, _ldt)| mfda),
73            open_range: None,
74            now_secs,
75            clustering_reversed,
76        }
77    }
78
79    /// Feed one decoded range-tombstone bound marker, running the same start/end/
80    /// boundary FSM the delta-scan path uses. `bound_kind` is the raw Cassandra
81    /// `ClusteringPrefix.Kind` ordinal; `deleted_at_secondary` is present only for
82    /// boundary markers (kind 2/5) and carries the new range's deletion time.
83    pub(super) fn feed_range_marker(
84        &mut self,
85        bound_values: Vec<Value>,
86        bound_kind: u8,
87        deleted_at_primary: i64,
88        deleted_at_secondary: Option<i64>,
89    ) -> Result<()> {
90        match bound_kind {
91            // Simple start bound: open a range (INCL_START=1, EXCL_START=7).
92            1 | 7 => {
93                self.open_range = Some((bound_values, bound_kind == 1, deleted_at_primary));
94            }
95            // Simple end bound: close the open range (EXCL_END=0, INCL_END=6).
96            0 | 6 => {
97                self.open_range = None;
98            }
99            // EXCL_END_INCL_START_BOUNDARY: close prev, open new range (inclusive start).
100            2 => {
101                let d = deleted_at_secondary.unwrap_or(deleted_at_primary);
102                self.open_range = Some((bound_values, true, d));
103            }
104            // INCL_END_EXCL_START_BOUNDARY: close prev, open new range (exclusive start).
105            5 => {
106                let d = deleted_at_secondary.unwrap_or(deleted_at_primary);
107                self.open_range = Some((bound_values, false, d));
108            }
109            unknown => {
110                return Err(Error::corruption(format!(
111                    "read-shadow: unknown range tombstone bound kind {unknown} — cannot represent \
112                     faithfully (no-heuristics mandate, issue #28)"
113                )));
114            }
115        }
116        Ok(())
117    }
118
119    /// `true` when the row's clustering values must be extracted for a coverage
120    /// check — i.e. a range tombstone is currently open. The tombstone-free common
121    /// case returns `false`, so the caller skips the per-row clustering clone.
122    pub(super) fn needs_clustering(&self) -> bool {
123        self.open_range.is_some()
124    }
125
126    /// `true` when a user-facing `SELECT` must HIDE this row: it is shadowed by the
127    /// partition tombstone or the open range tombstone (all of the row's data is
128    /// older than the deletion), or it is TTL-expired. `clustering` is the row's
129    /// clustering-key values (needed only for range coverage; pass an empty slice
130    /// when [`Self::needs_clustering`] is `false`).
131    pub(super) fn row_hidden(&self, header: &RowHeader, clustering: &[Value]) -> bool {
132        // Partition-level deletion shadows every row whose data predates it.
133        if let Some(d) = self.partition_deletion {
134            if header.shadowed_by_deletion_at(d) {
135                return true;
136            }
137        }
138        // Open range tombstone shadows covered rows whose data predates it.
139        if let Some((start, inclusive, d)) = &self.open_range {
140            if self.at_or_after_start(clustering, start, *inclusive)
141                && header.shadowed_by_deletion_at(*d)
142            {
143                return true;
144            }
145        }
146        // Read-time TTL expiry.
147        header.row_liveness_expired(self.now_secs)
148    }
149
150    /// Whether `row_ck` lies at/after the range's start bound in PHYSICAL storage
151    /// order. While a range is open its end has not yet been seen, so a covered row
152    /// need only satisfy the start bound. Comparison honours the per-column
153    /// clustering order: for a DESC column the physical order is the reverse of
154    /// value order, so we reverse that component's comparison — the same rule
155    /// [`crate::storage::write_engine::mutation::ClusteringKey::compare`] applies
156    /// (`ordering.reverse()` when the column is `ClusteringOrder::Desc`). Using the
157    /// threaded `clustering_reversed` flags keeps this schema-authoritative with no
158    /// per-row allocation (we compare the borrowed `Value` slices in place rather
159    /// than materialising `ClusteringKey`/`TableSchema` per row). A prefix start
160    /// bound (fewer components than the clustering arity) treats its missing
161    /// components as −∞, per Cassandra clustering-bound semantics.
162    fn at_or_after_start(&self, row_ck: &[Value], start: &[Value], inclusive: bool) -> bool {
163        // Unbounded (open) start covers everything.
164        if start.is_empty() {
165            return true;
166        }
167        for (i, (a, b)) in row_ck.iter().zip(start.iter()).enumerate() {
168            // Clustering-key values share a type per component, so `partial_cmp`
169            // yields a total order here; fall back to Equal defensively.
170            let mut ord = a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal);
171            // DESC column: physical order is the reverse of value order.
172            if self.clustering_reversed.get(i).copied().unwrap_or(false) {
173                ord = ord.reverse();
174            }
175            match ord {
176                std::cmp::Ordering::Less => return false,
177                std::cmp::Ordering::Greater => return true,
178                std::cmp::Ordering::Equal => {}
179            }
180        }
181        // All compared components equal.
182        match row_ck.len().cmp(&start.len()) {
183            // Row extends a shorter (prefix) start bound: its missing components are
184            // −∞, so the row is strictly after the start ⇒ covered.
185            std::cmp::Ordering::Greater => true,
186            // Row is a prefix of a longer start bound (fewer clustering components
187            // than the bound) — not a well-formed data row; treat as before start.
188            std::cmp::Ordering::Less => false,
189            // Exact boundary point: covered only when the start bound is inclusive.
190            std::cmp::Ordering::Equal => inclusive,
191        }
192    }
193
194    /// The effective covering deletion timestamp (µs) for a row with clustering prefix
195    /// `clustering`: the partition tombstone `markedForDeleteAt`, folded with the open
196    /// range tombstone's `deleted_at` when the row falls inside the range. `None` when
197    /// nothing covers the row. Shared by per-cell shadow filtering (issue #1741,
198    /// Finding 1) and the primary-key-liveness decision below.
199    pub(super) fn covering_deleted_at(&self, clustering: &[Value]) -> Option<i64> {
200        let mut cover = self.partition_deletion;
201        if let Some((start, inclusive, d)) = &self.open_range {
202            if self.at_or_after_start(clustering, start, *inclusive) {
203                cover = Some(cover.map_or(*d, |c| c.max(*d)));
204            }
205        }
206        cover
207    }
208
209    /// Per-cell shadow context for the row-data cell loop (issue #1741, Finding 1):
210    /// `(covering_deleted_at, now_secs)`. A data cell is dropped when its effective
211    /// write ts <= `covering_deleted_at`, or it is TTL-expired at `now_secs`.
212    pub(super) fn cell_context(&self, clustering: &[Value]) -> (Option<i64>, i64) {
213        (self.covering_deleted_at(clustering), self.now_secs)
214    }
215
216    /// Whether a data cell is dropped by read-side shadow/TTL filtering (issue #1741,
217    /// Finding 1): its effective write ts is shadowed by the covering deletion
218    /// (`eff_ts <= cover`, matching `DeletionTime.deletes`), OR it is TTL-expired at
219    /// `now` (`eff_exp <= now`). A cell with no authoritative write ts is NEVER
220    /// shadowed (no-heuristics: we hide only what authoritative metadata proves
221    /// stale). Associated function so both the row-data cell loop and its unit pins
222    /// share one decision.
223    pub(super) fn cell_shadowed_or_expired(
224        cover: Option<i64>,
225        now: i64,
226        eff_ts: Option<i64>,
227        eff_exp: Option<i64>,
228    ) -> bool {
229        let shadowed = matches!((cover, eff_ts), (Some(d), Some(t)) if t <= d);
230        let expired = eff_exp.is_some_and(|e| e <= now);
231        shadowed || expired
232    }
233}

492    #[test]

494        let shadow = PartitionShadow::open(0, Some((2000, 12345)), vec![]);
495        // Row marker older than the deletion (1000 <= 2000) but a folded element ts
496        // (3000) newer than it → NOT shadowed → visible.
497        let mut h = row(1000, None, None, false);
498        h.max_data_cell_timestamp = Some(3000);
499        assert!(!shadow.row_hidden(&h, &[]));
500        // Without the fold (max cell ts == row ts == 1000 <= 2000) → shadowed.
501        let mut h_prefold = row(1000, None, None, false);
502        h_prefold.max_data_cell_timestamp = Some(1000);
503        assert!(shadow.row_hidden(&h_prefold, &[]));
504    }
