3impl V5CompressedLegacyParser {
4    /// Within-partition clustering-slice variant of [`parse_block_emit`] (Issue
5    /// #954, Epic #951).
6    ///
7    /// When `row_body_window` is `Some((body_start, body_end))` (byte offsets
8    /// into `data`, in the SAME domain as `data` indices) the FIRST partition's
9    /// row-body parse is bounded to that window:
10    ///   - after the partition header is decoded, the row cursor is fast-forwarded
11    ///     to `max(after_header, body_start)` (skipping rows that precede the
12    ///     requested clustering slice), and
13    ///   - the row loop stops once the cursor reaches `body_end` (skipping rows
14    ///     after the slice).
15    ///
16    /// `body_start`/`body_end` are the byte extent of the row-index block(s) that
17    /// the authoritative BTI row index resolved as covering the requested
18    /// clustering range, so this decodes O(matched rows + index-block slack)
19    /// rather than the whole partition. The post-scan `evaluate_leaf` backstop
20    /// trims the block-granularity over-read, so the returned rows are a SUPERSET
21    /// of the exact slice and the final query output is byte-identical.
22    ///
23    /// The start fast-forward is only applied when the schema has NO static
24    /// columns (the caller enforces this): a static row precedes the clustering
25    /// rows and must be merged into each clustering row, so skipping past it would
26    /// drop static values. The end bound is always safe to apply.
27    ///
28    /// Every row this method actually decodes bumps the `work_counters`
29    /// `rows_decoded` counter so a test can prove the decode was bounded to the
30    /// slice. With `row_body_window == None` this is byte-for-byte
31    /// [`parse_block_emit`]'s original behaviour, and the counter is still bumped
32    /// (the seek path reports its full-partition decode too).
33    pub fn parse_block_emit_windowed<F>(
34        &self,
35        data: &[u8],
36        schema: Option<&TableSchema>,
37        reader: &crate::storage::sstable::reader::types::SSTableReader,
38        row_body_window: Option<(usize, usize)>,
39        mut emit: F,
40    ) -> Result<()>
41    where
42        F: FnMut((TableId, RowKey, ScanRow)) -> Result<std::ops::ControlFlow<()>>,
43    {
44        if data.is_empty() {
45            return Ok(());
46        }
47
48        // V5CompressedLegacy format stores cells WITHOUT column names,
49        // relying on schema to interpret the binary data. Schema is REQUIRED.
50        let schema = schema.ok_or_else(|| {
51            Error::schema(format!(
52                "V5CompressedLegacy format requires schema for {}.{} (cells lack column names in binary data)",
53                self.keyspace, self.table_name
54            ))
55        })?;
56
57        // Issue #1046: build the header→schema column resolution ONCE per block.
58        let resolution = RowColumnResolution::build(schema, reader);
59
60        log::debug!(
61            "V5CompressedLegacy: Parsing block for {}.{} ({} bytes)",
62            self.keyspace,
63            self.table_name,
64            data.len()
65        );
66        log::debug!(
67            "V5CompressedLegacy: Schema has {} columns",
68            schema.columns.len()
69        );
70        for (i, col) in schema.columns.iter().enumerate() {
71            log::debug!("  Column {}: {} ({})", i, col.name, col.data_type);
72        }
73        log::debug!(
74            "V5CompressedLegacy: First 64 bytes of data: {}",
75            hex::encode(&data[..std::cmp::min(64, data.len())])
76        );
77        debug!(
78            "V5CompressedLegacy: Parsing block for {}.{} ({} bytes)",
79            self.keyspace,
80            self.table_name,
81            data.len()
82        );
83
84        let mut emitted: usize = 0;
85        let mut offset = 0;
86        // Issue #1741 (F2): read-time TTL clock — captured ONCE per parser (== once per
87        // read/scan operation) in `V5CompressedLegacyParser::new`. The windowed scan
88        // driver reuses this parser across every block; sampling `self.now_secs` here
89        // (not the wall clock per block) keeps a boundary-crossing scan consistent.
90        let now_secs = self.now_secs;
91        let table_id = TableId::new(format!("{}.{}", self.keyspace, self.table_name));
92
93        // Cassandra partition key size limits (used in header validation)
94        // - CASSANDRA_MAX_KEY_SIZE: 64KB limit per Apache Cassandra specification
95        // - FORMAT_MAX_KEY_SIZE: u8 max value - V5CompressedLegacy format limitation
96        const CASSANDRA_MAX_KEY_SIZE: usize = 65536; // 64KB per Cassandra spec
97        const FORMAT_MAX_KEY_SIZE: usize = 255; // u8 max value - format limitation
98
99        // Parse ALL partitions in block (Issue #2 fix: previously only parsed one partition)
100        let mut partition_index = 0;
101        let mut skipped_partitions = 0;
102        while offset < data.len() {
103            log::debug!(
104                "V5CompressedLegacy: === PARTITION {} at offset {} (block size: {}) ===",
105                partition_index,
106                offset,
107                data.len()
108            );
109
110            // CRITICAL FIX (Issue #164): Validate partition header format before attempting parse
111            //
112            // Most compressed blocks contain EXACTLY ONE partition. After parsing the first
113            // partition's row data and trailing VInt, we should NOT assume there's another
114            // partition just because offset < data.len().
115            //
116            // Partition header format validation:
117            // - Byte 0: Flags (typically 0x00, sometimes has partition-level flags)
118            // - Byte 1: Partition key length (u8, typically 16 for UUID)
119            // - Bytes 2+: Partition key data
120            //
121            // If we don't see a valid partition header structure, we've reached the end
122            // of partitions in this block (remaining bytes are likely padding or metadata).
123            if offset >= data.len() {
124                break; // End of block
125            }
126
127            // Check if this looks like a partition header (flags byte + reasonable key length)
128            // Partition keys can be up to 64KB per Cassandra spec (composite keys, text, etc.)
129            if offset + 2 > data.len() {
130                log::debug!(
131                    "V5CompressedLegacy: Not enough bytes for partition header at offset {} (need 2, have {}), stopping",
132                    offset,
133                    data.len() - offset
134                );
135                break;
136            }
137
138            let flags = data[offset];
139            let key_len = data[offset + 1] as usize;
140
141            // Validate partition header:
142            // - Key length must be non-zero and within format's limit (u8 max = 255 bytes)
143            //   Note: Cassandra spec allows 64KB keys, but V5CompressedLegacy format uses u8 length
144            // - Must have enough bytes for the header (size depends on format version)
145            //
146            // VG3: oa format (hasUIntDeletionTime) uses a compact DeletionTime:
147            //   LIVE = 1 byte; DELETED = 12 bytes.  The minimum is therefore 1 byte.
148            // nb format always uses 12 bytes (4 + 8).
149            // NOTE: No heuristic validation of flags (Issue #258, #28 no-heuristics mandate)
150            let deletion_time_min = if self.has_uint_deletion_time() { 1 } else { 12 };
151            let header_min_size = 1 + 1 + key_len + deletion_time_min;
152            if key_len == 0
153                || key_len > FORMAT_MAX_KEY_SIZE.min(CASSANDRA_MAX_KEY_SIZE)
154                || offset + header_min_size > data.len()
155            {
156                log::warn!(
157                    "V5CompressedLegacy: Skipping malformed partition header at offset {} \
158                     (flags=0x{:02x}, key_len={}, need {} bytes, have {}, partition={}): header validation failed",
159                    offset,
160                    flags,
161                    key_len,
162                    header_min_size,
163                    data.len() - offset,
164                    partition_index
165                );
166                // Try to skip to next potential partition boundary
167                skipped_partitions += 1;
168                offset += 1; // Minimal forward progress to avoid infinite loop
169                continue; // Skip this partition, try next
170            }
171
172            // Try to parse partition header
173            match self.parse_partition_header_full(data, offset) {
174                Ok((partition_key, new_offset, partition_deletion)) => {
175                    let header_size = new_offset - offset;
176                    offset = new_offset;
177
178                    // Issue #1741: per-partition read-side shadowing, active ONLY for
179                    // user-facing query reads (`self.read_shadowing`). Captures the
180                    // partition-level deletion and tracks the open range tombstone as
181                    // rows are walked, so partition/range-tombstone-shadowed and
182                    // TTL-expired rows are hidden (matching Cassandra SELECT). `None`
183                    // for physical consumers (verify / get_all_entries) which must see
184                    // every on-disk row. Un-gated (not behind write-support).
185                    let mut shadow = self.read_shadowing.then(|| {
186                        PartitionShadow::open(
187                            now_secs,
188                            partition_deletion,
189                            clustering_reversed_flags(schema),
190                        )
191                    });
192
193                    // Issue #954: when a within-partition clustering-slice window is
194                    // supplied, fast-forward the row cursor of the FIRST partition to
195                    // the first row-index block covering the slice (skipping rows
196                    // before it). Applied only to `partition_index == 0` because the
197                    // seek decodes exactly one target partition. The end bound
198                    // (`body_end`) stops the row loop below. The caller guarantees a
199                    // start fast-forward is only requested when the schema has no
200                    // static columns, so this never skips a static prefix.
201                    let row_body_end = match row_body_window {
202                        Some((body_start, body_end)) if partition_index == 0 => {
203                            if body_start > offset && body_start <= data.len() {
204                                // Issue #1741 (Finding 1): a range tombstone that
205                                // OPENS before `body_start` can cover rows inside the
206                                // requested clustering slice. Fast-forwarding straight
207                                // to `body_start` would skip those markers, so the open
208                                // range would never be fed into `PartitionShadow` and a
209                                // slice read could return rows a full scan correctly
210                                // hides. When shadowing is active, replay the markers
211                                // from the partition body start up to `body_start` into
212                                // the shadow FSM first. If priming cannot faithfully
213                                // reconstruct the state (unrepresentable marker or an
214                                // undecodable row), fall back to a full-partition decode
215                                // from the body start rather than skip the markers: the
216                                // post-scan clustering backstop still trims the rows
217                                // before the slice, so correctness holds and only the
218                                // fast path is lost. When shadowing is off (physical
219                                // read) there is nothing to prime — keep the byte-for-
220                                // byte fast-forward (no-RT fast path unaffected).
221                                //
222                                // Issue #1741 (Finding 2): priming decodes the pre-window
223                                // prefix, regressing the row-index fast-forward from
224                                // O(slice) to O(prefix+slice) on a wide partition. Pay it
225                                // ONLY when a range tombstone can actually reach the slice.
226                                // Partition-deletion shadowing needs NO prefix decode (it is
227                                // captured from the header into `shadow`). Range tombstones:
228                                // when the SSTable's authoritative EncodingStats prove there
229                                // are no deletions (hence no range tombstones), keep the
230                                // O(slice) `offset = body_start` fast-forward and skip
231                                // priming. Priming itself is now marker-ONLY — it skips row
232                                // bodies via framing (no cell decode), feeding only the RT
233                                // markers into the FSM — so even when it runs it never
234                                // re-decodes the prefix cells.
235                                let primed = match shadow.as_mut() {
236                                    Some(_) if !self.sstable_may_have_range_tombstones() => true,
237                                    Some(sh) => self.prime_shadow_before_window(
238                                        data, offset, body_start, schema, sh,
239                                    ),
240                                    None => true,
241                                };
242                                if primed {
243                                    offset = body_start;
244                                }
245                            }
246                            Some(body_end)
247                        }
248                        _ => None,
249                    };
250
251                    log::debug!(
252                        "V5CompressedLegacy: Partition {} - Parsed partition key: {} bytes (header consumed {} bytes, now at offset {})",
253                        partition_index,
254                        partition_key.0.len(),
255                        header_size,
256                        offset
257                    );
258                    log::debug!(
259                        "V5CompressedLegacy: Partition {} - Row data starts at offset {}, remaining: {} bytes",
260                        partition_index,
261                        offset,
262                        data.len() - offset
263                    );
264                    log::debug!(
265                        "V5CompressedLegacy: Partition {} - Row data hex (first 128 bytes): {}",
266                        partition_index,
267                        hex::encode(&data[offset..std::cmp::min(offset + 128, data.len())])
268                    );
269
270                    debug!(
271                        "V5CompressedLegacy: Parsed partition key: {} bytes, now at offset {}",
272                        partition_key.0.len(),
273                        offset
274                    );
275
276                    // Parse ALL rows in this partition (Issue #166 fix: multi-row partition support)
277                    //
278                    // V5CompressedLegacy partitions can contain multiple rows with different clustering keys.
279                    // We use structural parsing (peek_is_partition_header) to detect partition boundaries,
280                    // not flag value heuristics (Issue #258, #28 no-heuristics mandate).
281                    // We parse rows in a loop until we encounter:
282                    // - End of block (offset >= data.len())
283                    // - END_OF_PARTITION marker (flags == 0x01, Issue #229 fix)
284                    // - Next partition header (detected via peek_is_partition_header)
285                    // - Parse error (invalid row data)
286
287                    // Issue #480 FIX: Static cell handling
288                    //
289                    // Cassandra static rows are stored once per partition (before clustering rows).
290                    // They should NOT be emitted as separate result entries — instead their column
291                    // values must be merged into each clustering row that follows in the partition.
292                    //
293                    // We accumulate static cells here and inject them into every clustering row.
294                    let mut static_cells: HashMap<Arc<str>, Value> = HashMap::new();
295                    let mut row_count = 0;
296                    loop {
297                        // Issue #954: stop at the clustering-slice end bound. The
298                        // row-index block extent (`body_end`) is the authoritative
299                        // upper byte bound of the rows that may fall in the requested
300                        // clustering range; rows past it are outside the slice, so we
301                        // stop decoding (the post-scan backstop already trims any
302                        // block-granularity over-read within the window).
303                        if let Some(body_end) = row_body_end {
304                            if offset >= body_end {
305                                break;
306                            }
307                        }
308
309                        // Issue #229 FIX: Check for END_OF_PARTITION marker BEFORE attempting row parse
310                        //
311                        // Per Cassandra's UnfilteredSerializer.java (lines 102, 730-732):
312                        // When END_OF_PARTITION (0x01) is set in the flags byte, nothing follows.
313                        // The partition is complete and we should move to the next partition.
314                        if offset < data.len() && Self::is_end_of_partition(data[offset]) {
315                            log::debug!(
316                                "V5CompressedLegacy: Partition {} complete via END_OF_PARTITION marker at offset {} ({} rows parsed)",
317                                partition_index, offset, row_count
318                            );
319                            offset += 1; // Skip the END_OF_PARTITION marker byte
320                            break; // Move to next partition
321                        }
322
323                        // Issue #229 FIX: Check for range tombstone marker
324                        //
325                        // Per Cassandra's UnfilteredSerializer.java (lines 103, 735-738):
326                        // When IS_MARKER (0x02) is set, this is a range tombstone boundary, not a row.
327                        // We skip these markers for now (full implementation would parse deletion ranges).
328                        if offset < data.len() && Self::is_range_tombstone_marker(data[offset]) {
329                            log::debug!(
330                                "V5CompressedLegacy: Range tombstone marker at offset {} (partition {}), skipping",
331                                offset, partition_index
332                            );
333                            // Issue #1741: when read-side shadowing is active, decode
334                            // the marker (bounds + deletion time) and feed the
335                            // range-tombstone FSM so covered rows are shadowed;
336                            // otherwise (physical read) only advance past it.
337                            if let Some(sh) = shadow.as_mut() {
338                                match self.parse_range_tombstone_marker_full(data, offset, schema) {
339                                    Ok((
340                                        bound_values,
341                                        bound_kind,
342                                        del_primary,
343                                        del_secondary,
344                                        next_offset,
345                                    )) => {
346                                        if let Err(e) = sh.feed_range_marker(
347                                            bound_values,
348                                            bound_kind,
349                                            del_primary,
350                                            del_secondary,
351                                        ) {
352                                            log::debug!(
353                                                "V5CompressedLegacy: range tombstone FSM error at offset {}: {}",
354                                                offset, e
355                                            );
356                                            break; // Unrepresentable marker, end partition
357                                        }
358                                        offset = next_offset;
359                                        continue; // Continue to next row/marker
360                                    }
361                                    Err(e) => {
362                                        log::debug!(
363                                            "V5CompressedLegacy: Failed to parse range tombstone marker at offset {}: {}",
364                                            offset, e
365                                        );
366                                        break; // Can't parse marker, end partition
367                                    }
368                                }
369                            }
370                            match self.skip_range_tombstone_marker(data, offset, schema) {
371                                Ok(next_offset) => {
372                                    offset = next_offset;
373                                    continue;
374                                }
375                                Err(e) => {
376                                    log::debug!(
377                                        "V5CompressedLegacy: Failed to skip range tombstone marker at offset {}: {}",
378                                        offset, e
379                                    );
380                                    break;
381                                }
382                            }
383                        }
384
385                        match self.parse_row_data_with_offset(
386                            data,
387                            offset,
388                            Some(schema),
389                            reader,
390                            false,
391                            &resolution,
392                            shadow.as_ref(),
393                        ) {
394                            Ok((
395                                mut cells,
396                                _row_cell_meta,
397                                row_header_opt,
398                                next_offset,
399                                is_static,
400                                _complex_meta,
401                            )) => {
402                                // Update offset to point to the next row or partition
403                                offset = next_offset;
404                                row_count += 1;
405
406                                log::debug!(
407                                    "V5CompressedLegacy: Partition {} Row {} - Parsed {} cells, now at offset {} (is_static={})",
408                                    partition_index,
409                                    row_count,
410                                    cells.len(),
411                                    offset,
412                                    is_static
413                                );
414
415                                if let Some(ref header) = row_header_opt {
416                                    log::debug!(
417                                        "V5CompressedLegacy: Row {} metadata - timestamp={:?}, ttl={:?}, deletion={:?}",
418                                        row_count,
419                                        header.timestamp, header.ttl, header.local_deletion_time
420                                    );
421                                }
422
423                                debug!(
424                                    "V5CompressedLegacy: Parsed {} cells from row {} (is_static={})",
425                                    cells.len(),
426                                    row_count,
427                                    is_static
428                                );
429
430                                // Issue #480 FIX: Static row handling
431                                //
432                                // Static rows are stored once per partition and contain values for
433                                // STATIC columns (e.g. `static_data TEXT STATIC`). They must NOT
434                                // be emitted as standalone result rows. Instead, store the static
435                                // column values and merge them into each subsequent clustering row.
436                                if is_static {
437                                    log::debug!(
438                                        "V5CompressedLegacy: Partition {} - Storing {} static cells for merging into clustering rows",
439                                        partition_index,
440                                        cells.len()
441                                    );
442                                    // Issue #1741 (Finding 1): a static row can itself be
443                                    // shadowed by the partition tombstone (static write ts
444                                    // <= markedForDeleteAt) or expired by its own TTL. If
445                                    // so its cells are STALE and must NOT be merged into a
446                                    // surviving clustering row, or a SELECT would resurface
447                                    // the deleted/expired static value. Static rows carry no
448                                    // clustering key, so an open range tombstone never covers
449                                    // them (empty clustering). No-op when shadowing is off.
450                                    let static_hidden = shadow.as_ref().is_some_and(|sh| {
451                                        row_header_opt
452                                            .as_ref()
453                                            .is_some_and(|h| sh.row_hidden(h, &[]))
454                                    });
455                                    static_cells =
456                                        if static_hidden { HashMap::new() } else { cells };
457                                    // Do NOT push to results — static rows are not result rows
458                                    // Continue to next row/marker in partition
459                                } else {
460                                    // Merge static cells into this clustering row (Issue #480)
461                                    for (k, v) in &static_cells {
462                                        cells.entry(k.clone()).or_insert_with(|| v.clone());
463                                    }
464
465                                    // Issue #1741: hide rows shadowed by a partition or
466                                    // range tombstone, or expired by TTL, matching a
467                                    // Cassandra SELECT. Active only for query reads
468                                    // (shadow is `Some`). A row reduced to only its
469                                    // primary key by per-cell filtering is hidden here
470                                    // too: the dropped cells still fold into the row
471                                    // aggregate, so `row_hidden` sees it shadowed/expired.
472                                    let hidden = shadow.as_ref().is_some_and(|sh| {
473                                        row_header_opt.as_ref().is_some_and(|h| {
474                                            let clustering = if sh.needs_clustering() {
475                                                extract_clustering_values(&cells, schema)
476                                            } else {
477                                                Vec::new()
478                                            };
479                                            sh.row_hidden(h, &clustering)
480                                        })
481                                    });
482
483                                    // Issue #505/#932: row-tombstone display rule now
484                                    // lives in the shared `build_display_row` helper.
485                                    let row_value =
486                                        build_display_row(cells, row_header_opt.as_ref(), schema);
487
488                                    // Issue #954: count each clustering row actually
489                                    // decoded out of Data.db so a slice query can be
490                                    // proven to decode O(matched rows + index block),
491                                    // not the whole partition. Counted at the row
492                                    // grain (here), distinct from the per-partition
493                                    // `partitions_decoded`. Gated `not(tombstones)`
494                                    // because the mutator is compiled only there.
495                                    #[cfg(not(feature = "tombstones"))]
496                                    crate::storage::sstable::work_counters::add_rows_decoded(1);
497
498                                    if !hidden {
499                                        match emit((
500                                            table_id.clone(),
501                                            partition_key.clone(),
502                                            row_value,
503                                        ))? {
504                                            std::ops::ControlFlow::Continue(()) => emitted += 1,
505                                            // Consumer dropped (streaming receiver gone): stop parsing.
506                                            std::ops::ControlFlow::Break(()) => return Ok(()),
507                                        }
508                                    }
509                                }
510
511                                // Check if we're at the end of the partition
512                                if offset >= data.len() {
513                                    debug!(
514                                        "V5CompressedLegacy: Partition {} complete: {} rows parsed (end of block)",
515                                        partition_index, row_count
516                                    );
517                                    break; // End of block
518                                }
519
520                                // CRITICAL FIX (Issue #166): NO HEURISTICS - Try-parse approach
521                                //
522                                // Instead of guessing based on byte patterns (e.g., checking if flags <= 0x20
523                                // or validating key_len ranges), we ACTUALLY TRY TO PARSE the next structure.
524                                //
525                                // Why heuristics fail:
526                                // - Row with small value (e.g., boolean=0x0A) can look like key_len
527                                // - Row flags=0x00 or 0x20 pass "<= 0x20" checks meant for partitions
528                                // - Any byte-pattern guessing will eventually fail on edge cases
529                                //
530                                // The only reliable approach: try to parse as partition header.
531                                // If that succeeds, it's a partition. If it fails, continue with rows.
532                                if self.peek_is_partition_header(data, offset) {
533                                    debug!(
534                                        "V5CompressedLegacy: Partition {} complete: {} rows parsed (next partition detected at offset {})",
535                                        partition_index, row_count, offset
536                                    );
537                                    break; // Next partition starts here
538                                }
539
540                                // Peek failed - not a partition header, so continue parsing rows
541                                debug!(
542                                    "V5CompressedLegacy: Partition {} - Continuing to row {} at offset {} (peek confirmed this is NOT a partition header)",
543                                    partition_index, row_count + 1, offset
544                                );
545                            }
546                            Err(e) => {
547                                // End of valid data in partition
548                                debug!(
549                                    "V5CompressedLegacy: Partition {} ended after {} rows: {}",
550                                    partition_index, row_count, e
551                                );
552                                if row_count == 0 {
553                                    // If we couldn't parse even one row, log as error
554                                    log::error!(
555                                        "V5CompressedLegacy: Partition {} - Failed to parse first row at offset {}: {}",
556                                        partition_index, offset, e
557                                    );
558                                }
559                                break; // End of valid data in partition
560                            }
561                        }
562                    }
563
564                    partition_index += 1;
565                }
566                Err(e) => {
567                    log::warn!(
568                        "V5CompressedLegacy: Failed to parse partition header at offset {} \
569                         (partition={}): {}. Attempting to continue to next partition.",
570                        offset,
571                        partition_index,
572                        e
573                    );
574                    // Try to skip forward to find next partition
575                    skipped_partitions += 1;
576                    offset += 1;
577                    continue; // Skip this partition, try next
578                }
579            }
580        }
581
582        if skipped_partitions > 0 {
583            log::warn!(
584                "V5CompressedLegacy: Successfully parsed {} entries, skipped {} malformed partitions",
585                emitted,
586                skipped_partitions
587            );
588        }
589
590        debug!(
591            "V5CompressedLegacy: Parsed {} total entries from block",
592            emitted
593        );
594
595        Ok(())
596    }

598    /// Issue #1741 (Finding 1): replay the range-tombstone (and any other)
599    /// unfiltereds between the partition body start (`start`) and the
600    /// clustering-slice window start (`body_start`) into `shadow`, so an open
601    /// range tombstone that begins BEFORE the slice correctly shadows covered
602    /// rows inside it. Rows in this prefix are NOT emitted — only markers move
603    /// the FSM. `body_start` aligns to the start of an unfiltered (a BTI row-index
604    /// block boundary), so markers before it are fully contained in the prefix.
605    ///
606    /// Returns `true` when the prefix was replayed cleanly all the way to
607    /// `body_start` (the caller may then fast-forward). Returns `false` when the
608    /// state cannot be faithfully reconstructed (an unrepresentable marker, an
609    /// undecodable row framing, an early END_OF_PARTITION, or a non-advancing
610    /// parse); the caller then declines the fast-forward and decodes the full
611    /// partition, which is still correct because the post-scan backstop trims rows
612    /// before the slice.
613    ///
614    /// Issue #1741 (Finding 2): this is a MARKER-ONLY scan. Data rows in the prefix
615    /// are skipped via their framing (`skip_row_framing`) WITHOUT decoding any cell
616    /// values — only range-tombstone markers move the FSM. It therefore adds no
617    /// per-cell decode / allocation over the pre-window prefix; only the (cheap)
618    /// row framing (flags + clustering + row_size VInts) is parsed to advance. It
619    /// runs solely on the slice-read fast-forward branch when shadowing is active
620    /// AND the SSTable may contain range tombstones (see the call site gate).
621    fn prime_shadow_before_window(
622        &self,
623        data: &[u8],
624        start: usize,
625        body_start: usize,
626        schema: &TableSchema,
627        shadow: &mut PartitionShadow,
628    ) -> bool {
629        let mut offset = start;
630        while offset < body_start {
631            if offset >= data.len() {
632                return false;
633            }
634            let flags = data[offset];
635            // An END_OF_PARTITION before the window means the window is not in this
636            // partition — bail to the safe full-decode path.
637            if Self::is_end_of_partition(flags) {
638                return false;
639            }
640            if Self::is_range_tombstone_marker(flags) {
641                match self.parse_range_tombstone_marker_full(data, offset, schema) {
642                    Ok((bound_values, bound_kind, del_primary, del_secondary, next_offset)) => {
643                        if shadow
644                            .feed_range_marker(bound_values, bound_kind, del_primary, del_secondary)
645                            .is_err()
646                        {
647                            return false;
648                        }
649                        if next_offset <= offset {
650                            return false; // non-advancing parse guard
651                        }
652                        offset = next_offset;
653                        continue;
654                    }
655                    Err(_) => return false,
656                }
657            }
658            // A data row in the prefix: advance past it via framing ONLY (no cell
659            // decode), so the fast-forward stays O(prefix framing), not O(prefix
660            // cells). The prefix carries no static row (the caller only requests a
661            // start fast-forward when the schema has no static columns).
662            match self.skip_row_framing(data, offset, schema) {
663                Ok(next_offset) => {
664                    if next_offset <= offset {
665                        return false; // non-advancing parse guard
666                    }
667                    offset = next_offset;
668                }
669                Err(_) => return false,
670            }
671        }
672        true
673    }

675    /// Issue #1741 (Finding 2): compute the offset immediately after the row at
676    /// `offset` by parsing ONLY its framing — flags, clustering prefix, and the
677    /// `row_size` VInt — never decoding cell values. Mirrors the authoritative
678    /// offset arithmetic in `parse_row_data_with_offset`
679    /// (`next = row_metadata_offset + row_size_vint_len + row_size`) so a marker-only
680    /// prefix scan can skip data rows cheaply. Returns an error on truncation or an
681    /// out-of-range `row_size`, which the caller treats as "cannot prime".
682    pub(super) fn skip_row_framing(
683        &self,
684        data: &[u8],
685        offset: usize,
686        schema: &TableSchema,
687    ) -> Result<usize> {
688        let (row_flags, extended_flags, flags_size) = self.parse_row_flags(data, offset)?;
689        let after_flags = offset + flags_size;
690        let is_static = extended_flags
691            .map(|ef| (ef & EXTENDED_IS_STATIC) != 0)
692            .unwrap_or(false);
693        // Static rows carry no clustering prefix; regular rows do.
694        let row_metadata_offset = if is_static {
695            after_flags
696        } else {
697            let (_clustering, after_clustering) =
698                self.parse_clustering_prefix(data, after_flags, schema)?;
699            after_clustering
700        };
701        let (row_header, row_size) =
702            self.parse_row_metadata(data, row_metadata_offset, row_flags, extended_flags)?;
703        let next_offset = (row_metadata_offset + row_header.row_size_vint_len) + row_size as usize;
704        if next_offset > data.len() {
705            return Err(Error::corruption(
706                "prime_shadow: row framing extends past decompressed block".to_string(),
707            ));
708        }
709        Ok(next_offset)
710    }

712    /// Parse all partitions in a decompressed block, returning per-row timestamps.
713    ///
714    /// This is the compaction-specific variant of [`parse_block`].  It returns
715    /// `(TableId, RowKey, ScanRow, row_timestamp_micros)` so that the k-way merger
716    /// can perform timestamp-accurate last-write-wins ordering rather than
717    /// falling back to `SystemTime::now()`.
718    ///
719    /// Row tombstones are emitted as `Value::Tombstone(RowTombstone)` with their
720    /// actual `deletion_time`.  Cell tombstones within live rows are stored as
721    /// `Value::Tombstone(CellTombstone)` inside the `Value::Map`, again carrying
722    /// the cell-level deletion timestamp.
723    ///
724    /// The `row_timestamp_micros` in the returned tuple is the row-level write
725    /// timestamp decoded from the `HAS_TIMESTAMP` field in the row header
726    /// (`min_timestamp + delta`).  For row tombstones the same timestamp also
727    /// appears in `TombstoneInfo::deletion_time`.
728    ///
729    /// Normal user-facing scan/get paths should use [`parse_block`] instead.
730    /// (Issue #505)
731    pub fn parse_block_with_timestamps(
732        &self,
733        data: &[u8],
734        schema: Option<&TableSchema>,
735        reader: &crate::storage::sstable::reader::types::SSTableReader,
736    ) -> Result<Vec<(TableId, RowKey, ScanRow, i64)>> {
737        // Thin wrapper that collects the streaming emit variant into a Vec, so
738        // every existing caller/test is byte-for-byte unchanged (issue #827).
739        let mut results: Vec<(TableId, RowKey, ScanRow, i64)> = Vec::new();
740        self.parse_block_with_timestamps_emit(data, schema, reader, |entry| {
741            results.push(entry);
742            Ok(std::ops::ControlFlow::Continue(()))
743        })?;
744        Ok(results)
745    }

747    /// Streaming variant of [`parse_block_with_timestamps`]: invokes `emit` for
748    /// each parsed `(TableId, RowKey, ScanRow, row_timestamp_micros)` entry rather
749    /// than collecting into a `Vec`, so the compaction read path can forward
750    /// rows into a bounded channel without materialising the whole block at once
751    /// (issue #827). Returning `ControlFlow::Break` from `emit` stops parsing
752    /// early — used when the streaming consumer is dropped.
753    ///
754    /// The tombstone/timestamp semantics are byte-identical to
755    /// [`parse_block_with_timestamps`] (Issue #505/#533): a row tombstone is
756    /// emitted as `Value::Tombstone` carrying its `markedForDeleteAt`, and the
757    /// fourth tuple element is the row write timestamp for live rows.
758    pub fn parse_block_with_timestamps_emit<F>(
759        &self,
760        data: &[u8],
761        schema: Option<&TableSchema>,
762        reader: &crate::storage::sstable::reader::types::SSTableReader,
763        mut emit: F,
764    ) -> Result<()>
765    where
766        F: FnMut((TableId, RowKey, ScanRow, i64)) -> Result<std::ops::ControlFlow<()>>,
767    {
768        if data.is_empty() {
769            return Ok(());
770        }
771
772        let schema = schema.ok_or_else(|| {
773            Error::schema(format!(
774                "V5CompressedLegacy (compaction) format requires schema for {}.{}",
775                self.keyspace, self.table_name
776            ))
777        })?;
778
779        let mut offset = 0;
780        let mut skipped_partitions = 0;
781
782        // Wrap `emit` so a Break is observable here as well as inside the
783        // one-partition parser (which stops its inner row loop on Break). This
784        // lets the outer loop terminate promptly when the consumer is dropped.
785        // `Cell` so the wrapping closure can borrow it shared while the outer
786        // loop also reads it between calls.
787        let broke = std::cell::Cell::new(false);
788        let mut tracking_emit = |entry| -> Result<std::ops::ControlFlow<()>> {
789            let flow = emit(entry)?;
790            if matches!(flow, std::ops::ControlFlow::Break(())) {
791                broke.set(true);
792            }
793            Ok(flow)
794        };
795
796        while offset < data.len() {
797            match self.parse_one_partition_with_timestamps(
798                &data[offset..],
799                Some(schema),
800                reader,
801                // The whole block is present; never request a refill. A trailing
802                // parse failure is terminal here (matches the legacy
803                // `Err(_) => break` behaviour of the original loop).
804                //
805                // Compaction NEVER shadows: this parser is built without
806                // `read_shadowing`, so the merger sees the raw tombstones/expired
807                // cells it needs to reconcile across generations (#1741).
808                true,
809                &mut tracking_emit,
810            )? {
811                ParseStep::Emitted(consumed) => {
812                    if consumed == 0 {
813                        // Defensive: avoid an infinite loop on a zero-byte
814                        // partition (should not happen — a header is >= 2 bytes).
815                        skipped_partitions += 1;
816                        offset += 1;
817                    } else {
818                        offset += consumed;
819                    }
820                }
821                // `at_final_chunk = true` collapses NeedMore into Done: there is
822                // no further chunk to append, so a truncated tail is end-of-data.
823                ParseStep::NeedMore | ParseStep::Done => break,
824            }
825            // Propagate an early Break from `emit` (consumer dropped).
826            if broke.get() {
827                break;
828            }
829        }
830
831        if skipped_partitions > 0 {
832            log::warn!(
833                "V5CompressedLegacy (compaction): skipped {} malformed partitions",
834                skipped_partitions
835            );
836        }
837
838        Ok(())
839    }

841    /// Parse exactly ONE partition from the front of `data`, emitting each row
842    /// via `emit`, and report how the parse terminated (issue #827).
843    ///
844    /// This isolates the body of the outer partition loop so the sliding-window
845    /// compaction driver can drain one partition at a time and `drain(0..consumed)`
846    /// from its window between calls. The crucial distinction over the legacy
847    /// monolithic loop is `NeedMore` vs `Done`:
848    ///
849    /// - [`ParseStep::Emitted(consumed)`] — a full partition was parsed and
850    ///   terminated by an END_OF_PARTITION marker or a confirmed next-partition
851    ///   header. `consumed` bytes may be drained from the window.
852    /// - [`ParseStep::NeedMore`] — `data` is (possibly) truncated mid-partition
853    ///   and `!at_final_chunk`, so the caller must append the next chunk and
854    ///   retry. NEVER returned when `at_final_chunk` is true.
855    /// - [`ParseStep::Done`] — genuine end of partitions, or (when
856    ///   `at_final_chunk`) a trailing truncation that cannot be resolved by more
857    ///   data. Terminal.
858    ///
859    /// `at_final_chunk` flips a mid-partition parse failure between a refill
860    /// request (`NeedMore`) and a terminal stop. The legacy code conflated
861    /// parse-error with end-of-partitions (`Err(_) => break`); doing that
862    /// mid-stream would silently drop every partition after a chunk boundary, so
863    /// we return `NeedMore` whenever the buffer may simply be truncated and we
864    /// are not yet at the final chunk.
865    /// Read-side shadowing (issue #1741) is driven by `self.read_shadowing`: when the
866    /// parser was built for a user-facing query read, rows shadowed by a partition/
867    /// range tombstone or expired by TTL are dropped before they reach `emit`. The
868    /// compaction read path builds the parser WITHOUT shadowing (`read_shadowing ==
869    /// false`) — it needs the raw tombstones/expired cells preserved to reconcile
870    /// across generations. Un-gated: read correctness does not depend on `write-support`.
871    pub fn parse_one_partition_with_timestamps<F>(
872        &self,
873        data: &[u8],
874        schema: Option<&TableSchema>,
875        reader: &crate::storage::sstable::reader::types::SSTableReader,
876        at_final_chunk: bool,
877        emit: &mut F,
878    ) -> Result<ParseStep>
879    where
880        F: FnMut((TableId, RowKey, ScanRow, i64)) -> Result<std::ops::ControlFlow<()>>,
881    {
882        if data.is_empty() {
883            return Ok(ParseStep::Done);
884        }
885
886        let schema = schema.ok_or_else(|| {
887            Error::schema(format!(
888                "V5CompressedLegacy (compaction) format requires schema for {}.{}",
889                self.keyspace, self.table_name
890            ))
891        })?;
892
893        // Issue #1046: per-PARTITION resolution build (this driver is re-entered once
894        // per partition by the sliding-window caller; allocations scale with partition
895        // count, not row count). Borrows header strings + schema columns for the loop.
896        let resolution = RowColumnResolution::build(schema, reader);
897
898        let table_id = TableId::new(format!("{}.{}", self.keyspace, self.table_name));
899
900        // #1741 (roborev HIGH): size the header need-more decision correctly for
901        // the oa/da DeletionTime form. A DELETED oa/da partition carries the full
902        // 12-byte DeletionTime, not the 1-byte LIVE sentinel; peeking the
903        // discriminator (via `partition_header_readiness`) ensures a deleted header
904        // split across a NON-FINAL chunk returns `NeedMore` instead of being
905        // mis-parsed and emitted as `Emitted(1)` (which desynced the scan). The
906        // nb path is unchanged (fixed 12-byte signed form).
907        match self.partition_header_readiness(data) {
908            // Invalid header shape (zero/over-long key) → malformed; advance by
909            // one byte so the outer loop can resynchronise. Mirrors the legacy
910            // `offset += 1; continue` skip-a-byte recovery.
911            PartitionHeaderReadiness::Malformed => return Ok(ParseStep::Emitted(1)),
912            // Header (or its deleted-form DeletionTime) is split across the chunk
913            // boundary. On a non-final chunk request more bytes; on the final
914            // chunk no more will arrive, so the legacy loop treated this as the
915            // end of parseable partitions.
916            PartitionHeaderReadiness::Incomplete => {
917                return Ok(if at_final_chunk {
918                    ParseStep::Done
919                } else {
920                    ParseStep::NeedMore
921                });
922            }
923            // Every byte of the header (incl. its full DeletionTime) is present.
924            PartitionHeaderReadiness::Ready => {}
925        }
926
927        let (partition_key, mut offset, partition_deletion) = match self
928            .parse_partition_header_full(data, 0)
929        {
930            Ok(v) => v,
931            // Defense-in-depth: `Ready` guarantees the DeletionTime is fully
932            // present, so a parse failure here cannot be truncation — it is a
933            // genuinely corrupt header (e.g. a non-0x80 byte with bit 7 set).
934            // On a non-final chunk, however, only re-request more bytes if the
935            // header is not yet complete; otherwise skip a byte to make forward
936            // progress (returning NeedMore on a complete buffer would loop
937            // forever). Since `Ready` holds, that reduces to the legacy
938            // skip-a-byte resync.
939            Err(_) => {
940                if !at_final_chunk
941                    && self.partition_header_readiness(data) == PartitionHeaderReadiness::Incomplete
942                {
943                    return Ok(ParseStep::NeedMore);
944                }
945                return Ok(ParseStep::Emitted(1));
946            }
947        };
948
949        // Issue #1741: read-side shadowing is active ONLY when the parser was built
950        // for a user-facing query read (`self.read_shadowing`); the compaction caller
951        // builds the parser without it, leaving this `None`.
952        // F2: reuse the parser's once-per-read `now_secs` (see `new`), not a per-block
953        // wall-clock sample, so all blocks of one scan share a single `now`.
954        let now_secs = self.now_secs;
955        let mut shadow = self.read_shadowing.then(|| {
956            PartitionShadow::open(
957                now_secs,
958                partition_deletion,
959                clustering_reversed_flags(schema),
960            )
961        });
962
963        let mut static_cells: HashMap<Arc<str>, Value> = HashMap::new();
964
965        // Finding 1 (#827): buffer this partition's emitted rows locally and only
966        // forward them to the external `emit` once the partition is CONFIRMED
967        // complete (a `ParseStep::Emitted` return). If the buffer is truncated
968        // mid-partition (`NeedMore`) after one or more rows were parsed, we must
969        // emit NOTHING and let the caller refill and re-parse this partition from
970        // its start — otherwise the already-forwarded rows would be re-emitted on
971        // the retry, duplicating them in the streaming compaction output.
972        //
973        // The buffer is bounded by ONE partition's rows (the documented
974        // `max_partition_size` bound), not the whole file, so memory stays
975        // bounded as required by the #827 deliverable.
976        let mut pending: Vec<(TableId, RowKey, ScanRow, i64)> = Vec::new();
977
978        // Flush the buffered rows to the external `emit`, honouring an early
979        // `Break`. Returns the `ParseStep` to surface to the caller: on `Break`
980        // we still report the bytes consumed for this (complete) partition so the
981        // caller drains correctly, but stop forwarding the remaining buffered
982        // rows. `flushed_break` becomes true so the driver can stop promptly.
983        macro_rules! flush_and_emitted {
984            ($consumed:expr, $pending:expr, $emit:expr) => {{
985                for entry in $pending.drain(..) {
986                    match $emit(entry)? {
987                        std::ops::ControlFlow::Continue(()) => {}
988                        std::ops::ControlFlow::Break(()) => break,
989                    }
990                }
991                Ok(ParseStep::Emitted($consumed))
992            }};
993        }
994
995        loop {
996            // END_OF_PARTITION (0x01): partition complete, consume the marker.
997            if offset < data.len() && Self::is_end_of_partition(data[offset]) {
998                offset += 1;
999                return flush_and_emitted!(offset, pending, emit);
1000            }
1001
1002            // Consumed everything but never saw END_OF_PARTITION: the partition
1003            // may continue in the next chunk. On NeedMore emit NOTHING (drop the
1004            // buffered rows) so the caller can refill and re-parse from the start
1005            // without duplicating already-buffered rows (Finding 1).
1006            if offset >= data.len() {
1007                if at_final_chunk {
1008                    return flush_and_emitted!(offset, pending, emit);
1009                }
1010                return Ok(ParseStep::NeedMore);
1011            }
1012
1013            if Self::is_range_tombstone_marker(data[offset]) {
1014                // Issue #1741: on the user-facing scan path decode the marker and
1015                // feed the range-tombstone FSM so covered rows are shadowed; the
1016                // compaction path (shadow == None) only advances past it.
1017                if let Some(sh) = shadow.as_mut() {
1018                    match self.parse_range_tombstone_marker_full(data, offset, schema) {
1019                        Ok((bv, bk, dp, ds, next_offset)) => {
1020                            if sh.feed_range_marker(bv, bk, dp, ds).is_err() {
1021                                if at_final_chunk {
1022                                    return flush_and_emitted!(offset, pending, emit);
1023                                }
1024                                return Ok(ParseStep::NeedMore);
1025                            }
1026                            offset = next_offset;
1027                            continue;
1028                        }
1029                        Err(_) => {
1030                            if at_final_chunk {
1031                                return flush_and_emitted!(offset, pending, emit);
1032                            }
1033                            return Ok(ParseStep::NeedMore);
1034                        }
1035                    }
1036                }
1037                match self.skip_range_tombstone_marker(data, offset, schema) {
1038                    Ok(next_offset) => {
1039                        offset = next_offset;
1040                        continue;
1041                    }
1042                    Err(_) => {
1043                        // Marker body truncated? request more unless final.
1044                        if at_final_chunk {
1045                            return flush_and_emitted!(offset, pending, emit);
1046                        }
1047                        return Ok(ParseStep::NeedMore);
1048                    }
1049                }
1050            }
1051
1052            match self.parse_row_data_with_offset(
1053                data,
1054                offset,
1055                Some(schema),
1056                reader,
1057                false,
1058                &resolution,
1059                shadow.as_ref(),
1060            ) {
1061                Ok((
1062                    mut cells,
1063                    _row_cell_meta,
1064                    row_header_opt,
1065                    next_offset,
1066                    is_static,
1067                    _complex_meta,
1068                )) => {
1069                    offset = next_offset;
1070
1071                    // For a row tombstone the authoritative timestamp is
1072                    // markedForDeleteAt (HAS_TIMESTAMP is absent for pure row
1073                    // deletes). For a live row it is the row write timestamp.
1074                    // Both the merger tuple `row_ts` and the emitted
1075                    // Value::Tombstone must agree, so resolve once here (#505).
1076                    let row_tombstone = row_header_opt.as_ref().filter(|h| h.is_row_tombstone());
1077                    // Issue #932: a HAS_DELETION row may ALSO carry a liveness
1078                    // timestamp (surviving cells written strictly after the
1079                    // deletion). Prefer the liveness timestamp as the row
1080                    // timestamp so those cells inherit it and are NOT shadowed by
1081                    // the older row deletion during reconcile; fall back to
1082                    // markedForDeleteAt only for a PURE row tombstone (which has no
1083                    // HAS_TIMESTAMP).
1084                    let row_ts = row_header_opt
1085                        .as_ref()
1086                        .and_then(|h| h.timestamp)
1087                        .or_else(|| row_tombstone.map(|h| h.row_tombstone_deletion_time()))
1088                        .unwrap_or(0);
1089
1090                    if is_static {
1091                        // Issue #1741 (Finding 1): on the shadowing (user-facing) path,
1092                        // drop a static row's cells when the static row is itself shadowed
1093                        // by the partition tombstone or expired by its own TTL, so a
1094                        // surviving clustering row does not resurface stale static data.
1095                        // No-op on the compaction path (shadow == None), which must keep
1096                        // the raw static cells for cross-generation reconcile.
1097                        let static_hidden = shadow.as_ref().is_some_and(|sh| {
1098                            row_header_opt
1099                                .as_ref()
1100                                .is_some_and(|h| sh.row_hidden(h, &[]))
1101                        });
1102                        static_cells = if static_hidden { HashMap::new() } else { cells };
1103                    } else {
1104                        for (k, v) in &static_cells {
1105                            cells.entry(k.clone()).or_insert_with(|| v.clone());
1106                        }
1107
1108                        // Issue #1741: on the user-facing scan path, hide rows
1109                        // shadowed by a partition/range tombstone or expired by TTL
1110                        // (matching a Cassandra SELECT). No-op on the compaction path
1111                        // (shadow == None), which must preserve the row for the merger.
1112                        // A row reduced to only its primary key by per-cell filtering is
1113                        // hidden here too: the dropped cells still fold into the row
1114                        // aggregate, so `row_hidden` sees it shadowed/expired.
1115                        let hidden = shadow.as_ref().is_some_and(|sh| {
1116                            row_header_opt.as_ref().is_some_and(|h| {
1117                                let clustering = if sh.needs_clustering() {
1118                                    extract_clustering_values(&cells, schema)
1119                                } else {
1120                                    Vec::new()
1121                                };
1122                                sh.row_hidden(h, &clustering)
1123                            })
1124                        });
1125
1126                        // Issue #505/#932: row-tombstone display rule now lives in the
1127                        // shared `build_display_row` helper. This `ScanRow` drives BOTH
1128                        // the user-facing streaming scan and the compaction read; both
1129                        // consume the same carrier.
1130                        let row_value = build_display_row(cells, row_header_opt.as_ref(), schema);
1131
1132                        // Finding 1: buffer the row instead of forwarding it now.
1133                        // It is flushed to `emit` only when the partition is
1134                        // confirmed complete (a `flush_and_emitted!` return). A
1135                        // mid-partition `NeedMore` discards `pending` and the
1136                        // caller re-parses from the partition start.
1137                        if !hidden {
1138                            pending.push((
1139                                table_id.clone(),
1140                                partition_key.clone(),
1141                                row_value,
1142                                row_ts,
1143                            ));
1144                        }
1145                    }
1146
1147                    if offset >= data.len() {
1148                        // End of the buffer without an explicit END_OF_PARTITION:
1149                        // the partition may continue in the next chunk.
1150                        if at_final_chunk {
1151                            return flush_and_emitted!(offset, pending, emit);
1152                        }
1153                        return Ok(ParseStep::NeedMore);
1154                    }
1155                    if self.peek_is_partition_header(data, offset) {
1156                        // Next partition starts here — current one is complete.
1157                        return flush_and_emitted!(offset, pending, emit);
1158                    }
1159                }
1160                Err(_) => {
1161                    // A row failed to parse. The legacy loop unconditionally
1162                    // `break`s here (end-of-partition). Mid-stream that may
1163                    // instead be a row straddling the chunk boundary, so request
1164                    // more bytes unless this is the final chunk.
1165                    if at_final_chunk {
1166                        return flush_and_emitted!(offset, pending, emit);
1167                    }
1168                    return Ok(ParseStep::NeedMore);
1169                }
1170            }
1171        }
1172    }
1173}
