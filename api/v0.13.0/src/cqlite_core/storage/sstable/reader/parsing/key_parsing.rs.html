16    /// Enhanced composite key parsing for Cassandra 5.0 multi-component keys with improved format detection
17    pub(in crate::storage::sstable::reader) fn parse_composite_key(
18        &self,
19        key_data: &[u8],
20        schema: Option<&crate::schema::TableSchema>,
21    ) -> Result<RowKey> {
22        if key_data.is_empty() {
23            return Ok(RowKey::new(Vec::new()));
24        }
25
26        // SCHEMA-DRIVEN KEY PARSING: Use exact comparator types when available
27        if let Some(schema) = self.get_table_schema(schema) {
28            return self.parse_key_with_schema(key_data, &schema);
29        }
30
31        // Modern formats should never reach this non-schema fallback path
32        match self.header.cassandra_version {
33            crate::parser::header::CassandraVersion::V5_0NewBig
34            | crate::parser::header::CassandraVersion::V5_0Bti
35            | crate::parser::header::CassandraVersion::V5_0NewBigFormat
36            | crate::parser::header::CassandraVersion::V5_0Uncompressed
37            | crate::parser::header::CassandraVersion::V5_0ComplexTypes
38            | crate::parser::header::CassandraVersion::V5_0TypedCollections
39            | crate::parser::header::CassandraVersion::V5_0WideRows
40            | crate::parser::header::CassandraVersion::V5_0FormatG => Err(Error::Schema(format!(
41                "Non-schema key parsing fallback not allowed for modern format {:?}. \
42                     Use SchemaAwareReader with proper schema registry.",
43                self.header.cassandra_version
44            ))),
45            _ => {
46                // Legacy formats can return raw key data as last resort
47                #[cfg(feature = "legacy-heuristics")]
48                {
49                    log::warn!(
50                        "No schema available - returning raw key data for key of length {} (use SchemaAwareReader)",
51                        key_data.len()
52                    );
53                    Ok(RowKey::new(key_data.to_vec()))
54                }
55                #[cfg(not(feature = "legacy-heuristics"))]
56                {
57                    Err(Error::Schema(
58                        "Non-schema key parsing requires legacy-heuristics feature for legacy compatibility.".to_string()
59                    ))
60                }
61            }
62        }
63    }

116/// Decode key component using exact comparator type (pure function for testing)
117pub(crate) fn decode_key_component_impl(
118    component_data: &[u8],
119    comparator: &ComparatorType,
120) -> Result<Vec<u8>> {
121    // For key components, we typically preserve the byte-comparable encoding
122    // but can validate format based on comparator type
123
124    match comparator {
125        ComparatorType::Uuid => {
126            if component_data.len() != 16 {
127                return Err(Error::corruption("Invalid UUID key component length"));
128            }
129        }
130        ComparatorType::Int => {
131            if component_data.len() != 4 {
132                return Err(Error::corruption("Invalid Int key component length"));
133            }
134        }
135        ComparatorType::BigInt => {
136            if component_data.len() != 8 {
137                return Err(Error::corruption("Invalid BigInt key component length"));
138            }
139        }
140        ComparatorType::Text => {
141            // Validate UTF-8 for text keys
142            if std::str::from_utf8(component_data).is_err() {
143                return Err(Error::corruption("Invalid UTF-8 in text key component"));
144            }
145        }
146        _ => {
147            // For other types, accept as-is for now
148        }
149    }
150
151    // Return the byte-comparable encoding as-is
152    // The comparator validation ensures format correctness
153    Ok(component_data.to_vec())
154}

156/// Parse composite key using Cassandra 5.0+ vint-based format (pure function for testing)
157pub(crate) fn parse_composite_key_v5_format_impl(key_data: &[u8]) -> Result<RowKey> {
158    if key_data.len() < 2 {
159        return Err(Error::corruption("Key too short for v5 format".to_string()));
160    }
161
162    let mut components = Vec::new();
163
164    // Parse component count (vint)
165    let (remaining, component_count) = parse_vint_length(key_data)
166        .map_err(|_| Error::corruption("Failed to parse component count".to_string()))?;
167    let mut offset = key_data.len() - remaining.len();
168
169    if component_count == 0 || component_count > 256 {
170        return Err(Error::corruption(format!(
171            "Invalid component count: {}",
172            component_count
173        )));
174    }
175
176    log::debug!(
177        "Parsing v5 composite key with {} components",
178        component_count
179    );
180
181    // Parse each component
182    for i in 0..component_count {
183        if offset >= key_data.len() {
184            break;
185        }
186
187        // Parse component length (vint)
188        let (remaining, component_len) = parse_vint_length(&key_data[offset..])
189            .map_err(|_| Error::corruption(format!("Failed to parse component {} length", i)))?;
190        offset = key_data.len() - remaining.len();
191
192        if component_len > 0 && offset + component_len <= key_data.len() {
193            components.extend_from_slice(&key_data[offset..offset + component_len]);
194            offset += component_len;
195
196            // Add component separator (except for last component)
197            if i < component_count - 1 {
198                components.push(0x00);
199            }
200        }
201    }
202
203    log::debug!("Parsed v5 composite key: {} total bytes", components.len());
204    Ok(RowKey::new(components))
205}

207/// Parse composite key using legacy u16-length prefixed format (pure function for testing)
208pub(crate) fn parse_composite_key_legacy_format_impl(key_data: &[u8]) -> Result<RowKey> {
209    if key_data.len() < 3 || key_data[0] != 0x00 {
210        return Err(Error::corruption(
211            "Not legacy composite key format".to_string(),
212        ));
213    }
214
215    let mut offset = 0;
216    let mut components = Vec::new();
217
218    while offset < key_data.len() {
219        if offset + 2 > key_data.len() {
220            break;
221        }
222
223        // Read component length (big-endian u16)
224        let component_len = u16::from_be_bytes([key_data[offset], key_data[offset + 1]]) as usize;
225        offset += 2;
226
227        if offset + component_len > key_data.len() {
228            break;
229        }
230
231        components.extend_from_slice(&key_data[offset..offset + component_len]);
232        components.push(0x00); // Component separator
233        offset += component_len;
234
235        // Check for end-of-components marker
236        if offset < key_data.len() && key_data[offset] == 0x00 {
237            break;
238        }
239    }
240
241    // Remove trailing separator if present
242    if components.last() == Some(&0x00) {
243        components.pop();
244    }
245
246    log::debug!(
247        "Parsed legacy composite key: {} total bytes",
248        components.len()
249    );
250    Ok(RowKey::new(components))
251}

253/// Parse clustering key format (pure function for testing)
254pub(crate) fn parse_clustering_key_format_impl(key_data: &[u8]) -> Result<RowKey> {
255    // Clustering keys in Cassandra 5.0 might use a different format
256    // Check for clustering key markers or patterns
257
258    if key_data.len() < 4 {
259        return Err(Error::corruption(
260            "Too short for clustering key".to_string(),
261        ));
262    }
263
264    // Check if this looks like a clustering key by analyzing the structure
265    // Clustering keys often have type info followed by the actual key data
266    if key_data[0] <= 0x1F {
267        // Potential type marker
268        let mut offset = 1;
269
270        // Skip type information
271        while offset < key_data.len() && key_data[offset] <= 0x1F {
272            offset += 1;
273        }
274
275        if offset < key_data.len() {
276            let clustering_data = &key_data[offset..];
277            log::debug!(
278                "Parsed clustering key: {} bytes after {} byte type prefix",
279                clustering_data.len(),
280                offset
281            );
282            return Ok(RowKey::new(clustering_data.to_vec()));
283        }
284    }
285
286    Err(Error::corruption("Not clustering key format".to_string()))
287}

289/// Parse key using exact schema information (pure function for testing)
290pub(crate) fn parse_key_with_schema_impl(
291    key_data: &[u8],
292    schema: &TableSchema,
293    cassandra_version: CassandraVersion,
294) -> Result<RowKey> {
295    use crate::parser::vint::parse_vint_length;
296
297    // Check if this format uses byte-comparable encoding
298    let is_byte_comparable = matches!(cassandra_version, CassandraVersion::V5_0NewBigFormat);
299
300    if is_byte_comparable {
301        // Use byte-comparable decoder
302        let (_, components) = super::byte_comparable::decode_byte_comparable_key(key_data)
303            .map_err(|_| Error::corruption("Failed to decode byte-comparable partition key"))?;
304
305        log::debug!(
306            "parse_key_with_schema: Decoded {} byte-comparable components",
307            components.len()
308        );
309
310        // Build compound key from decoded components
311        let total_size: usize = components.iter().map(|c| c.len()).sum();
312        let mut compound_key_data = Vec::with_capacity(total_size);
313        for component in components {
314            compound_key_data.extend_from_slice(&component);
315        }
316
317        return Ok(RowKey::new(compound_key_data));
318    }
319
320    // Handle single-component partition keys (no VInt prefix - raw bytes)
321    // Multi-component (composite) partition keys have VInt length prefixes per component
322    if schema.partition_keys.len() == 1 {
323        // Single partition key column: key_data is the raw value (no length prefix)
324        let partition_column = &schema.partition_keys[0];
325        let comparator =
326            ComparatorType::from_data_type(&partition_column.data_type).map_err(|e| {
327                Error::Schema(format!(
328                    "Invalid partition key type '{}': {}",
329                    partition_column.data_type, e
330                ))
331            })?;
332
333        log::debug!(
334            "parse_key_with_schema: Single partition key column '{}' ({}), raw key length: {}",
335            partition_column.name,
336            partition_column.data_type,
337            key_data.len()
338        );
339
340        // Decode and return the single component directly
341        let decoded_component = decode_key_component_impl(key_data, &comparator)?;
342        return Ok(RowKey::new(decoded_component));
343    }
344
345    // Multi-component (composite) partition keys: VInt-based parsing
346    let mut offset = 0;
347    let mut key_components = Vec::new();
348
349    log::debug!(
350        "parse_key_with_schema: Composite partition key with {} components, key length: {}",
351        schema.partition_keys.len(),
352        key_data.len()
353    );
354
355    // Parse partition key components using exact comparator types
356    for partition_column in &schema.partition_keys {
357        if offset >= key_data.len() {
358            break;
359        }
360
361        // Parse component length (vint) - only for composite keys
362        let (remaining, component_len) = parse_vint_length(&key_data[offset..])
363            .map_err(|_| Error::corruption("Failed to parse partition key component length"))?;
364        offset = key_data.len() - remaining.len();
365
366        if component_len > remaining.len() {
367            return Err(Error::corruption(
368                "Partition key component length exceeds available data",
369            ));
370        }
371
372        // Extract component data
373        let component_data = &remaining[..component_len];
374
375        let comparator =
376            ComparatorType::from_data_type(&partition_column.data_type).map_err(|e| {
377                Error::Schema(format!(
378                    "Invalid partition key type '{}': {}",
379                    partition_column.data_type, e
380                ))
381            })?;
382
383        // Decode component using exact comparator type
384        let decoded_component = decode_key_component_impl(component_data, &comparator)?;
385        key_components.push(decoded_component);
386
387        offset += component_len;
388    }
389
390    // Parse clustering key components if present
391    if offset < key_data.len() {
392        for clustering_column in &schema.clustering_keys {
393            if offset >= key_data.len() {
394                break;
395            }
396
397            // Parse component length (vint)
398            let (remaining, component_len) =
399                parse_vint_length(&key_data[offset..]).map_err(|_| {
400                    Error::corruption("Failed to parse clustering key component length")
401                })?;
402            offset = key_data.len() - remaining.len();
403
404            if component_len > remaining.len() {
405                return Err(Error::corruption(
406                    "Clustering key component length exceeds available data",
407                ));
408            }
409
410            // Extract component data
411            let component_data = &remaining[..component_len];
412
413            // DEPRECATED: This should use SchemaAwareReader with proper comparators
414            let comparator =
415                ComparatorType::from_data_type(&clustering_column.data_type).map_err(|e| {
416                    Error::Schema(format!(
417                        "Invalid clustering key type '{}' - use SchemaAwareReader: {}",
418                        clustering_column.data_type, e
419                    ))
420                })?;
421
422            // Decode component using exact comparator type
423            let decoded_component = decode_key_component_impl(component_data, &comparator)?;
424            key_components.push(decoded_component);
425
426            offset += component_len;
427        }
428    }
429
430    // Create compound key from decoded components
431    let total_size: usize = key_components.iter().map(|c| c.len()).sum();
432    let mut compound_key_data = Vec::with_capacity(total_size);
433    for component in key_components {
434        compound_key_data.extend_from_slice(&component);
435    }
436
437    Ok(RowKey::new(compound_key_data))
438}
