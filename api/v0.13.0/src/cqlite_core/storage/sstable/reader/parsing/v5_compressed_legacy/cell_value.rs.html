5impl V5CompressedLegacyParser {
6    /// Parse a single cell value WITHOUT column name (schema-order format)
7    ///
8    /// Cell format in V5CompressedLegacy follows Cassandra 5.0 cell serialization:
9    /// - First byte: Cell flags (bitset, valid range: 0x00-0x1F)
10    ///   - 0x01 = IS_DELETED_MASK (tombstone)
11    ///   - 0x02 = IS_EXPIRING_MASK (has TTL)
12    ///   - 0x04 = HAS_EMPTY_VALUE_MASK (no value bytes)
13    ///   - 0x08 = USE_ROW_TIMESTAMP_MASK (use row timestamp)
14    ///   - 0x10 = USE_ROW_TTL_MASK (use row TTL)
15    /// - Conditional timestamp/TTL/deletion fields (based on flags)
16    /// - Value data (if HAS_EMPTY_VALUE not set)
17    ///
18    /// See CASSANDRA_5_CELL_DESERIALIZATION_FORMAT.md for complete specification.
19    ///
20    /// Returns: `(value, cell_own_timestamp, expiration, new_offset)` where:
21    /// - `cell_own_timestamp`: the cell's own decoded timestamp in µs, or `None`
22    ///   when the cell inherits the row-level timestamp (`USE_ROW_TIMESTAMP` flag).
23    /// - `expiration`: TTL / localDeletionTime pair when the cell is expiring, or
24    ///   `None` when the cell has no TTL.
25    pub(super) fn parse_cell_value_schema_order(
26        &self,
27        data: &[u8],
28        mut offset: usize,
29        column: &crate::schema::Column,
30        // AUTHORITATIVE on-disk SerializationHeader marshal type for this column
31        // (issue #1080). Used as a fallback to resolve a `frozen<udt>` whose
32        // supplied schema short form carries no field defs and no UdtRegistry is
33        // wired: the header type is the full `FrozenType(UserType(ks,name,fields))`
34        // marshal string, decoded structurally rather than guessed. `None` on the
35        // header-empty synthetic path and on internal recursive calls.
36        header_type: Option<&str>,
37        _reader: &crate::storage::sstable::reader::types::SSTableReader,
38    ) -> Result<(Value, Option<i64>, Option<CellExpiration>, usize)> {
39        // Cell flag constants (from Cassandra 5.0 Cell.Serializer)
40        const CELL_IS_DELETED: u8 = 0x01;
41        const CELL_IS_EXPIRING: u8 = 0x02;
42        const CELL_HAS_EMPTY_VALUE: u8 = 0x04;
43        const CELL_USE_ROW_TIMESTAMP: u8 = 0x08;
44        const CELL_USE_ROW_TTL: u8 = 0x10;
45
46        // Read cell flags byte
47        if offset >= data.len() {
48            return Err(Error::corruption(format!(
49                "Cell '{}': unexpected end at flags byte",
50                column.name
51            )));
52        }
53        let flags = data[offset];
54
55        // CRITICAL FIX (Issue #191): Validate flags are in valid range (0x00-0x1F)
56        // Bits 0x20, 0x40, 0x80 are row-level flags and should NEVER appear in cell flags.
57        // If we see these bits, the offset is misaligned (reading row data at cell position).
58        if flags > 0x1F {
59            return Err(Error::corruption(format!(
60                "Cell '{}': invalid cell flags 0x{:02x} at offset {} (bits 0x20/0x40/0x80 indicate offset misalignment)",
61                column.name, flags, offset
62            )));
63        }
64
65        offset += 1;
66
67        // Decode flags
68        let is_deleted = (flags & CELL_IS_DELETED) != 0;
69        let is_expiring = (flags & CELL_IS_EXPIRING) != 0;
70        let has_empty_value = (flags & CELL_HAS_EMPTY_VALUE) != 0;
71        let use_row_timestamp = (flags & CELL_USE_ROW_TIMESTAMP) != 0;
72        let use_row_ttl = (flags & CELL_USE_ROW_TTL) != 0;
73
74        log::debug!(
75            "V5CompressedLegacy: Cell '{}' flags=0x{:02x} (deleted={}, expiring={}, empty={}, use_row_ts={}, use_row_ttl={})",
76            column.name, flags, is_deleted, is_expiring, has_empty_value, use_row_timestamp, use_row_ttl
77        );
78
79        // === PHASE 2: Parse conditional fields between flags and value ===
80        // Based on Cassandra 5.0 Cell.Serializer format specification
81
82        // Step 1: Read timestamp (if not using row timestamp)
83        // Issue #505: capture the actual cell timestamp so deleted cells can carry it
84        // in a Value::Tombstone.
85        //
86        // Fix #629 (C2): Cell timestamp delta is UNSIGNED VInt per Cassandra
87        // SerializationHeader.java:165: out.writeUnsignedVInt(timestamp - stats.minTimestamp).
88        let mut cell_timestamp: Option<i64> = None;
89        if !use_row_timestamp {
90            let (remaining, timestamp_delta) = parse_vuint(&data[offset..]).map_err(|e| {
91                Error::corruption(format!(
92                    "Cell '{}': failed to parse timestamp delta as VUInt at offset {}: {:?}",
93                    column.name, offset, e
94                ))
95            })?;
96            let bytes_consumed = data[offset..].len() - remaining.len();
97            offset += bytes_consumed;
98            let absolute_ts = self.min_timestamp.wrapping_add(timestamp_delta as i64);
99            log::debug!(
100                "V5CompressedLegacy: Cell '{}' timestamp_delta={} (min_timestamp={}) absolute={}",
101                column.name,
102                timestamp_delta,
103                self.min_timestamp,
104                absolute_ts,
105            );
106            cell_timestamp = Some(absolute_ts);
107        }
108
109        // Step 2: Read localDeletionTime (if deleted or expiring, and not using row TTL)
110        // Captured as absolute epoch-seconds for CellExpiration.expires_at_seconds.
111        let mut cell_local_deletion_time: Option<i64> = None;
112        if !use_row_ttl && (is_deleted || is_expiring) {
113            let (remaining, deletion_delta) = parse_vuint(&data[offset..]).map_err(|e| {
114                Error::corruption(format!(
115                    "Cell '{}': failed to parse localDeletionTime delta as VUInt at offset {}: {:?}",
116                    column.name, offset, e
117                ))
118            })?;
119            let bytes_consumed = data[offset..].len() - remaining.len();
120            offset += bytes_consumed;
121            let abs_ldt = self
122                .min_local_deletion_time
123                .wrapping_add(deletion_delta as i64);
124            log::debug!(
125                "V5CompressedLegacy: Cell '{}' deletion_delta={} (min_local_deletion_time={}) abs_ldt={}",
126                column.name,
127                deletion_delta,
128                self.min_local_deletion_time,
129                abs_ldt
130            );
131            cell_local_deletion_time = Some(abs_ldt);
132        }
133
134        // Step 3: Read TTL (if expiring and not using row TTL)
135        // Captured as absolute TTL seconds for CellExpiration.ttl_seconds.
136        let mut cell_ttl_seconds: Option<i32> = None;
137        if !use_row_ttl && is_expiring {
138            let (remaining, ttl_delta) = parse_vuint(&data[offset..]).map_err(|e| {
139                Error::corruption(format!(
140                    "Cell '{}': failed to parse TTL delta as VUInt at offset {}: {:?}",
141                    column.name, offset, e
142                ))
143            })?;
144            let bytes_consumed = data[offset..].len() - remaining.len();
145            offset += bytes_consumed;
146            // Absolute TTL = min_ttl + delta (seconds).  Clamp to i32 range for the
147            // CellExpiration.ttl_seconds field (Cassandra caps TTL at ~630M seconds).
148            let abs_ttl = self.min_ttl.unwrap_or(0).wrapping_add(ttl_delta as i64);
149            log::debug!(
150                "V5CompressedLegacy: Cell '{}' ttl_delta={} (min_ttl={:?}) abs_ttl={}",
151                column.name,
152                ttl_delta,
153                self.min_ttl,
154                abs_ttl
155            );
156            cell_ttl_seconds = Some(abs_ttl.min(i32::MAX as i64) as i32);
157        }
158
159        // Build per-cell expiration metadata (used when the flag is set).
160        // Available at both return sites below — the tombstone path also uses
161        // cell_timestamp so we compute expiration here before the tombstone check.
162        let cell_expiration: Option<CellExpiration> =
163            match (is_expiring, cell_local_deletion_time, cell_ttl_seconds) {
164                (true, Some(expires_at), Some(ttl_secs)) => Some(CellExpiration {
165                    ttl_seconds: ttl_secs,
166                    expires_at_seconds: expires_at,
167                }),
168                // use_row_ttl path: expiration info comes from the row header (caller handles it).
169                _ => None,
170            };
171
172        // Step 4: Cell path for complex columns (multi-cell collections/UDTs)
173        // For now, skip this - we'll add in a future iteration when we handle complex columns.
174        // Simple columns (int, text, boolean, uuid, etc.) don't have cell paths.
175
176        // === End of Phase 2 conditional field parsing ===
177
178        // CRITICAL: Inverted logic for HAS_EMPTY_VALUE_MASK
179        // Flag NOT set (0x04 absent) = cell HAS value → read value bytes
180        // Flag SET (0x04 present) = cell has NO value → return empty/null immediately
181        let has_value = !has_empty_value;
182
183        // Handle deleted cells (tombstones)
184        // According to Cassandra 5.0 Cell.Serializer, deleted cells:
185        // 1. Have IS_DELETED flag set
186        // 2. May have deletion metadata (timestamp, localDeletionTime)
187        // 3. Do NOT have value data (even if HAS_EMPTY_VALUE not set)
188        //
189        // Issue #505: emit Value::Tombstone(CellTombstone) so that the compaction
190        // merger can apply cell-level shadowing semantics.  The actual deletion
191        // timestamp is carried in the tombstone for timestamp-based LWW ordering.
192        if is_deleted {
193            let deletion_time = cell_timestamp.unwrap_or(0);
194            log::debug!(
195                "V5CompressedLegacy: Cell '{}' is tombstone (deleted), returning Tombstone(deletion_time={})",
196                column.name, deletion_time
197            );
198            return Ok((
199                Value::Tombstone(TombstoneInfo {
200                    deletion_time,
201                    tombstone_type: TombstoneType::CellTombstone,
202                    // On-disk `localDeletionTime` (GC clock, seconds) for the cell
203                    // tombstone; `0` when not surfaced (#873).
204                    local_deletion_time: cell_local_deletion_time.unwrap_or(0),
205                    ttl: None,
206                    range_start: None,
207                    range_end: None,
208                }),
209                cell_timestamp,
210                cell_expiration,
211                offset,
212            ));
213        }
214
215        // Handle empty cells (no value bytes to read)
216        if !has_value {
217            log::debug!(
218                "V5CompressedLegacy: Cell '{}' has HAS_EMPTY_VALUE flag, returning empty value",
219                column.name
220            );
221            // Issue #1077: the empty (zero-length) value MUST decode to the empty
222            // value of the column's DECLARED type — never blindly `Text("")`.
223            // An empty `blob` is `Blob([])` (sstabledump renders `"0x"`), an empty
224            // text/ascii/varchar is `Text("")`. Mirrors the clustering-key EMPTY
225            // handling above; fixed-width types should not normally carry an empty
226            // value, so treat that as NULL with a warning.
227            let empty_value = match column.data_type.to_lowercase().as_str() {
228                "text" | "varchar" | "ascii" => Value::Text(String::new()),
229                "blob" => Value::Blob(Vec::new()),
230                _ => {
231                    log::warn!(
232                        "V5CompressedLegacy: EMPTY value for cell '{}' (type {}), treating as NULL",
233                        column.name,
234                        column.data_type
235                    );
236                    Value::Null
237                }
238            };
239            return Ok((empty_value, cell_timestamp, cell_expiration, offset));
240        }
241
242        // At this point, we have a live cell with value data
243        // The value parsing logic below is unchanged from the original implementation
244
245        // Parse based on column type (data_type is a String with CQL type name)
246        // CRITICAL: Normalize type name to lowercase for case-insensitive matching
247        // Schema may provide "TEXT", "INT", etc. (uppercase) while match arms use lowercase
248        let normalized_type = column.data_type.to_lowercase();
249        let value = match normalized_type.as_str() {
250            "boolean" => {
251                // Boolean: [0x08][u8 value]
252                if offset >= data.len() {
253                    return Err(Error::corruption(format!(
254                        "Cell '{}': unexpected end at boolean value",
255                        column.name
256                    )));
257                }
258                let bool_byte = data[offset];
259                offset += 1;
260                Value::Boolean(bool_byte != 0)
261            }
262
263            "int" => {
264                // Integer (i32): fixed-width 4 bytes (no length prefix in Cassandra 5.0)
265                if offset + 4 > data.len() {
266                    return Err(Error::corruption(format!(
267                        "Cell '{}': need 4 bytes for int, only {} available",
268                        column.name,
269                        data.len() - offset
270                    )));
271                }
272                let int_val = i32::from_be_bytes([
273                    data[offset],
274                    data[offset + 1],
275                    data[offset + 2],
276                    data[offset + 3],
277                ]);
278                offset += 4;
279                Value::Integer(int_val)
280            }
281
282            "text" | "varchar" | "ascii" => {
283                // Text: [VInt len][text bytes]
284                // V5CompressedLegacy uses VInt length encoding for variable-length types
285                let (remaining, text_len) = parse_vuint(&data[offset..]).map_err(|e| {
286                    Error::corruption(format!(
287                        "Cell '{}': failed to parse text length as VInt: {:?}",
288                        column.name, e
289                    ))
290                })?;
291                let text_len = text_len as usize;
292                let bytes_consumed = data[offset..].len() - remaining.len();
293                offset += bytes_consumed;
294
295                if offset + text_len > data.len() {
296                    return Err(Error::corruption(format!(
297                        "Cell '{}': need {} bytes for text, only {} available",
298                        column.name,
299                        text_len,
300                        data.len() - offset
301                    )));
302                }
303
304                let text_bytes = &data[offset..offset + text_len];
305                let text = String::from_utf8(text_bytes.to_vec()).map_err(|e| {
306                    Error::corruption(format!(
307                        "Cell '{}': invalid UTF-8 in text value: {}",
308                        column.name, e
309                    ))
310                })?;
311
312                offset += text_len;
313                Value::Text(text)
314            }
315
316            "uuid" | "timeuuid" => {
317                // UUID/TimeUUID: fixed-width 16 bytes (no length prefix in Cassandra 5.0 writer)
318                if offset + 16 > data.len() {
319                    return Err(Error::corruption(format!(
320                        "Cell '{}': need 16 bytes for UUID, only {} available",
321                        column.name,
322                        data.len() - offset
323                    )));
324                }
325
326                let uuid_bytes: [u8; 16] = data[offset..offset + 16]
327                    .try_into()
328                    .map_err(|_| Error::corruption("UUID byte conversion failed"))?;
329
330                offset += 16;
331                Value::Uuid(uuid_bytes)
332            }
333
334            "decimal" => {
335                // Decimal: [VInt total_len][i32 scale][unscaled bytes]
336                if offset >= data.len() {
337                    return Err(Error::corruption(format!(
338                        "Cell '{}': unexpected end at decimal length",
339                        column.name
340                    )));
341                }
342
343                let (remaining, total_len) = parse_vuint(&data[offset..]).map_err(|e| {
344                    Error::corruption(format!(
345                        "Cell '{}': failed to parse decimal length as VInt: {:?}",
346                        column.name, e
347                    ))
348                })?;
349                let total_len = total_len as usize;
350                let bytes_consumed = data[offset..].len() - remaining.len();
351                offset += bytes_consumed;
352
353                if offset + total_len > data.len() {
354                    return Err(Error::corruption(format!(
355                        "Cell '{}': need {} bytes for decimal, only {} available",
356                        column.name,
357                        total_len,
358                        data.len() - offset
359                    )));
360                }
361
362                // First 4 bytes: scale (i32 BE)
363                if total_len < 4 {
364                    return Err(Error::corruption(format!(
365                        "Cell '{}': decimal length {} too small for scale",
366                        column.name, total_len
367                    )));
368                }
369                let scale = i32::from_be_bytes([
370                    data[offset],
371                    data[offset + 1],
372                    data[offset + 2],
373                    data[offset + 3],
374                ]);
375
376                // Remaining bytes: unscaled value
377                let unscaled = data[offset + 4..offset + total_len].to_vec();
378                offset += total_len;
379
380                Value::Decimal { scale, unscaled }
381            }
382
383            "bigint" => {
384                // BigInt: fixed-width 8 bytes (no length prefix in Cassandra 5.0)
385                if offset + 8 > data.len() {
386                    return Err(Error::corruption(format!(
387                        "Cell '{}': need 8 bytes for bigint, only {} available",
388                        column.name,
389                        data.len() - offset
390                    )));
391                }
392                let val = i64::from_be_bytes([
393                    data[offset],
394                    data[offset + 1],
395                    data[offset + 2],
396                    data[offset + 3],
397                    data[offset + 4],
398                    data[offset + 5],
399                    data[offset + 6],
400                    data[offset + 7],
401                ]);
402                offset += 8;
403                Value::BigInt(val)
404            }
405
406            "counter" => {
407                // Counter cells can arrive in two formats:
408                //
409                // 1. Real Cassandra CounterContext: [VInt length][header_size:i16][indices][shards]
410                //    The counter value is the sum of all shard counts.
411                //
412                // 2. CQLite writer format (raw i64): [VInt(8)][8 bytes big-endian i64]
413                //    The writer serialises Value::Counter as a plain 8-byte integer with a
414                //    length prefix of 8, identical to how BigInt is written.
415                //
416                // We try CounterContext first and fall back to the raw-i64 interpretation
417                // when the length prefix equals exactly 8 (the size of a raw i64).
418
419                // Read the VInt length prefix.
420                let (remaining, context_len) = parse_vuint(&data[offset..]).map_err(|e| {
421                    Error::corruption(format!(
422                        "Cell '{}': failed to parse counter context length as VInt: {:?}",
423                        column.name, e
424                    ))
425                })?;
426                let context_len = context_len as usize;
427                let len_bytes_consumed = data[offset..].len() - remaining.len();
428                offset += len_bytes_consumed;
429
430                log::debug!(
431                    "V5CompressedLegacy: Counter '{}' context_len={} (len prefix: {} bytes)",
432                    column.name,
433                    context_len,
434                    len_bytes_consumed
435                );
436
437                if offset + context_len > data.len() {
438                    return Err(Error::corruption(format!(
439                        "Cell '{}': need {} bytes for counter context, only {} available",
440                        column.name,
441                        context_len,
442                        data.len() - offset
443                    )));
444                }
445
446                // Try the full CounterContext parse first.
447                match Self::parse_counter_context(data, offset, &column.name) {
448                    Ok((total, consumed)) if consumed == context_len => {
449                        // Successfully parsed a proper CounterContext.
450                        offset += consumed;
451                        log::debug!(
452                            "V5CompressedLegacy: Counter '{}' value={} (CounterContext), total consumed {} bytes",
453                            column.name,
454                            total,
455                            len_bytes_consumed + context_len
456                        );
457                        Value::Counter(total)
458                    }
459                    _ if context_len == 8 => {
460                        // A real Cassandra CounterContext is at minimum 36 bytes
461                        // (2 header + 2 indices + 32 body for 1 shard), so
462                        // context_len == 8 can only be produced by the CQLite writer
463                        // which serialises Counter as a raw big-endian i64.
464                        // This intentionally swallows any parse_counter_context error
465                        // for 8-byte payloads, which is safe since a valid
466                        // CounterContext can never be 8 bytes.
467                        //
468                        // Bounds already verified by the context_len check above.
469                        let val = i64::from_be_bytes([
470                            data[offset],
471                            data[offset + 1],
472                            data[offset + 2],
473                            data[offset + 3],
474                            data[offset + 4],
475                            data[offset + 5],
476                            data[offset + 6],
477                            data[offset + 7],
478                        ]);
479                        offset += 8;
480                        log::debug!(
481                            "V5CompressedLegacy: Counter '{}' value={} (raw i64 fallback), total consumed {} bytes",
482                            column.name,
483                            val,
484                            len_bytes_consumed + 8
485                        );
486                        Value::Counter(val)
487                    }
488                    Err(e) => return Err(e),
489                    Ok((_, consumed)) => {
490                        return Err(Error::corruption(format!(
491                            "Counter '{}': VInt length ({}) doesn't match parsed context size ({})",
492                            column.name, context_len, consumed
493                        )));
494                    }
495                }
496            }
497
498            "double" => {
499                // Double: 8 bytes, f64 big-endian (NO length prefix)
500                if offset + 8 > data.len() {
501                    return Err(Error::corruption(format!(
502                        "Cell '{}': need 8 bytes for double, only {} available",
503                        column.name,
504                        data.len() - offset
505                    )));
506                }
507                let val = f64::from_be_bytes([
508                    data[offset],
509                    data[offset + 1],
510                    data[offset + 2],
511                    data[offset + 3],
512                    data[offset + 4],
513                    data[offset + 5],
514                    data[offset + 6],
515                    data[offset + 7],
516                ]);
517                offset += 8;
518                Value::Float(val)
519            }
520
521            "timestamp" => {
522                // Timestamp: 8 bytes, i64 milliseconds big-endian (NO length prefix, per Cassandra spec)
523                if offset + 8 > data.len() {
524                    return Err(Error::corruption(format!(
525                        "Cell '{}': need 8 bytes for timestamp, only {} available",
526                        column.name,
527                        data.len() - offset
528                    )));
529                }
530                let millis = i64::from_be_bytes([
531                    data[offset],
532                    data[offset + 1],
533                    data[offset + 2],
534                    data[offset + 3],
535                    data[offset + 4],
536                    data[offset + 5],
537                    data[offset + 6],
538                    data[offset + 7],
539                ]);
540                offset += 8;
541                Value::Timestamp(millis)
542            }
543
544            "date" => {
545                // Date: [VInt len=4][i32 BE days]
546                if offset >= data.len() {
547                    return Err(Error::corruption(format!(
548                        "Cell '{}': unexpected end at date length",
549                        column.name
550                    )));
551                }
552
553                let (remaining, date_len) = parse_vuint(&data[offset..]).map_err(|e| {
554                    Error::corruption(format!(
555                        "Cell '{}': failed to parse date length as VInt: {:?}",
556                        column.name, e
557                    ))
558                })?;
559                let date_len = date_len as usize;
560                let bytes_consumed = data[offset..].len() - remaining.len();
561                offset += bytes_consumed;
562
563                if date_len != 4 {
564                    return Err(Error::corruption(format!(
565                        "Cell '{}': expected date length 4, got {}",
566                        column.name, date_len
567                    )));
568                }
569
570                if offset + 4 > data.len() {
571                    return Err(Error::corruption(format!(
572                        "Cell '{}': need 4 bytes for date, only {} available",
573                        column.name,
574                        data.len() - offset
575                    )));
576                }
577
578                let stored = u32::from_be_bytes([
579                    data[offset],
580                    data[offset + 1],
581                    data[offset + 2],
582                    data[offset + 3],
583                ]);
584                offset += 4;
585                // Cassandra DATE: 4-byte unsigned int with Integer.MIN_VALUE offset
586                let days_since_epoch = stored.wrapping_add(i32::MIN as u32) as i32;
587                Value::Date(days_since_epoch)
588            }
589
590            "duration" => {
591                // Duration: [VInt len][months VInt][days VInt][nanos VInt]
592                // Format: Variable-length encoding with 3 VInt components
593                if offset >= data.len() {
594                    return Err(Error::corruption(format!(
595                        "Cell '{}': unexpected end at duration length",
596                        column.name
597                    )));
598                }
599
600                let (remaining, duration_len) = parse_vuint(&data[offset..]).map_err(|e| {
601                    Error::corruption(format!(
602                        "Cell '{}': failed to parse duration length as VInt: {:?}",
603                        column.name, e
604                    ))
605                })?;
606                let duration_len = duration_len as usize;
607                let bytes_consumed = data[offset..].len() - remaining.len();
608                offset += bytes_consumed;
609
610                if offset + duration_len > data.len() {
611                    return Err(Error::corruption(format!(
612                        "Cell '{}': need {} bytes for duration, only {} available",
613                        column.name,
614                        duration_len,
615                        data.len() - offset
616                    )));
617                }
618
619                // Parse three VInt components from the duration_len bytes
620                let duration_bytes = &data[offset..offset + duration_len];
621
622                // Parse months (signed VInt)
623                let (remaining, months) = parse_vint(duration_bytes).map_err(|e| {
624                    Error::corruption(format!(
625                        "Cell '{}': failed to parse duration months: {:?}",
626                        column.name, e
627                    ))
628                })?;
629                let pos = duration_bytes.len() - remaining.len();
630
631                // Parse days (signed VInt)
632                let (remaining, days) = parse_vint(&duration_bytes[pos..]).map_err(|e| {
633                    Error::corruption(format!(
634                        "Cell '{}': failed to parse duration days: {:?}",
635                        column.name, e
636                    ))
637                })?;
638                let pos = duration_bytes.len() - remaining.len();
639
640                // Parse nanoseconds (signed VInt)
641                let (remaining, nanos) = parse_vint(&duration_bytes[pos..]).map_err(|e| {
642                    Error::corruption(format!(
643                        "Cell '{}': failed to parse duration nanos: {:?}",
644                        column.name, e
645                    ))
646                })?;
647
648                // Verify we consumed all duration bytes
649                if !remaining.is_empty() {
650                    warn!(
651                        "V5CompressedLegacy: Duration '{}' has {} extra bytes after parsing",
652                        column.name,
653                        remaining.len()
654                    );
655                }
656
657                // months/days are i32 in Cassandra's DurationType. Reject
658                // (rather than silently truncate via `as i32`) any encoded value
659                // outside the i32 range so a corrupt encoding errors instead of
660                // wrapping (issue #1632, item b).
661                let months = i32::try_from(months).map_err(|_| {
662                    Error::corruption(format!(
663                        "Cell '{}': duration months out of i32 range",
664                        column.name
665                    ))
666                })?;
667                let days = i32::try_from(days).map_err(|_| {
668                    Error::corruption(format!(
669                        "Cell '{}': duration days out of i32 range",
670                        column.name
671                    ))
672                })?;
673
674                offset += duration_len;
675                Value::Duration {
676                    months,
677                    days,
678                    nanos,
679                }
680            }
681
682            "float" => {
683                // Float: 4 bytes, f32 big-endian (NO length prefix, fixed size)
684                if offset + 4 > data.len() {
685                    return Err(Error::corruption(format!(
686                        "Cell '{}': need 4 bytes for float, only {} available",
687                        column.name,
688                        data.len() - offset
689                    )));
690                }
691
692                let val = f32::from_be_bytes([
693                    data[offset],
694                    data[offset + 1],
695                    data[offset + 2],
696                    data[offset + 3],
697                ]);
698                offset += 4;
699                Value::Float(val as f64) // Convert f32 to f64 for storage
700            }
701
702            "smallint" | "short" => {
703                // SmallInt: [VInt len=2][i16 BE]
704                if offset >= data.len() {
705                    return Err(Error::corruption(format!(
706                        "Cell '{}': unexpected end at smallint length",
707                        column.name
708                    )));
709                }
710
711                let (remaining, len) = parse_vuint(&data[offset..]).map_err(|e| {
712                    Error::corruption(format!(
713                        "Cell '{}': failed to parse smallint length as VInt: {:?}",
714                        column.name, e
715                    ))
716                })?;
717                let len = len as usize;
718                let bytes_consumed = data[offset..].len() - remaining.len();
719                offset += bytes_consumed;
720
721                if len != 2 {
722                    return Err(Error::corruption(format!(
723                        "Cell '{}': expected smallint length 2, got {}",
724                        column.name, len
725                    )));
726                }
727
728                if offset + 2 > data.len() {
729                    return Err(Error::corruption(format!(
730                        "Cell '{}': need 2 bytes for smallint, only {} available",
731                        column.name,
732                        data.len() - offset
733                    )));
734                }
735
736                let val = i16::from_be_bytes([data[offset], data[offset + 1]]);
737                offset += 2;
738                Value::SmallInt(val)
739            }
740
741            "tinyint" | "byte" => {
742                // TinyInt: [VInt len=1][i8]
743                if offset >= data.len() {
744                    return Err(Error::corruption(format!(
745                        "Cell '{}': unexpected end at tinyint length",
746                        column.name
747                    )));
748                }
749
750                let (remaining, len) = parse_vuint(&data[offset..]).map_err(|e| {
751                    Error::corruption(format!(
752                        "Cell '{}': failed to parse tinyint length as VInt: {:?}",
753                        column.name, e
754                    ))
755                })?;
756                let len = len as usize;
757                let bytes_consumed = data[offset..].len() - remaining.len();
758                offset += bytes_consumed;
759
760                if len != 1 {
761                    return Err(Error::corruption(format!(
762                        "Cell '{}': expected tinyint length 1, got {}",
763                        column.name, len
764                    )));
765                }
766
767                if offset >= data.len() {
768                    return Err(Error::corruption(format!(
769                        "Cell '{}': need 1 byte for tinyint, only {} available",
770                        column.name,
771                        data.len() - offset
772                    )));
773                }
774
775                let val = data[offset] as i8;
776                offset += 1;
777                Value::TinyInt(val)
778            }
779
780            "time" => {
781                // Time: [VInt len=8][i64 BE nanoseconds since midnight]
782                if offset >= data.len() {
783                    return Err(Error::corruption(format!(
784                        "Cell '{}': unexpected end at time length",
785                        column.name
786                    )));
787                }
788                let (remaining, time_len) = parse_vuint(&data[offset..]).map_err(|e| {
789                    Error::corruption(format!(
790                        "Cell '{}': failed to parse time length as VInt: {:?}",
791                        column.name, e
792                    ))
793                })?;
794                let time_len = time_len as usize;
795                let bytes_consumed = data[offset..].len() - remaining.len();
796                offset += bytes_consumed;
797                if time_len != 8 {
798                    return Err(Error::corruption(format!(
799                        "Cell '{}': expected time length 8, got {}",
800                        column.name, time_len
801                    )));
802                }
803                if offset + 8 > data.len() {
804                    return Err(Error::corruption(format!(
805                        "Cell '{}': need 8 bytes for time value, only {} available",
806                        column.name,
807                        data.len() - offset
808                    )));
809                }
810                let nanos = i64::from_be_bytes([
811                    data[offset],
812                    data[offset + 1],
813                    data[offset + 2],
814                    data[offset + 3],
815                    data[offset + 4],
816                    data[offset + 5],
817                    data[offset + 6],
818                    data[offset + 7],
819                ]);
820                offset += 8;
821                Value::Time(nanos)
822            }
823
824            "inet" => {
825                // Inet: [VInt len][address bytes] (len is 4 for IPv4, 16 for IPv6)
826                if offset >= data.len() {
827                    return Err(Error::corruption(format!(
828                        "Cell '{}': unexpected end at inet length",
829                        column.name
830                    )));
831                }
832
833                let (remaining, len) = parse_vuint(&data[offset..]).map_err(|e| {
834                    Error::corruption(format!(
835                        "Cell '{}': failed to parse inet length as VInt: {:?}",
836                        column.name, e
837                    ))
838                })?;
839                let len = len as usize;
840                let bytes_consumed = data[offset..].len() - remaining.len();
841                offset += bytes_consumed;
842
843                if len != 4 && len != 16 {
844                    return Err(Error::corruption(format!(
845                        "Cell '{}': invalid inet length {}, expected 4 or 16",
846                        column.name, len
847                    )));
848                }
849
850                if offset + len > data.len() {
851                    return Err(Error::corruption(format!(
852                        "Cell '{}': need {} bytes for inet, only {} available",
853                        column.name,
854                        len,
855                        data.len() - offset
856                    )));
857                }
858
859                let bytes = data[offset..offset + len].to_vec();
860                offset += len;
861                Value::Inet(bytes)
862            }
863
864            // Complex types: frozen, tuple, UDT
865            type_str if type_str.starts_with("frozen<") => {
866                // Frozen types: unwrap inner type and route to appropriate parser
867                let inner_type = self.extract_frozen_inner_type(type_str)?;
868
869                log::debug!(
870                    "V5CompressedLegacy: Parsing frozen type '{}' -> inner type '{}'",
871                    type_str,
872                    inner_type
873                );
874
875                // Issue #1340: extract the AUTHORITATIVE marshal element type(s)
876                // from the on-disk SerializationHeader marshal type (`header_type`).
877                // When an element is a `frozen<UDT>`, threading the marshal type lets
878                // it decode to a typed `Value::Frozen(Value::Udt)` registry-free
879                // (precedence: header marshal → registry → Blob, no byte-pattern
880                // inference — no-heuristics #28). Extracted once per frozen cell
881                // (before the element loop); the result borrows from `header_type`,
882                // so the per-element loop is allocation-free.
883                let marshal_elems = header_type.and_then(Self::extract_marshal_collection_elements);
884                // Shared for list & set (both are `Sequence`): the borrowed element
885                // marshal type, or `None` for a map / absent / mismatched marshal.
886                let sequence_marshal_elem = match &marshal_elems {
887                    Some(MarshalCollectionElements::Sequence(m)) => Some(*m),
888                    _ => None,
889                };
890
891                // Route to appropriate frozen collection parser
892                let (inner_value, new_offset) = if inner_type.starts_with("list<") {
893                    let schema_elem = self.extract_collection_element_type(&inner_type, "list")?;
894                    let element_type =
895                        Self::prefer_udt_marshal_element(sequence_marshal_elem, &schema_elem);
896                    self.parse_frozen_list_value(data, offset, element_type, column, _reader)?
897                } else if inner_type.starts_with("set<") {
898                    let schema_elem = self.extract_collection_element_type(&inner_type, "set")?;
899                    let element_type =
900                        Self::prefer_udt_marshal_element(sequence_marshal_elem, &schema_elem);
901                    self.parse_frozen_set_value(data, offset, element_type, column, _reader)?
902                } else if inner_type.starts_with("map<") {
903                    let (schema_key, schema_val) = self.extract_map_types(&inner_type)?;
904                    let (marshal_key, marshal_val) = match &marshal_elems {
905                        Some(MarshalCollectionElements::Map(k, v)) => (Some(*k), Some(*v)),
906                        _ => (None, None),
907                    };
908                    let key_type = Self::prefer_udt_marshal_element(marshal_key, &schema_key);
909                    let value_type = Self::prefer_udt_marshal_element(marshal_val, &schema_val);
910                    self.parse_frozen_map_value(
911                        data, offset, key_type, value_type, column, _reader,
912                    )?
913                } else if Self::is_udt_type(&column.data_type) {
914                    // Frozen UDT - parse using UDT parser
915                    // The column.data_type contains the full Cassandra type string including UserType
916                    log::debug!(
917                        "V5CompressedLegacy: Parsing frozen UDT column '{}' type='{}'",
918                        column.name,
919                        column.data_type
920                    );
921
922                    // Parse UDT definition from the type string
923                    let udt_def = Self::parse_udt_type_definition(&column.data_type)?;
924
925                    // First read the VInt-prefixed blob length
926                    let (remaining, blob_len_raw) = parse_vuint(&data[offset..]).map_err(|e| {
927                        Error::corruption(format!(
928                            "Frozen UDT '{}': failed to parse blob length: {:?}",
929                            column.name, e
930                        ))
931                    })?;
932                    if blob_len_raw > MAX_CELL_VALUE_LENGTH {
933                        return Err(Error::corruption(format!(
934                            "Frozen UDT '{}': blob_len {} exceeds maximum {}",
935                            column.name, blob_len_raw, MAX_CELL_VALUE_LENGTH
936                        )));
937                    }
938                    let blob_len = blob_len_raw as usize;
939                    let bytes_consumed = data[offset..].len() - remaining.len();
940                    offset += bytes_consumed;
941
942                    if offset + blob_len > data.len() {
943                        return Err(Error::corruption(format!(
944                            "Frozen UDT '{}': need {} bytes but only {} available",
945                            column.name,
946                            blob_len,
947                            data.len() - offset
948                        )));
949                    }
950
951                    // Parse UDT value from the blob
952                    let udt_data = &data[offset..offset + blob_len];
953                    let (udt_value, _) = self.parse_udt_value(udt_data, 0, &udt_def, column)?;
954                    offset += blob_len;
955
956                    (udt_value, offset)
957                } else if let Some(udt_def) = self
958                    .udt_registry
959                    .as_ref()
960                    .and_then(|reg| reg.get_udt(&self.keyspace, &inner_type).cloned())
961                {
962                    // frozen<short_udt_name>: look up the concrete UDT definition in the
963                    // registry (Issue #502).  This handles type strings like
964                    // `frozen<person>` where "person" is a registered UDT rather than a
965                    // collection or a full marshal-format UserType string.
966                    log::debug!(
967                        "V5CompressedLegacy: Resolving frozen UDT '{}' via registry for column '{}'",
968                        inner_type,
969                        column.name,
970                    );
971
972                    // Read VUInt-prefixed blob length (same framing as tuple and
973                    // marshal-format UDT cells).
974                    let (remaining, blob_len_raw) = parse_vuint(&data[offset..]).map_err(|e| {
975                        Error::corruption(format!(
976                            "Frozen UDT '{}' (column '{}'): failed to parse blob length: {:?}",
977                            inner_type, column.name, e
978                        ))
979                    })?;
980                    if blob_len_raw > MAX_CELL_VALUE_LENGTH {
981                        return Err(Error::corruption(format!(
982                            "Frozen UDT '{}' (column '{}'): blob_len {} exceeds maximum {}",
983                            inner_type, column.name, blob_len_raw, MAX_CELL_VALUE_LENGTH
984                        )));
985                    }
986                    let blob_len = blob_len_raw as usize;
987                    let len_bytes_consumed = data[offset..].len() - remaining.len();
988                    offset += len_bytes_consumed;
989
990                    if offset + blob_len > data.len() {
991                        return Err(Error::corruption(format!(
992                            "Frozen UDT '{}' (column '{}'): need {} bytes but only {} available",
993                            inner_type,
994                            column.name,
995                            blob_len,
996                            data.len() - offset
997                        )));
998                    }
999
1000                    let udt_data = &data[offset..offset + blob_len];
1001                    let (udt_value, _) = self.parse_udt_value(udt_data, 0, &udt_def, column)?;
1002                    offset += blob_len;
1003
1004                    (udt_value, offset)
1005                } else if let Some(ht) =
1006                    header_type.filter(|ht| Self::marshal_is_top_level_frozen_udt(ht))
1007                {
1008                    // Issue #1080: NO UdtRegistry is wired and the supplied schema
1009                    // short form `frozen<person_type>` carries no field defs, but
1010                    // the AUTHORITATIVE on-disk SerializationHeader marshal type for
1011                    // this column is the full
1012                    // `FrozenType(UserType(ks,hexname,field:Type,...))`. Decode the
1013                    // UDT STRUCTURALLY from that header type (no guessing, issue #28)
1014                    // rather than dropping the column (which also broke the Err→break
1015                    // loop, silently losing all trailing columns).
1016                    self.decode_frozen_udt_from_header_type(data, offset, ht, column)?
1017                } else {
1018                    // Detect bare identifiers that look like unregistered UDT names.
1019                    // A bare identifier has no '<' (not a container or tuple) and does not
1020                    // match any known CQL primitive type.  If we reach this branch with
1021                    // such an identifier it means the UDT was not in the registry — return
1022                    // an actionable schema error rather than silently producing a Blob.
1023                    //
1024                    // Legitimate fall-through types handled below:
1025                    //   • tuple<...>  (contains '<')
1026                    //   • known primitives: int, text, uuid, boolean, blob, float, double,
1027                    //     decimal, varint, bigint, counter, timestamp, date, time, duration,
1028                    //     inet, smallint, tinyint, varchar, ascii, timeuuid
1029                    const KNOWN_PRIMITIVES: &[&str] = &[
1030                        "int",
1031                        "bigint",
1032                        "counter",
1033                        "smallint",
1034                        "tinyint",
1035                        "text",
1036                        "varchar",
1037                        "ascii",
1038                        "uuid",
1039                        "timeuuid",
1040                        "boolean",
1041                        "blob",
1042                        "float",
1043                        "double",
1044                        "decimal",
1045                        "varint",
1046                        "timestamp",
1047                        "date",
1048                        "time",
1049                        "duration",
1050                        "inet",
1051                    ];
1052                    let is_container = inner_type.contains('<');
1053                    let is_primitive = KNOWN_PRIMITIVES.contains(&inner_type.as_str());
1054                    if !is_container && !is_primitive {
1055                        // Bare identifier that is neither a container nor a primitive —
1056                        // this is an unregistered UDT name.
1057                        return Err(Error::schema(format!(
1058                            "frozen<{inner}>: UDT '{inner}' not found in registry for keyspace '{}'; \
1059                             register it before reading",
1060                            self.keyspace,
1061                            inner = inner_type,
1062                        )));
1063                    }
1064                    // Non-collection / primitive frozen type — recurse normally.
1065                    // The recursive call now returns 4 elements; we only need value + offset.
1066                    let mut inner_column = column.clone();
1067                    inner_column.data_type = inner_type.clone();
1068                    let (inner_val, _inner_ts, _inner_exp, inner_off) = self
1069                        .parse_cell_value_schema_order(
1070                            data,
1071                            offset,
1072                            &inner_column,
1073                            None,
1074                            _reader,
1075                        )?;
1076                    (inner_val, inner_off)
1077                };
1078
1079                offset = new_offset;
1080
1081                // Wrap in Frozen
1082                Value::Frozen(Box::new(inner_value))
1083            }
1084
1085            type_str if type_str.starts_with("tuple<") => {
1086                // Tuple types: parse fixed number of elements
1087                self.parse_tuple_value(data, &mut offset, type_str, column, _reader)?
1088            }
1089
1090            // Non-frozen collections: list, set, map
1091            // TODO(Issue #162, Task 3): Multi-cell collection parsing
1092            //
1093            // Collections in V5CompressedLegacy are stored as MULTIPLE CELLS with path identifiers,
1094            // NOT as single blob values. The current single-cell parser cannot handle this.
1095            //
1096            // Format (from sstabledump analysis):
1097            //   {"name": "scores", "deletion_info": {...}},  // Collection tombstone
1098            //   {"name": "scores", "path": ["uuid1"], "value": 23},  // Element 1
1099            //   {"name": "scores", "path": ["uuid2"], "value": 99},  // Element 2
1100            //
1101            // Required implementation:
1102            //   1. Parse cell path (clustering key bytes) for each collection element
1103            //   2. Detect collection tombstone cell (has deletion_info, no path/value)
1104            //   3. Read N element cells (each with path + value)
1105            //   4. Aggregate elements into Value::List/Set/Map based on column type
1106            //   5. Handle different path encodings:
1107            //      - list<T>: path is UUID bytes (timeuuid for ordering)
1108            //      - set<T>: path is serialized element value (key), value is empty
1109            //      - map<K,V>: path is serialized key, value is serialized value
1110            //
1111            // This is a fundamental architectural change requiring cell-level parsing
1112            // before column-level aggregation. For now, return stub to unblock downstream work.
1113            type_str
1114                if type_str.starts_with("list<")
1115                    || type_str.starts_with("set<")
1116                    || type_str.starts_with("map<") =>
1117            {
1118                warn!(
1119                    "V5CompressedLegacy: Non-frozen collection '{}' type '{}' requires multi-cell parsing (not yet implemented). \
1120                     Collections are stored as multiple cells with path identifiers, requiring cell-level aggregation. \
1121                     Returning empty collection as placeholder. See Issue #162 Task 3 for implementation plan.",
1122                    column.name, column.data_type
1123                );
1124
1125                // Return empty collection based on type
1126                if type_str.starts_with("list<") {
1127                    Value::List(Vec::new())
1128                } else if type_str.starts_with("set<") {
1129                    Value::Set(Vec::new())
1130                } else {
1131                    Value::Map(Vec::new())
1132                }
1133            }
1134
1135            // Issue #1080 / roborev job 1363: marshal-form frozen UDT. When the
1136            // schema is DERIVED FROM the on-disk header (rather than supplied as a
1137            // CQL short form), `column.data_type` is the authoritative marshal
1138            // string `org.apache.cassandra.db.marshal.FrozenType(...UserType...)`,
1139            // which does NOT start with CQL `frozen<` and so misses the arm above.
1140            // Decode it structurally from that marshal type (same authoritative
1141            // path as the supplied-schema header fallback) instead of blobbing it.
1142            // `marshal_is_top_level_frozen_udt` accepts ONLY a top-level
1143            // `FrozenType(UserType(...))` (NOT a frozen collection that contains a
1144            // UDT, e.g. `FrozenType(ListType(UserType(...)))` — roborev 1365), and a
1145            // non-frozen top-level UDT is routed to the complex branch by
1146            // `is_complex_column`, so reaching here means a single-cell frozen UDT
1147            // → wrap in `Value::Frozen` (consistent with the CQL `frozen<` arm).
1148            _ if Self::marshal_is_top_level_frozen_udt(&column.data_type) => {
1149                let (udt_value, new_offset) = self.decode_frozen_udt_from_header_type(
1150                    data,
1151                    offset,
1152                    &column.data_type,
1153                    column,
1154                )?;
1155                offset = new_offset;
1156                Value::Frozen(Box::new(udt_value))
1157            }
1158
1159            // TODO(Issue #162): UDT parsing requires schema registry access
1160            // For now, UDTs fall through to blob. Future implementation will:
1161            // - Extract UDT name from type_str
1162            // - Look up UDT definition in schema registry
1163            // - Parse fields according to UDT schema
1164            // - Return Value::Udt(UdtValue)
1165
1166            // Default: treat as VInt-length-prefixed blob
1167            // CRITICAL: V5CompressedLegacy format uses VInt encoding for blob/bytes lengths,
1168            // NOT simple u8 length prefix. This allows blobs > 255 bytes.
1169            _ => {
1170                if offset >= data.len() {
1171                    return Err(Error::corruption(format!(
1172                        "Cell '{}': unexpected end at blob length (type: {})",
1173                        column.name, column.data_type
1174                    )));
1175                }
1176
1177                // Parse blob length as unsigned VInt (can be > 255 bytes)
1178                let (remaining, blob_len) = parse_vuint(&data[offset..]).map_err(|e| {
1179                    Error::corruption(format!(
1180                        "Cell '{}': failed to parse blob length as VInt: {:?}",
1181                        column.name, e
1182                    ))
1183                })?;
1184                let blob_len = blob_len as usize;
1185                let bytes_consumed = data[offset..].len() - remaining.len();
1186                offset += bytes_consumed;
1187
1188                if offset + blob_len > data.len() {
1189                    return Err(Error::corruption(format!(
1190                        "Cell '{}': need {} bytes for blob, only {} available (type: {})",
1191                        column.name,
1192                        blob_len,
1193                        data.len() - offset,
1194                        column.data_type
1195                    )));
1196                }
1197
1198                let blob_bytes = data[offset..offset + blob_len].to_vec();
1199                offset += blob_len;
1200                Value::Blob(blob_bytes)
1201            }
1202        };
1203
1204        Ok((value, cell_timestamp, cell_expiration, offset))
1205    }
1206}
