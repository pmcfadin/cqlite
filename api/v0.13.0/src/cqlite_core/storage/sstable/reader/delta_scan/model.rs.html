1//! Delta-scan record data model (Epic #696, Issue #697 types).
2//!
3//! These types form the public surface of the delta-scan layer: the
4//! discriminated [`DeltaRecord`] union and its supporting key/cell/metadata
5//! structs, plus the [`ScanSummary`]/[`ScanSummaryHandle`] aggregate-statistics
6//! pair.  The streaming driver that produces them lives in the sibling
7//! [`super::scan`] module.

16// ---------------------------------------------------------------------------
17// ScanSummary — scan-level aggregate statistics (Issue #700, DS4)
18// ---------------------------------------------------------------------------
19
20/// Summary statistics produced by a [`scan_delta`](super::scan_delta) run.
21///
22/// The caller receives a [`ScanSummaryHandle`] alongside the record stream;
23/// after the stream is drained the handle's counters reflect the full scan.
24///
25/// ## Element tombstones (Issue #493 / DS4)
26///
27/// Non-frozen collection cells may contain **element-level removals**
28/// (`s = s - {x}` for sets, individual map-key deletions).  V1 of the
29/// delta-scan layer cannot represent these at element granularity; they are
30/// detected, counted here, and reported as a warning — but not silently
31/// dropped.  Consumers that need element-level fidelity must wait for full
32/// Issue #493 implementation.
33#[derive(Debug, Clone)]
34pub struct ScanSummary {
35    /// Total number of element-level collection tombstones detected across
36    /// all collection columns in this scan.
37    ///
38    /// A non-zero value means the delta contains `s = s - {x}` style
39    /// removals that are not represented in the emitted [`DeltaRecord`]s
40    /// (Issue #493 follow-up).
41    pub element_tombstones_detected: u64,
42}
43
44/// Handle to in-progress scan summary counters.
45///
46/// Returned by [`scan_delta`](super::scan_delta) alongside the record receiver.
47/// After the receiver is drained (stream complete), call [`ScanSummaryHandle::read`]
48/// to obtain the final [`ScanSummary`].
49///
50/// The handle is cheaply clonable — all clones share the same counters.
51#[derive(Debug, Clone)]
52pub struct ScanSummaryHandle {
53    element_tombstones: Arc<AtomicU64>,
54}
55
56impl ScanSummaryHandle {
57    pub(super) fn new() -> Self {
58        Self {
59            element_tombstones: Arc::new(AtomicU64::new(0)),
60        }
61    }
62
63    /// Increment the element-tombstone counter by `n`.
64    pub(super) fn add_element_tombstones(&self, n: u64) {
65        self.element_tombstones.fetch_add(n, Ordering::Relaxed);
66    }
67
68    /// Read the current scan summary.  Call after draining the record receiver
69    /// for a complete picture.
70    pub fn read(&self) -> ScanSummary {
71        ScanSummary {
72            element_tombstones_detected: self.element_tombstones.load(Ordering::Relaxed),
73        }
74    }
75}

78// RowKeys

81/// Combined partition key + clustering columns for a single row (or partition).
82///
83/// For records that address a full partition (`PartitionDelete`,
84/// `StaticUpsert`, `RangeDelete`) the `clustering` vec is empty.  For
85/// row-addressed records it holds the decoded clustering column values in
86/// primary-key definition order.
87///
88/// Values are decoded according to the `TableSchema` in effect at scan time;
89/// the no-heuristics mandate applies — every component uses schema-authorised
90/// types.
91#[derive(Debug, Clone, PartialEq)]
92pub struct RowKeys {
93    /// Decoded partition-key components in definition order.
94    ///
95    /// Multi-component partition keys (composite) appear as multiple elements.
96    pub partition: Vec<Value>,
97
98    /// Decoded clustering-key components in definition order.
99    ///
100    /// Empty when the record is partition-scoped (partition delete, static
101    /// upsert, or range-delete — in which case bounds carry the clustering
102    /// information instead).
103    pub clustering: Vec<Value>,
104}
105
106impl RowKeys {
107    /// Create a partition-scoped key (no clustering components).
108    pub fn partition_only(partition: Vec<Value>) -> Self {
109        Self {
110            partition,
111            clustering: Vec::new(),
112        }
113    }
114
115    /// Create a fully-specified row key.
116    pub fn new(partition: Vec<Value>, clustering: Vec<Value>) -> Self {
117        Self {
118            partition,
119            clustering,
120        }
121    }
122}

124// ---------------------------------------------------------------------------
125// CellMeta — row liveness info
126// ---------------------------------------------------------------------------
127
128/// Liveness metadata for a row's primary-key cell.
129///
130/// Cassandra writes a liveness-info record alongside a row when the row was
131/// created with `INSERT` (not `UPDATE`).  `UPDATE` statements produce rows
132/// with no liveness info; those rows have `liveness: None` in
133/// [`DeltaRecord::Upsert`].
134///
135/// The `writetime` here is the row-level primary-key liveness timestamp,
136/// equivalent to `__ts` in the Parquet envelope.
137#[derive(Debug, Clone, PartialEq)]
138pub struct CellMeta {
139    /// Writetime in microseconds since the Unix epoch.
140    pub writetime: i64,
141
142    /// Expiry time in microseconds since the Unix epoch, if a TTL was set on
143    /// the `INSERT`.  `None` means no TTL.
144    pub expires_at: Option<i64>,
145}
146
147impl CellMeta {
148    /// Create a liveness record with no TTL.
149    pub fn new(writetime: i64) -> Self {
150        Self {
151            writetime,
152            expires_at: None,
153        }
154    }
155
156    /// Create a liveness record with a TTL expiry time.
157    pub fn with_ttl(writetime: i64, expires_at: i64) -> Self {
158        Self {
159            writetime,
160            expires_at: Some(expires_at),
161        }
162    }
163}

165// ---------------------------------------------------------------------------
166// CellDelta
167// ---------------------------------------------------------------------------
168
169/// Per-cell change record carried inside [`DeltaRecord::Upsert`] and
170/// [`DeltaRecord::StaticUpsert`].
171///
172/// A `null` [`value`][Self::value] represents a cell tombstone
173/// (`DELETE col FROM t WHERE …`).  A non-null value is the cell content as
174/// decoded by the schema-aware reader — no heuristics.
175///
176/// ## Collection columns (v1 limitation)
177///
178/// Non-frozen collection columns carry per-element writetimes and element
179/// tombstones that v1 cannot faithfully represent at element granularity.
180/// For those columns:
181///
182/// - `value` holds the elements present in this generation (for an append
183///   `s = s + {…}` that is only the appended elements — correct delta
184///   semantics).
185/// - `writetime` is the maximum element writetime.
186/// - `replaced` is `true` when the generation carries a collection tombstone
187///   (i.e. an overwrite `s = {…}`), signalling consumers to replace rather
188///   than merge.
189///
190/// Full element-level fidelity is a tracked follow-up (Issue #493).
191#[derive(Debug, Clone, PartialEq)]
192pub struct CellDelta {
193    /// The decoded cell value, or `None` for a cell tombstone.
194    pub value: Option<Value>,
195
196    /// Writetime in microseconds since the Unix epoch.
197    pub writetime: i64,
198
199    /// TTL expiry time in microseconds since the Unix epoch.
200    ///
201    /// `None` means no TTL was set on this cell.  The delta-scan layer never
202    /// resolves TTLs at scan time — whether a cell is expired is left to the
203    /// downstream consumer (idempotent output guarantee).
204    pub expires_at: Option<i64>,
205
206    /// `true` when this collection cell carries a collection-level tombstone,
207    /// meaning the downstream consumer must **replace** rather than merge the
208    /// prior collection state.
209    ///
210    /// Always `false` for scalar (non-collection) columns.
211    pub replaced: bool,
212}
213
214impl CellDelta {
215    /// Create a simple value cell with no TTL and no collection-replace flag.
216    pub fn value(value: Value, writetime: i64) -> Self {
217        Self {
218            value: Some(value),
219            writetime,
220            expires_at: None,
221            replaced: false,
222        }
223    }
224
225    /// Create a cell tombstone (no value, just a deletion timestamp).
226    pub fn tombstone(writetime: i64) -> Self {
227        Self {
228            value: None,
229            writetime,
230            expires_at: None,
231            replaced: false,
232        }
233    }
234
235    /// Create a value cell with a TTL expiry time.
236    pub fn value_with_ttl(value: Value, writetime: i64, expires_at: i64) -> Self {
237        Self {
238            value: Some(value),
239            writetime,
240            expires_at: Some(expires_at),
241            replaced: false,
242        }
243    }
244
245    /// Create a collection cell that replaces (not merges) prior state.
246    pub fn collection_replace(value: Value, writetime: i64) -> Self {
247        Self {
248            value: Some(value),
249            writetime,
250            expires_at: None,
251            replaced: true,
252        }
253    }
254}

256// ---------------------------------------------------------------------------
257// RangeBound
258// ---------------------------------------------------------------------------
259
260/// One bound of a range-tombstone clustering-key range.
261///
262/// ## Prefix bounds
263///
264/// Cassandra range tombstones may specify bounds that are **prefixes** of the
265/// full clustering key — for example, a table with clustering columns
266/// `(year INT, month INT, day INT)` can have a range tombstone covering all
267/// rows in `year = 2024`.  In that case:
268///
269/// ```text
270/// start = RangeBound { values: [Value::Integer(2024)], inclusive: true  }
271/// end   = RangeBound { values: [Value::Integer(2024)], inclusive: true  }
272/// ```
273///
274/// Trailing clustering components that are absent from the bound are simply
275/// not present in [`values`][Self::values].  Consumers must treat a prefix
276/// bound as matching all rows whose clustering key begins with the given
277/// prefix (within the inclusive/exclusive constraint).
278///
279/// ## Empty bound
280///
281/// An empty `values` vec with `inclusive: false` represents an open
282/// (unbounded) end — i.e. from/to the beginning or end of the partition.
283#[derive(Debug, Clone, PartialEq)]
284pub struct RangeBound {
285    /// The clustering-key prefix values for this bound, in primary-key order.
286    ///
287    /// May be shorter than the full clustering-key arity (prefix bound).
288    /// An empty vec means an open (unbounded) end.
289    pub values: Vec<Value>,
290
291    /// Whether the bound is inclusive (`>=` / `<=`) or exclusive (`>` / `<`).
292    pub inclusive: bool,
293}
294
295impl RangeBound {
296    /// Create a bound from a full or partial set of clustering values.
297    pub fn new(values: Vec<Value>, inclusive: bool) -> Self {
298        Self { values, inclusive }
299    }
300
301    /// Convenience: create an inclusive bound.
302    pub fn inclusive(values: Vec<Value>) -> Self {
303        Self {
304            values,
305            inclusive: true,
306        }
307    }
308
309    /// Convenience: create an exclusive bound.
310    pub fn exclusive(values: Vec<Value>) -> Self {
311        Self {
312            values,
313            inclusive: false,
314        }
315    }
316
317    /// An open (unbounded) end — matches from/to the start or end of the partition.
318    pub fn open() -> Self {
319        Self {
320            values: Vec::new(),
321            inclusive: false,
322        }
323    }
324
325    /// Returns `true` if this is a prefix bound (fewer values than the full
326    /// clustering-key arity of the table, or the full arity is not known).
327    ///
328    /// Note: callers that know the table's clustering-key arity should compare
329    /// `self.values.len()` against that arity directly rather than relying on
330    /// this method.
331    pub fn is_prefix(&self) -> bool {
332        // An empty bound is not a prefix — it is an open bound.
333        // Any non-empty bound with at least one value is potentially a prefix;
334        // the caller is responsible for comparing against the full arity.
335        !self.values.is_empty()
336    }
337}

339// ---------------------------------------------------------------------------
340// DeltaRecord
341// ---------------------------------------------------------------------------
342
343/// A single change record emitted by the delta-scan API.
344///
345/// Records stream in SSTable order (partition, then clustering).  The scan
346/// makes no cross-SSTable decisions — no merge, no GC-grace filtering.
347///
348/// ## Variant reference
349///
350/// | Variant | CQL operation | Key scope |
351/// |---------|---------------|-----------|
352/// | `Upsert` | `INSERT`/`UPDATE` on regular columns | `(pk, ck)` |
353/// | `StaticUpsert` | `UPDATE` on static columns | `pk` |
354/// | `RowDelete` | `DELETE FROM t WHERE pk=? AND ck=?` | `(pk, ck)` |
355/// | `RangeDelete` | `DELETE FROM t WHERE pk=? AND ck>=? AND ck<?` | `pk` + bounds |
356/// | `PartitionDelete` | `DELETE FROM t WHERE pk=?` | `pk` |
357///
358/// ## Downstream merge keys
359///
360/// - `Upsert` / `RowDelete` — reconcile on `(partition, clustering)`.
361/// - `StaticUpsert` — reconcile on `partition`.
362/// - `PartitionDelete` / `RangeDelete` — apply as predicates using `__ts`
363///   (i.e. `deleted_at`) to decide last-write-wins per cell.
364#[derive(Debug, Clone, PartialEq)]
365pub enum DeltaRecord {
366    /// A row-level insert or update of regular (non-static) columns.
367    ///
368    /// `liveness` is `Some` when the row was created with `INSERT` and carries
369    /// a primary-key liveness timestamp.  `UPDATE` statements produce rows
370    /// with `liveness: None`.
371    ///
372    /// Each element of `cells` is a `(column_id, delta)` pair.  Only the
373    /// columns actually written in this generation appear; absent columns are
374    /// not present (they are null in the Parquet envelope).
375    Upsert {
376        /// Partition key + clustering columns for this row.
377        keys: RowKeys,
378        /// Row liveness info, present only for `INSERT` operations.
379        liveness: Option<CellMeta>,
380        /// Per-column cell deltas for non-static columns modified in this row.
381        cells: Vec<(ColumnId, CellDelta)>,
382    },
383
384    /// An update of one or more static columns for a partition.
385    ///
386    /// Static columns belong to the partition, not individual rows.  The
387    /// clustering key is empty in `partition_key`.
388    StaticUpsert {
389        /// Partition key (no clustering columns).
390        partition_key: RowKeys,
391        /// Per-column cell deltas for static columns modified.
392        cells: Vec<(ColumnId, CellDelta)>,
393    },
394
395    /// A row-level tombstone (`DELETE FROM t WHERE pk=? AND ck=?`).
396    RowDelete {
397        /// Partition key + clustering columns identifying the deleted row.
398        keys: RowKeys,
399        /// Deletion timestamp in microseconds since the Unix epoch
400        /// (`markedForDeleteAt` in Cassandra internals).
401        deleted_at: i64,
402    },
403
404    /// A range tombstone covering a contiguous clustering-key range within a
405    /// single partition.
406    ///
407    /// `start` and `end` may be prefix bounds (see [`RangeBound`]).
408    ///
409    /// ```text
410    /// DELETE FROM t WHERE pk=1 AND ck >= 'a' AND ck < 'm'
411    /// ```
412    RangeDelete {
413        /// Partition key (no clustering columns — bounds carry the range).
414        partition_key: RowKeys,
415        /// Inclusive or exclusive lower clustering-key bound.
416        start: RangeBound,
417        /// Inclusive or exclusive upper clustering-key bound.
418        end: RangeBound,
419        /// Deletion timestamp in microseconds since the Unix epoch.
420        deleted_at: i64,
421    },
422
423    /// A partition-level tombstone (`DELETE FROM t WHERE pk=?`).
424    ///
425    /// Supersedes every row and cell in the partition whose writetime is older
426    /// than `deleted_at`.
427    PartitionDelete {
428        /// Partition key (no clustering columns).
429        partition_key: RowKeys,
430        /// Deletion timestamp in microseconds since the Unix epoch.
431        deleted_at: i64,
432    },
433}
434
435impl DeltaRecord {
436    /// Return the partition-key portion of any record type.
437    pub fn partition_key(&self) -> &[Value] {
438        match self {
439            DeltaRecord::Upsert { keys, .. } => &keys.partition,
440            DeltaRecord::StaticUpsert { partition_key, .. } => &partition_key.partition,
441            DeltaRecord::RowDelete { keys, .. } => &keys.partition,
442            DeltaRecord::RangeDelete { partition_key, .. } => &partition_key.partition,
443            DeltaRecord::PartitionDelete { partition_key, .. } => &partition_key.partition,
444        }
445    }
446
447    /// Return the `__op` discriminator string used in the Parquet envelope.
448    pub fn op_name(&self) -> &'static str {
449        match self {
450            DeltaRecord::Upsert { .. } => "upsert",
451            DeltaRecord::StaticUpsert { .. } => "static_upsert",
452            DeltaRecord::RowDelete { .. } => "row_delete",
453            DeltaRecord::RangeDelete { .. } => "range_delete",
454            DeltaRecord::PartitionDelete { .. } => "partition_delete",
455        }
456    }
457}
