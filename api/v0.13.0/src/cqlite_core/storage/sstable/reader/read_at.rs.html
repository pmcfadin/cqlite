1//! Positional (`pread`-style) read backends for the SSTable point-read path
2//! (issue #1573, Epic C / C2).
3//!
4//! # Why this exists
5//!
6//! The historical point-read path served every read through a single
7//! [`BlockSource`](super::source::BlockSource) behind an
8//! `Arc<Mutex<BlockSource>>`. That mutex exists ONLY to guard a mutable seek
9//! cursor, and it was held **across disk I/O**, so N concurrent point reads on
10//! one SSTable serialized one-at-a-time (the convoy). The BTI variant sidestepped
11//! the mutex by paying `File::open` **per lookup** instead — trading the convoy
12//! for per-op `open(2)` syscalls and an fd-exhaustion risk under load.
13//!
14//! Both pathologies share one root cause: a *stateful* file cursor. [`ReadAt`]
15//! removes the state. A positioned read carries its offset as a call parameter,
16//! not as shared mutable file state, so:
17//!
18//! - **No mutable position** — the trait has no seek cursor.
19//! - **No `&mut self`** — every method takes `&self`, so concurrent callers share
20//!   one value with no exclusive borrow and no mutex.
21//! - **No seek** — the offset is a parameter of the read.
22//!
23//! A positioned read on a shared file descriptor is independent of every other
24//! positioned read (POSIX `pread`/`FileExt::read_at` does not touch the fd's
25//! offset), so the lock is not merely moved off the point path — it is removed.
26//!
27//! # Backends
28//!
29//! | Backend | `read_at` | Notes |
30//! |---------|-----------|-------|
31//! | [`PlainFileReadAt`] | `FileExt::read_at` (Unix, lock-free) / `Mutex`-guarded `seek_read` (Windows) on ONE shared `File` | Unix `pread` is positional so reads are independent (no lock); Windows `seek_read` moves the shared cursor, so it is serialized under a `Mutex`. Positioned reads on a shared fd are independent — no lock. |
32//! | [`MmapReadAt`] | slice indexing into the resident map | Bounds-checked; zero syscalls per read. |
33//! | [`DirectReadAt`] (Unix) | aligned positioned read honoring the `O_DIRECT` 4K rule | A per-call aligned bounce buffer keeps `&self` (no shared cursor). |
34//!
35//! The trait is intentionally shaped so the windowed streaming scan can adopt it
36//! later (audit F3) without requiring it now; this change migrates only the
37//! point-read path (design.md Decision 4).
38

45/// A positional byte source: reads are addressed by an explicit `offset`, never
46/// by a shared mutable cursor.
47///
48/// Every method takes `&self`. Two concurrent `read_at` calls at different
49/// offsets are independent — neither observes nor disturbs the other's position,
50/// because there is no position. This is the whole point of the refactor
51/// (issue #1573): concurrent point reads on one SSTable no longer serialize on a
52/// cross-I/O mutex, and BTI lookups no longer `open(2)` per call.
53pub(crate) trait ReadAt: Send + Sync {
54    /// Read up to `buf.len()` bytes starting at absolute file `offset`, returning
55    /// the number of bytes read. A short read (fewer than `buf.len()`) is legal
56    /// at/near EOF, exactly like POSIX `pread`.
57    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize>;
58
59    /// Total length of the backing file in bytes.
60    fn len(&self) -> u64;
61
62    /// Read EXACTLY `buf.len()` bytes starting at `offset`, erroring with
63    /// [`std::io::ErrorKind::UnexpectedEof`] if the source is exhausted first.
64    ///
65    /// Default impl loops [`read_at`](Self::read_at) so every backend gets an
66    /// exact-read helper for free; backends may override if they can do better.
67    fn read_exact_at(&self, offset: u64, buf: &mut [u8]) -> Result<()> {
68        let mut filled = 0usize;
69        while filled < buf.len() {
70            let n = self.read_at(offset + filled as u64, &mut buf[filled..])?;
71            if n == 0 {
72                return Err(Error::Io(std::io::Error::new(
73                    std::io::ErrorKind::UnexpectedEof,
74                    format!(
75                        "read_exact_at: reached EOF after {filled} of {} bytes at offset 0x{offset:x}",
76                        buf.len()
77                    ),
78                )));
79            }
80            filled += n;
81        }
82        Ok(())
83    }
84}

86/// Delegating impl so an `Arc<dyn ReadAt>` (or `Arc<ConcreteBackend>`) is itself a
87/// `ReadAt`. Lets the point path hold a single `Arc<dyn ReadAt>` and lets a test
88/// wrap the reader's existing shared source without unsizing gymnastics.
89impl<T: ReadAt + ?Sized> ReadAt for Arc<T> {
90    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize> {
91        (**self).read_at(offset, buf)
92    }
93    fn len(&self) -> u64 {
94        (**self).len()
95    }
96    fn read_exact_at(&self, offset: u64, buf: &mut [u8]) -> Result<()> {
97        (**self).read_exact_at(offset, buf)
98    }
99}

176/// Positional reads over a plain file handle.
177///
178/// Wraps ONE shared `std::fs::File`, opened once at reader open and shared for the
179/// reader's lifetime (design.md Decision 3).
180///
181/// On **Unix**, `FileExt::read_at` issues a `pread` that does not touch the fd's
182/// offset, so concurrent reads on this single handle are independent — no lock, no
183/// per-lookup `open(2)`.
184///
185/// On **Windows** there is no positional `pread`: `seek_read` MOVES the handle's
186/// file pointer, so two concurrent reads on one shared handle would race the cursor
187/// and return the wrong bytes (silent corruption). The Windows arm therefore guards
188/// the handle with a `std::sync::Mutex` and performs the seek+read under the lock,
189/// serializing Windows point reads for correctness while still opening the fd only
190/// once. The Unix arm keeps its lock-free `Arc<File>` (issue #1573 roborev finding).
191pub(crate) struct PlainFileReadAt {
192    /// Unix: shared handle, read lock-free via positional `pread` (no cursor).
193    #[cfg(unix)]
194    file: Arc<std::fs::File>,
195    /// Windows: `seek_read` mutates the shared cursor, so the handle is `Mutex`-
196    /// guarded and read+seek happen atomically under the lock (see struct docs).
197    #[cfg(windows)]
198    file: std::sync::Mutex<std::fs::File>,
199    len: u64,
200}
201
202impl PlainFileReadAt {
203    /// Wrap an already-open file handle whose length is `len`.
204    pub(crate) fn new(file: std::fs::File, len: u64) -> Self {
205        Self {
206            #[cfg(unix)]
207            file: Arc::new(file),
208            #[cfg(windows)]
209            file: std::sync::Mutex::new(file),
210            len,
211        }
212    }
213
214    /// Open `path` read-only for positioned reads, recording one `open(2)` in the
215    /// read-work counters (consumer C2). This is the single per-reader open the
216    /// BTI point path used to pay per lookup.
217    pub(crate) fn open(path: &std::path::Path, len: u64) -> Result<Self> {
218        crate::storage::sstable::read_work_counters::record_file_open();
219        let file = std::fs::File::open(path)?;
220        Ok(Self::new(file, len))
221    }
222}
223
224impl ReadAt for PlainFileReadAt {
225    #[cfg(unix)]
226    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize> {
227        use std::os::unix::fs::FileExt;
228        Ok(self.file.read_at(buf, offset)?)
229    }
230
231    #[cfg(windows)]
232    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize> {
233        // Windows has no positional `pread`; `seek_read` MOVES the handle's file
234        // pointer, so two concurrent `read_at` calls on the one shared handle would
235        // race the cursor and return the wrong bytes (silent corruption). We hold a
236        // `Mutex` across the seek+read so Windows point reads are serialized and
237        // correct, while still opening the fd only once (issue #1573 roborev
238        // finding). This intentionally trades Windows concurrency for correctness;
239        // the Unix arm above needs no lock because `FileExt::read_at` is positional
240        // and never touches the fd offset. Recover a poisoned lock in-place (a
241        // panicking reader cannot leave the handle in a torn state) rather than
242        // `unwrap`/`expect` in library code.
243        use std::os::windows::fs::FileExt;
244        let file = self.file.lock().unwrap_or_else(|e| e.into_inner());
245        Ok(file.seek_read(buf, offset)?)
246    }
247
248    fn len(&self) -> u64 {
249        self.len
250    }
251}

253/// Positional reads over a memory-mapped file.
254///
255/// A slice read out of the resident map — zero syscalls, and trivially `&self`
256/// (the map is immutable). Shares the reader's existing `Arc<Mmap>` when the
257/// backend is mmap, so no extra mapping or fd is created.
258pub(crate) struct MmapReadAt {
259    mmap: Arc<Mmap>,
260}
261
262impl MmapReadAt {
263    pub(crate) fn new(mmap: Arc<Mmap>) -> Self {
264        Self { mmap }
265    }
266}
267
268impl ReadAt for MmapReadAt {
269    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize> {
270        let data: &[u8] = &self.mmap;
271        let len = data.len() as u64;
272        // At/past EOF: zero bytes, matching `File`/pread semantics.
273        if offset >= len {
274            return Ok(0);
275        }
276        let start = offset as usize;
277        let avail = &data[start..];
278        let n = avail.len().min(buf.len());
279        buf[..n].copy_from_slice(&avail[..n]);
280        Ok(n)
281    }
282
283    fn len(&self) -> u64 {
284        self.mmap.len() as u64
285    }
286}

288/// I/O alignment (bytes) for `O_DIRECT`: the offset, length, and buffer must all
289/// be a multiple of the device logical block size. 4096 is a safe superset of the
290/// common 512/4096 sector sizes (mirrors `source::DIRECT_IO_ALIGN`).
291#[cfg(unix)]
292const DIRECT_IO_ALIGN: usize = 4096;
293
294/// Positional reads over a file opened with direct I/O (`O_DIRECT` / `F_NOCACHE`).
295///
296/// `O_DIRECT` mandates aligned transfers, so each `read_at` allocates its OWN
297/// aligned bounce buffer, `pread`s an aligned superset of the requested range,
298/// and copies out the requested sub-range. Because the bounce buffer is per-call
299/// (never shared), the backend keeps `&self` with no shared mutable cursor — two
300/// concurrent `read_at`s never touch the same buffer.
301#[cfg(unix)]
302pub(crate) struct DirectReadAt {
303    file: Arc<std::fs::File>,
304    len: u64,
305}
306
307#[cfg(unix)]
308impl DirectReadAt {
309    /// Open `path` with direct I/O for positioned reads. Records one `open(2)`
310    /// (consumer C2). Returns an error if the platform/filesystem refuses direct
311    /// I/O so the caller can fall back to buffered.
312    pub(crate) fn open(path: &std::path::Path, len: u64) -> Result<Self> {
313        let file = super::source::open_direct_file(path)?;
314        crate::storage::sstable::read_work_counters::record_file_open();
315        Ok(Self {
316            file: Arc::new(file),
317            len,
318        })
319    }
320}
321
322#[cfg(unix)]
323impl ReadAt for DirectReadAt {
324    fn read_at(&self, offset: u64, buf: &mut [u8]) -> Result<usize> {
325        use std::os::unix::fs::FileExt;
326        if offset >= self.len || buf.is_empty() {
327            return Ok(0);
328        }
329        let align = DIRECT_IO_ALIGN as u64;
330        let aligned_off = (offset / align) * align;
331        let head = (offset - aligned_off) as usize;
332        // Round the aligned span up to cover [offset, offset+buf.len()).
333        let want_end = offset
334            .checked_add(buf.len() as u64)
335            .ok_or_else(|| Error::corruption("direct read_at: offset+len overflow"))?
336            .min(self.len);
337        let span = (want_end - aligned_off) as usize;
338        let aligned_len = span
339            .checked_next_multiple_of(DIRECT_IO_ALIGN)
340            .unwrap_or(usize::MAX & !(DIRECT_IO_ALIGN - 1));
341        let mut bounce = AlignedBuf::new(aligned_len, DIRECT_IO_ALIGN)?;
342        let got = self.file.read_at(bounce.as_mut_slice(), aligned_off)?;
343        if got <= head {
344            return Ok(0);
345        }
346        let usable = &bounce.as_slice()[head..got];
347        let n = usable.len().min(buf.len());
348        buf[..n].copy_from_slice(&usable[..n]);
349        Ok(n)
350    }
351
352    fn len(&self) -> u64 {
353        self.len
354    }
355}

357/// A heap allocation whose start address is aligned to a requested boundary, as
358/// required for `O_DIRECT` transfer buffers. Local to the point-read path so it
359/// stays independent of the scan-cursor `AlignedBuf` in `source.rs`.
360#[cfg(unix)]
361struct AlignedBuf {
362    ptr: std::ptr::NonNull<u8>,
363    layout: std::alloc::Layout,
364}
365
366// SAFETY: `AlignedBuf` uniquely owns its allocation for its whole lifetime; the
367// raw pointer is never aliased or shared, so moving it across threads is sound.
368#[cfg(unix)]
369unsafe impl Send for AlignedBuf {}
370
371#[cfg(unix)]
372impl AlignedBuf {
373    fn new(size: usize, align: usize) -> Result<Self> {
374        let size = size.max(align);
375        let layout = std::alloc::Layout::from_size_align(size, align)
376            .map_err(|e| Error::Io(std::io::Error::new(std::io::ErrorKind::InvalidInput, e)))?;
377        // SAFETY: `layout` has non-zero size (size >= align >= 1).
378        let raw = unsafe { std::alloc::alloc_zeroed(layout) };
379        let ptr = std::ptr::NonNull::new(raw).ok_or_else(|| {
380            Error::Io(std::io::Error::new(
381                std::io::ErrorKind::OutOfMemory,
382                "aligned alloc failed",
383            ))
384        })?;
385        Ok(Self { ptr, layout })
386    }
387
388    fn as_slice(&self) -> &[u8] {
389        // SAFETY: `ptr` is valid for `layout.size()` initialised bytes (zeroed on
390        // alloc, then overwritten by reads) and borrowed immutably here.
391        unsafe { std::slice::from_raw_parts(self.ptr.as_ptr(), self.layout.size()) }
392    }
393
394    fn as_mut_slice(&mut self) -> &mut [u8] {
395        // SAFETY: `ptr` is valid for `layout.size()` bytes and uniquely borrowed.
396        unsafe { std::slice::from_raw_parts_mut(self.ptr.as_ptr(), self.layout.size()) }
397    }
398}
399
400#[cfg(unix)]
401impl Drop for AlignedBuf {
402    fn drop(&mut self) {
403        // SAFETY: `ptr`/`layout` are exactly the values from the matching
404        // `alloc_zeroed`, freed exactly once here.
405        unsafe { std::alloc::dealloc(self.ptr.as_ptr(), self.layout) };
406    }
407}

508    /// The `O_DIRECT` backend's per-call 4K alignment math (`aligned_off`/`head`/
509    /// `want_end`/`checked_next_multiple_of`), aligned bounce buffer, and
510    /// short-read/EOF handling must return EXACTLY the bytes a plain positioned
511    /// read would, at every awkward alignment: offset 0, an unaligned offset, an
512    /// offset spanning a 4K block boundary, and a partial final block at/near EOF.
513    /// `PlainFileReadAt` over the same file is the oracle. Guarded `#[cfg(unix)]`
514    /// and SKIPS GRACEFULLY when the temp filesystem refuses `O_DIRECT` (common on
515    /// tmpfs/overlayfs) — mirroring `source::direct_cursor_reads_and_seeks`.
516    #[cfg(unix)]
517    #[test]
518    fn direct_read_at_matches_plain_oracle_at_boundaries() {
519        // File size deliberately NOT a 4096 multiple → forces a partial final
520        // block and non-trivial `aligned_len` rounding.
521        let len = DIRECT_IO_ALIGN * 2 + 1234;
522        let mut data = vec![0u8; len];
523        // Deterministic pseudo-random content (LCG) so mismatches are meaningful.
524        let mut x = 0x1234_5678u32;
525        for b in data.iter_mut() {
526            x = x.wrapping_mul(1_664_525).wrapping_add(1_013_904_223);
527            *b = (x >> 24) as u8;
528        }
529        let path = temp_file(&data, "direct");
530        let oracle = PlainFileReadAt::open(&path, data.len() as u64).unwrap();
531
532        // Open may fail on filesystems without O_DIRECT support → skip cleanly.
533        let direct = match DirectReadAt::open(&path, data.len() as u64) {
534            Ok(d) => d,
535            Err(e) => {
536                eprintln!("O_DIRECT open unsupported here ({e}); skipping DirectReadAt test");
537                std::fs::remove_file(&path).ok();
538                return;
539            }
540        };
541        // Some filesystems accept the O_DIRECT open but refuse aligned reads
542        // (EINVAL) → probe once and skip cleanly rather than fail.
543        let mut probe = [0u8; 16];
544        if let Err(e) = direct.read_at(0, &mut probe) {
545            eprintln!("O_DIRECT read unsupported here ({e}); skipping DirectReadAt test");
546            std::fs::remove_file(&path).ok();
547            return;
548        }
549
550        // (offset, length) pairs exercising each alignment corner.
551        let align = DIRECT_IO_ALIGN as u64;
552        let cases: &[(u64, usize)] = &[
553            (0, 100),                  // offset 0, within first block
554            (37, 200),                 // unaligned, within first block
555            (align - 10, 50),          // spans the first 4K boundary
556            (align, DIRECT_IO_ALIGN),  // aligned start, spans into next block
557            (align + 7, 5000),         // unaligned start crossing multiple blocks
558            ((len - 300) as u64, 300), // exact final bytes (partial final block)
559            ((len - 10) as u64, 5),    // near EOF within the partial final block
560        ];
561        for &(off, n) in cases {
562            let mut want = vec![0u8; n];
563            oracle.read_exact_at(off, &mut want).unwrap();
564            let mut got = vec![0u8; n];
565            direct.read_exact_at(off, &mut got).unwrap();
566            assert_eq!(got, want, "DirectReadAt mismatch at off={off} n={n}");
567        }
568
569        // Short read at EOF: request more than remains → exactly the tail, and
570        // byte-for-byte identical to the oracle's short read.
571        let tail_off = (len - 100) as u64;
572        let mut big = vec![0u8; 500];
573        let got_n = direct.read_at(tail_off, &mut big).unwrap();
574        let mut oracle_big = vec![0u8; 500];
575        let oracle_n = oracle.read_at(tail_off, &mut oracle_big).unwrap();
576        assert_eq!(got_n, oracle_n, "short-read length must match oracle");
577        assert_eq!(got_n, 100, "should read exactly the 100 remaining bytes");
578        assert_eq!(&big[..got_n], &oracle_big[..oracle_n]);
579
580        // Fully past EOF → 0 bytes (matches File/pread and MmapReadAt).
581        let mut none = [0u8; 32];
582        assert_eq!(direct.read_at(len as u64, &mut none).unwrap(), 0);
583        assert_eq!(direct.read_at(len as u64 + align, &mut none).unwrap(), 0);
584
585        // read_exact_at fully past EOF → UnexpectedEof, consistent with the
586        // other backends.
587        let err = direct.read_exact_at(len as u64, &mut none).unwrap_err();
588        match err {
589            Error::Io(e) => assert_eq!(e.kind(), std::io::ErrorKind::UnexpectedEof),
590            other => panic!("expected UnexpectedEof, got {other:?}"),
591        }
592
593        std::fs::remove_file(&path).ok();
594    }
595}
