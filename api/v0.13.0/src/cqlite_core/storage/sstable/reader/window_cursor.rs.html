1//! Sliding decompressed-byte window helpers (issue #1589).
2//!
3//! Both the user-facing windowed scan (`scan_stream_windowed`) and the streaming
4//! compaction driver (`data_access::compaction`) keep a `Vec<u8>` of decompressed
5//! bytes, parse CONFIRMED partitions from the FRONT, and refill from the tail as
6//! new chunks arrive. Consuming the front with `window.drain(0..consumed)` memmoves
7//! the ENTIRE residual tail on every confirmed partition — Θ(P·W) bytes moved per
8//! window for P tiny partitions over a W-byte residual (the pathological case is a
9//! partition-dense table: thousands of one-row partitions packed into one window).
10//!
11//! This module hosts the test-only byte-movement [`probe`] that measures that cost
12//! (issue #1589), and the [`WindowCursor`] cursor that reduces it to Θ(W) per
13//! window.
14
15/// A decompressed-byte window with a FRONT CURSOR (issue #1589).
16///
17/// Invariant: `start <= buf.len()`. The logical window (the bytes not yet consumed)
18/// is `buf[start..]`. [`consume`](Self::consume) advances `start` with NO memmove;
19/// [`refill`](Self::refill) reclaims the consumed prefix `buf[..start]` with a
20/// SINGLE compaction — once per refill, not once per confirmed partition — then
21/// appends the new chunk. So each byte is physically moved at most once per refill
22/// it survives (Θ(W) per window over the whole window lifetime), replacing the
23/// per-partition `window.drain(0..consumed)` that moved Θ(P·W).
24///
25/// Window sizing / backpressure semantics are unchanged: [`as_slice`](Self::as_slice)
26/// yields exactly the same unconsumed bytes the prior `Vec` did after each drain;
27/// only the physical byte movement differs.
28pub(crate) struct WindowCursor {
29    buf: Vec<u8>,
30    start: usize,
31}
32
33impl WindowCursor {
34    /// A new, empty window.
35    pub(crate) fn new() -> Self {
36        Self {
37            buf: Vec::new(),
38            start: 0,
39        }
40    }
41
42    /// The unconsumed bytes — the logical window the parser reads. Cursor-relative.
43    #[inline]
44    pub(crate) fn as_slice(&self) -> &[u8] {
45        &self.buf[self.start..]
46    }
47
48    /// Number of unconsumed bytes remaining in the logical window.
49    #[inline]
50    pub(crate) fn len(&self) -> usize {
51        self.buf.len() - self.start
52    }
53
54    /// Whether the logical window is empty (all appended bytes consumed).
55    #[inline]
56    pub(crate) fn is_empty(&self) -> bool {
57        self.start >= self.buf.len()
58    }
59
60    /// Consume `n` bytes from the FRONT of the logical window by advancing the
61    /// cursor — NO memmove. `n` is clamped to the unconsumed length, exactly
62    /// matching the prior `drain(0..n.min(len))` semantics (a parser reporting
63    /// `consumed > remaining` cannot advance the cursor past the window end).
64    #[inline]
65    pub(crate) fn consume(&mut self, n: usize) {
66        // `capped <= self.len()` and `start + self.len() == buf.len()`, so the
67        // cursor never runs past the buffer; no overflow (capped <= buf.len()).
68        let capped = n.min(self.len());
69        self.start += capped;
70    }
71
72    /// Append a freshly decompressed `chunk`. First reclaims the already-consumed
73    /// prefix `buf[..start]` with a SINGLE compaction (one memmove of the residual
74    /// unconsumed tail), resetting the cursor to 0, THEN extends with `chunk`. The
75    /// residual tail is therefore moved at most once PER REFILL — never once per
76    /// confirmed partition (issue #1589).
77    pub(crate) fn refill(&mut self, chunk: &[u8]) {
78        if self.start > 0 {
79            // Residual (unconsumed) bytes this single compaction moves. `copy_within`
80            // + `truncate` moves exactly `residual` bytes — the same work a
81            // `drain(0..start)` would do, but done once per refill, not per partition.
82            let residual = self.len();
83            #[cfg(feature = "scan-offload-probe")]
84            probe::note_bytes_memmoved(residual);
85            self.buf.copy_within(self.start.., 0);
86            self.buf.truncate(residual);
87            self.start = 0;
88        }
89        #[cfg(feature = "scan-offload-probe")]
90        probe::note_bytes_appended(chunk.len());
91        self.buf.extend_from_slice(chunk);
92    }
93}

95/// Test-only probe (issue #1589) for the sliding window's byte-movement cost.
96///
97/// Records the total bytes physically memmoved by the window's consume/compact path
98/// and the total decompressed bytes appended, so a guard test can prove that a
99/// partition-dense scan moves O(appended) bytes total — each byte at most ~once per
100/// window — rather than the Θ(P·W) the per-partition front-drain moved. Compiled
101/// ONLY under the non-default `scan-offload-probe` feature, so it adds zero cost and
102/// no public surface in normal/release builds; both the windowed scan and the
103/// streaming compaction driver feed this one counter, so it covers both sites.
104#[cfg(feature = "scan-offload-probe")]
105#[doc(hidden)]
106pub mod probe {
107    use std::sync::atomic::{AtomicBool, AtomicUsize, Ordering};
108
109    static ARMED: AtomicBool = AtomicBool::new(false);
110    static BYTES_MEMMOVED: AtomicUsize = AtomicUsize::new(0);
111    static BYTES_APPENDED: AtomicUsize = AtomicUsize::new(0);
112
113    /// Arm the probe and reset the counters. Call before driving a scan/compaction.
114    pub fn arm() {
115        BYTES_MEMMOVED.store(0, Ordering::SeqCst);
116        BYTES_APPENDED.store(0, Ordering::SeqCst);
117        ARMED.store(true, Ordering::SeqCst);
118    }
119
120    /// Disarm the probe (restores the production no-op state).
121    pub fn disarm() {
122        ARMED.store(false, Ordering::SeqCst);
123    }
124
125    /// Record `n` bytes physically memmoved while consuming/compacting the window.
126    pub(crate) fn note_bytes_memmoved(n: usize) {
127        if ARMED.load(Ordering::Relaxed) {
128            BYTES_MEMMOVED.fetch_add(n, Ordering::Relaxed);
129        }
130    }
131
132    /// Record `n` bytes appended (decompressed) into the window, if armed.
133    pub(crate) fn note_bytes_appended(n: usize) {
134        if ARMED.load(Ordering::Relaxed) {
135            BYTES_APPENDED.fetch_add(n, Ordering::Relaxed);
136        }
137    }
138
139    /// Total bytes physically memmoved by the window during the armed run.
140    pub fn recorded_bytes_memmoved() -> usize {
141        BYTES_MEMMOVED.load(Ordering::Relaxed)
142    }
143
144    /// Total decompressed bytes appended into the window during the armed run.
145    pub fn recorded_bytes_appended() -> usize {
146        BYTES_APPENDED.load(Ordering::Relaxed)
147    }
148}

150#[cfg(test)]
151mod tests {
152    use super::WindowCursor;
153
154    #[test]
155    fn consume_advances_cursor_without_memmove() {
156        let mut w = WindowCursor::new();
157        w.refill(b"abcdefgh");
158        assert_eq!(w.as_slice(), b"abcdefgh");
159        assert_eq!(w.len(), 8);
160        assert!(!w.is_empty());
161        w.consume(3);
162        assert_eq!(w.as_slice(), b"defgh");
163        assert_eq!(w.len(), 5);
164        w.consume(5);
165        assert!(w.is_empty());
166        assert_eq!(w.len(), 0);
167        assert_eq!(w.as_slice(), b"");
168    }
169
170    #[test]
171    fn consume_clamps_to_remaining() {
172        let mut w = WindowCursor::new();
173        w.refill(b"xyz");
174        // A parser reporting more consumed than remaining must not run past the end
175        // (mirrors the prior `drain(0..take.min(window.len()))` clamp).
176        w.consume(100);
177        assert!(w.is_empty());
178        assert_eq!(w.len(), 0);
179    }
180
181    #[test]
182    fn refill_compacts_consumed_prefix_and_preserves_residual() {
183        let mut w = WindowCursor::new();
184        w.refill(b"aabbccdd");
185        w.consume(6); // consume "aabbcc", residual "dd"
186        assert_eq!(w.as_slice(), b"dd");
187        w.refill(b"eeff"); // compact residual "dd" once, then append "eeff"
188        assert_eq!(w.as_slice(), b"ddeeff");
189        assert_eq!(w.len(), 6);
190        // A second consume + refill continues from the compacted base.
191        w.consume(2); // "dd"
192        w.refill(b"gg");
193        assert_eq!(w.as_slice(), b"eeffgg");
194    }
195
196    #[test]
197    fn refill_from_empty_and_fully_consumed_windows() {
198        let mut w = WindowCursor::new();
199        // Refill into an empty window: no residual to compact.
200        w.refill(b"first");
201        w.consume(5); // fully consumed
202        assert!(w.is_empty());
203        // Refill after full consumption: compaction removes everything, then appends.
204        w.refill(b"second");
205        assert_eq!(w.as_slice(), b"second");
206        assert_eq!(w.len(), 6);
207    }
