1//! Backing store abstraction for SSTable block I/O.
2//!
3//! The SSTable reader accesses Data.db through a single seekable byte source.
4//! Historically this was always a buffered file handle (`BufReader<File>`).
5//! [`BlockSource`] generalizes that to two interchangeable backends:
6//!
7//! - [`BlockSource::Buffered`]: standard buffered async file I/O. Reads go
8//!   through the OS page cache with kernel read-ahead. This is the safe,
9//!   universally-portable default and is used for small files (where mmap
10//!   setup overhead is not worth it) or when mmap is disabled.
11//! - [`BlockSource::Mapped`]: a memory-mapped view of the file. The OS maps
12//!   the file into the process address space and serves reads straight from
13//!   the page cache with no per-block `read` syscall and no extra copy into a
14//!   buffered reader. This is well-suited to local SSTable analysis where the
15//!   same files are scanned repeatedly across queries (the page cache is
16//!   shared and reused), which is also the strategy Cassandra itself uses for
17//!   its read path (`disk_access_mode: mmap`).
18//!
19//! Both variants implement [`tokio::io::AsyncRead`] and
20//! [`tokio::io::AsyncSeek`], so every existing call site
21//! (`seek` / `stream_position` / `read` / `read_exact`) works against either
22//! backend without modification. The mmap backend is purely in-memory, so its
23//! poll methods always complete synchronously (`Poll::Ready`).
24

39/// A seekable byte source backing an [`SSTableReader`](super::types::SSTableReader).
40///
41/// See the module documentation for the trade-offs between the two backends.
42pub(crate) enum BlockSource {
43    /// Buffered file I/O through the OS page cache. `len` caches the immutable
44    /// file length so the block-I/O layer never re-probes it with `seek(End)`
45    /// per chunk (issue #1586).
46    Buffered {
47        reader: BufReader<File>,
48        len: Option<u64>,
49    },
50    /// Memory-mapped file view served directly from the page cache.
51    Mapped(MmapCursor),
52    /// Direct I/O (`O_DIRECT` / `F_NOCACHE`) that bypasses the page cache.
53    #[cfg(unix)]
54    Direct(DirectCursor),
55}
56
57impl BlockSource {
58    /// Create a buffered (non-mmap) source with the length resolved lazily on the
59    /// first [`Self::len`] call. Test-only: production paths know the length at
60    /// open and use [`Self::buffered_sized`] (issue #1586).
61    #[cfg(test)]
62    pub(crate) fn buffered(file: File) -> Self {
63        BlockSource::Buffered {
64            reader: BufReader::new(file),
65            len: None,
66        }
67    }
68
69    /// Create a buffered source with its (immutable) file length already known,
70    /// so the block-I/O layer never issues a size probe at all (issue #1586).
71    pub(crate) fn buffered_sized(file: File, len: u64) -> Self {
72        BlockSource::Buffered {
73            reader: BufReader::new(file),
74            len: Some(len),
75        }
76    }
77
78    /// The total byte length of the backing file. Resident for mmap/direct; for a
79    /// buffered source it is cached (derived once via `metadata`, SSTables being
80    /// immutable) rather than re-probed with `seek(End)` per chunk (issue #1586).
81    pub(crate) async fn len(&mut self) -> io::Result<u64> {
82        match self {
83            BlockSource::Buffered { reader, len } => match *len {
84                Some(l) => Ok(l),
85                None => {
86                    let l = reader.get_ref().metadata().await?.len();
87                    *len = Some(l);
88                    Ok(l)
89                }
90            },
91            BlockSource::Mapped(c) => Ok(c.mmap.len() as u64),
92            #[cfg(unix)]
93            BlockSource::Direct(c) => Ok(c.len),
94        }
95    }
96
97    /// Create a memory-mapped source from a previously mapped file.
98    pub(crate) fn mapped(mmap: Arc<Mmap>) -> Self {
99        BlockSource::Mapped(MmapCursor::new(mmap))
100    }
101
102    /// Create a direct-I/O source (page-cache-bypassing) over `cursor`.
103    #[cfg(unix)]
104    pub(crate) fn direct(cursor: DirectCursor) -> Self {
105        BlockSource::Direct(cursor)
106    }
107
108    /// Returns `true` when this source is backed by a memory map.
109    #[cfg(test)]
110    pub(crate) fn is_mmap(&self) -> bool {
111        matches!(self, BlockSource::Mapped(_))
112    }
113
114    /// Returns `true` when this source is backed by direct I/O.
115    #[cfg(all(test, unix))]
116    pub(crate) fn is_direct(&self) -> bool {
117        matches!(self, BlockSource::Direct(_))
118    }
119}

121/// Template for minting fresh, independent [`BlockSource`]s — one per scan.
122///
123/// Issue #815: concurrent scans on a single `SSTableReader` must not share a
124/// mutable file position or chunk index, otherwise their seeks interleave and
125/// corrupt each other's reads (the bug #805 fixed by serializing with a mutex).
126/// Instead of serializing, each scan now opens its own [`ScanCursor`] from this
127/// template, so they run in parallel while staying correct. SSTable files are
128/// immutable, so minting extra views is always safe.
129pub(crate) enum ScanSource {
130    /// Reopen the file for each scan, giving it its own OS file handle and seek
131    /// position. The per-handle cost is a small buffered reader.
132    Buffered {
133        /// Immutable file length captured at reader open (issue #1586).
134        file_len: u64,
135    },
136    /// Share the underlying read-only memory map; each scan gets its own cursor
137    /// position over the same mapped bytes (just an `Arc` clone, no new mapping).
138    Mapped(Arc<Mmap>),
139    /// Reopen the file with direct I/O for each scan, giving it its own
140    /// page-cache-bypassing handle and aligned read-ahead window.
141    #[cfg(unix)]
142    Direct {
143        /// Read-ahead window (bytes) for each scan's [`DirectCursor`].
144        window: usize,
145        /// Immutable file length captured at reader open (issue #1586), used by
146        /// the buffered fallback when direct I/O is refused.
147        file_len: u64,
148    },
149}
150
151impl ScanSource {
152    /// Mint a fresh, independent [`BlockSource`] for one scan.
153    pub(crate) async fn open(&self, path: &Path) -> Result<BlockSource> {
154        // A5 read-work counters (FILE_OPENS; consumer C2): each arm below counts the
155        // exact number of open(2)s it performs — the Buffered arm and the Direct
156        // buffered-fallback each do one File::open, and the Direct cursor open(2) is
157        // counted inside `DirectCursor::open_direct`. The Mapped arm reuses the
158        // shared `Arc<Mmap>` (no open(2)), so it records nothing. No-op in release
159        // (design.md Decision 1/2). This is the per-scan open site C2 targets.
160        use crate::storage::sstable::read_work_counters::record_file_open;
161        Ok(match self {
162            ScanSource::Buffered { file_len } => {
163                record_file_open();
164                BlockSource::buffered_sized(File::open(path).await?, *file_len)
165            }
166            ScanSource::Mapped(mmap) => BlockSource::mapped(mmap.clone()),
167            #[cfg(unix)]
168            ScanSource::Direct { window, file_len } => {
169                // Reopen with O_DIRECT for this scan. If the platform/filesystem
170                // refuses direct I/O, degrade to buffered rather than failing the
171                // scan (mirrors the open()-time fallback).
172                match DirectCursor::open(path, *window) {
173                    Ok(cursor) => BlockSource::direct(cursor),
174                    Err(e) => {
175                        log::warn!(
176                            "Direct-I/O reopen of {} for scan failed ({}); using buffered I/O",
177                            path.display(),
178                            e
179                        );
180                        record_file_open();
181                        BlockSource::buffered_sized(File::open(path).await?, *file_len)
182                    }
183                }
184            }
185        })
186    }
187}

246/// An in-memory read cursor over a memory-mapped file.
247///
248/// Implements the same `AsyncRead`/`AsyncSeek` contract as a buffered file
249/// reader, but every operation completes immediately since the data is already
250/// resident (modulo lazy page faults handled transparently by the OS).
251pub(crate) struct MmapCursor {
252    mmap: Arc<Mmap>,
253    /// Current read position. May legally point past the end of the map, in
254    /// which case reads yield zero bytes (matching `File` EOF semantics).
255    pos: u64,
256}
257
258impl MmapCursor {
259    fn new(mmap: Arc<Mmap>) -> Self {
260        Self { mmap, pos: 0 }
261    }
262}
263
264impl AsyncRead for MmapCursor {
265    fn poll_read(
266        self: Pin<&mut Self>,
267        _cx: &mut Context<'_>,
268        buf: &mut ReadBuf<'_>,
269    ) -> Poll<io::Result<()>> {
270        let this = self.get_mut();
271        let data: &[u8] = &this.mmap;
272        let len = data.len() as u64;
273        // At or past EOF: read nothing and leave the position untouched. A real
274        // `File` preserves a seeked-past-EOF position across a zero-byte read,
275        // so we must not clamp `pos` back to `len` here (that divergence was the
276        // source-level parity bug). Returning zero bytes still drives
277        // `read_exact` to `UnexpectedEof`, which the block-header readers use to
278        // detect EOF.
279        if this.pos >= len {
280            return Poll::Ready(Ok(()));
281        }
282        let pos = this.pos as usize;
283        let remaining = &data[pos..];
284        let n = remaining.len().min(buf.remaining());
285        buf.put_slice(&remaining[..n]);
286        this.pos += n as u64;
287        Poll::Ready(Ok(()))
288    }
289}

291impl AsyncSeek for MmapCursor {
292    fn start_seek(self: Pin<&mut Self>, position: SeekFrom) -> io::Result<()> {
293        let this = self.get_mut();
294        let len = this.mmap.len() as u64;
295        let new_pos = match position {
296            SeekFrom::Start(offset) => offset,
297            SeekFrom::End(offset) => offset_from(len, offset)?,
298            SeekFrom::Current(offset) => offset_from(this.pos, offset)?,
299        };
300        this.pos = new_pos;
301        Ok(())
302    }
303
304    fn poll_complete(self: Pin<&mut Self>, _cx: &mut Context<'_>) -> Poll<io::Result<u64>> {
305        Poll::Ready(Ok(self.get_mut().pos))
306    }
307}

309/// Apply a signed offset to a base position, rejecting seeks before byte 0.
310///
311/// Seeking past the end is permitted (matching `std`/`tokio` `File` behaviour);
312/// subsequent reads simply return zero bytes.
313fn offset_from(base: u64, offset: i64) -> io::Result<u64> {
314    let result = if offset >= 0 {
315        // Overflowing the u64 address space is rejected rather than clamped, to
316        // mirror `std::fs::File`, which errors on a seek that overflows.
317        base.checked_add(offset as u64).ok_or_else(|| {
318            io::Error::new(
319                io::ErrorKind::InvalidInput,
320                "invalid seek to an overflowing position",
321            )
322        })?
323    } else {
324        base.checked_sub(offset.unsigned_abs()).ok_or_else(|| {
325            io::Error::new(
326                io::ErrorKind::InvalidInput,
327                "invalid seek to a negative position",
328            )
329        })?
330    };
331    Ok(result)
332}

334/// Open a file handle in cache-bypassing (direct-I/O) mode for the current
335/// platform. Shared by the scan-cursor [`DirectCursor`] and the point-read
336/// [`DirectReadAt`](super::read_at::DirectReadAt) so both honor the exact same
337/// per-platform primitive.
338///
339/// - Linux/Android: `O_DIRECT`.
340/// - macOS: `F_NOCACHE` (the per-fd equivalent; no `O_DIRECT`).
341/// - other Unix: unsupported → the caller falls back to buffered I/O.
342#[cfg(any(target_os = "linux", target_os = "android"))]
343pub(crate) fn open_direct_file(path: &Path) -> io::Result<std::fs::File> {
344    use std::os::unix::fs::OpenOptionsExt;
345    std::fs::OpenOptions::new()
346        .read(true)
347        .custom_flags(libc::O_DIRECT)
348        .open(path)
349}
350
351#[cfg(target_os = "macos")]
352pub(crate) fn open_direct_file(path: &Path) -> io::Result<std::fs::File> {
353    use std::os::unix::io::AsRawFd;
354    let file = std::fs::File::open(path)?;
355    // SAFETY: `fcntl` with `F_NOCACHE` on a freshly opened, valid fd.
356    let rc = unsafe { libc::fcntl(file.as_raw_fd(), libc::F_NOCACHE, 1) };
357    if rc == -1 {
358        return Err(io::Error::last_os_error());
359    }
360    Ok(file)
361}

374/// I/O alignment (bytes) for direct reads. `O_DIRECT` requires the file
375/// offset, transfer length, and user buffer to all be aligned to the logical
376/// block size of the underlying device. 4096 is a safe superset of the common
377/// 512/4096-byte sector sizes, so aligning to it satisfies every mainstream
378/// local filesystem.
379#[cfg(unix)]
380const DIRECT_IO_ALIGN: usize = 4096;
381
382/// A page-cache-bypassing read cursor over a file opened with direct I/O.
383///
384/// Direct I/O (`O_DIRECT` on Linux, `F_NOCACHE` on macOS) serves reads straight
385/// from the device without populating or evicting the OS page cache. This is
386/// the right backend for a single large scan (a Data.db file bigger than a
387/// configurable fraction of system RAM): it keeps that one-shot scan from
388/// thrashing the page cache and evicting everything else the host has warm.
389///
390/// Because `O_DIRECT` mandates aligned transfers, the cursor maintains its own
391/// aligned read-ahead buffer: each refill issues one aligned `pread` of up to
392/// `window` bytes, and `poll_read` copies the requested sub-range out of it.
393/// The window therefore doubles as the prefetch granularity.
394///
395/// Like [`MmapCursor`], the actual `pread` runs synchronously inside
396/// `poll_read`; direct reads are uncached disk I/O, so callers should keep this
397/// backend to the blocking-friendly contexts where the buffered/mmap backends
398/// already perform synchronous page faults.
399#[cfg(unix)]
400pub(crate) struct DirectCursor {
401    file: std::fs::File,
402    /// Total file length, used for EOF detection.
403    len: u64,
404    /// Logical read position (may sit past EOF; reads then yield zero bytes).
405    pos: u64,
406    /// Read-ahead window size (bytes), already rounded up to [`DIRECT_IO_ALIGN`].
407    window: usize,
408    /// Aligned scratch buffer holding the most recently read window.
409    buf: AlignedBuf,
410    /// File offset of `buf[0]` (always a multiple of [`DIRECT_IO_ALIGN`]).
411    buf_off: u64,
412    /// Number of valid bytes currently in `buf`.
413    buf_len: usize,
414}
415
416#[cfg(unix)]
417impl DirectCursor {
418    /// Open `path` with direct I/O and a `window`-byte aligned read-ahead buffer.
419    ///
420    /// `window` is rounded up to [`DIRECT_IO_ALIGN`] (and never less than one
421    /// alignment unit). Returns an error if the platform or filesystem does not
422    /// support direct I/O, so callers can fall back to buffered reads.
423    pub(crate) fn open(path: &Path, window: usize) -> io::Result<Self> {
424        let file = Self::open_direct(path)?;
425        // A5 read-work counter (FILE_OPENS; consumer C2): one per SUCCESSFUL
426        // direct-I/O open(2) (used by both the reader's cold-open direct path and
427        // per-scan direct reopens). Counted after `open_direct` succeeds so a failed
428        // direct open that falls back to a buffered open (which counts its own
429        // open(2)) is not double-counted. No-op in release (design.md Decision 1/2).
430        crate::storage::sstable::read_work_counters::record_file_open();
431        let len = file.metadata()?.len();
432        let align = DIRECT_IO_ALIGN;
433        // Round the window up to the alignment, but saturate instead of
434        // overflowing (a near-`usize::MAX` `direct_io_prefetch_bytes` would
435        // otherwise panic in debug / wrap to 0 in release). A huge window then
436        // simply fails the allocation below and the caller falls back to
437        // buffered I/O, rather than crashing.
438        let window = window
439            .max(align)
440            .checked_next_multiple_of(align)
441            .unwrap_or(usize::MAX & !(align - 1));
442        let buf = AlignedBuf::new(window, align)?;
443        Ok(Self {
444            file,
445            len,
446            pos: 0,
447            window,
448            buf,
449            buf_off: 0,
450            buf_len: 0,
451        })
452    }
453
454    /// Open a file handle in cache-bypassing mode for the current platform.
455    fn open_direct(path: &Path) -> io::Result<std::fs::File> {
456        open_direct_file(path)
457    }
458
459    /// Ensure the byte at `self.pos` is resident in `buf`, refilling with one
460    /// aligned `pread` if not.
461    fn ensure_buffered(&mut self) -> io::Result<()> {
462        let covered = self.buf_len > 0
463            && self.pos >= self.buf_off
464            && self.pos < self.buf_off + self.buf_len as u64;
465        if covered {
466            return Ok(());
467        }
468        let align = DIRECT_IO_ALIGN as u64;
469        let aligned_off = (self.pos / align) * align;
470        let slice = self.buf.as_mut_slice();
471        debug_assert_eq!(slice.len(), self.window);
472        // A single aligned pread. O_DIRECT returns the full window unless it
473        // straddles EOF, where it returns just the available tail (a legal
474        // short read). `read_at` does not advance the file's own offset, so
475        // concurrent cursors over reopened handles never interfere.
476        use std::os::unix::fs::FileExt;
477        let n = self.file.read_at(slice, aligned_off)?;
478        self.buf_off = aligned_off;
479        self.buf_len = n;
480        Ok(())
481    }
482}

484#[cfg(unix)]
485impl AsyncRead for DirectCursor {
486    fn poll_read(
487        self: Pin<&mut Self>,
488        _cx: &mut Context<'_>,
489        buf: &mut ReadBuf<'_>,
490    ) -> Poll<io::Result<()>> {
491        let this = self.get_mut();
492        // At or past EOF: yield zero bytes, leaving position untouched (matching
493        // `File`/`MmapCursor` semantics so `read_exact` reports `UnexpectedEof`).
494        if this.pos >= this.len {
495            return Poll::Ready(Ok(()));
496        }
497        if let Err(e) = this.ensure_buffered() {
498            return Poll::Ready(Err(e));
499        }
500        let rel = (this.pos - this.buf_off) as usize;
501        if rel >= this.buf_len {
502            // The aligned read landed entirely before `pos` (only possible at a
503            // short EOF read); nothing more to hand back.
504            return Poll::Ready(Ok(()));
505        }
506        let available = &this.buf.as_slice()[rel..this.buf_len];
507        let n = available.len().min(buf.remaining());
508        buf.put_slice(&available[..n]);
509        this.pos += n as u64;
510        Poll::Ready(Ok(()))
511    }
512}

531/// A heap allocation whose start address is aligned to a requested boundary,
532/// as required for `O_DIRECT` transfer buffers.
533#[cfg(unix)]
534struct AlignedBuf {
535    ptr: std::ptr::NonNull<u8>,
536    layout: std::alloc::Layout,
537}
538
539// SAFETY: `AlignedBuf` uniquely owns its allocation for its whole lifetime; the
540// raw pointer is never aliased or shared, so moving it across threads is sound.
541#[cfg(unix)]
542unsafe impl Send for AlignedBuf {}
543
544#[cfg(unix)]
545impl AlignedBuf {
546    fn new(size: usize, align: usize) -> io::Result<Self> {
547        let size = size.max(align);
548        let layout = std::alloc::Layout::from_size_align(size, align)
549            .map_err(|e| io::Error::new(io::ErrorKind::InvalidInput, e))?;
550        // SAFETY: `layout` has non-zero size (size >= align >= 1).
551        let raw = unsafe { std::alloc::alloc_zeroed(layout) };
552        let ptr = std::ptr::NonNull::new(raw)
553            .ok_or_else(|| io::Error::new(io::ErrorKind::OutOfMemory, "aligned alloc failed"))?;
554        Ok(Self { ptr, layout })
555    }
556
557    fn as_slice(&self) -> &[u8] {
558        // SAFETY: `ptr` is valid for `layout.size()` initialised bytes (zeroed
559        // on alloc, then overwritten by reads) and borrowed immutably here.
560        unsafe { std::slice::from_raw_parts(self.ptr.as_ptr(), self.layout.size()) }
561    }
562
563    fn as_mut_slice(&mut self) -> &mut [u8] {
564        // SAFETY: `ptr` is valid for `layout.size()` bytes and uniquely borrowed.
565        unsafe { std::slice::from_raw_parts_mut(self.ptr.as_ptr(), self.layout.size()) }
566    }
567}
568
569#[cfg(unix)]
570impl Drop for AlignedBuf {
571    fn drop(&mut self) {
572        // SAFETY: `ptr`/`layout` are exactly the values returned from the
573        // matching `alloc_zeroed`, freed exactly once here.
574        unsafe { std::alloc::dealloc(self.ptr.as_ptr(), self.layout) };
575    }

760    /// Direct-I/O cursor reads back exactly what was written, including across
761    /// the aligned read-ahead window boundary and at a non-aligned EOF. Skips
762    /// when the temp filesystem refuses `O_DIRECT` (common on tmpfs/overlayfs).
763    #[cfg(unix)]
764    #[tokio::test]
765    async fn direct_cursor_reads_and_seeks() {
766        use tokio::io::{AsyncReadExt, AsyncSeekExt};
767
768        // A size that is NOT a multiple of the alignment, spanning multiple
769        // windows, to exercise the partial-final-block path.
770        let len = DIRECT_IO_ALIGN * 3 + 123;
771        let mut data = vec![0u8; len];
772        for (i, b) in data.iter_mut().enumerate() {
773            *b = (i % 251) as u8;
774        }
775
776        let dir = std::env::temp_dir();
777        let path = dir.join(format!("cqlite_directcursor_{}.bin", std::process::id()));
778        tokio::fs::write(&path, &data).await.unwrap();
779
780        // Small window (2 alignment units) so a full read crosses windows.
781        let mut cursor = match DirectCursor::open(&path, DIRECT_IO_ALIGN * 2) {
782            Ok(c) => c,
783            Err(e) => {
784                eprintln!("O_DIRECT unsupported here ({e}); skipping direct cursor test");
785                tokio::fs::remove_file(&path).await.ok();
786                return;
787            }
788        };
789
790        // Sequential read of the whole file.
791        let mut got = Vec::with_capacity(len);
792        let mut chunk = [0u8; 1000];
793        loop {
794            let n = cursor.read(&mut chunk).await.unwrap();
795            if n == 0 {
796                break;
797            }
798            got.extend_from_slice(&chunk[..n]);
799        }
800        assert_eq!(got, data, "direct sequential read must match contents");
801
802        // Seek to a non-aligned offset and read a window-crossing span.
803        cursor.seek(SeekFrom::Start(4090)).await.unwrap();
804        let mut window = [0u8; 16];
805        cursor.read_exact(&mut window).await.unwrap();
806        for (k, b) in window.iter().enumerate() {
807            assert_eq!(*b, ((4090 + k) % 251) as u8, "byte {k} after seek");
808        }
809
810        // Reading past EOF reports UnexpectedEof.
811        cursor
812            .seek(SeekFrom::Start((len - 4) as u64))
813            .await
814            .unwrap();
815        let mut tail = [0u8; 8];
816        let err = cursor.read_exact(&mut tail).await.unwrap_err();
817        assert_eq!(err.kind(), io::ErrorKind::UnexpectedEof);
818
819        tokio::fs::remove_file(&path).await.ok();
820    }
