1//! Bulletproof chunk-based decompression for SSTable Data.db files
2//!
3//! This module implements the proper decompression strategy for Cassandra SSTable files
4//! using CompressionInfo.db metadata to decompress chunks on-demand.
5
6use super::compression_info::CompressionInfo;
7use crate::parser::header::CassandraVersion;
8use crate::{Error, Result};
9use rustc_hash::FxHashMap;
10use std::io::{Read, Seek, SeekFrom};
11
12/// Chunk-based decompressor for SSTable Data.db files
13pub struct ChunkDecompressor {
14    /// Compression metadata from CompressionInfo.db
15    compression_info: CompressionInfo,
16    /// Cache of decompressed chunks
17    chunk_cache: FxHashMap<usize, Vec<u8>>,
18    /// Maximum number of chunks to cache
19    max_cached_chunks: usize,
20    /// Data file path for error reporting
21    data_file_path: Option<String>,
22    /// Cached Data.db length, captured on the first chunk read (issue #1586).
23    ///
24    /// The file is an immutable SSTable, so its size is derived exactly once
25    /// with a single seek probe and then reused for every subsequent chunk's
26    /// `compressed_chunk_size` bounds math — instead of re-seeking to `End(0)`
27    /// (and back) on every chunk read.
28    cached_data_file_size: Option<u64>,
29    /// The reader's logical position after the most recent chunk read within the
30    /// current `read_data` / `decompress_chunk_by_index` call (issue #1586).
31    ///
32    /// Sequential chunk reads leave the reader positioned exactly at the next
33    /// chunk's offset, so the explicit `seek(Start(offset))` can be skipped when
34    /// it would be a no-op. Reset to `None` at each public entry point so it is
35    /// only ever trusted within a single call against a single reader.
36    stream_pos: Option<u64>,
37}
38
39impl ChunkDecompressor {
40    /// Create a new chunk decompressor with compression metadata.
41    ///
42    /// The `cassandra_version` parameter is accepted for API compatibility but is no longer
43    /// used: the NB format (all Cassandra 5.0 files) always has inline CRC32 in Data.db
44    /// regardless of version, which is handled deterministically in decompress_chunk().
45    pub fn new(
46        compression_info: CompressionInfo,
47        _cassandra_version: CassandraVersion,
48    ) -> Result<Self> {
49        compression_info.validate()?;
50
51        Ok(Self {
52            compression_info,
53            chunk_cache: FxHashMap::default(),
54            max_cached_chunks: 16, // Cache up to 16 chunks (16 * 16KB = 256KB max memory)
55            data_file_path: None,
56            cached_data_file_size: None,
57            stream_pos: None,
58        })
59    }

81    /// Read data from compressed SSTable at specified offset and length
82    pub fn read_data<R: Read + Seek>(
83        &mut self,
84        reader: &mut R,
85        offset: u64,
86        length: usize,
87    ) -> Result<Vec<u8>> {
88        let mut result = Vec::with_capacity(length);
89        let mut remaining = length;
90        let mut current_offset = offset;
91
92        // The reader's logical position is unknown at the start of a call (the
93        // caller may have passed a fresh reader), so the first chunk read always
94        // seeks; subsequent sequential reads within this call skip the redundant
95        // seek (issue #1586).
96        self.stream_pos = None;
97
98        while remaining > 0 {
99            // Determine which chunk contains this offset
100            let chunk_index = self.compression_info.chunk_for_offset(current_offset);
101            let offset_in_chunk = self.compression_info.offset_within_chunk(current_offset);
102
103            // Get the decompressed chunk
104            let chunk_data = self.get_decompressed_chunk(reader, chunk_index)?;
105
106            // Extract the requested data from this chunk
107            let chunk_start = offset_in_chunk as usize;
108            let chunk_end = std::cmp::min(chunk_start + remaining, chunk_data.len());
109
110            if chunk_start >= chunk_data.len() {
111                return Err(Error::InvalidFormat(format!(
112                    "Offset {} beyond chunk {} size {}",
113                    chunk_start,
114                    chunk_index,
115                    chunk_data.len()
116                )));
117            }
118
119            let chunk_slice = &chunk_data[chunk_start..chunk_end];
120            result.extend_from_slice(chunk_slice);
121
122            let bytes_read = chunk_slice.len();
123            remaining -= bytes_read;
124            current_offset += bytes_read as u64;
125        }
126
127        Ok(result)
128    }

130    /// Get a decompressed chunk, using cache if available
131    fn get_decompressed_chunk<R: Read + Seek>(
132        &mut self,
133        reader: &mut R,
134        chunk_index: usize,
135    ) -> Result<Vec<u8>> {
136        // Check cache first
137        if let Some(cached_chunk) = self.chunk_cache.get(&chunk_index) {
138            return Ok(cached_chunk.clone());
139        }
140
141        // Decompress the chunk
142        let chunk_data = self.decompress_chunk(reader, chunk_index)?;
143
144        // Add to cache (with LRU eviction if necessary)
145        if self.chunk_cache.len() >= self.max_cached_chunks {
146            // Simple eviction: remove first entry
147            if let Some(first_key) = self.chunk_cache.keys().next().copied() {
148                self.chunk_cache.remove(&first_key);
149            }
150        }
151
152        self.chunk_cache.insert(chunk_index, chunk_data.clone());
153        Ok(chunk_data)
154    }
155
156    /// Exact uncompressed size Cassandra wrote for chunk `chunk_index`, derived from
157    /// `data_length` and `chunk_length`.
158    ///
159    /// Cassandra lays chunks out so that random access works via
160    /// `position / chunk_length`: every chunk that holds data up to the final byte is
161    /// exactly `chunk_length`, the chunk containing the final byte is the partial
162    /// remainder, and any chunk whose start is at/after `data_length` is empty (some
163    /// flush/close paths append a degenerate trailing chunk — e.g. `system/local`:
164    /// data_length 708, chunk_length 16384, chunk 0 = 708 bytes, chunk 1 = 0). This is
165    /// the precise per-chunk invariant: stricter than `<= chunk_length` (it still
166    /// rejects a short chunk that has addressable data after it — corruption that would
167    /// also break Cassandra's own position→chunk mapping), but not the over-strict
168    /// `== chunk_length` that rejected valid partial/trailing chunks.
169    fn expected_decompressed_len(&self, chunk_index: usize) -> u64 {
170        let chunk_length = self.compression_info.chunk_length as u64;
171        let start = (chunk_index as u64).saturating_mul(chunk_length);
172        self.compression_info
173            .data_length
174            .saturating_sub(start)
175            .min(chunk_length)
176    }

178    /// Decompress a specific chunk from the compressed data file
179    fn decompress_chunk<R: Read + Seek>(
180        &mut self,
181        reader: &mut R,
182        chunk_index: usize,
183    ) -> Result<Vec<u8>> {
184        // Get compressed chunk offset
185        let compressed_offset = self
186            .compression_info
187            .compressed_chunk_offset(chunk_index)
188            .ok_or_else(|| Error::InvalidFormat(format!("No offset for chunk {}", chunk_index)))?;
189
190        // Determine record size (compressed payload + 4-byte inline CRC) using the
191        // Data.db length. The SSTable is immutable, so its size is derived once
192        // with a single seek probe and cached — never re-probed per chunk (#1586).
193        let file_size = match self.cached_data_file_size {
194            Some(size) => size,
195            None => {
196                let current_pos = reader.stream_position().map_err(Error::Io)?;
197                let size = reader.seek(SeekFrom::End(0)).map_err(Error::Io)?;
198                reader
199                    .seek(SeekFrom::Start(current_pos))
200                    .map_err(Error::Io)?;
201                self.cached_data_file_size = Some(size);
202                // The probe restored the reader to `current_pos`.
203                self.stream_pos = Some(current_pos);
204                size
205            }
206        };
207
208        // compressed_chunk_size returns the full record delta including the 4-byte inline CRC.
209        // CompressedSequentialWriter.java:203: chunkOffset += compressedLength + 4
210        let record_size = self
211            .compression_info
212            .compressed_chunk_size(chunk_index, file_size)
213            .ok_or_else(|| {
214                Error::InvalidFormat(format!("Cannot determine size for chunk {}", chunk_index))
215            })?;
216
217        // Bug #639 fix: subtract the 4-byte inline CRC from the delta.
218        // The old code passed all (delta) bytes to the decompressor, which included the
219        // trailing CRC and caused decompression failures on well-formed chunks.
220        if record_size < 4 {
221            return Err(Error::InvalidFormat(format!(
222                "Chunk {} record size {} is too small (minimum 4 bytes for inline CRC)",
223                chunk_index, record_size
224            )));
225        }
226        let compressed_len = (record_size - 4) as usize;
227
228        // Seek to compressed chunk offset and read compressed payload only.
229        // Skip the seek when the previous sequential chunk read already left the
230        // reader positioned exactly here (issue #1586 step 3).
231        if self.stream_pos != Some(compressed_offset) {
232            reader
233                .seek(SeekFrom::Start(compressed_offset))
234                .map_err(Error::Io)?;
235        }
236        // Position is unknown until the record's bytes are fully read.
237        self.stream_pos = None;
238
239        let mut compressed_data = vec![0u8; compressed_len];
240        reader.read_exact(&mut compressed_data).map_err(Error::Io)?;
241
242        // Read the 4-byte inline CRC32 (big-endian) and validate it over the compressed bytes.
243        // Authority: CompressedSequentialWriter.java:192 + read path lines 275-282.
244        let mut crc_bytes = [0u8; 4];
245        reader.read_exact(&mut crc_bytes).map_err(Error::Io)?;
246        // The record (payload + 4-byte CRC) has been consumed; the reader now sits
247        // at the next chunk's offset, so a subsequent sequential read can skip its
248        // seek (issue #1586).
249        self.stream_pos = Some(compressed_offset.saturating_add(record_size));
250        let stored_crc = u32::from_be_bytes(crc_bytes);
251        let computed_crc = crc32fast::hash(&compressed_data);
252        if stored_crc != computed_crc {
253            let file_info = match &self.data_file_path {
254                Some(path) => format!(" in file {}", path),
255                None => String::new(),
256            };
257            return Err(Error::InvalidFormat(format!(
258                "CRC32 mismatch for chunk {} at offset 0x{:x}{}: stored=0x{:08x}, computed=0x{:08x}, compressed_len={}",
259                chunk_index, compressed_offset, file_info, stored_crc, computed_crc, compressed_len
260            )));
261        }
262
263        log::debug!(
264            "Reading chunk {} at offset {} ({} bytes compressed, CRC OK)",
265            chunk_index,
266            compressed_offset,
267            compressed_len
268        );
269
270        // Incompressible-chunk fallback (Bug #639):
271        // When compressedLength >= maxCompressedLength, Cassandra stored the chunk uncompressed.
272        // CompressedSequentialWriter.java:160-177: if compressedLen >= maxCompressedLen, use raw buffer.
273        let max_compressed_length = self.compression_info.max_compressed_length as usize;
274        if compressed_len >= max_compressed_length {
275            log::debug!(
276                "Chunk {} is incompressible (compressed_len={} >= max_compressed_length={}), returning raw bytes",
277                chunk_index, compressed_len, max_compressed_length
278            );
279            // A raw chunk's stored bytes ARE the uncompressed bytes, so they must match
280            // the exact size Cassandra wrote for this chunk index — the same invariant
281            // the compressed path checks below (see `expected_decompressed_len`).
282            let expected_size = self.expected_decompressed_len(chunk_index);
283            if compressed_data.len() as u64 != expected_size {
284                return Err(Error::InvalidFormat(format!(
285                    "Raw (incompressible) chunk {} size {} != expected {} (data_length {}, chunk_length {}) — corrupt or misdecoded",
286                    chunk_index,
287                    compressed_data.len(),
288                    expected_size,
289                    self.compression_info.data_length,
290                    self.compression_info.chunk_length,
291                )));
292            }
293            return Ok(compressed_data);
294        }
295
296        // Decompress based on algorithm
297        let decompressed = match self.compression_info.algorithm.as_str() {
298            "LZ4Compressor" => self.decompress_lz4_chunk(&compressed_data, chunk_index),
299            "SnappyCompressor" => self.decompress_snappy_chunk(&compressed_data, chunk_index),
300            "DeflateCompressor" => self.decompress_deflate_chunk(&compressed_data, chunk_index),
301            "ZstdCompressor" => self.decompress_zstd_chunk(&compressed_data, chunk_index),
302            algorithm => Err(Error::UnsupportedFormat(format!(
303                "Unknown compression algorithm: {}",
304                algorithm
305            ))),
306        }?;
307
308        // Validate the decompressed chunk size against the exact size Cassandra wrote
309        // for this chunk index (see `expected_decompressed_len`). This handles full
310        // chunks, the partial final-data chunk, and degenerate empty trailing chunks,
311        // while still rejecting a short non-final chunk that has addressable data after
312        // it (corruption). The inline CRC above is the primary integrity guard.
313        let expected_size = self.expected_decompressed_len(chunk_index);
314        if decompressed.len() as u64 != expected_size {
315            return Err(Error::InvalidFormat(format!(
316                "Decompressed chunk {} size {} != expected {} (data_length {}, chunk_length {}) — corrupt or misdecoded",
317                chunk_index,
318                decompressed.len(),
319                expected_size,
320                self.compression_info.data_length,
321                self.compression_info.chunk_length,
322            )));
323        }
324
325        Ok(decompressed)
326    }

328    /// Decompress LZ4 chunk - Cassandra uses 4-byte little-endian length prefix
329    /// Upper bound on a single decompressed chunk: the configured uncompressed
330    /// chunk length (the final chunk is shorter, never larger). Used to bound the
331    /// streaming Deflate/Zstd decoders against decompression bombs. Falls back to
332    /// the 128MB global cap if chunk_length is unset/zero.
333    ///
334    /// Only the Snappy/Deflate/Zstd paths consume this bound, so the method is
335    /// gated to match its callers and stay dead-code-free under minimal builds.
336    #[cfg(any(feature = "snappy", feature = "deflate", feature = "zstd"))]
337    fn chunk_size_guard(&self) -> u64 {
338        match self.compression_info.chunk_length {
339            0 => 128 * 1024 * 1024,
340            n => n as u64,
341        }
342    }

344    fn decompress_lz4_chunk(&self, compressed_data: &[u8], chunk_index: usize) -> Result<Vec<u8>> {
345        let file_info = match &self.data_file_path {
346            Some(path) => format!(" in file {}", path),
347            None => String::new(),
348        };
349
350        if compressed_data.len() < 4 {
351            return Err(Error::InvalidFormat(format!(
352                "LZ4 compressed data too short for chunk {}{} (need at least 4 bytes for length prefix, got {})",
353                chunk_index, file_info, compressed_data.len()
354            )));
355        }
356
357        // CRITICAL: Cassandra's LZ4Compressor prepends a 4-byte little-endian length prefix
358        // See: org.apache.cassandra.io.compress.LZ4Compressor.decompress() lines 169-172
359        // This is NOT the lz4_flex size-prepended format (which uses varint encoding)
360        let decompressed_length = u32::from_le_bytes([
361            compressed_data[0],
362            compressed_data[1],
363            compressed_data[2],
364            compressed_data[3],
365        ]) as usize;
366
367        // Validate the LZ4 length prefix against the exact size Cassandra wrote for
368        // this chunk index (full / partial-final / empty-trailing — see
369        // `expected_decompressed_len`). Fails fast on a misdecoded prefix before
370        // attempting decompression.
371        let expected_size = self.expected_decompressed_len(chunk_index) as usize;
372        if decompressed_length != expected_size {
373            return Err(Error::InvalidFormat(format!(
374                "LZ4 length prefix mismatch for chunk {} at offset 0x{:x}: expected {}, got {} (first 4 bytes: {:02x} {:02x} {:02x} {:02x}){}",
375                chunk_index,
376                self.compression_info
377                    .chunk_offsets
378                    .get(chunk_index)
379                    .unwrap_or(&0),
380                expected_size,
381                decompressed_length,
382                compressed_data[0],
383                compressed_data[1],
384                compressed_data[2],
385                compressed_data[3],
386                file_info
387            )));
388        }
389
390        // Skip the 4-byte length prefix and decompress the actual LZ4 data
391        let lz4_data = &compressed_data[4..];
392
393        match lz4_flex::decompress(lz4_data, decompressed_length) {
394            Ok(decompressed) => {
395                if decompressed.len() != decompressed_length {
396                    return Err(Error::InvalidFormat(format!(
397                        "LZ4 decompression size mismatch for chunk {} at offset 0x{:x}: expected {}, got {}{}",
398                        chunk_index,
399                        self.compression_info
400                            .chunk_offsets
401                            .get(chunk_index)
402                            .unwrap_or(&0),
403                        decompressed_length,
404                        decompressed.len(),
405                        file_info
406                    )));
407                }
408                Ok(decompressed)
409            }
410            Err(e) => Err(Error::InvalidFormat(format!(
411                "LZ4 decompression failed for chunk {} at offset 0x{:x}: {}{}",
412                chunk_index,
413                self.compression_info
414                    .chunk_offsets
415                    .get(chunk_index)
416                    .unwrap_or(&0),
417                e,
418                file_info
419            ))),
420        }
421    }

423    /// Decompress Snappy chunk - strict mode for modern formats
424    fn decompress_snappy_chunk(
425        &self,
426        compressed_data: &[u8],
427        chunk_index: usize,
428    ) -> Result<Vec<u8>> {
429        #[cfg(feature = "snappy")]
430        {
431            use snap::raw::Decoder;
432
433            // Decompression-bomb guard (issue #1588): a raw Snappy block advertises
434            // its decompressed length as a leading varint, and `decompress_vec`
435            // pre-allocates that size BEFORE decoding. Reject an over-limit advertised
436            // length up front — a chunk decodes to at most `chunk_length` bytes — so a
437            // crafted block cannot force a huge allocation (mirrors the Deflate/Zstd
438            // chunk guards, which bound by `chunk_size_guard()`).
439            let max_chunk = self.chunk_size_guard();
440            let advertised = snap::raw::decompress_len(compressed_data).map_err(|e| {
441                Error::InvalidFormat(format!(
442                    "Snappy length decode failed for chunk {}: {}",
443                    chunk_index, e
444                ))
445            })?;
446            if advertised as u64 > max_chunk {
447                return Err(Error::InvalidFormat(format!(
448                    "Snappy chunk {} advertises {} bytes, beyond chunk_length {} (decompression-bomb guard)",
449                    chunk_index, advertised, max_chunk
450                )));
451            }
452
453            let mut decoder = Decoder::new();
454            match decoder.decompress_vec(compressed_data) {
455                Ok(decompressed) => Ok(decompressed),
456                Err(e) => Err(Error::InvalidFormat(format!(
457                    "Snappy decompression failed for chunk {} at offset 0x{:x}: {}. No fallback allowed for modern formats.",
458                    chunk_index,
459                    self.compression_info
460                        .chunk_offsets
461                        .get(chunk_index)
462                        .unwrap_or(&0),
463                    e
464                ))),
465            }
466        }
467
468        #[cfg(not(feature = "snappy"))]
469        {
470            let _ = (compressed_data, chunk_index); // Suppress unused warnings
471            Err(Error::UnsupportedFormat(
472                "Snappy support not compiled in".to_string(),
473            ))
474        }
475    }

477    /// Decompress Deflate chunk - strict mode for modern formats
478    fn decompress_deflate_chunk(
479        &self,
480        compressed_data: &[u8],
481        chunk_index: usize,
482    ) -> Result<Vec<u8>> {
483        #[cfg(feature = "deflate")]
484        {
485            // Cassandra's DeflateCompressor uses java.util.zip.Deflater/Inflater,
486            // which emit zlib-wrapped streams (2-byte header 0x78 0x9c + Adler-32
487            // trailer), NOT raw DEFLATE. Decode with ZlibDecoder to match. (#1082)
488            use flate2::read::ZlibDecoder;
489            use std::io::Read;
490
491            // Decompression-bomb guard: a chunk decompresses to at most
492            // `chunk_length` bytes (the final chunk is shorter). Cap the reader at
493            // chunk_length + 1 so a crafted zlib stream cannot expand into an
494            // unbounded Vec before we validate the size (roborev).
495            let max_chunk = self.chunk_size_guard();
496            let mut decoder = ZlibDecoder::new(compressed_data).take(max_chunk + 1);
497            let mut decompressed = Vec::new();
498
499            match decoder.read_to_end(&mut decompressed) {
500                Ok(_) if decompressed.len() as u64 > max_chunk => Err(Error::InvalidFormat(format!(
501                    "Deflate chunk {} expands beyond chunk_length {} (decompression-bomb guard)",
502                    chunk_index, max_chunk
503                ))),
504                Ok(_) => Ok(decompressed),
505                Err(e) => Err(Error::InvalidFormat(format!(
506                    "Deflate decompression failed for chunk {} at offset 0x{:x}: {}. No fallback allowed for modern formats.",
507                    chunk_index,
508                    self.compression_info
509                        .chunk_offsets
510                        .get(chunk_index)
511                        .unwrap_or(&0),
512                    e
513                ))),
514            }
515        }
516
517        #[cfg(not(feature = "deflate"))]
518        {
519            let _ = (compressed_data, chunk_index); // Suppress unused warnings
520            Err(Error::UnsupportedFormat(
521                "Deflate support not compiled in".to_string(),
522            ))
523        }
524    }

526    /// Decompress Zstd chunk - strict mode for modern formats
527    fn decompress_zstd_chunk(&self, compressed_data: &[u8], chunk_index: usize) -> Result<Vec<u8>> {
528        #[cfg(feature = "zstd")]
529        {
530            use std::io::Read;
531            use zstd::stream::read::Decoder as ZstdDecoder;
532
533            // Decompression-bomb guard: stream through a reader capped at
534            // chunk_length + 1 rather than `decode_all`, which would pre-allocate
535            // whatever content size the frame declares (roborev).
536            let max_chunk = self.chunk_size_guard();
537            let mut decoder = match ZstdDecoder::new(compressed_data) {
538                Ok(d) => d.take(max_chunk + 1),
539                Err(e) => {
540                    return Err(Error::InvalidFormat(format!(
541                        "Zstd decoder init failed for chunk {}: {}",
542                        chunk_index, e
543                    )))
544                }
545            };
546            let mut decompressed = Vec::new();
547            match decoder.read_to_end(&mut decompressed) {
548                Ok(_) if decompressed.len() as u64 > max_chunk => Err(Error::InvalidFormat(format!(
549                    "Zstd chunk {} expands beyond chunk_length {} (decompression-bomb guard)",
550                    chunk_index, max_chunk
551                ))),
552                Ok(_) => Ok(decompressed),
553                Err(e) => Err(Error::InvalidFormat(format!(
554                    "Zstd decompression failed for chunk {} at offset 0x{:x}: {}. No fallback allowed for modern formats.",
555                    chunk_index,
556                    self.compression_info
557                        .chunk_offsets
558                        .get(chunk_index)
559                        .unwrap_or(&0),
560                    e
561                ))),
562            }
563        }
564
565        #[cfg(not(feature = "zstd"))]
566        {
567            let _ = (compressed_data, chunk_index); // Suppress unused warnings
568            Err(Error::UnsupportedFormat(
569                "Zstd support not compiled in".to_string(),
570            ))
571        }
572    }

594    /// Number of compressed chunk records (the CompressionInfo.db offset-table length).
595    pub fn chunk_count(&self) -> usize {
596        self.compression_info.chunk_offsets.len()
597    }
598
599    /// Decompress a single chunk by index, validating its inline CRC32 trailer.
600    ///
601    /// Exposed for strict per-chunk parity tests that must exercise every chunk record
602    /// — including degenerate trailing chunks that `read_all_data` (bounded by
603    /// `data_length`) never reads.
604    pub fn decompress_chunk_by_index<R: Read + Seek>(
605        &mut self,
606        reader: &mut R,
607        chunk_index: usize,
608    ) -> Result<Vec<u8>> {
609        // Single-chunk entry point: the reader position is unknown here too, so
610        // force the first (only) read to seek (issue #1586).
611        self.stream_pos = None;
612        self.get_decompressed_chunk(reader, chunk_index)
613    }
614
615    /// Get compression info
616    pub fn compression_info(&self) -> &CompressionInfo {
617        &self.compression_info
618    }
619}

621/// Utility function to create a chunk decompressor from CompressionInfo.db file
622pub fn create_decompressor_from_file(
623    compression_info_path: &std::path::Path,
624) -> Result<ChunkDecompressor> {
625    let compression_data = std::fs::read(compression_info_path).map_err(Error::Io)?;
626
627    // Parse CompressionInfo using the deterministic Cassandra format.
628    // (Bug #638: old heuristic parse_alternative_format violated the no-heuristics mandate
629    // and has been removed.  The standard parse() is authoritative for all supported files.)
630    let compression_info = CompressionInfo::parse(&compression_data)?;
631
632    log::info!("Loaded compression info:");
633    log::info!("   Algorithm: {}", compression_info.algorithm);
634    log::info!("   Chunk Length: {} bytes", compression_info.chunk_length);
635    log::info!("   Data Length: {} bytes", compression_info.data_length);
636    log::info!("   Chunk Count: {}", compression_info.chunk_offsets.len());
637
638    ChunkDecompressor::new(compression_info, CassandraVersion::V5_0Release)
639}

686    /// A `Read + Seek` wrapper that counts every `seek()` call (which includes
687    /// `stream_position()`, since the default `Seek::stream_position` is
688    /// `seek(SeekFrom::Current(0))`). Used to prove the chunk read path does not
689    /// re-derive the immutable file size with a seek probe on every chunk
690    /// (issue #1586).
691    struct SeekCountingReader {
692        inner: std::io::Cursor<Vec<u8>>,
693        seeks: std::rc::Rc<std::cell::Cell<usize>>,
694    }
695
696    impl std::io::Read for SeekCountingReader {
697        fn read(&mut self, buf: &mut [u8]) -> std::io::Result<usize> {
698            self.inner.read(buf)
699        }
700    }
701
702    impl Seek for SeekCountingReader {
703        fn seek(&mut self, pos: SeekFrom) -> std::io::Result<u64> {
704            self.seeks.set(self.seeks.get() + 1);
705            self.inner.seek(pos)
706        }
707    }

709    /// Build a synthetic `k`-chunk LZ4 Data.db body plus its `CompressionInfo`.
710    /// Each chunk record is `[LE u32 uncompressed_len][lz4 bytes][BE u32 CRC32]`
711    /// exactly as Cassandra's `CompressedSequentialWriter` lays them out.
712    #[cfg(feature = "lz4")]
713    fn build_multichunk_lz4(
714        k: usize,
715        chunk_len: usize,
716        last_len: usize,
717    ) -> (Vec<u8>, CompressionInfo) {
718        let mut data_db: Vec<u8> = Vec::new();
719        let mut offsets: Vec<u64> = Vec::new();
720        let mut data_length: u64 = 0;
721        for i in 0..k {
722            offsets.push(data_db.len() as u64);
723            let unclen = if i + 1 == k { last_len } else { chunk_len };
724            let uncompressed: Vec<u8> = (0..unclen).map(|j| ((i * 7 + j) % 251) as u8).collect();
725            data_length += unclen as u64;
726            let mut payload = (unclen as u32).to_le_bytes().to_vec();
727            payload.extend_from_slice(&lz4_flex::compress(&uncompressed));
728            let crc = crc32fast::hash(&payload);
729            data_db.extend_from_slice(&payload);
730            data_db.extend_from_slice(&crc.to_be_bytes());
731        }
732        let info = CompressionInfo {
733            algorithm: "LZ4Compressor".to_string(),
734            chunk_length: chunk_len as u32,
735            data_length,
736            chunk_offsets: offsets,
737            option_pairs: vec![],
738            max_compressed_length: i32::MAX as u32,
739        };
740        (data_db, info)
741    }

743    /// Issue #1586: reading every chunk of a K-chunk file must NOT re-probe the
744    /// immutable file size with a `seek(End)`/`seek(Start)` dance on each chunk.
745    /// A full `read_all_data` scan must issue `<= K + O(1)` seeks, not the ~4*K
746    /// seeks the per-chunk size probe produced. On unfixed code this asserts
747    /// ~4*K and FAILS; after threading the cached file size it passes.
748    #[cfg(feature = "lz4")]
749    #[test]
750    fn read_all_data_does_not_reprobe_file_size_per_chunk() {
751        let k = 8usize;
752        let (data_db, info) = build_multichunk_lz4(k, 200, 50);
753        let mut decompressor = ChunkDecompressor::new(info, CassandraVersion::V5_0Release).unwrap();
754
755        let seeks = std::rc::Rc::new(std::cell::Cell::new(0usize));
756        let mut reader = SeekCountingReader {
757            inner: std::io::Cursor::new(data_db),
758            seeks: seeks.clone(),
759        };
760
761        let out = decompressor
762            .read_all_data(&mut reader)
763            .expect("read_all_data");
764        assert_eq!(out.len(), 7 * 200 + 50, "all decompressed bytes returned");
765
766        // Budget: one initial size probe (a small constant) plus at most one
767        // positioning seek per chunk. The per-chunk file-size re-probe (3 extra
768        // seeks/chunk) must be gone.
769        let total = seeks.get();
770        let budget = k + 4;
771        assert!(
772            total <= budget,
773            "expected <= {budget} seeks for {k} chunks (K + O(1)), got {total} \
774             — per-chunk file-size re-probe not eliminated (issue #1586)"
775        );
776    }
