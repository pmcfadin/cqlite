1//! Crash-durability barrier for the flush -> WAL-truncate handoff (issue #1392).
2//!
3//! When a memtable flush finishes, the same data lives in two places: the newly
4//! written SSTable components on disk and the still-intact WAL. Truncating the
5//! WAL discards the WAL copy, so it is only safe *after* the SSTable is truly
6//! durable.
7//!
8//! Fsyncing each component file (Data.db, Filter.db, ...) persists the file
9//! *contents*, but on POSIX filesystems it does **not** persist the parent
10//! directory's entries. After a power loss the freshly created dirents can be
11//! missing even though the file contents were flushed — leaving the data in
12//! neither a reachable SSTable nor the (already truncated) WAL.
13//!
14//! This module closes that gap: on every WAL-truncating flush it fsyncs the
15//! **full ancestor chain** from the SSTable's leaf directory up to (and
16//! including) the configured data root, deepest-first, *after* the component
17//! files are synced and *before* the WAL is truncated. It also makes the
18//! truncate-failure path explicit (leave the WAL intact as a durable replay
19//! marker) rather than silently swallowing the error.
20//!
21//! Syncing the whole leaf→data-root chain unconditionally — rather than only
22//! the directories a *given* flush attempt happened to create — closes a
23//! retry/partial-failure hazard (issue #1392): if an earlier attempt created
24//! `data_dir/ks/table` but crashed before its full directory-sync chain
25//! completed, the retry sees those dirs already present, so a
26//! "sync-only-what-I-created" barrier would sync just the leaf and truncate the
27//! WAL while the keyspace/table dirents were still not durable. Directory fsync
28//! is cheap and idempotent, so re-syncing an already-durable dir is harmless
29//! and removes the hazard entirely.
30//!
31//! The [`DurabilityBarrier`] trait is a test seam: production uses
32//! [`RealDurabilityBarrier`], while tests inject fakes that record the ordering
33//! of filesystem operations or fault a specific step.
34

39/// Outcome of a successful [`finalize_flush_durability`] call.
40///
41/// In every variant the SSTable has already been fully written and published
42/// (its components' contents are fsynced and its directory entries persisted).
43/// The variant tells the caller how to reconcile flush state (issue #1392):
44///
45/// * [`FlushDurabilityOutcome::Durable`] — the WAL was truncated, or was left
46///   intact but is still a valid replay marker. The flush is clean; the caller
47///   commits state (clears the memtable, advances the generation) and returns
48///   success.
49/// * [`FlushDurabilityOutcome::WalTruncateFailedAfterCommit`] — the WAL truncate
50///   failed *after* `set_len(0)` had already zeroed the WAL. The WAL is no
51///   longer replayable, so the published SSTable is the **only** durable copy
52///   of the flushed mutations. The caller MUST still commit flush state (clear
53///   the memtable and advance the generation) so a retry writes a NEW
54///   generation instead of overwriting the published SSTable, and then surface
55///   the carried error. Returning the error *without* committing state would
56///   let a retry reuse this generation and `File::create` would overwrite the
57///   sole durable copy — a crash mid-retry would then lose the data entirely.
58#[derive(Debug)]
59pub(crate) enum FlushDurabilityOutcome {
60    /// The handoff completed cleanly; the WAL is truncated or still replayable.
61    Durable,
62    /// The SSTable is durably published but the WAL truncate failed after
63    /// zeroing the WAL. State MUST be committed before the error is surfaced.
64    WalTruncateFailedAfterCommit(Error),
65}
66
67/// Seam over the two durability-critical operations performed at the end of a
68/// flush. Splitting them behind a trait lets tests assert their ordering and
69/// inject faults without touching the real filesystem.
70pub(crate) trait DurabilityBarrier {
71    /// Fsync `dir` so that directory entries created during the flush (the new
72    /// SSTable's component files) are persisted.
73    fn sync_dir(&self, dir: &Path) -> Result<()>;
74
75    /// Truncate the WAL now that the SSTable is durable.
76    ///
77    /// Returns [`TruncateError`] on failure so the caller can distinguish a
78    /// failure that leaves the WAL intact ([`TruncateError::BeforeMutation`])
79    /// from one that has already zeroed it
80    /// ([`TruncateError::AfterMutation`], issue #1392).
81    fn truncate_wal(&self, wal: &mut WriteAheadLog) -> std::result::Result<(), TruncateError>;
82}

84/// Production barrier: performs real directory fsyncs and WAL truncation.
85#[derive(Debug, Default, Clone, Copy)]
86pub(crate) struct RealDurabilityBarrier;
87
88impl DurabilityBarrier for RealDurabilityBarrier {
89    fn sync_dir(&self, dir: &Path) -> Result<()> {
90        sync_directory(dir)
91    }
92
93    fn truncate_wal(&self, wal: &mut WriteAheadLog) -> std::result::Result<(), TruncateError> {
94        wal.truncate_checked()
95    }
96}

98/// Make a directory's entries durable.
99///
100/// On Unix this opens the directory and calls `fsync` on the resulting
101/// descriptor, which is the portable POSIX way to persist newly created
102/// dirents. On non-Unix targets directory fsync is not available the same way;
103/// the per-file fsyncs performed by the component writers remain the durability
104/// guarantee there, so this is a documented no-op.
105#[cfg(unix)]
106fn sync_directory(dir: &Path) -> Result<()> {
107    let handle = std::fs::File::open(dir).map_err(|e| {
108        Error::Storage(format!(
109            "Failed to open data directory {} for fsync: {e}",
110            dir.display()
111        ))
112    })?;
113    handle.sync_all().map_err(|e| {
114        Error::Storage(format!(
115            "Failed to fsync data directory {}: {e}",
116            dir.display()
117        ))
118    })
119}
120
121#[cfg(not(unix))]
122fn sync_directory(_dir: &Path) -> Result<()> {
123    Ok(())
124}

126/// Like [`std::fs::create_dir_all`], but maps failures to [`Error::Storage`]
127/// and treats a raced/pre-existing directory as success (issue #1392).
128///
129/// Directory *creation* no longer needs to record which ancestors it
130/// materialized: the flush durability barrier fsyncs the entire leaf→data-root
131/// chain unconditionally (see [`dirs_to_sync`]), so there is nothing to track.
132pub(crate) fn create_dir_all(path: &Path) -> Result<()> {
133    if path.as_os_str().is_empty() {
134        return Ok(());
135    }
136    match std::fs::create_dir(path) {
137        Ok(()) => Ok(()),
138        // Raced/pre-existing: another actor already made it, or it existed.
139        Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists && path.is_dir() => Ok(()),
140        // Parent missing: create ancestors first, then retry.
141        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
142            match path.parent() {
143                Some(parent) if !parent.as_os_str().is_empty() => {
144                    create_dir_all(parent)?;
145                }
146                _ => {
147                    return Err(Error::Storage(format!(
148                        "Failed to create directory {}: {e}",
149                        path.display()
150                    )));
151                }
152            }
153            match std::fs::create_dir(path) {
154                Ok(()) => Ok(()),
155                Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists && path.is_dir() => Ok(()),
156                Err(e) => Err(Error::Storage(format!(
157                    "Failed to create directory {}: {e}",
158                    path.display()
159                ))),
160            }
161        }
162        Err(e) => Err(Error::Storage(format!(
163            "Failed to create directory {}: {e}",
164            path.display()
165        ))),
166    }
167}

169/// Compute the ordered list of directories to fsync so that every dirent along
170/// the newly written SSTable's path is persisted before the WAL is truncated
171/// (issue #1392).
172///
173/// Walks from the leaf SSTable directory up to (and including) `data_root`,
174/// deepest-first. This is the whole ancestor chain that could hold a
175/// not-yet-durable dirent for the SSTable — the leaf (its component dirents),
176/// the table directory (the leaf's own entry), the keyspace directory (the
177/// table's entry), and the data root (the keyspace's entry).
178///
179/// Syncing the FULL chain on every flush — regardless of which dirs this
180/// particular attempt created — is deliberate (issue #1392): a retry after a
181/// partially completed earlier attempt sees the ancestors already present, so a
182/// "sync-only-what-I-created" barrier would skip them and truncate the WAL
183/// while they were still not durable. Directory fsync is idempotent, so
184/// re-syncing an already-durable dir is harmless.
185///
186/// The result is ordered deepest-first so each child directory is synced before
187/// its parent — matching the POSIX convention that a parent's fsync only needs
188/// to follow the creation (and here the sync) of the child it references. The
189/// walk never ascends above `data_root`: it stops as soon as it reaches it, and
190/// defensively stops if `leaf` is not under `data_root` at all.
191fn dirs_to_sync(leaf: &Path, data_root: &Path) -> Vec<PathBuf> {
192    let mut dirs: Vec<PathBuf> = Vec::new();
193    let mut cur = leaf;
194    loop {
195        dirs.push(cur.to_path_buf());
196        if cur == data_root {
197            break;
198        }
199        match cur.parent() {
200            // Only ascend while we stay within the configured data root; never
201            // fsync directories above it.
202            Some(parent) if !parent.as_os_str().is_empty() && parent.starts_with(data_root) => {
203                cur = parent;
204            }
205            _ => break,
206        }
207    }
208    dirs
209}

211/// Complete the durability handoff at the end of a flush.
212///
213/// `data_path` is the SSTable's Data.db path; its parent directory is the leaf
214/// SSTable directory (falling back to `data_dir` if `data_path` has no parent).
215/// `data_dir` is the write engine's configured data root. Ordering is
216/// load-bearing (issue #1392):
217///
218/// 1. fsync the **full ancestor chain** from the leaf SSTable directory up to
219///    (and including) `data_dir`, deepest-first, so every directory entry along
220///    the new SSTable's path is durable.
221/// 2. Only then truncate the WAL — the WAL copy of the data is discarded here.
222///
223/// The full chain is synced on EVERY flush, not just the directories this
224/// attempt created (see [`dirs_to_sync`]). Fsyncing only the leaf — or only the
225/// dirs a given attempt created — would persist the component dirents but could
226/// leave the leaf's (or keyspace's) own entry in its parent unsynced, so a
227/// crash after the WAL truncate could lose the entire new SSTable directory.
228/// This is especially dangerous on a RETRY after a partially completed earlier
229/// attempt: the ancestors already exist, so a "created this attempt" list would
230/// be empty and only the leaf would be synced. Directory fsync is cheap and
231/// idempotent, so re-syncing already-durable ancestors is harmless.
232///
233/// If any directory fsync fails, the error is propagated and the WAL is
234/// **not** truncated, so recovery can still replay from the WAL.
235///
236/// When the WAL is already empty (`wal.size() == 0`) — e.g. a flush under
237/// `Durability::Disabled` or any flush whose mutations were never WAL'd — the
238/// truncate phase is skipped entirely and [`FlushDurabilityOutcome::Durable`]
239/// is returned (issue #1392). There is no recovery marker to preserve or
240/// destroy, so entering the truncate phase would only risk a spurious
241/// after-commit error on an already-durable flush. The directory-sync barrier
242/// above still runs unconditionally.
243///
244/// The WAL truncate is phase-aware (issue #1392):
245///
246/// * [`TruncateError::BeforeMutation`] — the truncate failed without touching
247///   the WAL contents, so the WAL is still intact. The error is surfaced as a
248///   warning and the WAL is deliberately left in place: it remains a durable
249///   replay marker, and replay is idempotent (identical mutations reconcile by
250///   last-write-wins timestamp), so this degrades to a redundant re-flush
251///   rather than data loss.
252/// * [`TruncateError::AfterMutation`] — the truncate failed *after* `set_len(0)`
253///   already zeroed the WAL. The WAL is **no longer** a replayable marker, so
254///   this returns [`FlushDurabilityOutcome::WalTruncateFailedAfterCommit`]
255///   carrying the error. The published SSTable is now the ONLY durable copy of
256///   the flushed data, so the caller must treat the flush as COMMITTED for
257///   state purposes (clear the memtable and advance the generation) BEFORE
258///   surfacing the error — otherwise a retry would reuse this generation and
259///   `File::create` would overwrite the sole durable copy, losing data on a
260///   crash mid-retry (issue #1392).
261///
262/// Every emitted SSTable component's *contents* must already be fsynced by the
263/// component writers (see [`crate::storage::sstable::writer::SSTableWriter::finish`])
264/// before this function runs. Ordering is load-bearing: component-content
265/// fsyncs (inside `writer.finish()`) happen first, then this function fsyncs
266/// the directory, then it truncates the WAL.
267pub(crate) fn finalize_flush_durability(
268    barrier: &dyn DurabilityBarrier,
269    data_path: &Path,
270    data_dir: &Path,
271    wal: &mut WriteAheadLog,
272) -> Result<FlushDurabilityOutcome> {
273    // Step 1: persist the new SSTable's dirents BEFORE dropping the WAL copy.
274    // (The component *contents* were already fsynced inside `writer.finish()`.)
275    // This fsyncs the FULL ancestor chain from the leaf SSTable directory up to
276    // (and including) the data root, deepest-first, so every directory entry
277    // along the SSTable's path is durable — not just the leaf, and regardless
278    // of whether an earlier flush attempt already created some ancestors
279    // (issue #1392).
280    let sstable_dir = data_path.parent().unwrap_or(data_dir);
281    for dir in dirs_to_sync(sstable_dir, data_dir) {
282        barrier.sync_dir(&dir)?;
283    }
284
285    // Step 2: it is now durable-safe to truncate the WAL.
286    //
287    // Skip the truncate phase entirely when the WAL is already empty (issue
288    // #1392): a flush under `Durability::Disabled`, or any flush whose
289    // mutations were never WAL'd, has nothing to truncate — there is no
290    // recovery marker to preserve or destroy. Entering the truncate phase
291    // anyway would let a post-`set_len(0)` seek/sync fault surface as a
292    // spurious `WalTruncateFailedAfterCommit` on a flush that is already fully
293    // durable in the published SSTable. The directory-sync barrier above still
294    // runs unconditionally, so the durability guarantee is unchanged.
295    if wal.size() == 0 {
296        return Ok(FlushDurabilityOutcome::Durable);
297    }
298
299    match barrier.truncate_wal(wal) {
300        Ok(()) => Ok(FlushDurabilityOutcome::Durable),
301        Err(TruncateError::BeforeMutation(e)) => {
302            // The WAL was never mutated; it is still a valid replay marker.
303            log::warn!(
304                "WAL truncate failed before mutating the WAL ({e}). Leaving the \
305                 WAL intact as a durable replay marker; the next startup will \
306                 replay it idempotently (last-write-wins), so no data is lost."
307            );
308            Ok(FlushDurabilityOutcome::Durable)
309        }
310        Err(TruncateError::AfterMutation(e)) => {
311            // The WAL was already zeroed by set_len(0) before the failure, so it
312            // is NO LONGER a replayable marker. The published SSTable is now the
313            // ONLY durable copy of these mutations. Report the flush as COMMITTED
314            // (carrying the error) so the caller advances the generation and
315            // clears the memtable BEFORE surfacing the failure — a retry must
316            // write a NEW generation rather than overwrite the published SSTable
317            // (which `File::create` would do at the reused generation), which
318            // would lose the data on a crash mid-retry (issue #1392).
319            Ok(FlushDurabilityOutcome::WalTruncateFailedAfterCommit(
320                Error::Storage(format!(
321                    "WAL truncate failed AFTER zeroing the WAL ({e}); the WAL is \
322                     no longer a replayable recovery marker. The flushed SSTable \
323                     is durable and the generation has been advanced so a retry \
324                     will not overwrite it."
325                )),
326            ))
327        }
328    }
329}

339    /// A single durability-relevant filesystem operation, recorded by the test
340    /// seam to assert ordering (the directory fsync MUST precede the WAL
341    /// truncate).
342    #[derive(Debug, Clone, PartialEq, Eq)]
343    enum FsOp {
344        /// fsync of a directory whose dirents changed during the flush (the leaf
345        /// SSTable directory and/or a newly created ancestor). Carries the path
346        /// so tests can assert *which* directories were synced and in what order.
347        SyncDir(PathBuf),
348        /// Truncation of the WAL to zero length.
349        WalTruncate,
350    }

352    fn test_mutation(id: i32, name: &str) -> Mutation {
353        let table_id = TableId::new("test_ks", "test_table");
354        let pk = PartitionKey::single("id", Value::Integer(id));
355        let ops = vec![CellOperation::Write {
356            column: "name".to_string(),
357            value: Value::Text(name.to_string()),
358        }];
359        Mutation::new(table_id, pk, None, ops, 1234567890, None)
360    }

362    /// Records the order of operations; `sync_dir` optionally faults.
363    struct RecordingBarrier {
364        ops: RefCell<Vec<FsOp>>,
365        fail_dir_sync: bool,
366        /// Fault the truncate BEFORE mutating the WAL (WAL stays intact).
367        fail_truncate_before_mutation: bool,
368        /// Fault the truncate AFTER `set_len(0)` has already zeroed the WAL.
369        fail_truncate_after_mutation: bool,
370    }
371
372    impl RecordingBarrier {
373        fn new() -> Self {
374            Self {
375                ops: RefCell::new(Vec::new()),
376                fail_dir_sync: false,
377                fail_truncate_before_mutation: false,
378                fail_truncate_after_mutation: false,
379            }
380        }
381    }
382
383    impl DurabilityBarrier for RecordingBarrier {
384        fn sync_dir(&self, dir: &Path) -> Result<()> {
385            if self.fail_dir_sync {
386                return Err(Error::Storage("injected dir fsync fault".to_string()));
387            }
388            self.ops.borrow_mut().push(FsOp::SyncDir(dir.to_path_buf()));
389            Ok(())
390        }
391
392        fn truncate_wal(&self, wal: &mut WriteAheadLog) -> std::result::Result<(), TruncateError> {
393            if self.fail_truncate_before_mutation {
394                // WAL contents untouched.
395                return Err(TruncateError::BeforeMutation(Error::Storage(
396                    "injected pre-mutation truncate fault".to_string(),
397                )));
398            }
399            if self.fail_truncate_after_mutation {
400                // Model a failure that occurs AFTER set_len(0): actually zero the
401                // WAL (mutating it), then report an after-mutation fault. The WAL
402                // is now empty and is no longer a replayable marker.
403                wal.truncate().map_err(TruncateError::AfterMutation)?;
404                return Err(TruncateError::AfterMutation(Error::Storage(
405                    "injected post-set_len truncate fault".to_string(),
406                )));
407            }
408            self.ops.borrow_mut().push(FsOp::WalTruncate);
409            wal.truncate().map_err(TruncateError::BeforeMutation)
410        }
411    }

416        let dir = TempDir::new().unwrap();

515    // Issue #1392: a WAL truncate that fails AFTER `set_len(0)` has already
516    // zeroed the WAL must NOT be swallowed as "WAL intact". The WAL is no longer
517    // a replayable marker, so finalize returns
518    // `WalTruncateFailedAfterCommit(err)` — the SSTable is committed (so the
519    // caller advances the generation and clears the memtable) but the error is
520    // carried so it is still surfaced to the caller.
521    #[test]
522    fn faulted_truncate_after_mutation_reports_committed_with_error() {
523        let dir = TempDir::new().unwrap();
524        let mut wal = WriteAheadLog::create(dir.path()).unwrap();
525        wal.append(&test_mutation(11, "Dave")).unwrap();
526        wal.sync().unwrap();
527
528        let mut barrier = RecordingBarrier::new();
529        barrier.fail_truncate_after_mutation = true;
530
531        let outcome = finalize_flush_durability(
532            &barrier,
533            &dir.path().join("nb-1-big-Data.db"),
534            dir.path(),
535            &mut wal,
536        )
537        .expect("a post-mutation truncate failure is COMMITTED, not a hard error");
538        let err = match outcome {
539            FlushDurabilityOutcome::WalTruncateFailedAfterCommit(e) => e,
540            FlushDurabilityOutcome::Durable => {
541                panic!("post-mutation truncate failure must not report Durable")
542            }
543        };
544        assert!(
545            matches!(err, Error::Storage(msg) if msg.contains("no longer a replayable recovery marker")),
546            "post-mutation truncate failure must carry a storage error"
547        );
548
549        // The dir was synced (recorded); the WAL was mutated (zeroed) so it is
550        // NOT a replay marker anymore — this is precisely why the error must be
551        // propagated rather than swallowed.
552        assert_eq!(
553            barrier.ops.into_inner(),
554            vec![FsOp::SyncDir(dir.path().to_path_buf())]
555        );

565    // Issue #1392 (roborev LOW, FINDING 1): a flush whose WAL is already empty
566    // (e.g. `Durability::Disabled`, or any flush whose mutations were never
567    // WAL'd) must SKIP the WAL truncate phase entirely. The directory-sync
568    // barrier still runs, but `truncate_wal` is never invoked — so a
569    // truncate-phase fault cannot turn an already-durable flush into a spurious
570    // `WalTruncateFailedAfterCommit` error.
571    #[test]
572    fn empty_wal_skips_truncate_phase() {
573        let dir = TempDir::new().unwrap();
574        // A freshly created WAL with nothing appended has size 0.
575        let mut wal = WriteAheadLog::create(dir.path()).unwrap();
576        assert_eq!(wal.size(), 0, "a fresh, unwritten WAL must have size 0");
577
578        // Arm the truncate to fault AFTER mutating the WAL: if the truncate
579        // phase were entered at all, this would surface as
580        // `WalTruncateFailedAfterCommit`. Skipping the phase must avoid it.
581        let mut barrier = RecordingBarrier::new();
582        barrier.fail_truncate_after_mutation = true;
583
584        let outcome = finalize_flush_durability(
585            &barrier,
586            &dir.path().join("nb-1-big-Data.db"),
587            dir.path(),
588            &mut wal,
589        )
590        .expect("an empty-WAL flush must not enter the truncate phase");
591        assert!(
592            matches!(outcome, FlushDurabilityOutcome::Durable),
593            "empty WAL: the truncate phase is skipped, so the flush stays Durable"
594        );
595
596        // The directory was still synced, but the WAL truncate was never
597        // attempted (no WalTruncate op recorded and the armed fault never fired).
598        assert_eq!(
599            barrier.ops.into_inner(),
600            vec![FsOp::SyncDir(dir.path().to_path_buf())],
601            "dir fsync still runs; the truncate phase is skipped for an empty WAL"
602        );
603    }

623    // Issue #1392 (FINDING 1): on the FIRST flush into a brand-new
624    // keyspace/table path, the barrier must fsync the leaf AND every ancestor up
625    // to the data root — deepest first — BEFORE the WAL truncate, so a crash
626    // cannot lose the freshly created directory tree.
627    #[test]
628    fn first_flush_fsyncs_new_ancestor_dirs_before_truncate() {
629        let root = TempDir::new().unwrap();
630        let data_dir = root.path();
631        // WAL lives at the data_dir root (as in a real engine layout).
632        let mut wal = WriteAheadLog::create(data_dir).unwrap();
633        wal.append(&test_mutation(1, "Alice")).unwrap();
634        wal.sync().unwrap();
635
636        // Leaf SSTable dir: data_dir/test_ks/test_table (both ancestors new).
637        let ks_dir = data_dir.join("test_ks");
638        let leaf_dir = ks_dir.join("test_table");
639        let data_path = leaf_dir.join("nb-1-big-Data.db");
640
641        let barrier = RecordingBarrier::new();
642        finalize_flush_durability(&barrier, &data_path, data_dir, &mut wal).unwrap();
643
644        // Expected: fsync the leaf (its component dirents), then the keyspace dir
645        // (persists the leaf's own entry), then the data_dir (persists the
646        // keyspace's own entry) — deepest-first — and only THEN truncate the WAL.
647        assert_eq!(
648            barrier.ops.into_inner(),
649            vec![
650                FsOp::SyncDir(leaf_dir),
651                FsOp::SyncDir(ks_dir),
652                FsOp::SyncDir(data_dir.to_path_buf()),
653                FsOp::WalTruncate,
654            ],
655            "every ancestor dir up to the data root must be fsynced (deepest \
656             first) before the WAL truncate"
657        );
658        assert_eq!(wal.replay().unwrap().mutations.len(), 0);
659    }
