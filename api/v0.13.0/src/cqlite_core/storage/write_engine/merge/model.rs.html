1//! Merge data model — the entities that flow through the k-way merge stream.
2//!
3//! Extracted from `merge.rs` (issue #945) as a pure, behavior-preserving
4//! relocation: `MergeEntry`, `RowData`, `CellData`, `ComplexDeletion`,
5//! `MergeStep`, and `MergeStats` plus their builders. The parent `merge`
6//! module re-exports every item, so external `write_engine::merge::MergeEntry`
7//! paths keep resolving unchanged. No logic moved here — see `mod.rs` for the
8//! merge machinery and `reconcile.rs` for the reconciliation kernel.

22/// Entry in the merge stream
23///
24/// Represents a single row from one of the input SSTables. This is the
25/// fundamental unit that flows through the merge heap.
26#[cfg(feature = "write-support")]
27#[derive(Debug, Clone, PartialEq, Eq)]
28pub struct MergeEntry {
29    /// Which SSTable this came from (0 = newest)
30    pub run_index: usize,
31    /// Partition key with token
32    pub key: DecoratedKey,
33    /// Clustering key (None for tables without clustering)
34    pub clustering_key: Option<ClusteringKey>,
35    /// Timestamp in microseconds since Unix epoch
36    pub timestamp: i64,
37    /// Row data (live cells or tombstone)
38    pub row_data: RowData,
39    /// Complex (collection / UDT) deletion markers for the multi-cell columns
40    /// of this `(pk, ck)` (issue #886 substrate).
41    ///
42    /// Carried through the merge so the per-cell-path collection/UDT followup
43    /// (#844) and shadow-before-purge (#887) can preserve collection/UDT
44    /// deletion timestamps. `reconcile_cluster` unions and preserves these
45    /// across a cluster, but reconciliation does **not yet consult** them and
46    /// the writer does not yet apply them — defaults to empty, so output is
47    /// byte-unchanged. Population (per-element reader emit) lands in #899.
48    pub complex_deletions: Vec<ComplexDeletion>,
49    /// Range-deletion marker covering a span of clustering keys (issue #886
50    /// substrate).
51    ///
52    /// A first-class slot so range tombstones can flow through the merge stream
53    /// instead of being skipped by the parser; applying them to shadow covered
54    /// cells is the follow-up #846. Carried (and timestamp-max-preserved)
55    /// through `reconcile_cluster` but **not yet consulted** by reconciliation
56    /// or the writer, so output is byte-unchanged. `None` when this entry
57    /// carries no range deletion.
58    pub range_deletion: Option<RangeTombstone>,
59    /// Row-level deletion that COEXISTS with this entry's surviving live cells
60    /// (issue #932).
61    ///
62    /// `Some((deletion_time_micros, local_deletion_time_secs))` when a row
63    /// tombstone whose timestamp is OLDER than the surviving cells must be kept
64    /// alongside `RowData::Live`. `RowData::Live` has no row-deletion slot, so —
65    /// mirroring the carry-only `complex_deletions` / `range_deletion` fields —
66    /// the coexisting deletion rides here instead of forcing a `RowData` change
67    /// across ~100 `RowData::Live` sites. `reconcile_cluster` folds it into the
68    /// winning row deletion and re-attaches it to the emitted live entry;
69    /// `merge_entry_to_mutation` emits it as `Mutation::row_tombstone`. Without
70    /// this, a partial compaction that keeps newer cells would DROP the row
71    /// deletion and let older cells of other columns (in SSTables not part of the
72    /// compaction) resurrect. `None` for a row with no coexisting deletion.
73    pub row_deletion: Option<(i64, i32)>,
74    /// Partition-level deletion `(markedForDeleteAt µs, localDeletionTime s)`
75    /// carried by a synthetic partition-tombstone carrier entry (issue #1072).
76    ///
77    /// `Some(..)` ONLY on the carrier entry the reader emits for a partition
78    /// header that has a tombstone (empty `RowData::Live`, `clustering_key:
79    /// None`). The merge extracts the MAX `markedForDeleteAt` across sources,
80    /// applies it as the OUTERMOST per-cell `<=` shadow floor for the whole
81    /// partition, and re-emits the surviving tombstone via
82    /// `merge_entry_to_mutation` (→ `Mutation::partition_tombstone`). Without
83    /// this a newer partition tombstone in one SSTable failed to shadow older
84    /// live rows in another, resurrecting the deleted partition. `None` for every
85    /// non-carrier entry.
86    pub partition_deletion: Option<(i64, i32)>,
87}

89impl MergeEntry {
90    /// Create a new merge entry.
91    ///
92    /// The carry-only #886 substrate fields (`complex_deletions`,
93    /// `range_deletion`) default to empty/`None`; attach them with
94    /// [`with_complex_deletions`](Self::with_complex_deletions) /
95    /// [`with_range_deletion`](Self::with_range_deletion) once the reader emit
96    /// surfaces them (#899).
97    pub fn new(
98        run_index: usize,
99        key: DecoratedKey,
100        clustering_key: Option<ClusteringKey>,
101        timestamp: i64,
102        row_data: RowData,
103    ) -> Self {
104        Self {
105            run_index,
106            key,
107            clustering_key,
108            timestamp,
109            row_data,
110            complex_deletions: Vec::new(),
111            range_deletion: None,
112            row_deletion: None,
113            partition_deletion: None,
114        }
115    }
116
117    /// Attach complex-deletion markers (issue #886 substrate; carry-only).
118    #[must_use]
119    pub fn with_complex_deletions(mut self, complex_deletions: Vec<ComplexDeletion>) -> Self {
120        self.complex_deletions = complex_deletions;
121        self
122    }
123
124    /// Attach a coexisting row-level deletion (issue #932).
125    ///
126    /// `deletion_time` is `markedForDeleteAt` (microseconds); `ldt` is the
127    /// `localDeletionTime` (GC-clock seconds, `0` = not surfaced). Used when a
128    /// row's surviving cells are newer than a row tombstone that must still be
129    /// preserved so it keeps shadowing older cells in non-compacted SSTables.
130    #[must_use]
131    pub fn with_row_deletion(mut self, deletion_time: i64, ldt: i32) -> Self {
132        self.row_deletion = Some((deletion_time, ldt));
133        self
134    }
135
136    /// Attach a range-deletion marker (issue #886 substrate; carry-only).
137    #[must_use]
138    pub fn with_range_deletion(mut self, range_deletion: RangeTombstone) -> Self {
139        self.range_deletion = Some(range_deletion);
140        self
141    }
142
143    /// Attach a partition-level deletion (issue #1072).
144    ///
145    /// `deletion_time` is `markedForDeleteAt` (microseconds); `ldt` is the
146    /// `localDeletionTime` (GC-clock seconds). Marks this entry as the synthetic
147    /// partition-tombstone carrier so the merge applies the partition floor and
148    /// re-emits the surviving tombstone.
149    #[must_use]
150    pub fn with_partition_deletion(mut self, partition_deletion: (i64, i32)) -> Self {
151        self.partition_deletion = Some(partition_deletion);
152        self
153    }
154
155    /// True when this entry has NO writer-emittable content at all — an empty
156    /// `RowData::Live { cells: vec![] }` with no row tombstone and no carried
157    /// deletion metadata. Routing such an entry to the writer would create a
158    /// PHANTOM live empty (pure-PK) row at timestamp 0 (`DataWriter::merge_row_group`
159    /// treats a no-op mutation as a primary-key insert), so `merge` filters it.
160    ///
161    /// Epic #899 Phase C: a COMPLEX DELETION is consumed by the writer (a real
162    /// `CellOperation::ComplexDeletion` marker), so a complex-deletion-only entry
163    /// is NOT a no-op.
164    ///
165    /// Issue #933: a RANGE DELETION is now ALSO consumed by the compaction writer —
166    /// `merge_entry_to_mutation` threads it onto the mutation's `range_tombstones`,
167    /// which the writer interleaves as on-disk bound markers AND uses to shadow
168    /// covered same-partition rows. A range-deletion-only carrier is therefore NO
169    /// LONGER a no-op and MUST reach the writer (dropping it would resurrect the
170    /// covered rows it shadowed — roborev #959 High #2). In practice
171    /// `reconcile_cluster` never emits a truly-empty entry (one with empty cells,
172    /// no row tombstone, and no carried metadata reconciles to `None`), so this
173    /// guard is now effectively unreachable but kept as a defensive phantom-row
174    /// filter.
175    #[must_use]
176    pub fn is_metadata_only_no_op(&self) -> bool {
177        matches!(&self.row_data, RowData::Live { cells } if cells.is_empty())
178            && self.complex_deletions.is_empty()
179            && self.range_deletion.is_none()
180            && self.row_deletion.is_none()
181            && self.partition_deletion.is_none()
182    }
183
184    /// True when this entry is a synthetic partition-tombstone carrier (issue
185    /// #1072): an empty live row whose only payload is `partition_deletion`.
186    /// Produced by [`Self::build_merge_entry`] from a reader `PartitionDelete`.
187    /// Such an entry produces NO output row (it yields a
188    /// `Mutation::partition_tombstone`, not a clustering row), so the merge must
189    /// not count it toward output row counts — but it MUST reach the writer.
190    #[must_use]
191    pub fn is_partition_delete_carrier(&self) -> bool {
192        self.partition_deletion.is_some()
193    }
194}

196/// Ord implementation for min-heap routing ONLY (not LWW winner selection).
197///
198/// This orders entries so the heap yields them grouped by partition and
199/// clustering key. The actual equal-timestamp Delete-vs-Live winner is chosen
200/// in `merge_partition_rows` (timestamp → liveness → run_index), NOT here.
201///
202/// Order by:
203/// 1. Token (ascending)
204/// 2. Key bytes (ascending, for hash collisions)
205/// 3. Clustering key (ascending, schema-aware)
206/// 4. Run index (ascending) - stable routing tiebreak only
207#[cfg(feature = "write-support")]
208impl Ord for MergeEntry {
209    fn cmp(&self, other: &Self) -> Ordering {
210        // Primary: by token
211        match self.key.token.cmp(&other.key.token) {
212            Ordering::Equal => {
213                // Secondary: by key bytes (hash collision resolution)
214                match self.key.key.cmp(&other.key.key) {
215                    Ordering::Equal => {
216                        // Tertiary: by clustering key
217                        match (&self.clustering_key, &other.clustering_key) {
218                            (None, None) => {
219                                // Quaternary: by run_index (lower = newer)
220                                self.run_index.cmp(&other.run_index)
221                            }
222                            (None, Some(_)) => Ordering::Less,
223                            (Some(_), None) => Ordering::Greater,
224                            (Some(a), Some(b)) => {
225                                // Use fallback Ord (not schema-aware at this level)
226                                // Schema-aware comparison happens during partition merge
227                                match a.cmp(b) {
228                                    Ordering::Equal => {
229                                        // Equal clustering keys: prefer lower run_index
230                                        self.run_index.cmp(&other.run_index)
231                                    }
232                                    other_ord => other_ord,
233                                }
234                            }
235                        }
236                    }
237                    other_ord => other_ord,
238                }
239            }
240            other_ord => other_ord,
241        }
242    }
243}

245#[cfg(feature = "write-support")]
246impl PartialOrd for MergeEntry {
247    fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
248        Some(self.cmp(other))
249    }
250}

270/// Cell data with timestamp, optional TTL, and (for complex columns) cell path.
271///
272/// ## Per-cell merge metadata (issue #886 — byte-parity foundation)
273///
274/// To reconcile per-cell and per-element data byte-faithfully (Cassandra
275/// `Cells#reconcile`), the merge entry must carry more than a single row-level
276/// timestamp. The fields below thread that richer state from the reader toward
277/// the followup behaviors (#844 per-cell-path collection/UDT merge, #848
278/// tombstone-vs-expiring TTL tie-break). They are **carried but not yet acted
279/// on** by reconciliation — this struct change is plumbing only and must not
280/// alter output bytes.
281///
282/// Where the reader does not yet surface a value the field is left `None`; the
283/// dependent issues fill it in once the reader is extended.
284#[cfg(feature = "write-support")]
285#[derive(Debug, Clone, PartialEq, Eq)]
286pub struct CellData {
287    /// Column name
288    pub column: String,
289    /// Column value
290    pub value: Value,
291    /// Cell timestamp (microseconds)
292    pub timestamp: i64,
293    /// TTL in seconds (None = no expiration)
294    pub ttl: Option<u32>,
295    /// Cell path for a complex (collection / non-frozen UDT) element — the
296    /// serialized element key/index that distinguishes one element of a
297    /// multi-cell column from another (issue #886 substrate).
298    ///
299    /// **Carry-only.** This field is threaded through the merge entry so that
300    /// per-element reconciliation can become byte-faithful, but nothing
301    /// populates or consumes it yet: the reader still collapses collections to
302    /// a single whole-column [`CellData`] and the writer does not read this
303    /// field. Population (per-element reader emit) and consumption (per-path
304    /// merge #844) land in the follow-up #899. `None` for simple cells.
305    pub cell_path: Option<Vec<u8>>,
306    /// Local deletion time in **seconds** since the Unix epoch for this cell
307    /// (the on-disk `localDeletionTime`), used by gc_grace purging and
308    /// expiring-cell tie-breaks (issue #886 substrate).
309    ///
310    /// For an expiring (TTL) cell this is the cell's expiration instant; for a
311    /// cell tombstone it is when the delete was applied.
312    ///
313    /// Populated for complex (collection / UDT) elements on the compaction read
314    /// path (epic #899, Phase C); `None` for simple cells whose LDT the reader
315    /// does not surface, and for live simple cells.
316    pub local_deletion_time: Option<i32>,
317    /// True when this `CellData` represents a single ELEMENT of a non-frozen
318    /// complex column (a list/set member or a map entry), as opposed to a simple
319    /// single-cell column (epic #899, Phase C).
320    ///
321    /// When `true`, [`cell_path`](Self::cell_path) is the element's authoritative
322    /// on-disk path and the merge→mutation step emits a
323    /// [`CellOperation::WriteComplexElement`] (preserving per-element
324    /// ts/ttl/ldt/path) rather than a whole-column `Write`. `false` for every
325    /// simple cell (whole-column collapse no longer happens on the production
326    /// path).
327    ///
328    /// [`CellOperation::WriteComplexElement`]: crate::storage::write_engine::mutation::CellOperation::WriteComplexElement
329    pub is_complex_element: bool,
330    /// Authoritative IS_DELETED (0x01) flag for a complex element (epic #899,
331    /// Phase C). Carried verbatim from [`ComplexElement::is_deleted`] — NOT
332    /// re-derived from value/ttl/ldt shape, so an expiring SET member (empty
333    /// value, ttl + ldt set) is correctly NOT treated as a tombstone
334    /// (no-heuristics mandate). Always `false` for simple cells (a simple cell
335    /// tombstone rides in [`value`](Self::value) as `Value::Tombstone`).
336    ///
337    /// [`ComplexElement::is_deleted`]: crate::storage::sstable::reader::compaction_row::ComplexElement::is_deleted
338    pub is_deleted: bool,
339    /// On-disk HAS_EMPTY_VALUE (0x04) flag for a complex element (epic #899,
340    /// Phase C). `true` for a SET member (whose identity lives in the cell_path)
341    /// and any genuinely empty-value element, so the writer reproduces the same
342    /// on-disk emptiness rather than re-deriving it from the decoded value.
343    /// Always `false` for simple cells.
344    pub has_empty_value: bool,
345}

367/// Complex (collection / non-frozen UDT) deletion marker for one column
368/// (issue #886 substrate).
369///
370/// Cassandra writes a complex-deletion marker ahead of a multi-cell column's
371/// elements to delete every element written at or before `marked_for_delete_at`.
372/// A merged complex deletion is dropped unless it **strictly supersedes** the
373/// active one (Cassandra commit `bd244649`). CQLite currently reduces this to a
374/// boolean and discards the timestamps; this first-class entity preserves them
375/// so per-path merge (#844) and shadow-before-purge (#887) can be byte-faithful.
376///
377/// **Carry-only.** Carried on [`MergeEntry`] (unioned through
378/// `reconcile_cluster`) but not yet populated by the reader or applied during
379/// the merge — that is the follow-up #899/#887.
380#[cfg(feature = "write-support")]
381#[derive(Debug, Clone, PartialEq, Eq)]
382pub struct ComplexDeletion {
383    /// Name of the complex column this deletion covers.
384    pub column: String,
385    /// Deletion timestamp (`markedForDeleteAt`) in microseconds since the epoch.
386    pub marked_for_delete_at: i64,
387    /// Local deletion time in seconds since the epoch.
388    pub local_deletion_time: i32,
389}
