1//! JSON schema handling for the schema aggregator.
2//!
3//! Parses JSON schema files (minimal single-table format and full
4//! multi-table + UDT format) into the intermediate [`ParsedSchema`] format,
5//! and converts the JSON-shaped definitions into [`TableSchema`]/[`UdtTypeDef`].
6
7use std::collections::HashMap;
8use std::path::Path;
9
10use crate::error::{Error, Result};
11use crate::schema::{ClusteringColumn, Column, KeyColumn, TableSchema};
12use crate::types::UdtTypeDef;
13
14use super::{ParsedSchema, SchemaAggregator};
15
16/// JSON schema formats
17#[derive(Debug, serde::Deserialize)]
18#[serde(untagged)]
19pub(super) enum JsonSchemaFormat {
20    /// Minimal format: single table with "table" field
21    Minimal(MinimalTableSchema),
22    /// Full format: with "tables" array and optional "udts" array
23    Full(FullSchema),
24}
25
26/// Minimal JSON schema format (single table)
27#[derive(Debug, serde::Deserialize)]
28pub(super) struct MinimalTableSchema {
29    keyspace: String,
30    table: String,
31    columns: Vec<JsonColumn>,
32    #[serde(default)]
33    partition_keys: Vec<String>,
34    #[serde(default)]
35    primary_key: Vec<String>, // Synonym for partition_keys when no clustering
36    #[serde(default)]
37    clustering_keys: Vec<JsonClusteringKey>,
38    /// Dropped-column drop times (column → drop_time_micros), used for
39    /// dropped-column filtering during compaction (#904/#847).
40    #[serde(default)]
41    dropped_columns: HashMap<String, i64>,
42}
43
44/// Full JSON schema format (multiple tables + UDTs)
45/// Note: Both `udts` and `tables` are optional to support UDT-only or table-only files
46#[derive(Debug, serde::Deserialize)]
47pub(super) struct FullSchema {
48    keyspace: String,
49    #[serde(default)]
50    udts: Vec<JsonUdt>,
51    #[serde(default)]
52    tables: Vec<JsonTable>,
53}
54
55/// JSON table definition
56#[derive(Debug, serde::Deserialize)]
57struct JsonTable {
58    name: String,
59    columns: Vec<JsonColumn>,
60    #[serde(default)]
61    partition_keys: Vec<String>,
62    #[serde(default)]
63    primary_key: Vec<String>,
64    #[serde(default)]
65    clustering_keys: Vec<JsonClusteringKey>,
66    /// Dropped-column drop times (column → drop_time_micros), used for
67    /// dropped-column filtering during compaction (#904/#847).
68    #[serde(default)]
69    dropped_columns: HashMap<String, i64>,
70}
71
72/// JSON column definition
73#[derive(Debug, serde::Deserialize)]
74struct JsonColumn {
75    name: String,
76    #[serde(alias = "data_type")]
77    r#type: String,
78    #[serde(default)]
79    nullable: bool,
80}
81
82/// JSON clustering key definition
83#[derive(Debug, serde::Deserialize)]
84struct JsonClusteringKey {
85    name: String,
86    #[serde(alias = "data_type")]
87    r#type: String,
88    #[serde(default)]
89    order: Option<String>,
90}
91
92/// JSON UDT definition
93#[derive(Debug, serde::Deserialize)]
94struct JsonUdt {
95    name: String,
96    fields: Vec<JsonUdtField>,
97}
98
99/// JSON UDT field definition
100#[derive(Debug, serde::Deserialize)]
101struct JsonUdtField {
102    name: String,
103    #[serde(alias = "data_type")]
104    r#type: String,
105    #[serde(default = "default_nullable")]
106    nullable: bool,
107}
108
109fn default_nullable() -> bool {
110    true
111}

113impl SchemaAggregator {
114    /// Parse a JSON file (either minimal or full format)
115    pub(super) async fn parse_json_file(&self, path: &Path) -> Result<Option<ParsedSchema>> {
116        let content = std::fs::read_to_string(path)?;
117
118        let json_schema: JsonSchemaFormat = serde_json::from_str(&content)
119            .map_err(|e| Error::schema(format!("Invalid JSON in {}: {}", path.display(), e)))?;
120
121        match json_schema {
122            JsonSchemaFormat::Minimal(minimal) => self.parse_minimal_format(path, minimal).await,
123            JsonSchemaFormat::Full(full) => self.parse_full_format(path, full).await,
124        }
125    }
126
127    /// Parse minimal JSON format (single table)
128    async fn parse_minimal_format(
129        &self,
130        _path: &Path,
131        minimal: MinimalTableSchema,
132    ) -> Result<Option<ParsedSchema>> {
133        let table_schema = self.convert_minimal_to_table_schema(minimal)?;
134        let keyspace = table_schema.keyspace.clone();
135
136        let mut tables = HashMap::new();
137        // Key by fully-qualified name (keyspace.table) to avoid multi-keyspace collisions
138        let qualified_name = format!("{}.{}", table_schema.keyspace, table_schema.table);
139        tables.insert(qualified_name, table_schema);
140
141        Ok(Some(ParsedSchema {
142            keyspace,
143            tables,
144            udts: HashMap::new(),
145        }))
146    }
147
148    /// Parse full JSON format (multiple tables + UDTs)
149    async fn parse_full_format(
150        &self,
151        _path: &Path,
152        full: FullSchema,
153    ) -> Result<Option<ParsedSchema>> {
154        let keyspace = full.keyspace.clone();
155        let mut tables = HashMap::new();
156        let mut udts = HashMap::new();
157
158        // Parse UDTs
159        for udt_json in full.udts {
160            let udt_def = self.convert_json_udt_to_typedef(&keyspace, udt_json)?;
161            // Key by fully-qualified name (keyspace.typename) to avoid multi-keyspace collisions
162            let qualified_name = format!("{}.{}", udt_def.keyspace, udt_def.name);
163            udts.insert(qualified_name, udt_def);
164        }
165
166        // Parse tables
167        for table_json in full.tables {
168            let table_schema = self.convert_json_table_to_table_schema(&keyspace, table_json)?;
169            // Key by fully-qualified name (keyspace.table) to avoid multi-keyspace collisions
170            let qualified_name = format!("{}.{}", table_schema.keyspace, table_schema.table);
171            tables.insert(qualified_name, table_schema);
172        }
173
174        Ok(Some(ParsedSchema {
175            keyspace,
176            tables,
177            udts,
178        }))
179    }
180
181    /// Convert minimal JSON format to TableSchema
182    pub(super) fn convert_minimal_to_table_schema(
183        &self,
184        minimal: MinimalTableSchema,
185    ) -> Result<TableSchema> {
186        // Determine partition keys (prefer partition_keys, fallback to primary_key)
187        let partition_key_names = if !minimal.partition_keys.is_empty() {
188            minimal.partition_keys
189        } else if !minimal.primary_key.is_empty() {
190            minimal.primary_key
191        } else {
192            return Err(Error::schema(
193                "Table must have partition_keys or primary_key".to_string(),
194            ));
195        };
196
197        // Build columns
198        let columns: Vec<Column> = minimal
199            .columns
200            .iter()
201            .map(|col| Column {
202                name: col.name.clone(),
203                data_type: col.r#type.clone(),
204                nullable: col.nullable,
205                default: None,
206                is_static: false, // TODO: minimal schema doesn't track static columns yet
207            })
208            .collect();
209
210        // Build partition keys
211        let partition_keys: Vec<KeyColumn> = partition_key_names
212            .iter()
213            .enumerate()
214            .map(|(pos, name)| {
215                let col = minimal
216                    .columns
217                    .iter()
218                    .find(|c| &c.name == name)
219                    .ok_or_else(|| {
220                        Error::schema(format!("Partition key '{}' not found in columns", name))
221                    })?;
222
223                Ok(KeyColumn {
224                    name: col.name.clone(),
225                    data_type: col.r#type.clone(),
226                    position: pos,
227                })
228            })
229            .collect::<Result<Vec<_>>>()?;
230
231        // Build clustering keys
232        let clustering_keys: Vec<ClusteringColumn> = minimal
233            .clustering_keys
234            .iter()
235            .enumerate()
236            .map(|(pos, ck)| ClusteringColumn {
237                name: ck.name.clone(),
238                data_type: ck.r#type.clone(),
239                position: pos,
240                order: ck.order.as_deref().map(|s| s.into()).unwrap_or_default(),
241            })
242            .collect();
243
244        let schema = TableSchema {
245            keyspace: minimal.keyspace,
246            table: minimal.table,
247            partition_keys,
248            clustering_keys,
249            columns,
250            comments: HashMap::new(),
251            dropped_columns: minimal.dropped_columns,
252        };
253
254        schema.validate()?;
255        Ok(schema)
256    }

258    /// Convert JSON table to TableSchema
259    fn convert_json_table_to_table_schema(
260        &self,
261        keyspace: &str,
262        table_json: JsonTable,
263    ) -> Result<TableSchema> {
264        let partition_key_names = if !table_json.partition_keys.is_empty() {
265            table_json.partition_keys
266        } else if !table_json.primary_key.is_empty() {
267            table_json.primary_key
268        } else {
269            return Err(Error::schema(format!(
270                "Table '{}' must have partition_keys or primary_key",
271                table_json.name
272            )));
273        };
274
275        let columns: Vec<Column> = table_json
276            .columns
277            .iter()
278            .map(|col| Column {
279                name: col.name.clone(),
280                data_type: col.r#type.clone(),
281                nullable: col.nullable,
282                default: None,
283                is_static: false, // TODO: JSON schema doesn't track static columns yet
284            })
285            .collect();
286
287        let partition_keys: Vec<KeyColumn> = partition_key_names
288            .iter()
289            .enumerate()
290            .map(|(pos, name)| {
291                let col = table_json
292                    .columns
293                    .iter()
294                    .find(|c| &c.name == name)
295                    .ok_or_else(|| {
296                        Error::schema(format!(
297                            "Partition key '{}' not found in columns of table '{}'",
298                            name, table_json.name
299                        ))
300                    })?;
301
302                Ok(KeyColumn {
303                    name: col.name.clone(),
304                    data_type: col.r#type.clone(),
305                    position: pos,
306                })
307            })
308            .collect::<Result<Vec<_>>>()?;
309
310        let clustering_keys: Vec<ClusteringColumn> = table_json
311            .clustering_keys
312            .iter()
313            .enumerate()
314            .map(|(pos, ck)| ClusteringColumn {
315                name: ck.name.clone(),
316                data_type: ck.r#type.clone(),
317                position: pos,
318                order: ck.order.as_deref().map(|s| s.into()).unwrap_or_default(),
319            })
320            .collect();
321
322        let schema = TableSchema {
323            keyspace: keyspace.to_string(),
324            table: table_json.name,
325            partition_keys,
326            clustering_keys,
327            columns,
328            comments: HashMap::new(),
329            dropped_columns: table_json.dropped_columns,
330        };
331
332        schema.validate()?;
333        Ok(schema)
334    }
335
336    /// Convert JSON UDT to UdtTypeDef
337    fn convert_json_udt_to_typedef(&self, keyspace: &str, udt_json: JsonUdt) -> Result<UdtTypeDef> {
338        let mut udt_def = UdtTypeDef::new(keyspace.to_string(), udt_json.name);
339
340        for field in udt_json.fields {
341            let field_type = crate::schema::CqlType::parse(&field.r#type)?;
342            udt_def = udt_def.with_field(field.name, field_type, field.nullable);
343        }
344
345        Ok(udt_def)
346    }
