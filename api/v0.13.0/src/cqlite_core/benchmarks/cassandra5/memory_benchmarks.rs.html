44        let mut results = Vec::new();

53            100.0,  // 100MB - large

59        println!(

62        );
63
64        for &size_mb in &test_sizes_mb {
65            println!("\n📏 Memory benchmark for {:.0} MB file", size_mb);
66
67            // Skip very large tests if insufficient memory
68            if size_mb >= 500.0 && !self.has_sufficient_memory(size_mb) {
69                println!(
70                    "   ⏭️ Skipping {:.0} MB test due to insufficient system memory",

73                continue;
74            }
75
76            // Test different memory usage patterns
77            let memory_tests = vec![
78                ("Sequential", MemoryAccessPattern::Sequential),
79                ("Random", MemoryAccessPattern::Random),
80                ("Streaming", MemoryAccessPattern::Streaming),
81                ("Chunked", MemoryAccessPattern::Chunked),

103    /// Benchmark memory usage for specific file size and access pattern
104    async fn benchmark_memory_usage(
105        &self,
106        size_mb: f64,
107        pattern: MemoryAccessPattern,
108        pattern_name: &str,
109        targets: &PRDTargets,
110    ) -> Result<BenchmarkResult> {
111        let benchmark_name = format!("Memory_{}_{:.0}MB", pattern_name, size_mb);
112
113        // Generate test data
114        let test_data = generate_test_data(size_mb);
115        let mut memory_monitor = MemoryMonitor::new();
116
117        // Force garbage collection before starting
118        #[cfg(not(target_env = "msvc"))]
119        {
120            // On non-MSVC targets, try to trigger GC-like behavior
121            std::hint::black_box(&test_data);
122        }
123
124        let timer = PrecisionTimer::start();
125        memory_monitor.sample();
126
127        // Simulate different access patterns
128        let operations_count = match pattern {
129            MemoryAccessPattern::Sequential => {
130                self.sequential_access(&test_data, &mut memory_monitor)
131                    .await?
132            }
133            MemoryAccessPattern::Random => {
134                self.random_access(&test_data, &mut memory_monitor).await?
135            }
136            MemoryAccessPattern::Streaming => {
137                self.streaming_access(&test_data, &mut memory_monitor)
138                    .await?
139            }
140            MemoryAccessPattern::Chunked => {
141                self.chunked_access(&test_data, &mut memory_monitor).await?
142            }
143        };
144
145        let duration = timer.elapsed_duration();
146        let memory_usage_mb = memory_monitor.peak_usage_mb();
147        let avg_memory_mb = memory_monitor.average_usage_mb();
148
149        // Calculate metrics
150        let throughput_mb_per_sec = size_mb / duration.as_secs_f64();
151        let memory_efficiency = size_mb / memory_usage_mb.max(0.1);
152        let operations_per_second = operations_count as f64 / duration.as_secs_f64();
153
154        // Check PRD compliance - memory is the primary concern
155        let meets_target = memory_usage_mb <= targets.memory_limit_mb
156            && throughput_mb_per_sec >= targets.parse_speed_mb_per_sec * 0.5; // Allow 50% slower for memory-focused test
157
158        let target_comparison = if meets_target {
159            format!(
160                "✅ Meets memory targets ({:.1} MB ≤ {})",
161                memory_usage_mb, targets.memory_limit_mb
162            )
163        } else {
164            format!(
165                "❌ Exceeds memory limit ({:.1} MB > {})",
166                memory_usage_mb, targets.memory_limit_mb
167            )
168        };
169
170        let mut details = HashMap::new();
171        details.insert("peak_memory_mb".to_string(), memory_usage_mb);
172        details.insert("average_memory_mb".to_string(), avg_memory_mb);
173        details.insert("memory_per_mb_ratio".to_string(), memory_usage_mb / size_mb);
174        details.insert("access_pattern".to_string(), pattern as u8 as f64);
175
176        println!(
177            "     ✅ {} {:.0}MB: {:.1} MB peak memory ({:.2}x file size), {:.2} MB/s",
178            pattern_name,
179            size_mb,
180            memory_usage_mb,
181            memory_usage_mb / size_mb,
182            throughput_mb_per_sec
183        );
184
185        Ok(BenchmarkResult {
186            benchmark_name,
187            file_size_mb: size_mb,
188            duration,
189            throughput_mb_per_sec,
190            memory_usage_mb,
191            memory_efficiency,
192            compression_ratio: None,
193            operations_per_second,
194            meets_prd_target: meets_target,
195            target_comparison,
196            details,
197        })
198    }

200    /// Sequential memory access pattern - simulates reading entire file sequentially
201    async fn sequential_access(&self, data: &[u8], monitor: &mut MemoryMonitor) -> Result<usize> {
202        let chunk_size = 64 * 1024; // 64KB chunks
203        let mut operations = 0;
204        let mut position = 0;
205
206        while position < data.len() {
207            let end = (position + chunk_size).min(data.len());
208            let _chunk = &data[position..end];
209
210            // Simulate processing the chunk
211            std::hint::black_box(_chunk);
212
213            operations += 1;
214            position += chunk_size;
215
216            // Sample memory every 100 operations
217            if operations % 100 == 0 {
218                monitor.sample();
219            }
220        }
221
222        monitor.sample();
223        Ok(operations)
224    }

226    /// Random memory access pattern - simulates random seeks within the file
227    async fn random_access(&self, data: &[u8], monitor: &mut MemoryMonitor) -> Result<usize> {
228        let chunk_size = 4 * 1024; // 4KB chunks for random access
229        let num_operations = 1000.min(data.len() / chunk_size);
230
231        // Create pseudo-random access pattern
232        let mut seed = 12345u64;
233
234        for i in 0..num_operations {
235            // Simple LCG for pseudo-random numbers
236            seed = seed.wrapping_mul(1_103_515_245).wrapping_add(12345);
237            let offset = (seed as usize) % (data.len().saturating_sub(chunk_size));
238
239            let _chunk = &data[offset..offset + chunk_size];
240            std::hint::black_box(_chunk);
241
242            // Sample memory every 100 operations
243            if i % 100 == 0 {
244                monitor.sample();
245            }
246        }
247
248        monitor.sample();
249        Ok(num_operations)
250    }

252    /// Streaming memory access pattern - simulates continuous streaming with minimal buffering
253    async fn streaming_access(&self, data: &[u8], monitor: &mut MemoryMonitor) -> Result<usize> {
254        let stream_buffer_size = 8 * 1024; // 8KB streaming buffer
255        let mut operations = 0;
256        let mut position = 0;
257
258        // Allocate a small streaming buffer
259        let mut buffer = vec![0u8; stream_buffer_size];
260
261        while position < data.len() {
262            let copy_size = stream_buffer_size.min(data.len() - position);
263            buffer[..copy_size].copy_from_slice(&data[position..position + copy_size]);
264
265            // Simulate processing the streaming buffer
266            std::hint::black_box(&buffer[..copy_size]);
267
268            operations += 1;
269            position += copy_size;
270
271            // Sample memory every 50 operations for streaming
272            if operations % 50 == 0 {
273                monitor.sample();
274            }
275        }
276
277        monitor.sample();
278        Ok(operations)
279    }

281    /// Chunked memory access pattern - simulates processing file in large chunks with memory management
282    async fn chunked_access(&self, data: &[u8], monitor: &mut MemoryMonitor) -> Result<usize> {
283        let chunk_size = 1024 * 1024; // 1MB chunks
284        let mut operations = 0;
285        let mut position = 0;
286
287        while position < data.len() {
288            let end = (position + chunk_size).min(data.len());
289
290            // Create a separate chunk buffer (simulates allocation/deallocation)
291            let chunk = data[position..end].to_vec();
292            std::hint::black_box(&chunk);
293
294            // Simulate processing time
295            tokio::time::sleep(Duration::from_millis(1)).await;
296
297            operations += 1;
298            position = end;
299
300            monitor.sample();
301
302            // Drop the chunk explicitly to help memory management
303            drop(chunk);
304        }
305
306        monitor.sample();
307        Ok(operations)
308    }

310    /// Analyze memory efficiency across all test results
311    async fn analyze_memory_efficiency(
312        &self,
313        results: &[BenchmarkResult],
314        targets: &PRDTargets,
315    ) -> Result<BenchmarkResult> {
316        let benchmark_name = "Memory_Efficiency_Analysis".to_string();
317
318        // Calculate efficiency metrics
319        let memory_results: Vec<_> = results
320            .iter()
321            .filter(|r| r.benchmark_name.starts_with("Memory_"))
322            .collect();
323
324        if memory_results.is_empty() {
325            return Err(crate::Error::invalid_operation(
326                "No memory results to analyze".to_string(),
327            ));
328        }
329
330        let avg_efficiency: f64 = memory_results
331            .iter()
332            .map(|r| r.memory_efficiency)
333            .sum::<f64>()
334            / memory_results.len() as f64;
335
336        let max_memory_usage: f64 = memory_results
337            .iter()
338            .map(|r| r.memory_usage_mb)
339            .fold(0.0, f64::max);
340
341        let memory_scaling_factor = self.calculate_memory_scaling(&memory_results);
342
343        let meets_target = max_memory_usage <= targets.memory_limit_mb && avg_efficiency >= 1.0; // Should process at least 1MB per 1MB of memory
344
345        let target_comparison = if meets_target {
346            format!(
347                "✅ Memory efficiency acceptable (max: {:.1} MB, avg efficiency: {:.2})",
348                max_memory_usage, avg_efficiency
349            )
350        } else {
351            format!(
352                "❌ Memory efficiency needs improvement (max: {:.1} MB > {}, efficiency: {:.2})",
353                max_memory_usage, targets.memory_limit_mb, avg_efficiency
354            )
355        };
356
357        let mut details = HashMap::new();
358        details.insert("average_efficiency".to_string(), avg_efficiency);
359        details.insert("max_memory_usage_mb".to_string(), max_memory_usage);
360        details.insert("memory_scaling_factor".to_string(), memory_scaling_factor);
361        details.insert("tests_analyzed".to_string(), memory_results.len() as f64);
362
363        println!(
364            "     📊 Memory Efficiency Analysis: {:.2} avg efficiency, {:.1} MB max usage",
365            avg_efficiency, max_memory_usage
366        );
367
368        Ok(BenchmarkResult {
369            benchmark_name,
370            file_size_mb: 0.0, // Aggregate result
371            duration: Duration::from_secs(0),
372            throughput_mb_per_sec: 0.0,
373            memory_usage_mb: max_memory_usage,
374            memory_efficiency: avg_efficiency,
375            compression_ratio: None,
376            operations_per_second: 0.0,
377            meets_prd_target: meets_target,
378            target_comparison,
379            details,
380        })
381    }

383    /// Calculate memory scaling factor (how memory usage scales with file size)
384    fn calculate_memory_scaling(&self, results: &[&BenchmarkResult]) -> f64 {
385        if results.len() < 2 {
386            return 1.0;
387        }
388
389        // Sort by file size
390        let mut sorted_results = results.to_vec();
391        sorted_results.sort_by(|a, b| {
392            a.file_size_mb
393                .partial_cmp(&b.file_size_mb)
394                .unwrap_or(std::cmp::Ordering::Equal)
395        });
396
397        // Calculate average scaling factor
398        let mut scaling_factors = Vec::new();
399
400        for i in 1..sorted_results.len() {
401            let prev = &sorted_results[i - 1];
402            let curr = &sorted_results[i];
403
404            if prev.file_size_mb > 0.0 {
405                let size_ratio = curr.file_size_mb / prev.file_size_mb;
406                let memory_ratio = curr.memory_usage_mb / prev.memory_usage_mb.max(0.1);
407                let scaling_factor = memory_ratio / size_ratio;
408                scaling_factors.push(scaling_factor);
409            }
410        }
411
412        if scaling_factors.is_empty() {
413            1.0
414        } else {
415            scaling_factors.iter().sum::<f64>() / scaling_factors.len() as f64
416        }
417    }

419    /// Detect potential memory leaks by running repeated operations
420    async fn detect_memory_leaks(&self, targets: &PRDTargets) -> Result<BenchmarkResult> {
421        let benchmark_name = "Memory_Leak_Detection".to_string();
422
423        println!("   🔍 Running memory leak detection test...");
424
425        let test_size_mb = 10.0; // Use moderate size for leak detection
426        let iterations = 20; // Run multiple iterations
427        let mut memory_samples = Vec::new();
428        let mut monitor = MemoryMonitor::new();
429
430        let timer = PrecisionTimer::start();
431
432        for i in 0..iterations {
433            // Generate and process test data
434            let test_data = generate_test_data(test_size_mb);
435
436            // Simulate various operations that might leak memory
437            let _processed = self.simulate_processing(&test_data).await?;
438
439            // Sample memory after each iteration
440            monitor.sample();
441            memory_samples.push(monitor.peak_usage_mb());
442
443            // Drop test data explicitly
444            drop(test_data);
445
446            // Force cleanup attempts
447            if i % 5 == 0 {
448                // Give time for cleanup
449                tokio::time::sleep(Duration::from_millis(10)).await;
450            }
451
452            if i % 5 == 0 {
453                println!(
454                    "     Iteration {}/{}: {:.1} MB",
455                    i + 1,
456                    iterations,
457                    memory_samples[i]
458                );
459            }
460        }
461
462        let duration = timer.elapsed_duration();
463
464        // Analyze memory growth pattern
465        let memory_growth = self.analyze_memory_growth(&memory_samples);
466        let final_memory = memory_samples.last().copied().unwrap_or(0.0);
467        let initial_memory = memory_samples.first().copied().unwrap_or(0.0);
468
469        // Leak detection: significant growth over iterations suggests a leak
470        let has_potential_leak = memory_growth > 0.1; // >0.1 MB growth per iteration
471        let exceeds_memory_limit = final_memory > targets.memory_limit_mb;
472
473        let meets_target = !has_potential_leak && !exceeds_memory_limit;
474
475        let target_comparison = if meets_target {
476            format!(
477                "✅ No memory leaks detected ({:.2} MB/iter growth)",
478                memory_growth
479            )
480        } else if has_potential_leak {
481            format!(
482                "⚠️ Potential memory leak detected ({:.2} MB/iter growth)",
483                memory_growth
484            )
485        } else {
486            format!(
487                "❌ Memory usage too high ({:.1} MB > {})",
488                final_memory, targets.memory_limit_mb
489            )
490        };
491
492        let mut details = HashMap::new();
493        details.insert("memory_growth_mb_per_iter".to_string(), memory_growth);
494        details.insert("initial_memory_mb".to_string(), initial_memory);
495        details.insert("final_memory_mb".to_string(), final_memory);
496        details.insert("iterations".to_string(), iterations as f64);
497        details.insert(
498            "potential_leak".to_string(),
499            if has_potential_leak { 1.0 } else { 0.0 },
500        );
501
502        println!(
503            "     🔍 Leak Detection: {:.2} MB/iter growth, final: {:.1} MB",
504            memory_growth, final_memory
505        );
506
507        Ok(BenchmarkResult {
508            benchmark_name,
509            file_size_mb: test_size_mb * iterations as f64,
510            duration,
511            throughput_mb_per_sec: (test_size_mb * iterations as f64) / duration.as_secs_f64(),
512            memory_usage_mb: final_memory,
513            memory_efficiency: (test_size_mb * iterations as f64) / final_memory.max(0.1),
514            compression_ratio: None,
515            operations_per_second: iterations as f64 / duration.as_secs_f64(),
516            meets_prd_target: meets_target,
517            target_comparison,
518            details,
519        })
520    }

522    /// Analyze memory growth pattern across iterations
523    fn analyze_memory_growth(&self, samples: &[f64]) -> f64 {
524        if samples.len() < 2 {
525            return 0.0;
526        }
527
528        // Calculate linear regression to find growth trend
529        let n = samples.len() as f64;
530        let sum_x: f64 = (0..samples.len()).map(|i| i as f64).sum();
531        let sum_y: f64 = samples.iter().sum();
532        let sum_xy_product: f64 = samples.iter().enumerate().map(|(i, &y)| i as f64 * y).sum();
533        let sum_x2: f64 = (0..samples.len()).map(|i| (i as f64).powi(2)).sum();
534
535        // Linear regression slope (growth rate)
536
537        (n * sum_xy_product - sum_x * sum_y) / (n * sum_x2 - sum_x.powi(2))
538    }

541    async fn simulate_processing(&self, data: &[u8]) -> Result<Vec<u8>> {
542        // Simulate various operations that might leak memory
543        let mut result = Vec::new();
544
545        // 1. Buffer allocations and deallocations
546        for chunk in data.chunks(1024) {
547            let mut buffer = chunk.to_vec();
548            buffer.reverse(); // Some processing
549            result.extend_from_slice(&buffer);
550        }
551
552        // 2. String operations
553        let _string_data = String::from_utf8_lossy(data);
554
555        // 3. HashMap operations
556        let mut map = HashMap::new();
557        for (i, &byte) in data.iter().enumerate().take(100) {
558            map.insert(i, byte);
559        }
560
561        // 4. Nested allocations
562        let nested: Vec<Vec<u8>> = data.chunks(1024).map(|chunk| chunk.to_vec()).collect();
563        let _flattened: Vec<u8> = nested.into_iter().flatten().collect();
564
565        Ok(result)
566    }

598    #[cfg(target_os = "linux")]
599    fn get_memory_linux(&self) -> f64 {
600        use std::fs;
601
602        if let Ok(meminfo) = fs::read_to_string("/proc/meminfo") {
603            for line in meminfo.lines() {
604                if line.starts_with("MemAvailable:") {
605                    let parts: Vec<&str> = line.split_whitespace().collect();
606                    if parts.len() >= 2 {
607                        if let Ok(kb) = parts[1].parse::<f64>() {
608                            return kb / 1024.0 / 1024.0; // KB to GB
609                        }
610                    }
611                }
612            }
613        }
614        8.0 // Default fallback
615    }
