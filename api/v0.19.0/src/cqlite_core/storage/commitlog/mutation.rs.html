1//! Decoded CommitLog mutation representation + the mutation-body decoder
2//! (issue #2389).
3//!
4//! A record body (validated by [`super::frame`]) is a serialized Cassandra
5//! `Mutation` in **messaging** form (`Mutation.serializer` /
6//! `UnfilteredRowIteratorSerializer` with `isForSSTable() == false`). The
7//! grammar below is verified byte-for-byte against real Cassandra 5.0.2 and
8//! 5.0.8 segments:
9//!
10//! ```text
11//! mutation:  uvint numUpdates, then numUpdates × partition-update
12//! update:    tableId(16), UnfilteredRowIterator
13//! iterator:  writeWithVIntLength(partitionKey), iterFlags(1),
14//!            EncodingStats(3 uvints), [static columns], regular columns,
15//!            [partition deletion], [static row], [row estimate uvint],
16//!            unfiltereds… END_OF_PARTITION(0x01)
17//! row:       flags(1) [ext flags], clustering values, [liveness], [deletion],
18//!            [column subset], cells…
19//! cell:      flags(1), [ts delta], [localDeletion], [ttl], [path], [value]
20//! ```
21//!
22//! This module fully decodes the common insert path (simple columns, all
23//! columns present, no clustering/static/complex/subset/deletion). Constructs
24//! it does not fully model (clustering columns — the `ClusteringPrefix`
25//! presence-header is not decoded, see `decode_rows`'s bail — static rows,
26//! complex/collection columns, column subsets, range tombstone markers,
27//! row/partition deletions) are surfaced honestly: the affected partition's
28//! [`PartitionUpdate::rows_decoded`] is set `false` rather than guessing byte
29//! offsets — each mutation record is independently CRC-framed and
30//! length-delimited, so bailing on one body never disturbs the others.
31

37// ---- UnfilteredRowIterator flags -----------------------------------------
38const ITER_IS_EMPTY: u8 = 0x01;
39const ITER_HAS_PARTITION_DELETION: u8 = 0x04;
40const ITER_HAS_STATIC_ROW: u8 = 0x08;
41const ITER_HAS_ROW_ESTIMATE: u8 = 0x10;
42
43// ---- Unfiltered (row / marker) flags -------------------------------------
44const ROW_END_OF_PARTITION: u8 = 0x01;
45const ROW_IS_MARKER: u8 = 0x02;
46const ROW_HAS_TIMESTAMP: u8 = 0x04;
47const ROW_HAS_TTL: u8 = 0x08;
48const ROW_HAS_DELETION: u8 = 0x10;
49const ROW_HAS_ALL_COLUMNS: u8 = 0x20;
50const ROW_HAS_COMPLEX_DELETION: u8 = 0x40;
51const ROW_EXTENSION_FLAG: u8 = 0x80;

53// ---- Cell flags -----------------------------------------------------------
54const CELL_IS_DELETED: u8 = 0x01;
55const CELL_IS_EXPIRING: u8 = 0x02;
56const CELL_HAS_EMPTY_VALUE: u8 = 0x04;
57const CELL_USE_ROW_TIMESTAMP: u8 = 0x08;
58const CELL_USE_ROW_TTL: u8 = 0x10;

60/// A decoded CommitLog mutation: one or more partition updates.
61#[derive(Debug, Clone, PartialEq, Eq)]
62pub struct Mutation {
63    /// The partition updates carried by this mutation (one per table touched).
64    /// May be a PREFIX of the mutation's true update count — see
65    /// [`Mutation::updates_complete`].
66    pub updates: Vec<PartitionUpdate>,
67    /// `false` when fewer updates were represented than the mutation declared.
68    /// A final update whose body is only partially decoded still counts as
69    /// represented; an unmodeled partial update makes this false only when
70    /// additional declared updates remain hidden after it.
71    pub updates_complete: bool,
72}

74/// A single-table partition update within a mutation.
75#[derive(Debug, Clone, PartialEq, Eq)]
76pub struct PartitionUpdate {
77    /// 16-byte table id (Cassandra `TableId`).
78    pub table_id: [u8; 16],
79    /// Raw partition-key bytes (schema-typed decoding is the caller's choice).
80    pub partition_key: Vec<u8>,
81    /// Whether the regular column-name block was successfully parsed.
82    /// `true` includes a parsed zero-name block; `false` means no such block
83    /// was read (including the empty-partition fast path and static-row bailout).
84    pub columns_read: bool,
85    /// Regular column names from the messaging header, in wire order.
86    pub column_names: Vec<String>,
87    /// Whether the partition carries a partition-level deletion.
88    pub has_partition_deletion: bool,
89    /// Fully-decoded rows. Empty when `rows_decoded` is `false`.
90    pub rows: Vec<DecodedRow>,
91    /// `true` when every row/cell was decoded; `false` when a schema was
92    /// unavailable or the partition used a construct this decoder does not
93    /// model (surfaced honestly rather than guessed).
94    pub rows_decoded: bool,
95    /// The partition's `EncodingStats.minTimestamp` (microseconds): the base every row/cell
96    /// timestamp delta is added to. `None` when the update carries no `EncodingStats`
97    /// (an empty-partition carrier) — never a fabricated 0.
98    pub min_timestamp: Option<i64>,
99    /// Why `rows` could not be decoded (`rows_decoded == false`), as a stable human string
100    /// naming the unmodeled construct; `None` when rows were decoded.
101    pub undecoded_cause: Option<&'static str>,
102}
103
104impl PartitionUpdate {
105    /// The table id formatted as a canonical UUID string.
106    pub fn table_id_uuid(&self) -> String {
107        format_table_id(&self.table_id)
108    }
109}

111/// A decoded row within a partition update.
112#[derive(Debug, Clone, PartialEq, Eq)]
113pub struct DecodedRow {
114    /// Clustering-column values in order (raw bytes; empty for a no-clustering
115    /// table).
116    pub clustering: Vec<Vec<u8>>,
117    /// Decoded cells for the row's present columns.
118    pub cells: Vec<DecodedCell>,
119    /// Whether the row itself is a row deletion (tombstone).
120    pub is_row_deletion: bool,
121    /// Row liveness timestamp (`min_timestamp + delta`), `None` when the row carries no
122    /// row-level timestamp (`ROW_HAS_TIMESTAMP` unset).
123    pub writetime: Option<i64>,
124    /// The row deletion's `markedForDeleteAt` (`min_timestamp + delta`) when the row is a
125    /// deletion, else `None`.
126    pub deletion_timestamp: Option<i64>,
127    /// Whether the row's primary-key liveness marker carries a TTL (`ROW_HAS_TTL`), i.e. it is
128    /// an expiring marker. Needed to reconcile equal-timestamp markers exactly.
129    pub liveness_expiring: bool,
130}

132/// A decoded cell (one column value within a row).
133#[derive(Debug, Clone, PartialEq, Eq)]
134pub struct DecodedCell {
135    /// Column name.
136    pub column: String,
137    /// Raw value bytes, or `None` for a tombstone / empty-value cell.
138    pub value: Option<Vec<u8>>,
139    /// Whether the cell is a tombstone (deleted).
140    pub deleted: bool,
141    /// Cell timestamp (`min_timestamp + delta`, or the row's when `CELL_USE_ROW_TIMESTAMP`);
142    /// `None` only when the cell inherits from a row that has no timestamp.
143    pub writetime: Option<i64>,
144    /// Whether the cell is expiring (`CELL_IS_EXPIRING`): it carries a TTL and a local
145    /// deletion time that this decoder does not retain.
146    pub expiring: bool,
147}

149/// Decode a mutation record body into a [`Mutation`].
150///
151/// `schemas` supplies per-table-id schemas for full clustering/cell decode; a
152/// table with no schema decodes structurally (table id + partition key + column
153/// names) with `rows_decoded == false`.
154///
155/// # Errors
156/// [`Error::CorruptCommitLogFrame`] if the body runs out of bytes mid-field
157/// (should not happen for a CRC-validated record, but is handled defensively —
158/// never panics).
159pub fn decode_mutation(body: &[u8], schemas: &SchemaSet) -> Result<Mutation> {
160    let mut c = Cursor::new(body);
161    let num_updates = c.uvint()?;
162    // Guard against a maliciously huge declared count spinning the loop below:
163    // each update needs at least a few bytes (a 16-byte table id alone), so a
164    // count exceeding the body length is already provably impossible — reject
165    // it outright rather than relying on each iteration's short-read failure
166    // to terminate a `for _ in 0..num_updates` driven by an untrusted u64
167    // (roborev finding — a hang risk even though it fails fast today, review-
168    // first pass).
169    if num_updates > body.len() as u64 {
170        return Err(Error::CorruptCommitLogFrame(format!(
171            "mutation declares {num_updates} updates, impossible for a {}-byte body",
172            body.len()
173        )));
174    }
175    let mut updates = Vec::with_capacity(num_updates.min(1024) as usize);
176    let mut updates_complete = true;
177    for update_index in 0..num_updates {
178        // Each update is decoded either fully (cursor left at the next update) or
179        // partially (structural fields only — no schema, or an unmodeled
180        // construct). A partial decode cannot locate a following update's
181        // offset, so stop. The current update is still represented in `updates`;
182        // completeness is false only if declared updates remain hidden after it.
183        let (update, consumed_fully) = decode_partition_update(&mut c, schemas)?;
184        updates.push(update);
185        if !consumed_fully {
186            updates_complete = update_index + 1 == num_updates;
187            break;
188        }
189    }
190    Ok(Mutation {
191        updates,
192        updates_complete,
193    })
194}

196fn decode_partition_update(
197    c: &mut Cursor<'_>,
198    schemas: &SchemaSet,
199) -> Result<(PartitionUpdate, bool)> {
200    let table_id = c.take_array16()?;
201    let pk_len = c.uvint()? as usize;
202    let partition_key = c.take(pk_len)?.to_vec();
203    let iter_flags = c.u8()?;
204
205    let mut update = PartitionUpdate {
206        table_id,
207        partition_key,
208        column_names: Vec::new(),
209        columns_read: false,
210        // Set from the authoritative iter_flags bit immediately, before any
211        // early return — has_partition_deletion is a structural fact readable
212        // right here, independent of rows_decoded, so it must not silently
213        // default to false on the static-row/empty-partition paths below.
214        // Those early returns previously left it false even when the
215        // partition genuinely carried a deletion, reporting an
216        // authoritative-looking "not deleted" to callers (CLI JSON, library
217        // consumers) that was simply never checked (roborev finding, review-
218        // first pass).
219        has_partition_deletion: iter_flags & ITER_HAS_PARTITION_DELETION != 0,
220        rows: Vec::new(),
221        rows_decoded: false,
222        min_timestamp: None,
223        undecoded_cause: Some("rows not decoded"),
224    };
225
226    if iter_flags & ITER_IS_EMPTY != 0 {
227        // Empty partition (e.g. a partition-only delete carrier). The update body
228        // is exactly tableId + pk + flags — fully consumed.
229        update.rows_decoded = true;
230        update.undecoded_cause = None;
231        return Ok((update, true));
232    }
233
234    // EncodingStats: minTimestamp, minLocalDeletionTime, minTTL (all uvints).
235    // Verified against cassandra-5.0.8 `EncodingStats.Serializer`: the wire value is
236    // `minTimestamp - TIMESTAMP_EPOCH` (2015-09-22T00:00:00Z in microseconds).
237    let min_ts_raw = c.uvint()?;
238    let _min_ldt = c.uvint()?;
239    let _min_ttl = c.uvint()?;
240    // cassandra-5.0.8 `EncodingStats.Serializer.serialize` writes
241    // `out.writeUnsignedVInt(stats.minTimestamp - TIMESTAMP_EPOCH)` and `deserialize` reads
242    // `in.readUnsignedVInt() + TIMESTAMP_EPOCH` — Java `long` arithmetic. A pre-2015-09-22
243    // timestamp therefore has a NEGATIVE delta, written as its two's-complement 64-bit pattern
244    // (a 9-byte vint); reading those bits as `i64` and adding with wrap-around reproduces
245    // Cassandra exactly. (`i64::try_from` wrongly rejected every such valid timestamp.)
246    let min_timestamp = Some((min_ts_raw as i64).wrapping_add(TIMESTAMP_EPOCH_MICROS));
247    update.min_timestamp = min_timestamp;
248
249    let has_static = iter_flags & ITER_HAS_STATIC_ROW != 0;
250    if has_static {
251        // Static rows: NOT modeled. Whether Cassandra's
252        // UnfilteredRowIteratorSerializer writes a static-columns block gated
253        // on this per-partition iterator flag (the prior assumption here), or
254        // unconditionally based on the table's static-column set regardless
255        // of whether THIS iterator instance carries a static row, is
256        // unverified against real Cassandra 5.0 source (unavailable in this
257        // environment — see the appendix-h doc). Bail BEFORE reading any
258        // columns block at all: under either assumption, guessing wrong here
259        // would misalign the cursor and — worse — would leave
260        // update.column_names populated with misparsed bytes that then get
261        // surfaced to callers (including the CLI's structural-only output)
262        // as if authoritative, even though rows_decoded is false (roborev
263        // finding, review-first pass).
264        update.undecoded_cause = Some("static row not decoded");
265        return Ok((update, false));
266    }
267    update.column_names = read_column_names(c)?;
268    update.columns_read = true;
269
270    // Constructs we do not fully model: bail structurally (honest, no guessing).
271    // The cursor is left mid-body, so this update is NOT fully consumed.
272    if update.has_partition_deletion {
273        update.undecoded_cause = Some("partition-deletion timestamp not decoded");
274        return Ok((update, false));
275    }
276
277    if iter_flags & ITER_HAS_ROW_ESTIMATE != 0 {
278        let _row_estimate = c.uvint()?;
279    }
280
281    let schema = match schemas.get(&table_id) {
282        Some(s) => s,
283        None => {
284            update.undecoded_cause = Some("table schema not supplied");
285            return Ok((update, false)); // structural-only decode
286        }
287    };
288
289    let Some(base_ts) = min_timestamp else {
290        update.undecoded_cause = Some("EncodingStats minTimestamp not decoded");
291        return Ok((update, false));
292    };
293    match decode_rows(c, schema, &update.column_names, base_ts) {
294        Ok(rows) => {
295            update.rows = rows;
296            update.rows_decoded = true;
297            update.undecoded_cause = None;
298            // decode_rows consumed through END_OF_PARTITION.
299            Ok((update, true))
300        }
301        Err(RowBail::Malformed(cause)) => Err(Error::CorruptCommitLogFrame(format!(
302            "malformed mutation body for table {}: {cause}",
303            update.table_id_uuid()
304        ))),
305        Err(RowBail::Unsupported(cause)) => {
306            // A construct beyond this decoder's model was hit; keep the
307            // structural fields and report rows as not decoded.
308            update.rows.clear();
309            update.rows_decoded = false;
310            update.undecoded_cause = Some(cause);
311            Ok((update, false))
312        }
313    }
314}

316/// Why the row decode stopped: a shape the decoder deliberately does not model (surfaced as an
317/// incomplete, `unmeasured` update) versus a structurally MALFORMED body (truncated mid-row, a
318/// timestamp delta out of range) that must be a commit-log error, never a quiet success.
319enum RowBail {
320    Unsupported(&'static str),
321    Malformed(&'static str),
322}
323type RowResult<T> = std::result::Result<T, RowBail>;
324
325fn decode_rows(
326    c: &mut Cursor<'_>,
327    schema: &CommitLogSchema,
328    column_names: &[String],
329    base_ts: i64,
330) -> RowResult<Vec<DecodedRow>> {
331    // Clustered tables: NOT modeled. Cassandra's `ClusteringPrefix.Serializer`
332    // writes a `writeUnsignedVInt` PRESENCE HEADER (2 bits per clustering
333    // column, batched per 32 columns: 0=present, 1=empty, 2=null) before the
334    // present/non-empty values — see the SSTable reader's
335    // `row_decoder/row_framing.rs` (issue #213) for the canonical decode of
336    // this exact header. A naive per-column value loop (no header) misreads
337    // the header bytes as the first value's bytes, silently misaligning every
338    // subsequent field. Bail honestly rather than guess, matching this
339    // module's existing posture for other unmodeled constructs (range
340    // tombstone markers, partition/static-row deletions) — decoding it is a
341    // follow-up, not a v1 requirement.
342    if !schema.clustering.is_empty() {
343        return Err(RowBail::Unsupported("clustered table not decoded"));
344    }
345    // Complex columns (collection/tuple/UDT/vector): NOT modeled. Cassandra
346    // serializes a complex column as an entirely different wire shape (an
347    // optional complex-deletion time, then a count-prefixed set of
348    // (cell-path, cell) pairs) — not the single simple Cell this module's
349    // read_cell decodes. The module doc previously claimed this was
350    // "surfaced honestly", but nothing actually checked for it: a schema
351    // with e.g. a `list<text>` column would silently misalign every
352    // subsequent row while still reporting rows_decoded == true (roborev
353    // finding, review-first pass). Checked once here (a schema-level
354    // property, not a per-row one) rather than per column per row.
355    if column_names
356        .iter()
357        .any(|name| !schema.column_type(name).is_some_and(is_simple_scalar_type))
358    {
359        return Err(RowBail::Unsupported("complex column not decoded"));
360    }
361    let mut rows = Vec::new();
362    loop {
363        let flags = c.u8().map_err(|_| short_row())?;
364        if flags & ROW_END_OF_PARTITION != 0 {
365            break;
366        }
367        if flags & ROW_IS_MARKER != 0 {
368            // Range tombstone marker — not modeled.
369            return Err(RowBail::Unsupported("range tombstone marker not decoded"));
370        }
371        if flags & ROW_EXTENSION_FLAG != 0 {
372            let _ext = c.u8().map_err(|_| short_row())?;
373        }
374
375        // Clustering values: schema.clustering is guaranteed empty at this
376        // point (decode_rows bails to structural-only above whenever it
377        // isn't), so there is nothing to read here. No dead read-loop is kept
378        // around it — re-enabling clustering support means deliberately
379        // writing the ClusteringPrefix presence-header decode (see the bail
380        // site's comment), not just deleting the guard (roborev finding,
381        // review-first pass).
382        let clustering = Vec::new();
383
384        // Liveness.
385        let mut row_writetime = None;
386        if flags & ROW_HAS_TIMESTAMP != 0 {
387            row_writetime = Some(add_delta(base_ts, c.uvint().map_err(|_| short_row())?)?);
388        }
389        let liveness_expiring = flags & ROW_HAS_TTL != 0;
390        if liveness_expiring {
391            let _ttl = c.uvint().map_err(|_| short_row())?;
392            let _ldt = c.uvint().map_err(|_| short_row())?;
393        }
394        let mut is_row_deletion = false;
395        let mut deletion_timestamp = None;
396        if flags & ROW_HAS_DELETION != 0 {
397            // Row deletion time (two uvints in the messaging delta encoding).
398            let mfda = c.uvint().map_err(|_| short_row())?;
399            let _ldt = c.uvint().map_err(|_| short_row())?;
400            deletion_timestamp = Some(add_delta(base_ts, mfda)?);
401            is_row_deletion = true;
402        }
403        if flags & ROW_HAS_COMPLEX_DELETION != 0 {
404            return Err(RowBail::Unsupported("complex column not decoded"));
405        }
406
407        // Column subset — not modeled; only the all-columns case is decoded.
408        if flags & ROW_HAS_ALL_COLUMNS == 0 {
409            return Err(RowBail::Unsupported("column-subset row not decoded"));
410        }
411
412        // One simple cell per regular column, in header order.
413        let mut cells = Vec::with_capacity(column_names.len());
414        for name in column_names {
415            let type_name = schema
416                .column_type(name)
417                .ok_or(RowBail::Unsupported("column type not in schema"))?;
418            let cell = read_cell(c, name, type_name, base_ts, row_writetime)?;
419            cells.push(cell);
420        }
421
422        rows.push(DecodedRow {
423            clustering,
424            cells,
425            is_row_deletion,
426            writetime: row_writetime,
427            deletion_timestamp,
428            liveness_expiring,
429        });
430    }
431    Ok(rows)
432}

434fn read_cell(
435    c: &mut Cursor<'_>,
436    column: &str,
437    type_name: &str,
438    base_ts: i64,
439    row_writetime: Option<i64>,
440) -> RowResult<DecodedCell> {
441    let flags = c.u8().map_err(|_| short_row())?;
442    let deleted = flags & CELL_IS_DELETED != 0;
443    let expiring = flags & CELL_IS_EXPIRING != 0;
444    let empty = flags & CELL_HAS_EMPTY_VALUE != 0;
445
446    let writetime = if flags & CELL_USE_ROW_TIMESTAMP == 0 {
447        Some(add_delta(base_ts, c.uvint().map_err(|_| short_row())?)?)
448    } else {
449        row_writetime
450    };
451    let use_row_ttl = flags & CELL_USE_ROW_TTL != 0;
452    if (deleted || expiring) && !use_row_ttl {
453        let _ldt = c.uvint().map_err(|_| short_row())?;
454    }
455    if expiring && !use_row_ttl {
456        let _ttl = c.uvint().map_err(|_| short_row())?;
457    }
458
459    // A value is present unless the cell is a tombstone or explicitly empty.
460    let value = if deleted || empty {
461        None
462    } else {
463        Some(read_typed_value(c, type_name)?)
464    };
465
466    Ok(DecodedCell {
467        column: column.to_string(),
468        value,
469        deleted,
470        writetime,
471        expiring,
472    })
473}

475/// `EncodingStats.TIMESTAMP_EPOCH` (cassandra-5.0.8 `EncodingStats.java`): 2015-09-22T00:00:00Z
476/// in microseconds. The serialized `minTimestamp` is relative to it.
477pub(crate) const TIMESTAMP_EPOCH_MICROS: i64 = 1_442_880_000_000_000;
478
479/// `base + delta` with overflow treated as an unmodeled construct (never a wrapped value).
480fn add_delta(base: i64, delta: u64) -> RowResult<i64> {
481    i64::try_from(delta)
482        .ok()
483        .and_then(|d| base.checked_add(d))
484        .ok_or(RowBail::Malformed("timestamp delta overflows i64"))
485}
486
487fn short_row() -> RowBail {
488    RowBail::Malformed("mutation body ended mid-row")
489}

491/// Read one CQL value: fixed-length types have no length prefix; variable
492/// types are unsigned-vint-length-prefixed (`AbstractType.writeValue`).
493fn read_typed_value(c: &mut Cursor<'_>, type_name: &str) -> RowResult<Vec<u8>> {
494    match cql_fixed_len(type_name) {
495        Some(n) => c.take(n).map(|s| s.to_vec()).map_err(|_| short_row()),
496        None => {
497            let len = c.uvint().map_err(|_| short_row())? as usize;
498            c.take(len).map(|s| s.to_vec()).map_err(|_| short_row())
499        }
500    }
501}

503/// Read a `Columns` block: `uvint count`, then `count` × vint-length-prefixed
504/// column name bytes.
505fn read_column_names(c: &mut Cursor<'_>) -> Result<Vec<String>> {
506    let count = c.uvint()?;
507    // Guard the declared count against the cursor's remaining bytes before
508    // spinning the loop below `count` times on an untrusted u64: each column
509    // name needs at least its own 1-byte length vint, so a count exceeding the
510    // remaining bytes is provably impossible. Without this, a CRC-valid body of
511    // mostly 0x00 bytes (each decoding as a zero-length name) could turn a
512    // ~128MB body into ~128M Vec<String> entries (~3GB), defeating the <128MB
513    // memory target — the same class as decode_mutation's num_updates guard.
514    let remaining = c.remaining() as u64;
515    if count > remaining {
516        return Err(Error::CorruptCommitLogFrame(format!(
517            "column-names block declares {count} names, impossible for {remaining} remaining bytes"
518        )));
519    }
520    let mut names = Vec::with_capacity(count.min(4096) as usize);
521    for _ in 0..count {
522        let len = c.uvint()? as usize;
523        let bytes = c.take(len)?;
524        names.push(String::from_utf8_lossy(bytes).into_owned());
525    }
526    Ok(names)
527}

529/// A minimal forward-only byte cursor with unsigned-vint support.
530struct Cursor<'a> {
531    bytes: &'a [u8],
532    pos: usize,
533}
534
535impl<'a> Cursor<'a> {
536    fn new(bytes: &'a [u8]) -> Self {
537        Self { bytes, pos: 0 }
538    }
539
540    /// Bytes not yet consumed — an upper bound for any count-prefixed block, so
541    /// a hostile declared count can be rejected before it drives a loop.
542    fn remaining(&self) -> usize {
543        self.bytes.len().saturating_sub(self.pos)
544    }
545
546    fn u8(&mut self) -> Result<u8> {
547        let b = *self.bytes.get(self.pos).ok_or_else(short)?;
548        self.pos += 1;
549        Ok(b)
550    }
551
552    fn take(&mut self, n: usize) -> Result<&'a [u8]> {
553        let end = self.pos.checked_add(n).ok_or_else(short)?;
554        if end > self.bytes.len() {
555            return Err(short());
556        }
557        let out = &self.bytes[self.pos..end];
558        self.pos = end;
559        Ok(out)
560    }
561
562    fn take_array16(&mut self) -> Result<[u8; 16]> {
563        let s = self.take(16)?;
564        let mut a = [0u8; 16];
565        a.copy_from_slice(s);
566        Ok(a)
567    }
568
569    /// Decode a Cassandra unsigned VInt (`writeUnsignedVInt`).
570    ///
571    /// Delegates to [`crate::parser::vint::decode_unsigned`] — the ONE canonical
572    /// read-side VInt decoder (Issue #1638, Epic J/J4), already used by the
573    /// SSTable reader and already fuzzed (`fuzz/fuzz_targets/fuzz_vint.rs`) —
574    /// rather than maintaining a second, independent bit-assembly here. (An
575    /// earlier local reimplementation had a 9-byte shift-overflow bug that the
576    /// canonical decoder already special-cases correctly.)
577    fn uvint(&mut self) -> Result<u64> {
578        let (value, consumed) =
579            crate::parser::vint::decode_unsigned(&self.bytes[self.pos..]).map_err(|_| short())?;
580        self.pos += consumed;
581        Ok(value)
582    }
583}
584
585fn short() -> Error {
586    Error::CorruptCommitLogFrame("mutation body ended mid-field".to_string())
587}
