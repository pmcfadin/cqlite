1//! The one closed rule for "what is this CommitLog mutation's writetime" (issue #4205, D4).
2//!
3//! A [`PartitionUpdate`]'s writetime is the MAXIMUM timestamp it contributes: every row
4//! liveness timestamp, every cell timestamp, and every row-deletion `markedForDeleteAt`.
5//! Anything the decoder did not model (a partition deletion, a clustered table, a complex
6//! column, a missing schema, …) is [`Writetime::Unmeasured`] naming the cause — never a
7//! guessed or zero value (#4159 doctrine).
8
9use crate::storage::commitlog::mutation::PartitionUpdate;
10
11/// A mutation's writetime, or the stated reason it could not be measured.
12#[derive(Debug, Clone, Copy, PartialEq, Eq)]
13pub enum Writetime {
14    /// Microseconds since the Unix epoch (Cassandra's cell timestamp unit).
15    Known(i64),
16    /// The decoder does not model this mutation's shape; the cause is stable text.
17    Unmeasured(&'static str),
18}
19
20/// Compute `update`'s writetime. See the module doc for the rule.
21pub fn mutation_writetime(update: &PartitionUpdate) -> Writetime {
22    if !update.rows_decoded {
23        return Writetime::Unmeasured(update.undecoded_cause.unwrap_or("rows not decoded"));
24    }
25    let mut max: Option<i64> = None;
26    let mut bump = |t: Option<i64>| {
27        if let Some(t) = t {
28            max = Some(max.map_or(t, |m| m.max(t)));
29        }
30    };
31    for row in &update.rows {
32        bump(row.writetime);
33        bump(row.deletion_timestamp);
34        for cell in &row.cells {
35            bump(cell.writetime);
36        }
37    }
38    match max {
39        Some(t) => Writetime::Known(t),
40        None => Writetime::Unmeasured("mutation carries no timestamped rows"),
41    }
42}

44/// One fact a CommitLog mutation must have in the SSTables for the mutation to count as flushed
45/// (issue #4205, roborev job 299 M1): the SSTables must hold the same column at a timestamp
46/// STRICTLY above the mutation's, or an EXACTLY equal cell (see the `state` fields), or a
47/// covering deletion at or above it. At an equal timestamp Cassandra resolves cells by
48/// tombstone/expiring state, local deletion time, TTL and then value bytes
49/// (`Cells.resolveRegular`, `LivenessInfo.supersedes`, cassandra-5.0.8), so an equal timestamp
50/// alone proves nothing. Only meaningful for a table without clustering columns (the only shape
51/// the decoder models).
52#[derive(Debug, Clone, PartialEq, Eq)]
53pub enum Need {
54    /// A cell (or cell tombstone) of `column` written at `timestamp`.
55    Cell {
56        /// Column name.
57        column: String,
58        /// Cell timestamp.
59        timestamp: i64,
60        /// `Some(value bytes)` for a live, non-expiring cell; `None` for a tombstone or an
61        /// expiring cell (whose TTL / local deletion time the decoder does not retain), which
62        /// can never be proven equal to an SSTable cell at the same timestamp.
63        value: Option<Vec<u8>>,
64    },
65    /// The row's primary-key liveness marker written at this timestamp.
66    Liveness {
67        /// Marker timestamp.
68        timestamp: i64,
69        /// Whether the marker carries a TTL (an expiring marker cannot be proven equal).
70        expiring: bool,
71    },
72    /// A row deletion with this `markedForDeleteAt`.
73    Deletion(i64),
74}
75
76/// The [`Need`]s of `update`. `Err(cause)` when the update has a shape the coverage proof does not
77/// model (undecoded rows, a clustered row, a cell without a timestamp): the verdict is then
78/// unmeasured with that cause, never a guess.
79pub fn mutation_needs(update: &PartitionUpdate) -> Result<Vec<Need>, &'static str> {
80    if !update.rows_decoded {
81        return Err(update.undecoded_cause.unwrap_or("rows not decoded"));
82    }
83    let mut needs = Vec::new();
84    for row in &update.rows {
85        if !row.clustering.is_empty() {
86            return Err("clustered row: SSTable cell coverage is not modeled");
87        }
88        if let Some(t) = row.writetime {
89            needs.push(Need::Liveness {
90                timestamp: t,
91                expiring: row.liveness_expiring,
92            });
93        }
94        if let Some(t) = row.deletion_timestamp {
95            needs.push(Need::Deletion(t));
96        }
97        for cell in &row.cells {
98            let Some(timestamp) = cell.writetime else {
99                return Err("a cell carries no timestamp");
100            };
101            needs.push(Need::Cell {
102                column: cell.column.clone(),
103                timestamp,
104                value: if cell.deleted || cell.expiring {
105                    None
106                } else {
107                    Some(cell.value.clone().unwrap_or_default())
108                },
109            });
110        }
111    }
112    Ok(needs)
113}
