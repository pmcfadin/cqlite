12/// A closed classification for a cell version's reconciliation outcome.
13#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
14#[serde(rename_all = "kebab-case")]
15pub enum Verdict {
16    Winner,
17    ShadowedByTimestamp,
18    ShadowedByTombstone(TombstoneKind),
19    Expired,
20    Purgeable,
21    DroppedColumn,
22}

26#[serde(rename_all = "lowercase")]
27pub enum TombstoneKind {

35/// The `Cells.resolveRegular` rule that decided a cell-vs-cell reconciliation
36/// (cassandra-5.0.8 `db/rows/Cells.java:79-129`), in evaluation order (issue #4193 review M2).
37/// Reported by [`DecidedBy::Winner`] and the cell case of [`DecidedBy::Tombstone`]; it is the
38/// rule `reconcile_rules::resolve_cell` applied, never a re-derivation.
39#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
40#[serde(rename_all = "snake_case")]
41pub enum ReconcileRule {
42    /// The versions' write timestamps differ: the strictly newer one wins.
43    Timestamp,
44    /// Equal timestamps: a tombstone beats a live cell.
45    TombstoneOverLive,
46    /// Equal timestamps: an expiring (TTL) cell beats a live cell (CASSANDRA-14592).
47    ExpiringOverLive,
48    /// Equal timestamps: a tombstone beats an expiring cell.
49    TombstoneOverExpiring,
50    /// Equal timestamps, both tombstones or both expiring: the later `localDeletionTime` wins.
51    LaterLocalDeletionTime,
52    /// Equal timestamps and `localDeletionTime`, both expiring: the lower TTL wins.
53    LowerTtl,
54    /// Everything else tied: the greater serialized value bytes win (a full tie keeps the
55    /// first-seen cell, indistinguishable from the other).
56    GreaterValue,
57}
58
59/// The fact that decided a cell version's verdict.
60#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
61#[serde(rename_all = "snake_case", tag = "decision")]
62pub enum DecidedBy {
63    Winner {
64        run_index: usize,
65        writetime: i64,
66        /// The `Cells.resolveRegular` rule that picked this winner over the shadowed version.
67        /// `None` where no cell-vs-cell comparison happened (a surviving cell).
68        #[serde(default, skip_serializing_if = "Option::is_none")]
69        rule: Option<ReconcileRule>,
70    },
71    Tombstone {
72        kind: TombstoneKind,
73        run_index: usize,
74        deletion_time: i64,
75        local_deletion_time: i32,
76        droppable_at_now: bool,
77        /// Set only when the tombstone is a CELL that beat another cell version at an equal
78        /// timestamp (`TombstoneOverLive` / `TombstoneOverExpiring`); `None` for partition,
79        /// range, row and complex-column shadowing.
80        #[serde(default, skip_serializing_if = "Option::is_none")]
81        rule: Option<ReconcileRule>,
82    },
83    DropTime(i64),
84    Expiry {
85        expires_at: i64,
86        now: i64,
87    },
88    GcGrace {
89        ldt: i32,
90        gc_before: i64,
91        now: i64,
92    },
93    None,
94}

96/// One cell version observed by reconciliation.
97#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
98pub struct CellDecision {
99    pub run_index: usize,
100    pub clustering: Option<ClusteringKey>,
101    pub column: String,
102    /// The element path of a non-frozen collection cell (`CellData::cell_path`); `None` for a
103    /// simple cell or a frozen value. Carried for `cqlite diff` (#4201); `explain` does not render it.
104    #[serde(default, skip_serializing_if = "Option::is_none")]
105    pub cell_path: Option<Vec<u8>>,
106    pub value: Option<Value>,
107    pub writetime: i64,
108    /// The CQL TTL is rendered as a signed value in the public trace schema,
109    /// matching the accepted OpenSpec shape.  The on-disk cell model stores
110    /// this as `u32`; the merge boundary performs the checked-width conversion
111    /// before emitting the event.
112    pub ttl: Option<i32>,
113    pub expires_at: Option<i64>,
114    pub verdict: Verdict,
115    pub decided_by: DecidedBy,
116}

118/// One tombstone marker observed while reconciling a partition.
119#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
120pub struct TombstoneRecord {
121    pub kind: TombstoneKind,
122    pub run_index: usize,
123    pub clustering: Option<ClusteringKey>,
124    pub column: Option<String>,
125    /// The element path of a CELL tombstone on a collection element (`cell_path`), else `None`.

128    pub deletion_time: i64,
129    pub local_deletion_time: i32,
130    pub range_start: Option<ClusteringBound>,
131    pub range_end: Option<ClusteringBound>,
132    pub droppable_at_now: bool,
133}

144/// Static-dispatch destination for reconciliation observations.
145pub trait TraceSink {
146    /// Whether constructing a trace event can have an observable effect.
147    ///
148    /// This associated constant lets the merge skip trace-only cloning and
149    /// allocation for [`NoTrace`] before constructing an event.
150    const ENABLED: bool = true;
151
152    fn cell(&mut self, decision: CellDecision);
153    fn tombstone(&mut self, tombstone: TombstoneRecord);
154    fn generation_probe(&mut self, run_index: usize, outcome: ProbeOutcome);
155}
156
157/// The default sink. It is a zero-sized type and all methods inline to no-op.
158#[derive(Debug, Default, Clone, Copy)]
159pub struct NoTrace;
160
161impl TraceSink for NoTrace {
162    const ENABLED: bool = false;
163
164    #[inline]
165    fn cell(&mut self, _decision: CellDecision) {}
166
167    #[inline]
168    fn tombstone(&mut self, _tombstone: TombstoneRecord) {}
169
170    #[inline]
171    fn generation_probe(&mut self, _run_index: usize, _outcome: ProbeOutcome) {}
172}

174/// An in-memory sink used by the explain surface and integration tests.
175#[derive(Debug, Default, Clone, PartialEq)]
176pub struct RecordingSink {
177    cells: Vec<CellDecision>,
178    tombstones: Vec<TombstoneRecord>,
179    probes: Vec<(usize, ProbeOutcome)>,
180}
181
182impl RecordingSink {
183    #[must_use]
184    pub fn new() -> Self {
185        Self::default()
186    }
187
188    #[must_use]
189    pub fn cells(&self) -> &[CellDecision] {
190        &self.cells
191    }
192
193    #[must_use]
194    pub fn tombstones(&self) -> &[TombstoneRecord] {
195        &self.tombstones
196    }
197
198    #[must_use]
199    pub fn probes(&self) -> &[(usize, ProbeOutcome)] {
200        &self.probes
201    }
202
203    #[must_use]
204    pub fn into_parts(
205        self,
206    ) -> (
207        Vec<CellDecision>,
208        Vec<TombstoneRecord>,
209        Vec<(usize, ProbeOutcome)>,
210    ) {
211        (self.cells, self.tombstones, self.probes)
212    }
213}
214
215impl TraceSink for RecordingSink {
216    fn cell(&mut self, decision: CellDecision) {
217        self.cells.push(decision);
218    }
219
220    fn tombstone(&mut self, tombstone: TombstoneRecord) {
221        self.tombstones.push(tombstone);
222    }
223
224    fn generation_probe(&mut self, run_index: usize, outcome: ProbeOutcome) {
225        self.probes.push((run_index, outcome));
226    }
227}
228
229impl<T: TraceSink + ?Sized> TraceSink for &mut T {
230    const ENABLED: bool = T::ENABLED;
231
232    #[inline]
233    fn cell(&mut self, decision: CellDecision) {
234        (**self).cell(decision);
235    }
236
237    #[inline]
238    fn tombstone(&mut self, tombstone: TombstoneRecord) {
239        (**self).tombstone(tombstone);
240    }
241
242    #[inline]
243    fn generation_probe(&mut self, run_index: usize, outcome: ProbeOutcome) {
244        (**self).generation_probe(run_index, outcome);
245    }
246}
