23/// Open `path` ONCE and probe it for every requested key, delegating to
24/// [`probe_reader_async`] (issue #2346) once the reader is open.
25///
26/// Runs the async open + seek on the shared bridge runtime. The reader is opened
27/// with buffered I/O (never mmap/direct — the file may be deleted after the
28/// read), matching the full-scan producer's file-lifetime contract (issue #591).
29#[cfg(not(feature = "tombstones"))]
30pub(super) fn probe_path(
31    path: &Path,
32    keys: &[Vec<u8>],
33    schema: &TableSchema,
34    udt_registry: Option<&UdtRegistry>,
35    scan_cancel: ScanCancel,
36    collect_notes: bool,
37) -> Result<ProbeOutcome> {
38    let path = path.to_path_buf();
39    let schema = schema.clone();
40    let keys: Vec<Vec<u8>> = keys.to_vec();
41    let udt_registry = udt_registry.cloned();
42    block_on_async(async move {
43        let mut config = Config::default();
44        config.storage.use_mmap = false;
45        config.storage.disk_access_mode = DiskAccessMode::Buffered;
46        let platform = Arc::new(Platform::new(&config).await?);
47        let mut reader = SSTableReader::open(&path, &config, platform).await?;
48        // Issue #2349: decode a `frozen<UDT>`-in-collection cell structurally on the
49        // point-read seek path too, matching the full-scan cold reader.
50        if let Some(registry) = udt_registry {
51            reader.set_udt_registry(registry);
52        }
53        probe_reader_async(&reader, &keys, &schema, &scan_cancel, collect_notes).await
54    })
55}

57/// Probe an ALREADY-OPEN, possibly-SHARED `reader` (issue #2346) — the
58/// reader-based analogue of [`probe_path`], used by
59/// [`build_single_partition_merger_from_readers`]. `reader` is an `Arc` clone
60/// (the caller keeps the original for the `NeedsScan` fail-safe path), moved
61/// into the bridged future so no borrow needs to outlive this call.
62#[cfg(not(feature = "tombstones"))]
63pub(super) fn probe_reader(
64    reader: Arc<SSTableReader>,
65    keys: &[Vec<u8>],
66    schema: &TableSchema,
67    scan_cancel: ScanCancel,
68    collect_notes: bool,
69) -> Result<ProbeOutcome> {
70    let schema = schema.clone();
71    let keys: Vec<Vec<u8>> = keys.to_vec();
72    block_on_async(async move {
73        probe_reader_async(&reader, &keys, &schema, &scan_cancel, collect_notes).await
74    })
75}

77/// Tombstones-build fallback: the single-partition SEEK machinery
78/// (`successor_partition_offset`, `point_read_whole_section`, …) is
79/// `not(tombstones)` only, so under the alternate `tombstones` feature every
80/// candidate degrades to a full scan filtered to the key set — correct, just
81/// without the seek speedup. Keeps the public builder API identical across builds.
82#[cfg(feature = "tombstones")]
83pub(super) fn probe_path(
84    _path: &Path,
85    _keys: &[Vec<u8>],
86    _schema: &TableSchema,
87    _udt_registry: Option<&UdtRegistry>,
88    _scan_cancel: ScanCancel,
89    _collect_notes: bool,
90) -> Result<ProbeOutcome> {
91    Ok(ProbeOutcome {
92        probe: PathProbe::NeedsScan,
93        notes: Vec::new(),
94    })
95}
