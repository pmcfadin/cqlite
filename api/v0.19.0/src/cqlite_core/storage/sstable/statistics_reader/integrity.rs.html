1//! The strict `Statistics.db` layout + checksum check, shared by every reader (#4612).
2//!
3//! `StatisticsReader` used to parse whatever it could from a damaged file, and the merge then
4//! silently re-ordered columns and re-based every write timestamp onto the wrong
5//! `EncodingStats` baseline. Every in-scope consumer now reads the file through this module, so
6//! a truncated or corrupt `Statistics.db` is a named [`Error::Corruption`] carrying the path,
7//! never a degraded open. The layout check is a port of Cassandra's own integrity rules:
8//!
9//! `cassandra-5.0.8` `io/sstable/metadata/MetadataSerializer.java:146-212` (`deserialize`):
10//! `[i32 count][crc32]`, then `count` x `[i32 type][i32 offset]` followed by a `crc32` over the
11//! count and every pair, then each component's bytes followed by a `crc32` of those bytes; a
12//! component's length is the next offset minus its own (the last one runs to end of file). Every
13//! checksum mismatch is a `CorruptSSTableException` (`:214-227`). `hasMetadataChecksum()` is true
14//! for every in-scope format (BIG `na`+ at `format/big/BigFormat.java:404`; BTI `da` at
15//! `format/bti/BtiFormat.java:349-352`).
16
17use crate::error::{Error, Result};
18use std::io::Read;
19use std::path::Path;

21/// The most bytes read from one `Statistics.db`. A real file is a few KiB (a TOC, a handful of
22/// histograms and the serialization header); the cap only bounds a hostile or runaway file so the
23/// read can never be unbounded (#4613 item 5). A file larger than this is refused, never
24/// truncated to fit.
25pub const MAX_STATISTICS_BYTES: u64 = 64 * 1024 * 1024;
26
27/// Number of `MetadataType` constants in `cassandra-5.0.8` (`MetadataType.java:28-34`).
28const METADATA_TYPE_COUNT: i32 = 4;
29
30fn corrupt(path: &Path, reason: impl std::fmt::Display) -> Error {
31    Error::corruption(format!(
32        "Statistics.db {} failed Cassandra's integrity check: {reason}",
33        path.display()
34    ))
35}
36
37fn checked(path: &Path, bytes: Vec<u8>) -> Result<Vec<u8>> {
38    if bytes.len() as u64 > MAX_STATISTICS_BYTES {
39        return Err(corrupt(
40            path,
41            format!("larger than the {MAX_STATISTICS_BYTES} byte cap"),
42        ));
43    }
44    validate_metadata_layout(&bytes).map_err(|reason| corrupt(path, reason))?;
45    Ok(bytes)
46}

48/// Read `path` under [`MAX_STATISTICS_BYTES`] and refuse it unless it passes the strict layout and
49/// checksum check. A missing or unreadable file is an [`Error::Io`]; a damaged one is an
50/// [`Error::Corruption`] naming `path`.
51pub fn read_checked_statistics(path: &Path) -> Result<Vec<u8>> {
52    checked(path, read_bounded_statistics(path)?)
53}
54
55/// Read `path` under [`MAX_STATISTICS_BYTES`] + 1 bytes (the extra byte lets [`checked`] tell an
56/// oversize file from one exactly at the cap) with no integrity check, for callers that apply
57/// their own classification (`verify`).
58pub fn read_bounded_statistics(path: &Path) -> std::io::Result<Vec<u8>> {
59    let file = std::fs::File::open(path)?;
60    let mut bytes = Vec::new();
61    file.take(MAX_STATISTICS_BYTES + 1)
62        .read_to_end(&mut bytes)?;
63    Ok(bytes)
64}
65
66/// [`read_checked_statistics`] for async callers.
67pub async fn read_checked_statistics_async(path: &Path) -> Result<Vec<u8>> {
68    use tokio::io::AsyncReadExt;
69    let file = tokio::fs::File::open(path).await?;
70    let hint = file
71        .metadata()
72        .await
73        .map_or(0, |m| m.len().min(MAX_STATISTICS_BYTES + 1));
74    let mut bytes = Vec::with_capacity(hint as usize);
75    file.take(MAX_STATISTICS_BYTES + 1)
76        .read_to_end(&mut bytes)
77        .await?;
78    checked(path, bytes)
79}

81/// Refuse the whole input set unless every PRESENT generation's `Statistics.db` passes the
82/// strict check (a missing one is left to its existing handling). For the entry points whose downstream helpers (baseline seeding, schema
83/// normalization) must never run on a degraded read. The error names the failing path.
84pub fn require_checked_statistics(data_paths: &[std::path::PathBuf]) -> Result<()> {
85    for data_path in data_paths {
86        let name = data_path
87            .file_name()
88            .and_then(|n| n.to_str())
89            .unwrap_or_default();
90        let Some(stem) = name.strip_suffix("Data.db") else {
91            continue;
92        };
93        match read_checked_statistics(&data_path.with_file_name(format!("{stem}Statistics.db"))) {
94            Ok(_) => {}
95            // A genuinely missing Statistics.db keeps its prior, separately-handled behaviour
96            // (the reader proceeds without it); only a PRESENT damaged one is refused here.
97            Err(Error::Io(e)) if e.kind() == std::io::ErrorKind::NotFound => {}
98            Err(e) => return Err(e),
99        }
100    }
101    Ok(())
102}

104/// `MetadataType` ordinals (`MetadataType.java:28-34`, cassandra-5.0.8).
105const VALIDATION: i32 = 0;
106const STATS: i32 = 2;
107const HEADER: i32 = 3;
108
109/// Every TOC ordinal must be a known `MetadataType` (Cassandra indexes
110/// `MetadataType.values()[ordinal]`, so an unknown one cannot be read), each at most once
111/// (`serialize` writes one entry per type from an `EnumMap`), and VALIDATION, STATS and HEADER
112/// must be present: `MetadataCollector.finalizeMetadata` (`:362-390`) always writes all four and
113/// `BigSSTableReaderLoadingBuilder:77` / `BtiTableReaderLoadingBuilder:63` load exactly
114/// STATS + HEADER + VALIDATION. COMPACTION is written but not required by any loader, so a file
115/// without it is accepted. Without HEADER the parser would fall back and rebuild the wrong
116/// EncodingStats / column baselines, which is the defect this module exists to refuse.
117fn check_component_types(ordinals: &[i32]) -> std::result::Result<(), String> {
118    let mut seen = [false; METADATA_TYPE_COUNT as usize];
119    for &ordinal in ordinals {
120        let slot = usize::try_from(ordinal)
121            .ok()
122            .and_then(|i| seen.get_mut(i))
123            .ok_or_else(|| format!("TOC names unknown metadata type ordinal {ordinal}"))?;
124        if std::mem::replace(slot, true) {
125            return Err(format!("TOC names metadata type ordinal {ordinal} twice"));
126        }
127    }
128    for (required, name) in [
129        (VALIDATION, "VALIDATION"),
130        (STATS, "STATS"),
131        (HEADER, "HEADER"),
132    ] {
133        if !seen[required as usize] {
134            return Err(format!(
135                "TOC has no {name} component (required by Cassandra's loaders)"
136            ));
137        }
138    }
139    Ok(())
140}

142/// Why a `Statistics.db` failed the strict check.
143pub fn validate_metadata_layout(bytes: &[u8]) -> std::result::Result<(), String> {
144    let int_at = |at: usize| -> std::result::Result<i32, String> {
145        bytes
146            .get(at..at + 4)
147            .map(|b| i32::from_be_bytes([b[0], b[1], b[2], b[3]]))
148            .ok_or_else(|| {
149                format!(
150                    "truncated: needs 4 bytes at offset {at}, file is {} bytes",
151                    bytes.len()
152                )
153            })
154    };
155    let mut crc = crc32fast::Hasher::new();
156
157    let count = int_at(0)?;
158    crc.update(&count.to_be_bytes());
159    let expected = int_at(4)?;
160    let actual = crc.clone().finalize() as i32;
161    if actual != expected {
162        return Err(format!(
163            "TOC count checksum mismatch (stored {expected:#x}, computed {actual:#x})"
164        ));
165    }
166    if count < 1 {
167        return Err(format!("TOC declares {count} components"));
168    }
169    // `MetadataType` (`io/sstable/metadata/MetadataType.java`, cassandra-5.0.8) has exactly
170    // VALIDATION, COMPACTION, STATS, HEADER; `deserialize` fills one entry per type, so a larger
171    // count cannot be a Cassandra file. Refuse it BEFORE any allocation or per-pair work.
172    if count > METADATA_TYPE_COUNT {
173        return Err(format!(
174            "TOC declares {count} components but Cassandra defines only {METADATA_TYPE_COUNT} \
175             metadata types (VALIDATION, COMPACTION, STATS, HEADER)"
176        ));
177    }
178    let count = usize::try_from(count).map_err(|_| "TOC count overflows".to_string())?;
179    // 4 (count) + 4 (crc) + 8 per pair + 4 (crc): reject a count the file cannot hold before
180    // looking at any pair.
181    let toc_end = count
182        .checked_mul(8)
183        .and_then(|pairs| pairs.checked_add(12))
184        .ok_or_else(|| "TOC size overflows".to_string())?;
185    if toc_end > bytes.len() {
186        return Err(format!(
187            "truncated: TOC of {count} components needs {toc_end} bytes, file is {} bytes",
188            bytes.len()
189        ));
190    }
191    let mut offsets = Vec::new(); // never pre-sized from an untrusted count
192    let mut ordinals = Vec::new();
193    for i in 0..count {
194        let at = 8 + 8 * i;
195        let ordinal = int_at(at)?;
196        ordinals.push(ordinal);
197        crc.update(&ordinal.to_be_bytes());
198        let offset = int_at(at + 4)?;
199        crc.update(&offset.to_be_bytes());
200        offsets.push(
201            usize::try_from(offset).map_err(|_| format!("component {i} has a negative offset"))?,
202        );
203    }
204    let expected = int_at(8 + 8 * count)?;
205    let actual = crc.finalize() as i32;
206    if actual != expected {
207        return Err(format!(
208            "TOC checksum mismatch (stored {expected:#x}, computed {actual:#x})"
209        ));
210    }
211    check_component_types(&ordinals)?;
212
213    for (i, &start) in offsets.iter().enumerate() {
214        let end = offsets.get(i + 1).copied().unwrap_or(bytes.len());
215        if start < toc_end || end < start + 4 || end > bytes.len() {
216            return Err(format!(
217                "component {i} spans {start}..{end}, outside the {} byte file or shorter than its checksum",
218                bytes.len()
219            ));
220        }
221        let body = &bytes[start..end - 4];
222        let expected = int_at(end - 4)?;
223        let actual = crc32fast::hash(body) as i32;
224        if actual != expected {
225            return Err(format!(
226                "component {i} checksum mismatch (stored {expected:#x}, computed {actual:#x})"
227            ));
228        }
229    }
230    Ok(())
231}

237    /// Build a file the way `MetadataSerializer.serialize` (`:67-112`) lays it out.
238    fn build(components: &[&[u8]]) -> Vec<u8> {
239        let typed: Vec<(i32, &[u8])> = components
240            .iter()
241            .enumerate()
242            .map(|(i, b)| (i as i32, *b))
243            .collect();
244        build_typed(&typed)
245    }
246
247    /// As [`build`], with explicit type ordinals (checksums recomputed so the file reaches the
248    /// ordinal check).
249    fn build_typed(components: &[(i32, &[u8])]) -> Vec<u8> {
250        let mut out = Vec::new();
251        let count = components.len() as i32;
252        out.extend(count.to_be_bytes());
253        let mut crc = crc32fast::Hasher::new();
254        crc.update(&count.to_be_bytes());
255        out.extend((crc.clone().finalize() as i32).to_be_bytes());
256        let mut position = 4 + 8 * components.len() + 8;
257        for (ordinal, body) in components.iter() {
258            out.extend(ordinal.to_be_bytes());
259            crc.update(&ordinal.to_be_bytes());
260            out.extend((position as i32).to_be_bytes());
261            crc.update(&(position as i32).to_be_bytes());
262            position += body.len() + 4;
263        }
264        out.extend((crc.finalize() as i32).to_be_bytes());
265        for (_, body) in components {
266            out.extend(*body);
267            out.extend((crc32fast::hash(body) as i32).to_be_bytes());
268        }
269        out
270    }
