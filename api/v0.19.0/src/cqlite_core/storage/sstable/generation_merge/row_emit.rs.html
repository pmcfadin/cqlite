1//! Row emission for the cross-generation read drivers (issue #4640).
2//!
3//! The k-way merge yields ONE `CellData` per collection element (keyed by its
4//! `cell_path`). Emitting each as a `(column, value)` pair, as this module did before
5//! #4640, handed N duplicate pairs to the executor for an N-element collection, and the
6//! name-keyed row map kept only the last. Every emission here therefore routes the
7//! surviving cells through [`assemble_read_cells`], the ONE reassembler (#2324/#2339, the
8//! Flight producer's), which folds per-element cells into `Value::List`/`Set`/`Map` and
9//! fails closed on a column that mixes simple and complex cells or whose elements cannot
10//! be ordered (no heuristics, #28).
11//!
12//! The assembly is projection-scoped and UDT-aware (owner ruling Q51): [`ReadAssembly`]
13//! carries the SELECT's needed columns (`storage::read_projection`) and the reader's
14//! UDT registry, and calls [`assemble_read_cells_with_udts`], the assembler the Flight
15//! producer uses. Columns the query does not read are never assembled, so a table that
16//! merely HAS a UDT collection cannot fail a query that selects other columns.
17

29/// One reconciled metadata row inside the merge task, before per-cell metadata is
30/// attached: `(partition key bytes, ScanRow row carrier, [(column,
31/// write_timestamp_micros)])`.
32pub(super) type MergedMetaRow = (Vec<u8>, ScanRow, Vec<(String, i64)>);
33
34/// What the merged read needs to reassemble collections, captured ONCE per read on the
35/// caller's task (before any hop to a blocking thread).
36#[derive(Clone)]
37pub(super) struct ReadAssembly {
38    /// The SELECT's columns plus the key columns; `None` = every column.
39    needed: Option<HashSet<String>>,
40    /// The readers' UDT registry (the manager's; no second registry is built).
41    pub(super) registry: Option<UdtRegistry>,
42    keyspace: String,
43}
44
45impl ReadAssembly {
46    pub(super) fn capture(
47        readers: &[Arc<SSTableReader>],
48        schema: &crate::schema::TableSchema,
49    ) -> Self {
50        let needed = crate::storage::read_projection::current().map(|n| {
51            let mut set: HashSet<String> = (*n).clone();
52            // Key columns surface as pseudo-cells the row needs regardless of projection.
53            set.extend(schema.partition_keys.iter().map(|k| k.name.clone()));
54            set.extend(schema.clustering_keys.iter().map(|k| k.name.clone()));
55            set
56        });
57        Self {
58            needed,
59            registry: readers.iter().find_map(|r| r.udt_registry.clone()),
60            keyspace: schema.keyspace.clone(),
61        }
62    }
63
64    fn assemble(
65        &self,
66        cells: Vec<crate::storage::write_engine::merge::CellData>,
67        schema: &crate::schema::TableSchema,
68    ) -> Result<RowCells> {
69        let scope = self.registry.as_ref().map(|registry| UdtScope {
70            registry,
71            keyspace: &self.keyspace,
72        });
73        assemble_read_cells_with_udts(cells, schema, self.needed.as_ref(), scope)
74    }
75}

77/// Convert one reconciled partition's merge rows into the live scan rows the read
78/// path emits: drop cell tombstones, drop row tombstones, drop rows left empty,
79/// and reassemble per-element collection cells into whole collections.
80///
81/// The single emission kernel shared by the materializing plain merge and the
82/// streaming driver so tombstone filtering can never drift between them (issue
83/// #1334: emit the interned-name `ScanRow` carrier the read path consumes).
84pub(super) fn partition_live_rows(
85    row_key: &RowKey,
86    rows: Vec<MergeEntry>,
87    shadow: &ReadShadow,
88    schema: &crate::schema::TableSchema,
89    asm: &ReadAssembly,
90) -> Result<Vec<(RowKey, ScanRow)>> {
91    // Issue #1849: partition-tombstone cover for this partition's rows, applied as a
92    // read-side shadow floor alongside per-cell read-time TTL expiry (post-merge).
93    let cover = partition_cover(&rows);
94    let mut out = Vec::new();
95    for entry in rows {
96        let vis = row_visibility::RowVis::of(&entry); // before `row_data` is moved out
97        match entry.row_data {
98            RowData::Live { cells } => {
99                // Read-time visibility: drop cell tombstones + TTL-expired /
100                // partition-shadowed data cells; `None` hides the whole row.
101                let Some(surviving) = shadow.filter_visible(cover, vis, cells) else {
102                    continue;
103                };
104                let row_cells: RowCells = asm.assemble(surviving, schema)?;
105                out.push((row_key.clone(), ScanRow::Row(row_cells)));
106            }
107            // Row tombstone: the row is deleted across all generations — suppress.
108            RowData::Tombstone { .. } => {}
109        }
110    }
111    Ok(out)
112}

114/// Convert one reconciled partition's merge rows into metadata-carrying rows: the
115/// live cells (collections reassembled, #4640) plus each surviving cell's `(column,
116/// write_timestamp)`. Shared by the range and point-targeted branches of the metadata
117/// merge so the tombstone filtering + timestamp capture live once.
118pub(super) fn push_metadata_rows(
119    key_bytes: &[u8],
120    rows: Vec<MergeEntry>,
121    out: &mut Vec<MergedMetaRow>,
122    shadow: &ReadShadow,
123    schema: &crate::schema::TableSchema,
124    asm: &ReadAssembly,
125) -> Result<()> {
126    // Issue #1849: same read-side partition cover + per-cell TTL/shadow filter as the
127    // plain path, so the WRITETIME/TTL projection agrees with `SELECT *`.
128    let cover = partition_cover(&rows);
129    for entry in rows {
130        let vis = row_visibility::RowVis::of(&entry); // before `row_data` is moved out
131        if let RowData::Live { cells } = entry.row_data {
132            // `filter_live` drops cell tombstones + TTL-expired / partition-shadowed
133            // data cells; `None` hides the whole row (partition-shadowed).
134            let Some(surviving) = shadow.filter_visible(cover, vis, cells) else {
135                continue;
136            };
137            // One entry per surviving CELL (a collection contributes one per element).
138            let timestamps: Vec<(String, i64)> = surviving
139                .iter()
140                .map(|c| (c.column.clone(), c.timestamp))
141                .collect();
142            let row_cells: RowCells = asm.assemble(surviving, schema)?;
143            out.push((key_bytes.to_vec(), ScanRow::Row(row_cells), timestamps));
144        }
145        // Row tombstones suppress the row entirely (no emission).
146    }
147    Ok(())
148}
