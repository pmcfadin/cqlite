6/// Names of the schema's STATIC columns.
7pub(super) fn static_column_names(schema: &crate::schema::TableSchema) -> HashSet<String> {
8    (schema.columns.iter())
9        .filter(|c| c.is_static)
10        .map(|c| c.name.clone())
11        .collect()
12}

14/// The per-entry facts [`ReadShadow::filter_visible`] needs besides the cells (all `Copy`, read
15/// off the `MergeEntry` BEFORE its `row_data` is moved out).
16#[derive(Clone, Copy)]
17pub(super) struct RowVis {
18    pub(super) liveness: RowLiveness,
19    /// `entry.clustering_key.is_none()`: the partition's static row.
20    pub(super) static_row: bool,
21    /// The entry's coexisting row deletion `(markedForDeleteAt µs, localDeletionTime s)`.
22    pub(super) row_deletion: Option<(i64, i32)>,
23}
24
25impl RowVis {
26    pub(super) fn of(entry: &super::MergeEntry) -> Self {
27        Self {
28            liveness: entry.row_liveness,
29            static_row: entry.clustering_key.is_none(),
30            row_deletion: entry.row_deletion,
31        }
32    }
33}
34
35impl ReadShadow {
36    /// [`ReadShadow::filter_live`] plus the row-level visibility rule: `None` also when the row
37    /// has neither a LIVE primary-key liveness marker nor a live non-key (data) cell.
38    ///
39    /// Cassandra 5.0.8 `BTreeRow.purge` -> `update` keeps a row iff it has a live liveness marker
40    /// OR a live data cell; deleted cells are purged and never keep a row. The previous "drop
41    /// rows left empty" rule lost the liveness-only / explicit-null row (marker, no live cell)
42    /// and, on a clustered table, kept a marker-less row whose only surviving cells were its
43    /// clustering pseudo-cells. Partition/clustering pseudo-cells (`key_columns`) are structural,
44    /// kept verbatim by `filter_live`, and never count as data.
45    ///
46    /// A STATIC cell is never data of a CLUSTERING row either: Cassandra 5.0.8
47    /// `AbstractRow.hasLiveData` (`db/rows/AbstractRow.java:48`) consults only the row's OWN
48    /// liveness and its OWN `cells()`, and the static cells belong to the separate static `Row`
49    /// (`AbstractRow.isStatic`: `clustering() == Clustering.STATIC_CLUSTERING`), which a read
50    /// emits on its own. So a live static value must not keep a markerless, tombstone-only or
51    /// expired clustering row visible. `static_row` is the static row itself (no clustering
52    /// key), whose own static cells ARE its data.
53    ///
54    /// `row_deletion` is the entry's coexisting row deletion (`MergeEntry::row_deletion`, kept only
55    /// when newer than some surviving cell): a marker it covers is NOT liveness.
56    pub(super) fn filter_visible(
57        &self,
58        cover: Option<i64>,
59        vis: RowVis,
60        cells: Vec<CellData>,
61    ) -> Option<Vec<CellData>> {
62        let RowVis {
63            liveness,
64            static_row,
65            row_deletion,
66        } = vis;
67        let surviving = self.filter_live(cover, liveness.marker_timestamp, cells)?;
68        let has_data = surviving.iter().any(|c| {
69            !self.key_columns.contains(&c.column)
70                && (static_row || !self.static_columns.contains(&c.column))
71        });
72        // Roborev job 307: the marker counts only if it survives the row's coexisting deletion —
73        // Cassandra 5.0.8 `DeletionTime.deletes(ts)` is `ts <= markedForDeleteAt` (equal loses).
74        let marker_survives = row_deletion
75            .zip(liveness.marker_timestamp)
76            .is_none_or(|((deleted_at, _), marker)| marker > deleted_at);
77        (has_data || (marker_survives && liveness.marker_live_at(self.now_secs)))
78            .then_some(surviving)
79    }
