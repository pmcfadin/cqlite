1//! Cassandra `uuid` / `timeuuid` ordering and byte-comparable production (issue #4534, #4531).
2//!
3//! `uuid` (`UUIDType`) and `timeuuid` (`TimeUUIDType`) do NOT order by their raw 16 bytes, and
4//! they do not order the same way as each other. Both orderings, and the matching OSS50
5//! `asComparableBytes` swizzles the `Rows.db` row index is built from, are ported here from the
6//! pinned `cassandra-5.0.8` sources (`src/java/org/apache/cassandra/db/marshal/`):
7//!
8//! * `UUIDType.compareCustom` (`UUIDType.java:74-119`): version nibble first; for version 1 the
9//!   reordered timestamp (signed long compare); otherwise unsigned msb; then UNSIGNED lsb.
10//! * `AbstractTimeUUIDType.compareCustom` (`:57-87`): reordered timestamp, then the lsb compared
11//!   with SIGNED per-byte semantics (`signedBytesToNativeLong`, `:72-75` — CASSANDRA-8730).
12//! * `UUIDType.asComparableBytes` (`:122-140`) and `AbstractTimeUUIDType.asComparableBytes`
13//!   (`:89-102`).
14//!
15//! `uuid` vs `timeuuid` is decided by the DECLARED column type, never by inspecting the bytes
16//! (#28): every caller passes the schema's type. A `timeuuid` value that is not version 1 is a
17//! `MarshalException` in Cassandra; here it simply orders by the same formulas.
18
19use std::cmp::Ordering;
20
21fn msb(u: &[u8; 16]) -> u64 {
22    u64::from_be_bytes([u[0], u[1], u[2], u[3], u[4], u[5], u[6], u[7]])
23}
24
25fn lsb(u: &[u8; 16]) -> u64 {
26    u64::from_be_bytes([u[8], u[9], u[10], u[11], u[12], u[13], u[14], u[15]])
27}
28
29fn version(msb: u64) -> u64 {
30    (msb >> 12) & 0xf
31}
32
33/// `AbstractTimeUUIDType.reorderTimestampBytes` (`:85-90`).
34fn reorder_timestamp_bytes(input: u64) -> u64 {
35    (input << 48) | ((input << 16) & 0xFFFF_0000_0000) | (input >> 32)
36}
37
38/// `UUIDType.compareCustom` — the order of a declared `uuid` column.
39pub(crate) fn uuid_type_cmp(a: &[u8; 16], b: &[u8; 16]) -> Ordering {
40    let (m1, m2) = (msb(a), msb(b));
41    let (v1, v2) = (version(m1), version(m2));
42    if v1 != v2 {
43        return v1.cmp(&v2);
44    }
45    let first = if v1 == 1 {
46        // reordered longs are non-negative (top bits are the 12-bit time_hi), so the signed
47        // `Long.compare` equals the unsigned one.
48        reorder_timestamp_bytes(m1).cmp(&reorder_timestamp_bytes(m2))
49    } else {
50        m1.cmp(&m2) // UnsignedLongs.compare
51    };
52    first.then_with(|| lsb(a).cmp(&lsb(b))) // UnsignedLongs.compare
53}
54
55/// `AbstractTimeUUIDType.compareCustom` — the order of a declared `timeuuid` column.
56pub(crate) fn timeuuid_type_cmp(a: &[u8; 16], b: &[u8; 16]) -> Ordering {
57    let first =
58        (reorder_timestamp_bytes(msb(a)) as i64).cmp(&(reorder_timestamp_bytes(msb(b)) as i64));
59    first.then_with(|| {
60        // signedBytesToNativeLong: lsb ^ 0x0080808080808080, then a SIGNED long compare.
61        let s = |x: u64| (x ^ 0x0080_8080_8080_8080) as i64;
62        s(lsb(a)).cmp(&s(lsb(b)))
63    })
64}

66/// Order two `Value::Uuid`s as a declared `uuid` (`timeuuid == false`) or `timeuuid`.
67pub(crate) fn compare_values(
68    left: &crate::types::Value,
69    right: &crate::types::Value,
70    timeuuid: bool,
71) -> crate::Result<Ordering> {
72    match (left, right) {
73        (crate::types::Value::Uuid(l), crate::types::Value::Uuid(r)) => Ok(if timeuuid {
74            timeuuid_type_cmp(l, r)
75        } else {
76            uuid_type_cmp(l, r)
77        }),
78        _ => Err(crate::Error::Schema(
79            "Type mismatch: expected uuid values".to_string(),
80        )),
81    }
82}

84/// `timeuuid` order of two serialized cell paths (a list element's position). A path that is not
85/// 16 bytes (never produced for a list) falls back to raw bytes. The write-engine merge and the
86/// single-generation SELECT decode (#4645) both order list cells by path with it.
87pub(crate) fn timeuuid_path_cmp(a: &[u8], b: &[u8]) -> Ordering {
88    match (<[u8; 16]>::try_from(a), <[u8; 16]>::try_from(b)) {
89        (Ok(x), Ok(y)) => timeuuid_type_cmp(&x, &y),
90        _ => a.cmp(b),
91    }
92}

94/// `UUIDType.asComparableBytes` (`:122-140`): the 16 bytes a declared `uuid` contributes to a
95/// byte-comparable key.
96pub(crate) fn uuid_comparable_bytes(u: &[u8; 16]) -> [u8; 16] {
97    let m = msb(u);
98    let v = version(m);
99    let hi = if v == 1 {
100        reorder_timestamp_bytes(m)
101    } else {
102        (v << 60) | ((m >> 4) & 0x0FFF_FFFF_FFFF_F000) | (m & 0xFFF)
103    };
104    concat(hi, lsb(u))
105}
106
107/// `AbstractTimeUUIDType.asComparableBytes` (`:89-102`): reordered timestamp msb, then the lsb
108/// XOR `0x8080808080808080`.
109pub(crate) fn timeuuid_comparable_bytes(u: &[u8; 16]) -> [u8; 16] {
110    concat(
111        reorder_timestamp_bytes(msb(u)),
112        lsb(u) ^ 0x8080_8080_8080_8080,
113    )
114}
115
116fn concat(hi: u64, lo: u64) -> [u8; 16] {
117    let mut out = [0u8; 16];
118    out[..8].copy_from_slice(&hi.to_be_bytes());
119    out[8..].copy_from_slice(&lo.to_be_bytes());
120    out
121}

123#[cfg(test)]
124mod tests {
125    use super::*;
126
127    fn v1(time_low: u32, time_mid: u16, time_hi: u16, lsb_bytes: [u8; 8]) -> [u8; 16] {
128        let mut u = [0u8; 16];
129        u[0..4].copy_from_slice(&time_low.to_be_bytes());
130        u[4..6].copy_from_slice(&time_mid.to_be_bytes());
131        u[6..8].copy_from_slice(&((1u16 << 12) | (time_hi & 0x0fff)).to_be_bytes());
132        u[8..].copy_from_slice(&lsb_bytes);
133        u
134    }
135
136    /// The two orders differ ONLY in the lsb tie-break, at an identical v1 timestamp
137    /// (CASSANDRA-8730): lsb byte 0x7f vs 0x80.
138    #[test]
139    fn lsb_tiebreak_is_unsigned_for_uuid_and_signed_for_timeuuid() {
140        let lo = v1(5, 5, 5, [0x7f, 0, 0, 0, 0, 0, 0, 0]);
141        let hi = v1(5, 5, 5, [0x80, 0, 0, 0, 0, 0, 0, 0]);
142        assert_eq!(
143            uuid_type_cmp(&lo, &hi),
144            Ordering::Less,
145            "unsigned: 0x7f < 0x80"
146        );
147        assert_eq!(
148            timeuuid_type_cmp(&lo, &hi),
149            Ordering::Greater,
150            "signed: 0x7f > 0x80 (-128)"
151        );
152    }
153
154    #[test]
155    fn timeuuid_orders_by_timestamp_not_by_raw_time_low() {
156        // later timestamp but smaller time_low: raw bytes would sort it first.
157        let early = v1(0xFFFF_FFFF, 0, 1, [0; 8]);
158        let late = v1(0, 0, 2, [0; 8]);
159        assert!(early[..] > late[..], "raw bytes disagree");
160        assert_eq!(timeuuid_type_cmp(&early, &late), Ordering::Less);
161        assert_eq!(uuid_type_cmp(&early, &late), Ordering::Less);
162    }
163
164    #[test]
165    fn comparable_bytes_are_order_isomorphic() {
166        let samples = [
167            v1(1, 0, 0, [0; 8]),
168            v1(2, 0, 0, [0xff; 8]),
169            v1(0, 1, 0, [0x7f, 0, 0, 0, 0, 0, 0, 0]),
170            v1(0, 0, 1, [0x80, 0, 0, 0, 0, 0, 0, 0]),
171            v1(0xFFFF_FFFF, 0xFFFF, 0x0FFF, [0xff; 8]),
172        ];
173        for a in &samples {
174            for b in &samples {
175                assert_eq!(
176                    timeuuid_comparable_bytes(a).cmp(&timeuuid_comparable_bytes(b)),
177                    timeuuid_type_cmp(a, b)
178                );
179                assert_eq!(
180                    uuid_comparable_bytes(a).cmp(&uuid_comparable_bytes(b)),
181                    uuid_type_cmp(a, b)
182                );
183            }
184        }
185        // mixed versions
186        let mut v4 = [0x11u8; 16];
187        v4[6] = 0x40 | (v4[6] & 0x0f);
188        let mut v3 = [0xeeu8; 16];
189        v3[6] = 0x30 | (v3[6] & 0x0f);
190        for (a, b) in [(v4, v3), (v3, v4), (v4, samples[0]), (samples[0], v3)] {
191            assert_eq!(
192                uuid_comparable_bytes(&a).cmp(&uuid_comparable_bytes(&b)),
193                uuid_type_cmp(&a, &b)
194            );
195        }
196    }
