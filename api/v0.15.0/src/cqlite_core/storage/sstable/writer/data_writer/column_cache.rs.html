1//! Schema-constant ordered column lists, memoized once per writer (issue #1674,
2//! R3).
3//!
4//! Part of the `data_writer` responsibility split (issue #1118): this module
5//! holds one `impl DataWriter` block. `use super::*` pulls the shared writer
6//! types, serialization/schema helpers, and crate imports re-exported from
7//! `data_writer/mod.rs`. No emitted bytes change.
8//!
9//! The schema is fixed for a writer's lifetime, yet `regular_columns` /
10//! `static_columns` were re-filtered and re-sorted up to 3× per row, and the
11//! sort comparator (`column_order_key` → `is_complex_column`) allocated a
12//! lowercased `String` on every comparison. These accessors resolve through a
13//! lazily-built [`OrderedCols`](super::OrderedCols) cache instead, so the sort +
14//! `to_lowercase` classification runs exactly once (O(C)), never per row.
15
16use super::*;
17
18impl DataWriter {
19    /// Get regular (non-PK, non-CK, non-static) columns from schema.
20    ///
21    /// Cassandra's column bitmap only covers regular columns — partition key
22    /// and clustering key columns are serialized separately in the partition
23    /// header and clustering prefix. Within the regular set, simple columns
24    /// sort before complex columns, then by name. The order is served from the
25    /// per-writer cache (issue #1674, R3), so no per-row sort runs.
26    ///
27    /// Test-only helper (issue #1674, R3): the per-row hot paths read the cached
28    /// indices directly (`cached_cols(...).regular`) to stay allocation-free
29    /// (issue #1673 ratchet); this `Vec<&Column>` form exists for order-assertion
30    /// tests (`scenarios_3`).
31    #[cfg(test)]
32    pub(super) fn regular_columns<'a>(&self, schema: &'a TableSchema) -> Vec<&'a Column> {
33        self.cached_cols(schema)
34            .regular
35            .iter()
36            .map(|&idx| &schema.columns[idx])
37            .collect()
38    }
39
40    /// Get static columns from schema in Cassandra serialization-header order.
41    ///
42    /// Test-only helper (issue #1674, R3): per-row hot paths read the cached
43    /// indices directly (`cached_cols(...).static_`) to stay allocation-free
44    /// (issue #1673 ratchet); this `Vec<&Column>` form exists for order-assertion
45    /// tests (`scenarios_3`).
46    #[cfg(test)]
47    pub(super) fn static_columns<'a>(&self, schema: &'a TableSchema) -> Vec<&'a Column> {
48        self.cached_cols(schema)
49            .static_
50            .iter()
51            .map(|&idx| &schema.columns[idx])
52            .collect()
53    }
54
55    /// Regular columns paired with their cached `is_complex` flag, in Cassandra
56    /// serialization-header order (issue #1674, R3). Lets the cell-emission loop
57    /// read complexity from the cache instead of re-lowercasing each column's
58    /// type per row.
59    pub(super) fn regular_columns_with_complex<'a>(
60        &self,
61        schema: &'a TableSchema,
62    ) -> Vec<(&'a Column, bool)> {
63        let cache = self.cached_cols(schema);
64        cache
65            .regular
66            .iter()
67            .map(|&idx| (&schema.columns[idx], cache.is_complex[idx]))
68            .collect()
69    }
70
71    /// `is_complex_column` for the column named `col_name`, read from the
72    /// per-writer cache (issue #1674, R3) so `to_lowercase` never runs per row.
73    /// Unknown names (no matching schema column) are non-complex, matching the
74    /// prior `find(...).map(...).unwrap_or(false)` behaviour.
75    pub(super) fn column_is_complex(&self, schema: &TableSchema, col_name: &str) -> bool {
76        schema
77            .columns
78            .iter()
79            .position(|c| c.name == col_name)
80            .map(|idx| self.cached_cols(schema).is_complex[idx])
81            .unwrap_or(false)
82    }
83
84    /// Borrow the schema-constant ordered column cache, building it on first use
85    /// (issue #1674, R3). The schema is fixed for the writer's lifetime, so this
86    /// runs the filter + sort + `is_complex_column` classification exactly ONCE;
87    /// every later row read is `O(C)` index resolution with no `to_lowercase`.
88    ///
89    /// Invariant: a `DataWriter` sees exactly ONE schema for its lifetime
90    /// (`SSTableWriter` owns one `schema` and threads it into every call). The
91    /// cache is built from the FIRST schema and reused; since `is_complex` /
92    /// index resolution below assumes the passed `schema` matches, the
93    /// `debug_assert` fails loudly in debug/test if a caller ever swaps schemas
94    /// (a differently-sized column list) rather than emitting wrong bytes.
95    pub(super) fn cached_cols(&self, schema: &TableSchema) -> &OrderedCols {
96        let cols = self
97            .column_cache
98            .get_or_init(|| Self::build_ordered_cols(schema));
99        debug_assert_eq!(
100            cols.is_complex.len(),
101            schema.columns.len(),
102            "DataWriter column cache built from a different schema than now passed \
103             (one-schema-per-writer invariant, issue #1674)"
104        );
105        cols
106    }
107
108    /// Compute the ordered regular/static column index lists + per-column
109    /// `is_complex` classification (issue #1674, R3).
110    ///
111    /// Byte-order invariant: the ordering MUST equal the previous
112    /// `sort_by_key(column_order_key)` — i.e. the same `(is_complex, name)` tuple
113    /// key — so the emitted column bitmap / cell order stays byte-identical. Names
114    /// are unique within a table, so ties on the key never occur and sort
115    /// stability is irrelevant.
116    fn build_ordered_cols(schema: &TableSchema) -> OrderedCols {
117        // One `to_lowercase` per column (O(C)), never per row.
118        let is_complex: Vec<bool> = schema
119            .columns
120            .iter()
121            .map(|column| is_complex_column(&column.data_type))
122            .collect();
123        // Same key as `column_order_key(column)` = `(is_complex, name)`, resolved
124        // through the precomputed `is_complex` slice.
125        let order_key = |idx: usize| (is_complex[idx], schema.columns[idx].name.as_str());
126
127        let mut regular: Vec<usize> = schema
128            .columns
129            .iter()
130            .enumerate()
131            .filter(|(_, column)| {
132                !column.is_static
133                    && !schema.is_partition_key(&column.name)
134                    && !schema.is_clustering_key(&column.name)
135            })
136            .map(|(idx, _)| idx)
137            .collect();
138        regular.sort_by(|&a, &b| order_key(a).cmp(&order_key(b)));
139
140        let mut static_: Vec<usize> = schema
141            .columns
142            .iter()
143            .enumerate()
144            .filter(|(_, column)| column.is_static)
145            .map(|(idx, _)| idx)
146            .collect();
147        static_.sort_by(|&a, &b| order_key(a).cmp(&order_key(b)));
148
149        OrderedCols {
150            regular,
151            static_,
152            is_complex,
153        }
154    }
155}

157/// A thread-local recording scope for `is_complex_column` calls on the WRITE
158/// path (issue #1674, R3), mirroring the `#2428` parallel-test-pollution-immune
159/// design of the `work_counters` scopes (`cell_data_clone_scope` et al.). The
160/// writer's `is_complex_column` allocates a lowercased `String` per call, so this
161/// scope makes it observable that the schema-constant ordered column lists and
162/// per-column complexity classification are computed ONCE per writer (an `O(C)`
163/// pass) rather than re-derived per row (`O(R·C·log C)`).
164///
165/// Kept HERE, alongside the cache it guards, rather than in the (already
166/// over-threshold, epic #1116) `work_counters.rs`. `#[cfg(test)]`: the
167/// `record()` call in `is_complex_column` is likewise `#[cfg(test)]`-gated, so
168/// production pays ZERO added cost. Also gated on `feature = "write-support"`
169/// because `is_complex_column` and this module only compile with that feature.
170#[cfg(all(test, feature = "write-support"))]
171pub(crate) mod is_complex_scope {
172    use std::cell::Cell;
173
174    thread_local! {
175        /// `Some(count)` while an [`IsComplexScope`] is active on this thread,
176        /// `None` otherwise. Only `is_complex_column` calls on this thread bump it.
177        static SCOPED: Cell<Option<u64>> = const { Cell::new(None) };
178    }
179
180    /// Bump the active scope on the current thread, if any. A no-op on threads
181    /// (production writes, other tests) with no active scope.
182    pub(crate) fn record() {
183        SCOPED.with(|c| {
184            if let Some(v) = c.get() {
185                c.set(Some(v.saturating_add(1)));
186            }
187        });
188    }
189
190    /// A per-thread recording scope for `is_complex_column`. Open one before
191    /// driving row writes and read [`count`](IsComplexScope::count) after. Immune
192    /// to concurrent tests on other threads (issue #2428); dropping it clears the
193    /// scope. Deliberately `!Send` (holds a `PhantomData<*const ()>`).
194    pub(crate) struct IsComplexScope {
195        _not_send: std::marker::PhantomData<*const ()>,
196    }
197
198    impl IsComplexScope {
199        /// Begin recording on the current thread. Panics if a scope is already
200        /// active on this thread (one scope per assertion; nesting unsupported).
201        pub(crate) fn new() -> Self {
202            SCOPED.with(|c| {
203                assert!(
204                    c.get().is_none(),
205                    "an IsComplexScope is already active on this thread (nesting unsupported)"
206                );
207                c.set(Some(0));
208            });
209            Self {
210                _not_send: std::marker::PhantomData,
211            }
212        }
213
214        /// `is_complex_column` increments recorded on this thread since the scope
215        /// opened.
216        pub(crate) fn count(&self) -> u64 {
217            SCOPED.with(|c| c.get().unwrap_or(0))
218        }
219    }
220
221    impl Drop for IsComplexScope {
222        fn drop(&mut self) {
223            SCOPED.with(|c| c.set(None));
224        }
225    }
226}
