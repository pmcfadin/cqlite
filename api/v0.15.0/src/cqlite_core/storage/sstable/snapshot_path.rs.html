1//! Authoritative snapshot-aware SSTable path parsing (issue #2384).
2//!
3//! Cassandra lays SSTables out as
4//! `.../{keyspace}/{table_name}-{table_id}/...-Data.db`. For a **snapshot** the
5//! components live one level deeper, under
6//! `.../{table_name}-{table_id}/snapshots/{snapshot_name}/...-Data.db`
7//! (see `Directories.SNAPSHOT_SUBDIR` in Cassandra's `Directories.java`).
8//!
9//! Every in-crate parser that derives `{keyspace}.{table}` identity from a path —
10//! header keyspace/table, schema-registry resolution, and `SSTableManager` keying —
11//! MUST route through this module so a snapshot read resolves the REAL
12//! `{keyspace}.{table}` rather than `snapshots`/`{snapshot_name}`.
13//!
14//! This is pure directory-segment STRUCTURE matching (a fixed Cassandra naming
15//! shape), not a data/byte heuristic — it is compliant with the no-heuristics
16//! mandate (issue #28).
17
18use std::path::Path;
19
20/// Return `true` when `name` matches Cassandra's `{table_name}-{table_id}` dir
21/// shape: a non-empty table name, a `-`, then EXACTLY 32 lowercase hex chars.
22///
23/// Cassandra encodes the table UUID as 32 lowercase hex digits (no hyphens) in
24/// the on-disk directory name (`TableMetadata`/`Directories.java`). Table names
25/// may themselves contain hyphens, so only the final `-{32-hex}` suffix is the
26/// id — we split on the LAST hyphen.
27fn is_table_id_dir(name: &str) -> bool {
28    match name.rsplit_once('-') {
29        Some((table_name, id)) => {
30            !table_name.is_empty()
31                && id.len() == 32
32                && id
33                    .chars()
34                    .all(|c| c.is_ascii_digit() || matches!(c, 'a'..='f'))
35        }
36        None => false,
37    }
38}

40/// Resolve the real `{table_name}-{table_id}` directory for a Data.db `path`,
41/// transparently walking up past a `snapshots/{snapshot_name}` layer.
42///
43/// For a normal SSTable the resolved directory is simply the Data.db file's
44/// parent. For a snapshot (`.../{table}-{id}/snapshots/{tag}/...-Data.db`) it is
45/// the grandparent-of-`snapshots` directory.
46///
47/// # Snapshot detection is STRUCTURAL and guarded
48///
49/// A path is only treated as a snapshot when BOTH hold:
50/// 1. the grandparent directory is literally named `snapshots`, AND
51/// 2. the directory ABOVE `snapshots` matches the `{table}-{32-hex}` shape
52///    ([`is_table_id_dir`]).
53///
54/// Guard #2 prevents misresolving an ORDINARY table that happens to live in a
55/// keyspace named `snapshots`
56/// (`.../data/snapshots/{table}-{id}/...-Data.db`): there the dir above
57/// `snapshots` is the keyspace root (e.g. `data`), which is NOT a table dir, so
58/// the path is correctly treated as a normal (non-snapshot) layout.
59///
60/// # Known limitation — ID-less snapshot dirs (follow-up #2415)
61///
62/// The [`is_table_id_dir`] guard requires the `-{32-hex}` id suffix, so it only
63/// fires for Cassandra-shaped ID-ful snapshots. CQLite's own write engine emits
64/// ID-LESS table dirs `{ks}/{table}/` (no uuid suffix — `writer/mod.rs`); a
65/// snapshot of one (`{ks}/{table}/snapshots/{tag}/...-Data.db`) does NOT match the
66/// guard, so the walk-up is skipped and header identity misparses as
67/// keyspace=`snapshots`, table=`{tag}`. This is inherently unresolvable from the
68/// path alone — an ID-less snapshot and an ordinary table in a keyspace literally
69/// named `snapshots` are structurally identical. The robust fix (threading
70/// authoritative keyspace/table into the reader) is tracked as #2415. See the
71/// `idless_snapshot_currently_unresolved_pending_followup` test below.
72pub(crate) fn resolve_table_dir(path: &Path) -> Option<&Path> {
73    // Parent of the Data.db file: `{table_name}-{table_id}` (normal) or
74    // `{snapshot_name}` (snapshot layout).
75    let parent = path.parent()?;
76
77    if let Some(maybe_snapshots) = parent.parent() {
78        let is_snapshots_dir =
79            maybe_snapshots.file_name().and_then(|n| n.to_str()) == Some("snapshots");
80        if is_snapshots_dir {
81            // Only walk up past `snapshots` when the directory above it is a real
82            // `{table}-{32-hex}` dir. Otherwise this is an ordinary table inside a
83            // keyspace literally named `snapshots`.
84            if let Some(real_table_dir) = maybe_snapshots.parent() {
85                let looks_like_table = real_table_dir
86                    .file_name()
87                    .and_then(|n| n.to_str())
88                    .map(is_table_id_dir)
89                    .unwrap_or(false);
90                if looks_like_table {
91                    return Some(real_table_dir);
92                }
93            }
94        }
95    }
96
97    Some(parent)
98}

100/// Extract the keyspace name for a Data.db `path` (snapshot-aware).
101///
102/// The keyspace is the parent of the resolved `{table_name}-{table_id}` dir.
103/// Returns `None` when the path is too shallow to contain a keyspace directory.
104pub(crate) fn extract_keyspace(path: &Path) -> Option<String> {
105    resolve_table_dir(path)
106        .and_then(|table_dir| table_dir.parent())
107        .and_then(|keyspace_dir| keyspace_dir.file_name())
108        .and_then(|n| n.to_str())
109        .map(|s| s.to_string())
110}
111
112/// Extract the table name for a Data.db `path` (snapshot-aware), stripping the
113/// trailing `-{table_id}` from the resolved directory when present.
114///
115/// When the resolved directory name has no `-{id}` suffix (unusual, e.g. a flat
116/// test dir) the whole directory name is returned unchanged. Returns `None` when
117/// the path has no directory component at all.
118pub(crate) fn extract_table_name(path: &Path) -> Option<String> {
119    let dir_name = resolve_table_dir(path)?.file_name()?.to_str()?;
120
121    // Split on the LAST hyphen so table names containing hyphens survive; only a
122    // trailing 32-hex id is stripped. Non-id suffixes are left intact.
123    match dir_name.rsplit_once('-') {
124        Some((table_name, id)) if is_table_id_suffix(id) => Some(table_name.to_string()),
125        _ => Some(dir_name.to_string()),
126    }
127}
128
129/// Return `true` when `id` is EXACTLY 32 lowercase hex chars (a Cassandra table
130/// id suffix). Used to decide whether a trailing `-{id}` should be stripped.
131fn is_table_id_suffix(id: &str) -> bool {
132    id.len() == 32
133        && id
134            .chars()
135            .all(|c| c.is_ascii_digit() || matches!(c, 'a'..='f'))
136}

201    #[test]
202    fn is_table_id_dir_shape() {
203        assert!(is_table_id_dir("t-9f3a1c2d4e5f6071829304a5b6c7d8e9"));
204        assert!(is_table_id_dir("my-t-00112233445566778899aabbccddeeff"));
205        // Wrong length.
206        assert!(!is_table_id_dir("t-9f3a"));
207        // Uppercase is not Cassandra's on-disk shape.
208        assert!(!is_table_id_dir("t-9F3A1C2D4E5F6071829304A5B6C7D8E9"));
209        // Non-hex.
210        assert!(!is_table_id_dir("t-zzzz1c2d4e5f6071829304a5b6c7d8e9"));
211        // No hyphen / empty table.
212        assert!(!is_table_id_dir("snapshots"));
213        assert!(!is_table_id_dir("-9f3a1c2d4e5f6071829304a5b6c7d8e9"));
214    }

244    }
