1//! Next-partition (successor) offset resolution for the single-partition seek
2//! window (issue #953 / #951; O(depth) local walk, issue #2058).
3//!
4//! The within-SSTable single-partition seek bounds its decompression window to
5//! exactly one partition's byte extent `[target_offset, successor_offset)` using
6//! authoritative index/trie layout metadata (never a heuristic boundary scan).
7//! Both helpers are gated `#[cfg(not(feature = "tombstones"))]` like the seek path
8//! they serve: their callers are `scan_single_partition_clustering`,
9//! `read_single_partition_for_compaction`, and `big_reverse_partition_rows`, which
10//! the `tombstones` build compiles out (it serves single-partition reads via a full
11//! scan + filter, not a seek).
12//!
13//! Split out of `partition_lookup.rs` (campsite / epic #1116) so the B4 key-cache
14//! wiring (issue #1570) does not push that file over the source-size threshold.
15
16#[cfg(not(feature = "tombstones"))]
17use super::SSTableReader;
18#[cfg(not(feature = "tombstones"))]
19use crate::{Error, Result};
20
21#[cfg(not(feature = "tombstones"))]
22impl SSTableReader {
23    /// Authoritatively resolve the UNCOMPRESSED `Data.db` offset of the partition
24    /// that immediately FOLLOWS the partition starting at `target_offset` (whose
25    /// partition key is `partition_key`), used to bound the within-SSTable seek's
26    /// decompression window to exactly one partition's byte extent (issue #953 / #951
27    /// MEDIUM).
28    ///
29    /// The successor's start offset is the partition's exclusive END: a partition
30    /// occupies `[target_offset, successor_offset)`, so decompressing the chunks
31    /// covering that half-open range materializes every byte of the target
32    /// partition (including a row/cell that spans multiple compression chunks)
33    /// without reading any of the next partition. This is authoritative metadata
34    /// (the index/trie's own partition layout), NOT a heuristic boundary scan.
35    ///
36    /// Returns:
37    /// - `Ok(Some(off))` — the next partition's start offset (`off > target_offset`).
38    /// - `Ok(None)` — `target_offset` is the LAST partition (no successor); the
39    ///   caller bounds the end with the authoritative data-section length.
40    ///
41    /// Resolution is per index format:
42    /// - **BTI (`da`)** — the successor is resolved by a SINGLE O(depth) strict-ceiling
43    ///   walk of the `Partitions.db` trie keyed on `partition_key`'s byte-comparable
44    ///   token ([`partition_successor_in_bti_slice`]). Because the trie stores
45    ///   partitions in byte-comparable key order — which for `Murmur3Partitioner`
46    ///   equals token order equals `Data.db` layout order — the trie IN-ORDER
47    ///   successor is exactly the OFFSET successor (the smallest partition start
48    ///   strictly greater than `target_offset`). This replaces the pre-#2058
49    ///   whole-trie DFS that enumerated + sorted EVERY partition offset into a
50    ///   `OnceLock` array on the first seek. A defensive `> target_offset` guard
51    ///   fails safe to `None` (data-section-length bound, a safe over-read that never
52    ///   truncates) should a resolved successor not exceed the target — a case the
53    ///   real-fixture oracle test proves does not occur (`tests/issue_2058_*`).
54    /// - **BIG (`nb`)** — `Index.db` `partition_entries` are sorted by key (==
55    ///   `Data.db` order); the successor is the smallest `data_offset` strictly
56    ///   greater than `target_offset`.
57    ///
58    /// [`partition_successor_in_bti_slice`]: crate::storage::sstable::bti::partition_successor_in_bti_slice
59    pub(crate) async fn successor_partition_offset(
60        &self,
61        target_offset: u64,
62        partition_key: &[u8],
63    ) -> Result<Option<u64>> {
64        if self.bti_partitions_db.is_some() {
65            return self.bti_successor_partition_offset(target_offset, partition_key);
66        }
67
68        // BIG (`nb`): scan the sorted Index.db entries for the smallest data_offset
69        // strictly greater than target_offset. `partition_entries` are emitted in
70        // key (== Data.db) order, but we take the min over `> target` defensively
71        // rather than rely on positional adjacency.
72        if let Some(index_reader) = &self.index_reader {
73            // Issue #2412 Stage 2: a lazily-opened reader defers the full parse to
74            // first use — this successor scan IS that first use. No-op for an
75            // eagerly-opened reader.
76            index_reader.ensure_materialized(&self.scan_cancel).await?;
77            let successor = index_reader
78                .get_partition_entries()
79                .iter()
80                .map(|e| e.data_offset)
81                .filter(|&o| o > target_offset)
82                .min();
83            return Ok(successor);
84        }
85
86        // No index available: cannot resolve a successor authoritatively.
87        Ok(None)
88    }
89
90    /// BTI (`da`) next-partition successor via the O(depth) local strict-ceiling walk
91    /// (issue #2058), resolving a WIDE successor's `RowsOffset` through `Rows.db`.
92    ///
93    /// The walk keys on `partition_key`'s byte-comparable token, so the successor it
94    /// returns is byte-identical to the pre-#2058 whole-trie DFS + sorted-offset
95    /// `partition_point(<= target_offset)`, for every partition (proven by the
96    /// real-fixture oracle test). No shared state / `OnceLock`, so concurrent point
97    /// reads on the same reader never race or double-walk.
98    fn bti_successor_partition_offset(
99        &self,
100        target_offset: u64,
101        partition_key: &[u8],
102    ) -> Result<Option<u64>> {
103        use crate::storage::sstable::bti::{
104            encode_partition_key_for_bti_trie_uncounted, partition_successor_in_bti_slice,
105            resolve_rows_db_entry_uncounted, BtiPartitionLocation,
106        };
107
108        let Some(partitions_db) = &self.bti_partitions_db else {
109            return Ok(None);
110        };
111
112        // Encode WITHOUT the C4 `KEY_HASH_CALLS` counter: a point read already hashed
113        // this exact key once (the candidate prune), and the one-hash-per-read
114        // invariant (issue #1575) must hold — the successor bound is not a new query
115        // key, it re-encodes the same one.
116        let encoded = encode_partition_key_for_bti_trie_uncounted(partition_key);
117
118        let location = partition_successor_in_bti_slice(partitions_db.as_slice(), &encoded)
119            .map_err(|e| {
120                Error::corruption(format!(
121                    "BTI Partitions.db next-partition successor walk failed while resolving the \
122                     seek end bound (key len={}): {e}",
123                    partition_key.len()
124                ))
125            })?;
126
127        let successor_offset = match location {
128            // Narrow successor: its Data.db start is the target's exclusive end.
129            Some(BtiPartitionLocation::DataOffset(off)) => Some(off),
130            // Wide successor: recover its Data.db start (`data_position`) via Rows.db.
131            // Uncounted so the L1 clustering-window `ROWS_DB_ENTRY_RESOLVES == 1`
132            // invariant (issue #1647) is not perturbed by this seek-bound resolve.
133            Some(BtiPartitionLocation::RowsOffset(rows_offset)) => {
134                let rows_db = self.bti_rows_db.as_ref().ok_or_else(|| {
135                    Error::corruption(format!(
136                        "BTI successor walk returned RowsOffset({rows_offset}) but this reader has \
137                         no Rows.db; the SSTable is structurally invalid (Rows.db is required for \
138                         wide partitions)."
139                    ))
140                })?;
141                let header =
142                    resolve_rows_db_entry_uncounted(rows_db.as_slice(), rows_offset as usize)
143                        .map_err(|e| {
144                            Error::corruption(format!(
145                        "BTI Rows.db row-index entry at RowsOffset({rows_offset}) is unreadable \
146                             while resolving the next-partition seek bound: {e}"
147                    ))
148                        })?;
149                Some(header.data_position)
150            }
151            // No successor: `partition_key` is the LAST partition.
152            None => None,
153        };
154
155        // Fail-safe (never a truncating bound): the resolved successor MUST start
156        // strictly after the target partition. If it does not (a pathological trie /
157        // token-collision shape the oracle test proves never occurs on real data),
158        // return `None` so the caller bounds the end with the authoritative
159        // data-section length — a safe over-read, never a truncation.
160        Ok(successor_offset.filter(|&off| off > target_offset))
161    }
162}
