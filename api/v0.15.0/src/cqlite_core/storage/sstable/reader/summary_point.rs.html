1//! Summary-guided bounded-interval BIG point lookup (issue #2412 §B, Stage 3).
2//!
3//! When a BIG (`nb`/uncompressed) reader was lazily opened over a usable `Summary.db`
4//! (design §A), a `WHERE pk = ?` point read resolves the partition's `Data.db` offset
5//! by:
6//! 1. binary-searching `Summary.db` for the sample interval covering the key
7//!    ([`SummaryReader::find_by_key`](crate::storage::sstable::summary_reader::SummaryReader::find_by_key)), then
8//! 2. reading/parsing exactly ONE `Index.db` interval (≤ `min_index_interval` entries)
9//!    from that position ([`lookup_key_in_interval`]).
10//!
11//! This REPLACES the pre-Stage-3 fallback that materialized the whole `Index.db` map
12//! (`IndexReader::ensure_materialized`) on the point-read path. The interval is bounded
13//! by two authoritative `Summary.db` sample positions, so an interval MISS is an
14//! authoritative absence — the whole-file `scan_for_key` oracle is not consulted for
15//! the common case (design §B; the FellBack/no-summary path keeps it, #1572).
16//!
17//! No-heuristics (issue #28): every seek offset and interval bound is an authoritative
18//! `Summary.db` sample position / `Index.db` entry framing — nothing is inferred from
19//! value bytes.
20
21use super::SSTableReader;
22use crate::storage::sstable::summary_reader::interval::lookup_key_in_interval;
23use crate::Result;
24
25impl SSTableReader {
26    /// Whether a BIG point lookup should resolve through ONE `Summary.db`-bounded
27    /// `Index.db` interval (design §B) rather than the full resident partition map.
28    ///
29    /// `true` iff BOTH hold:
30    /// - a usable `Summary.db` (at least one sample entry) bounds the walk, AND
31    /// - the lazy `Index.db` map has NOT already been materialized.
32    ///
33    /// The second clause preserves performance for a reader whose full map is already
34    /// resident (e.g. a prior full scan called `ensure_materialized`): the in-memory
35    /// O(1) probe beats a fresh bounded disk read, and the scan-path re-probes that
36    /// route through `lookup_partition_with_index` after materializing keep using the
37    /// resident map. It also means the FellBack (absent-`Summary.db`) reader — which
38    /// is eagerly materialized at open — never takes this path.
39    pub(super) fn should_use_summary_interval(&self) -> bool {
40        let summary_usable = self
41            .summary_reader
42            .as_ref()
43            .map(|s| !s.get_entries().is_empty())
44            .unwrap_or(false);
45        let index_lazy = self
46            .index_reader
47            .as_ref()
48            .map(|ir| !ir.is_materialized())
49            .unwrap_or(false);
50        summary_usable && index_lazy
51    }
52
53    /// Whether `partition_key`'s covering `Summary.db` interval is END-BOUNDED — i.e.
54    /// bounded ABOVE by a NEXT authoritative summary sample (`end_position.is_some()`),
55    /// not the last (read-to-EOF) interval.
56    ///
57    /// This is the #1572-safe gate for treating an interval MISS as an AUTHORITATIVE
58    /// absence (design §B, hardened): a tail-truncated `Index.db` (the #1572
59    /// degraded-input class — whole trailing entries dropped, or a mid-entry cut at
60    /// the end) can only lose the HIGHEST-token partitions, which live in the LAST
61    /// interval. An end-bounded interval `[sample_i, sample_{i+1})` is delimited by a
62    /// next sample whose position sits BEFORE any tail truncation, so its bytes are
63    /// intact and a miss within it is genuinely authoritative. The LAST interval
64    /// (`end_position == None`, read to EOF) could hide a dropped tail entry, so a miss
65    /// there keeps the whole-file `scan_for_key` fallback (`big_get_with_resolution`),
66    /// preserving the #1572 get/scan agreement on degraded inputs.
67    ///
68    /// A pure in-memory binary search (no disk read) — `big_get_with_resolution`
69    /// consults it only on an interval miss to classify the absence.
70    pub(super) fn covering_interval_is_end_bounded(&self, partition_key: &[u8]) -> bool {
71        self.summary_reader
72            .as_ref()
73            .and_then(|s| s.find_by_key(partition_key))
74            .map(|iv| iv.end_position.is_some())
75            .unwrap_or(false)
76    }
77
78    /// Resolve `partition_key` to its uncompressed `Data.db` offset via the
79    /// `Summary.db`-guided bounded `Index.db` interval (issue #2412 §B).
80    ///
81    /// Returns:
82    /// - `Ok(Some((offset, size)))` — an exact partition entry found within the one
83    ///   bounded interval (`size` is `0` — BIG `Index.db` stores no partition size —
84    ///   consistent with the resident-map path; callers parse forward from `offset`).
85    /// - `Ok(None)` — the key is genuinely absent from the interval between two
86    ///   authoritative `Summary.db` samples: an AUTHORITATIVE absence. The BIG point
87    ///   path (`big_get_with_resolution`) treats this as a definitive "not found"
88    ///   without a whole-file `scan_for_key` (design §B).
89    ///
90    /// Records exactly one real probe (`INDEX_PROBES`, mirroring the resident-map
91    /// path's accounting) plus one bounded interval parse
92    /// (`cqlite.sstable.index_interval_parses_total`, emitted inside
93    /// [`lookup_key_in_interval`], design §F) — never a full `index_parses_total`.
94    ///
95    /// Precondition: [`Self::should_use_summary_interval`] holds. When the summary /
96    /// index handles are unexpectedly absent (they should not be, given the
97    /// precondition), returns `Ok(None)` rather than fabricating a hit.
98    pub(super) async fn lookup_partition_via_summary_interval(
99        &self,
100        partition_key: &[u8],
101    ) -> Result<Option<(u64, u32)>> {
102        use crate::observability::{self as obs, catalog};
103
104        let format = self.sstable_format_label();
105        let (Some(summary), Some(index_reader)) =
106            (self.summary_reader.as_ref(), self.index_reader.as_ref())
107        else {
108            return Ok(None);
109        };
110        let Some(interval) = summary.find_by_key(partition_key) else {
111            // Empty summary — the precondition rules this out, but never fabricate.
112            return Ok(None);
113        };
114        let header = summary.get_header();
115        let min_index_interval = header.min_index_interval;
116        // Roborev job 1709 (High): the entry cap for the read-to-EOF LAST interval
117        // must be downsampling-aware, so `sampling_level` is threaded through —
118        // never a guess, the already-parsed header field (design doc §A1/no-heuristics).
119        let sampling_level = header.sampling_level;
120
121        // One real bounded interval probe (not a B4 cache hit): mirror the
122        // resident-map path's `INDEX_PROBES` accounting so a repeated read served
123        // from the B4 cache still records zero probes. The distinct
124        // `index_interval_parses_total` is emitted inside `lookup_key_in_interval`.
125        crate::storage::sstable::read_work_counters::record_index_probe();
126
127        // The reader's CURRENT (possibly #2383-rebound) Index.db path (issue #2356
128        // roborev): a lazy warm reader rebound across a snapshot teardown must open the
129        // live hardlink here, not the dead open-time path (#2352 class).
130        let index_db_path = index_reader.index_path();
131        let lookup = lookup_key_in_interval(
132            &index_db_path,
133            interval,
134            partition_key,
135            min_index_interval,
136            sampling_level,
137        )
138        .await?;
139
140        match lookup.entry {
141            Some(entry) => {
142                obs::add_counter(
143                    catalog::READ_PARTITION_LOOKUP,
144                    1,
145                    &[
146                        (catalog::attr::RESULT, "hit".into()),
147                        (catalog::attr::LOOKUP_ROUTE, "index".into()),
148                        (catalog::attr::SSTABLE_FORMAT, format.into()),
149                    ],
150                );
151                // Issue #2059: cache this present-key resolution (keyed on the
152                // reader's generation identity) so a repeat point read of the same
153                // key skips BOTH the summary search AND the bounded `Index.db`
154                // interval parse (post-#2412 this is the load-bearing latency win —
155                // a hit touches ZERO interval parses). BIG `Index.db` records no size.
156                self.key_cache_insert(
157                    partition_key,
158                    crate::storage::cache::PartitionLoc::new(entry.data_offset, entry.data_size),
159                );
160                Ok(Some((entry.data_offset, entry.data_size)))
161            }
162            None => {
163                obs::add_counter(
164                    catalog::READ_PARTITION_LOOKUP,
165                    1,
166                    &[
167                        (catalog::attr::RESULT, "miss".into()),
168                        (catalog::attr::LOOKUP_ROUTE, "index".into()),
169                        (catalog::attr::SSTABLE_FORMAT, format.into()),
170                    ],
171                );
172                // Authoritative absence between two summary samples — the caller does
173                // NOT fall back to a whole-file scan_for_key (design §B).
174                Ok(None)
175            }
176        }
177    }
