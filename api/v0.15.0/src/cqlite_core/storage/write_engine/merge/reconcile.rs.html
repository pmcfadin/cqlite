1//! Clustering-group reconciliation kernel (issue #945 decomposition).
2//!
3//! `KWayMerger::reconcile_cluster_with_overlap_counted` used to be a single
4//! ~470-line function. This module hosts the same logic decomposed into named,
5//! separately testable steps threaded through a [`ReconcileState`] accumulator.
6//! The orchestrator in `mod.rs` calls the steps **in a fixed, parity-load-bearing
7//! order** — see [`ReconcileState`] for the contract. This is a pure refactor:
8//! the per-step bodies are lifted verbatim from the original `// Step N` blocks,
9//! so compaction output (and the #1037 purge tally) is byte-for-byte identical.
10//!
11//! ## Step order (do not reorder — parity-critical, Cassandra `f66fa14f`)
12//!
13//! 1. [`fold_row_deletions`](ReconcileState::fold_row_deletions) — Step 1:
14//!    effective row deletion (max `deletion_time` + paired LDT), carried
15//!    complex/range deletions, key/run_index.
16//! 2. [`resolve_cell_winners`](ReconcileState::resolve_cell_winners) — Step 2:
17//!    per-`(column, cell_path)` last-write-wins winner resolution.
18//! 3. [`apply_complex_deletions`](ReconcileState::apply_complex_deletions) —
19//!    Step 2b: strict-supersede + shadow-before-purge. MUST run before the
20//!    row-tombstone / gc-grace filters so a surviving complex deletion cannot
21//!    resurrect a covered element on a later purge.
22//! 4. [`shadow_by_row_deletion`](ReconcileState::shadow_by_row_deletion) —
23//!    Step 3: drop cells with `ts <= row_del`.
24//! 5. [`filter_dropped_columns`](ReconcileState::filter_dropped_columns) —
25//!    Step 3b: drop cells `ts <= drop_time`, capturing the pre-purge
26//!    `had_data_before` phantom-row guard state.
27//! 5b. [`expire_ttl_cells`](ReconcileState::expire_ttl_cells) — Step 3b′
28//!    (#1382): turn each expired live TTL cell (`localDeletionTime < now_secs`)
29//!    into a cell tombstone so Step 3c purges it under the same gc/overlap gate.
30//! 6. [`purge_gc_grace`](ReconcileState::purge_gc_grace) — Step 3c: drop
31//!    purgeable tombstones (overlap-safe), tallying genuine purges (#1037).
32//! 7. [`build`](ReconcileState::build) — Step 4: phantom-row guard + emit the
33//!    merged `MergeEntry`.
34

49/// Per-cell reconcile key: `(column, cell_path)` so each element of a multi-cell
50/// column reconciles independently (epic #899). Simple cells have
51/// `cell_path == None`.
52type CellKey = (String, Option<Vec<u8>>);
53
54/// In-flight working set for reconciling one clustering-key group.
55///
56/// Built by [`Self::new`], then mutated through the step methods in the fixed
57/// order documented at the module level. The load-bearing invariants the
58/// original ~470-line function depended on are preserved:
59///
60/// - **Step 2b before Step 3/3c** — a surviving complex deletion shadows its
61///   covered elements before any marker is purged (`f66fa14f`).
62/// - **`had_data_before` captured pre-purge** — recorded in
63///   [`Self::filter_dropped_columns`] (after row-tombstone + dropped-column
64///   filtering) and consulted in [`Self::build`] AFTER the gc-grace purge, so a
65///   clustered row whose only non-key data is a purgeable cell tombstone is
66///   recognized as data-less and emits nothing instead of a phantom key-only
67///   live row (#921 finding 3).
68pub(super) struct ReconcileState {
69    /// Clustering key shared by every entry in the group (carry-through).
70    clustering_key: Option<ClusteringKey>,
71    /// Partition key, taken from the first entry seen.
72    key: Option<DecoratedKey>,
73    /// Lowest run_index seen (newest file) for stable downstream ordering.
74    run_index: usize,
75    /// Effective row deletion timestamp (`markedForDeleteAt`, micros).
76    row_del: Option<i64>,
77    /// `localDeletionTime` (GC-clock seconds) paired with the winning `row_del`.
78    row_del_ldt: i32,
79    /// First-seen cell-key order for deterministic output.
80    order: Vec<CellKey>,
81    /// Winning cell per `(column, cell_path)`.
82    winners: HashMap<CellKey, CellData>,
83    /// Carried complex (collection / UDT) deletion markers.
84    complex_deletions: Vec<ComplexDeletion>,
85    /// Carried range deletion (timestamp-max preserved).
86    range_deletion: Option<RangeTombstone>,
87    /// Cells surviving row-tombstone shadowing (Step 3 output).
88    after_row_del: Vec<CellData>,
89    /// Cells surviving dropped-column filtering (Step 3b output), then mutated
90    /// by the gc-grace purge.
91    surviving: Vec<CellData>,
92    /// Whether the row held non-key data BEFORE the gc-grace purge (phantom-row
93    /// guard, captured pre-purge).
94    had_data_before: bool,
95}

98    /// Initialize an empty accumulator for the group sharing `clustering_key`.
99    pub(super) fn new(clustering_key: Option<ClusteringKey>) -> Self {
100        Self {
101            clustering_key,
102            key: None,
103            run_index: usize::MAX,
104            row_del: None,
105            row_del_ldt: 0,
106            order: Vec::new(),
107            winners: HashMap::new(),
108            complex_deletions: Vec::new(),
109            range_deletion: None,
110            after_row_del: Vec::new(),
111            surviving: Vec::new(),
112            had_data_before: false,
113        }
114    }

116    /// True once an input entry has been folded (i.e. the group is non-empty).
117    /// Mirrors the original `let key = key?` early-out: an empty group emits
118    /// nothing.
119    pub(super) fn has_key(&self) -> bool {
120        self.key.is_some()
121    }

123    /// Step 1 — fold the per-entry deletion metadata: effective row deletion
124    /// (max `deletion_time` with its PAIRED source LDT, #873), the carried
125    /// complex-deletion union (first-seen, de-duplicated) and the
126    /// timestamp-max range deletion, plus the carry-through key / run_index.
127    ///
128    /// The per-cell winner resolution (the `RowData::Live` arm) is split out
129    /// into [`Self::resolve_cell_winners`]; both passes iterate `cluster_rows`
130    /// in heap-routing order, and the accumulators are independent, so the
131    /// result is identical to the original single combined loop.
132    pub(super) fn fold_row_deletions(&mut self, cluster_rows: &[MergeEntry]) {
133        for entry in cluster_rows {
134            if self.key.is_none() {
135                self.key = Some(entry.key.clone());
136            }
137            self.run_index = self.run_index.min(entry.run_index);
138
139            for cd in &entry.complex_deletions {
140                if !self.complex_deletions.contains(cd) {
141                    self.complex_deletions.push(cd.clone());
142                }
143            }
144            if let Some(rd) = &entry.range_deletion {
145                let replace = match &self.range_deletion {
146                    None => true,
147                    Some(current) => rd.deletion_time > current.deletion_time,
148                };
149                if replace {
150                    self.range_deletion = Some(rd.clone());
151                }
152            }
153
154            // Issue #932: a coexisting row deletion carried on a LIVE entry (a
155            // read-back row that held BOTH a row tombstone and surviving newer
156            // cells) contributes to the winning row deletion exactly as a
157            // standalone `RowData::Tombstone` does — capture its paired LDT when
158            // it sets the new max so the rebuilt deletion preserves its
159            // wall-clock `localDeletionTime` (#873).
160            if let Some((del_ts, del_ldt)) = entry.row_deletion {
161                if self.row_del.is_none_or(|d| del_ts > d) {
162                    self.row_del = Some(del_ts);
163                    self.row_del_ldt = del_ldt;
164                }
165            }
166
167            if let RowData::Tombstone {
168                deletion_time,
169                local_deletion_time,
170            } = &entry.row_data
171            {
172                // When this tombstone's deletion_time becomes (or sets) the new
173                // max, capture its paired LDT too so the winning tombstone's
174                // source `localDeletionTime` survives reconciliation (#873).
175                if self.row_del.is_none_or(|d| *deletion_time > d) {
176                    self.row_del = Some(*deletion_time);
177                    self.row_del_ldt = *local_deletion_time;
178                }
179            }
180        }
181    }
182
183    /// Step 2 — per-cell winner resolution keyed by `(column, cell_path)` so
184    /// each element of a multi-cell column reconciles independently (epic #899).
185    /// Preserves first-seen key order while resolving winners in `winners`.
186    ///
187    /// `cluster_rows` is in heap-routing order (run_index ascending within equal
188    /// keys), so when two cells tie on both timestamp and liveness the
189    /// first-seen (newer file) is kept (see
190    /// [`reconcile_rules::cell_wins`], the shared tie-break rule).
191    pub(super) fn resolve_cell_winners(&mut self, cluster_rows: &[MergeEntry]) {
192        for entry in cluster_rows {
193            if let RowData::Live { cells } = &entry.row_data {
194                for cell in cells {
195                    let cell_key: CellKey = (cell.column.clone(), cell.cell_path.clone());
196                    // Issue #1665: the HashMap `entry()` API hashes ONCE on the
197                    // vacant path (the old `get()`+`insert()` hashed twice) and
198                    // lets `order` reuse the slot's owned key. Output is
199                    // byte-identical: the vacant/occupied arms below are the same
200                    // two branches as the former `None`/`Some` match, and the
201                    // `order.push` still precedes the `insert` so first-seen order
202                    // is preserved for the equal-timestamp tie-break.
203                    match self.winners.entry(cell_key) {
204                        Entry::Vacant(slot) => {
205                            self.order.push(slot.key().clone());
206                            slot.insert(cell.clone());
207                        }
208                        Entry::Occupied(mut slot) => {
209                            // Higher timestamp wins. At EQUAL timestamp a cell
210                            // DELETION (tombstone) beats a LIVE or EXPIRING
211                            // (TTL) cell, decided BEFORE any localDeletionTime
212                            // compare (parity Cassandra `a62c749`,
213                            // `Cells#reconcile`; issue #848 / #498). At equal ts
214                            // + equal deletion-status, keep the first-seen
215                            // (newer file) winner. The tie-break is the SHARED
216                            // [`reconcile_rules::cell_wins`] rule (issue #947),
217                            // also used by the flush/write path.
218                            if reconcile_rules::cell_wins(cell, slot.get()) {
219                                slot.insert(cell.clone());
220                            }
221                        }
222                    }
223                }
224            }
225        }
226    }

228    /// Step 2b — COMPLEX-DELETION reconcile: strict-supersede + shadow-before-purge
229    /// (issue #887). Runs AFTER per-cell winner resolution and BEFORE the
230    /// row-tombstone / dropped-column filters, so an element shadowed by a
231    /// surviving complex deletion cannot be resurrected by a later purge of the
232    /// deletion marker.
233    ///
234    /// 1. STRICT SUPERSEDE (parity Cassandra `bd244649`): per complex column the
235    ///    ACTIVE deletion is the one with the greatest `marked_for_delete_at`. A
236    ///    merged deletion supersedes the active one only when its mfda is STRICTLY
237    ///    GREATER — EQUAL timestamps do NOT supersede. Reduce the carried
238    ///    first-seen union to ONE deletion per column NAME (the strict max),
239    ///    preserving first-seen column order for deterministic output. Matched BY
240    ///    NAME (`ComplexDeletion.column`), never by header identity (consistent
241    ///    with #888's match-by-name substrate).
242    ///
243    /// 2. SHADOW BEFORE PURGE (parity Cassandra `f66fa14f`): for the surviving
244    ///    deletion on a column, drop every per-element winner of THAT column
245    ///    (matched by NAME, only complex ELEMENTS carrying a `cell_path`) whose own
246    ///    timestamp is `<= marked_for_delete_at`. The boundary mirrors the
247    ///    cell-vs-deletion rule (#498): an element with ts STRICTLY GREATER than
248    ///    mfda survives; `<=` is shadowed. A simple cell sharing the name is never
249    ///    collapsed by a complex marker.
250    pub(super) fn apply_complex_deletions(&mut self) {
251        if !self.complex_deletions.is_empty() {
252            // Strict-supersede: collapse to the max-mfda deletion per column name.
253            let mut active: HashMap<String, ComplexDeletion> = HashMap::new();
254            let mut active_order: Vec<String> = Vec::new();
255            for cd in self.complex_deletions.drain(..) {
256                match active.get_mut(&cd.column) {
257                    None => {
258                        active_order.push(cd.column.clone());
259                        active.insert(cd.column.clone(), cd);
260                    }
261                    // STRICTLY GREATER supersedes; equal/lesser does NOT
262                    // (bd244649). Shared rule (issue #947).
263                    Some(existing)
264                        if reconcile_rules::complex_deletion_supersedes(
265                            cd.marked_for_delete_at,
266                            existing.marked_for_delete_at,
267                        ) =>
268                    {
269                        *existing = cd;
270                    }
271                    Some(_) => {}
272                }
273            }
274
275            // Shadow-before-purge: drop covered per-element winners of each column
276            // whose ts <= mfda (matched by NAME), so purging the marker later cannot
277            // resurrect them (f66fa14f). Elements strictly newer than mfda survive.
278            for column in &active_order {
279                if let Some(cd) = active.get(column) {
280                    let mfda = cd.marked_for_delete_at;
281                    let winners = &mut self.winners;
282                    self.order.retain(|cell_key| {
283                        let (cell_column, cell_path) = cell_key;
284                        // Only complex elements (those with a cell_path) of THIS
285                        // column are candidates for shadowing.
286                        if cell_column != column || cell_path.is_none() {
287                            return true;
288                        }
289                        match winners.get(cell_key) {
290                            // ts > mfda survives; ts <= mfda is shadowed (purged).
291                            // Shared shadow-before-purge boundary (issue #947).
292                            Some(cell)
293                                if reconcile_rules::element_survives_complex_deletion(
294                                    cell.timestamp,
295                                    mfda,
296                                ) =>
297                            {
298                                true
299                            }
300                            Some(_) => {
301                                winners.remove(cell_key);
302                                false
303                            }
304                            None => false,
305                        }
306                    });
307                }
308            }
309
310            // Rebuild the carried union as the surviving per-column deletions in
311            // first-seen column order.
312            self.complex_deletions = active_order
313                .into_iter()
314                .filter_map(|column| active.remove(&column))
315                .collect();
316        }
317    }

319    /// Step 3 — apply row-tombstone shadowing per cell. A cell whose timestamp is
320    /// `<= row_del` is shadowed (`<=` lets the tombstone win at equal ts, #498).
321    /// Cells written strictly after `row_del` survive. This shadowing applies to
322    /// cell tombstones too: a row tombstone at ts=T supersedes a cell tombstone at
323    /// ts<=T (real Cassandra semantics).
324    ///
325    /// COMPLEX-DELETION shadowing has already been applied in Step 2b above,
326    /// BEFORE this row-tombstone filter, so a surviving complex deletion cannot
327    /// resurrect a covered element on a later purge (issue #887, parity
328    /// `f66fa14f`). The output `after_row_del` is kept so the dropped-column
329    /// stage (Step 3b) can tell whether its purge is what emptied the row of real
330    /// data (phantom-row guard).
331    pub(super) fn shadow_by_row_deletion(&mut self, purges: &mut PurgeCounts) {
332        let row_del = self.row_del;
333        // Issue #2163 (roborev r5): clustering-key pseudo-cells are intentionally
334        // RETAINED in the cell list for read-back (see `extract_clustering_key`
335        // and the `had_data_before` computation in `filter_dropped_columns`
336        // below) — they are not real data, so shadowing one must not inflate
337        // `tombstones_suppressed`. Build the SAME `is_data_cell` exclusion
338        // `filter_dropped_columns`/`build` already use, computed here (before the
339        // mutable `winners` borrow) so a clustering-key cell shadowed by the row
340        // tombstone is excluded from the count.
341        let ck_names: HashSet<&str> = self
342            .clustering_key
343            .as_ref()
344            .map(|ck| ck.columns.iter().map(|(n, _)| n.as_str()).collect())
345            .unwrap_or_default();
346        let winners = &mut self.winners;
347        self.after_row_del = std::mem::take(&mut self.order)
348            .into_iter()
349            .filter_map(|cell_key| winners.remove(&cell_key))
350            .filter(|cell| match row_del {
351                Some(d) => {
352                    let survives = cell.timestamp > d;
353                    // Issue #2163: a LIVE DATA cell dropped here was SHADOWED by
354                    // the row tombstone (suppressed) — count it, distinct from a
355                    // gc/overlap-safe purge. A cell that is itself a tombstone,
356                    // or a clustering-key pseudo-cell (retained for read-back,
357                    // never real data), is not "live data suppressed" and is not
358                    // counted.
359                    if !survives
360                        && !KWayMerger::is_cell_tombstone(cell)
361                        && !ck_names.contains(cell.column.as_str())
362                    {
363                        purges.suppressed += 1;
364                    }
365                    survives
366                }
367                None => true,
368            })
369            .collect();
370    }

372    /// Step 3b — dropped-column filtering (Cassandra `cb34ad47`,
373    /// `compaction.purge`). A column dropped at `drop_time` discards every cell
374    /// whose `timestamp <= drop_time`; a cell written strictly after the drop
375    /// (the column was re-added) survives. Scoped per column via the
376    /// `dropped_columns` map (#904 plumbing, #847 filter).
377    ///
378    /// EXACT PER-CELL GRANULARITY (#922): `cell.timestamp` is the cell's OWN
379    /// writetime, so a row mixing a pre-drop dropped-column cell with a post-drop
380    /// cell of ANOTHER column purges the dropped cell by its own writetime while
381    /// the sibling survives by its.
382    ///
383    /// Also captures the pre-purge `had_data_before` for the phantom-row guard:
384    /// clustering-key columns are intentionally left in the cell list (see
385    /// `extract_clustering_key`) for read-back, so we must remember whether the
386    /// row HAD non-key data before the gc-grace purge runs (the
387    /// `has_data_after` / `purged_to_empty` determination is DEFERRED to
388    /// [`Self::build`], after Step 3c — #921 finding 3).
389    pub(super) fn filter_dropped_columns(&mut self, dropped_columns: &HashMap<String, i64>) {
390        // Issue #1665: capture the pre-purge phantom-row guard state FIRST — while
391        // `after_row_del` is still populated — so we can then MOVE it into the
392        // survivor filter instead of deep-cloning every survivor. `after_row_del`
393        // is DEAD after this method (VERIFIED against the reconcile step sequence
394        // in `merge/mod.rs`: shadow_by_row_deletion → filter_dropped_columns →
395        // expire_ttl_cells → purge_gc_grace → build, none of which read it again,
396        // and `build` destructures `self` with `..`), so `mem::take` is safe and
397        // output is byte-identical (same `had_data_before`, same `surviving`
398        // contents and order — only the survivor clone is eliminated).
399        let ck_names: HashSet<&str> = self
400            .clustering_key
401            .as_ref()
402            .map(|ck| ck.columns.iter().map(|(n, _)| n.as_str()).collect())
403            .unwrap_or_default();
404        let is_data_cell = |cell: &CellData| !ck_names.contains(cell.column.as_str());
405        self.had_data_before = self.after_row_del.iter().any(is_data_cell);
406        drop(ck_names);
407
408        self.surviving = std::mem::take(&mut self.after_row_del)
409            .into_iter()
410            .filter(|cell| match dropped_columns.get(&cell.column) {
411                Some(drop_time) => cell.timestamp > *drop_time,
412                None => true,
413            })
414            .collect();
415    }

417    /// Step 3b′ — TTL EXPIRY (issue #1382, parity Cassandra `ExpiringCell`
418    /// / `TTLExpiryTest`). Runs AFTER dropped-column filtering and BEFORE the
419    /// gc-grace purge (Step 3c) so an expired cell is first turned into a cell
420    /// tombstone and then purged by the SAME gc/overlap gate as any other cell
421    /// tombstone — matching Cassandra, which treats an expiring cell whose
422    /// `localDeletionTime` is past as a tombstone with `localDeletionTime =`
423    /// the expiry instant and the cell's own `markedForDeleteAt` unchanged.
424    ///
425    /// NO-HEURISTICS (#28): a cell is "expiring" ONLY when the reader surfaced
426    /// its authoritative expiry metadata — BOTH `ttl` (Some) AND
427    /// `local_deletion_time` (Some) — from the on-disk cell (never inferred
428    /// from the value's byte shape). `now_secs` is the compaction's pinned
429    /// evaluation instant (`None` disables expiry: a strict no-op).
430    ///
431    /// A cell that is ALREADY a tombstone (`is_cell_tombstone`, which also covers
432    /// an `is_deleted` complex-element tombstone) is left untouched — it has no
433    /// live value to expire.
434    ///
435    /// COMPLEX / COLLECTION / UDT ELEMENTS (#1537): an expiring ELEMENT of a
436    /// non-frozen complex column (`is_complex_element`) is expired here too, not
437    /// just simple cells (the #1382 scope gap — F2). Cassandra
438    /// `AbstractCell.purge(DeletionPurger, nowInSec)` treats an expired expiring
439    /// cell UNIFORMLY whether simple or complex: it converts it to
440    /// `BufferCell.tombstone(column, timestamp(), localDeletionTime() - ttl(), path())` —
441    /// PRESERVING the element's cell path and setting the tombstone's
442    /// `localDeletionTime` to `ldt - ttl` (the cell's creation time). So an
443    /// expired element becomes an element-level tombstone at the SAME path,
444    /// carrying its own IS_DELETED flag (epic #899), which the per-element writer
445    /// emits as an element tombstone.
446    ///
447    /// LDT NORMALIZATION (#921 finding 2): on-disk `localDeletionTime` is an
448    /// UNSIGNED 32-bit GC-clock second count carried as a wrapped `i32`;
449    /// reinterpret the bits as unsigned (`i64::from(ldt as u32)`) before the
450    /// compare so a far-future expiry is not treated as already-past.
451    pub(super) fn expire_ttl_cells(&mut self, now_secs: Option<i64>) {
452        let Some(now) = now_secs else {
453            return; // Expiry disabled — strict no-op.
454        };
455        for cell in &mut self.surviving {
456            // An existing tombstone (simple cell tombstone OR a complex-element
457            // tombstone via `is_deleted`) has no live value to expire.
458            if KWayMerger::is_cell_tombstone(cell) {
459                continue;
460            }
461            let (Some(ttl), Some(ldt)) = (cell.ttl, cell.local_deletion_time) else {
462                continue; // Not an authoritative expiring cell.
463            };
464            // PARITY — off-by-one at `now == ldt` (issue #1537). Cassandra
465            // `Cell.isLive(nowInSec)` (`Cell.java`) is LIVE iff
466            // `nowInSec < localDeletionTime`, so an expiring cell is EXPIRED iff
467            // `localDeletionTime <= nowInSec`. Keep the cell LIVE only when its
468            // expiry instant (unsigned GC-clock seconds) is STRICTLY in the
469            // future; `ldt == now` means already expired.
470            if i64::from(ldt as u32) > now {
471                continue; // Not yet expired (strictly-future expiry).
472            }
473            // PARITY — tombstone `localDeletionTime` == `ldt - ttl` (issue #1537).
474            // Cassandra `AbstractCell.purge` converts an expired expiring cell via
475            // `BufferCell.tombstone(column, timestamp(), localDeletionTime() - ttl(),
476            // path())`. CQLite stores an expiring cell's on-disk
477            // `local_deletion_time` as the EXPIRY instant (`now + ttl`), so the
478            // parity-correct tombstone value is `ldt - ttl` = the cell's
479            // creation-time-in-seconds. gc grace is therefore measured from the
480            // cell's creation time (see the comment in `AbstractCell.purge`).
481            //
482            // GUARD: this cannot underflow for a VALID expiring cell — `ldt`
483            // (= creation + ttl) is always `>= ttl`. A `saturating_sub` on the
484            // unsigned seconds keeps a malformed pair from ever panicking (it
485            // floors at 0, an ancient/purgeable tombstone) with no `unwrap`.
486            let creation_secs: u32 = (ldt as u32).saturating_sub(ttl);
487            let tombstone_ldt: i64 = i64::from(creation_secs);
488            // Convert the expired live cell into a (cell / complex-element)
489            // tombstone whose `localDeletionTime` is the creation-time instant
490            // (`ldt - ttl`) and whose `markedForDeleteAt` is the cell's own write
491            // timestamp (unchanged). Step 3c then purges it exactly like any other
492            // cell tombstone once its LDT is < gcBefore and the overlap gate
493            // allows. `ttl` is cleared: a tombstone carries no TTL.
494            cell.value = crate::types::Value::Tombstone(Box::new(crate::types::TombstoneInfo {
495                deletion_time: cell.timestamp,
496                tombstone_type: crate::types::TombstoneType::CellTombstone,
497                local_deletion_time: tombstone_ldt,
498                ttl: None,
499                range_start: None,
500                range_end: None,
501            }));
502            cell.ttl = None;
503            // A complex ELEMENT additionally carries its deletion via the
504            // authoritative IS_DELETED flag (epic #899). Set it so the per-element
505            // writer emits an element tombstone (preserving the element's
506            // cell_path, Cassandra `path()`) rather than a live element, and so the
507            // gc-grace purge's `cell.is_deleted` branch (Step 3c (a)) treats it as
508            // a purgeable complex-element tombstone. The cell path is untouched.
509            if cell.is_complex_element {
510                cell.is_deleted = true;
511            }
512            // PARITY: the converted tombstone's `localDeletionTime` is the
513            // creation-time instant (`ldt - ttl`), so update `cell.local_deletion_time`
514            // to match. `cell_effective_ldt` prefers this field, so both the
515            // gc-grace purge (Step 3c) and the writer surface the creation-time
516            // LDT — measuring gc grace from creation, per `AbstractCell.purge`.
517            // Stored as the wrapped `i32` GC-clock value (re-widened unsigned by
518            // `cell_effective_ldt`); `creation_secs` fits `i32` for any real
519            // creation time.
520            cell.local_deletion_time = Some(creation_secs as i32);
521        }
522    }
523
524    /// Step 3c — gc_grace / gcBefore tombstone PURGING (issue #845, parity
525    /// Cassandra `8d47ebb2`). A tombstone whose on-disk `localDeletionTime`
526    /// (GC-clock seconds) is STRICTLY LESS THAN `gcBefore` is purgeable and
527    /// dropped; one within grace (`>= gcBefore`) is retained. Runs AFTER the
528    /// complex-deletion shadow-before-purge (Step 2b) and the row-tombstone /
529    /// dropped-column filters above, so the now-redundant marker cannot
530    /// resurrect data within this set.
531    ///
532    /// OVERLAP SAFETY (#921 finding 1, #935): `gc_before_secs` is the EFFECTIVE
533    /// cutoff (caller collapses it to `None` for an unbounded partial compaction,
534    /// making this a strict no-op). With a bound, a tombstone is purged only when
535    /// BOTH its gc grace has elapsed AND its own deletion timestamp
536    /// (`markedForDeleteAt`) is STRICTLY LESS THAN `max_purgeable_timestamp` — so
537    /// it provably shadows nothing in a non-included overlapping SSTable.
538    ///
539    /// LDT NORMALIZATION (#921 finding 2): on-disk `localDeletionTime` is an
540    /// UNSIGNED 32-bit GC-clock second count carried as a wrapped `i32`;
541    /// reinterpret the bits as unsigned (`i64::from(ldt as u32)`) before every
542    /// compare so a far-future tombstone is not purged immediately. All three
543    /// sites below (cell / row / complex-deletion), plus the expiring-cell check
544    /// in `expire_ttl_cells`, MUST widen unsigned — a signed compare would treat
545    /// a value in `[2^31, 2^32)` (a negative `i32`, e.g. `i32::MIN == 2^31`) as
546    /// ANCIENT and purge a not-yet-expired far-future tombstone, resurrecting any
547    /// older cells it shadows. Pinned by `tests/issue_1386_wrapped_negative_ldt.rs`.
548    ///
549    /// DIVERGENCE FROM CASSANDRA — RECORDED DECISION (#1386): CQLite SILENTLY
550    /// reinterprets the LDT bits as unsigned (year-2106 GC-clock semantics) and
551    /// never flags the value as "suspect". Cassandra 5.0's deletion-time handling
552    /// additionally treats an out-of-range `localDeletionTime` as suspicious
553    /// (e.g. `Cell`/`DeletionTime` validation paths log / mark it). CQLite's
554    /// posture is deliberately reinterpret-unsigned WITHOUT suspect-marking: a
555    /// far-future tombstone round-trips and purges purely on the unsigned GC-clock
556    /// compare, matching Cassandra's PURGE DECISION while omitting its diagnostics.
557    ///
558    /// Each genuine gc/overlap-safe purge increments the matching `purges` field
559    /// (issue #1037); last-write-wins reconciliation collapse is NOT counted.
560    pub(super) fn purge_gc_grace(
561        &mut self,
562        gc_before_secs: Option<i64>,
563        max_purgeable_timestamp: i64,
564        purges: &mut PurgeCounts,
565    ) {
566        if let Some(gc_before) = gc_before_secs {
567            // (a) Cell tombstones: drop any purgeable simple cell tombstone (and
568            // purgeable complex-element tombstone) from the surviving set. A cell
569            // whose `local_deletion_time` is not surfaced (`None`) is conservative-
570            // ly RETAINED — we never purge on unknown LDT (no-heuristics mandate).
571            self.surviving.retain(|cell| {
572                if KWayMerger::is_cell_tombstone(cell) || cell.is_deleted {
573                    // #921 finding 1: a simple cell tombstone surfaces its LDT in
574                    // its `Value::Tombstone` payload, not `CellData.local_deletion_time`
575                    // (which the reader fills only for expiring cells). Consult both
576                    // via `cell_effective_ldt` so a purgeable cell tombstone is
577                    // actually purged here — matching the survivor pre-pass.
578                    let gc_purgeable = match KWayMerger::cell_effective_ldt(cell) {
579                        Some(ldt) => i64::from(ldt as u32) < gc_before,
580                        // Unknown LDT: never purge (no-heuristics mandate).
581                        None => false,
582                    };
583                    // #935 overlap gate: the cell tombstone's own write timestamp
584                    // (`markedForDeleteAt`) must be STRICTLY BELOW the min outside
585                    // timestamp to prove it shadows nothing in a non-included
586                    // overlapping SSTable. `i64::MAX` for a full compaction lets
587                    // every gc-purgeable tombstone through unchanged.
588                    let overlap_purgeable = cell.timestamp < max_purgeable_timestamp;
589                    // RETAIN unless BOTH gates allow the purge.
590                    let keep = !(gc_purgeable && overlap_purgeable);
591                    if !keep {
592                        // True gc/overlap-safe purge of a cell tombstone (simple
593                        // or complex-element), issue #1037.
594                        purges.cell_tombstones += 1;
595                    }
596                    keep
597                } else {
598                    true
599                }
600            });
601
602            // (b) Row tombstone: a purgeable row deletion is dropped so no
603            // tombstone entry is emitted for it. Row-tombstone shadowing already
604            // ran above (`after_row_del`), so cells it covered are gone.
605            //
606            // UNKNOWN-LDT RETENTION (#921 finding 2): `row_del_ldt == 0` is the
607            // "LDT not surfaced" placeholder; treat 0 as UNKNOWN and RETAIN the
608            // row tombstone. Only a real, non-zero, surfaced LDT strictly below
609            // `gcBefore` purges.
610            //
611            // #935 overlap gate: also require the row deletion's own timestamp
612            // (`markedForDeleteAt` = `row_del`) to be STRICTLY BELOW the min
613            // outside timestamp so it shadows nothing in a non-included
614            // overlapping SSTable. `i64::MAX` (full compaction) is a no-op.
615            if self.row_del_ldt != 0
616                && i64::from(self.row_del_ldt as u32) < gc_before
617                && self.row_del.is_some_and(|d| d < max_purgeable_timestamp)
618            {
619                self.row_del = None;
620                // True gc/overlap-safe purge of a row tombstone (issue #1037).
621                purges.row_tombstones += 1;
622            }
623
624            // (c) Complex-deletion markers: drop each purgeable marker. The
625            // strict-supersede reduction + shadow-before-purge already ran in
626            // Step 2b, so a covered element is gone before its marker is purged.
627            // #935 overlap gate: also require the marker's `marked_for_delete_at`
628            // to be STRICTLY BELOW the min outside timestamp.
629            self.complex_deletions.retain(|cd| {
630                let gc_purgeable = i64::from(cd.local_deletion_time as u32) < gc_before;
631                let overlap_purgeable = cd.marked_for_delete_at < max_purgeable_timestamp;
632                let keep = !(gc_purgeable && overlap_purgeable);
633                if !keep {
634                    // True gc/overlap-safe purge of a complex-deletion marker
635                    // (issue #1037).
636                    purges.complex_deletions += 1;
637                }
638                keep
639            });
640        }
641    }
642
643    /// Step 4 — build the merged result, applying the phantom key-only-row guard
644    /// (#921 finding 3): recompute `purged_to_empty` AFTER the gc-grace
645    /// cell-tombstone purge, so a CLUSTERED row whose only non-key data was a
646    /// purgeable cell tombstone emits NOTHING instead of a phantom key-only live
647    /// row. A row that was always key-only (a genuine row marker) keeps
648    /// `had_data_before == false` and is preserved.
649    ///
650    /// Returns `None` for an empty group (the original `let key = key?`
651    /// early-out) or a truly absent row.
652    pub(super) fn build(self, purges: &mut PurgeCounts) -> Option<MergeEntry> {
653        let ReconcileState {
654            clustering_key,
655            key,
656            run_index,
657            row_del,
658            row_del_ldt,
659            complex_deletions,
660            range_deletion,
661            surviving,
662            had_data_before,
663            ..
664        } = self;
665
666        let key = key?; // empty group => nothing to emit
667
668        let ck_names: HashSet<&str> = clustering_key
669            .as_ref()
670            .map(|ck| ck.columns.iter().map(|(n, _)| n.as_str()).collect())
671            .unwrap_or_default();
672        let is_data_cell = |cell: &CellData| !ck_names.contains(cell.column.as_str());
673        let has_data_after = surviving.iter().any(is_data_cell);
674        let purged_to_empty = had_data_before && !has_data_after;
675        drop(ck_names);
676
677        // Issue #2163 (roborev r7): a cell tombstone (simple or complex-element,
678        // the SAME predicate `purge_gc_grace`'s retain closure uses) that
679        // SURVIVES into `surviving` — whether because gc-grace/overlap kept it,
680        // or because purging was inactive for this merge entirely (no
681        // `gc_before_secs`, e.g. an unbounded partial compaction) — is a marker
682        // RETAINED into the output, matching the row/range/partition tombstone
683        // "emitted" contract established above. Counted here from the FINAL
684        // surviving set (not inside `purge_gc_grace`'s conditional retain) so a
685        // merge with purging disabled still counts its retained cell
686        // tombstones; a cell tombstone always satisfies `is_data_cell` (it is
687        // never a clustering-key pseudo-cell), so its presence here already
688        // implies `purged_to_empty == false` and the row below is genuinely
689        // emitted with it intact.
690        let retained_cell_tombstones = surviving
691            .iter()
692            .filter(|cell| KWayMerger::is_cell_tombstone(cell) || cell.is_deleted)
693            .count() as u64;
694        purges.emitted += retained_cell_tombstones;
695
696        // Attach the carried deletion metadata to whichever entry is emitted so it
697        // is not dropped by reconciliation (#886 plumbing preservation).
698        let has_carried_metadata = !complex_deletions.is_empty() || range_deletion.is_some();
699
700        // Emit a live row only when real data survives. `!purged_to_empty`
701        // suppresses a clustered row whose only data was a dropped column (phantom
702        // key-only row); a genuine row marker (always key-only) has
703        // `had_data_before == false` so it is preserved.
704        let built = if !surviving.is_empty() && !purged_to_empty {
705            // `surviving` is non-empty, so `max()` is `Some`; `unwrap_or(0)` only
706            // guards the type and never triggers.
707            let row_ts = surviving.iter().map(|c| c.timestamp).max().unwrap_or(0);
708            let live = MergeEntry::new(
709                run_index,
710                key,
711                clustering_key,
712                row_ts,
713                RowData::Live { cells: surviving },
714            );
715            // Issue #932: a row deletion that survived purging COEXISTS with the
716            // surviving (strictly-newer) cells. Carry it on the live entry so the
717            // merge→mutation step emits a `HAS_DELETION` row holding both — the row
718            // deletion keeps shadowing older cells of OTHER columns living in
719            // SSTables not part of a partial compaction, preventing resurrection.
720            // Step 3 already dropped the cells this deletion covers (ts <= row_del),
721            // so the deletion shadows nothing within `surviving`.
722            Some(match row_del {
723                Some(deletion_time) => {
724                    // Issue #2163: this row-tombstone marker is RETAINED alongside
725                    // newer live cells (the common "retained-but-coexisting" case,
726                    // NOT the sole-output case below) — count it as emitted too, so
727                    // `tombstones_emitted` covers every marker that survives
728                    // reconciliation into the output, not only the row-absent case.
729                    purges.emitted += 1;
730                    live.with_row_deletion(deletion_time, row_del_ldt)
731                }
732                None => live,
733            })
734        } else if let Some(deletion_time) = row_del {
735            // No surviving data. If a row tombstone exists, keep the row shadowed
736            // so downstream still emits the deletion (preserves #505/#498 absence).
737            // Issue #2163: this row-tombstone marker is RETAINED into the output —
738            // count it as emitted (a marker carried forward, not purged).
739            purges.emitted += 1;
740            Some(MergeEntry::new(
741                run_index,
742                key,
743                clustering_key,
744                deletion_time,
745                RowData::Tombstone {
746                    deletion_time,
747                    // Preserve the source LDT paired with the winning deletion_time
748                    // (#873) instead of resetting it to 0; the writer encodes it
749                    // verbatim and gc_grace decisions stay faithful.
750                    local_deletion_time: row_del_ldt,
751                },
752            ))
753        } else if has_carried_metadata {
754            // No surviving data AND no row tombstone, but the cluster still carries
755            // complex/range deletion metadata. Emit a metadata-only entry (an empty
756            // `Live` row) so it survives reconciliation and reaches downstream
757            // consumers (#844/#846/#899).
758            Some(MergeEntry::new(
759                run_index,
760                key,
761                clustering_key,
762                0,
763                RowData::Live { cells: vec![] },
764            ))
765        } else {
766            // Truly empty/absent row.
767            None
768        };
769
770        built.map(|entry| {
771            let entry = if complex_deletions.is_empty() {
772                entry
773            } else {
774                entry.with_complex_deletions(complex_deletions)
775            };
776            match range_deletion {
777                Some(rd) => entry.with_range_deletion(rd),
778                None => entry,
779            }
780        })
781    }
