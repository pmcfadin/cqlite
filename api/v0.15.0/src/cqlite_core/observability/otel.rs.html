1//! OpenTelemetry runtime wiring (feature = "observability").
2//!
3//! This module is only compiled when the `observability` feature is on. It owns
4//! the OTLP trace + metric exporter setup, the parent-based trace-ID-ratio
5//! sampler, the global providers, the lazily-built metric instruments keyed by
6//! the [`catalog`](crate::observability::catalog) names, and the
7//! [`ObservabilityGuard`] that flushes/shuts down on drop.
8//!
9//! The public re-exports live in [`crate::observability`]; everything here is
10//! reached through those.

32/// Concrete SDK tracer provider stashed by [`init`] so [`tracing_layer`] can
33/// build its layer from an `SdkTracer` (which implements `PreSampledTracer`).
34/// When `init` is not called (or is inert), a default no-export provider is
35/// lazily created so the layer type stays monomorphic and simply drops spans.
36static TRACER_PROVIDER: OnceLock<SdkTracerProvider> = OnceLock::new();
37static METRICS_ACTIVE: AtomicBool = AtomicBool::new(false);

39/// Build the OTel `Resource` describing this process.
40fn build_resource(cfg: &ObservabilityConfig) -> Resource {
41    Resource::builder()
42        .with_service_name(cfg.service_name.clone())
43        .with_attribute(KeyValue::new(
44            semconv::attribute::SERVICE_VERSION,
45            cfg.service_version.clone(),
46        ))
47        .build()
48}

50/// Parent-based trace-ID-ratio sampler from the configured ratio.
51fn build_sampler(ratio: f64) -> Sampler {
52    Sampler::ParentBased(Box::new(Sampler::TraceIdRatioBased(ratio.clamp(0.0, 1.0))))
53}

55fn otlp_protocol(p: OtelProtocol) -> Protocol {
56    match p {
57        OtelProtocol::Grpc => Protocol::Grpc,
58        OtelProtocol::Http => Protocol::HttpBinary,
59    }
60}

62fn build_span_exporter(cfg: &ObservabilityConfig) -> Result<SpanExporter> {
63    let builder = match cfg.protocol {
64        OtelProtocol::Grpc => SpanExporter::builder()
65            .with_tonic()
66            .with_endpoint(cfg.endpoint.clone())
67            .with_protocol(otlp_protocol(cfg.protocol))
68            .with_timeout(cfg.timeout)
69            .build(),
70        OtelProtocol::Http => SpanExporter::builder()
71            .with_http()
72            .with_endpoint(cfg.endpoint.clone())
73            .with_protocol(otlp_protocol(cfg.protocol))
74            .with_timeout(cfg.timeout)
75            .build(),
76    };
77    builder.map_err(|e| Error::configuration(format!("OTLP span exporter init failed: {e}")))
78}

80fn build_metric_exporter(cfg: &ObservabilityConfig) -> Result<MetricExporter> {
81    let builder = match cfg.protocol {
82        OtelProtocol::Grpc => MetricExporter::builder()
83            .with_tonic()
84            .with_endpoint(cfg.endpoint.clone())
85            .with_protocol(otlp_protocol(cfg.protocol))
86            .with_timeout(cfg.timeout)
87            .build(),
88        OtelProtocol::Http => MetricExporter::builder()
89            .with_http()
90            .with_endpoint(cfg.endpoint.clone())
91            .with_protocol(otlp_protocol(cfg.protocol))
92            .with_timeout(cfg.timeout)
93            .build(),
94    };
95    builder.map_err(|e| Error::configuration(format!("OTLP metric exporter init failed: {e}")))
96}

98/// RAII guard returned by [`init`](crate::observability::init).
99///
100/// Dropping it force-flushes and shuts down the trace and metric providers so
101/// buffered telemetry is exported before the process exits. An *inert* guard
102/// (built when observability is disabled) does nothing on drop.
103pub struct ObservabilityGuard {
104    tracer_provider: Option<SdkTracerProvider>,
105    meter_provider: Option<SdkMeterProvider>,
106}
107
108impl std::fmt::Debug for ObservabilityGuard {
109    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
110        f.debug_struct("ObservabilityGuard")
111            .field("active", &self.tracer_provider.is_some())
112            .finish()
113    }
114}
115
116impl ObservabilityGuard {
117    /// An inert guard that installs and flushes nothing.
118    pub(crate) fn inert() -> Self {
119        Self {
120            tracer_provider: None,
121            meter_provider: None,
122        }
123    }
124
125    /// Whether this guard owns live exporters.
126    pub fn is_active(&self) -> bool {
127        self.tracer_provider.is_some() || self.meter_provider.is_some()
128    }
129
130    /// Explicitly flush all pending telemetry without shutting down. Useful for
131    /// tests; a normal program can just rely on the drop behaviour.
132    pub fn force_flush(&self) {
133        if let Some(tp) = &self.tracer_provider {
134            let _ = tp.force_flush();
135        }
136        if let Some(mp) = &self.meter_provider {
137            let _ = mp.force_flush();
138        }
139    }
140}
141
142impl Drop for ObservabilityGuard {
143    fn drop(&mut self) {
144        if let Some(tp) = &self.tracer_provider {
145            let _ = tp.force_flush();
146            let _ = tp.shutdown();
147        }
148        if let Some(mp) = &self.meter_provider {
149            let _ = mp.force_flush();
150            let _ = mp.shutdown();
151            METRICS_ACTIVE.store(false, Ordering::Relaxed);
152        }
153    }
154}

156/// Initialise OTLP trace + metric export and install the global providers.
157///
158/// Returns an inert [`ObservabilityGuard`] when `cfg.enabled == false` (no
159/// exporters, no global providers touched). Otherwise installs SDK providers as
160/// the OpenTelemetry globals and returns a guard that flushes on drop.
161///
162/// Note: this does NOT install a `tracing` subscriber. Callers compose their own
163/// subscriber and add [`tracing_layer`] to it — see the module docs on
164/// [`crate::observability`].
165pub fn init(cfg: ObservabilityConfig) -> Result<ObservabilityGuard> {
166    // Issue #2163 (roborev r4): plumb the presence-oracle verification config
167    // field into the runtime switch the read path actually consults — applied
168    // UNCONDITIONALLY (even when `cfg.enabled == false`, i.e. OTel export itself
169    // is off), since the switch also drives the confirmation-scan + warning-log
170    // side effect independent of metric export. `apply_config` itself honors
171    // env-overrides-config precedence.
172    crate::storage::sstable::reader::presence_verification::apply_config(
173        cfg.verify_presence_oracle,
174    );
175
176    if !cfg.enabled {
177        return Ok(ObservabilityGuard::inert());
178    }
179
180    let resource = build_resource(&cfg);
181
182    // Tracing pipeline.
183    let span_exporter = build_span_exporter(&cfg)?;
184    let tracer_provider = SdkTracerProvider::builder()
185        .with_batch_exporter(span_exporter)
186        .with_sampler(build_sampler(cfg.sampling_ratio))
187        .with_resource(resource.clone())
188        .build();
189    // Set the global provider (for cross-crate context propagation) and stash a
190    // concrete clone so `tracing_layer` can build its layer from the SDK tracer
191    // (the boxed global tracer does not implement `PreSampledTracer`).
192    let _ = TRACER_PROVIDER.set(tracer_provider.clone());
193    global::set_tracer_provider(tracer_provider.clone());
194
195    // Metrics pipeline.
196    let metric_exporter = build_metric_exporter(&cfg)?;
197    let reader = PeriodicReader::builder(metric_exporter).build();
198    let meter_provider = SdkMeterProvider::builder()
199        .with_reader(reader)
200        .with_resource(resource)
201        .build();
202    global::set_meter_provider(meter_provider.clone());
203    METRICS_ACTIVE.store(true, Ordering::Relaxed);
204
205    // Eagerly seed the always-on baseline instruments so a fresh scrape of a
206    // just-started server shows them at 0 rather than absent (issue #2288).
207    register_baseline_instruments();
208
209    Ok(ObservabilityGuard {
210        tracer_provider: Some(tracer_provider),
211        meter_provider: Some(meter_provider),
212    })
213}

220/// Eagerly emit a zero data point for the always-on baseline instruments so they
221/// are visible in a scrape of a freshly-started server, before any real activity
222/// (issue #2288).
223///
224/// `cqlite.errors.total` otherwise registers *lazily* — it appears in a metrics
225/// backend only on its first increment — so "metric name absent from the
226/// backend" was ambiguous between *no errors occurred* and *error counting isn't
227/// wired*. A single `add(0)` builds the instrument and publishes a `0` series so
228/// absence unambiguously means "not wired". The baseline uses an empty attribute
229/// set (no invented `{category, subsystem}` values that would pollute the bounded
230/// taxonomy); real errors add their own labeled series alongside it.
231pub(crate) fn register_baseline_instruments() {
232    instruments().errors_total.add(0, &[]);
233}

240/// Return the `tracing` layer that bridges spans/events into OpenTelemetry, or
241/// `None` if observability has not been initialised.
242///
243/// **Call [`init`] before composing this layer.** The layer is bound at
244/// construction time to the exporting tracer provider that `init` installs, so
245/// it can only be built once that provider exists. When `init` has not run (or
246/// ran with a disabled config), this returns `None` — an honestly inert result —
247/// rather than a layer permanently bound to a non-exporting provider. `Option<L>`
248/// itself implements `Layer`, so the `None` case is a no-op and callers compose
249/// it directly:
250///
251/// ```ignore
252/// use tracing_subscriber::prelude::*;
253/// let _guard = cqlite_core::observability::init(cfg)?;   // installs the provider
254/// tracing_subscriber::registry()
255///     .with(tracing_subscriber::fmt::layer())
256///     .with(tracing_subscriber::EnvFilter::from_default_env())
257///     .with(cqlite_core::observability::tracing_layer())  // Some(..) after init, else None
258///     .init();
259/// ```
260pub fn tracing_layer<S>() -> Option<impl tracing_subscriber::Layer<S>>
261where
262    S: tracing::Subscriber + for<'span> tracing_subscriber::registry::LookupSpan<'span>,
263{
264    // Only build a layer once `init` has installed the exporting provider. No
265    // ephemeral fallback: a layer bound to a throwaway no-export provider can
266    // never start exporting, so returning `None` (inert) is the correct
267    // behaviour when uninitialised.
268    let provider = TRACER_PROVIDER.get()?;
269    let tracer = provider.tracer(SCOPE);
270    Some(tracing_opentelemetry::layer().with_tracer(tracer))
271}

827/// Extract a W3C `traceparent` header and set the resulting remote context as
828/// the parent of `span`. No-op when `traceparent` is absent/empty/unparseable.
829///
830/// Uses the standard [`TraceContextPropagator`] to parse the header into an
831/// OpenTelemetry [`Context`](opentelemetry::Context), then
832/// `tracing-opentelemetry`'s `set_parent` to link the `tracing` span to that
833/// remote span. Only the `traceparent` header is consulted (no baggage /
834/// tracestate), keeping the surface minimal and the behaviour identical to
835/// other CQLite hosts.
836pub(crate) fn set_span_parent_from_traceparent(span: &tracing::Span, traceparent: Option<&str>) {
837    use opentelemetry::propagation::{Extractor, TextMapPropagator};
838    use opentelemetry::trace::TraceContextExt;
839    use opentelemetry_sdk::propagation::TraceContextPropagator;
840    use tracing_opentelemetry::OpenTelemetrySpanExt;
841
842    let header = match traceparent {
843        Some(h) if !h.trim().is_empty() => h,
844        _ => return,
845    };
846
847    /// Single-header carrier exposing only `traceparent` to the propagator.
848    struct TraceParentCarrier<'a>(&'a str);
849    impl Extractor for TraceParentCarrier<'_> {
850        fn get(&self, key: &str) -> Option<&str> {
851            if key.eq_ignore_ascii_case("traceparent") {
852                Some(self.0)
853            } else {
854                None
855            }
856        }
857        fn keys(&self) -> Vec<&str> {
858            vec!["traceparent"]
859        }
860    }
861
862    let propagator = TraceContextPropagator::new();
863    let cx = propagator.extract(&TraceParentCarrier(header));
864    // Only re-parent when extraction produced a valid remote span context;
865    // otherwise leave the span attached to its in-process parent.
866    if cx.span().span_context().is_valid() {
867        span.set_parent(cx);
868    }
869}
