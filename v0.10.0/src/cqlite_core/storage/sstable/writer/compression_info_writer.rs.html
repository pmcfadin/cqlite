1//! CompressionInfo.db writer - writes compression metadata
2//!
3//! Generates the CompressionInfo.db component that describes how Data.db chunks
4//! are compressed. This file is required for readers to decompress Data.db.
5//!
6//! # Binary Format (Cassandra 5.0 NB)
7//!
8//! ```text
9//! [u16 BE: algorithm_name_length]    ← Length of algorithm name
10//! [bytes: algorithm_name]            ← "LZ4Compressor", "SnappyCompressor", etc.
11//! [4 bytes: padding]                 ← Fixed 0x00000000 padding
12//! [u32 BE: chunk_length]             ← Uncompressed chunk size (typically 65536)
13//! [u32 BE: options/flags]            ← Options field (0x7FFFFFFF typical)
14//! [u64 BE: compressed_data_length]   ← Total compressed Data.db size
15//! [u32 BE: chunk_count]              ← Number of chunks
16//! [u64 BE * chunk_count: offsets]    ← Byte offset of each chunk in Data.db
17//! [u32 BE * chunk_count: crcs]       ← CRC32 of each compressed chunk (optional)
18//! [u32 BE: metadata_crc]             ← CRC32 of all preceding bytes
19//! ```
20//!
21//! References:
22//! - Parser: `cqlite-core/src/storage/sstable/compression_info.rs`
23//! - Format docs: `docs/sstables-definitive-guide/chapters/09-compression.md`
24

31/// Compression algorithm identifier
32#[derive(Debug, Clone, Copy, PartialEq, Eq)]
33pub enum CompressionAlgorithm {
34    /// LZ4 compression (fast, moderate ratio)
35    Lz4,
36    /// Snappy compression (very fast, lower ratio)
37    Snappy,
38    /// Deflate/zlib compression (slower, better ratio)
39    Deflate,
40    /// Zstd compression (balanced speed/ratio)
41    Zstd,
42    /// No compression (passthrough)
43    None,
44}
45
46impl CompressionAlgorithm {
47    /// Get the Cassandra algorithm name string
48    pub fn cassandra_name(&self) -> &'static str {
49        match self {
50            CompressionAlgorithm::Lz4 => "LZ4Compressor",
51            CompressionAlgorithm::Snappy => "SnappyCompressor",
52            CompressionAlgorithm::Deflate => "DeflateCompressor",
53            CompressionAlgorithm::Zstd => "ZstdCompressor",
54            CompressionAlgorithm::None => "NoopCompressor",
55        }
56    }
57
58    /// Parse from Cassandra algorithm name
59    pub fn from_cassandra_name(name: &str) -> Option<Self> {
60        match name {
61            "LZ4Compressor" | "org.apache.cassandra.io.compress.LZ4Compressor" => {
62                Some(CompressionAlgorithm::Lz4)
63            }
64            "SnappyCompressor" | "org.apache.cassandra.io.compress.SnappyCompressor" => {
65                Some(CompressionAlgorithm::Snappy)
66            }
67            "DeflateCompressor" | "org.apache.cassandra.io.compress.DeflateCompressor" => {
68                Some(CompressionAlgorithm::Deflate)
69            }
70            "ZstdCompressor" | "org.apache.cassandra.io.compress.ZstdCompressor" => {
71                Some(CompressionAlgorithm::Zstd)
72            }
73            "NoopCompressor" | "org.apache.cassandra.io.compress.NoopCompressor" => {
74                Some(CompressionAlgorithm::None)
75            }
76            _ => None,
77        }
78    }
79}

81/// Metadata about compressed Data.db
82///
83/// Collected during compression and written to CompressionInfo.db
84#[derive(Debug, Clone)]
85pub struct CompressionMetadata {
86    /// Compression algorithm used
87    pub algorithm: CompressionAlgorithm,
88    /// Uncompressed chunk size in bytes (typically 65536)
89    pub chunk_length: u32,
90    /// Total compressed data length (Data.db file size)
91    pub compressed_length: u64,
92    /// Byte offset of each compressed chunk in Data.db
93    pub chunk_offsets: Vec<u64>,
94    /// CRC32 checksum of each compressed chunk (optional)
95    pub chunk_crcs: Vec<u32>,
96}
97
98impl CompressionMetadata {
99    /// Create new compression metadata
100    pub fn new(algorithm: CompressionAlgorithm, chunk_length: u32) -> Self {
101        Self {
102            algorithm,
103            chunk_length,
104            compressed_length: 0,
105            chunk_offsets: Vec::new(),
106            chunk_crcs: Vec::new(),
107        }
108    }
109
110    /// Add a new chunk offset and optional CRC
111    pub fn add_chunk(&mut self, offset: u64, crc: Option<u32>) {
112        self.chunk_offsets.push(offset);
113        if let Some(crc_value) = crc {
114            self.chunk_crcs.push(crc_value);
115        }
116    }
117
118    /// Set the total compressed data length
119    pub fn set_compressed_length(&mut self, length: u64) {
120        self.compressed_length = length;
121    }
122
123    /// Get the number of chunks
124    pub fn chunk_count(&self) -> usize {
125        self.chunk_offsets.len()
126    }
127}

129/// CompressionInfo.db file writer
130///
131/// Writes compression metadata to disk in Cassandra's binary format.
132#[derive(Debug)]
133pub struct CompressionInfoWriter {
134    /// Output file path
135    path: PathBuf,
136}
137
138impl CompressionInfoWriter {
139    /// Create a new CompressionInfo.db writer
140    pub fn new(path: PathBuf) -> Self {
141        Self { path }
142    }
143
144    /// Write compression metadata to file
145    ///
146    /// # Arguments
147    /// * `metadata` - Compression metadata collected during Data.db writing
148    ///
149    /// # Returns
150    /// Ok(()) on success, or error if write fails
151    pub fn write(&self, metadata: &CompressionMetadata) -> Result<()> {
152        let file = File::create(&self.path).map_err(|e| {
153            Error::Storage(format!(
154                "Failed to create CompressionInfo.db at {}: {}",
155                self.path.display(),
156                e
157            ))
158        })?;
159        let mut writer = BufWriter::new(file);
160
161        // Build the binary content first to compute CRC
162        let content = self.build_content(metadata)?;
163
164        // Compute CRC32 of content (excluding the CRC itself)
165        let mut hasher = Hasher::new();
166        hasher.update(&content);
167        let crc32 = hasher.finalize();
168
169        // Write content + CRC
170        writer.write_all(&content).map_err(|e| {
171            Error::Storage(format!("Failed to write CompressionInfo.db content: {}", e))
172        })?;
173
174        // Write trailing CRC32
175        writer.write_all(&crc32.to_be_bytes()).map_err(|e| {
176            Error::Storage(format!("Failed to write CompressionInfo.db CRC: {}", e))
177        })?;
178
179        writer
180            .flush()
181            .map_err(|e| Error::Storage(format!("Failed to flush CompressionInfo.db: {}", e)))?;
182
183        Ok(())
184    }
185
186    /// Build the binary content (everything before the trailing CRC)
187    fn build_content(&self, metadata: &CompressionMetadata) -> Result<Vec<u8>> {
188        let mut content = Vec::new();
189
190        // Algorithm name
191        let algorithm_name = metadata.algorithm.cassandra_name();
192        let name_bytes = algorithm_name.as_bytes();
193
194        if name_bytes.len() > u16::MAX as usize {
195            return Err(Error::InvalidInput(format!(
196                "Algorithm name too long: {} bytes (max {})",
197                name_bytes.len(),
198                u16::MAX
199            )));
200        }
201
202        // Algorithm name length (u16 BE)
203        content.extend_from_slice(&(name_bytes.len() as u16).to_be_bytes());
204
205        // Algorithm name bytes
206        content.extend_from_slice(name_bytes);
207
208        // Fixed 4-byte padding (0x00000000) - ensures 8-byte alignment for chunk_length field
209        content.extend_from_slice(&[0u8; 4]);
210
211        // Chunk length (u32 BE)
212        content.extend_from_slice(&metadata.chunk_length.to_be_bytes());
213
214        // Options/flags field (u32 BE) - typically 0x7FFFFFFF
215        content.extend_from_slice(&0x7FFFFFFFu32.to_be_bytes());
216
217        // Compressed data length (u64 BE)
218        content.extend_from_slice(&metadata.compressed_length.to_be_bytes());
219
220        // Chunk count (u32 BE)
221        if metadata.chunk_offsets.len() > u32::MAX as usize {
222            return Err(Error::InvalidInput(format!(
223                "Too many chunks: {} (max {})",
224                metadata.chunk_offsets.len(),
225                u32::MAX
226            )));
227        }
228        content.extend_from_slice(&(metadata.chunk_offsets.len() as u32).to_be_bytes());
229
230        // Chunk offsets (u64 BE each)
231        for offset in &metadata.chunk_offsets {
232            content.extend_from_slice(&offset.to_be_bytes());
233        }
234
235        // Chunk CRCs (u32 BE each) - only if we have them
236        if !metadata.chunk_crcs.is_empty() {
237            if metadata.chunk_crcs.len() != metadata.chunk_offsets.len() {
238                return Err(Error::InvalidInput(format!(
239                    "Chunk CRC count ({}) doesn't match chunk count ({})",
240                    metadata.chunk_crcs.len(),
241                    metadata.chunk_offsets.len()
242                )));
243            }
244
245            for crc in &metadata.chunk_crcs {
246                content.extend_from_slice(&crc.to_be_bytes());
247            }
248        }
249
250        Ok(content)
251    }
252
253    /// Build content to a buffer instead of file (for testing)
254    pub fn build_to_vec(&self, metadata: &CompressionMetadata) -> Result<Vec<u8>> {
255        let content = self.build_content(metadata)?;
256
257        // Compute CRC32
258        let mut hasher = Hasher::new();
259        hasher.update(&content);
260        let crc32 = hasher.finalize();
261
262        // Combine content + CRC
263        let mut result = content;
264        result.extend_from_slice(&crc32.to_be_bytes());
265
266        Ok(result)
267    }
268}

336    #[test]
337    fn test_compression_info_writer_build_content() {
338        let mut metadata = CompressionMetadata::new(CompressionAlgorithm::Lz4, 65536);
339        metadata.add_chunk(0, None);
340        metadata.add_chunk(8192, None);
341        metadata.set_compressed_length(16000);
342
343        let writer = CompressionInfoWriter::new(PathBuf::from("/tmp/test"));
344        let content = writer.build_content(&metadata).unwrap();
345
346        // Verify structure:
347        // 2 bytes: name length (13 for "LZ4Compressor")
348        // 13 bytes: algorithm name
349        // 4 bytes: padding
350        // 4 bytes: chunk length
351        // 4 bytes: options
352        // 8 bytes: compressed length
353        // 4 bytes: chunk count
354        // 16 bytes: 2 chunk offsets (8 bytes each)
355        // Total: 55 bytes (no CRCs)
356
357        assert_eq!(content.len(), 55);
358
359        // Check algorithm name length
360        assert_eq!(&content[0..2], &[0x00, 0x0D]); // 13 in BE
361
362        // Check algorithm name
363        assert_eq!(&content[2..15], b"LZ4Compressor");
364
365        // Check padding
366        assert_eq!(&content[15..19], &[0x00, 0x00, 0x00, 0x00]);
367
368        // Check chunk length (65536 = 0x00010000)
369        assert_eq!(&content[19..23], &[0x00, 0x01, 0x00, 0x00]);
370
371        // Check options (0x7FFFFFFF)
372        assert_eq!(&content[23..27], &[0x7F, 0xFF, 0xFF, 0xFF]);
373
374        // Check compressed length (16000 = 0x3E80)
375        assert_eq!(
376            &content[27..35],
377            &[0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x3E, 0x80]
378        );
379
380        // Check chunk count (2)
381        assert_eq!(&content[35..39], &[0x00, 0x00, 0x00, 0x02]);
382
383        // Check first offset (0)
384        assert_eq!(
385            &content[39..47],
386            &[0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00]
387        );
388
389        // Check second offset (8192 = 0x2000)
390        assert_eq!(
391            &content[47..55],
392            &[0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x20, 0x00]
393        );
394    }

417    #[test]
418    fn test_compression_info_writer_write_file() {
419        let temp_dir = TempDir::new().unwrap();
420        let path = temp_dir.path().join("nb-1-big-CompressionInfo.db");
421
422        let mut metadata = CompressionMetadata::new(CompressionAlgorithm::Lz4, 65536);
423        metadata.add_chunk(0, Some(0xDEADBEEF));
424        metadata.set_compressed_length(32768);
425
426        let writer = CompressionInfoWriter::new(path.clone());
427        writer.write(&metadata).unwrap();
428
429        // Verify file was created
430        assert!(path.exists());
431
432        // Read and verify content
433        let bytes = std::fs::read(&path).unwrap();
434
435        // Should have content + 4-byte CRC
436        assert!(bytes.len() > 4);
437
438        // Verify CRC32 is valid
439        let content_len = bytes.len() - 4;
440        let stored_crc = u32::from_be_bytes([
441            bytes[content_len],
442            bytes[content_len + 1],
443            bytes[content_len + 2],
444            bytes[content_len + 3],
445        ]);
446
447        let mut hasher = Hasher::new();
448        hasher.update(&bytes[..content_len]);
449        let computed_crc = hasher.finalize();
450
451        assert_eq!(stored_crc, computed_crc, "CRC mismatch");
452    }
